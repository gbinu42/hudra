ܛܟܣܐ ܘܩܢܘܢ̈ܐ
ܕܡܫܬܡܫܝܢ ܒܥܐܕܐ
ܕܨܠܝܒܐ ܣܓ݂ܝܕܐ.
ܕܡܸܬ݂ܥܒܸ݁ܕ݂ ܒܟ݂ܠܗܹܝܢ ܫܢܲܝ̈ܵܐ ܕܥܵܠܡܵܐ ܒܲܬ݂ܠܵܬܲܥܣܲܪ
ܒܐܝܼܠܘܿܢ ܝܲܪܚܵܐ. ܟܠ ܝܵܘܡܵܐ ܕܟ݂ܵܐܹܢ.
( ܘܐܵܦ ܗܵܕܹܐ ܙܵܕܹ݁ܩ ܠܡܵܘܕܵܥܘܼ. ܕܐܸܢ ܒܚܲܕ݂ܒ݁ܫܲܒܵܐ ܐܵܘ ܒܲܬ݂ܪܹܝܢܒ݁ܫܲܒܵܐ ܐܵܦ ܒܲܬ݂ܠܵܬ݂ܒ݁ܫܲܒܵܐ ܢܵܦܹ݁ܠ ܥܹܐܕ݂ܵܐ ܕܲܨܠܝܼܒ݂ܵܐ: ܒܚܲܕ݂ܒ݁ܫܲܒܵܐ ܕܫܲܒ݂ܥܵܐ ܕܒ݂ܵܬܲܪ ܥܹܐܕ݂ܵܐ ܗܵܘܝܵܐ ܡܲܥܲܠܬܵܐ. ܘܐܸܢ ܒܐܲܪܒܲܥܒ݁ܫܲܒܵܐ ܐܵܘ ܒܚܲܡܫܵܒ݁ܫܲܒܵܐ ܐܵܘ ܒܲܥܪܘܼܒ݂ܬܵܐ ܐܵܘ ܒܫܲܒ݁ܬ݂ܵܐ ܢܵܦܹ݁ܠ ܥܹܐܕ݂ܵܐ ܕܲܨܠܝܼܒ݂ܵܐ: ܒܚܲܕ݂ܒ݁ܫܲܒܵܐ ܕܲܬ݂ܡܵܢܝܵܐ ܕܒ݂ܵܬܲܪ ܥܹܐܕ݂ܵܐ ܗܵܘܝܵܐ ܡܲܥܲܠܬܵܐ: ܕܠܵܐ ܦܘܼܠܵܓ݂. )
	ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܕ݂ܥܹܐܕܹ̈ܐ.
ܢܵܘܕܸ݁ܐ ܡܵܪܝ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܒܬܸܫ̈ܒ݁ܚܵܬ݂ܵܐ ܪ̈ܘܼܚܵܢܵܝܵܬܵܐ: ܘܢܸܣܓ݁ܘܿܕ݂ ܠܡܵܪܘܼܬ݂ܵܟ݂ ܒܣܸܓ݂̈ܕܵܬ݂ܵܐ ܐܲܪ̈ܥܵܢܵܝܵܬ݂ܵܐ: ܘܲܢܫܲܒܲܚ ܠܲܟ݂ܝܵܢܵܟ݂ ܟܲܣܝܵܐ ܘܲܓ݂ܢܝܼܙܵܐ ܒܡܲܚܫ̈ܒ݂ܵܬ݂ܵܐ ܕܲܟ݂̈ܝܵܬ݂ܵܐ ܘܠܵܐ ܡܛܵܘ̈ܫܵܬ݂ܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܡܲܪܡܝܼܬ݂ܵܐ. ܠܵܟ݂ ܝܵܐ̇ܝܵܐ. ܒܟ݂ܠܵܗ̇ ܫܲܒܬ݂ܵܐ. ܚܙܝܼ. ܡܙܡܘܪ̈ܐ ܣܗ‐ܣܙ
ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܐܲܝܟ݂ ܥܸܛܪܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܕ݂ܥܹܐܕܹ̈ܐ: ܠܲܫܡܵܐ ܪܲܒܵܐ ܘܲܕ݂ܚܝܼܠܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܘܲܡܒܲܪܟ݂ܵܐ ܘܛܘܼܒ݂ܬ݂ܵܢܵܐ ܘܠܵܐ ܡܸܬ݁ܕܲܪܟܵܢܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܘܲܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܕܲܠܘܵܬ݂ ܓܸܢܣܲܢ: ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	ܡܵܐ ܚܲܒ݁ܝܼܒ݂ܝܼܢ ܡܲܫܟ݁ܢܲܝ̈ܟ ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ. ܐܲܝܟ݂ ܥܸܛܪܵܐ ܕܒܸܣܡܹ̈ܐ ܛܵܒܹ̈ܐ ܘܪܹܝܚܵܐ ܕܦܝܼܪܡܵܐ ܒܲܣܝܼܡܵܐ: ܩܲܒܸ݁ܠ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ ܒܵܥܘܼܬ݂ܵܐ ܘܲܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ܀
	ܣܲܟ݁ܝܲܬ݀ ܢܲܦ̮ܫܝ ܘܐܸܬ݂ܪܲܓ݂ܪܓܲܬ݀ ܠܕ݂ܵܪ̈ܵܘܗܝ ܕܡܵܪܝܵܐ. ܐܲܝܟ݂ ܥܸܛܪܵܐ ܕܒܸܣܡܹ̈ܐ ܛܵܒܹ̈ܐ ܘܪܹܝܚܵܐ ܕܦܝܼܪܡܵܐ ܒܲܣܝܼܡܵܐ: ܩܲܒܸ݁ܠ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ ܒܵܥܘܼܬ݂ܵܐ ܘܲܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ܀
	ܠܹܒ݁ܝ ܘܒܸܣܪܝ ܫܲܒܲܚܘ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ. ܐܲܝܟ݂ ܥܸܛܪܵܐ ܕܒܸܣܡܹ̈ܐ ܛܵܒܹ̈ܐ ܘܪܹܝܚܵܐ ܕܦܝܼܪܡܵܐ ܒܲܣܝܼܡܵܐ: ܩܲܒܸ݁ܠ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ ܒܵܥܘܼܬ݂ܵܐ ܘܲܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ܀
	ܫܲܒܲܚ: ܐܲܝܟ݂ ܥܸܛܪܵܐ ܕܒܸܣܡܹ̈ܐ ܛܵܒܹ̈ܐ ܘܪܹܝܚܵܐ ܕܦܝܼܪܡܵܐ ܒܲܣܝܼܡܵܐ: ܩܲܒܸ݁ܠ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ ܒܵܥܘܼܬ݂ܵܐ ܘܲܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ܀
	ܥܵܠܲܡ: ܐܲܝܟ݂ ܥܸܛܪܵܐ ܕܒܸܣܡܹ̈ܐ ܛܵܒܹ̈ܐ ܘܪܹܝܚܵܐ ܕܦܝܼܪܡܵܐ ܒܲܣܝܼܡܵܐ: ܩܲܒܸ݁ܠ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ ܒܵܥܘܼܬ݂ܵܐ ܘܲܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ܀
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܥܲܠ ܦܝܼܪܡܵܐ. ܢܲܣܸܩ ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܟ݂ܠܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܐܵܘ ܗܵܕܹܐ. ܡܫܝܼܚܵܐ ܕܩܲܒܸ݁ܠ ܕܡܵܐ ܕܣܵܗܕܹ̈ܐ ܒܝܘܿܡ ܩܸܛܠܲܝ̈ܗܘܿܢ: ܩܲܒܸ݁ܠ ܒܸܣܡܵܐ ܗܵܢܵܐ ܡܢ ܐܝܼܕܲܝ̈ ܚܲܠܵܫܘܼܬ݂ܝ ܒܛܲܝܒ݁ܘܼ ܕܲܚܢܵܢܵܟ݂ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܠܵܟ݂ܘܼ ܡܵܪܵܐ: ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܕ݂ܥܹܐܕܹ̈ܐ ܘܲܕ݁ܕ݂ܘܼܟ݂ܪ̈ܵܢܹܐ. ܘܥܲܠ ܐܲܦܲܝ̈ ܟܠܗܘܿܢ ܥܘܼܕ݂ܪ̈ܵܢܲܝܟ ܘܛܲܝ̈ܒ݁ܘܵܬ݂ܵܟ݂ ܕܲܠܘܵܬܲܢ ܕܠܵܐ ܡܸܬ݂ܦܲܪܥܝܼܢ: ܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ ܒܥܹܕ݁ܬܵܟ݂ ܡܟܲܠܲܠܬܵܐ ܡܲܠܝܲܬ݂ ܟܠ ܥܘܼܕ݂ܪ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܵܐ ܘܒ݂ܵܪܘܿܝܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
	ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
      ܢܵܦ݁ܩܝܼܢ. ܒܠܵܟ݂ܘܼܡܵܪܵܐ ܠܕ݂ܵܪܬܵܐ ܥܲܡ ܨܠܝܼܒ݂ܵܐ ܘܐܹܘܲܢܓܲܠܝܼܘܿܢ ܘܢܲܗܝܼܪܹ̈ܐ ܘܦܝܼܪܡܵܐ. ܘܲܡܨܲܠܹܝܢ ܠܒܲܪ ܒܕ݂ܵܪܬܵܐ ܡܢ ܝܵܘܡܵܢ ܥܲܕ݂ ܠܩܘܼܕܵܫܥܹܕ݁ܬܵܐ.
	ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܹܝܢ ܕܥܹܐܕܹ̈ܐ. ܒܓ݂ܘܼܕ݂ܵܐ ܡܸܬ݂ܐܲܡܪܵܐ: ܗ: ܙܲܒ݂ܢ̈ܝܼܢ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ܀
	ܚܕܝܼܬ݂ ܟܲܕ݂ ܐܵܡܪܝܼܢ ܗ݇ܘܵܘ ܠܝܼ: ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ܀
	 ܠܒܲܝܬܹ݁ܗ ܕܡܵܪܝܵܐ ܐܵܙܠ݇ܝܼܢ ܚ݇ܢܲܢ: ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ܀
	ܫܲܒܲܚ: ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ܀
	ܥܵܠܲܡ: ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ܀
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܡ. ܐܲܢ݇ܬ݁ܘܼ ܡܵܪܝ ܫܲܪܝܼܪܵܐܝܼܬ݂ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܘܲܡܢܲܛܪܵܢܵܐ ܐܲܡܝܼܢܵܐ ܕܚܲܝܲܝ̈ܢ: ܘܠܵܟ݂ ܡܵܪܝ ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
      ܥܘܿܢܝܼܬ݂ܵܐ ܕܲܩܕ݂ܵܡ. ܪܹܫ ܩܵܠܵܐ. ܒܵܟ݂ ܡܵܪܝܵܐ ܣܲܒ݁ܪܹܬ݂ ܠܵܐ ܐܸܒ݂ܗܲܬ݂ ܠܥܵܠܲܡ: ܘܐܵܦ ܟܠ ܕܲܡܣܲܒ݁ܪܝܼܢ ܒܵܟ݂ ܠܵܐ ܢܸܒ݂ܗܬ݂ܘܿܢ. ܕܨܘܼܪܬܵܐ. ܡܸܛܠ ܕܡܵܐ ܒܵܗܬ݁ܝܼܢ ܐܲܝܠܹܝܢ ܕܲܬ݂ܟ݂ܝܼܠܝܼܢ ܥܠܲܝܟ. ܠܵܐ ܒܵܗܬ݁ܝܼܢܲܢ ܝܼܫܘܿܥ ܒܲܨܠܝܼܒ݂ܵܟ݂. ܡܸܛܠ ܚܲܝܠܵܟ݂ ܪܲܒܵܐ ܕܲܟ݂ܣܸܐ ܒܹܗ. ܐܸܢ ܚܲܢܦܹ̈ܐ ܘܲܝܗ݇ܘܼܕܵܝܹ̈ܐ. ܡܒܲܙܚܝܼܢ ܒܵܗ ܒܟ݂ܵܪܘܿܙܘܼܬ݂ܵܟ݂. ܐܸܠܵܐ ܠܲܡܒܲܛܵܠܘܼ ܫܪܵܪܵܐ ܡܬ݂ܘܿܡ ܠܵܐ ܡܸܫܟ݁ܚܝܼܢ. ܬܪ̈ܲܝܗܘܿܢ ܓܹܝܪ ܫܲܘܝܵܐܝܼܬ݂ ܗܵܐ ܩܵܥܹܝܢ ܥܲܠ ܙܵܟ݂ܘܼܬ݂ܵܟ݂. ܝܗ݇ܘܼܕܵܝܹ̈ܐ ܗܵܐ ܐܸܬ݂ܒܲܕܲܪܘ. ܘܝܘܼܠܦܵܢܵܐ ܕܚܲܢܦܹ̈ܐ ܐܸܫܬ݁ܪܝܼ. ܘܗܵܐ ܣܵܗܕܝܼܢ ܐܲܟܲܚ݇ܕ݂. ܕܪܲܒ݁ܘܼ ܚܲܝܠܵܟ݂ ܡܵܪܝܵܐ:

ܬܢܝܼ

.
ܘܫܲܒܲܚ. ܠܵܐ ܒܵܗܬ݁ܝܼܢܲܢ ܝܼܫܘܿܥ ܒܲܨܠܝܼܒ݂ܵܟ݂. ܡܸܛܠ ܚܲܝܠܵܟ݂ ܪܲܒܵܐ ܕܲܟ݂ܣܸܐ ܒܹܗ. ܐܸܢ ܚܲܢܦܹ̈ܐ ܘܲܝܗ݇ܘܼܕܵܝܹ̈ܐ. ܡܒܲܙܚܝܼܢ ܒܵܗ ܒܟ݂ܵܪܘܿܙܘܼܬ݂ܵܟ݂. ܐܸܠܵܐ ܠܲܡܒܲܛܵܠܘܼ ܫܪܵܪܵܐ ܡܬ݂ܘܿܡ ܠܵܐ ܡܸܫܟ݁ܚܝܼܢ. ܬܪ̈ܲܝܗܘܿܢ ܓܹܝܪ ܫܲܘܝܵܐܝܼܬ݂ ܗܵܐ ܩܵܥܹܝܢ ܥܲܠ ܙܵܟ݂ܘܼܬ݂ܵܟ݂. ܝܗ݇ܘܼܕܵܝܹ̈ܐ ܗܵܐ ܐܸܬ݂ܒܲܕܲܪܘ. ܘܝܘܼܠܦܵܢܵܐ ܕܚܲܢܦܹ̈ܐ ܐܸܫܬ݁ܪܝܼ. ܘܗܵܐ ܣܵܗܕܝܼܢ ܐܲܟܲܚ݇ܕ݂. ܕܪܲܒ݁ܘܼ ܚܲܝܠܵܟ݂ ܡܵܪܝܵܐ.
ܘܐܵܡܪܝܼܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܡܵܪܝܵܐ ܩܪܹܝ̇ܬ݂ܵܟ݂: ܠܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ: ܘܠܲܒ݂ܛܝܼܠܘܼܬܹܗ ܕܨܸܒ݂ܝܵܢܵܟ݂ ܕܲܥܠܲܝܢ: ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܐܵܡܪܝܼܢ. ܡܵܪܝܵܐ ܩܪܹܝܬ݂ܵܟ݂ ܐܲܢ݇ܬ݁ ܥܢܝܼܢܝ. ܚܙܝܼ. ܡܙܡܘܪ̈ܐ ܕܡܪܝܐ ܕܩܪܝܬܟ
      ܡܵܪܝܵܐ ܩܪܹܝܬ݂ܵܟ݂. ܒܥܲܡ ܓܲܒ݂ܪܹ̈ܐ. ܡܸܛܠ ܗܵܢܵܐ ܒܲܪܟ݂ܵܟ݂ ܐܲܠܵܗܵܐ ܠܥܵܠܲܡ* ܐܲܪܡܵܐ ܣܲܝܦܵܐ ܒܚܲܨܲܝ̈ܟ ܓܲܢ݇ܒܵܪܵܐ* ܗܸܕ݂ܪܵܟ݂ ܘܫܘܼܒ݂ܚܵܟ݂ ܘܫܘܼܒ݂ܚܵܟ݂ ܙܵܟܹ݁ܐ* ܪܟܲܒ݂ ܥܲܠ ܡܸܠܬ݂ܵܐ ܕܩܘܼܫܬܵܐ ܘܡܘܼܟܵܟ݂ܵܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ* ܢܵܡܘܿܣܵܟ݂ ܒܕܸܚܠܬ݂ܵܐ ܕܝܲܡܝܼܢܵܟ݂.
ܫܲܒܲܚ... ܥܵܠܲܡ... ܘܲܡܣܲܝܟܹܝܢ ܐܲܟ݂ܚ݇ܕ݂ܵܐ ܒܬܲܪ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ. ܡܵܪܝܵܐ ܩܪܹܝܬ݂ܵܟ݂ ܐܲܢ݇ܬ݁ ܥܢܝܼܢܝ: ܨܘܼܬ݂ ܡܸܠܲܝ̈ ܘܩܲܒܸ݁ܠ ܨܠܘܿܬ݂ܝ.
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܒ݂ܵܬܲܪ: ܫܡܲܥ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܒܲܚܢܵܢܵܟ݂: ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܒܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂: ܘܚܘܼܢ ܠܚܲܛܵܝܘܼܬܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܪ̈ܲܚܡܲܝܟ: ܐܵܣܝܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܡܵܪܵܐ ܕܟ݂ܠ...
	ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
      ܥܘܿܢܝܼܬ݂ܵܐ ܕܒ݂ܵܬܲܪ. ܒܲܫܠܝܼܚܹ̈ܐ ܓܒܲܝ̈ܵܐ. ܡܵܢ̇ܵܘ ܓܲܒ݂ܪܵܐ ܕܐܸܬ݁ܕܟܲܪܬܵܝܗܝ: ܐܲܢ݇ܬ݁ܘܼ ܦܪܲܩܬܵܢ ܡܢ ܣܵܢܐܲܝ̈ܢ: ܐܲܢ݇ܬ݁ ܪܲܨܸܨܬ݁ ܪܹ̈ܫܵܘܗܝ ܕܠܹܘܝܵܬܲܢ. ܕܨܘܼܪܬܵܐ. ܗܵܘ ܕܦܲܪܩܲܢ ܡܢ ܫܘܼܠܛܵܢܹܗ ܕܚܸܫܘܿܟ݂ܵܐ. ܒܚܲܫܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ. ܙܟܲܝܬܵܝܗܝ ܠܡܵܘܬܵܐ ܡܫܝܼܚܵܐ ܡܲܚܝܵܢܲܢ. ܘܲܗܘܲܝܬ݁ ܢܘܼܚܵܡܵܐ ܘܪܹܫܝܼܬ݂ܵܐ ܠܲܩܝܵܡܬܵܐ ܕܡܝܼܬܹ̈ܐ. ܘܐܸܣܬܲܠܲܩܬ݁ ܒܬܸܫܒ݁ܘܿܚܬܵܐ. ܘܥܲܠ ܟܘܪܣܲܝ ܫܘܼܒ݂ܚܵܐ ܡܙܲܝܚܝܼܢ ܠܵܟ݂ ܟܪ̈ܘܿܒܹܐ ܘܲܣܪ̈ܵܦܹܐ. ܥܲܡ ܡܲܠܲܐܟܹ̈ܐ ܐܵܦ ܚܢܲܢ ܡܫܲܒ݁ܚܝܼܢܲܢ ܠܵܟ݂ ܡܵܪܝܵܐ ܘܐܵܡܪܝܼܢܲܢ. ܠܵܟ݂ ܢܸܒ݂ܪܘܿܟ݂ ܘܠܵܟ݂ ܢܸܣܓ݁ܘܿܕ݂. ܟܠ ܡܵܐ ܕܒܲܫܡܲܝܵܐ ܘܡܵܐ ܕܒܲܐܪܥܵܐ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ: ܬܢܝܼ. ܘܫܲܒܲܚ. ܒܵܗ.
ܘܐܵܡܪܝܼܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ:
ܟܵܪܘܿܙܘܼܬ݂ܵܐ. ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ ܘܐܲܠܵܗܵܐ ܕܟ݂ܠܹܗ ܒܘܼܝܵܐܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܦܵܪܘܿܩܲܢ ܘܝܲܨܘܼܦܬ݂ܵܢܲܢ ܘܲܡܦܲܪܢܣܵܢܵܐ ܕܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ ܘܩܘܼܝܵܡܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ ܘܲܕ݂ܟ݂ܠܗܹܝܢ ܥܹ̈ܕ݂ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܐܲܬ݂ܪܲܢ ܘܥܲܠ ܟܠ ܐܲܬ݂ܪ̈ܵܘܵܢ: ܘܥܲܠ ܐܲܝܠܹܝܢ ܕܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܥܵܡ̇ܪܝܼܢ ܒܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܘܼܙܵܓ݂ܵܘ̈ܗܝ ܕܐܵܐܲܪ ܘܟܲܗܝܼܢܘܼܬ݂ܵܐ ܕܫܲܢ݇ܬܵܐ ܘܥܲܠ̈ܠܵܬ݂ܵܐ ܕܦܹܐܪܹ̈ܐ ܘܥܲܠ ܬܘܼܩܵܢܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܲܠܟܹ̈ܐ ܐܲܚܝܼܕܲܝ̈ ܫܘܼܠܛܵܢܹܗ ܕܥܵܠܡܵܐ ܗܵܢܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܩܲܫܝܼܫܹ̈ܐ ܘܲܡܫܲܡܫܵܢܹ̈ܐ ܐܲܪ̈ܬܵܕ݁ܘܿܟ݁ܣܹܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܥܲܬ݁ܝܼܪ ܒܪ̈ܲܚܡܵܘܗܝ ܘܲܫܦܝܼܥ ܒܲܚܢܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܡܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ ܐܝܼܬ݂ܵܘܗܝ ܘܲܠܕ݂ܵܪܕܵܪ̈ܝܼܢ ܩܲܝܵܡ ܫܘܼܠܛܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܟ݂ܝܵܢܹܗ ܐܝܼܬ݂ܵܘܗܝ ܛܵܒ݂ܵܐ ܘܝܵܗܘܿܒ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹܐ ܒܡܵܘܬܹ݁ܗ ܕܚܲܛܵܝܵܐ ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܢ ܥܵܘܠܹܗ ܘܢܸܚܸܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܲܒ݂ܨܵܘܡܹܗ ܩܲܕ݁ܝܼܫܵܐ ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܐܲܦܨܲܚ ܠܲܫܡܲܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܹܗ ܗܘ ܟܝܵܢܵܐܝܼܬ݂ ܐܝܼܬ݂ ܠܵܐ ܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܘܝܲܬ݁ܝܼܪܵܐܝܼܬ݂ ܕܐܲܝܠܹܝܢ ܕܲܡܗܲܝܡܢܝܼܢ ܒܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܣܓܵܐ ܥܲܡܲܢ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܢܲܦ̮̈ܫܵܬܲܢ ܘܕܲܚܕ݂ܵܕܹ̈ܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܲܓ݂ܥܸܠ.
ܘܥܵܢܹ̇ܝܢ: ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ.
	ܘܲܡܨܲܠܸܐ ܟܵܗܢܵܐ. ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܡܲܓ݂ܥܠܝܼܢܲܢ ܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ: ܘܡܸܢܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܫܵܐ̇ܠܝܼܢܲܢ ܫܘܼܒ݂ܩܵܢܵܐ ܕܚܵܘܒܹ̈ܐ ܘܕܲܚܛܵܗܹ̈ܐ: ܗܲܒ݂ܠܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܪ̈ܲܚܡܲܝܟ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ ܟܵܪܘܿܙܘܼܬ݂ܵܐ: ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ.
ܘܐܵܡ̇ܪܝܼܢ ܒܩܵܠܵܐ ܦܫܝܼܛܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܫܲܒܲܚ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܥܵܠܲܡ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܒ݂ܵܬܲܪ ܕܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܕ݂ܥܹܐܕܹ̈ܐ. ܠܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ ܕܩܲܕ݁ܝܼܫ ܒܲܟ݂ܝܵܢܹܗ: ܘܲܫܒ݂ܝܼܚ ܒܐܝܼܬ݂ܘܼܬܹܗ: ܘܪܵܡ ܘܲܡܥܲܠܲܝ ܡܢ ܟܠ ܒܐܲܠܵܗܘܼܬܹܗ. ܟܝܵܢܵܐ ܩܲܕ݁ܝܼܫܵܐ ܘܲܣܓ݂ܝܼܕ݂ܵܐ ܘܲܡܒܲܪܟ݂ܵܐ ܕܡܢ ܥܵܠܲܡ. ܡܵܘܕܹ݁ܝܢܲܢ ܘܣܵܓ݂ܕ݁ܝܼܢܲܢ ܘܲܡܫܲܒ݁ܚܝܼܢܲܢ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ. ܒܲܪܸܟ݂ ܡܵܪܝ: ܐܲܪܟܸ݁ܢܘ ܪܹ̈ܫܲܝܟ݁ܘܿܢ ܠܲܣܝܵܡܝܼܕ݂ܵܐ ܘܩܲܒܸ݁ܠܘ ܒܘܼܪܟ݁ܬ݂ܵܐ.
	ܘܐܵܡܲܪ ܪܲܒܵܐ. ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܬܸܫܡܸܫܬܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ. ܘܡܲܩܸܦ ܨܠܘܿܬ݂ܵܐ. ܘܟܲܕ݂ ܫܵܠ̈ܡܵܢ ܢܲܦ̮̈ܫܵܬܲܢ ܠܲܚܕ݂ܵܐ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܓܡܝܼܪܬܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܟܠܲܢ ܒܲܚܕ݂ܵܐ ܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ: ܢܸܫܬ݁ܘܸܐ ܕܢܲܣܸܩ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ...
	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܣܘܼܝܵܟܹ̈ܐ. ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܘܓܲܐܝܘܼܬ݂ܵܐ. ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܨܓ‐ܨܗ
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܬܸܫܒ݁ܘܿܚܬܵܐ ܚܕܲܬܵܐ. ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܨܘ‐ܨܚ
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܟ݂ܠܗܘܿܢ ܥܹܐܕܹ̈ܐ ܡܵܪ̈ܵܢܵܝܹܐ. ܠܲܡܕܲܒ݁ܪܵܢܘܼܬ݂ܵܟ݂ ܡܵܪܝ ܬܡܝܼܗܬܵܐ ܘܠܵܐ ܡܸܬ݂ܡܲܠܠܵܢܝܼܬ݂ܵܐ: ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܘܒܲܚܢܵܢܵܐ ܐܸܬ݂ܓܲܡܪܲܬ݀ ܘܐܸܣܬܲܥܪܲܬ݀ ܘܐܸܫܬܲܡܠܝܲܬ݀ ܠܚܘܼܕܵܬܹܗ ܘܲܠܦܘܼܪܩܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ ܚܲܠܵܫܵܐ: ܒܪܹܫܝܼܬ݂ܵܐ ܕܡܸܢܲܢ: ܡܲܣܩܝܼܢܲܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕܬ݂ܵܐ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
 ܫܘܼܪܵܝܵܐ. ܒܪܲܒ݁ܘܼ ܡܵܪܲܢ. ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܢܙܘܼܥܘܼܢ ܥܲܡܡܹ̈ܐ.

 ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.*

ܝܵܬܸ݁ܒ݂ ܥܲܠ ܟܪ̈ܘܿܒܹܐ ܬܙܘܼܥ ܐܲܪܥܵܐ* ܡܵܪܝܵܐ ܪܲܒ݁ ܒܨܸܗܝܘܿܢ* ܘܪܵܡ ܥܲܠ ܟܠܗܘܿܢ ܥܲܡܡܹ̈ܐ* ܢܵܘܕ݁ܘܿܢ ܠܲܫܡܵܟ݂ ܪܲܒܵܐ ܘܲܕ݂ܚܝܼܠܵܐ ܕܩܲܕ݁ܝܼܫܘܼ* ܥܘܼܫܢܹܗ ܕܡܲܠܟܵܐ ܪܵܚܹܡ ܕܝܼܢܵܐ* ܐܲܢ݇ܬ݁ ܬܲܩܸܢܬ݁ ܬܪܝܼܨܘܼܬ݂ܵܐ ܘܕܝܼܢܵܐ* ܘܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ ܒܝܲܥܩܘܿܒ݂ ܐܲܢ݇ܬ݁ ܥܒܲܕ݂ܬ݁* ܪܲܡܪܸܡܘ ܠܡܵܪܝܵܐ ܐܲܠܵܗܲܢ* ܘܲܣܓ݂ܘܿܕ݂ܘ ܠܟ݂ܘܼܒ݂ܫܵܐ ܕܪܸ̈ܓ݂ܠܵܘܗܝ* ܩܲܕ݁ܝܼܫܘܼ ܡܘܼܫܹܐ ܘܐܲܗܪܘܿܢ ܒܟ݂ܘܼܡܪ̈ܵܘܗܝ* ܘܲܫܡܘܼܐܹܝܠ ܒܩܵܪ̈ܝܲܝ ܫܡܹܗ* ܕܲܩܪܵܘ ܠܡܵܪܝܵܐ ܘܲܥܢܵܐ ܐܸܢܘܿܢ* ܘܲܒ݂ܥܲܡܘܼܕ݂ܵܐ ܕܲܥܢܵܢܵܐ ܡܲܠܸܠ ܥܲܡܗܘܿܢ* ܢܛܲܪܘ ܣܵܗܕܘܼܬܹܗ ܘܲܩܝܵܡܵܐ ܕܝܲܗ݇ܒ݂ ܠܗܘܿܢ* ܡܵܪܝܵܐ ܐܲܠܵܗܵܢ ܐܲܢ݇ܬ݁ ܥܢܲܝܬ݁ ܐܸܢܘܿܢ* ܐܲܠܵܗܵܐ ܬܵܒ݂ܘܿܥܵܐ ܗܘܲܝܬ݁ ܠܗܘܿܢ* ܦܪܘܿܥ ܐܸܢܘܿܢ ܐܲܝܟ݂ ܥܒ݂ܵܕܲܝ̈ܗܘܿܢ* ܪܲܡܪܸܡܘ ܠܡܵܪܝܵܐ ܐܲܠܵܗܲܢ* ܘܲܣܓ݂ܘܿܕ݂ܘ ܠܛܘܼܪܹܗ ܩܲܕ݁ܝܼܫܵܐ* ܡܸܛܠ ܕܩܲܕ݁ܝܼܫܘܼ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ* ܫܲܒܲܚ. ܥܵܠܲܡ.
ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܕܒ݂ܵܣܵܠܝܼܩܹ̈ܐ. ܪܹܫ ܩܵܠܵܐ. ܐܲܡܠܸܟ݂ ܐܲܠܵܗܵܐ ܥܲܠ ܥܲܡܡܹ̈ܐ: ܐܲܡܠܸܟ݂ ܡܵܪܝܵܐ ܠܥܵܠܲܡ. ܕܨܘܼܪܬܵܐ. ܡܲܠܟ݁ܘܼܬܹܗ ܡܲܠܟ݁ܘܼܬ݂ ܥܵܠܲܡ. ܘܫܘܼܠܛܵܢܹܗ ܠܕ݂ܵܪܕܵܪ̈ܝܼܢ. ܨܠܝܼܒ݂ܵܟ݂ ܐܲܡܠܸܟ݂ ܒܲܫܡܲܝܵܐ. ܘܲܨܠܝܼܒ݂ܵܟ݂ ܐܲܡܠܸܟ݂ ܒܐܲܪܥܵܐ. ܘܲܨܠܝܼܒ݂ܵܟ݂ ܟܲܠܸܠ ܐܸܢܘܿܢ. ܠܟܸܢܫܹ̈ܐ ܕܐܵܘܕ݁ܝܼܘ ܒܲܨܠܝܼܒ݂ܵܟ݂: ܬܢܝܼ. ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ. ܕܨܘܼܪܬܵܐ. ܗܵܘ ܕܒܹܗ ܐܝܼܬ݂ ܠܲܢ ܦܘܼܪܩܵܢܵܐ ܘܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ. ܨܠܝܼܒ݂ܵܐ ܕܐܸܬ݂ܚ݇ܙܝܼ ܒܲܫܡܲܝܵܐ. ܘܐܸܬ݂ܓ݁ܠܝܼ ܒܪ̈ܲܚܡܹܐ ܠܐܲܪ̈ܥܵܢܹܐ. ܗܘ ܪܲܡܪܸܡ ܠܓܸܢܣܲܢ ܫܝܼܛܵܐ. ܘܥܲܠܝܼ ܣܵܡܹܗ ܒܲܫܡܲܝܵܐ. ܗܸܕ݂ܪܵܟ݂ ܘܫܘܼܒ݂ܚܵܟ݂ ܘܫܘܼܒ݂ܚܵܟ݂ ܙܵܟܹ݁ܐ. ܕܨܘܼܪܬܵܐ. ܥܒܲܕ݂ ܙܵܟ݂ܘܼܬ݂ܵܐ ܒܲܕ݂ܪܵܥܹܗ. ܨܠܝܼܒ݂ܵܐ ܙܟ݂ܵܐ ܨܠܝܼܒ݂ܵܐ ܙܵܟܹ݁ܐ. ܨܠܝܼܒ݂ܵܐ ܚܲܝܒܹܗ ܠܣܵܛܵܢܵܐ. ܘܐܲܒ݂ܗܸܬ݂ ܠܓ݂ܘܼܕܹ̈ܐ ܕܨܵܠܘܿܒ݂ܵܘ̈ܗܝ. ܘܚܲܕ݁ܝܼ ܠܟܸܢܫܹ̈ܐ ܕܣܵܓ݂ܘܿܕ݂ܵܘ̈ܗܝ. ܐܲܢܗܲܪܘ ܒܲܪ̈ܩܵܘܗܝ ܠܬܹܒܹܝܠ. ܕܨܘܼܪܬܵܐ. ܐܸܬ݂ܟܲܣܝܼܘ ܫܡܲܝ̈ܵܐ ܡܢ ܙܝܼܘܹܗ ܡܫܲܒ݁ܚܵܐ. ܨܠܝܼܒ݂ܵܐ ܕܢܘܼܗܪܵܐ ܕܐܸܬ݂ܚܵܘܝܼ. ܠܩܘܼܣܛܲܢܛܝܼܢܘܿܣ ܒܲܫܡܲܝܵܐ. ܐܲܝܟ݂ ܪܲܒ݁ ܚܲܝܠܵܐ ܐܵܙܹܠ ܗ݇ܘܵܐ. ܒܪܹܫ ܡܲܫܪܝܼܬܹܗ ܠܲܩܪܵܒ݂ܵܐ. ܡܢ ܥܲܡܵܐ ܕܠܵܐ ܡܪܲܚܡܵܢ. ܕܨܘܼܪܬܵܐ. ܬܘܼܒ݂ ܦܘܼܩܕܵܢܵܐ ܚܲܕ݂ܬ݂ܵܐ ܝܵܗܹܒ݂ ܐ݇ܢܵܐ ܠܟ݂ܘܿܢ. ܨܠܝܼܒ݂ܵܟ݂ ܡܵܪܲܢ ܕܐܸܬ݂ܚܲܦܝܼ. ܡܢ ܒܝܼܫܘܼܬ݂ܗܘܿܢ ܕܙܵܩܘܿܦܹ̈ܐ. ܦܪܲܚܘ ܙܲܠܝܼܩܵܘ̈ܗܝ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܐܲܢܗܲܪܘ ܠܥܵܠܡܵܐ ܡܢ ܣܵܟܹܗ. ܡܒܲܛܸܠ ܩܪ̈ܵܒܹܐ ܡܢ ܣܵܘܦܹܝ̈ܗ ܕܐܲܪܥܵܐ. ܕܨܘܼܪܬܵܐ. ܘܲܫܪܵܐ ܣܝܵܓ݂ܵܐ ܕܩܵܐܹܡ ܗ݇ܘܵܐ ܒܲܡܨܲܥܬ݂ܵܐ ܘܲܒ݂ܥܸܠܕܒ݂ܵܒ݂ܘܼܬ݂ܵܐ ܒܒܸܣܪܹܗ. ܨܠܝܼܒ݂ܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ. ܫܲܝܸܢ ܠܥܘܼܡܩܵܐ ܬܲܚܬܵܝܹ̈ܐ. ܕܪܵܘܡܵܐ ܠܵܐ ܣܢܝܼܩ ܥܲܠ ܫܲܝܢܵܐ. ܫܲܝܸܢ ܡܵܪܝ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ. ܕܲܢܚܵܘܸܐ ܓܲܢ݇ܒܵܪܘܼܬܹܗ. ܕܨܘܼܪܬܵܐ. ܐܲܪܝܼܡ ܪܹܫܹܗ ܘܪܲܡܪܡܹܗ. ܨܠܝܼܒ݂ܵܐ ܙܲܟ݁ܝܹܗ ܠܲܟ݂ܝܵܢܲܢ. ܨܠܝܼܒ݂ܵܐ ܐܲܩܝܼܡ ܡܲܦܘܼܠܬܲܢ. ܨܠܝܼܒ݂ܵܐ ܪܲܡܪܸܡ ܕܲܚܝܼܚܲܢ. ܘܥܲܠܝܼ ܣܵܡܹܗ ܒܲܫܡܲܝܵܐ. ܬܹܒܹܝܠ ܘܟ݂ܠܗܘܿܢ ܥܵܡܘܿܪܹ̈ܝܗ. ܕܨܘܼܪܬܵܐ. ܘܫܲܝܸܢ ܒܲܕ݂ܡܵܐ ܕܲܙܩܝܼܦܹܗ ܒܐܝܼܕ݂ܵܘ̈ܗܝ. ܒܲܨܠܝܼܒ݂ܵܟ݂ ܐܸܫܬܲܝܲܢܢ ܟܠ. ܘܒܲܨܠܝܼܒ݂ܵܟ݂ ܐܸܬ݂ܚܲܕܲܬ݂ ܟܠ. ܘܒܲܨܠܝܼܒ݂ܵܟ݂ ܡܵܪܝ ܢܲܛܲܪ ܠܲܢ. ܡܢ ܨܸܢܥܵܬܹ̈ܗ ܕܲܢܟ݂ܝܼܠܵܐ. ܥܲܠ ܥܲܡܵܟ݂ ܒܘܼܪܟ݁ܬ݂ܵܟ݂ ܠܥܵܠܲܡ. ܕܨܘܼܪܬܵܐ. ܡܲܢ ܠܵܐ ܢܸܕ݂ܚܲܠ ܠܵܟ݂ ܡܲܠܟܵܐ ܕܟ݂ܠܗܘܿܢ ܥܲܡܡܹ̈ܐ. ܒܲܨܠܝܼܒ݂ܵܟ݂ ܡܵܪܝ ܢܸܫܬܲܝܢܘܼܢ. ܟܵܗܢܹ̈ܐ ܘܡܲܠܟܹ̈ܐ ܡܪ̈ܲܥܝܵܢܲܝܟ. ܘܒܲܨܠܝܼܒ݂ܵܟ݂ ܬܸܪܘܲܙ ܥܹܕ݁ܬܵܟ݂. ܕܲܒ݂ܡܘܼܟܵܟ݂ܵܟ݂ ܪܲܡܪܸܡܬܵܗ. ܚܢܲܢ ܥܲܡܵܟ݂ ܘܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬ݂ܵܟ݂. ܕܨܘܼܪܬܵܐ. ܢܸܬ݂ܒܲܪ ܩܸܫܬܵܐ ܒܲܩܪܵܒ݂ܵܐ. ܘܲܢܡܲܠܸܠ ܫܠܵܡܵܐ ܥܲܡ ܥܲܡܡܹ̈ܐ. ܒܲܨܠܝܼܒ݂ܵܟ݂ ܡܵܪܝ ܢܸܫܬܲܝܲܢ. ܥܵܠܡܵܐ ܕܲܩܪ̈ܵܒܹܐ ܫܲܓ݂ܫܘܼܗܝ. ܘܲܒ݂ܦܘܼܩܕܵܢܵܟ݂ ܢܸܬ݂ܒܲܛܲܠ. ܣܲܝܦܵܐ ܕܚܵܪܹܒ݂ ܡܲܫܪܝܼܬܲܢ. ܚܵܘܝܼ ܡܵܪܝܵܐ ܦܘܼܪܩܵܢܹܗ. ܕܨܘܼܪܬܵܐ. ܠܲܝܬ݁ ܒܐ݇ܢܵܫ ܐ݇ܚܪܹܝܢ ܦܘܼܪܩܵܢܵܐ. ܒܲܨܠܝܼܒ݂ܵܐ ܗܘܵܐ ܦܘܼܪܩܵܢܵܐ. ܠܥܲܡܡܹ̈ܐ ܕܗܲܝܡܸܢܘ ܒܲܨܠܝܼܒ݂ܵܐ. ܘܒܲܨܠܝܼܒ݂ܵܐ ܗܘܵܐ ܚܘܼܝܵܒ݂ܵܐ. ܠܥܲܡܵܐ ܕܲܟ݂ܦܲܪ ܒܲܨܠܝܼܒ݂ܵܐ. ܡܸܛܠ ܕܡܲܒ݁ܘܼܥܵܐ ܕܚܲܝܹ̈ܐ ܥܲܡܵܟ݂ ܗ݇ܘ. ܕܨܘܼܪܬܵܐ. ܢܸܗܘܘܿܢ ܒܹܗ ܡܥܝܼܢܵܐ ܕܡܲܝ̈ܵܐ ܕܢܵܒ݂ܥܝܼܢ ܠܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ. ܕܵܡܹܐ ܨܠܝܼܒ݂ܵܐ ܩܲܕ݁ܝܼܫܵܐ. ܠܡܲܒ݁ܘܼܥܵܐ ܕܪܵܕܹ݁ܐ ܒܲܥܕܸܢ. ܘܐܸܫܬ݁ܝܼܘ ܡܸܢܹܗ ܚܲܟ݁ܝܼܡܹ̈ܐ. ܘܐܵܦ ܣܲܟ݂ܠܹ̈ܐ ܩܢܵܘ ܒܘܼܝܵܢܵܐ. ܢܸܗܘܸܐ ܐܲܝܟ݂ ܐܝܼܠܵܢܵܐ ܕܲܢܨܝܼܒ݂ ܥܲܠ ܬܲܦܵܐ ܕܡܲܝ̈ܵܐ. ܕܨܘܼܪܬܵܐ. ܐܲܝܟ݂ ܙܲܝܬܵܐ ܗܕ݂ܝܼܪܵܐ ܕܪܵܘܪ̈ܒ݂ܵܢ ܣܵܘܟܵܘ̈ܗܝ. ܕܵܡܹܐ ܨܠܝܼܒ݂ܵܐ ܩܲܕ݁ܝܼܫܵܐ. ܠܐܝܼܠܵܢ ܚܲܝܹ̈ܐ ܒܓ݂ܵܘ ܥܹܕ݁ܬܵܐ. ܕܦܹܐܪ̈ܵܘܜ ܥܵܗܢܝܼܢ ܠܡܹܐܟ݂ܘܼܠܬܵܐ. ܘܛܲܪ̈ܦܵܘܗܝ ܚܵܫܚܝܼܢ ܠܐܵܣܝܘܼܬ݂ܵܐ. ܫܲܦܝܼܪ ܒܚܸܙܘܹܗ ܡܢ ܒܢܲܝ̈ܢܵܫܵܐ. ܕܨܘܼܪܬܵܐ. ܐܲܝܟ݂ ܢܘܼܗܪܵܐ ܕܫܲܒ݂ܥܵܐ ܝܵܘܡ̈ܝܼܢ. ܕܵܡܹܐ ܨܠܝܼܒ݂ܵܐ ܩܲܕ݁ܝܼܫܵܐ. ܠܫܸܡܫܵܐ ܕܕ݂ܵܢܲܚ ܒܲܪܩܝܼܥܵܐ. ܕܙܲܠܝܼܩܵܘ̈ܗܝ ܦܵܪܚܝܼܢ ܒܐܵܐܲܪ. ܘܢܘܼܗܪܹܗ ܡܲܦܨܲܚ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܣܹܐܡܵܐ ܓܲܒ݂ܝܵܐ ܕܲܒ݂ܚܝܼܪ ܒܐܲܪܥܵܐ. ܕܨܘܼܪܬܵܐ. ܩܲܒ݁ܠܘܼܗܝ ܗܵܟܹܝܠ ܒܡܵܪܝܵܐ ܒܟ݂ܠ ܚܲܕ݂ܘܵܐ. ܕܵܡܹܐ ܨܠܝܼܒ݂ܵܐ ܩܲܕ݁ܝܼܫܵܐ. ܠܡܲܪܓܵܢܝܼܬ݂ܵܐ ܕܠܵܐ ܡܘܼܡܵܐ. ܕܙܲܒ݁ܢܘܼܗ ܪ̈ܲܒܲܝ ܬܲܓܵܪܹ̈ܐ. ܘܲܥܬܲܪܘ ܡܸܢܵܗ ܘܐܸܬ݂ܪܲܡܪܲܡܘ. ܠܲܡܗܲܝܡܢܹ̈ܐ ܢ̇ܵܛܲܪ ܡܵܪܝܵܐ. ܕܵܡܹܐ ܟܸܢܫܵܐ ܕܲܡܗܲܝܡܢܹ̈ܐ. ܟܲܕ݂ ܩܵܐܹܡ ܒܓ݂ܵܘ ܥܹܕܵܬ݂̈ܵܐ. ܠܟܸܢܫܹ̈ܐ ܕܥܝܼܪܹ̈ܐ ܕܲܡܙܲܝܚܝܼܢ. ܠܲܫܟ݂ܝܼܢܬܵܐ ܐܲܠܵܗܵܝܬܵܐ. ܐܲܝܟ݂ ܪ̈ܣܝܼܣܹܐ ܕܢܵܚܬ݁ܝܼܢ ܥܲܠ ܐܲܪܥܵܐ. ܕܵܡܹܐ ܨܠܝܼܒܹܗ ܕܡܲܚܝܵܢܲܢ. ܠܗܵܘ ܕܐܲܪܝܼܡ ܡܘܼܫܹܐ ܒܡܲܕ݂ܒ݁ܪܵܐ. ܗܵܘ ܠܥܸܒ݂ܪ̈ܵܝܹܐ ܐܲܚܝܼ ܗ݇ܘܵܐ. ܘܗܵܢܵܐ ܠܐܲܪ̈ܒܲܥܦܸܢܝܵܬ݂ܵܐ. ܢܸܫܬܲܒ݂ܗܲܪ ܟܠ ܕܝ̇ܵܡܹܐ ܒܹܗ. ܠܵܐ ܒܵܗܬ݁ܝܼܢܲܢ ܒܲܨܠܝܼܒ݂ܵܟ݂. ܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ. ܕܒܹܗܘ ܦܪܲܩܬܵܝܗܝ ܠܲܟ݂ܝܵܢܲܢ. ܡܢ ܫܘܼܥܒܵܕ݂ܵܗ ܕܲܚܛܝܼܬ݂ܵܐ. ܡܢ ܣܵܘܦܹܝ̈ܗ ܠܣܵܘܦܹܝ̈ܗ ܕܐܲܪܥܵܐ. ܨܠܝܼܒ݂ܵܐ ܕܲܡܬ݂ܝܼܚ ܫܘܼܠܛܵܢܹܗ. ܒܲܫܡܲܝܵܐ ܘܲܠܬܲܚܬ݁ ܒܐܲܪܥܵܐ. ܫܲܝܢܵܟ݂ ܢܲܡܠܸܟ݂ ܥܲܠ ܥܹܕ݁ܬܵܟ݂ ܕܬܸܙܡܲܪ ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܟܠܝܘܿܡ. ܡܢ ܡܲܕ݂ܢܵܚܵܘ̈ܜ ܕܫܸܡܫܵܐ ܘܲܥܕܲܡܵܐ ܠܡܲܥܪ̈ܵܒ݂ܵܘܜ. ܕܨܘܼܪܬܵܐ. ܘܬܸܫܒ݁ܘܿܚܬܹ݁ܗ ܐܸܬ݂ܡܲܠܝܲܬ݀ ܐܲܪܥܵܐ. ܫܘܼܦܪܹܗ ܪܓ݂ܝܼܓ݂ܵܐ ܕܲܨܠܝܼܒ݂ܵܐ. ܐܲܙܠܸܓ݂ ܥܲܠ ܟܠܵܗ ܬܹܒܹܝܠ. ܘܲܩܢܵܘ ܣܲܒ݂ܪܵܐ ܫܲܪܝܼܪܵܐ. ܫܡܲܝܵܢܹ̈ܐ ܘܐܲܪ̈ܥܵܢܵܝܹܐ. ܒܲܕ݂ܪܵܥܵܟ݂ ܥܲܫܝܼܢܵܐ ܒܲܕܲܪܬ݁ ܠܲܒ݂ܥܸܠܕܒ݂ܵܒܲܝ̈ܟ. ܕܨܘܼܪܬܵܐ. ܫܸܡܫܵܐ ܘܣܲܗܪܵܐ ܩܵܡܘ ܒܡܸܕ݂ܝܵܪ̈ܲܝܗܘܿܢ. ܐܘܿ ܨܠܝܼܒ݂ܵܐ ܕܲܡܠܸܐ ܚܢܵܢܹ̈ܐ. ܡܲܢ ܐܲܦ̮ܩܵܟ݂ ܠܒܹܝܬ݂ ܐܘܼܪ̈ܚܵܬ݂ܵܐ. ܒܸܪ̈ܝܵܬ݂ܵܐ ܚܙܹܝܬ݂ ܕܲܫܓ݂ܝܼ̈ܫܵܢ. ܘܢܸܦ݁ܩܹܬ݂ ܕܐܹܫܲܝܸܢ ܐܸܢܹܝܢ. ܢܲܛܲܪܲܝܢܝ ܐܲܝܟ݂ ܒܵܒ݂ܬ݂ܵܐ ܕܥܲܝܢܵܐ. ܨܠܝܼܒ݂ܵܟ݂ ܢܸܗܘܸܐ ܢ̇ܵܛܘܿܪܵܐ. ܠܟܸܢܫܹ̈ܐ ܣܵܓ݂ܕܲܝ̈ ܠܲܨܠܝܼܒ݂ܵܟ݂. ܘܒܲܨܠܝܼܒ݂ܵܟ݂ ܢܲܛܲܪ ܠܥܹܕ݁ܬܵܟ݂. ܡܢ ܓܘܼܕܵܦܵܐ ܕܡܲܪ̈ܵܚܹܐ. ܚܢܲܢ ܥܲܡܵܟ݂ ܘܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬ݂ܵܟ݂. ܒܲܨܠܝܼܒ݂ܵܟ݂ ܡܵܪܝ ܢܸܬ݂ܢܛܲܪ. ܟܸܢܫܵܐ ܕܙܲܝܚ ܥܲܕ݂ܥܹܕ݂ܵܟ݂. ܘܲܒ݂ܛܸܠܵܠܹܗ ܢܸܣܬܲܬܲܪ. ܡܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹ̈ܗ. ܕܲܨܠܘܿܬ݂ܵܐ܀ ܝܲܡܝܼܢܵܟ݂ ܡܵܪܝܵܐ ܐܲܓܸ݁ܢ ܥܠܲܝ. ܐܘܿ ܕܲܦܫܝܼܛܵܐ ܝܲܡܝܼܢܹܗ. ܘܲܦܬ݂ܝܼܚ ܬܲܪܥܹܗ ܠܬܲܝ̇ܵܒܹ̈ܐ. ܦܬܲܚ ܠܵܗ ܬܲܪܥܵܐ ܠܲܨܠܘܿܬܲܢ. ܘܬܸܥܘܿܠ ܩܕ݂ܵܡܲܝܟ ܒܵܥܘܼܬܲܢ. ܕܒܲܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ ܣܲܒܲܪ݇ܢ. ܡܫܝܼܚܵܐ ܕܒܲܫܡܹܗ ܟܢܝܼܫܝܼܢܲܢ. ܘܥܲܠ ܛܲܝܒ݁ܘܼܬܹܗ ܬܟ݂ܝܼܠܝܼܢܲܢ. ܢܲܛܲܪ ܠܟܸܢܫܲܢ ܒܲܨܠܝܼܒ݂ܵܟ݂. ܕܢܸܫܬܲܡܠܸܐ ܒܲܢ ܨܸܒ݂ܝܵܢܵܟ݂. ܒܟ݂ܠܙܒܲܢ ܘܲܒ݂ܟ݂ܠܥܸܕܵܢ. ܨܠܘܿܬ݂ܵܗ̇ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܡܲܪܝܲܡ. ܐܸܡܹܗ ܕܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ. ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂. ܫܘܼܪܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ. ܒܥܵܘ ܠܡܵܪܝܵܐ ܘܐܸܬ݂ܚܲܝܲܠܘ. ܢܒ݂ܝܼܹ̈ܐ ܨܲܠܵܘ ܥܲܠ ܫܲܝܢܵܐ. ܘܲܫܠܝܼܚܹ̈ܐ ܥܲܠ ܫܠܵܡܵܐ. ܘܣܵܗܕܹ̈ܐ ܘܟ݂ܵܗܢܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ. ܨܠܘܿܬ݂ܟ݂ܘܿܢ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܲܢ. ܫܲܒܲܚ. ܨܠܝܼܒ݂ܵܟ݂ ܐܲܡܠܸܟ݂ ܒܲܫܡܲܝܵܐ. ܘܲܨܠܝܼܒ݂ܵܟ݂ ܐܲܡܠܸܟ݂ ܒܐܲܪܥܵܐ. ܘܲܨܠܝܼܒ݂ܵܟ݂ ܟܲܠܸܠ ܐܸܢܘܿܢ. ܠܟܸܢܫܹ̈ܐ ܕܐܵܘܕ݁ܝܼܘ ܒܲܨܠܝܼܒ݂ܵܟ݂. ܥܵܠܲܡ. ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ. ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ. ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ. ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
ܡܲܩܦܝܼܢ ܒܩܵܠܵܐ ܬܲܪ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ
ܘܡܲܩܦܝܼܢ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܥܲܡ ܩܵܢܘܿܢܵܗ̇: ܐܲܝܟ݂ ܕܲܒ݂ܫܘܼܪܵܝܵܐ.
ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܫܡܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܩܲܕ݁ܝܼܫܵܐ ܢܸܫܬܲܒܲܚ: ܘܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܬܸܣܬ݁ܓܸܕ݂: ܘܡܵܪܘܼܬ݂ܵܟ݂ ܬܸܬ݂ܝܲܩܲܪ: ܘܪܲܒ݁ܘܼܬ݂ܵܟ݂ ܬܸܬ݂ܢܲܨܲܚ: ܘܐܝܼܬ݂ܘܼܬ݂ܵܟ݂ ܬܸܬ݂ܪܲܡܪܲܡ: ܘܥܲܠ ܥܲܡܵܟ݂ ܘܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬ݂ܵܟ݂: ܢܓ݂ܘܼܠܘܼܢ ܪ̈ܲܚܡܹܐ ܡܬ݂ܘܿܡܵܝܹ̈ܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܡܵܪܵܐ ܕܟ݂ܠ.
	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܚܵܬ݂ܵܗ̇. ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܒܪܝܼܟ݂ܵܐ ܡܵܪܝ ܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܲܣܓ݂ܝܼܕ݂ܵܐ ܡܵܪܘܼܬ݂ܵܟ݂: ܩܲܕ݁ܝܼܫ ܘܲܫܒ݂ܝܼܚ ܘܲܡܫܲܒܲܚ: ܘܪܵܡ ܘܲܡܥܲܠܲܝ ܫܡܵܐ ܣܓ݂ܝܼܕ݂ܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
	ܨܠܘܿܬ݂ܵܐ. ܥܘܼܕ݂ܪܵܢܵܐ ܡܵܪܝ ܕܪ̈ܲܚܡܲܝܟ ܘܣܘܼܝܵܥܵܐ ܪܲܒܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܚܲܝܠܵܐ ܟܲܣܝܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܘܝܲܡܝܼܢܵܟ݂ ܕܡܲܠܝܵܐ ܪ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ: ܐܲܓܸ݁ܢ ܘܐܲܠܘܵܐ ܠܲܡܚܝܼܠܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܡܢ ܒܲܝܬܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ ܡܠܸܐ ܟܠ ܥܘܼܕ݂ܪ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܒܲܨܠܘܿܬ݂ ܟܠܗܘܿܢ ܩܲܕ݁ܝܼܫܹ̈ܐ ܡܪ̈ܲܥܝܵܢܲܝܟ: ܡܵܪܵܐ ܕܟ݂ܠ...
	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ ܒܲܪܸܟ݂ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ. ܒܒ݂ܘܼܪܟ݁ܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܢܸܬ݂ܒܲܪܟ݂ܘܼܢ ܥܲܒ݂ܕܲܝ̈ܟ: ܘܒܲܒ݂ܛܝܼܠܘܼܬܹܗ ܕܨܸܒ݂ܝܵܢܵܟ݂ ܢܸܬ݂ܢܲܛܪܘܼܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܘܫܲܝܢܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܲܫܠܵܡܵܐ ܢܲܓ݁ܝܼܪܵܐ ܕܡܵܪܘܼܬ݂ܵܟ݂: ܢܲܡܠܸܟ݂ ܒܥܲܡܵܟ݂ ܘܲܒ݂ܥܹܕ݁ܬܵܟ݂. ܟܠܗܘܿܢ ܝܵܘܡܵܬܹ̈ܗ ܕܥܵܠܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ...

ܘܡܲܩܦܝܼܢ. ܨܠܘܿܬ݂ܵܗ̇ ܡܵܪܝ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܗ̇ ܕܐܸܡܵܐ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܬܲܟ݂ܫܲܦ̮ܬܵܗ̇ ܘܬܲܚܢܲܢܬܵܗ̇ ܕܡܲܠܝܲܬ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܐ: ܡܵܪܬ݁ܝ ܡܲܪܝܲܡ ܛܘܼܒ݂ܵܢܝܼܬ݂ܵܐ: ܘܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ ܙܲܟܵܝܵܐ: ܘܥܘܼܕ݂ܪܵܢܵܐ ܐܲܠܵܗܵܝܵܐ: ܘܒ݂ܵܥܘܼܬܹܗ ܕܡܵܪܝ ܝܘܿܚܲܢܵܢ ܡܲܥܡܕ݂ܵܢܵܐ: ܘܕܲܫܠܝܼܚܹ̈ܐ ܛܘܼܒ݂ܵܢܹ̈ܐ: ܘܕܲܐܒ݂ܵܗ̈ܵܬ݂ܵܐ ܢܲܨܝܼܚܹ̈ܐ ܡܣܲܒ݁ܪ̈ܵܢܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܘܙܵܪ̈ܥܲܝ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܕܲܐܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܡܵܪܝ ܐܲܒ݂ܵܐ ܩܵܬ݂ܘܿܠܝܼܩܵܐ: ܘܡܵܪܝ ܐܸܣܛܲܦܵܢܘܿܣ ܒܘܼܟ݂ܪܵܐ ܕܣܵܗܕܹ̈ܐ: ܘܲܕ݂ܓܲܢ݇ܒܵܪ ܚܲܝܠܵܐ ܡܵܪܝ ܓܝܼܘܲܪܓܝܼܣ ܣܵܗܕܵܐ ܢܲܨܝܼܚܵܐ: ܘܲܕ݂ܡܵܪܝ ܦܠܵܢ: ܘܲܕ݂ܡܵܪܝ ܐܵܘܓܹ݁ܝܢ ܛܘܼܒ݂ܵܢܵܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܣܝܼܥܬܹ݁ܗ ܪܘܼܚܵܢܵܝܬܵܐ. ܘܕܲܫܡܘܿܢܝܼ ܘܕܲܒ݂ܢܹܝ̈ܗ̇: ܘܲܕ݂ܟ݂ܠܗܘܿܢ ܣܵܗܕܵܘ̈ܗܝ ܘܩܲܕ݁ܝܼ̈ܫܵܘܗܝ ܕܡܵܪܲܢ: ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܫܘܼܪܵܐ ܪܵܡܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܚܲܣܝܼܢܵܐ: ܕܬܸܦܪܘܿܩ ܘܲܬ݂ܦܲܨܸܐ ܘܲܬ݂ܫܵܘܙܸܒ݂ ܘܲܬ݂ܢܲܛܲܪ ܠܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ ܡܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹܗ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹ̇ܝܢ: ܐܵܡܹܝܢ.
ܘܟܸܢ ܚܵܬ݂ܡ̇ܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܪ̈ܵܝܹܐ. ܘܢܵܣܹ̇ܒ݂ ܟܵܗܢܵܐ ܨܠܝܼܒ݂ܵܐ ܒܐܝܼܕܹܗ ܕܝܲܡܝܼܢܵܐ: ܘܩܵܐܹ̇ܡ ܒܓܲܒܵܐ ܬܲܝܡܢܵܝܵܐ: ܘܡܲܗܦܸܟ݂ ܐܲܦܵܘ̈ܗܝ ܠܥܲܡܵܐ ܘܐܵܡܲܪ: ܒܲܪܸܟ݂ ܡܵܪܝ ܒܦܘܼܩܕܵܢܟ݂ܘܿܢ. ܘܥܵܢܹ̇ܝܢ: ܒܦܘܼܩܕܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܡܵܪܲܢ: ܘܫܘܼܒ݂ܚܵܐ ܠܲܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ. ܘܡܲܪܟ݁ܢܝܼܢ ܥܲܡܵܐ ܪܹ̈ܫܲܝܗܘܿܢ.
ܘܡܲܩܸܦ ܚܘܼܬܵܡܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ: ܨܸܡܚܹܗ ܕܐܲܒ݂ܵܐ ܡܬ݂ܘܿܡܵܝܵܐ: ܝܼܠܝܼܕ݂ܵܐ ܕܠܵܐ ܫܘܼܪܵܝܵܐ: ܠܥܸܠ ܡܢ ܙܲܒ݂ܢܹ̈ܐ ܘܲܗܘܲܝ̈ܵܐ: ܕܠܲܝܬ݁ ܠܲܢ ܣܲܒ݂ܪܵܐ ܘܣܘܼܟܵܝܵܐ: ܐܸܠܵܐ ܐܸܢ ܐܲܢ݇ܬ݁ ܒܵܪܘܿܝܵܐ. ܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܘܲܓ݂ܒܲܝ̈ܵܐ: ܕܲܫܦܲܪܘ ܠܵܟ݂ ܡܢ ܫܘܼܪܵܝܵܐ: ܥܒܸܕ݂ ܠܲܚܛܵܗܲܝ̈ܢ ܚܘܼܣܵܝܵܐ: ܘܲܠܣܲܟ݂ܠ̈ܘܵܬܲܢ ܫܪܵܝܵܐ: ܘܲܠܥܵܩ̈ܵܬܲܢ ܦܘܼܨܵܝܵܐ: ܘܲܠܫܸܐ̈ܠܵܬܲܢ ܦܘܼܢܵܝܵܐ: ܘܡܲܢܲܥܲܝܢ ܠܢܘܼܗܪܵܟ݂ ܓܲܐܝܵܐ: ܘܢܲܛܲܪܲܝܢ ܒܢܝܼܫܵܟ݂ ܚܲܝܵܐ: ܡܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܟܣܲܝ̈ܵܐ ܘܲܓ݂ܠܲܝ̈ܵܐ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ). ܗܵܫܵܐ ♰ ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܪܵܫܹ̇ܡ ܨܠܝܼܒ݂ܵܐ ܥܲܠ ܥܲܡܵܐ: ܟܲܕ݂ ܦܵܫܹ̇ܛ ܐܝܼܕܹܗ ܠܥܸܠ ܠܐܲܦܲܝ̈ ܡܲܥܪܒ݂ܵܐ: ܘܡܲܝܬܸ݁ܐ ܠܵܗ̇ ܠܡܲܕ݂ܢܚܵܐ: ܘܟܸܢ ܦܵܫܹ̇ܛ ܠܵܗ̇ ܠܓܲܒܵܐ ܕܬܲܝܡܢܵܝܵܐ: ܘܡܵܘܒܸ݁ܠ ܠܵܗ̇ ܠܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ. ܐܲܝܟܲܢܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܸܣܬ݁ܡܸܟ݂ ܥܲܠ ܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܓܲܒܵܐ ܣܸܡܵܠܵܝܵܐ ܕܟܸܢܫܵܐ.
ܘܥܵܢܹ̇ܝܢ: ܡܫܝܼܚܵܐ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬ݂ܵܟ݂: ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܟܵܗܢܘܼܬ݂ܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
ܒܲܫܪܵܪܵܐ ܓܹܝܪ ܕܠܵܐ ܦܘܼܠܵܓ݂ܵܐ: ܡܗܲܝܡܢܝܼܢܲܢ ܒܚܲܕ݂ ܐܲܠܵܗܵܐ: ܐܲܒ݂ܵܐ ܐܲܚܝܼܕ݂ ܟܠ. ܥܵܒ݂ܘܿܕ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܐܲܝܠܹܝܢ ܕܡܸܬ݂ܚܲܙ̈ܝܵܢ ܘܲܕ݂ܠܵܐ ܡܸܬ݂ܚܲܙ̈ܝܵܢ. ܘܲܒ݂ܚܲܕ݂ ܡܵܪܝܵܐ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ. ܒܪܹܗ ܕܐܲܠܵܗܵܐ ܝܼܚܝܼܕ݂ܵܝܵܐ: ܒܘܼܟ݂ܪܵܐ ܕܟ݂ܠܗܹܝܢ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܗܵܘ̇ ܕܡܢ ܐܲܒ݂ܘܼܗܝ ܐܸܬ݂ܝܼܠܸܕ݂ ܩܕ݂ܵܡ ܟܠܗܘܿܢ ܥܵܠܡܹ̈ܐ: ܘܠܵܐ ܐܸܬ݂ܥܒܸܕ݂. ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܕܡܢ ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܒܲܪ ܟܝܵܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܕܒ݂ܐܝܼ̈ܕ݂ܵܘܗܝ ܐܸܬ݁ܬܲܩܲܢܘ ܥܵܠܡܹ̈ܐ: ܘܐܸܬ݂ܒ݁ܪܝܼ ܟܠ ܡܸܕ݁ܡ. ܗܵܘ̇ ܕܡܸܛܠܵܬܲܢ ܒܢܲܝ̈ܢܵܫܵܐ: ܘܡܸܛܠ ܦܘܼܪܩܵܢܲܢ: ܢܚܸܬ݂ ܡܢ ܫܡܲܝܵܐ: ܘܐܸܬ݂ܓܲܫܲܡ ܡܢ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܘܲܗܘܵܐ ܒܲܪܢܵܫܵܐ: ܘܐܸܬ݂ܒ݁ܛܸܢ ܘܐܸܬ݂ܝܼܠܸܕ݂ ܡܢ ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܘܚܲܫ ܘܐܸܙܕ݁ܩܸܦ: ܒܝܵܘܡܲܝ̈ ܦܲܢܛܝܼܘܿܣ ܦܝܼܠܵܛܘܿܣ. ܘܐܸܬ݂ܩ̄ܒܲܪ: ܘܩܵܡ ܠܲܬ݂ܠܵܬ݂ܵܐ ܝܵܘܡܝܼ̈ܢ ܐܲܝܟ݂ ܕܲܟ݂ܬ݂ܝܼܒ݂. ܘܲܣܠܸܩ ܠܲܫܡܲܝܵܐ: ܘܝܼܬܸܒ݂ ܡܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܘܬ݂ܘܼܒ݂ ܥܬ݂ܝܼܕ݂ ܠܡܹܐܬ݂ܵܐ ܠܲܡܕ݂ܵܢ ܠܡܝܼܬܹ̈ܐ ܘܲܠܚܲܝܹ̈ܐ. ܘܲܒ݂ܚܲܕ݂ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܪܘܼܚܵܐ ܕܲܫܪܵܪܵܐ: ܗܵܘ̇ ܕܡܢ ܐܲܒ݂ܵܐ ܢܵܦܹ݁ܩ: ܪܘܼܚܵܐ ܡܲܚܝܵܢܵܐ. ܘܒܲܚܕ݂ܵܐ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܘܲܫܠܝܼܚܵܝܬܵܐ ܩܵܬ݂ܘܿܠܝܼܩܝܼ. ܘܡܵܘܕܹ݁ܝܢܲܢ ܒܲܚܕ݂ܵܐ ܡܲܥܡܘܿܕ݂ܝܼܬ݂ܵܐ: ܠܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ. ܘܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܲܒ݂ܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ: ܐܵܡܹܝܢ.
