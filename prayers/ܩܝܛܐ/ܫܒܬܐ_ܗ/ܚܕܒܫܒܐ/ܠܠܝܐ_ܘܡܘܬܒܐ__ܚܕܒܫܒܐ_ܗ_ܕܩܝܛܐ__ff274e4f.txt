ܛܲܟ݂ܣܵܐ ܕܬܸܫܡܸܫܬܵܐ
ܕܠܸܠܝܵܐ ܘܕܨܲܦܪܵܐ
 ܒܫܸܡ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕܫܵܐ ܠܥܵܠܡܝܼܢ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟ݂: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟ݂: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬ݂ܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹܝ̇ܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.

ܡܵܘܬ݁ܒ݂ܵܐ
	ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܡܵܘܬ݁ܒ݂ܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ. ܨܠܘܿܬܲܢ ܡܵܪܝ ܬܸܒ݂ܣܲܡ ܠܵܟ݂: ܘܒ݂ܵܥܘܼܬܲܢ ܬܸܥܘܿܠ ܩܘܼܕ݂ܡܲܝܟ: ܘܡܢ ܒܹܝܬ݂ ܓܲܙܹܗ ܪܲܒܵܐ ܕܲܚܢܵܢܵܟ݂: ܢܸܬ݂ܦܲܢ̈ܝܵܢ ܫܹܐ̈ܠܵܬ݂ܵܐ ܠܲܣܢܝܼܩܘܼܬܲܢ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܦܵܪܘܿܩܲܢ. ܥܘܼܕ݂ܪܵܢ ܛܲܝܒܘܼܬ݂ܵܟ݂ ܐܲܠܘܵܐ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܡܫܝܼܚܵܐ ܕܐܸܬ݂ܵܐ ܠܦܘܼܪܩܵܢܲܢ. ܐܸܬ݁ܕܲܟ݂ܪ ܥܹܕ݁ܬܵܟ݂ ܕܲܩܢܲܝܬ݁ ܡܢ ܩܕ݂ܝܼܡ. ܐܸܬ݁ܕܲܟ݂ܪ ܡܵܪܝ ܥܹܕܬܵܟ݂ ܕܲܩܢܲܝܬ݁ ܡܢ ܩܕ݂ܝܼܡ. ܘܛܲܪ ܠܵܗ̇ ܫܠܵܡܵܐ ܢܲܓܝܼܪܵܐ. ܐܲܢ݇ܬ݁ ܕܐܘܼܪܚܵܐ܀ ܚܵܘܵܢܝ ܡܵܪܝܵܐ ܐܘܼܪܚܵܟ݂ ܘܐܹܗܲܠܸܟ݂ ܒܩܘܼܫܬܵܐ. ܗܲܒ݂ܠܲܢ ܡܵܪܝ ܕܢܸܪܕܸ݁ܐ ܘܢܸܗܦܘܿܟ݂ ܒܲܫܠܵܡܵܐ. ܐܲܝܟ݂ ܝܲܥܩܘܿܒ݂ ܡܸܢ ܒܹܝܬ݂ ܠܵܒ݂ܵܢ. ܕܡܸܛܪܵܐ܀ ܥܲܝܢܹ̈ܐ ܕܟ݂ܠܢܵܫ ܠܘܵܬ݂ܵܟ݂ ܬܲܠ̈ܝܵܢ. ܠܛܲܠܵܐ ܡܣܲܟܹ݁ܝܢܲܢ ܠܡܸܛܪܵܐ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܐܸܡܹܗ ܕܲܡܫܝܼܚܵܐ. ܬܸܗܘܸܐ ܫܘܼܪܵܐ ܠܢܲܦ̮̈ܫܵܬܲܢ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܒܥܵܘ ܠܡܵܪܝܵܐ ܘܐܸܬ݂ܚܲܝܲܠܘ. ܢܒ݂ܝܼܹ̈ܐ ܘܲܫܠܝܼܚܹ̈ܐ ܘܣܵܗܕܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ. ܨܠܘܿܬ݂ܟ݂ܘܿܢ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܲܢ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܐܲܝܟ݂ ܡܕ݂ܝܼܢ݇ܬܵܐ ܕܲܟ݂ܪܝܼܟ݂ ܠܵܗ̇ ܫܘܼܪܵܐ. ܨܠܝܼܒ݂ܵܟ݂ ܦܵܪܘܿܩܲܢ ܫܘܼܪܵܐ ܢܸܗܘܸܐ ܠܲܢ. ܘܒܹܗ ܢܸܙܟܹܝܘܗܝ ܠܲܒ݂ܥܸܠܕܲܪܵܐ. ܕܐܲܒ݂ܘܼܢ܀ ܒܥܝܼ ܡܢ ܡܵܪܝܵܐ ܘܨܲܠܵܐ ܩܕ݂ܵܡܵܘܗܝ. ܐܲܒ݂ܘܼܢ ܩܲܕܝܼܫܵܐ ܐܲܦܝܼܣ ܠܲܡܫܝܼܚܵܐ. ܕܲܢܪܲܚܸܡ ܥܲܠ ܢܲܦ̮̈ܫܵܬܲܢ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܠܵܐ ܬܸܫܕܹܝܢܝ ܡܢ ܩܕ݂ܵܡܲܝܟ. ܡܫܝܼܚܵܐ ܡܠܸܐ ܪ̈ܲܚܡܹܐ ܢܓ݂ܘܼܠܘܼܢ ܥܠܲܝܢ ܪ̈ܲܚܡܲܝܟ. ܒܗ̇ܘ ܒܹܝܬ݂ ܕܝܼܢܵܐ ܕܟܹܐܢܘܼܬ݂ܵܟ݂. ܡܵܪܝܵܐ ܒܨܲܦܪܵܐ ܬܸܫܡܲܥ ܩܵܠܝ. ܒܫܵܥܬ݂ܵܐ ܡܵܐ ܕܬ݂ܵܩܹ̇ܢ ܟܘܼܪܣܝܵܟ݂ ܠܒ݂ܘܼܚܪܵܢܵܐ. ܠܵܐ ܬܲܥܠܲܢ ܕܥܲܢܝܼܕܹ̈ܐ. ܛܘܼܒ݂ܵܘܗܝ ܠܡܲܢ ܕܐܸܫܬ݁ܒܸܩ ܠܹܗ ܥܵܘܠܹܗ. ܒܐܲܬ݂ܪܵܐ ܕܚܝܼܠܵܐ ܬܸܕ݂ܢܲܚ ܠܝܼ ܡܵܪܝ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܐܸܚܙܸܐ ܚܢܵܢܵܟ݂. ܬܵܘ ܢܸܒ݂ܪܘܿܟ݂ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܹܗ. ܣܵܓ݂ܕ݁ܝܼܢܲܢ ܠܵܟ݂ ܒܵܪܘܿܝܲܢ ܘܲܡܡܝܼܬ݂ܵܢܲܢ ܘܲܡܢܲܚܡܵܢܲܢ.
ܐܲܩܸܦ. ܒܐ݇ܪܵܙܵܐ ܪܲܒܵܐ. ܬܵܘ ܚܙܵܘ ܥܒ݂ܵܕ݂ܵܘ̈ܗܝ ܕܐܲܠܵܗܵܐ. ܒܬܲܗܪܵܐ ܘܲܠܫܘܼܒ݂ܚܵܐ ܕܗܵܘ̇ ܒܵܪܘܿܝܵܐ ܚܲܟ݁ܝܼܡܵܐ. ܡܛܲܝܸܒ݂ ܘܲܡܥܝܼܪ ܟܠܝܘܿܡ ܠܲܡܠܝܼܠܹ̈ܐ. ܗܵܢܵܐ ܥܵܠܡܵܐ ܒܬ݂ܘܼܩܵܢܹܗ. ܫܘܼܚܠܵܦܹ̈ܐ ܓܸܝܪ ܕܬܸܕ݂ܡܘܼܪܬܵܐ ܡܠܲܚܡܝܼܢ ܒܹܗ. ܗܵܢ̇ܘܿܢ ܕܣܲܩܘܼܒ݂ܠܵܐ ܐܸܢܘܿܢ ܠܲܚܕ݂ܵܕܹ̈ܐ. ܢܘܼܪܵܐ ܘܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܘܐܵܐܲܪ ܐܲܫܝܼܕ݂ܵܐ. ܘܲܕ݂ܠܵܐ ܢܸܛܥܹܐ ܘܢܸܣܒܲܪ ܒܣܘܿܓ݂ܵܐܐ ܕܫܘܼܚܠܵܦܲܝ̈ܗܘܿܢ. ܕܥܵܒ݂ܘܿܕܹ̈ܐ ܣܲܓ݁ܝܼܐܹ̈ܐ ܐܝܼܬ݂. ܚܲܕ݂ ܓܘܼܫܡܵܐ ܫܩܲܠ ܥܲܒ݂ܕܵܗ̇ ܠܲܒ݂ܪܝܼܬ݂ܵܐ ܒܲܓ݂ܒ݂ܝܼܠܬܹ݁ܗ ܕܒܲܪܢܵܫܵܐ. ܘܒܹܗ ܐܵܘܕ݁ܥܲܢ ܕܗܘ ܐܝܼܬ݂ܵܘܗܝ ܡܵܪܵܐ ܕܟ݂ܠ. ܡܸܛܠ ܐܝܼܩܵܪܵܐ ܕܲܫܡܵܟ݂. ܐܲܢ݇ܬ݁ܘܼ ܢܘܼܗܪܵܐ ܡܫܲܕܲܪ ܐܲܢ݇ܬ݁ ܠܥܵܠܡܵܐ ܟܠܹܗ ܡܢ ܣܵܟܹܗ. ܒܝܲܕ݂ ܪܸܡܙܵܐ ܕܨܸܒ݂ܝܵܢܵܟ݂. ܘܲܠܚܸܫܘܿܟ݂ܵܐ ܕܠܸܠܝܵܐ ܡܲܥܒܲܪ ܦܘܼܩܕܵܢܵܟ݂ ܡܵܪܝܵܐ. ܡܢ ܥܲܡܵܐ ܕܡܵܘܕܸ݁ܐ ܠܵܟ݂. ܠܵܐ ܡܵܪܝ ܬܲܗܡܸܐ ܐܸܠܵܐ ܦܪܘܿܩܲܝܢ ܘܪܲܚܸܡܥܠܲܝܢ. ܕܲܨܠܘܿܬ݂ܵܐ. ܠܟ݂ܠܗܘܿܢ ܚܲܛܵܝܹ̈ܐ. ܫܘܼܚܠܵܦܵܐ: ܓ. ܩܵܠܵܐ: ܘ. ܒܪܘܼܟܵܒ݂ ܡܲܕ݂ܒܚܵܟ݂ ܡܵܪܝܵܐ

ܚܵܘ̈ܒܵܬܲܢ. ܕܲܢܫܲܒܚܵܟ݂ ܘܢܵܘܕܸܐ ܠܲܫܡܵܟ݂. ܡܵܪܝܵܐ ܪ̈ܲܚܡܲܝܟ ܠܥܵܠܲܡ. ܠܪ̈ܲܚܡܹܐ ܫܦܝܼܥܹ̈ܐ ܕܛܲܝܒܘܼܬ݂ܵܟ݂. ܡܲܠܟܵܐ ܡܫܝܼܚܵܐ ܩܵܪܹܝܢܲܢ. ܠܐܝܼܵܠܵܐ ܕܲܡܚܝܼܠܘܼܬܲܢ. ܡܸܛܠ ܙܲܒ݂ܢܹ̈ܐ ܕܲܫܓ݂ܝܼܫܝܼܢ. ܘܥܵܠܡܵܐ ܕܲܕ݂ܠܝܼܚ ܒܲܚܛܵܗܵܘ̈ܗܝ. ܥܢܵܢܹ̈ܐ ܘܲܪ̈ܣܝܼܣܹܐ. ܪܸܡܙܵܟ݂ ܣܵܥܲܪ ܠܟ݂ܠ ܡܸܕܸܡ. ܗܲܒ݂ܠܲܢ ܕܢܵܘܕܸܐ ܠܛܲܝܒܘܼܬ݂ܵܟ݂. ܩܪܹܝ̇ܬ݂ܵܟ݂ ܡܵܪܝܵܐ ܒܟ݂ܠܝܘܿܡ. ܠܵܟ݂ܘܼ ܡܵܪܲܢ ܩܵܪܹܝܢܲܢ. ܘܡܸܢܵܟ݂ ܪ̈ܲܚܡܹܐ ܫܵܐܠܝܼܢܲܢ. ܚܲܣܵܐ ܚܵܘܒܲܝ̈ܢ ܒܛܲܝܒܘܼܬ݂ܵܟ݂. ܕܐܲܢ݇ܬ݁ܘܼ ܐܝܼܬܲܝܟ ܐܝܼܵܠܲܢ. ܘܐܲܢ݇ܬ݁ܘܼ ܣܲܒ݂ܪܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ. ܒܲܚܢܵܢܵܟ݂ ܪܲܚܸܡ ܥܠܲܝܢ. ܘܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ ܘܚܲܝܠܹܗ. ܘܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܬܸܛܲܪ ܐܸܢܘܿܢ. ܦܲܓ݂ܪܵܐ ܘܢܲܦ̮ܫܵܐ ܫܲܘܝܵܐܝܼܬ݂. ܫܵܐܠܝܼܢ ܪ̈ܲܚܡܹܐ ܘܦܘܼܪܩܵܢܵܐ. ܡܢ ܓܲܙܵܐ ܕܒܲܣܝܼܡܘܼܬ݂ܵܟ݂. ܚܲܣܵܐ ܚܵܘܒܹ̈ܐ ܕܲܬ݂ܪ̈ܲܝܗܘܿܢ. ܐܘܿ ܒܵܪܘܿܝܵܐ ܕܲܬ݂ܪ̈ܲܝܗܘܿܢ. ܐܲܝܟ݂ ܕܲܡܥܵܕ݂ܵܐ ܗ݇ܝ ܛܲܝܒܘܼܬ݂ܵܟ݂. ܕܢܸܗܘܘܿܢ ܥܘܼܡܪܵܐ ܠܨܸܒ݂ܝܵܢܵܟ݂. ܢܫܲܒܲܚ ܘܢܸܚܕܸܐ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬܲܢ. ܠܵܟ݂ ܬܸܫܒܘܿܚܬܵܐ ܦܵܪܘܿܩܲܢ. ܕܚܲܣܝܼܬ݁ ܚܵܘܒܲܝ̈ܢ ܘܲܚܛܵܗܲܝ̈ܢ. ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂ ܕܲܠܘܵܬ݂ ܓܸܢܣܲܢ. ܘܲܒ݂ܝܲܕ݂ ܪ̈ܲܚܡܲܝܟ ܣܲܓܝܼ̈ܐܹܐ. ܒܪܵܘܡܵܐ ܕܲܠܥܸܠ ܐܵܘܬܸܒ݂ܬܵܢ. ܡܢ ܝܲܡܝܼܢܵܐ ܕܫܵܠܘܿܚܵܟ݂. ܣܓ݂ܝܼܕܲܬ݁ ܥܲܡ ܐܲܒ݂ܘܼܟ݂ ܘܪܘܼܚܵܟ݂. ܥܲܕܲܪܲܝܢ ܐܲܠܵܗܵܐ ܦܵܪܘܿܩܲܢ. ܫܲܕܲܪ ܡܵܪܲܢ ܥܘܼܕ݂ܪܵܢܵܐ. ܠܟܸܢܫܵܐ ܟܠܹܗ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܡܢ ܒܹܝܬ݂ ܓܲܙܹܗ ܕܲܚܢܵܢܵܟ݂. ܘܢܲܛܲܪ ܐܸܢܘܿܢ ܒܛܲܝܒܘܼܬ݂ܵܟ݂. ܘܲܙܪܘܿܥ ܫܲܝܢܵܟ݂ ܒܲܝܢܵܬ݂ܗܘܿܢ. ܕܢܵܘܕܘܿܢ ܠܲܫܡܵܟ݂ ܩܲܕܝܼܫܵܐ. ܡܫܲܒܚܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ. ܐܲܣܵܢܝ ܡܵܪܝܵܐ ܡܸܛܠ ܕܙܵܥܘ ܓܲܪ̈ܡܲܝ. ܡܫܝܼܚܵܐ ܐܵܣܝܵܐ ܫܲܪܝܼܪܵܐ. ܕܐܲܣܝܼ ܟܹܐܒ݂ܵܘ̈ܗܝ ܕܲܟ݂ܝܵܢܲܢ. ܡܸܛܠ ܚܘܼܒܹܗ ܕܲܠܘܵܬܲܢ. ܥܨܘܿܒ݂ ܫܘܼܡ̈ܵܬܲܢ ܒܲܚܢܵܢܵܟ݂. ܘܚܲܣܵܐ ܚܵܘܒܲܝ̈ܢ ܒܛܲܝܒܘܼܬ݂ܵܟ݂. ܘܣܝܼܡ ܠܲܢ ܡܵܪܝ ܣܲܡܵܐ ܕܪ̈ܲܚܡܲܝܟ. ܕܒܹܗ ܢܸܬ݂ܐܲܣܸܐ ܘܢܵܘܕܸܐ ܠܵܟ݂. ܡܸܛܠ ܕܡܲܒܘܼܥܵܐ ܕܚܲܝܹ̈ܐ ܥܲܡܵܟ݂ ܗ݇ܘ. ܓܲܙܵܐ ܕܛܘܼܒܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܘܲܠܡܹܐܢܵܐ ܕܟ݂ܠ ܥܘܼܕ݂ܪ̈ܵܢܝܼܢ. ܐܝܼܬܲܝܟ ܟܠܫܵܥ ܠܲܟ݂ܝܵܢܲܢ. ܘܠܵܟ݂ܘܼ ܚܵܝܪ̈ܵܢ ܢܲܦ̮̈ܫܵܬܲܢ. ܕܐܲܢ݇ܬ݁ܘܼ ܣܲܒ݂ܪܲܢ ܘܫܘܼܒ݂ܗܵܪܲܢ. ܐܲܣܵܐ ܟܹܐܒܲܝ̈ܢ ܘܟ݂ܘܼܪ̈ܗܵܢܲܝܢ. ܘܚܲܣܵܐ ܘܲܥܛܝܼ ܣܘܼܪ̈ܚܵܢܲܝܢ. ܡܸܛܠ ܕܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܛܵܒ݂ ܐܲܢ݇ܬ݁. ܛܵܒ݂ܵܐ ܕܓܲܒ݂ܠܲܢ ܒܛܲܝܒܘܼܬܹܗ. ܒܪܵܫܝܼܬ݂ ܒܪ̈ܲܚܡܵܘܗܝ ܡܢ ܥܲܦܪܵܐ. ܚܘܼܣ ܥܠܲܝܢ ܡܸܛܠ ܪ̈ܲܚܡܲܝܟ. ܘܢܲܛܲܪ ܥܹܕܬܵܟ݂ ܒܲܚܢܵܢܵܟ݂. ܡܢ ܒܝܼܫܵܐ ܘܚܲܝ̈ܠܵܘܵܬܹܗ. ܕܝܲܠܕܹܝ̈ܗ̇ ܢܲܣܩܘܼܢ ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚ ܩܘܼܕ݂ܫܵܐ. ܠܵܟ݂ ܒܲܠܚܘܿܕܲܝܟ ܚܛܹܝܬ݂. ܕܲܚܛܹܝܬ݂ ܡܵܪܲܢ ܝܵܕܲܥ ܐ݇ܢܵܐ. ܐܲܝܟ݂ ܝܵܪܬܵܐ ܕܦܲܪܲܚ ܢܸܟ݂ܣܵܘ̈ܗܝ. ܘܲܛܥܹ̇ܝܬ݂ ܡܢ ܓܙܵܪܵܐ ܕܥܵܢ̈ܵܟ݂. ܐܲܦܢܵܢܝ ܡܵܪܲܢ ܠܲܛܝܵܪܵܟ݂. ܘܚܲܠܸܛܲܝܢܝ ܥܲܡ ܩܲܕܝܼܫܲܝ̈ܟ. ܘܐܸܚܸܐ ܒܛܲܝܒܘܼ ܐܲܝܟ݂ ܡܵܟ݂ܣܵܐ. ܘܓܲܝܵܣܵܐ ܕܙܲܕܸܩܘ ܪ̈ܲܚܡܲܝܟ. ܥܘܼܕ݂ܪܵܢܲܢ ܒܲܫܡܹܗ ܕܡܵܪܝܵܐ. ܥܘܼܕ݂ܪܵܢܵܐ ܕܪ̈ܲܚܡܲܝܟ ܡܵܪܲܢ. ܐܲܠܘܵܐ ܠܟܸܢܫܲܢ ܒܟ݂ܠܥܸܕܵܢ. ܕܠܲܫܡܵܟ݂ ܪܲܒܵܐ ܩܵܪܹܝܢܲܢ. ܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ. ܐܲܝܟ݂ ܕܐܸܫܬܵܘܕܝܼܬ݁ ܒܲܣܒܲܪܬ݂ܵܟ݂. ܕܟ݂ܠ ܕܩܵܪܹܐ ܠܝܼ ܥܵܢܹܐ ܐ݇ܢܵܐ ܠܹܗ. ܘܲܕ݂ܢܵܩܹܫ ܦܵܬܲܚ ܐ݇ܢܵܐ ܠܹܗ. ܕܐܘܼܪܚܵܐ܀ ܟܲܕ݂ ܩܪܹܝ̇ܬ݂ܵܟ݂ ܥܢܲܝܬܵܢܝ. ܠܐܹܠܝܼܫܲܥ ܕܲܩܪܵܟ݂ ܡܵܪܲܢ. ܡܸܢ ܣܵܢܐܵܘ̈ܗܝ ܒܚܲܝܠܵܟ݂ ܢܲܛܲܪܬ݁. ܘܥܝܼܪܵܐ ܠܥܘܼܕ݂ܪܵܢܹܗ ܫܲܕܲܪܬ݁. ܘܲܚܢܲܢ ܕܒ݂ܐܘܼܪܚܵܐ ܪܵܕܹܝ̇ܢܲܢ. ܡܸܢ ܣܲܩܘܼܒ݂̈ܠܹܐ ܕܵܚ̇ܠܝܼܢܲܢ. ܒܵܟ݂ ܢܸܫܬܲܡܠܘܿܢ ܣܘܼܥܪ̈ܵܢܲܝܢ. ܐܲܟ݂ܡܵܐ ܕܝܵܕ݂ܥܲܬ݁ ܕܥܵܕ݂ܪܵܐ ܠܲܢ. ܕܡܸܛܪܵܐ܀ ܡܸܛܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܐܲܠܵܗܵܐ. ܡܸܛܠ ܡܐܵܐ ܘܥܸܣܪܝܼܢ ܐܲܠܦܝܼ̈ܢ. ܕܲܛܠܵܝܹ̈ܐ ܒܢܲܝ̈ ܢܝܼܢܘܵܝܹ̈ܐ. ܕܲܓܸ݁ܠܬ݁ ܡܵܪܝܵܐ ܠܲܢܒ݂ܝܼܵܟ݂. ܘܟܲܕ݂ ܡܸܠܬ݂ܵܟ݂ ܠܵܐ ܡܸܬ݁ܕܲܓ݁ܠܵܐ. ܪ̈ܲܚܡܲܝܟ ܕܐܸܬ݂ܚܵܘܝܼܘ ܬܲܡܵܢ. ܢܸܬ݂ܚܙܘܿܢ ܒܐܲܬ݂ܪܲܢ ܡܸܣܟܹ݁ܢܵܐ. ܕܲܨܗܸܐ ܠܡܸܛܪܵܐ ܘܲܠܛܲܠܵܐ. ܕܡܲܪܝܲܡ܀ ܕܡܲܪܝܲܡ܀ ܡܸܛܠ ܕܗܘܝܘܼ ܡܵܪܹܟ݂ܝ ܣܓ݂ܘܿܕ݂ܝ ܠܹܗ. ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܩܲܕܝܼܫܬܵܐ. ܐܸܡܹܗ ܕܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ. ܐܸܬ݂ܟܲܫܲܦܝ ܘܲܒ݂ܥܵܝ ܪ̈ܲܚܡܹܐ. ܥܲܠ ܚܲܛܵܝܹ̈ܐ ܕܠܵܐ ܢܹܐܒ݂ܕ݁ܘܼܢ. ܕܒܲܨܠܵܘܵܬܹ̈ܟ݂ܝ ܡܸܬ݂ܓܵܘܣܝܼܢ. ܨܠܘܿܬܹܟ݂ܝ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܲܢ. ܒܗܵܢ ܥܵܠܡܵܐ ܘܲܒ݂ܗ̇ܘ ܕܲܥܬ݂ܝܼܕ݂. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܛܲܪ ܪ̈ܲܚܡܲܝܟ ܠܩܲܪ̈ܝܼܒܲܝܟ. ܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܡܪ̈ܲܥܝܵܢܲܝܟ. ܘܙܲܕܝܼܩܹ̈ܐ ܕܲܫܦܲܪܘ ܩܘܼܕ݂ܡܲܝܟ. ܢܒ݂ܝܼܹ̈ܐ ܫܠܝܼܚܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ. ܘܣܵܗܕܹ̈ܐ ܘܟ݂ܵܗܢܹ̈ܐ ܘܥܵܢܘܵܝܹ̈ܐ. ܢܲܛܲܪ ܠܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܢܲܣܩܘܼܢ ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܚܲܕ݂ܬ݂ܵܐ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚ ܩܘܼܕ݂ܫܵܐ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܣܲܒ݂ܪܵܐ ܕܟ݂ܠܗܘܿܢ ܣܵܘܦܹܝ̈ܗ ܕܐܲܪܥܵܐ. ܨܠܝܼܒ݂ܵܐ ܕܘܼܟ݂ܪܵܢܵܐ ܠܥܵܠܲܡ ܢܸܗܘܸܐ ܠܙܲܕܝܼܩܵܐ. ܢܝܲܩܲܪ ܟܠܲܢ ܒܲܚܦܝܼܛܘܼ. ܠܝܵܘܡܵܐ ܫܒ݂ܝܼܚܵܐ ܕܕ݂ܘܼܟ݂ܪܵܢܹܗ. ܕܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕܝܼܫܵܐ. ܕܲܗܘܵܐ ܡܵܐܢܵܐ ܡܠܸܐ ܪ̈ܲܚܡܹܐ. ܕܚܵܫܲܚ ܠܬܸܫܡܸܫܬܵܐ ܕܡܵܪܹܗ. ܘܟ݂ܠܝܘܿܡ ܢܵܗܲܪ ܠܲܡܦܹܝܕܹܗ. ܨܠܘܿܬܹܗ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܲܢ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܘܲܥܒܸܕ݂ ܥܲܡܝ ܐܵܬ݀ܵܐܛܵܒ݂ܬ݂ܵܐ. ܒܝܵܘܡܵܐ ܪܲܒܵܐ ܕܡܹܐܬ݂ܝܼܬ݂ܵܟ݂. ܝܼܫܘܿܥ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ. ܕܲܥܬ݂ܝܼܕܲܬ݁ ܠܲܡܕ݂ܵܢ ܟܠܵܐ. ܢܓ݂ܘܼܠܘܼܢ ܪ̈ܲܚܡܲܝܟ ܥܲܠ ܚܵܘܒܲܝ̈ܢ. ܒܛܲܝܒܘܼܬ݂ܵܟ݂ ܬܚܲܣܸܐ ܚܛܵܗܲܝ̈ܢ. ܘܐܲܫܘܵܢ ܕܥܲܡܵܟ݂ ܢܸܬ݂ܒܲܣܲܡ. ܒܫܘܼܒ݂ܚܵܐ ܪܲܒܵܐ ܕܡܲܠܟܘܼܬ݂ܵܟ݂. ܢܸܪܘܙܘܼܢ ܘܲܢܫܲܒܚܘܼܢ. ܥܲܢܝܼ̈ܕܹܐ ܕܲܥܡܲܕ݂ܘ ܒܲܫܡܵܟ݂. ܘܣܲܟܝܼܘ ܠܕܸܢܚܹܗ ܕܓܸܠܝܵܢܵܟ݂. ܘܐܸܫܬܵܘܬܲܦܘܒܦܲܓ݂ܪܵܟ݂ ܘܲܕ݂ܡܵܟ݂. ܘܲܡܫܝܼܚܵܐ ܒܩܘܼܫܬܵܐ ܝܲܕ݂ܥܘܼܟ݂. ܚܲܣܵܐ ܐܸܢܘܿܢ ܒܲܚܢܵܢܵܟ݂. ܘܲܪܫܘܿܡ ܐܸܢܘܿܢ ܒܓ݂ܵܘ ܣܸܦܪܵܟ݂. ܘܐܲܫܘܵܐ ܐܸܢܘܿܢ ܠܲܓ݂ܢܘܿܢܵܟ݂܀
ܕܥܲܢܝܼܕܹ̈ܐ. ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܕܦܘܼܪܩܵܢܝ. ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܡܵܪܹܗ ܕܡܵܘܬܵܐ ܘܡܲܦܩܵܢܵܐ. ܪܘܼܚܵܟ݂ ܡܒܲܣܲܡܬܵܐ ܬܕܲܒ݁ܪܲܢܝ ܒܐܘܼܪܚܵܟ݂ ܕܚܲܝܹ̈ܐ. ܡܵܐ ܕܐܵܬܹܐ ܐܲܢ݇ܬ݁ ܒܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܐܹܒܲܪܸܟ݂ ܠܡܵܪܝܵܐ ܒܟ݂ܠܙܒܲܢ. ܒܪܝܼܟ݂ ܗ݇ܘ ܡܫܝܼܚܵܐ ܕܐܸܬ݁ܬܲܚܬܝܼ ܘܲܛܥܸܡ ܡܵܘܬܵܐ. ܘܝܲܗ݇ܒ݂ ܠܲܢ ܦܲܓ݂ܪܹܗ ܘܲܕ݂ܡܹܗ. ܘܐܲܚܝܼ ܠܐܵܕ݂ܵܡ ܘܲܠܝܲܠܕ݂ܵܘ̈ܗܝ ܦܪܲܩ ܒܲܨܠܝܼܒܹܗ.
ܫܲܒܲܚ. ܒܥܲܕ݂ܠܵܐ ܬܲܝܬܹ݁ܝܢ. ܠܵܐ ܡܨܸܐ ܪܸܥܝܵܢܵܐ ܡܲܕ݂ܪܸܟ݂. ܐܵܦܠܵܐ ܦܘܼܡܵܐ ܠܲܡܬܲܢܵܝܘܼ. ܥܲܠ ܥܘܼܫܢܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂. ܫܡܲܝܵܐ ܓܹܝܪ ܘܐܲܪܥܵܐ ܘܢܲܗܝܼܪܹ̈ܐ ܫܸܡܫܵܐ ܘܣܲܗܪܵܐ ܐܲܢ݇ܬ݁ ܒܪܲܝܬ݁ ܐܸܢܘܿܢ. ܢܸܣܬܲܟ݁ܠܘܿܢ ܗܵܟܹܝܠ ܐܲܝܠܹܝܢ ܕܣܵܓ݂ܕܝܼܢ ܠܒܸܪ̈ܝܵܬ݂ܵܐ. ܕܲܟ݂ܡܵܐ ܡܪܲܡܪܲܡ ܐܲܢ݇ܬ݁ ܒܚܲܝܠܵܐ ܘܫܘܼܠܛܵܢܵܐ. ܕܟ݂ܠ ܡܸܕܸ݁ܡ ܒܐܝܼܕܲܝ̈ܟ ܐܸܬ݁ܬܲܩܲܢ. ܒܝܲܕ݂ ܣܘܿܓ݂ܵܐܐ ܕܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂. ܒܵܥܹ̇ܝܢܲܢ ܢܲܛܲܪ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
ܥܵܠܲܡ. ܒܪܸܡܙܹܗ ܥܒܲܕ݂ ܒܪܝܼܬ݂ܵܐ ܟܠܵܗ̇ ܐܝܼܬ݂ܝܵܐ ܕܲܩܕ݂ܵܡ ܟܠ. ܘܠܲܚܡܵܗ̇ ܒܣܘܿܓ݂ܵܐܐ ܕܛܲܝܒ݁ܘܼܬܹܗ ܕܗܝ ܒܵܗ̇ ܬܹܐܠܲܦ ܥܠܵܘܗܝ. ܡܸܢܵܗ̇ ܥܲܒ݂ܕܵܗ̇ ܡܸܬ݂ܚܲܙܝܵܢܝܼܬ݂ܵܐ. ܡܢܵܬ݂ܵܐ ܐ݇ܚܪܹܬܵܐ ܡܸܬ݂ܝܲܕ݂ܥܵܢܝܼܬ݂ܵܐ. ܘܐܲܣܪܵܗ̇ ܠܟ݂ܠܵܗ̇ ܒܒܲܪܢܵܫܵܐ ܕܬ݂ܵܘܕܸ݁ܐ ܕܚܲܕ݂ܘܼ ܒܪܵܗ̇. ܕܲܒ݂ܝܲܕ݂ ܚܘܼܒܹ݁ܗ ܢܵܘܕܲܥ ܥܲܠ ܝܵܬܹܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ: ܚܘܼܣ ܥܠܲܝܢ ܐܲܢ݇ܬ݁ ܚܲܢܵܢܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܸܬ݂ܦܲܢ ܥܠܲܝܢ ܡܠܸܐ ܪ̈ܲܚܡܹܐ: ܘܠܵܐ ܡܵܪܝ ܬܲܗܡܸܐ ܚܝܵܪܵܟ݂ ܘܲܒ݂ܛܝܼܠܘܼܬ݂ܵܟ݂ ܡܸܢܲܢ: ܕܲܥܠܲܝܟ݁ܘܼ ܣܲܒ݂ܪܲܢ ܘܬ݂ܘܼܟ݂ܠܵܢܲܢ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ: ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
 ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܩܵܢܘܿܢܵܐ. * ܫܡܲܝܵܐ ܡܸܫܬܲܥܹܝܢ ܫܘܼܒ݂ܚܹܗ ܕܐܲܠܵܗܵܐ. * ܘܲܥܒܵܕ݂ ܐܝܼܕ݂ܵܘ̈ܗܝ ܡܚܲܘܸܐ ܪܩܝܼܥܵܐ. ܒܟ݂ܠ ܦܘܼܡ̈ܝܼܢ ܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܦܵܪܘܿܩܲܢ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܕܐܵܘܪܸܒ݂ܬ݁ ܠܥܲܒܹܕܲܝ̈ܟ ܘܢܲܨܲܚܬ݁ ܐܸܢܘܿܢ ܒܐܲܪ̈ܒܲܥܦܸܢܝܵܢ. ܢܲܛܲܪ ܠܥܸܕ݁ܬܵܟ݂ ܘܲܠܝܲܠܕܹܝ̈ܗ̇ ܒܣܘܿܓ݂ܵܐܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ * ܝܵܘܡܵܐ ܠܝܵܘܡܵܐ ܡܲܒܲܥ ܡܹܐܡܪܵܐ. * ܘܠܸܠܝܵܐ ܠܠܸܠܝܵܐ ܡܚܵܘܸܐ ܡܲܕ݁ܥܵܐ.

܀

 ܠܲܝܬ݁ ܡܹܐܡܪܵܐ ܐܵܦܠܵܐ ܡܸܠܹ̈ܐ. * ܕܠܵܐ ܢܸܫܬ݁ܡܲܥ ܒܩܵܠܗܘܿܢ.

܀

 ܒܟ݂ܠܵܗ̇ ܐܲܪܥܵܐ ܢܸܦܩܲܬ݀ ܣܒܲܪܬ݂ܗܘܿܢ. * ܘܲܒ݂ܣܵܘ̈ܦܹܝܗ̇ ܕܬܹܒܹܝܠ ܡܸܠܲܝ̈ܗܘܿܢ.

܀

 ܥܲܠ ܫܸܡܫܵܐ ܢܩܲܫ ܡܲܫܟ݁ܢܹܗ ܒܗܘܿܢ. * ܘܗܘ ܐܲܝܟ݂ ܚܲܬ݂ܢܵܐ ܕܢܵܦܹܩ ܡܢ ܒܹܝܬ݂ ܓܢܘܿܢܵܐ.

܀

 ܢܕ݂ܘܼܨ ܐܲܝܟ݂ ܓܲܢ݇ܒܵܪܵܐ ܠܡܸܪܗܲܛ ܐܘܼܪܚܹܗ. * ܡܢ ܣܵܘ̈ܦܲܝ ܫܡܲܝܵܐ ܡܲܦܩܹܗ.

܀

 ܘܡܲܣܡܟܹܗ ܥܲܠ ܣܵܘ̈ܦܲܝ ܫܡܲܝܵܐ. * ܘܠܲܝܬ݁ ܕܛܵܫܹܐ ܡܢ ܠܲܗܓܹ݁ܗ.

܀

 ܢܵܡܘܿܣܹܗ ܕܡܵܪܝܵܐ ܕܠܵܐ ܡܘܼܡ ܘܡܲܦܢܸܐ ܢܲܦ̮ܫܵܐ. * ܣܵܗܕܘܼܬܹܗ ܕܡܵܪܝܵܐ ܡܗܲܝܡܢܵܐ ܘܲܡܚܲܟ݁ܡܵܐ ܝܲܠܘܼܕܹ̈ܐ.

܀

 ܦܘܼܩܕܵܢܵܘ̈ܗܝ ܕܡܵܪܝܵܐ ܬܪܝܼܨܝܼܢ ܘܲܡܚܲܕܹܝܢ ܠܸܒܵܐ. * ܦܘܼܩܕܵܢܹܗ ܕܡܵܪܝܵܐ ܓܒܸܐ ܘܡܲܢܗܲܪ ܥܲܝܢܹ̈ܐ.

܀

܀

ܒܟ݂ܠ ܦܘܼܡ̈ܝܼܢ ܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܦܵܪܘܿܩܲܢ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܕܐܵܘܪܸܒ݂ܬ݁ ܠܥܲܒܹܕܲܝ̈ܟ ܘܢܲܨܲܚܬ݁ ܐܸܢܘܿܢ ܒܐܲܪ̈ܒܲܥܦܸܢܝܵܢ. ܢܲܛܲܪ ܠܥܸܕ݁ܬܵܟ݂ ܘܲܠܝܲܠܕܹܝ̈ܗ̇ ܒܣܘܿܓ݂ܵܐܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀
ܫܲܒܲܚ: ܥܵܠܲܡ. ܒܲܨܠܘܿܬ݂ܗܘܿܢ ܕܩܲܕ݁ܝܼܫܲܝ̈ܟ ܦܵܪܘܿܩܲܢ ܢܲܛܲܪ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܘܲܬ݂ܚܹܝܬ݂ ܟܸܢܦܹ̈ܐ ܕܲܒ݂ܛܝܼܠܘܼܬ݂ܵܟ݂ ܡܵܪܝ ܣܲܬܲܪ ܠܲܢ. ܡܢ ܢܸܟ݂ܝܵܢܵܘ̈ܗܝ ܕܡܵܪܘܿܕ݂ܵܐ ܣܵܢܹ̇ܐܗ ܕܲܟ݂ܝܵܢܲܢ܀
ܫܡܲܝܵܐ ܡܸܫܬܲܥܹܝܢ ܫܘܼܒ݂ܚܹܗ ܕܐܲܠܵܗܵܐ. * ܘܲܥܒܵܕ݂ ܐܝܼܕ݂ܵܘ̈ܗܝ ܡܚܲܘܸܐ ܪܩܝܼܥܵܐ. ܀ ܒܵܪܘܿܝܵܐ ܕܲܒ݂ܪܸܡܙܹܗ ܒܪܵܐ ܥܵܠܡܵܐ ܘܨܲܒ݁ܬܹܗ ܒܟ݂ܠ ܫܘܼܦܪ̈ܝܼܢ. ܘܟܲܢܸܫ ܘܐܲܣܪܹܗ ܠܬ݂ܘܼܩܵܢ ܥܲܡܠܹܗ ܒܲܨܠܸܡ ܐ݇ܢܵܫܵܐ. ܐܲܫܘܵܐ ܠܟ݂ܠܲܢ ܕܲܢܫܲܒ݁ܚܵܟ݂ ܘܢܵܘܕܸ݁ܐ ܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂.

ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܒܩܲܕ݂ܡܵܝܹ̈ܐ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܕܝܵܘܡܵܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ : ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ݂܀ ܗܵܘܿ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ݁ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡܠܲܟ݂ ܛܵܒ݂ܵܬ݂ܵܐ ܠܪܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕ݁ܬܹܗ: ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܠܵܐ ܨܵܒܹܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ:  ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܼܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܗܘܝܘ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܪ̈ܝܵܬ݂ܵܐ:  ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ ؛ ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܝܼܣ:  ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܝܼܣ : ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈  ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܵܘܿ ܕܲܒ݂ܪܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚܘ ܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂:  ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.

ܨ̈ܠܵܘܵܬ݂ܵܐ ܕܩܵܠܹ̈ܐ ܫܲܗܪܵܐ
ܨܠܘܿܬ݂ܵܐܕܩܲܕ݂ܡܵܝܵܐ. ܢܸܫܦܪܘܼܢ ܠܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܩܵܠܹ̈ܐ ܕܗܘܼܠܵܠܲܝ̈ܢ ܘܢܸܥ̈ܡܵܬ݂ܵܐ ܕܲܙܡܝܼܪ̈ܵܬܲܢ: ܘܩܲܒܸ݁ܠܡܢܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܦܹܐܪܹ̈ܐ ܡܠܝܼܠܹ̈ܐ ܕܣܸܦ̈ܘܵܬܲܢ: ܕܲܒ݂ܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܩܲܪܒ݂ܝܼܢܲܢ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܡܲܩܦܝܼܢ ܘܲܡܫܲܪܹܝܢ ܡܲܪܡܝܼܬ݂ܵܐ ܩܲܕ݂ܡܵܝܬܵܐ ܕܗܘܼܠܵܠܵܐ ܐܲܝܢܵܐ ܕܗܘܟܠܵܗ̇ ܒܩܵܠܵܗ̇: ܘܲܡܫܲܒ݁ܚܝܼܢ ܘܲܡܥܵܠܡܝܼܢ ܘܐܵܡܪܝܼܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.

	ܨܠܘܿܬ݂ܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܵܐ: ܫܘܼܒ݂ܚܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ: ܘܗܘܼܠܵܠܹ̈ܐ ܕܠܵܐ ܫܲܠܘܵܐ: ܘܬܸܫܒ̈ܚܵܬ݂ܵܐ ܕܠܵܐ ܣܵܟ݂ܵܐ: ܘܩܵܠܹ̈ܐ ܕܲܡܠܹܝܢ ܬܵܘܕ݁ܝܼܬ݂ܵܐ: ܚܲܝܵܒ݂ ܟܠܹܗ ܟܝܵܢܵܐ ܕܲܡܠܝܼ̈ܠܹܐ ܕܲܒ݂ܪܲܝܬ݁: ܠܡܲܣܵܩܘܼ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܲܡܫܲܡܫܝܼܢ ܟܠܵܗ̇ ܡܲܪܡܝܼܬ݂ܵܐ ܬܠܝܼܬ݂ܵܝܬܵܐ ܕܗܘܼܠܵܠܵܐ ܐܲܝܢܵܐ ܕܗܘ ܒܩܵܠܵܗ̇. ܐܲܝܟ݂ ܩܲܕ݂ܡܵܝܬܵܐ.
ܘܐܵܡܪܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܩܲܕܡܵܝܹ̈ܐ ܡܲܪܡܝܼܬ݂ܵܐ. ܘܐܵܦ ܐܸܢܵܐ ܐܲܛܸܢ ܐܸܢܘܿܢ ܒܠܵܐ ܥܲܡܝ. ܡܙܡܘܪܐ ܩܢܕ
 ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ. ܡܢ ܦܘܼܡܹ̈ܐ ܡܵܪܝ ܡܠܝܼܠܹ̈ܐ ܕܲܒ݂ܪܲܝܬ݁: ܘܡܢ ܠܸܫܵܢܹ̈ܐ ܡܫܲܒ݁ܚܵܢܹ̈ܐ ܕܪܲܟܸ݁ܒ݂ܬ݁: ܘܡܢ ܟܸܢܫܹ̈ܐ ܟܠܗܘܿܢ ܕܥܸܠܵܝܹ̈ܐ ܘܲܕ݂ܬܲܚܬܵܝܹ̈ܐ: ܫܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܲܕ݂ܡܵܪܘܼܬ݂ܵܟ݂ ܢܸܣܬ݁ܓܸܕ݂ ܘܢܸܫܬܲܒܲܚ ܘܢܸܬ݂ܝܲܩܲܪ ܘܢܸܬ݂ܪܲܡܪܲܡ ܘܢܸܬ݁ܬܵܘܕܸ݁ܐ ܘܢܸܬ݂ܒܲܪܲܟ݂ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܪܝܼܢ ܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ ܘܬܵܢܹܝܢ ܘܲܡܫܲܒܚܝܼܢ ܒܵܗ̇ ܘܡܲܩܦܝܼܢ:
ܕܠܸܠܝܵܐ. ܫܘܼܚܠܵܦܵܐ. ܒܐ݇ܪܵܙܵܐ ܪܲܒܵܐ. ܐܸܪܡܪܡܵܟ݂ ܡܵܪܝ ܡܲܠܟܵܐ:ܕܢܲܚ ܢܘܼܗܪܵܐ ܒܚܸܫܘܿܟ݂ܵܐ ܠܲܬ݂ܪ̈ܝܼܨܹܐ: ܐܲܢܗܲܪܘ ܒܲܪ̈ܩܵܘܗܝ ܠܬܸܒܹܝܠ. ܕܨܘܼܪܬܵܐ. ܘܟ݂ܠܹܗ ܬܚܹܝܬ݂ ܫܡܲܝܵܐ ܡܲܢܗܲܪ. ܡܲܠܟܵܐ ܡܫܝܼܚܵܐ ܕܲܕ݂ܢܲܚ ܢܘܼܗܪܹܗ ܒܐܲܪ̈ܒܲܥܦܸܢܝܵܢ. ܐܲܢ݇ܬ݁ ܒܪ̈ܲܚܡܲܝܟ ܣܲܓ݁ܝܼ̈ܐܹܐ ܚܘܼܣ ܥܲܠ ܥܸܕ݁ܬܵܟ݂. ܕܗܵܐ ܒܵܥܹ̇ܐ ܐܵܟܸܠܩܲܪܨܵܐ ܕܢܵܘܒܸ݁ܕ݂ ܝܲܠܕܹܝ̈ܗ̇. ܫܠܝܼܚܹ̈ܐ ܕܲܙܪܲܥܘ ܙܲܪܥܵܐ ܛܵܒ݂ܵܐ. ܒܓ݂ܵܘ ܢܲܦ̮ܫ̈ܵܬ݂ܵܐ ܕܲܒ݂ܢܲܝ̈ܢܵܫܵܐ. ܗܸܢܘܿܢ ܢܸܗܘܘܿܢ ܣܢܹܐܓ݂ܪܹ̈ܐ ܚܠܵܦܲܝܢ. ܘܲܢܦܝܼܣܘܼܢܵܟ݂ ܡܸܛܠ ܥܲܒ݂ܕܲܝ̈ܟ. ܡܵܪܵܐ ܛܵܒ݂ܵܐ ܪܲܚܸܡܥܠܲܝܢ: ܬܢܝܼ.
ܘܫܲܒܲܚ. ܡܲܠܟܵܐ ܡܫܝܼܚܵܐ ܕܲܕ݂ܢܲܚ ܢܘܼܗܪܹܗ ܒܐܲܪ̈ܒܲܥܦܸܢܝܵܢ. ܐܲܢ݇ܬ݁ ܒܪ̈ܲܚܡܲܝܟ ܣܲܓ݁ܝܼ̈ܐܹܐ ܚܘܼܣ ܥܲܠ ܥܸܕ݁ܬܵܟ݂. ܕܗܵܐ ܒܵܥܹ̇ܐ ܐܵܟܸܠܩܲܪܨܵܐ ܕܢܵܘܒܸ݁ܕ݂ ܝܲܠܕܹܝ̈ܗ̇. ܫܠܝܼܚܹ̈ܐ ܕܲܙܪܲܥܘ ܙܲܪܥܵܐ ܛܵܒ݂ܵܐ. ܒܓ݂ܵܘ ܢܲܦ̮ܫ̈ܵܬ݂ܵܐ ܕܲܒ݂ܢܲܝ̈ܢܵܫܵܐ. ܗܸܢܘܿܢ ܢܸܗܘܘܿܢ ܣܢܹܐܓ݂ܪܹ̈ܐ ܚܠܵܦܲܝܢ. ܘܲܢܦܝܼܣܘܼܢܵܟ݂ ܡܸܛܠ ܥܲܒ݂ܕܲܝ̈ܟ. ܡܵܪܵܐ ܛܵܒ݂ܵܐ ܪܲܚܸܡܥܠܲܝܢ.
ܥܵܠܲܡ. ܡܫܝܼܚܵܐ ܠܵܐ ܬܲܗܡܸܐ ܡܸܢܲܢ. ܘܠܵܐ ܬܲܪܚܸܩ ܡܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܒ݂ܵܟ݂ܘܼ ܡܵܪܝ ܐܸܬ݂ܓܵܘܲܣܢܲܢ. ܕܲܒܲܪܲܝܢ ܒܐܘܼܪܚܵܟ݂ ܕܚܲܝܹ̈ܐ ܕܲܢܙܲܡܲܪ ܠܵܟ݂ ܟܠܲܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ. ܢܹܐܡܲܪ. ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܐܸܡܹܗ ܕܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ: ܐܲܦܝܼܣܝ ܥܲܡܲܢ ܠܲܡܫܝܼܚܵܐ ܕܢܲܫܪܸܐ ܫܲܝܢܹܗ ܒܲܝܢܵܬܲܢ: ܘܲܢܢܲܛܲܪ ܠܲܢ ܡܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܒܵܚܵܐ. ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܢ ܟܠ ܠܸܫܵܢܝܼ̈ܢ: ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܘܗܸܕ݂ܪܵܐ ܘܪܘܼܡܪܵܡܵܐ ܡܢ ܟܠ ܒܸܪ̈ܝܵܢ: ܐܝܼܬ݂ܝܵܐ ܓܢܝܼܙܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܫܲܒܲܚ: ܘܥܵܠܲܡ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ. ܘܥܵܛ̇ܦܝܼܢ ܠܪܹܫܹܗ ܕܡܲܙܡܘܿܪܵܐ ܬܵܘ ܢܫܲܒܲܚ ܠܡܵܪܝܵܐ. * ܘܢܸܙܡܲܪ ܠܐܲܠܵܗܲܢ ܦܵܪܘܿܩܵܐ ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ..
ܘܐܵܡ̇ܪܝܼܢ ܗܦܵܟ݂ܬܵܐ. ܣܝܼܡܲܬ݂ ܚܲܝܹ̈ܐ ܒܲܟ݂ܬ݂ܵܒܹ̈ܐ: ܘܠܵܐ ܪܓ݂ܝܼܫܝܼܢ ܒܵܗ̇ ܡܵܝ̇ܘܿܬܹ̈ܐ. ܘܚܲܟ݁ܝܼܡܹ̈ܐ ܒܙܵܘܥܲܝ̈ ܢܲܦ̮ܫܵܐ: ܣܵܥ̇ܪܝܼܢ ܓܲܙܹ̈ܐ ܪ̈ܘܼܚܵܢܹܐ

܀

 ܢܸܒ݂ܥܵܐ ܚܲܠܝܵܐ ܒܢܵܡܘܿܣܹ̈ܐ: ܘܠܵܐ ܡܸܬ݂ܒܲܝܢܝܼܢ ܥܲܦܪ̈ܵܢܹܐ. ܘܝ̇ܵܕ݂ܘܿܥܹ̈ܐ ܒܪܸ̈ܓ݂ܫܲܝ ܗܵܘܢܵܐ: ܝܵܢ̇ܩܝܼܢ ܫܸܦܥܵܐ ܕܚܲܠܝܘܼܬܹܗ.
ܬܸܫܒܘܿܚܬܵܐ. ܘܲܒ݂ܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܕܹܝܢ ܐܵܡ̇ܪܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܗܵܕܹܐ ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܡܵܪܝ ܢܲܪܣܲܝ ܡܲܠܦܵܢܵܐ.
ܬܵܘܕ݁ܝܼ ܠܛܵܒ݂ܵܐ ܕܚܲܪܲܪ ܓܸܢܣܲܢ: ܡܢ ܥܲܒ݂ܕ݁ܘܼܬ݂ܵܐ ܕܒ݂ܝܼܫܵܐ ܘܡܵܘܬܵܐ܀ ܫܲܝܸܢ ܥܲܡܲܢ ܟܸܢܫܲܝ̈ ܪܵܘܡܵܐ: ܕܪܲܓ݁ܝܼܙܝܼܢ ܗ݇ܘܵܘ ܡܸܛܠ ܥܵܘܠܲܢ܀ ܒܪܝܼܟ݂ ܚܲܢܵܢܵܐ ܕܟܲܕ݂ ܠܵܐ ܒܥܲܝܢܵܝܗܝ: ܢܦܲܩ ܒܲܒ݂ܥܵܬܲܢ ܘܲܚܕ݂ܝܼ ܒܚܲܝܲܝ̈ܢ܀ ܘܨܵܪ ܕܸܡ̈ܘܵܬ݂ܵܐ ܕܐܲܒ݁ܝܼܕ݂ܘܼܬܲܢ: ܘܲܕ݂ܦܘܼܢܵܝܲܢ ܒܥܸܪܒܵܐ ܕܲܛܥܵܐ܀ ܝܵܪܬܵܐ ܘܲܒ݂ܪܵܐ ܩܪܵܝܗܝ ܠܲܟ݂ܝܵܢܲܢ: ܕܲܛܥܵܐ ܘܲܦܢܵܐ ܘܡܝܼܬ݂ ܘܐܸܬ݂ܢܲܚܲܡ܀ ܘܚܲܕ݁ܝܼ ܠܟܸܢܫܹ̈ܐ ܪ̈ܘܼܚܵܢܵܝܹܐ: ܒܲܬ݂ܝܵܒ݂ܘܼܬܲܢ ܘܲܒ݂ܢܘܼܚܵܡܲܢ܀ ܠܵܐ ܡܸܬ݂ܡܲܠܲܠ ܚܘܼܒܵܐ ܪܲܒܵܐ: ܕܚܵܘܝܼ ܨܹܐܕܲܝܢ ܪܵܚܹ̇ܡ ܓܸܢܣܲܢ܀ ܕܡܸܢܹܗ ܕܓܸܢܣܲܢ ܥܒܲܕ݂ ܡܸܨܥܵܝܵܐ: ܘܪܲܥܝܼ ܠܥܵܠܡܵܐ ܥܲܡ ܪܲܒܘܼܬܹܗ܀ ܪܲܒܵܐ ܗ݇ܝ ܡܸܢܲܢ ܘܡܢ ܟܠ ܒܸܪ̈ܝܵܢ: ܚܕܲܬܵܐ ܕܲܣܥܲܪ ܨܹܝܕ݂ ܐ݇ܢܵܫܘܼܬܲܢ܀ ܕܥܲܒ݂ܕܹ݁ܗ ܠܦܲܓ݂ܪܲܢ ܗܲܝܟܲܠ ܩܘܼܕ݂ܫܵܐ: ܕܲܢܡܲܠܸܐ ܒܹܗ ܣܸܓ݂ܕܲܬ݂ ܟܠܵܐ܀ ܬܵܘ ܐܲܪ̈ܥܵܢܹܐ ܘܲܫܡܲܝܵܢܹ̈ܐ: ܬܗܲܪܘ ܘܐܸܬ݁ܕܲܡܲܪܘ ܒܪܲܒ݁ܘܼܬ݂ ܕܲܪܓ݂ܵܐ܀ ܕܲܡܛܵܐ ܓܸܢܣܲܢ ܠܪܵܘܡܵܐ ܪܲܒܵܐ: ܕܐܲܠܵܗܘܼܬ݂ܵܐ ܕܠܵܐ ܡܸܬ݁ܕܲܪܟܵܐ܀ ܫܡܲܝܵܐ ܘܐܲܪܥܵܐ ܘܟ݂ܠܡܵܐ ܕܲܒ݂ܗܘܿܢ: ܢܵܘܕ݁ܘܿܢ ܥܲܡܲܢ ܠܡܵܘܪܸܒ݂ ܓܸܢܣܲܢ܀ ܕܚܲܕܸ݁ܬ݂ ܨܲܠܡܲܢ ܘܲܥܛܵܐ ܥܵܘܠܲܢ: ܘܲܩܪܵܢ ܒܲܫܡܹܗ ܘܫܲܥܒܸ݁ܕ݂ ܠܲܢ ܟܠ܀ ܫܵܘܹ̇ܐ ܠܫܘܼܒ݂ܚܵܐ ܡܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܗ̇ܘ ܕܐܲܪܝܼܡܲܢ ܠܥܸܠ ܡܢ ܟܠܵܐ܀ ܘܢܹܐܡܲܪ ܟܠܲܢ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ: ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀
ܘܩܵܝܡ̇ܝܼܢ: ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܐܲܝܟ݂ ܕܲܒ݂ܚܘܼܕ݂ܪܵܐ ܐܵܘ ܒܓܲܙܵܐ. ܐܵܘ ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ. ܕܠܵܐ ܚܲܒ݂ܪ̈ܵܬ݂ܵܗ̇.
(ܘܪܵܡܹܐ ܡܫܲܡܫܵܢܵܐ ܐܘܼܪܵܪܵܐ: ܘܐܵܣܲܪ ܙܘܼܢܵܪܵܐ: ܘܩܵܐܹ̇ܡ ܩܕ݂ܵܡ ܫܟ݂ܝܼܢܬܵܐ ܡܛܲܟ݁ܣܵܐܝܼܬ݂) ܘܐܵܡܲܪ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ: ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ ܘܐܲܠܵܗܵܐ ܕܟ݂ܠܹܗ ܒܘܼܝܵܐܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܦܵܪܘܿܩܲܢ ܘܝܲܨܘܼܦܬ݂ܵܢܲܢ ܘܲܡܦܲܪܢܣܵܢܵܐ ܕܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ ܘܩܘܼܝܵܡܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ ܘܲܕ݂ܟ݂ܠܗܹܝܢ ܥܹ̈ܕ݂ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܐܲܬ݂ܪܲܢ ܘܥܲܠ ܟܠ ܐܲܬ݂ܪ̈ܵܘܵܢ: ܘܥܲܠ ܐܲܝܠܹܝܢ ܕܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܥܵܡ̇ܪܝܼܢ ܒܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܘܼܙܵܓ݂ܵܘ̈ܗܝ ܕܐܵܐܲܪ ܘܟܲܗܝܼܢܘܼܬ݂ܵܐ ܕܫܲܢ݇ܬܵܐ ܘܥܲܠ̈ܠܵܬ݂ܵܐ ܕܦܹܐܪܹ̈ܐ ܘܥܲܠ ܬܘܼܩܵܢܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ1: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܲܠܟܹ̈ܐ ܐܲܚܝܼܕܲܝ̈ ܫܘܼܠܛܵܢܹܗ ܕܥܵܠܡܵܐ ܗܵܢܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܩܲܫܝܼܫܹ̈ܐ ܘܲܡܫܲܡܫܵܢܹ̈ܐ ܐܲܪ̈ܬܵܕ݁ܘܿܟ݁ܣܹܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ܗ̇ܘ ܕܥܲܬ݁ܝܼܪ ܒܪ̈ܲܚܡܵܘܗܝ ܘܲܫܦܝܼܥ ܒܲܚܢܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܡܸܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ ܐܝܼܬ݂ܵܘܗܝ ܘܲܠܕ݂ܵܪܕܵܪ̈ܝܼܢ ܩܲܝܵܡ ܫܘܼܠܛܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܟ݂ܝܵܢܹܗ ܐܝܼܬ݂ܵܘܗܝ ܛܵܒ݂ܵܐ ܘܝܵܗܘܿܒ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹܐ ܒܡܵܘܬܹ݁ܗ ܕܚܲܛܵܝܵܐ ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܸܢ ܥܵܘܠܹܗ ܘܢܸܚܸܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܲܒ݂ܝܲܠܕܹ݁ܗ ܐܵܘ ܕܲܒ݂ܕܸܢܚܹܗ ܐܵܘ ܕܲܒ݂ܨܵܘܡܹܗ ܐܵܘ ܕܲܒ݂ܡܲܥܲܠܬܹ݁ܗ ܐܵܘ ܕܒܲܩܝܵܡܬܹ݁ܗ ܐܵܘ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܐܵܘ ܕܲܒ݂ܡܲܚܲܬ݁ܬܹ݁ܗ ܐܵܘ ܕܒܲܨܠܝܼܒܹܗ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܐܲܦܨܲܚ ܠܲܫܡܲܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܹܗ ܗܘ ܟܝܵܢܵܐܝܼܬ݂ ܐܝܼܬ݂ ܠܵܐ ܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܘܝܲܬ݁ܝܼܪܵܐܝܼܬ݂ ܕܐܲܝܠܹܝܢ ܕܲܡܗܲܝܡܢܝܼܢ ܒܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܣܓܵܐ ܥܲܡܲܢ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
