ܛܲܟܼܣܵܐ ܕܬܸܫܡܸܫܬܵܐ
ܕܠܸܠܝܵܐ ܘܕܨܲܦܪܵܐ
 ܒܫܸܡ ܐܲܒܼܵܐ ܘܲܒܼܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕܫܵܐ ܠܥܵܠܡܝܼܢ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒܿܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܖ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒܿܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟܼ: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟܼ: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬܼܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟܼ: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟܼ: ܠܡܲܣܵܩܘܼ ܠܵܟܼ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬܼܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓܼܕ݁ܬ݂ܵܐ ܒܟܼܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟܼܠ.
ܘܥܵܢܹܝ̇ܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.

ܨܠܘܿܬ݂ܵܐ. ܚܲܝܸܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܠܲܡܚܝܼܠܘܼܬܲܢ ܒܲܚܢܵܢܵܟ݂: ܘܠܲܒܸ݁ܒ݂ ܘܥܲܕܲܪ ܠܲܙܥܘܿܪܘܼܬ݂ ܢܲܦ̮ܫܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܥܝܼܪ ܕܲܡܟ݁ܘܼܬ݂ ܪܸܥܝܵܢܲܢ: ܘܐܲܩܸܠ ܝܘܼܩܪܵܐ ܕܗܲܕܵܡܲܝ̈ܢ: ܘܐܲܫܝܼܓ݂ ܘܲܡܪܘܿܩ ܨܵܐܬ݂ܵܐ ܕܚܵܘܒܲܝ̈ܢ ܘܕܲܚܛܵܗܲܝ̈ܢ: ܘܐܲܢܗܲܪ ܠܚܸܫܘܿܟ݂ܘܼܬܹܗ ܕܡܲܕ݁ܥܲܢ: ܘܐܵܘܫܸܛ ܘܫܲܪܲܪ ܘܗܲܒ݂ ܒܲܢ ܚܲܝܠܵܐ ܘܐܝܼܕ݂ܵܐ ܕܥܘܼܕ݂ܖ̈ܵܢܹܐ: ܕܒ݂ܵܗ̇ ܢܩܘܼܡ ܘܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܨܠܘܿܬ݂ܵܐ. ܚܲܝܠܵܐ ܡܵܪܝ ܟܲܣܝܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܥܘܼܕ݂ܪܵܢܵܐ ܬܡܝܼܗܵܐ ܕܡܵܪܘܼܬ݂ܵܟ݂: ܘܣܘܼܝܵܥܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂: ܗܘ̤ ܢܚܲܝܸܠ ܠܲܡܚܝܼܠܘܼܬܹܗ ܕܲܟ݂ܝܵܢܲܢ ܚܲܠܵܫܵܐ: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܘܥܵܢܹܝ̇ܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܡܵܪܝܵܐ ܪܲܚܸܡ ܥܠܲܝܢ.

	ܨܠܘܿܬ݂ܵܐ. ܠܲܟ݂ܝܵܢ ܟܲܣܝܘܼܬ݂ܵܟ݂ ܕܠܵܐ ܡܸܬ݁ܕ݁ܪܸܟ݂ ܘܠܵܐ ܡܸܣܬܲܝܲܟ݂ ܡ̣ܢ ܚܘܼܫܵܒ݂ܵܐ ܘܣܘܼܟܵܠܵܐ ܕܟ݂ܠܗܘܿܢ ܥܒ݂ܝܼܕܹ̈ܐ: ܝܵܐܹ̇ܐ ܡܵܪܝ ܫܘܼܒ݂ܚܵܐ ܘܙܵܕ݂ܩ̇ܝܼܢ ܩܵܠܹ̈ܐ ܕܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ: ܘܦܲܐܝܵܐ ܘܡܸܬ݁ܬ݁ܚܝܼܒ݂ܵܐ ܣܸܓ݂ܕ݁ܬ݂ܵܐ: )ܡ̣ܢ ܟܠ ܕܲܒ݂ܪܲܝܬ݁ ܘܐܲܬ݂ܩܸܢܬ݁) ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

	ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܡܵܘܬ݁ܒ݂ܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ. ܨܠܘܿܬܲܢ ܡܵܪܝ ܬܸܒ݂ܣܲܡ ܠܵܟ݂: ܘܒ݂ܵܥܘܼܬܲܢ ܬܸܥܘܿܠ ܩܘܼܕ݂ܡܲܝܟ: ܘܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܹܗ ܪܲܒܵܐ ܕܲܚܢܵܢܵܟ݂: ܢܸܬ݂ܦܲܢ̈ܝܵܢ ܫܹܐ̈ܠܵܬ݂ܵܐ ܠܲܣܢܝܼܩܘܼܬܲܢ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܡܵܘܬ݁ܒ݂ܵܐ ܕܠܸܠܝܵܐ ܒܐܲܝܠܹܝܢ ܕܐܸܬ݁ܬ݁ܚܸܕ݂ܘ. ܒܲܪܸܟ݂ܝ ܢܲܦ̮ܫܝ ܠܡܵܪܝܵܐ: ܥܒܲܪ ܡ݂ܢ ܒܝܼܫܬܵܐ ܘܲܥܒܸܕ݂ ܛܵܒ݂ܬ݂ܵܐ. ܢܲܦ̮ܫܝ ܐܸܬ݁ܬ݁ܘܵܝ ܥܕܲܡܵܐ ܠܐܸܡܲܬ݂ܝ ܥܬ݂ܝܼܕܲܬ݁ܝ ܠܡܸܚܛܵܐ. ܐܵܦ ܡ݂ܢ ܕܝܼܢܵܐ ܙܘܼܥܝ ܘܐܸܣܬܲܪܲܕܝ. ܡܸܛܠ ܕܡܵܘܬܵܐ ܩ̣ܪܸܒ݂ ܘܲܡ̣ܛܵܐ. ܠܡܲܠܟܵܐ ܡܫܝܼܚܵܐ ܒܥܵܝ ܘܐܸܬ݂ܟܲܫܲܦܝ ܡܸܛܠ ܕܒ݂ܵܬܲܪ ܥܘܼܢܕܵܢܵܐ ܠܵܐ ܐܝܼܬ݂ ܬܝܵܒ݂ܘܼܬ݂ܵܐ. ܐܵܘܕܵܝ ܘܐܸܡܲܪܝ ܥܲܕ݂ܠܵܐ ܐܹܡܘܼܬ݂ ܡܵܪܝ ܘܪܲܚܸܡܥܠܲܝ: ܬܢܝܼ ܕܫܵܒܹ̇ܩ ܠܹܟ݂ܝ ܟܠܹܗ ܥܵܘ̣ܠܹܟ݂ܝ. ܐܘܿ ܠܹܟ݂ܝ ܢܲܦ̮ܫܝ ܕܲܩ̣ܪܸܒ݂ ܝܵܘܡܵܐ ܕܒܹܗ ܡܸܬ݁ܬܲܒ݂ܥܲܬ݁ܝ ܟܠ ܚܘܼܫܒܵܢ̈ܝܼܢ ܕܒ݂ܐܝܼܕܲܝ̈ܟ݁ܝ ܐܸܬ݂ܓ݁ܥܸܠܘ. ܕܐܵܘܒܸ݁ܕܬ݁ܝ ܐܸܢܘܿܢ ܒܝܘܼܩܪܹܗ ܕܥܵܘ̣ܠܹܟ݂ܝ. ܠܪ̈ܲܚܡܹܐ ܕܓܲܒ݂ܠܘܼܟ݂ܝ ܩܪܵܝ ܠܐܝܼܵܠܹܟ݂ܝ. ܘܲܒ݂ܝܲܕ݂ ܨܵܘܡܵܐ ܘܲܨܠܘܿܬ݂ܵܐ ܐܲܣܵܝ ܣܘܼܪ̈ܚܵܢܲܝܟ݁ܝ. ܘܲܓ݂ܥܵܝ ܠܘܵܬܹܗ ܐܲܢ݇ܬ݁ ܚܲܠܸܠܲܝܢܝ ܡܵܪܝ ܘܪܲܚܸܡܥܠܲܝ. ܐܸܬ݂ܦܢܵܝ ܢܲܦ̮ܫܝ ܠܲܢܝܵܚܹܟ݂ܝ. ܢܲܦ̮ܫܵܐ ܫܚܝܼܩܲܬ݂ ܒܲܬ݂ܘܵܬ݂ܵܐ. ܘܪܘܼܚܵܐ ܛܥܝܼܢܲܬ݂ ܡܘܼܟܵܟ݂ܵܐ. ܚܲܕܸ݁ܬ݂ ܒܝܼ ܒܪ̈ܲܚܡܲܝܟ ܡܫܝܼܚܵܐ. ܘܗܲܒ݂ܠܝܼ ܕܐܸܩܲܛܪܸܓ݂ ܠܲܥܒ݂ܵܕܹ̈ܐ ܒܝܼܫܹ̈ܐ ܕܣܸܥ̇ܪܹܬ݂. ܘܐܲܫܘܵܢܝ ܕܲܒ݂ܟ݂ܠܥܸܕܵܢܝܼ̈ܢ ܐܲܝܟ݂ ܕܡܵܨܝܵܐ ܬܝܵܒ݂ܘܼܬ݂ܝ ܐܸܩܲܪܹܒ݂ ܩܘܼܕ݂ܡܲܝܟ. ܕܠܵܐ ܒܗܵܘ̇ ܝܵܘܡܵܐ ܐܲܦ̈ܝܼܢ ܒܐܲܦ̈ܝܼܢ ܢܩܘܼܡܘܼܢ ܠܘܼܩܒܲܠܝ. ܡܵܐ ܕܲܥ̣ܬ݂ܝܼܕܲܬ݁ܝ ܠܲܡܕ݂ܵܢ ܠܟ݂ܠܵܗ̇ ܬܸܒܹܝܠ ܒܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ ܘܲܒ݂ܩܘܼܫܬܵܐ. ܕܲܨܠܘܿܬ݂ܵܐ. ܚܲܢܵܢܵܐ ܕܲܦܬ݂ܝܼܚ. ܫܘܼܚܠܵܦܵܐ:ܓ. ܩܵܠܵܐ:ܝܐ. ܚܙܝܼ: 113. ܐܲܩܸܦ. ܒܩܵܠܵܐ ܪܲܒܵܐ. ܩܲܪܝܼܒ݂ܘܼ ܦܘܼܪܩܵܢܹܗ ܠܕ݂ܵܚܠܵܘ̈ܗܝ. ܩܪܸܒ݂ ܠܹܗ ܫܘܼܠܵܡܵܐ ܕܚܲܝܲܝ̈. ܘܣܵܟ݂ ܠܵܐ ܥܹܗܕܹܬ݂ ܕܐܸܬ݁ܘܼܒ݂ ܘܐܸܬ݂ܦܢܸܐ. ܘܗܵܫܵܐ ܕܚܸܫܹ̇ܬ݂ ܕܙܲܒ݂ܢܵܐ ܗ݇ܘܼ ܕܐܸܬ݁ܘܼܒ݂ ܡ݂ܢ ܚܲܘܒܲܝ̈. ܣܲܪܗܸܒ݂ ܡܵܘܬܵܐ ܘܚܲܝܲܝ̈ ܫܩܲܠ. ܘܠܵܐ ܙܘܵܕܹ̈ܐ ܠܡܲܪܕ݁ܝܼܬ݂ܝ ܢܸܣܒܹ̇ܬ݂. ܘܠܵܐ ܢܲܚ̈ܬܹܐ ܠܡܸܫܬ݁ܘܼܬ݂ܵܐ ܛܲܝܒܹ̇ܬ݂. ܘܐܸܢ ܡܲܡܪܲܚ ܐ݇ܢܵܐ ܕܐܸܥܘܿܠ ܥܲܡ ܓܒ݂ܵܘ̈ܗܝ. ܛܵ̇ܪܹܕ݂ ܘܡܲܦܸܩ ܠܝܼ ܡ݂ܢ ܗܵܘ̇ ܛܘܼܒ݂ܵܐ ܕܒܹܗ ܡܸܬ݂ܒܲܣܡܝܼܢ ܙܲܕ݁ܝܼܩܹ̈ܐ. ܚܘܼܣ ܡܵܪܝ ܡܸܛܠ ܪ̈ܲܚܡܲܝܟ. ܘܲܦܪܘܿܩܲܝܢܝ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁. ܡܵܪܵܐ ܡܪܲܚܡܵܢܵܐ ܚܘܼܢܲܝܢܝ ܘܪܲܚܸܡ ܥܠܲܝ. ܡܵܪܝܵܐ ܩܕ݂ܵܡܲܝܟ ܗ݇ܝܼ ܟܠܵܗ̇ ܪܸܓܲܬ݂ܝ. ܪܓ݂ܝܼܓ݂ ܐ݇ܢܵܐ ܕܐܸܬ݁ܘܼܒ݂ ܡ݂ܢ ܚܵܘܒܲܝ̈. ܘܡܸܬ݂ܥܵܘܲܟ݂ ܐ݇ܢܵܐ ܒܝܘܼܩܪܹܗ ܕܥܵܘ̣ܠܵܐ. ܡܸܛܠ ܕܟ݂ܠܝܘܿܡ ܩܪܵܒ݂ܵܐ ܩܲܫܝܵܐ ܕܠܵܐ ܫܲܠܘܵܐ ܪܲܪ̈ܓ݂ܝܼܓ݂ܵܬ݂ܵܐ ܣܵ̇ܕܲܪ ܥܠܲܝ. ܘܡܵܐ ܕܚܵܒܹ̇ܬ݂ ܝܵܘܡܵܢ ܠܲܚܕ݂ܵܐ. ܘܚܵܪ ܚܵܒܹ̇ܬ݂ ܠܝܼ ܡ݂ܢ ܐ݇ܚܪܹܬ݂ܵܐ. ܘܠܵܐ ܒܵܛܹ̇ܠ ܣܵܟ݂ ܬܲܟ݂ܬ݁ܘܼܫܹܗ ܕܥܲܡܝ ܘܠܲܝܬ݁ ܡܿܢ ܕܲܡܦܲܨܸܐ ܠܝܼ. ܐܸܠܵܐ ܐܸܢ ܐܲܢ݇ܬ݁ ܡܫܝܼܚܵܐ ܚܵܐܹ̇ܢ ܠܚܲܛܵܝܹ̈ܐ. ܫܒ݂ܘܿܩ ܚܵܘܒܲܝ̈ ܒܲܚܢܵܢܵܟ݂. ܘܲܦܪܘܿܩܲܝܢܝ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁. ܡܵܪܵܐ ܡܪܲܚܡܵܢܵܐ ܐܲܦܢܵܢܝ ܘܪܲܚܸܡܥܠܲܝ .
ܕܲܨܠܘܿܬ݂ܵܐ. ܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܫܘܼܚܠܵܦܵܐ: ܕ. ܩܵܠܵܐ: ܒ. ܪܹܫ ܩܵܠܵܐ
ܥܘܼܕ݂ܪܵܢܲܢ ܒܲܫܡܹܗ ܕܡܵܪܝܵܐ. ܥܘܼܕ݂ܪܵܢ ܛܲܝܒܘܼܬ݂ܵܟ݂: ܐܲܠܘܵܐ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ ܡܵܪܝܵܐ. ܘܲܬ݂ܚܹܝܬ݂ ܟܸܢܦܵܘ̈ܗܝ ܕܲܨܠܝܼܒ݂ܵܟ݂: ܣܲܬܲܪ ܚܲܝܲܝ̈ܢ ܡ݂ܢ ܒܝܼܫܵܐ. ܘܐܲܢܗܲܪ ܙܵܘܥܹ̈ܐ ܕܚܘܼܫܵܒܲܝ̈ܢ: ܒܗܸܪܓܵܐ ܦܲܐܝܵܐ ܕܝܘܼܠܦܵܢܵܟ݂. ܝܵܨܘܿܦܹܗ ܕܓܸܢܣܲܢ: ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܪܲܚܸܡ ܥܠܲܝܢ. ܡܵܐ ܛܵܒ݂ ܘܡܵܐ ܫܲܦܝܼܪ. ܫܲܝܢܵܐ ܘܲܫܠܵܡܵܐ: ܫܵܐܹ̇ܠ ܐ݇ܢܵܐ ܒܟ݂ܠܙܒܲܢ ܡܵܪܝܵܐ. ܘܲܒ݂ܬܲܟ݂ܫܲܦ̮ܬܵܐ ܐܸܩܪܸܐ ܫܡܵܟ݂: ܘܐܸܦܢܸܐ ܘܐܹܩܘܼܡ ܩܕ݂ܵܡ ܡܲܕ݂ܒܚܵܟ݂. ܘܝܲܩ̈ܕܹܐ ܫܲܠܡܹ̈ܐ ܐܲܝܟ݂ ܕܡ݂ܢ ܩܕ݂ܝܼܡ: ܒܠܸܒܵܐ ܕܲܟ݂ܝܵܐ ܐܲܣܸܩ ܠܵܟ݂. ܝܵܨܘܿܦܹܗ ܕܓܸܢܣܲܢ: ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܪܲܚܸܡ ܥܠܲܝܢ. ܐܸܢܵܐ ܕܹܝܢ ܥܠܲܝܟ ܡܵܪܝܵܐ ܐܸܬ݁ܬܲܟ݂ܠܹܬ݂. ܥܲܠ ܬܘܼܟ݂ܠܵܢܵܐ ܗ݇ܘ̣ ܕܲܚܢܵܢܵܟ݂: ܩܵܪܹܝܢܲܢ ܠܵܟ݂ ܡܵܪܝܵܐ. ܦܬܲܚ ܠܲܦܝܵܣܲܢ ܬܪܲܥ ܪ̈ܲܚܡܲܝܟ: ܕܢܸܣܲܒ݂ ܡܸܢܹܗ ܫܘܼܒ݂ܩܵܢܵܐ. ܘܐܲܫܦܲܥ ܥܠܲܝܢ ܡܵܪܝ ܛܲܝܒܘܼܬ݂ܵܟ݂: ܐܲܝܟ݂ ܕܥܲܠ ܟܸܢܫܵܐ ܕܲܫܠܝܼܚܹ̈ܐ. ܛܵܒ݂ܵܐ ܡܢܲܛܪܵܢܲܢ: ܡܫܝܼܚܵܐ ܐܲܠܵܗܲܢ ܪܲܚܸܡ ܥܠܲܝܢ. ܡܸܛܠ ܕܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܛܵܒ݂ ܐܲܢ݇ܬ݁. ܛܵܒ݂ܵܐ ܕܛܲܟܸܣ ܟܠ ܒܸܖ̈ܝܵܢ ܒܪܸܡܙܵܐ ܣܓ݂ܝܼܕ݂ܵܐ ܕܚܲܝܠܹܗ. ܫܲܝܸܢ ܥܲܡܲܢ ܐܸܣܛܘܼܟ݂ܣܹ̈ܐ: ܘܐܲܦܨܲܚ ܠܬܹܒܹܝܠ ܡ݂ܢ ܣܵܟ݂ܵܗ̇. ܘܗܲܒ݂ ܥܘܼܕ݂ܪܵܢܵܐ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܘܲܡܠܝܼ ܦܘܼܡܲܢ ܬܵܘܕܝܼܬ݂ܵܐ. ܥܸܠܵܝ ܡ݂ܢ ܙܲܒ݂ܢܹ̈ܐ: ܟܝܵܢܵܐ ܕܠܵܐ ܫܘܼܪܵܝ ܪܲܚܸܡ ܥܠܲܝܢ. ܣܲܒ݂ܪܵܐ ܕܟ݂ܠܗܘܿܢ ܣܵܘܦܹܝ̈ܗ݁ ܕܐܲܪܥܵܐ. ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܵܐ ܕܫܲܪܝܼܪ̈ܵܘܗܝ: ܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܒܪ̈ܲܚܡܲܝܟ. ܘܒܲܛܸܠ ܡܸܢܵܗ̇ ܚܸܖ̈ܝܵܢܹܐ: ܘܣܸܕ݂ܩܹ̈ܐ ܘܦܘܼܠܵܓܹ̈ܐ ܕܣܵܢ̈ܐܹܐ. ܘܫܲܝܢܵܟ݂ ܢܲܡܠܸܟ݂ ܥܲܠ ܝܲܠܕܹܝ̈ܗ̇: ܘܢܸܫܬܲܡܠܸܐ ܒܲܢ ܨܸܒ݂ܝܵܢܵܟ݂. ܘܢܸܗܘܸܐ ܓܵܡܘܿܖܹ̈ܐ: ܕܨܸܒ݂ܝܵܢ ܡܵܪܘܼܬ݂ܵܟ݂ ܘܪܲܚܸܡ ܥܠܲܝܢ. ܝܲܡܝܼܢܵܟ݂ ܡܵܪܝܵܐ ܐܲܓܸܢ ܥܠܲܝ. ܝܲܡܝܼܢ ܪ̈ܲܚܡܲܝܟ ܡܵܪܹܐ ܟܠ: ܐܲܓܸܢ ܥܠܲܝܢ ܒܟ݂ܠܙܒܲܢ. ܘܒܲܨܠܝܼܒ݂ ܢܘܼܗܪܵܟ݂ ܢܸܬ݂ܒܲܪܲܟ݂: ܟܸܢܫܲܢ ܕܣܵܓܹܕ݂ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܘܲܕ݂ܚܘܿܩ ܡܸܢܲܢ ܠܲܢܟ݂ܝܼܠܵܐ: ܕܲܠܚܝܼܡ ܥܠܲܝܢ ܒܟ݂ܠܥܸܕܵܢ. ܛܵܒ݂ܵܐ ܕܡ݂ܢ ܥܵܠܲܡ: ܡܫܝܼܚܵܐ ܐܲܠܵܗܲܢ ܪܲܚܸܡ ܥܠܲܝܢ. ܕܐܘܼܪܚܵܐ܀ ܢܲܛܲܪܲܝܢܝ ܐܲܝܟ݂ ܒܵܒ݂ܬ݂ܵܐ ܕܥܲܝܢܵܐ. ܠܕ݂ܵܘܝܼܕ݂ ܕܲܥ̣ܪܲܩ ܡܸܢ ܫܵܐܘܿܠ: ܒܡܲܕ݂ܒ݁ܪܵܐ ܢ̣ܛܲܪܬܵܝܗܝ ܡܵܪܝܵܐ. ܘܲܚܠܵܦ ܚܲܫܹ̈ܐ ܕܐܲܟ݂ܣܸܢܝܵܐ: ܒܬ݂ܵܓ݂ܵܐ ܗܕ݂ܝܼܪܵܐ ܟܲܠܸܠܬܵܝܗܝ. ܘܐܸܢܵܐ ܥܲܒ݂ܕܵܟ݂ ܕܥܲܠ ܣܲܒ݂ܪܵܟ݂: ܒܝܘܼܬ݂ܪܵܢܵܐ ܕܢܲܦ̮ܫܝ ܪܵܗܹ̇ܛ ܐ݇ܢܵܐ. ܗܲܒ݂ܠܝܼ ܕܠܵܐ ܐܸܒ݂ܗܲܬ݂: ܡܸܛܠ ܪܘܼܫܡܵܟ݂ ܕܒܹܗ ܩܲܕܸ݁ܫܬܵܢܝ. ܕܡܸܛܪܵܐ܀ ܫܡܲܥ ܐܲܠܵܗܵܐ ܘܪܲܚܸܡ ܥܠܲܝ. ܠܵܐ ܗ݇ܘ̣ܵܐ ܠܡܸܛܪܵܐ ܫܚܝܼܡܵܐܝܼܬ݂: ܪܓ݂ܝܼܓ݂ܝܼܢܲܢ ܡܵܪܝܵܐ. ܐܸܠܵܐ ܡܸܠܬ݂ܵܟ݂ ܒܵܥܹ̇ܝܢܲܢ: ܕܐܝܼܬܹܝܗ̇ ܐܵܬ݂ܵܐ ܕܬܲܪܥܘܼܬ݂ܵܐ. ܕܪܘܼܓ݂ܙܵܟ݂ ܐܲܟ݂ܡܲܪ ܚܲܩ̈ܠܵܬܲܢ: ܛܲܠܵܟ݂ ܢܲܦܨܲܚ ܢܲܦ̮̈ܫܵܬܲܢ. ܥܲܬ݁ܝܼܪ ܒܡܵܘܗ̈ܒ݂ܵܬܹܗ: ܪܵܚܹ̇ܡ ܐ݇ܢܵܫܵܐ ܪܲܚܸܡ ܥܠܲܝܢ. ܕܡܲܪܝܲܡ܀ ܢܵܛܲܪ ܡܵܪܝܵܐ ܠܟ݂ܠܗܘܿܢ ܕܵܚܠܘ̈ܗܝ. ܒܲܨܠܘܿܬ݂ ܡܲܪܝܲܡ ܕܝܼܠܸܕ݂ܬ݂ܵܟ݂: ܡܫܝܼܚܵܐ ܡܵܪܹܗ ܕܓܸܢܣܲܢ. ܫܲܝܸܢ ܥܵܠܡܵܐ ܡ݂ܢ ܣܵܟܹܗ: ܘܩܲܝܸܡ ܥܹܕ݁ܬܵܟ݂ ܒܲܚܢܵܢܵܟ݂. ܕܒ݂ܵܗ̇ ܡܸܙܕܲܝܲܚ ܕܘܼܟ݂ܪܵܢܵܗ̇܆ ܕܡܲܪܝܲܡ ܐܸܡܵܐ ܕܐ݇ܢܵܫܘܼܬ݂ܵܟ݂. ܝܲܠܕܵܐ ܕܕ݂ܘܼܡܵܪܵܐ: ܦܹܐܪܵܐ ܒܪܝܼܟ݂ܵܐ ܪܲܚܸܡ ܥܠܲܝܢ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܡܲܫܪܝܼܬ݂ܵܐ ܕܡܲܠܲܐܟ݂ܵܘ̈ܗܝ ܕܡܵܪܝܵܐ. ܫܠܵܡܵܐ ܥܲܡܗܘܿܢ ܕܐܲܬ݂ܠܹܝܛܹ̈ܐ: ܕܲܫܦܲܪܘ ܠܚܘܼܒܵܟ݂ ܡܵܪܝܵܐ. ܕܒܲܨ̈ܠܵܘܵܬ݂ܗܘܿܢ ܡܸܬ݂ܥܲܕܪܝܼܢ: ܐܲܝܠܹܝܢ ܕܲܒ݂ܗܘܿܢ ܡܸܬ݂ܓܵܘܣܝܼܢ. ܫܘܼܒ݂ܚܵܐ ܠܚܲܝܠܵܐ ܕܲܟ݂ܣܸܐ ܒܗܘܿܢ: ܘܣܵܥܲܪ ܟܠܫܵܥ ܠܲܒ݂ܪܝܼܬ݂ܵܐ. ܟܠܝܼܠܵܐ ܕܫܲܪܝܼܪ̈ܵܘܗܝ: ܡܵܪܵܐ ܕܟܹܐܢܹ̈ܐ ܪܲܚܸܡ ܥܠܲܝܢ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܐܲܝܟ݂ ܡܕ݂ܝܼܢ݇ܬܵܐ ܕܲܟ݂ܪܝܼܟ݂ ܠܵܗ̇ ܫܘܼܪܵܐ. ܨܠܝܼܒ݂ܵܟ݂ ܡܫܝܼܚܵܐ ܡܲܚܝܵܢܲܢ: ܢܸܗܘܸܐ ܫܘܼܪܵܐ ܠܢܲܦ̮̈ܫܵܬܲܢ. ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡ݂ܢ ܒܝܼܫܵܐܘܚܲܝ̈ܠܵܘܵܬܹܗ. ܘܢܸܗܘܸܐ ܥܲܡܲܢ ܘܒܲܝܢܵܬܲܢ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ. ܘܢܸܓ݂ܡܘܿܪ ܨܸܒ݂ܝܵܢܵܟ݂: ܡܫܝܼܚܵܐ ܐܲܠܵܗܲܢ ܘܪܲܚܸܡ ܥܠܲܝܢ. ܕܐܲܒ݂ܘܼܢ܀ ܒܥܝܼ ܡ݂ܢ ܡܵܪܝܵܐ ܘܨܲܠܵܐ ܩܕ݂ܵܡܵܘܗܝ. ܐܲܒ݂ܘܼܢ ܪܵܚܡܹܗ ܕܲܡܫܝܼܚܵܐ: ܕܡܸܛܠ ܚܘܼܒܹܗ ܚܲܫܹ̈ܐ ܣܒܲܠ. ܐܲܦܝܼܣ ܘܲܒ݂ܥܝܼ ܡ݂ܢ ܡܵܪܵܟ݂: ܕܢܲܫܪܸܐ ܫܲܝܢܹܗ ܒܲܒ݂ܪܝܼܬܵܐ. ܘܢܲܩܒܸܠ ܐܲܦܵܘ̈ܗܝ ܒܚܲܕ݂ܘܼܬ݂ܵܐ: ܒܝܵܘܡܵܐ ܕܐܵܬܹܐ ܠܲܡܕ݂ܵܢ ܟܠ. ܐܘܿ ܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ: ܨܠܘܿܬ݂ܵܟ݂ ܬܸܗܘܸܐ ܠܲܢ ܒܟ݂ܠ ܥܸܕܵܢ̈ܝܼܢ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܡܸܛܠ ܕܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܛܵܒ݂ ܐܲܢ݇ܬ݁. ܛܵܒ݂ܵܐ ܕܓܲܒ݂ܠܹܗ ܠܲܟ݂ܝܵܢܲܢ: ܕܠܵܐ ܦܝܵܣܵܐ ܡ݂ܢ ܫܘܼܪܵܝܵܐ. ܐܲܢ݇ܬ݁ ܡܵܪܝ ܚܲܕܸܬ݂ ܨܘܼܪ̈ܵܬܲܢ: ܕܚܲܒܸܠ ܐܸܢܹܝܢ ܢܵܟ݂ܘܿܠܵܐ. ܘܲܓ݂ܡܘܿܪ ܨܹܐܕܲܝܢ ܒܲܥܒ݂ܵܕ݂ܵܐ: ܒܲܪ݇ܬ݂ ܩܵܠܵܐ ܕܐܸܡܲܪܬ݁ ܠܪ̈ܵܚܡܲܝܟ. ܕܟ݂ܠ ܕܐܵܟܹܠ ܦܲܓ݂ܪܵܟ݂ ܘܲܕ݂ܡܵܟ݂: ܡܸܫܬܵܘܬܲܦ ܥܲܡܵܟ݂ ܘܡܸܬ݂ܒܲܣܲܡ. ܐܲܪܝܼܡ ܩܵܠܹܗ ܘܙܵܥܲܬ̤ ܐܲܪܥܵܐ. ܩܵܠܵܐ ܕܲܩܪܵܝܗܝ ܠܠܵܥܵܙܲܪ: ܢܸܩܪܸܐ ܘܲܢܩܝܼܡ ܡܵܪܝ ܠܥܲܒ݂ܕܲܝ̈ܟ. ܘܢܸܠܒܫܘܼܢ ܐܸܣܛܲܠ ܬܸܫܒܘܿܚܬܵܐ: ܘܢܸܚܕܘܿܢ ܒܲܓ݂ܢܘܿܢ ܡܲܠܟܘܼܬ݂ܵܟ݂. ܥܲܡ ܟܹܐܢܹ̈ܐ ܕܣܲܟܝܼܘ ܠܕܸܢܚܵܟ݂: ܘܢܸܬ݂ܒܲܣܡܘܼܢ ܒܵܗ̇ ܒܲܚܙܵܬ݂ܵܟ݂. ܘܲܢܕ݂ܘܼܨܘܼܢ ܥܲܡܵܟ݂: ܒܦܵܬ݂ܘܿܪ ܡܲܠܟܘܼܬ݂ܵܟ݂ ܐܲܝܟ݂ ܫܘܼܘܕܵܝܵܟ݂܀
ܕܥܢܝ̈ܕܐ . ܗܸܢ̣ܘܿܢ ܥܵܒ݂ܪܝܼܢ ܘܐܲܢ݇ܬ݁ ܩܲܝܵܡ ܐܲܢ݇ܬ݁. ܥܵܠܡܵܐ ܗܵܢܵܐ ܟܠܹܗ ܥܵܒ݂ܘܿܪܵܐ ܗ݇ܘ̣ ܘܒ݂ܵܛܹ̇ܠ ܚܘܼܬ݂ܪܹܗ. ܘܡܵܘܦܸܐ ܫܘܼܒ݂ܗܵܪܹܗ. ܘܗܵܘܹܝ̇ܢ ܕܠܵܐ ܝܘܼܬ݂ܪܵܢ ܟܠ ܕܪܲܚܡ̣ܘܼܗܝ ܘܐܲܚ̣ܒ݂ܘܼܗܝ ܘܟܲܢܸܫ̣ܘ ܩܸܢܝܵܢܵܘ̈ܗܝ. ܘܕ݂ܵܥ̇ܟ݁ܝܼܢ ܠܲܡܦܹܝܕܲܝ̈ܗܘܿܢ ܥܲܡ ܕܚܲܡܸܫ ܣܲܟ݂̈ܠܵܬ݂ܵܐ. ܘܒ݂ܵܟܹܝ̇ܢ ܘܡܸܬ݁ܬܲܢܚܝܼܢ ܡܵܐ ܕܚܵܙܹܝ̇ܢ ܠܲܡܪ̈ܲܚܡܵܢܹܐ ܕܦܵܪܚܝܼܢ ܒܐܵܐܲܪ. ܠܐܘܼܪܥܹܗ ܕܚܲܬ݂ܢܵܐ ܫܒ݂ܝܼܚܵܐ. ܕܢܸܥܠܘܼܢ ܥܲܡܹܗ ܠܗܵܝ̇ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܠܵܐ ܡܸܫܬܲܪܝܵܐ. ܗܵܐ ܠܡܝܼܬܹܐ ܥܵܒܹܕ݂ ܐܲܢ݇ܬ݁ ܬܸܕ݂ܡܪ̈ܵܬ݂ܵܐ. ܗܵܐ ܠܡܝܼ̈ܬܹܐ ܡ݂ܢ ܩܲܒ݂ܪܹ̈ܐ ܩܵܪܹܐ ܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ. ܘܲܠܟ݂ܠܗܘܿܢ ܫܲܘܝܵܐܝܼܬ݂ ܡܢܲܚܸܡ ܐܲܢ݇ܬ݁ ܡ݂ܢ ܫܝܘܿܠ. ܐܸܠܵܐ ܡܦܝܼܣ ܐ݇ܢܵܐ ܠܵܟ݂ ܒܲܚܢܲܓ݂ܬ݂ܵܐ ܡܵܪܵܐ ܕܥܵܠܡܹ̈ܐ. ܕܡܵܐ ܕܡܸܬ݂ܓܲܠ̈ܝܵܢ ܟܲܣ̈ܝܵܬ݂ܵܐ ܕܲܩܢܘܿܡܝ. ܠܵܐ ܬܫܲܐܠܲܢܝ ܒܕ݂ܝܼܢܵܟ݂ ܟܹܐܢܵܐ. ܕܐܸܢܵܐ ܒܲܥܒ݂ܵܕܲܝ̈ ܒܝܼܫܹ̈ܐ ܟܠܝܘܿܡ ܐܲܪܓܸ݁ܙܬܵܟ݂. ܘܚܲܝܵܒ݂ ܐ݇ܢܵܐ ܩܕ݂ܵܡ ܟܹܐܢܘܼܬ݂ܵܟ݂. ܘܠܲܝܬ݁ ܠܝܼ ܓܲܠܝܘܼܬ݂ܵܐ ܐܲܦܹ̈ܐ. ܡܵܪܵܐ ܛܵܒ݂ܵܐ ܐܲܠܵܗܵܐ ܚܘܼܢܲܝܢܝ ܒܪ̈ܲܚܡܲܝܟ ܘܚܲܣܵܐ ܠܝܼ ܚܵܘܒܲܝ̈ ܘܪܲܚܸܡ ܥܠܲܝ.
ܠܐܲܝܟܵܐ ܐܹܙܲܠ ܡ݂ܢ ܪܘܼܚܵܟ݂. ܒܫܵܥܬ݂ܵܐ ܡܵܐ ܕܡ݂ܢ ܓܵܘ ܫܸܠܝܵܐ ܒܕܸܚܠܵܐ ܪܲܒܵܐ ܙܵܥܩܵܐ ܩܲܪܢܵܐ ܕܡܹܐܬ݂ܝܼܬ݂ܵܟ݂. ܡ݂ܢ ܫܡܲܝܵܐ ܛܵܝܣܝܼܢ ܠܡܸܚܲܬ݂ ܒܣܘܼܪܗܵܒ݂ܵܐ. ܬܸܓ݂ܡܹ̈ܐ ܬܡܝܼܗܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ. ܘܡܵܐ ܕܡ݂ܢ ܩܲܒ݂ܪܹ̈ܐ ܩܵܝ̇ܡܝܼܢ ܟܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܒܲܪܬܹܝܬ݂ܵܐ ܠܡܸܫܬܲܐܵܠܘܼ. ܪܵܥ̇ܠܝܼܢ ܚܲܝ̈ܠܵܘܵܬ݂ܵܐ ܥܸܠܵܝܹ̈ܐ. ܡ݂ܢ ܥܘܼܙܹܗ ܕܕ݂ܝܼܢܵܐ ܕܗܵܘܹ̇ܐ ܒܬܲܚܬܵܝܹ̈ܐ. ܘܡܵܐ ܕܲܟ݂ܪ̈ܘܿܒܹܐ ܛܥܝܼܢܝܼܢ ܡܙܲܝܚܝܼܢ ܠܵܟ݂ ܕܲܝܵܢܵܐ ܟܹܐܢܵܐ. ܐܸܠܵܐ ܒܗܵܘ̇ ܕܝܼܢܵܐ ܕܚܝܼܠܵܐ ܡܵܐ ܕܡܸܬ݂ܦܪܲܥ ܥܒ݂ܵܕ݂ܵܘ̈ܗܝ ܕܟ݂ܠܢܵܫ. ܪܲܚܸܡ ܥܠܲܝ ܐܲܢ݇ܬ݁ ܪܵܚܹܡ ܐ݇ܢܵܫܵܐ.
ܘܐܲܝܟܵܐ ܐܸܛܫܸܐ ܡ݂ܢ ܩܕ݂ܵܡܲܝܟ. ܒܫܵܥܬ݂ܵܐ ܗܵܝ̇ ܕܚܲܝ̈ܠܵܘܵܬ݂ܵܐ ܕܥܝܼܪܹ̈ܐ ܟܪܝܼܟ݂ܝܼܢ ܡܙܲܝܚܝܼܢ ܠܪܲܒ݁ܘܼܬ݂ܵܟ݂. ܘܡܵܐ ܕܡܲܙܥܩܝܼܢ ܘܩܵܥܹܝ̇ܢ ܟܠܗܘܿܢ ܒܩܵܠܵܐ ܕܩܲܪܢܵܐ ܘܫܝܼܦܘܿܪܵܐ. ܡܵܚܹܝ̇ܢ ܠܵܗ̇ ܚܸܒ݂̈ܠܹܐ ܩܫܲܝ̈ܵܐ ܠܐܲܪܥܵܐ ܕܬ݂ܵܘܠܸܕ݂ ܥܲܢܝܼ̈ܕܹܐ. ܘܩܵܝ̇ܡܝܼܢ ܝܲܠܕܵܘ̈ܗܝ ܕܐܵܕ݂ܵܡ ܥܲܦܪܵܢܵܐ. ܢܵܦܩ̇ܝܼܢ ܠܐܘܼܪܥܵܟ݂ ܡܵܪܲܢ ܒܐܘܿܫܲܥܢܹ̈ܐ. ܘܡܸܬ݂ܥܲܠܹܝܢ ܛܵܒܹ̈ܐ ܠܪܵܘܡܵܐ. ܘܒ݂ܝܼܫܹ̈ܐ ܦܵܝ̇ܫܝܼܢ ܥܲܠ ܐܲܪܥܵܐ. ܚܘܼܢܲܝܢܝ ܒܗܵܘ̇ ܝܵܘܡܵܟ݂ ܪܲܒܵܐ. ܡܵܐ ܕܐܵܬܹܝܬ݁ ܠܲܡܕ݂ܵܢ ܡܵܪܲܢ. ܘܪܲܚܸܡ ܥܠܲܝ ܐܲܢ݇ܬ݁ ܪܵܚܹܡ ܐ݇ܢܵܫܵܐ.
ܫܲܒܲܚ. ܫܲܒܲܚ. ܒܲܪ̈ܓ݂ܝܼܓ݂ܵܬ݂ܵܐ ܕܠܵܐ ܝܘܼܬ݂ܪܵܢ. ܟܠܝܘܿܡ ܡܲܪܡܸܐ ܠܝܼ ܒܥܸܠܕܒ݂ܵܒ݂ܝ. ܘܡܸܛܠ ܕܲܚ̣ܙܵܢܝ ܕܲܠܣܲܢ̈ܝܵܬ݂ܵܐ ܢܩܸܐ ܪܸܥܝܵܢܝ. ܟܠܝܘܿܡ ܩܪ̈ܵܒ݂ܵܘܗܝ ܣܵ̇ܕܲܪ ܥܠܝ. ܘܡܵܐ ܕܲܫܓܲܫ̣ ܢܲܦ̮ܫܝ ܟܲܣܝܵܐܝܼܬ݂. ܐܲܦ̣ܗܝܼ ܥܲܝܢܲܝ̈ ܒܚܵܘܪܵܐ ܙܲܠܝܼܠܵܐ. ܚܙܝܼ ܡܵܪܝ ܕܲܡܦܲܠܦܲܠ ܐ݇ܢܵܐ ܒܣܘܿܓ݂ܵܐܐ ܕܒ݂ܝܼܫ̈ܵܬ݂ܵܐ. ܡ݂ܢ ܝܲܕ݂ ܡܵܪܘܿܕ݂ܵܐ. ܕܗ̤ܘ ܒܲܚܣܵܡܹܗ ܥܲܒ݂ܕܲܢܝ ܕܐܸܗܘܸܐ ܡܘܼܝܵܩܵܐ. ܒܚܲܝܠܵܟ݂ ܡܵܪܝ ܡܬ݂ܘܿܡܵܝܵܐ. ܗܵܘ ܕܓܲܒ݂ܠܢܝ ܡ݂ܢ ܕܲܚܝܼܚܵܐ. ܣܚܘܿܦܲܝܗܝ ܠܲܛܪܘܿܢܵܐ ܕܲܫܒ݂ܵܢ̣ܝ ܘܪܲܚܸܡܥܠܲܝ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ̣ܵܐ: ܨܠܘܿܬ̣ܵܐ: ܚܘܼܣ ܥܠܲܝܢ ܐܲܢ݇ܬ݁ ܚܲܢܵܢܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܸܬܼܦܲܢ ܥܠܲܝܢ ܡܠܸܐ ܖ̈ܲܚܡܹܐ: ܘܠܵܐ ܡܵܪܝ ܬܲܗܡܸܐ ܚܝܵܪܵܟ݂ ܘܲܒ݂ܛܝܼܠܘܼܬ݂ܵܟ݂ ܡܸܢܲܢ: ܕܲܥܠܲܝܟ݁ܘܼ ܣܲܒܼܪܲܢ ܘܬܼܘܼܟܼܠܵܢܲܢ ܒܟܼܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟܼܠ: ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.

 ܘܲܚܢܲܢ ܥܲܡܵܐ ܕܝܼܠܹܗ ܘܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬܹܗ. * ܝܵܘܡܵܢܵܐ ܐܸܢ ܒܩܵܠܹܗ ܬܸܫܡ̱ܥܘܼܢ

܀

ܒܕܸܡܥܹ̈ܐ ܚܢܝܼ̈ܓ݂ܵܬ݂ܵܐ ܬܵܘ ܢܸܫܐܲܠ ܫܘܼܒ݂ܩܵܢ ܚܵܘ̈ܒܵܬ݂ܵܐ. ܘܢܸܩܪܸܐ ܟܠܫܵܥ ܠܪ̈ܲܚܡܹܐ ܫܦܝܼܥܹ̈ܐ ܕܚܲܢ̣ܘ ܠܲܟ݂ܝܵܢܲܢ. ܡܵܪܝܵܐ ܚܲܣܵܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀
ܫܲܒܲܚ: ܥܵܠܲܡ. ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܚܘܼܣ ܥܠܲܝ ܡܸܛܠ ܕܲܚܛܹ̇ܝܬ݂ ܠܵܟ݂. ܘܕܲܠܵܗ̇ ܠܢܲܦ̮ܫܝ ܡ݂ܢ ܓܵܘ ܝܲܡܵܐ ܕܚܵܘܒܹ̈ܐ ܕܣܹܥ̇ܪܹܬ݂. ܕܐܲܢ݇ܬ݁ܘܼ ܝܲܡܵܐ ܕܲܚܢܵܢܵܐ ܘܫܵܒܹ̇ܩ ܚܵܘ̈ܒܵܬ݂ܵܐ܀
ܬܵܘ ܢܫܲܒܲܚ ܠܡܵܪܝܵܐ. * ܘܢܸܙܡܲܪ ܠܐܲܠܵܗܲܢ ܦܵܪܘܿܩܵܐ܀ ܒܬܲܪܥܵܐ ܕܫܘܼܠܵܡܵܐ ܕܚܲܝܲܝ̈ܟ݁ܝ ܢܲܦ̮ܫܝ ܗܵܐ ܩܵܝ̇ܡܲܬ݁ܝ ܐܲܣܓܵܝ ܐܘܿܠܝ̈ܵܬ݂ܵܐ. ܘܩܲܪܸܒ݂ܝ ܦܝܵܣܵܐ ܘܕܸܡܥܹ̈ܐ ܕܫܵܘ̈ܝܵܢ ܠܲܬ݂ܝܵܒ݂ܘܼܬ݂ܵܐ. ܠܒ݂̇ܵܚܲܪ ܠܸܒܵܐ ܘܟ݂ܘܿܠ̈ܝܵܬ݂ܵܐ ܕܢܸܚܘܼܢܹܟ݂ܝ ܒܪ̈ܲܚܡܹܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀
ܬܸܫܒ݁ܘܿܚܬܵܐ. ܕܡܸܬ݂ܐܲܡܪܵܐ. ܒܗܵܢܵܐ ܫܵܒܼܘܿܥܵܐ. ܒܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܕܐܲܚܪ̈ܵܝܹܐ. ܐ. ܐܘܿ ܠܹܟ݂ܝ ܢܲܦ̮ܫܵܐ ܕܕܸܡܟܲܬ̤ ܒܥܵܘ̣ܠܵܐ. ܩܘܼܡܝ ܐܸܬ݂ܢܲܦ̄ܨܝ ܡ݂ܢ ܕܲܡܟ݁ܘܼܬܹܟ݂ܝ܀ ܒ. ܒܲܬ݂ܝܵܒ݂ܘܼܬ݂ܵܐ ܗܲܪܛܝ ܘܐܸܬ݂ܓܵܘܲܣܝ. ܕܕܲܡ ܢܸܬ݂ܪܲܥܸܐ ܟܹܐܢܹ̈ܐ ܒܕܸܡܥܲܝ̈ܟ݁ܝ܀ ܓ. ܓܠܵܝ ܬܲܚܦܝܼܬ݂ܵܐ ܕܲܫܟ݂ܝܼܪ̈ܵܬܹܟ݂ܝ. ܡ݂ܢ ܪܸܥܝܵܢܹܟ݂ܝ ܘܚܘܼܪܝ ܬܲܩܢܵܐܝܼܬ݂܀ ܕ. ܕܸܡܥܹ̈ܐ ܕܚܲܫܵܐ ܐܲܫܘܿܕ݂ܝ ܟܠܫܵܥ. ܩܕ݂ܵܡ ܕܲܝ̇ܵܢܵܐ ܥܲܠ ܣܲܟ݂ܠܘܼܬܹܟ݂ܝ܀ ܗ. ܗܵܘܬ݂ܵܐ ܕܚܸܫܟܵܐ ܡܛܲܝܒ݂ܵܐ ܠܥܲܡܠܲܝ̈ܟ݁ܝ. ܘܲܠܦܘܼܪܥܵܢܵܐ ܕܲܫܟ݂ܝܼܪ̈ܵܬܹܟ݂ܝ܀ ܘ. ܘܵܝܵܐ ܥܦܝܼܦܵܐ ܗܲܒ݂ܝ ܠܲܩܢܘܿܡܹܟ݂ܝ. ܕܢܸܫܡܲܥ ܟܹܐܢܵܐ ܘܢܸܬ݂ܪܲܥܸܐ ܠܹܟ݂ܝ܀ ܙ. ܙܘܼܥܝ ܘܐܸܣܬܲܪܲܕ݂ܝ ܛܵܒ݂ ܕܵܘܝܼܬ݂ܵܐ. ܡ݂ܢ ܓܹܗܲܢܵܐ ܕܐܸܬ݁ܛܲܝܒܲܬ̤ ܠܹܟ݂ܝ܀ ܚ. ܚܒܲܪܬ݂ܵܐ ܕܥܝܼܪܹ̈ܐ ܒܟ݂ܵܝ ܥܲܠ ܩܢܘܿܡܹܟ݂ܝ. ܕܥܵܘ̣ܠܹܟ݂ܝ ܐܲܙܸܠ̣ ܠܝܲܩܝܼܪܘܼܬܹܟ݂ܝ܀ ܛ. ܛܒܲܥ̣ܬ݁ܝ ܘܐܸܫܬܲܦܲܠܬ݁ܝ ܒܲܪ̈ܓ݂ܝܼܓ݂ܵܬ݂ܵܐ. ܘܠܵܐ ܐܸܬ݂ܢܲܦܲܨܬ݁ܝ ܡ݂ܢ ܛܘܼܒܵܥܹܟ݂ܝ܀ ܝ. ܝܲܥܪܵܐ ܘܟܘܼܒܹ̈ܐ ܚܢܲܩ̣ܘ ܚܘܼܫܵܒܲܝ̈ܟ݁ܝ. ܘܠܵܐ ܝܲܗ݇ܒ݂ܘ ܦܹܐܪܹ̈ܐ ܕܲܬ݂ܝܵܒ݂ܘܼܬ݂ܵܐ܀ ܟ. ܟܲܪܟ݂ܘܼܟ݂ܝ ܚܲܫܹ̈ܐ ܒܲܕ݂ܡܘܼܬ݂ ܓܲܠܠܹ̈ܐ. ܘܛܲܒ݁ܥܘܼܗ̇ ܠܐܸܠܦܹܟ݂ܝ ܒܝܲܡܵܐ ܕܚܵܘܒܹ̈ܐ܀ ܠ. ܠܵܐ ܣܵܟ݂ ܐܲܣܸܩ̣ܬ݁ܝ ܥܲܠ ܥܘܼܗܕ݂ܵܢܹܟ݂ܝ. ܝܵܘܡܵܐ ܕܒ݂ܵܚܲܪ ܟܠ ܟܲܣ̈ܝܵܬܹܟ݂ܝ܀ ܡ. ܡܵܢܵܐ ܬܸܥܒ݁ܕ݂ܝܼܢ ܡܵܐ ܕܡܲܪܗܸܒ݂ ܠܹܟ݂ܝ. ܩܵܠܹܗ ܕܚܲܬ݂ܢܵܐ ܕܬܸܦܩܝܼܢ ܠܐܘܼܪܥܹܗ܀ ܢܢ. ܢܘܼܗܪܵܐ ܕܥܲܡܠܲܝ̈ܟ݁ܝ ܕܥܸܟ݂ ܘܐܵܘܦ̣ܝܼ ܠܹܗ. ܘܥܲܡ ܣܲܟ݂̈ܠܵܬ݂ܵܐ ܠܒܲܪ ܡܸܫܬܲܢܩܲܬ݁ܝ܀ ܣ. ܣܲܒ݂ܝ ܠܹܟ݂ܝ ܡܸܫܚܵܐ ܐܘܿ ܕܵܘܝܼܬ݂ܵܐ. ܥܲܕ݂ ܐܝܼܬ݂ ܐܲܬ݂ܪܵܐ ܠܡܸܙܕܵܘܲܕ݂ܘܼ܀ ܥ. ܥܲܡ ܙܵܢܝܼܬ݂ܵܐ ܫܐܲܠܝ ܫܘܼܒ݂ܩܵܢܵܐ. ܘܲܒ݂ܝܲܕ݂ ܕܸܡܥܲܝ̈ܟܝ ܥܨܘܿܒ݂ܝ ܠܲܬ݂ܒ݂ܵܪܹܟ݂ܝ܀ ܦ. ܦܬ݂ܝܼܚܘܼ ܬܲܪܥܵܐ ܠܲܬ݂ܝܵܒ݂ܘܼܬ݂ܵܐ. ܩܘܿܫ ܒܲܬ݂ܘܵܬ݂ܵܐ ܟܲܕ݂ ܓܵܥ̇ܝܵܐ ܐܲܢ݇ܬ݁ܝ܀ ܨ. ܨܵܒܹ̇ܐ ܒܚܲܝܹ̈ܐ ܕܲܒ݂ܢܲܝ̈ܢܵܫܵܐ. ܚܘܼܣ ܥܲܠ ܢܲܦ̮ܫܵܗ̇ ܕܲܫܒ݂ܵܗ̇ ܒܝܼܫܵܐ܀ ܩ. ܩܲܒܸ݁ܠ ܒܵܥܘܼܬ݂ܝ ܘܚܘܼܣ ܥܲܠ ܚܲܝܲܝ̈. ܠܵܟ݂ ܩܵܪܹܐ ܐ݇ܢܵܐ ܪܵܚܹ̇ܡ ܐ݇ܢܵܫ̈ܵܐ܀ ܪ. ܪ̈ܲܚܡܲܝܟ ܢܦܝܼܣܘܼܢ ܥܲܠ ܚܲܝܵܒ݂ܘܼܬ݂ܝ. ܘܲܢܦܲܨܘܿܢܵܢܝ ܡ݂ܢ ܫܘܼܢܵܩܵܐ܀ ܫ. ܫܵܒܹ̇ܩ ܚܵܘܒܹ̈ܐ ܘܚܵܐܹ̇ܢ ܡܲܓܵܢ. ܫܒ݂ܘܿܩ ܣܘܼܪ̈ܚܵܢܲܝ ܒܕ݂ܝܼܢܵܟ݂ ܟܹܐܢܵܐ܀ ܬ. ܬܵܘܕ݁ܝܼ ܠܪ̈ܲܚܡܲܝܟ ܒܲܪ ܚܲܢܵܢܵܐ. ܡ݂ܢ ܚܲܛܵܝܹ̈ܐ ܕܐܲܦ̣ܢܝܼ ܚܢܵܢܵܟ݂܀ ܬ. ܬܵܘܕ݁ܝܼܬ݂ܵܐ ܠܵܟ݂ ܘܠܲܐܒ݂ܵܐ ܕܫܲܠ̣ܚܵܟ݂. ܘܲܠܪܘܼܚܩܘܼܕ݂ܫܵܐ ܦܵܪܲܩܠܹܝܛܵܐ܀ ܕܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ ܡ݂ܢ ܣܵ̇ܓ݂ܘܿܕ݂ܵܘ̈ܗܝ. ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܒܩܲܕ݂ܡܵܝܹ̈ܐ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܕܝܵܘܡܵܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ : ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ݂܀ ܗܵܘܿ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ̇ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡܠܲܟܼ ܛܵܒ݂ܵܬ݂ܵܐ ܠܪܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕ݁ܬܹܗ: ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܠܵܐ ܨܵܒܹܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ:  ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܼܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܗܘ̤ܝܘ̣ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܪ̈ܝܵܬ݂ܵܐ:  ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ ؛ ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܝܼܣ:  ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܝܼܣ : ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈  ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܵܘܿ ܕܲܒ݂ܪܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚܘ ܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂:  ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨ̈ܠܵܘܵܬ݂ܵܐ ܕܩܵܠܹ̈ܐ ܫܲܗܪܵܐ
ܨܠܘܿܬ݂ܵܐܕܩܲܕ݂ܡܵܝܵܐ. ܢܸܫܦܪܘܼܢ ܠܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܩܵܠܹ̈ܐ ܕܗܘܼܠܵܠܲܝ̈ܢ ܘܢܸܥ̈ܡܵܬ݂ܵܐ ܕܲܙܡܝܼܖ̈ܵܬܲܢ: ܘܩܲܒܸ݁ܠܡ̣ܢܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܦܹܐܖܹ̈ܐ ܡܠܝܼܠܹ̈ܐ ܕܣܸܦ̈ܘܵܬܲܢ: ܕܲܒ݂ܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܩܲܪܒ݂ܝܼܢܲܢ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܡܲܩܦܝܼܢ ܘܲܡܫܲܪܹܝܢ ܡܲܪܡܝܼܬ݂ܵܐ ܩܲܕ݂ܡܵܝܬܵܐ ܕܗܘܼܠܵܠܵܐ ܐܲܝܢܵܐ ܕܗܘ̤ܟܠܵܗ̇ ܒܩܵܠܵܗ̇: ܘܲܡܫܲܒ݁ܚܝܼܢ ܘܲܡܥܵܠܡܝܼܢ ܘܐܵܡܪܝܼܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܪܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܐܲܚܖ̈ܵܝܹܐ ܡܲܪܡܝܼܬ̣ܵܐ * ܗܵܝܕܹ̇ܝܢ ܫܲܒܲܚ̣ܘ ܡܘܼܫܹܐ ܘܲܒ̣ܢܲܝ̈ ܝܼܣܪܵܝܹܠ. ܡܲܙܡܘܖ̈ܐ ܩܢܐ‐ܩܢܒ.
ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܲܬ݂ܪܲܝܵܢܵܐ. ܐܲܫܘܵܐ ܠܲܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܕܥܲܡ ܥܝܼܖܹ̈ܐ ܘܟܸܢܫܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ: ܒܩܵܠܹ̈ܐ ܕܲܡܠܹܝܢ ܬܵܘܕ݁ܝܼܬ݂ܵܐ: ܢܸܙܡܲܪ ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ. ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܲܡܫܲܡܫܝܼܢ ܟܠܵܗ̇ ܡܲܪܡܝܼܬ݂ܵܐ ܬܪܲܝܵܢܝܼܬ݂ܵܐ ܕܗܘܼܠܵܠܵܐ ܕܐܲܝܢܵܐ ܕܗܘ̤ ܒܩܵܠܵܗ̇. ܐܲܝܟ݂ ܩܲܕ݂ܡܵܝܬܵܐ.
ܘܐܵܡܪܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܩܲܕܡܵܝܹ̈ܐ ܡܲܪܡܝܼܬ̣ܵܐ. ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܬܸܫܒ̇ܘܿܚܬܵܐ ܚܕܲܬܵܐ. ܚܙܝܼ: ܨܘܼܬ̣ ܫܡܲܝܵܐ ܘܐܹܡܲܠܸܠ. ܡܙܡܘܪܐ ܩܢܓ
ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܵܐ: ܫܘܼܒ݂ܚܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ: ܘܗܘܼܠܵܠܹ̈ܐ ܕܠܵܐ ܫܲܠܘܵܐ: ܘܬܸܫܒ̈ܚܵܬ݂ܵܐ ܕܠܵܐ ܣܵܟ݂ܵܐ: ܘܩܵܠܹ̈ܐ ܕܲܡܠܹܝܢ ܬܵܘܕ݁ܝܼܬ݂ܵܐ: ܚܲܝܵܒ݂ ܟܠܹܗ ܟܝܵܢܵܐ ܕܲܡܠܝܼ̈ܠܹܐ ܕܲܒ݂ܪܲܝܬ݁: ܠܡܲܣܵܩܘܼ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܲܡܫܲܡܫܝܼܢ ܟܠܵܗ̇ ܡܲܪܡܝܼܬ݂ܵܐ ܬܠܝܼܬ݂ܵܝܬܵܐ ܕܗܘܼܠܵܠܵܐ ܐܲܝܢܵܐ ܕܗܘ̤ ܒܩܵܠܵܗ̇. ܐܲܝܟ݂ ܩܲܕ݂ܡܵܝܬܵܐ.
ܘܐܵܡܪܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܩܲܕܡܵܝܹ̈ܐ ܡܲܪܡܝܼܬ̣ܵܐ. ܘܐܵܦ ܐܸܢܵܐ ܐܲܛܸܢ ܐܸܢܘܿܢ ܒܠܵܐ ܥܲܡܝ. ܡܙܡܘܪܐ ܩܢܕ
 ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ. ܡ̣ܢ ܦܘܼܡܹ̈ܐ ܡܵܪܝ ܡܠܝܼܠܹ̈ܐ ܕܲܒ݂ܪܲܝܬ݁: ܘܡ̣ܢ ܠܸܫܵܢܹ̈ܐ ܡܫܲܒ݁ܚܵܢܹ̈ܐ ܕܪܲܟܸ݁ܒ݂ܬ݁: ܘܡ̣ܢ ܟܸܢܫܹ̈ܐ ܟܠܗܘܿܢ ܕܥܸܠܵܝܹ̈ܐ ܘܲܕ݂ܬܲܚܬܵܝܹ̈ܐ: ܫܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܲܕ݂ܡܵܪܘܼܬ݂ܵܟ݂ ܢܸܣܬ݁ܓܸܕ݂ ܘܢܸܫܬܲܒܲܚ ܘܢܸܬ݂ܝܲܩܲܪ ܘܢܸܬ݂ܪܲܡܪܲܡ ܘܢܸܬ݁ܬܵܘܕܸ݁ܐ ܘܢܸܬ݂ܒܲܪܲܟ݂ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܪܝܼܢ ܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ ܘܬܵܢܹܝܢ ܘܲܡܫܲܒܚܝܼܢ ܒܵܗ̇ ܘܡܲܩܦܝܼܢ:
ܕܠܸܠܝܵܐ. ܒܡܵܪܝܵܐ ܥܸܕ݁ܬܵܟ݂. ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܦܘܼܪܩܵܢܝ: ܘܲܗܘܹ̇ܝܬ݂ ܐܲܝܟ݂ ܡܵܐܢܵܐ ܐܲܒ݁ܝܼܕ݂ܵܐ. ܛܸܒ݂ܥܸܬ݂ ܒܗܵܘܬ݂ܵܐ ܥܲܡܝܼܩܬܵܐ ܘܠܲܝܬ݁ ܒܵܗ̇ ܒܹܝܬ݂ ܩܝܵܡܵܐ. ܕܨܘܼܪܬܵܐ. ܐܲܝܟ݂ ܡܵܐܢܵܐ ܕܠܲܝܬ݁ ܒܹܗ ܚܲܫܚܘܼ. ܡܵܪܝܵܐ ܒܣܘܿܓ݂ܵܐܐ ܕܚܵܘܒܹ̈ܐ ܘܕܲܚܛܵܗܹ̈ܐ ܐܸܬ݁ܛܲܒ݁ܥܹܬ݂. ܘܲܗܘܹ̇ܝܬ݂ ܡܵܐܢܵܐ ܐܲܒ݁ܝܼܕ݂ܵܐ. ܘܠܲܝܬ݁ ܠܝܼ ܡܸܟܹ݁ܝܠ ܫܘܼܒ݂ܩܵܢܵܐ ܘܓܲܠܝܘܼܬ݂ ܐܲܦܹ̈ܐ. ܕܲܒ݂ܣܵܢ̈ܝܵܬ݂ܵܐ ܝܵܘܡܲܝ̈ ܓܲܡ̇ܪܹܬ݂. ܘܐܵܘܦ̣ܝܼܘ ܚܲܝܲܝ̈ ܒܕ݂ܵܘܘܿܢܹ̈ܐ. ܕܗܵܐ ܙܲܒ݂ܢܵܐ ܡܙܲܠܗܲܙ ܘܡܵܘܬܵܐ ܡܣܲܪܗܸܒ݂ ܘܓܸܠܝܵܢܵܟ݂ ܐܲܬ݂ܘ̱ܗܲܢ̣ܝ. ܩܕ݂ܵܡܲܝܟ ܕܸܡܥܲܝ̈ ܐܵܫܹ̇ܕ݂ ܐ݇ܢܵܐ. ܘܡܸܬ݂ܟܲܫܲܦ ܐ݇ܢܵܐ ܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܕܬܸܚܕܸ݁ܐ ܒܲܬ݂ܝܵܒ݂ܘܼܬ݂ܝ ܘܐܸܬ݂ܐܲܣܸܐ ܒܪ̈ܲܚܡܲܝܟ ܘܐܸܚܸܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂.
ܘܫܲܒܲܚ. ܡܵܪܝܵܐ ܒܣܘܿܓ݂ܵܐܐ ܕܚܵܘܒܹ̈ܐ ܘܕܲܚܛܵܗܹ̈ܐ ܐܸܬ݁ܛܲܒ݁ܥܹܬ݂. ܘܲܗܘܹ̇ܝܬ݂ ܡܵܐܢܵܐ ܐܲܒ݁ܝܼܕ݂ܵܐ. ܘܠܲܝܬ݁ ܠܝܼ ܡܸܟܹ݁ܝܠ ܫܘܼܒ݂ܩܵܢܵܐ ܘܓܲܠܝܘܼܬ݂ ܐܲܦܹ̈ܐ. ܕܲܒ݂ܣܵܢ̈ܝܵܬ݂ܵܐ ܝܵܘܡܲܝ̈ ܓܲܡ̇ܪܹܬ݂. ܘܐܵܘܦ̣ܝܼܘ ܚܲܝܲܝ̈ ܒܕ݂ܵܘܘܿܢܹ̈ܐ. ܕܗܵܐ ܙܲܒ݂ܢܵܐ ܡܙܲܠܗܲܙ ܘܡܵܘܬܵܐ ܡܣܲܪܗܸܒ݂ ܘܓܸܠܝܵܢܵܟ݂ ܐܲܬ݂ܘ̱ܗܲܢ̣ܝ. ܩܕ݂ܵܡܲܝܟ ܕܸܡܥܲܝ̈ ܐܵܫܹ̇ܕ݂ ܐ݇ܢܵܐ. ܘܡܸܬ݂ܟܲܫܲܦ ܐ݇ܢܵܐ ܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܕܬܸܚܕܸ݁ܐ ܒܲܬ݂ܝܵܒ݂ܘܼܬ݂ܝ ܘܐܸܬ݂ܐܲܣܸܐ ܒܪ̈ܲܚܡܲܝܟ ܘܐܸܚܸܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂.
ܥܵܠܲܡ. ܡܫܝܼܚܵܐ ܠܵܐ ܬܲܗܡܸܐ ܡܸܢܲܢ. ܘܠܵܐ ܬܲܪܚܸܩ ܡ݂ܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܒ݂ܵܟ݂ܘܼ ܡܵܪܝ ܐܸܬ݂ܓܵܘܲܣܢܲܢ. ܕܲܒܲܪܲܝܢ ܒܐܘܼܪܚܵܟ݂ ܕܚܲܝܹ̈ܐ ܕܲܢܙܲܡܲܪ ܠܵܟ݂ ܟܠܲܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ. ܢܹܐܡܲܪ. ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܐܸܡܹܗ ܕܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ: ܐܲܦܝܼܣܝ ܥܲܡܲܢ ܠܲܡܫܝܼܚܵܐ ܕܢܲܫܪܸܐ ܫܲܝܢܹܗ ܒܲܝܢܵܬܲܢ: ܘܲܢܢܲܛܲܪ ܠܲܢ ܡ̣ܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܒܵܚܵܐ. ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡ̣ܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡ̣ܢ ܟܠ ܠܸܫܵܢܝܼ̈ܢ: ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܘܗܸܕ݂ܪܵܐ ܘܪܘܼܡܪܵܡܵܐ ܡ̣ܢ ܟܠ ܒܸܪ̈ܝܵܢ: ܐܝܼܬ݂ܝܵܐ ܓܢܝܼܙܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܫܘܼܒܼܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟܼ ܡܵܪܝܵܐ.
 ܫܲܒܲܚ: ܘܥܵܠܲܡ. ܫܘܼܒܼܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟܼ ܡܵܪܝܵܐ. ܘܥܵܛ̇ܦܝܼܢ ܠܪܹܫܹܗ ܕܡܲܙܡܘܿܪܵܐ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܕܦܘܼܪܩܵܢܝ. * ܒܐܝܼܡܵܡܵܐ ܓܥܹܝ݁ܬ̣ ܘܲܒ̣ܠܸܠܝܵܐ ܩܕ݂ܵܡܲܝܟ ܫܘܼܒܼܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟܼ ܡܵܪܝܵܐ..
ܘܐܵܡ̇ܪܝܼܢ ܗܦܵܟ݂ܬܵܐ ܠܵܘ ܕܲܦܠܝܼܓ݂ ܐ݇ܢܵܐ ܥܲܠ ܥܘܼܕ݂ܪܵܢܵܟ݂ ܩܪܹܝ̇ܬ݂ܵܟ݂ ܗܵܫܵܐ: ܐܸܠܵܐ ܕܐܵܘܕܲܥ ܕܲܒ݂ܝܲܕ݂ ܚܲܝܠܵܟ݂ ܡܸܬ݂ܢܲܨܲܚ ܐ݇ܢܵܐ܀ ܡܸܛܠ ܗܵܢܵܐ ܒܢ̣ܵܐ ܠܹܗ ܡܵܪܝܵܐ ܒܲܝܬܵܐ ܒܐܲܪܥܵܐ: ܕܐܲܝܢܵܐ ܕܨܵܒܹ̇ܐ ܕܢܸܚܙܸܐ ܠܡܵܪܝܵܐ ܢܹܐܬܹܐ ܠܒܲܝܬܹܗ.
ܬܸܫܒܘܿܚܬܵܐ. ܘܲܒ݂ܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܕܹܝܢ ܐܵܡ̇ܪܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܗܵܕܹܐ ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܡܵܪܝ ܢܲܪܣܲܝ ܡܲܠܦܵܢܵܐ.
ܬܵܘܕ݁ܝܼ ܠܛܵܒ݂ܵܐ ܕܚܲܪܲܪ ܓܸܢܣܲܢ: ܡ̣ܢ ܥܲܒ݂ܕ݁ܘܼܬ݂ܵܐ ܕܒ݂ܝܼܫܵܐ ܘܡܵܘܬܵܐ܀ ܫܲܝܸܢ ܥܲܡܲܢ ܟܸܢܫܲܝ̈ ܪܵܘܡܵܐ: ܕܪܲܓ݁ܝܼܙܝܼܢ ܗ݇ܘ̣ܵܘ ܡܸܛܠ ܥܵܘ̣ܠܲܢ܀ ܒܪܝܼܟ݂ ܚܲܢܵܢܵܐ ܕܟܲܕ݂ ܠܵܐ ܒܥܲܝܢ̣ܵܝܗܝ: ܢ̣ܦܲܩ ܒܲܒ݂ܥܵܬܲܢ ܘܲܚ̣ܕ݂ܝܼ ܒܚܲܝܲܝ̈ܢ܀ ܘܨ̣ܵܪ ܕܸܡ̈ܘܵܬ݂ܵܐ ܕܐܲܒ݁ܝܼܕ݂ܘܼܬܲܢ: ܘܲܕ݂ܦܘܼܢܵܝܲܢ ܒܥܸܪܒܵܐ ܕܲܛ̣ܥܵܐ܀ ܝܵܪܬܵܐ ܘܲܒ݂ܪܵܐ ܩ̣ܪܵܝܗܝ ܠܲܟ݂ܝܵܢܲܢ: ܕܲܛ̣ܥܵܐ ܘܲܦܢ̣ܵܐ ܘܡ̣ܝܼܬ݂ ܘܐܸܬ݂ܢܲܚܲܡ܀ ܘܚܲܕ݁ܝܼ ܠܟܸܢܫܹ̈ܐ ܖ̈ܘܼܚܵܢܵܝܹܐ: ܒܲܬ݂ܝܵܒ݂ܘܼܬܲܢ ܘܲܒ݂ܢܘܼܚܵܡܲܢ܀ ܠܵܐ ܡܸܬ݂ܡܲܠܲܠ ܚܘܼܒܵܐ ܪܲܒܵܐ: ܕܚ̣ܵܘܝܼ ܨܹܐܕܲܝܢ ܪܵܚܹ̇ܡ ܓܸܢܣܲܢ܀ ܕܡܸܢܹܗ ܕܓܸܢܣܲܢ ܥ̣ܒܲܕ݂ ܡܸܨܥܵܝܵܐ: ܘܪܲܥ̣ܝܼ ܠܥܵܠܡܵܐ ܥܲܡ ܪܲܒܘܼܬܹܗ܀ ܪܲܒܵܐ ܗ݇ܝ̣ ܡܸܢܲܢ ܘܡ̣ܢ ܟܠ ܒܸܖ̈ܝܵܢ: ܚܕܲܬܵܐ ܕܲܣ̣ܥܲܪ ܨܹܝܕ݂ ܐ݇ܢܵܫܘܼܬܲܢ܀ ܕܥܲܒ݂ܕܹ݁ܗ ܠܦܲܓ݂ܪܲܢ ܗܲܝܟܲܠ ܩܘܼܕ݂ܫܵܐ: ܕܲܢܡܲܠܸܐ ܒܹܗ ܣܸܓ݂ܕܲܬ݂ ܟܠܵܐ܀ ܬܵܘ ܐܲܖ̈ܥܵܢܹܐ ܘܲܫܡܲܝܵܢܹ̈ܐ: ܬܗܲܪܘ ܘܐܸܬ݁ܕܲܡܲܪܘ ܒܪܲܒ݁ܘܼܬ݂ ܕܲܪܓ݂ܵܐ܀ ܕܲܡ̣ܛܵܐ ܓܸܢܣܲܢ ܠܪܵܘܡܵܐ ܪܲܒܵܐ: ܕܐܲܠܵܗܘܼܬ݂ܵܐ ܕܠܵܐ ܡܸܬ݁ܕܲܪܟܵܐ܀ ܫܡܲܝܵܐ ܘܐܲܪܥܵܐ ܘܟ݂ܠܡܵܐ ܕܲܒ݂ܗܘܿܢ: ܢܵܘܕ݁ܘܿܢ ܥܲܡܲܢ ܠܡܵܘܪܸܒ݂ ܓܸܢܣܲܢ܀ ܕܚܲܕܸ݁ܬ݂ ܨܲܠܡܲܢ ܘܲܥ̣ܛܵܐ ܥܵܘ̣ܠܲܢ: ܘܲܩ̣ܪܵܢ ܒܲܫܡܹܗ ܘܫܲܥܒܸ݁ܕ݂ ܠܲܢ ܟܠ܀ ܫܵܘܹ̇ܐ ܠܫܘܼܒ݂ܚܵܐ ܡ̣ܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܗ̇ܘ ܕܐܲܪܝܼܡܲܢ ܠܥܸܠ ܡ̣ܢ ܟܠܵܐ܀ ܘܢܹܐܡܲܪ ܟܠܲܢ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ: ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀
ܘܩܵܝܡ̇ܝܼܢ: ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܐܲܝܟ݂ ܕܲܒ݂ܚܘܼܕ݂ܪܵܐ ܐܵܘ ܒܓܲܙܵܐ. ܐܵܘ ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ. ܕܠܵܐ ܚܲܒ݂ܪ̈ܵܬ݂ܵܗ̇.
(ܘܪܵܡܹܐ ܡܫܲܡܫܵܢܵܐ ܐܘܼܪܵܪܵܐ: ܘܐܵܣܲܪ ܙܘܼܢܵܪܵܐ: ܘܩܵܐܹ̇ܡ ܩܕ݂ܵܡ ܫܟ݂ܝܼܢܬܵܐ ܡܛܲܟ݁ܣܵܐܝܼܬ݂) ܘܐܵܡܲܪ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ: ܐܲܒ݂ܵܐ ܕܖ̈ܲܚܡܹܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܐܲܒ݂ܵܐ ܕܖ̈ܲܚܡܹܐ ܘܐܲܠܵܗܵܐ ܕܟ݂ܠܹܗ ܒܘܼܝܵܐܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܦܵܪܘܿܩܲܢ ܘܝܲܨܘܼܦܬ݂ܵܢܲܢ ܘܲܡܦܲܪܢ̱ܣܵܢܵܐ ܕܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ ܘܩܘܼܝܵܡܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ ܘܲܕ݂ܟ݂ܠܗܹܝܢ ܥܹ̈ܕ݂ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܐܲܬ݂ܪܲܢ ܘܥܲܠ ܟܠ ܐܲܬ݂ܪ̈ܵܘܵܢ: ܘܥܲܠ ܐܲܝܠܹܝܢ ܕܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܥܵܡ̇ܪܝܼܢ ܒܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܘܼܙܵܓ݂ܵܘ̈ܗܝ ܕܐܵܐܲܪ ܘܟܲܗܝܼܢܘܼܬ݂ܵܐ ܕܫܲܢ݇ܬܵܐ ܘܥܲܠ̈ܠܵܬ݂ܵܐ ܕܦܹܐܪܹ̈ܐ ܘܥܲܠ ܬܘܼܩܵܢܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ1: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܲܠܟܹ̈ܐ ܐܲܚܝܼܕܲܝ̈ ܫܘܼܠܛܵܢܹܗ ܕܥܵܠܡܵܐ ܗܵܢܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܩܲܫܝܼܫܹ̈ܐ ܘܲܡܫܲܡܫܵܢܹ̈ܐ ܐܲܪ̈ܬܵܕ݁ܘܿܟ݁ܣܹܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ܗ̇ܘ ܕܥܲܬ݁ܝܼܪ ܒܪ̈ܲܚܡܵܘܗܝ ܘܲܫܦܝܼܥ ܒܲܚܢܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܡܸܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ ܐܝܼܬ݂ܵܘܗܝ ܘܲܠܕ݂ܵܪܕܵܪ̈ܝܼܢ ܩܲܝܵܡ ܫܘܼܠܛܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܟ݂ܝܵܢܹܗ ܐܝܼܬ݂ܵܘܗܝ ܛܵܒ݂ܵܐ ܘܝܵܗܘܿܒ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹܐ ܒܡܵܘܬܹ݁ܗ ܕܚܲܛܵܝܵܐ ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܸܢ ܥܵܘ̣ܠܹܗ ܘܢܸܚܸܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܲܒ݂ܝܲܠܕܹ݁ܗ ܐܵܘ ܕܲܒ݂ܕܸܢܚܹܗ ܐܵܘ ܕܲܒ݂ܨܵܘܡܹܗ ܐܵܘ ܕܲܒ݂ܡܲܥܲܠܬܹ݁ܗ ܐܵܘ ܕܒܲܩܝܵܡܬܹ݁ܗ ܐܵܘ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܐܵܘ ܕܲܒ݂ܡܲܚܲܬ݁ܬܹ݁ܗ ܐܵܘ ܕܒܲܨܠܝܼܒܹܗ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܐܲܦܨܲܚ ܠܲܫܡܲܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܹܗ ܗܘ̤ ܟܝܵܢܵܐܝܼܬ݂ ܐܝܼܬ݂ ܠܵܐ ܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܘܝܲܬ݁ܝܼܪܵܐܝܼܬ݂ ܕܐܲܝܠܹܝܢ ܕܲܡܗܲܝܡܢܝܼܢ ܒܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܣܓܵܐ ܥܲܡܲܢ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
