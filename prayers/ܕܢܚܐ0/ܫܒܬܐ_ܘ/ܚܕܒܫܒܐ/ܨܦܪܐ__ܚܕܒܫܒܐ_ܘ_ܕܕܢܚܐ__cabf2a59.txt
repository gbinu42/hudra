ܡܲܙܡܘܼܪܹ̈ܐ ܕܨܦܪܵܐ ܕܚܕܒܫܲܒܵܐ ܕܕܸܫܬܵܐ ܕܕܸܢܚܵܐ
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܡܵܪܝ ܫܲܠܝܼܛܵܐ ܕܪܹܫ ܥܲܝܢܵܐ. ܠܵܟ݂ ܟܝܵܢܵܐ ܟܲܣܝܵܐ ܘܲܓ݂ܢܝܼܙܵܐ ܚܕ݂ܵܢܵܝ ܒܐܝܼܬ݂ܘܼܬܹܗ. ܬܢܝܼ. ܕܒܲܩܢܘܿܡܹ̈ܐ ܬܠܵܬ݂ܵܐ ܡܸܣܬ݁ܓܸܕ݂ ܘܡܸܫܬܲܒܲܚ: ܘܕܲܒ݂ܝܘܿܪܕܢܵܢ ܢܲܗܪܵܐ ܚܵܘܝܼ ܦܘܼܪܫܵܢܵܐ ܕܕ݂ܝܼܠܵܝܵܬܹ̈ܗ ܒܩܵܠܹ̈ܐ ܕܡܸܨܛܲܒ݂ܝܵܢܘܼܬ݂ܵܐ: ܘܲܒ݂ܣܵܥܘܿܪܘܼܬ݂ܵܐ ܕܡܸܠܬ݂ܵܐ: ܘܲܒ݂ܡܲܓ݁ܢܵܢܘܼܬ݂ܵܐ ܕܪܘܼܚܵܐ ܕܥܲܠ ܪܹܫܹܗ: ܘܫܲܟܸ݁ܢ ܠܲܢ ܒܲܥܡܵܕܹܗ ܪܘܼܚܵܐ ܕܣܝܼܡܲܬ݂ ܒܢܲܝ̈ܵܐ: ܘܲܕ݂ܪܲܫ ܠܲܢ ܐܘܼܪܚܵܐ ܕܲܡܡܲܢܥܵܐ ܠܐܲܬ݂ܪܵܐ ܥܸܠܵܝܵܐ ܕܡܲܠܟ݁ܘܼܬܹܗ: ܘܠܵܐܘܵܢܹ̈ܐ ܕܲܫܡܵܘ̈ܗܝ ܡܫܲܒ݁ܚܹ̈ܐ: ܘܲܠܟܲܪ̈ܟܹܐ ܡܸܬ݂ܝܲܕ݂ܥܵܢܹ̈ܐ ܕܫܘܼܠܛܵܢܹܗ: ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ ܘܫܵܐ̇ܠܝܼܢܲܢ: ܩܲܒܸ݁ܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܬܸܥܘܿܠ ܠܲܩܕ݂ܵܡ ܫܟ݂ܝܼܢܬܵܐ ܕܐܝܼܩܵܪܵܟ݂: ܘܬܸܣܓ݁ܘܿܕ݂ ܩܕ݂ܵܡ ܟܘܼܪܣܝܵܐ ܕܪܲܒ݁ܘܼܬ݂ܵܟ݂: ܘܬܲܝܬܸܐ ܠܥܲܡܵܟ݂ ܟܠ ܝܘܼܬ݂ܪ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܘܲܬ݂ܫܲܟܸ݁ܢ ܠܲܢ ܟܠ ܡܵܘܗ̈ܒ݂ܵܢ ܘܟ݂ܠ ܥܘܼܕ݂ܪ̈ܵܢܝܼܢ: ܘܬܸܡܠܸܐ ܚܲܣܝܼܪ̈ܵܬܲܢ ܘܬܸܠܚܸܐ ܣܲܟ݂ܠܘ̈ܵܬܲܢ ܕܐܲܢ݇ܬ݁ܘܼ ܒܹܝܬ݂ ܓܵܘܣܵܐ ܕܐܲܝܠܹܝܢ ܕܲܬ݂ܟ݂ܝܼܠܝܼܢ ܥܠܲܝܟ* ܘܬܸܗܘܸܐ ܡܵܪܝ ܬܸܫܡܸܫܬܲܢ ܠܲܢܝܵܚܵܟ݂: ܘܲܨܠܘܿܬܲܢ ܘܫܲܗܪܲܢ ܠܲܦܝܵܣܵܟ݂: ܘܒ݂ܵܥܘܼܬܲܢ ܠܬܲܪܥܘܼܬ݂ܵܟ݂ ܘܬܲܟ݂ܫܲܦ̮ܬܲܢ ܠܐܝܼܩܵܪܵܟ݂: ܘܬܲܚܢܲܢܬܲܢ ܠܲܡܪܲܥܵܝܘܼܬ݂ܵܟ݂: ܘܪ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܢܸܗܘܘܿܢ ܠܚܘܼܣܵܝ ܚܵܘܒܹ̈ܐ ܕܥܲܡܵܟ݂: ܘܲܠܫܘܼܒ݂ܩܵܢ ܚܛܵܗܹ̈ܐ ܕܟ݂ܠܵܗ̇ ܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬ݂ܵܟ݂: ܕܲܓ݂ܒܲܝܬ݁ ܠܵܟ݂ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܪ̈ܲܚܡܲܝܟ: ܛܵܒ݂ܵܐ ܘܪܵܚܹ̇ܡ ܒܢܲܝ̈ܢܵܫܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܚܒܲܪܬ݂ܵܗ̇. ܠܵܟ݂ ܐܘܼܡܵܢܵܐ ܚܲܟ݁ܝܼܡܵܐ: ܕܲܠܨܲܠܡܵܐ ܕܡܢ ܩܕ݂ܝܼܡ ܫܒ݂ܝܼܚܵܐܝܼܬ݂ ܬܲܩܸܢ ܗ݇ܘܵܐ: ܚܙܵܐ ܣܵܣܵܐ ܕܥܵܘܠܵܐ ܐܸܟܲܠܬ݂ܹܗ: ܘܪܸܡܬ݂ܵܐ ܕܲܥܒܲܪ ܦܘܼܩܕܵܢܵܐ ܚܲܒܸ݁ܠܬ݂ܹܗ: ܘܬ݂ܵܘܠܥܵܐ ܕܲܚܛܝܼܬ݂ܵܐ ܐܵܘܒܕܲܬ݀ ܠܫܘܼܦܪܹܗ: ܘܲܗܦܲܟ݂ ܠܡܸܕ݂ܪܹܗ ܩܲܕ݂ܡܵܝܵܐ: ܒܘܼܟ݂ܪܵܐ ܚܲܕ݂ܬ݂ܵܐ ܕܲܒ݂ܢܘܼܗܪܹܗ ܡܸܬ݂ܓܵܘܙܠܵܢܵܐ ܪܘܼܚܵܐ ܚܲܕܸ݁ܬ݂ ܠܢܲܚܬܵܐ ܥܲܬ݁ܝܼܩܵܐ ܕܐܸܬ݁ܬܲܠܲܚ ܒܲܚܒ݂ܵܠܵܐ: ܘܲܥܛܲܦ ܨܲܒ݁ܥܹܗ ܒܡܲܝ̈ܵܐ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܘܐܲܣܩܹܗ ܡܢ ܬܲܡ̇ܢ ܫܒ݂ܝܼܚܵܐ ܘܡܲܦܪܓ݂ܵܐ ܕܥܵܗܹ̇ܢ ܠܬܲܟ݂ܣܝܼܬ݂ܵܐ ܕܡܲܠܟ݁ܘܼܬܹܗ: ܘܲܠܫܵܘܬܵܦܘܼܬ݂ܵܐ ܕܐܝܼܩܵܪܹܗ ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ ܘܲܡܦܝܼܣܝܼܢܲܢ ܘܫܵܐ̇ܠܝܼܢܲܢ: ܐܲܕ݂ܢܲܚ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܒܠܸܒܵܘ̈ܵܬܲܢ ܢܘܼܗܪܵܐ ܓܲܐܝܵܐ ܕܝܼܕܲܥܬ݂ܵܟ݂: ܘܢܲܒ݂ܪܸܫ ܢܲܦ̮ܫ̈ܵܬܲܢ ܒܪ̈ܲܚܡܲܝܟ: ܘܲܓ݂ܠܘܿܓ݂ ܡܢ ܐܲܦܲܝ̈ܢ ܘܹܐܠܵܐ ܚܹܫܘܿܟ݂ܬܵܐ ܕܲܩܪܝܼܡܵܐ ܥܲܠ ܡܲܕ݂ܥܲܢ: ܘܗܲܒ݂ܠܲܢ ܕܥܲܡ ܟܸܢܫܹ̈ܐ ܕܲܚܙܵܘ ܫܘܼܒ݂ܚܵܟ݂ ܥܲܠ ܝܘܿܪܕܢܵܢ: ܘܲܫܡܲܥܘ ܩܵܠܵܐ ܕܐܲܣܗܕ݂ ܥܲܠ ܒܪܘܼܬ݂ܵܟ݂: ܘܝܼܠܸܦ̮ܘ ܐ݇ܪ̈ܵܙܹܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂: ܢܣܲܟܸ݁ܐ ܠܡܵܘܠܵܕ݂ܵܐ ܬܠܝܼܬ݂ܵܝܵܐ ܕܡܢ ܩܲܒ݂ܪܵܐ: ܘܢܸܣܲܩ ܥܲܡܵܟ݂ ܒܫܘܼܒ݂ܚܵܐ ܠܲܫܡܲܝܵܐ ܗܵܘ̇ ܕܗܵܫܵܐ ܐܸܬ݂ܦܬܲܚ ܠܐܝܼܩܵܪܵܟ݂ ܝܵܘܡܵܢܵܐ: ܘܬܲܡܵܢ ܢܸܬ݂ܒܲܣܲܡ ܒܢܘܼܗܪܵܟ݂ ܗܵܘ̇ ܕܠܵܐ ܥܵܒܲܪ ܘܠܵܐ ܦܵܛܲܪ: ܘܠܵܐ ܡܲܕ݂ܪܸܟ݂ ܠܹܗ ܚܸܫܘܿܟ݂ܵܐ܀ ܒܵܪܘܿܝܹܗ ܕܢܘܼܗܪܵܐ ܒܛܲܝܒ݁ܘܼܬܹܗ: ܘܲܡܛܲܟ݁ܣܵܢܹܗ ܕܚܸܫܘܿܟ݂ܵܐ ܒܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܘܡܲܢܗܪܵܢܵܐ ܕܲܒ݂ܪܝܼܬ݂ܵܐ ܒܚܸܟ݂ܡܬܹܗ ܘܝܼܕܲܥܬܹܗ ܘܐܲܠܵܗܘܼܬܹܗ ܕܠܵܐ ܡܸܬ݁ܕܲܪܟܵܐ ܡܢ ܪ̈ܘܼܚܵܢܹܐ ܘܦܲܓ݂ܪ̈ܵܢܹܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܲܡܫܲܡܫܝܼܢ ܡܲܙܡܘܿܪܹ̈ܐ ܕܨܲܦܪܵܐ
ܘܲܡܫܲܪܹܝܢ ܒܩܵܠܵܐ. ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܟܠܵܗ̇ ܐܲܪܥܵܐ܀ ܘܐܵܡ̇ܪܝܼܢ ܒܪܵܫܝܼܬ݂ ܐܲܝܟ݂ ܕܲܒ݂ܚܘܼܕ݂ܪܵܐ ܐܸܢ ܐܝܼܬ݂. ܘܐܸܢ ܠܲܝܬ݁ ܒܪܵܫܝܼܬ݂: ܐܵܡ̇ܪܝܼܢ: ܝܵܗܹ̇ܒ݂ ܢܘܼܗܪܵܐ ܡܵܪܝܵܐ: ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܲܣܩܝܼܢܲܢ. ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܟܠܵܗ̇ ܐܲܪܥܵܐ܀ܘܡܲܩܦܝܼܢ ܦܠܘܿܚܘ ܠܡܵܪܝܵܐ ܒܚܲܕ݂ܘܼܬ݂ܵܐ.ܝܵܗܹܒ݂ ܢܘܼܗܪܵܐ ܡܵܪܝܵܐ: ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܲܣܩܝܼܢܲܢ* ܥܘܿܠܘ ܩܕ݂ܵܡܵܘܗܝ ܒܬܸܫܒ݁ܘܿܚܬܵܐ* ܕܲܥܘ ܕܗܘܝܘ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ*ܗܘܝܘ ܥܲܒ݂ܕܲܢ ܘܠܵܐ ܗ݇ܘܵܐ ܚܢܲܢ* ܥܲܡܹܗ ܚܢܲܢ ܘܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬܹܗ* ܥܘܿܠܘ ܒܬܲܪ̈ܥܵܘܗܝ ܒܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܲܠܕ݂ܵܪ̈ܵܘܗܝ ܒܫܘܼܒ݂ܚܵܐ* ܐܵܘܕܵܘ ܠܹܗ ܘܒܲܪܸܟ݂ܘ ܠܲܫܡܹܗ* ܡܸܛܠ ܕܛܵܒ݂ ܗ݇ܘܼ ܡܵܪܝܵܐ ܘܲܠܥܵܠܲܡ ܛܲܝܒ݁ܘܼܬܹܗ* ܘܲܠܕ݂ܵܪܕܵܪ̈ܝܼܢ ܗܲܝܡܵܢܘܼܬܹܗ܀ ܫܲܒܲܚ܀ ܥܵܠܲܡ܀ ܘܲܡܣܲܝܟ݂ܝܼܢ ܐܲܟ݂ܚ݇ܕ݂ܵܐ: ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܟܠܵܗ̇ ܐܲܪܥܵܐ. ܝܵܗܹܒ݂ ܢܘܼܗܪܵܐ ܡܵܪܝܵܐ ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܲܣܩܝܼܢܲܢ.
ܩܵܢܘܿܢܵܐ ܕܹܝܢ ܕܲܒ݂ܪܵܫܝܼܬ݂ ܗܵܟܲܢ ܡܸܬ݂ܐܡܲܪ: ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܟܠܵܗ̇ ܐܲܪܥܵܐ: ܒܪܵܫܝܼܬ݂ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ... ܥܕܲܡܵܐ ܕܫܵܠܹ̇ܡ ܟܠܹܗ ܬܲܪܥܵܐ. (ܘܐܸܢ ܐܝܼܬ݂ ܬܲܪܥܵܐ ܬܪܲܝܵܢܵܐ ܘܢܵܩܹ̇ܦ ܐܵܦ ܬܲܪܥܵܐ ܬܠܝܼܬ݂ܵܝܵܐ. ܬܲܪܥܵܐ ܬܪܲܝܵܢܵܐ ܕܲܒ݂ܪܵܫܝܼܬ݂: ܡܸܬ݂ܐܡܲܪ ܒܵܬܲܪ ܦܸܬ݂ܓ݂ܵܡܵܐ ܬܪܲܝܵܢܵܐ: ܗܵܢܵܘ ܕܹܝܢ: ܦܠܘܿܚܘ ܠܡܵܪܝܵܐ ܒܚܲܕ݂ܘܼܬ݂ܵܐ. ܘܬܲܪܥܵܐ ܬܠܝܼܬ݂ܵܝܵܐ ܕܲܒ݂ܪܵܫܝܼܬ݂. ܡܸܬ݂ܐܡܲܪ ܡܢ ܒܵܬܲܪ ܫܲܒܲܚ ܘܥܵܠܲܡ.(ܘܐܸܢ ܬܲܪܥܵܐ ܬܪܲܝܵܢܵܐ ܠܚܘܿܕ݂ ܐܝܼܬ݂: ܗܵܟ݂ܘܵܬ݂ ܒܵܬܲܪ ܫܲܒܲܚ. ܥܵܠܲܡ܀ ܟܲܕ݂ ܡܣܲܝܟ݂ܝܼܢ ܠܪܹܫ ܡܲܙܡܘܿܪܵܐ. ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܟܠܵܗ̇ ܐܲܪܥܵܐ܀ ܝܵܗܹ̇ܒ݂ ܢܘܼܗܪܵܐ ܡܵܪܝܵܐ ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܲܣܩܝܼܢܲܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܝܵܬܹܒ݂ ܒܣܸܬܵܪܹܗ: ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܕ݂ܥܹܐܕܹ̈ܐ. ܠܵܟ݂ ܡܵܪܝ ܫܘܼܒ݂ܚܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܲܣ̈ܩܵܢ: ܟܠܗܹܝܢ ܒܸܪ̈ܝܵܬ݂ܵܐ ܕܲܒ݂ܪܲܝܬ݁: ܕܐܲܢ݇ܬ݁ܘܼ ܠܗܹܝܢ ܢܘܼܗܪܵܐ ܕܲܫܪܵܪܵܐ ܒܲܠܚܘܿܕܲܝܟ: ܘܡܲܢܗܲܪ ܥܵܠܡܹ̈ܐ ܘܒܸܪ̈ܝܵܬ݂ܵܐ ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܪ̈ܲܚܡܵܘܗܝ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܒܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܕܹܝܢ ܘܲܒ݂ܕ݂ܘܼܟ݂ܪ̈ܵܢܹܐ ܐܵܡ̇ܪܝܼܢ: ܝܵܬܹܒ݂ ܒܣܸܬܵܪܹܗ ܕܲܡܪܲܝܡܵܐ. ܫܒ݂ܝܼܚ ܬܘܼܟ݂ܠܵܢܵܟ݂ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܝܵܬܹܒ݂ ܒܣܸܬܵܪܹܗ ܕܲܡܪܲܝܡܵܐ* ܘܲܒ݂ܛܸܠܵܠܹܗ ܕܐܲܠܵܗܵܐ ܡܸܫܬܲܒܲܚ* ܐܸܡܲܪ ܠܡܵܪܝܵܐ ܬܘܼܟ݂ܠܵܢܝ* ܒܹܝܬ݂ ܓܵܘܣܝ ܐܲܠܵܗܵܐ ܕܲܬ݂ܟ݂ܝܼܠ ܐ݇ܢܵܐ ܥܠܵܘܗܝ* ܕܗܘ ܢܦܲܨܹܝܟ݂ ܡܢ ܦܲܚܵܐ ܕܬ݂ܘܼܩܠܬ݂ܵܐ* ܘܡܢ ܡܲܡܠܠܵܐ ܕܲܣܪܝܼܩܘܼܬ݂ܵܐ* ܒܐܸܒ݂ܪ̈ܵܘܗܝ ܢܦܲܨܹܝܟ݂ ܘܲܬ݂ܚܘܿܬ݂ ܟܸܢܦܵܘ̈ܗܝ ܬܸܣܬܲܬܲܪ* ܙܲܝܢܵܐ ܢܸܚܕ݁ܪܵܟ݂ ܩܘܼܫܬܹ݁ܗ* ܠܵܐ ܬܸܕ݂ܚܲܠ ܡܢ ܕܸܚܠܬܹܗ ܕܠܸܠܝܵܐ* ܘܡܢ ܓܹܐܪܵܐ ܕܦܵܪܲܚ̇ ܒܐܝܼܡܵܡܵܐ* ܘܡܢ ܡܸܠܬ݂ܵܐ ܕܲܡܗܲܠܟ݂ܵܐ ܒܚܸܫܘܿܟ݂ܵܐ* ܘܡܢ ܪܘܼܚܵܐ ܕܫܵܝ̇ܕܵܐ ܒܛܲܗܪܵܐ* ܢܸܦܠܘܼܢ ܡܢ ܣܸܛܪܵܟ݂ ܐܲܠܦܹ̈ܐ* ܘܪܸ̈ܒܘܵܬ݂ܵܐ ܡܢ ܝܲܡܝܼܢܵܟ݂* ܘܲܠܘܵܬ݂ܵܟ݂ ܠܵܐ ܢܸܬ݂ܩܲܪܒ݂ܘܼܢ* ܐܸܠܵܐ ܒܥܲܝܢܲܝ̈ܟ ܒܲܠܚܘܿܕ݂ ܬܸܚܙܸܐ* ܒܦܘܼܪܥܵܢܗܘܿܢ ܕܪ̈ܲܫܝܼܥܹܐ ܬܸܚܙܸܐ* ܡܸܛܠ ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܝܵܐ ܬܘܼܟ݂ܠܵܢܝ* ܕܒܲܡܪܵܘܡܵܐ ܣܵܡܬ݁ ܡܲܥܡܪܵܟ݂* ܠܵܐ ܬܸܩܪܘܿܒ݂ ܠܵܟ݂ ܒܝܼܫܬܵܐ* ܘܲܡܚܘܿܬ݂ܵܐ ܠܵܐ ܬܸܬ݂ܩܲܪܲܒ݂ ܠܡܲܫܟ݁ܢܵܟ݂* ܡܸܛܠ ܕܲܠܡܲܠܲܐܟ݂ܵܘ̈ܗܝ ܢܦܲܩܸܕ݂ ܥܠܲܝܟ* ܕܲܢܢܲܛܪܘܼܢܵܟ݂ ܒܟ݂ܠܗܹܝܢ ܐܘܼܪ̈ܚܵܬ݂ܵܟ݂* ܘܥܲܠ ܕܪ̈ܵܥܲܝܗܘܿܢ ܢܸܫܩܠܘܼܢܵܟ݂* ܕܠܵܐ ܬܸܬ݁ܩܸܠ ܒܪܸܓ݂ܠܵܟ݂* ܥܲܠ ܓܵܪܣܵܐ ܘܥܲܠ ܚܲܪܡܵܢܵܐ ܬܸܕ݂ܪܘܿܟ݂* ܘܲܬ݁ܕ݂ܘܼܫ ܐܲܪܝܵܐ ܘܬܲܢܝܼܢܵܐ* ܡܸܛܠ ܕܠܝܼ ܒܥܵܐ ܐܹܦܲܨܹܝܘܗܝ ܘܐܹܥܲܫܢܝܼܘܗܝ* ܡܸܛܠ ܕܝܼܕܲܥ ܫܹܡܝ ܢܸܩܪܹܝܢܝ ܘܐܸܥܢܹܝܘܗܝ* ܥܲܡܹܗ ܐ݇ܢܵܐ ܒܐܘܼܠܨܵܢܵܐ* ܐܹܥܲܫܢܝܼܘܗܝ ܘܐܹܝܲܩܪܝܼܘܗܝ* ܢܘܼܓ݂ܪܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܐܹܣܲܒ݁ܥܝܼܘܗܝ ܘܐܹܚܵܘܹܝܘܗܝ ܦܘܼܪܩܵܢܝ܀ ܫܲܒܲܚ܀ ܥܵܠܲܡ܀ ܘܥܵܛ̇ܦܝܼܢ. ܝܵܬܹܒ݂ ܒܣܸܬܵܪܹܗ ܕܲܡܪܲܝܡܵܐ. ܫܒ݂ܝܼܚ ܬܘܼܟ݂ܠܵܢܵܟ݂ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ.
ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
[ܒܝܵܘܡܵܐ ܕܹܝܢ ܕܐܝܼܬ݂ ܒܹܗ ܒܪܵܫܝܼܬ݂... ܐܵܘ ܒܥܹܐܕܹ̈ܐ ܡܵܪ̈ܵܢܵܝܹܐ. ܐܵܡ̇ܪܝܼܢ ܝܵܬܹܒ݂ ܒܣܸܬܵܪܹܗ... ܒ: ܦܸܬ݂ܓ݂ܵܡܹ̈ܐ: ܘܡܲܩܦܝܼܢ: ܟܲܕ݂ ܬܲܩܢܹܗ ܒܵܪܘܿܝܵܐ ܠܗ̇ܘ ܢܘܼܗܪܵܐ ܬܗܲܪܘ ܒܹܗ ܡܲܠܲܐܟܹ̈ܐ: ܘܡܵܐ ܕܕ݂ܵܢܲܚ ܒܟ݂ܠ ܨܲܦܪ̈ܝܼܢ ܢܫܲܒ݁ܚܘܼܢܵܝܗܝ ܗܸܢܘܿܢ ܐܵܦ ܚܢܲܢ܀ ܘܬ݂ܵܢܹܝܢ ܠܟ݂ܠܹܗ ܡܲܙܡܘܿܪܵܐ ܒܙܵܘܓܹ̈ܐ. ܘܲܒ݂ܫܘܼܠܵܡܹܗ ܡܲܩܦܝܼܢ: ܟܲܕ݂ ܬܲܩܢܹܗ ܒܵܪܘܿܝܵܐ...܀ ܘܟܸܢ. ܫܲܒܲܚ. ܥܵܠܲܡ. ܘܡܲܩܦܝܼܢ ܟܲܕ݂ ܒܪܲܝܬܵܝܗܝ ܠܗܵܘ̇ ܢܘܼܗܪܵܐ ܐܲܫܝܼܕ݂ܵܐ: ܫܲܒ݁ܚܘܼܟ݂ ܪ̈ܘܼܚܵܢܹܐ: ܘܐܸܬ݂ܡܲܕܲܥܘ ܕܗܵܘ̇ ܕܲܒ݂ܪܵܝܗܝ ܠܗܵܘ̇ ܢܘܼܗܪܵܐ ܗܘܝܘܒܪܵܐ ܐܸܢܘܿܢ܀ ܘܬ݂ܵܢܹܝܢ ܠܪܹܫ ܡܲܙܡܘܿܪܵܐ: ܝܵܬܹܒ݂ ܒܣܸܬܵܪܹܗ...܀ ܒ: ܦܸܬ݂ܓ݂ܵܡܹ̈ܐ ܘܡܲܩܦܝܼܢ: ܡܢ ܡܲܠܲܐܟܹ̈ܐ ܡܸܫܬܲܒܲܚ ܐܲܢ݇ܬ݁: ܘܡܢ ܒܢܲܝ̈ܢܵܫܵܐ ܡܸܬ݂ܩܲܠܲܣ ܐܲܢ݇ܬ݁: ܘܟ݂ܠܗܘܿܢ ܐܲܟܲܚ݇ܕ݂ ܫܲܘܝܵܐܝܼܬ݂ ܩܵܥܹܝ̇ܢ ܘܐܵܡ̇ܪܝܼܢ: ܒܪܝܼܟ݂ܘܼ ܣܘܼܒܵܪܵܟ݂: (ܐܵܘ ܝܲܠܕܵܟ݂: ܐܵܘ ܥܡܵܕܵܟ݂: ܐܵܘ ܨܵܘܡܵܟ݂: ܐܵܘ ܒܪܝܼܟ݂ܵܐ ܗ݇ܝ ܩܝܵܡܬܵܟ݂: ܐܵܘܒܪܝܼܟ݂ܘܼ ܣܘܼܠܵܩܵܟ݂: ܐܵܘܒܪܝܼܟ݂ܵܐ ܗ݇ܝ ܡܲܚܲܬ݁ܬܹ݁ܗ ܕܪܘܼܚܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ:(ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ.]
ܨܠܘܿܬ݂ܵܐ. ܫܒ݂ܝܼܚ ܡܵܪܝ ܬܘܼܟ݂ܠܵܢܵܐ ܪܲܒܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܪܵܡ ܘܲܡܥܲܠܲܝ ܒܹܝܬ݂ ܓܵܘܣܵܐ ܬܡܝܼܗܵܐ ܕܡܵܪܘܼܬ݂ܵܟ݂: ܘܠܵܐ ܒܵܗ̇ܬ݁ܝܼܢ ܟܠ ܕܲܡܣܲܒ݁ܪܝܼܢ ܒܵܟ݂: ܘܲܬ݂ܟ݂ܝܼܠܝܼܢ ܥܠܲܝܟ ܘܩܵܪܹܝܢ ܠܲܫܡܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܵܟ݂ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܒܲܪܸܟ݂ܝ ܢܲܦ̮ܫܝ ܠܡܵܪܝܵܐ. ܠܵܟ݂ ܝܵܐܹ̇ܐ ܫܘܼܒ݂ܚܵܐ ܐܲܠܵܗܵܐ. ܒܲܪܸܟ݂ܝ ܢܲܦ̮ܫܝ ܠܡܵܪܝܵܐ* ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܝܼܪܸܒ݂ܬ݁ ܛܵܒ݂* ܙܝܼܘܵܐ ܘܗܸܕ݂ܪܵܐ ܠܒܸܫܬ݁* ܘܐܸܬ݂ܟܲܣܝܼܬ݁ ܢܘܼܗܪܵܐ ܐܲܝܟ݂ ܡܲܪܛܘܼܛܵܐ* ܡܬܲܚܬ݁ ܫܡܲܝܵܐ ܐܲܝܟ݂ ܝܵܪܝܼܥܬܵܐ* ܥܒܲܕ݂ ܒܡܲܝ̈ܵܐ ܡܸܕ݂ܝܵܪ̈ܵܘܗܝ* ܘܣܵܡ ܥܲܠ ܥܢܵܢܹ̈ܐ ܡܲܪܟܲܒ݂ܬ݂ܹܗ* ܘܲܡܗܲܠܸܟ݂ ܥܲܠ ܟܸܢܦܹ̈ܐ ܕܪܘܼܚܵܐ* ܥܒܲܕ݂ ܡܲܠܲܐܟ݂ܵܘ̈ܗܝ ܪܘܼܚ* ܘܲܡܫܲܡܫܵܢܵܘ̈ܗܝ ܢܘܼܪ ܝܵܩ̇ܕܵܐ* ܐܲܬ݂ܩܸܢ ܐܲܪܥܵܐ ܥܲܠ ܡܲܬ݂ܩܢܹܝ̈ܗ̇* ܕܠܵܐ ܬܙܘܼܥ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ* ܬܗܘܿܡܵܐ ܐܲܝܟ݂ ܡܲܪܛܘܼܛܵܐ ܟܲܣܝܼܬܵܝܗܝ* ܘܥܲܠ ܛܘܼܪܹ̈ܐ ܩܵܡܘ ܡܲܝ̈ܵܐ* ܡܢ ܟܐܵܬ݂ܵܟ݂ ܥܵܪܩ̇ܝܼܢ* ܘܡܢ ܩܵܠܵܐ ܕܪ̈ܲܥܡܲܝܟ ܡܸܬ݂ܪܲܗܒ݁ܝܼܢ* ܣܠܸܩܘ ܛܘܼܪܹ̈ܐ ܘܲܢܚܸܬ݂ܘ ܦܲܩ̈ܥܵܬ݂ܵܐ* ܒܐܲܬ݂ܪܵܐ ܕܬܲܩܸܢܬ݁ ܠܗܘܿܢ* ܬܚܘܼܡܵܐ ܣܵܡܬ݁ ܠܗܘܿܢ* ܕܠܵܐ ܢܸܥܒ݁ܪܘܼܢ ܘܲܢܟܲܣܘܿܢܵܗ̇ ܠܐܲܪܥܵܐ* ܫܲܕܲܪܬ݁ ܡܲܒܘܼܥܹ̈ܐ ܒܢܲܚ̈ܠܹܐ* ܘܒܲܝܢܵܬ݂ ܛܘܼܪܹ̈ܐ ܡܗܲܠܟ݂ܝܼܢ* ܘܡܲܫܩܹܝܢ ܟܠܵܗ̇ ܚܲܝܘܬ݂ܵܐ ܕܕܲܒ݂ܪܵܐ* ܘܣܵܒ݂ܥ̇ܝܼܢ ܥܪ̈ܵܕܹܐ ܕܲܨܗܹܝܢ* ܦܵܪܲܚܬ݂ܵܐ ܕܲܫܡܲܝܵܐ ܥܠܲܝܗܘܿܢ ܫܵܟ݂ܢܵܐ* ܘܡܢ ܒܲܝܢܵܬ݂ ܛܘܼܪܹ̈ܐ ܝܵܗ̇ܒ݁ܝܼܢ ܩܵܠܵܐ* ܡܲܫܩܸܐ ܛܘܼܪܹ̈ܐ ܡܢ ܡܸܕ݂ܝܵܪ̈ܵܘܗܝ* ܘܡܢ ܦܹܐܪ̈ܲܝ ܥܒ݂ܵܕܲܝ̈ܟ ܬܸܣܒܲܥ ܐܲܪܥܵܐ* ܡܵܘܥܸܐ ܥܡܝܼܪܵܐ ܠܲܒ݂ܥܝܼܪܵܐ* ܘܥܸܣܒܵܐ ܠܦܘܼܠܚܵܢܹܗ ܕܒܲܪܢܵܫܵܐ* ܠܡܲܦܵܩܘܼ ܠܲܚܡܵܐ ܡܢ ܐܲܪܥܵܐ* ܚܲܡܪܵܐ ܡܚܲܕܸ݁ܐ ܠܸܒܹ݁ܗ ܕܒܲܪܢܵܫܵܐ* ܘܡܲܦܨܲܚ ܐܲܦܵܘ̈ܗܝ ܒܡܸܫܚܵܐ* ܠܲܚܡܵܐ ܣܵܡܹ̇ܟ݂ ܠܸܒܹ݁ܗ ܕܒܲܪܢܵܫܵܐ* ܢܸܣܒ݁ܥܘܼܢ ܐܝܼܠܵܢܵܘ̈ܗܝ ܕܡܵܪܝܵܐ. ܠܵܟ݂ ܝܵܐܹ̇ܐ ܫܘܼܒ݂ܚܵܐ ܐܲܠܵܗܵܐ܀
ܡܲܙܡܘܿܪܵܐ ܗܵܢܵܐ ܕܐܵܬܹܐ ܡܸܬ݂ܐܡܲܪ ܒܙܵܘܓܹ̈ܐ ܘܒܲܠܒ݂ܵܟ݂ܬܵܐ.
ܫܲܒܲܚܘ ܥܲܒ݂ܕܵܘ̈ܗܝ ܕܡܵܪܝܵܐ. ܬܢܝܼ. ܫܲܒܲܚܘ ܠܲܫܡܹܗ ܕܡܵܪܝܵܐ* ܠܵܟ݂ ܝܵܐܹ̇ܐ ܫܘܼܒ݂ܚܵܐ ܐܲܠܵܗܵܐ: ܕܐܲܢ݇ܬ݁ܘܼ ܒܵܪܘܿܝܹܗ ܕܢܘܼܗܪܵܐ. ܢܸܗܘܸܐ ܫܡܹܗ ܕܡܵܪܝܵܐ ܡܒܲܪܲܟ݂. ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ* ܡܢ ܡܲܕ݂ܢܵܚܵܘ̈ܗܝ ܕܫܸܡܫܵܐ ܘܲܥܕܲܡܵܐ ܠܡܲܥܪ̈ܵܒ݂ܵܘܗܝ. ܪܲܒ݁ܘܼ ܫܡܹܗ ܕܡܵܪܝܵܐ* ܘܪܵܡ ܥܲܠ ܟܠܗܘܿܢ ܥܲܡܡܹ̈ܐ ܡܵܪܝܵܐ. ܘܥܲܠ ܫܡܲܝܵܐ ܐܝܼܩܵܪܹܗ* ܡܲܢܘܼ ܐܲܝܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ. ܕܝܵܬܹܒ݂ ܒܪܵܘܡܵܐ ܘܚܵܙܹ̇ܐ ܒܥܘܼܡܩܵܐ* ܕܝܵܬܹܒ݂ ܒܪܵܘܡܵܐ ܘܚܵܙܹ̇ܐ ܒܥܘܼܡܩܵܐ. ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ* ܡܪܝܼܡ ܡܢ ܩܸܩܠܬ݂ܵܐ ܠܒ݂ܵܝܫܵܐ. ܕܢܵܘܬ݁ܒ݂ܝܼܘܗܝ ܥܲܡ ܪܵܘܪ̈ܒ݂ܵܢܹܐ ܕܥܲܡܵܐ* ܡܵܘܬܸ݁ܒ݂ ܥܩܲܪܬ݂ܵܐ ܒܒܲܝܬܵܐ. ܘܐܸܡܵܐ ܕܲܒ݂ܢܲܝ̈ܵܐ ܟܲܕ݂ ܚܵܕ݂ܝܵܐ*ܫܲܒܲܚ. ܥܵܠܲܡ܀ ܫܲܒܲܚܘ ܥܲܒ݂ܕܵܘ̈ܗܝ ܕܡܵܪܝܵܐ. ܬܢܝܼ. ܫܲܒܲܚܘ ܠܲܫܡܹܗ ܕܡܵܪܝܵܐ. ܠܵܟ݂ ܝܵܐܹ̇ܐ ܫܘܼܒ݂ܚܵܐ ܐܲܠܵܗܵܐ: ܕܐܲܢ݇ܬ݁ܘܼ ܒܵܪܘܿܝܹܗ ܕܢܘܼܗܪܵܐ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܕ݁ܕ݂ܘܼܟ݂ܪ̈ܵܢܹܐ. ܐܲܢ݇ܬ݁ܘܼ ܡܵܪܝ ܒܵܪܘܿܝܹܗ ܕܢܘܼܗܪܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܲܡܛܲܟ݁ܣܵܢܹܗ ܕܚܸܫܘܿܟ݂ܵܐ ܒܚܸܟ݂ܡܬ݂ܵܟ݂: ܘܡܲܢܗܪܵܢܵܐ ܕܲܒ݂ܪܝܼܬ݂ܵܐ ܒܢܘܼܗܪܵܟ݂ ܫܒ݂ܝܼܚܵܐ: ܘܠܵܟ݂ ܡܵܪܝ ܝܵܐܹ̇ܐ ܫܘܼܒ݂ܚܵܐ ܐܲܡܝܼܢܵܐ ܕܠܵܐ ܫܲܠܘܵܐ: ܘܠܲܫܡܵܟ݂ ܡܸܬ݁ܬ݁ܚܝܼܒ݂ܵܐ ܬܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡ̇ܪܝܼܢ. ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܘܓܲܐܝܘܼܬ݂ܵܐ ܠܒܸܫ. ܣܵܓ݂ܕܝܼ̇ܢܲܢ ܠܵܟ݂ ܐܝܼܬ݂ܝܵܐ ܕܡܸܢ ܡܬ݂ܘܿܡ* ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܘܓܲܐܝܘܼܬ݂ܵܐ ܠܒܸܫ.ܠܒܸܫ ܡܵܪܝܵܐ ܥܘܼܫܢܵܐ ܘܐܸܬ݂ܥܲܫܲܢ* ܘܐܲܬ݂ܩ̄ܢܵܗ̇ ܠܬܹܒܹܝܠ ܕܠܵܐ ܬܙܘܼܥ* ܡܲܬ݂ܩܲܢܘܼ ܟܘܼܪܣܝܵܟ݂ ܡܢ ܩܕ݂ܝܼܡ* ܘܡܢ ܥܵܠܲܡ ܐܲܢ݇ܬ݁ܘܼ*ܐܸܬ݁ܬ݁ܪܝܼܡܘ ܢܲܗܪ̈ܵܘܵܬ݂ܵܐ ܡܵܪܝܵܐ* ܘܐܲܪܝܼܡܘ ܢܲܗܪ̈ܵܘܵܬ݂ܵܐ ܩܵܠܗܘܿܢ* ܐܸܬ݁ܬ݁ܪܝܼܡܘ ܢܲܗܪ̈ܵܘܵܬ݂ܵܐ ܒܕܲܟ݂ܝܘܼܬ݂ܵܐ* ܡܢ ܩܵܠܵܐ ܕܡܲܝ̈ܵܐ ܣܲܓ݁ܝܼ̈ܐܹܐ* ܥܲܫܝܼܢܝܼܢ ܡܲܚܫܘܼ̈ܠܵܘܗܝ ܕܝܲܡܵܐ* ܡܫܲܒܲܚ ܒܲܡܪ̈ܵܘܡܹܐ ܡܵܪܝܵܐ* ܘܣܵܗ̈ܕ݁ܘܵܬ݂ܵܟ݂ ܛܵܒ݂ ܡܗܲܝ̈ܡܢܵܢ* ܘܲܠܒܲܝܬܵܟ݂ ܝܵܐ̇ܝܵܐ ܩܲܕ݁ܝܼܫܘܼܬ݂ܵܐ* ܡܵܪܝܵܐ ܠܢܘܼܓ݂ܪܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ܀ ܣܵܓ݂ܕܝܼ̇ܢܲܢ ܠܵܟ݂ ܐܝܼܬ݂ܝܵܐ ܕܡܸܢ ܡܬ݂ܘܿܡ.
ܐܲܩܸܦ: ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡܢ ܫܡܲܝܵܐ* ܫܲܒ݁ܚܘܼܗܝ ܒܲܡܪ̈ܵܘܡܹܐ* ܫܲܒ݁ܚܘܼܗܝ ܟܠܗܘܿܢ ܡܲܠܲܐܟ݂ܵܘ̈ܗܝ* ܫܲܒ݁ܚܘܼܗܝ ܟܠܗܘܿܢ ܚܲܝܠܵܘܵܬܹ̈ܗ* ܫܲܒ݁ܚܘܼܗܝ ܫܸܡܫܵܐ ܘܣܲܗܪܵܐ*ܫܲܒ݁ܚܘܼܗܝ ܟܠ ܟܵܘܟ݁ܒܹ̈ܐ ܘܢܘܼܗܪܵܐ* ܫܲܒ݁ܚܘܼܗܝ ܫܡܲܝ̈ ܫܡܲܝܵܐ܀ ܡܲܝ̈ܵܐ ܕܲܠܥܸܠ ܡܢ ܫܡܲܝܵܐ ܢܫܲܒ݁ܚܘܼܢ ܠܲܫܡܹܗ ܕܡܵܪܝܵܐ܀ ܡܸܛܠ ܕܗܘ ܐܸܡܲܪ ܘܲܗܘܵܘ ܘܗܘ ܦ̮ܩܲܕ݂ ܘܐܸܬ݂ܒ݁ܪܝܼܘ* ܘܐܲܩܝܼܡ ܐܸܢܘܿܢ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ* ܢܵܡܘܿܣܵܐ ܝܲܗ݇ܒ݂ ܘܠܵܐ ܥܵܒܲܪ* ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡܢ ܐܲܪܥܵܐ* ܬܲܢܝܼܢܹ̈ܐ ܘܟ݂ܠܗܘܿܢ ܬܗܘܿܡܹ̈ܐ* ܢܘܼܪܵܐ ܘܒܲܪܕ݂ܵܐ ܘܬܲܠܓܵܐ ܘܲܓ݂ܠܝܼܕ݂ܵܐ* ܪ̈ܘܼܚܹܐ ܘܥܲܠܥܵܠܹ̈ܐ ܥܒ݂ܵܕ݂ܵܐ ܕܡܸܠܬܹܗ* ܛܘܼܪܹ̈ܐ ܘܟ݂ܠܗܹܝܢ ܪ̈ܵܡܵܬ݂ܵܐ* ܐܝܼܠܵܢܹ̈ܐ ܕܦܹܐܪܹ̈ܐ ܘܟ݂ܠܗܘܿܢ ܐܲܪ̈ܙܹܐ* ܚܲܝܘܬ݂ܵܐ ܘܟ݂ܠܵܗ̇ ܒܥܝܼܪܵܐ* ܪ̈ܲܚܫܵܐ ܘܛܲܝܪܵܐ ܘܦܵܪܲܚܬ݂ܵܐ* ܡܲܠܟܹ̈ܐ ܕܐܲܪܥܵܐ ܘܟ݂ܠܗܹܝܢ ܐܸܡ̈ܘܵܬ݂ܵܐ* ܪܵܘܪ̈ܒ݂ܵܢܹܐ ܘܟ݂ܠܗܘܿܢ ܕܲܝܵܢܹ̈ܐ ܕܐܲܪܥܵܐ* ܓܲܕ݁ܘܼܕܹ̈ܐ ܘܲܒ݂ܬ݂̈ܘܼܠܵܬ݂ܵܐ* ܣܵܒܹ̈ܐ ܘܲܥܠܲܝܡܹ̈ܐ ܢܫܲܒ݁ܚܘܼܢ ܠܲܫܡܹܗ ܕܡܵܪܝܵܐ* ܡܸܛܠ ܕܪܲܒ݁ ܗ݇ܘ ܫܡܹܗ ܒܲܠܚܘܿܕ݂ܵܘܗܝ* ܘܫܘܼܒ݂ܚܹܗ ܒܐܲܪܥܵܐ ܘܒܲܫܡܲܝܵܐ* ܡܪܝܼܡ ܩܲܪܢܵܐ ܕܥܲܡܹܗ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܕܟ݂ܠܗܘܿܢ ܙܲܕ݁ܝܼܩܵܘ̈ܗܝ* ܠܲܒ݂ܢܲܝ̈ ܝܼܣܪܵܝܹܠ ܥܲܡܵܐ ܕܩܲܪܝܼܒ݂ ܠܹܗ܀
ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܒܩܘܼܕ݂ܫܹܗ* ܫܲܒ݁ܚܘܼܗܝ ܒܲܪܩܝܼܥܵܐ ܕܥܘܼܫܢܹܗ* ܫܲܒ݁ܚܘܼܗܝ ܒܚܲܝܠܬ݂ܵܢܘܼܬܹܗ* ܫܲܒ݁ܚܘܼܗܝ ܒܣܘܿܓ݂ܐܵܐ ܕܪܲܒ݁ܘܼܬܹܗ* ܫܲܒ݁ܚܘܼܗܝ ܒܩܵܠܵܐ ܕܩܲܪܢܵܐ* ܫܲܒ݁ܚܘܼܗܝ ܒܩܝܼܬ݂ܵܪܹ̈ܐ ܘܲܒ݂ܟܸܢܵܪܹ̈ܐ* ܫܲܒ݁ܚܘܼܗܝ ܒܲܦܠܲܓܹ̈ܐ ܘܒܲܪ̈ܒ݂ܝܼܥܹܐ* ܫܲܒ݁ܚܘܼܗܝ ܒܡܸܢܹ̈ܐ ܚܲܠ̈ܝܵܬ݂ܵܐ* ܫܲܒ݁ܚܘܼܗܝ ܒܨܸܨܠܲܝ̈ ܫܸܡܥܵܐ ܫܲܒ݁ܚܘܼܗܝ ܒܩܵܠܵܐ ܘܒܲܩ̈ܥܵܬ݂ܵܐ* ܒܟ݂ܠ ܢܸܫܡܵܐ ܢܫܲܒܲܚ ܠܡܵܪܝܵܐ܀
ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܟܠܟ݂ܘܿܢ ܥܲܡܡܹ̈ܐ* ܫܲܒ݁ܚܵܝܗܝ ܟܠܗܹܝܢ ܐܸܡ̈ܘܵܬ݂ܵܐ* ܡܸܛܠ ܕܥܸܫܢܲܬ݀ ܥܠܲܝܢ ܛܲܝܒ݁ܘܼܬܹܗ* ܫܲܪܝܼܪܵܐܝܼܬ݂ ܡܵܪܝܵܐ ܗ݇ܘ ܠܥܵܠܲܡ܀ ܫܲܒܲܚ܀ ܥܵܠܲܡ܀ ܘܡܲܩܦܝܼܢ: ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܟܠܵܗ ܐܲܪܥܵܐ. ܝܵܗܹ̇ܒ݂ ܢܘܼܗܪܵܐ ܡܵܪܝܵܐ: ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܲܣܩܝܼܢܲܢ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܥܘܿܢܝܼܬ݂ܵܐ ܕܨܲܦܪܵܐ. ܠܵܟ݂ ܡܫܝܼܚܵܐ ܢܘܼܗܪܵܐ ܫܲܪܝܼܪܵܐ: ܨܸܡܚܵܐ ܫܒ݂ܝܼܚܵܐ ܕܡܢ ܐܲܒ݂ܵܐ: ܕܐܸܬ݂ܓ݁ܠܝܼ ܘܲܕ݂ܢܲܚ ܒܓ݂ܵܘ ܬܹܒܹܝܠ: ܠܚܘܼܕܵܬܹܗ ܘܲܠܦܘܼܪܩܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ ܒܪܹܫܝܼܬ݂ܵܐ ܕܡܸܢܲܢ: ܡܲܣܩܝܼܢܲܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݂ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ ܠܥܵܠܡܝܼܢ.
ܘܐܵܡ̇ܪܝܼܢ ܥܘܿܢܝܼܬ݂ܵܐ ܕܨܲܦܪܵܐ ܐܲܝܟ݂ ܕܲܒ݂ܚܘܼܕ݂ܪܵܐ. ܐܵܘ ܒܓܲܙܵܐ. ܘܬ݂ܵܢܹ̇ܝܢ ܠܵܗ̇܀ ܘܲܡܫܲܒ݁ܚܝܼܢ ܒܵܗ̇܀ ܘܡܲܩܦܝܼܢ ܥܵܠܲܡ. ܒܡܲܕ݂ܢܵܚܲܝ̈ ܨܲܦܪܵܐ. ܬܠܵܬ݂ ܥܘܿܢ̈ܝܵܬ݂ܵܐ ܩܲܕ݂ܡܵܝ̈ܵܬ݂ܵܐ.
ܕܨܲܦܪܵܐ. ܒܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝ̈ܵܐ. ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܟܠܵܗ̇ ܐܲܪܥܵܐ: ܦܠܘܿܚܘ ܠܡܵܪܝܵܐ ܒܚܲܕ݂ܘܼܬ݂ܵܐ. ܝܵܪܬܹ݁ܗ ܕܐܲܒ݂ܵܐ ܘܡܵܪܵܐ ܕܟ݂ܠ ܕܲܗܘܵܐ. ܚܲܕܵܐ ܠܥܹܕ݁ܬܵܟ݂ ܒܕܸܢܚܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ: ܬܢܝܼ. ܘܫܲܒܲܚ ܝܵܪܬܹ݁ܗ ܕܐܲܒ݂ܵܐ ܘܡܵܪܵܐ ܕܟ݂ܠ ܕܲܗܘܵܐ. ܚܲܕܵܐ ܠܥܹܕ݁ܬܵܟ݂ ܒܕܸܢܚܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ. ܥܵܠܲܡ. ܒܡܲܕ݂ܢܵܚܲܝ̈ ܨܲܦܪܵܐ ܠܵܟ݂ܘܼ ܡܵܪܝܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܗܲܒ݂ܠܲܢ ܒܲܚܢܵܢܵܟ݂ ܐܝܼܡܵܡܵܐ ܕܲܡܠܸܐ ܫܲܝܢܵܐ: ܘܲܥܒܸܕ݂ ܠܲܢ ܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ܀ ܣܲܒ݂ܪܲܢ ܠܵܐ ܬܸܦܣܘܿܩ ܘܬܲܪܥܵܟ݂ ܒܐܲܦܲܝ̈ܢ ܠܵܐ ܬܹܐܚܘܿܕ݂: ܘܲܒ݂ܛܝܼܠܘܼܬ݂ܵܟ݂ ܡܸܢܲܢ ܠܵܐ ܬܸܟ݂ܠܸܐ. ܘܐܲܟ݂ܡܵܐ ܕܫܵܘܹܝ̇ܢܲܢ ܐܲܠܵܗܵܐ ܠܵܐ ܬܸܦܪܥܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܒܲܠܚܘܿܕ݂ ܝܵܕܲܥ̇ ܡܚܝܼܠܘܼܬܲܢ܀ ܙܪܘܿܥ ܡܵܪܲܢ ܒܬܹܒܹܝܠ ܚܘܼܒܵܐ ܘܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ: ܘܩܲܝܸܡ ܟܵܗܢܹ̈ܐ ܘܡܲܠܟܹ̈ܐ ܘܕܲܝܵܢܹ̈ܐ. ܘܫܲܝܸܢ ܪܵܘܪ̈ܒ݂ܵܢܹܐ ܘܐܲܣܵܐ ܟܪ̈ܝܼܗܹܐ ܘܛܲܪ ܠܲܚܠܝܼܡܹ̈ܐ: ܘܚܲܣܵܐ ܚܵܘܒܹ̈ܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܢܘܼܗܪܵܐ ܕܢܲܚ. ܠܢܘܼܗܪܵܐ ܡܵܪܝ ܫܒ݂ܝܼܚܵܐ ܕܓܸܠܝܵܢܵܟ݂: ܘܲܠܕܸܢܚܵܐ ܦܨܝܼܚܵܐ ܕܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܗ̇ܘ ܕܠܹܗ ܚܵܝܪ̈ܵܢ ܘܲܡܣܲܒ݁ܪ̈ܵܢ ܘܲܡܣܲܟ̈ܝܵܢ ܟܠܗܹܝܢ ܒܸܪ̈ܝܵܬ݂ܵܐ ܕܲܒ݂ܪܲܝܬ݁: ܐܲܫܘܵܐ ܠܟ݂ܠܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܪ̈ܲܚܡܲܝܟ: ܕܒܹܗ ܢܸܚܕܸ݁ܐ ܘܢܸܬ݂ܒܲܣܲܡ ܥܲܡ ܫܲܪܝܼܪܹ̈ܐ ܒܢܲܝ̈ ܐ݇ܪ̈ܵܙܲܝܟ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ: ܡܵܪܵܐ ܕܟ݂ܠ ܠܥܵܠܡܝܼܢ.
ܘܐܵܡ̇ܪܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܗܵܕܹܐ: ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܩܲܕ݁ܝܼܫܵܐ ܡܵܪܝ ܐܲܦܪܹܝܡ ܡܲܠܦܵܢܵܐ ܣܘܼܪܝܵܝܵܐ: ܥܲܠ ܐܵܬ݂ܘ̈ܵܬ݂ܵܐ ܕܲܫܡܹܗ ܕܝܼܫܘܿܥ ܡܫܝܼܚܵܐ.
ܢܢ. ܢܘܼܗܪܵܐ ܕܢܲܚ ܠܙܲܕ݁ܝܼܩܹ̈ܐ: ܘܠܲܬ݂ܪ̈ܝܼܨܲܝ ܠܸܒܵܐ ܚܲܕ݂ܘܼܬ݂ܵܐ܀ ܝ. ܝܼܫܘܿܥ ܡܵܪܲܢ ܡܫܝܼܚܵܐ: ܕܢܲܚ ܠܲܢ ܡܢ ܥܘܼܒܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܐܸܬ݂ܵܐ ܘܐܲܦ̮ܩܲܢ ܡܢ ܚܸܫܘܿܟ݂ܵܐ: ܘܐܲܢܗܲܪ ܠܲܢ ܒܢܘܼܗܪܹܗ ܓܲܐܝܵܐ܀ ܐ. ܐܝܼܡܵܡܵܐ ܕܢܲܚ ܥܲܠ ܒܢܲܝ̈ܢܵܫܵܐ: ܘܲܥܪܲܩ ܫܘܼܠܛܵܢܹܗ ܕܚܸܫܘܿܟ݂ܵܐ. ܢܘܼܗܪܵܐ ܕܢܲܚ ܠܲܢ ܡܢ ܢܘܼܗܪܹܗ: ܘܐܲܢܗܲܪ ܥܲܝܢܲܝ̈ܢ ܚܸܫ̈ܘܿܟ݂ܵܬ݂ܵܐ܀ ܫ. ܫܘܼܒ݂ܚܹܗ ܐܲܕ݂ܢܲܚ ܒܓ݂ܵܘ ܬܹܒܹܝܠ: ܘܐܲܢܗܲܪ ܠܲܬ݂ܗܘܿܡܹ̈ܐ ܬܲܚܬܵܝܹ̈ܐ. ܕܥܸܟ݂ ܡܵܘܬܵܐ ܘܲܥܪܲܩ ܚܸܫܘܿܟ݂ܵܐ: ܘܐܸܬ݁ܬܲܒܲܪܘ ܬܲܪ̈ܥܹܐ ܕܲܫܝܘܿܠ܀ ܘ. ܘܐܲܢܗܲܪ ܠܟ݂ܠܗܹܝܢ ܒܸܪ̈ܝܵܬ݂ܵܐ: ܕܡܢ ܩܕ݂ܝܼܡ ܚܸܫ̈ܘܿܟ݂ܵܢ ܗ݇ܘܲܝ̈. ܩܵܡܘ ܡܝܼܬܹ̈ܐ ܫܵܟ݂ܒܲܝ̈ ܥܲܦܪܵܐ: ܘܫܲܒܲܚܘ ܕܲܗܘܵܐ ܠܗܘܿܢ ܦܘܼܪܩܵܢܵܐ܀ ܥ. ܥܒܲܕ݂ ܦܘܼܪܩܵܢܵܐ ܘܝܲܗ݇ܒ݂ ܠܲܢ ܚܲܝܹ̈ܐ: ܘܐܸܬ݂ܥܲܠܝܼ ܠܘܵܬ݂ ܐܲܒ݂ܘܼܗܝ ܠܪܵܘܡܵܐ. ܘܬ݂ܘܼܒ݂ ܐܵܬܹܐ ܒܫܘܼܒ݂ܚܹܗ ܪܲܒܵܐ: ܘܡܲܢܗܲܪ ܥܲܝܢܹ̈ܐ ܕܟ݂ܠ ܕܣܲܟ݁ܝܼܘ ܠܹܗ܀ ܡ: ܡܲܠܟܲܢ ܐܵܬܹܐ ܒܫܘܼܒ݂ܚܹܗ ܪܲܒܵܐ: ܢܲܢܗܲܪ ܫܪ̈ܵܓܲܝܢ ܘܢܸܦܘܿܩ ܠܐܘܼܪܥܹܗ. ܘܢܸܚܕܸ݁ܐ ܒܹܗ ܐܲܝܟ݂ ܕܲܚܕ݂ܝܼ ܒܲܢ: ܘܲܡܚܲܕܸ݁ܐ ܠܲܢ ܒܢܘܼܗܪܹܗ ܓܲܐܝܵܐ܀ ܫ. ܫܘܼܒ݂ܚܵܐ ܢܲܣܸܩ ܠܪܲܒ݁ܘܼܬܹܗ: ܘܟ݂ܠܲܢ ܢܵܘܕܸ݁ܐ ܠܐܲܒ݂ܘܼܗܝ ܪܵܡܵܐ. ܕܲܣܓ݂ܝܼܘ ܪ̈ܲܚܡܵܘܗܝ ܘܫܲܠܚܹܗ ܨܹܐܕܲܝܢ: ܘܲܥܒܲܕ݂ ܠܲܢ ܣܲܒ݂ܪܵܐ ܘܦܘܼܪܩܵܢܵܐ܀ ܝ. ܝܵܘܡܹܗ ܕܵܢܲܚ̇ ܡܢ ܫܸܠܝܵܐ: ܘܢܵܦܩ̇ܝܼܢ ܠܐܘܼܪܥܹܗ ܩܲܕ݁ܝܼ̈ܫܵܘܗܝ. ܘܡܲܢܗܪܝܼܢ ܠܲܡܦܹܝܕܲܝ̈ܗܘܿܢ: ܟܠ ܕܲܥܡܲܠܘ ܘܲܠܐܝܼܘ ܘܐܸܬ݁ܛܲܝܲܒ݂ܘ܀ ܚ. ܚܵܕܹܝ̇ܢ ܡܲܠܲܐܟܹ̈ܐ ܘܥܝܼܪ̈ܲܝ ܫܡܲܝܵܐ: ܒܫܘܼܒ݂ܚܵܐ ܕܟܹܐܢܹ̈ܐ ܘܲܕ݂ܙܲܕ݁ܝܼܩܹ̈ܐ. ܘܣܵܝ̇ܡܝܼܢ ܟܠܝܼ̈ܠܹܐ ܒܪܹ̈ܫܲܝܗܘܿܢ: ܘܐܲܟܲܚ݇ܕ݂ ܡܝܲܒ݁ܒ݂ܝܼܢ ܘܲܡܗܲܠܠܝܼܢ܀ ܐ. ܐܲܚܲܝ̈ ܩܘܼܡܘ ܘܐܸܬ݁ܛܲܝܲܒ݂ܘ: ܕܢܵܘܕܸ݁ܐ ܠܡܲܠܟܲܢ ܘܲܠܦܵܪܘܿܩܲܢ. ܕܐܵܬܹܐ ܒܫܘܼܒ݂ܚܹܗ ܘܲܡܚܲܕܸ݁ܐ ܠܲܢ: ܒܢܘܼܗܪܹܗ ܓܲܐܝܵܐ ܒܹܝܬ݂ ܡܲܠܟ݁ܘܼܬ݂ܵܐ܀
ܘܐܸܢ ܐܝܼܬ݂ ܣܘܿܓ݂ܝܼܬ݂ܵܐ ܕܝܵܘܡܵܐ ܠܗܵܪܟܵܐ ܡܸܬ݂ܐܲܡܪܵܐ.
ܘܡܲܩܦܝܼܢ ܒܥܸܩܒ݂ܵܗ̇ ܬܸܫܒ݁ܘܿܚܬܵܐ ܗܵܕܹܐ ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܡܵܪܝ ܢܲܪܣܲܝ ܡܲܠܦܵܢܵܐ.
ܢܘܼܗܪܵܐ ܕܕܸܢܚܹܗ ܕܲܡܫܝܼܚܵܐ: ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܠܲܫܡܲܝܵܐ܀ܛܘܼܥܝܲܝ ܒܲܕ݂ܡܘܼܬ݂ ܚܸܫܘܿܟ݂ܵܐ: ܦܪܝܼܣܵܐ ܗ݇ܘܵܬ݂ ܥܲܠ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܘܲܕ݂ܢܲܚ ܢܘܼܗܪܹܗ ܕܲܡܫܝܼܚܵܐ: ܘܲܩܢܵܐ ܥܵܠܡܵܐ ܒܘܼܝܵܢܵܐ܀ ܠܥܸܕܵܢ ܠܸܠܝܵܐ ܕܵܡܹ̇ܐ ܗ݇ܘܵܐ: ܙܲܒ݂ܢܵܐ ܕܡܢ ܐܵܕ݂ܵܡ ܘܲܠܟ݂ܵܐ. ܘܝܘܿܡ ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ: ܠܪܲܗܛܵܐ ܕܫܵܥܹ̈ܐ ܕܐܝܼܡܵܡܵܐ܀ ܨܲܦܪܵܐ ܟܲܢܝܹܗ ܐܵܦ ܡܵܪܲܢ: ܠܫܘܼܪܵܝܵܐ ܕܟ݂ܵܪܘܿܙܘܼܬܹܗ. ܘܥܸܕܵܢ ܪܲܡܫܵܐ ܠܫܘܼܠܵܡܵܐ: ܕܲܡܢܝܼܚ ܠܥܵܠܡܵܐ ܡܢ ܥܲܡܠܹܗ܀ ܠܗܵܢܵܐ ܣܲܒ݂ܪܵܐ ܡܣܲܟܹ݁ܝܢ ܗ݇ܘܵܘ: ܟܵܗܢܹ̈ܐ ܘܡܲܠܟܹ̈ܐ ܘܲܢܒ݂ܝܼܹ̈ܐ. ܘܐܲܢܝܼܚ ܐܸܢܘܿܢ ܒܵܪܘܿܝܵܐ: ܒܲܠܡܹܐܢܵܐ ܕܝܘܿܡ ܓܸܠܝܵܢܹܗ܀ ܒܓܸܠܝܵܢܹܗ ܢܵܚ̈ ܒܸܪ̈ܝܵܬ݂ܵܐ: ܕܲܫܚܝܼ̈ܩܵܢ ܗ݇ܘܲܝ̈ ܒܲܚܛܝܼܬ݂ܵܐ. ܘܫܲܪܝܼ ܥܵܠܡܵܐ ܡܸܬ݂ܗܲܓܸ݁ܐ: ܒܥܸܢܝܵܢܵܐ ܕܲܥܬ݂ܝܼ̈ܕ݂ܵܬ݂ܵܐ܀ ܕܝܵܬܹܩܹܐ ܚܕܲܬܵܐ ܐܲܫܠܸܡ: ܠܲܡܩܲܒ݁ܠܵܢܹ̈ܐ ܕܝܘܼܠܦܵܢܹܗ. ܛܲܒ݂ܥܵܗ̇ ܒܲܕ݂ܡܵܐ ܕܲܩܢܘܿܡܹܗ: ܕܠܵܐ ܢܸܬ݁ܕܲܓ݁ܠܘܼܢ ܫܘܼܘܕܵܝܵܘ̈ܗܝ܀ ܒܫܘܼܘܕܵܝܹ̈ܐ ܕܡܲܠܟ݁ܘܼܬ݂ ܪܵܘܡܵܐ: ܐܲܣܪܹܗ ܠܪܲܗܛܵܐ ܕܲܡܠܝܼܠܹ̈ܐ. ܘܗܵܐ ܡܣܲܟܹ݁ܝܢ ܠܹܗ ܠܓܸܠܝܵܢܹܗ: ܐܲܪ̈ܥܵܢܹܐ ܘܲܫܡܲܝܵܢܹ̈ܐ܀ ܩܪܸܒ݂ ܠܹܗ ܡܸܟܹ݁ܝܠ ܓܸܠܝܵܢܹܗ: ܕܡܲܠܟܵܐ ܕܐܲܡܠܸܟ݂ ܡܢ ܓܸܢܣܲܢ. ܬܵܘ ܢܸܬ݁ܛܲܝܲܒ݂ ܠܲܚܙܵܬܹܗ: ܥܲܡ ܟܸܢܫܹ̈ܐ ܕܲܫܡܲܝܵܢܹ̈ܐ܀ ܡܸܫܚܵܐ ܚܘܼܒܵܐ ܢܸܣܲܒ݂ ܠܲܢ: ܠܗ̇ܘ ܝܵܘܡܵܐ ܕܲܡܠܸܐ ܩܸܢܛܵܐ. ܕܠܵܐ ܢܸܫܡܲܥ ܗ̇ܝ ܒܲܪ݇ܬ݂ ܩܵܠܵܐ: ܕܠܵܐ ܝܵܕܲܥ̇ ܐ݇ܢܵܐ ܥܲܡܠܲܝ̈ܟ݁ܘܿܢ܀ ܒܟܲܟ݁ܪܵܐ ܕܡܸܠܲܬ݂ ܝܘܼܠܦܵܢܵܐ: ܢܸܬ݁ܬܲܓܲܪ ܥܲܕ݂ ܚܲܝܝܼܢܲܢ. ܕܢܸܫܬ݁ܘܸܐ ܕܢܸܫܡܲܥ ܗ̇ܘ ܩܵܠܵܐ: ܕܬ݂ܵܘ ܩܲܒܸ݁ܠܘ ܡܘܼܠܟܵܢܲܝ̈ܟ݁ܘܿܢ܀ ܒܣܲܒ݂ܪܵܐ ܕܚܲܝܹ̈ܐ ܕܲܥܬ݂ܝܼܕ݂ܝܼܢ: ܢܹܐܣܪܝܼܗ̇ ܠܐܸܠܦܵܐ ܕܪܸܥܝܵܢܲܢ. ܘܲܒ݂ܚܘܼܒܵܐ ܘܗܲܝܡܵܢܘܼܬ݂ܵܐ ܢܡܲܢܲܥ ܠܲܠܡܹܐܢ ܚܲܕ݂̈ܘܵܬ݂ܵܐ܀

ܨܠܘܿܬ݂ܵܐ. ܡܫܲܒ݁ܚܝܼܢܲܢ ܘܲܡܪܲܡܪܡܝܼܢܲܢ ܘܲܡܗܲܠܠܝܼܢܲܢ ܘܲܡܩܲܠܣܝܼܢܲܢ: ܠܲܟ݂ܝܵܢܵܐ ܟܲܣܝܵܐ ܘܲܓ݂ܢܝܼܙܵܐ: ܘܛܘܼܒ݂ܬ݂ܵܢܵܐ ܘܠܵܐ ܡܸܬ݁ܕܲܪܟܵܢܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܬܸܫܒ݁ܘܿܚܬܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܕ݂ܥܹܐܕܹ̈ܐ ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܡܵܪܝ ܬܹܐܵܕܘܿܪܘܿܣ ܡܦܲܫܩܵܢܵܐ ܬܹܒܹܝܠܵܝܵܐ. ܒܥܹܐܕܹ̈ܐ ܕܹܝܢ ܚܲܕ݂ ܦܸܬ݂ܓ݂ܵܡܵܐ ܒܩܲܢܟܹ̈ܐ ܡܸܬ݂ܐܡܲܪ ܘܲܐ݇ܚܪܹܝܢ ܒܗܲܝܟ݁ܠܵܐ.
ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ܀ ܬܢܝܼ ܓ ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ܀ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ܀ ܣܵܓ݂ܕ݁ܝܼ̇ܢܲܢ ܠܵܟ݂܀ ܡܫܲܒ݁ܚܝܼܢܲܢ ܠܵܟ݂܀ ܡܪܲܡܪܡܝܼܢܲܢ ܠܵܟ݂܀ ܐܝܼܬ݂ܝܵܐ ܕܡܢ ܡܬ݂ܘܿܡ܀ ܟܝܵܢܵܐ ܓܢܝܼܙܵܐ ܕܠܵܐ ܡܸܬ݁ܕ݁ܪܸܟ݂܀ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ܀ ܡܲܠܟܵܐ ܕܡܲܠܟܹ̈ܐ܀ ܘܡܵܪܵܐ ܕܡܵܪ̈ܵܘܵܬ݂ܵܐ܀ ܕܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ܀ ܗ̇ܘ ܕܐ݇ܢܵܫ ܡܢ ܒܢܲܝ̈ܢܵܫܵܐ ܠܵܐ ܚܙܵܝܗܝ܀ ܘܐܵܦܠܵܐ ܡܨܸܐ ܠܡܸܚܙܝܹܗ܀ ܩܲܕ݁ܝܼܫܵܐ ܒܲܠܚܘܿܕ݂ܵܘܗܝ܀ ܚܲܝܠܬ݂ܵܢܵܐ ܒܲܠܚܘܿܕ݂ܵܘܗܝ܀ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ ܒܲܠܚܘܿܕ݂ܵܘܗܝ܀ ܡܵܘܕܹ݁ܝܢܲܢ ܠܵܟ݂܀ ܒܝܲܕ݂ ܡܸܨܥܵܝܵܐ ܕܛܵܒ݂̈ܵܬܲܢ܀ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ܀ ܦܵܪܘܿܩܹܗ ܕܥܵܠܡܵܐ܀ ܘܲܒ݂ܪܹܗ ܕܲܡܪܲܝܡܵܐ܀ ܐܸܡܪܹܗ ܕܐܲܠܵܗܵܐ ܚܲܝܵܐ܀ ܫܵܩܹ̇ܠ ܚܛܝܼܬܹܗ ܕܥܵܠܡܵܐ܀ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܝ̇ܵܬܹܒ݂ ܡܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܘܼܗܝ܀ ܩܲܒܸ݁ܠ ܒܵܥܘܼܬܲܢ܀ ܕܐܲܢ݇ܬ݁ܘܼ ܐܲܠܵܗܲܢ܀ ܘܐܲܢ݇ܬ݁ܘܼ ܡܵܪܲܢ܀ ܘܐܲܢ݇ܬ݁ܘܼ ܡܲܠܟܲܢ܀ ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܲܢ܀ ܘܐܲܢ݇ܬ݁ܘܼ ܫܵܒ݂ܘܿܩܵܐ ܕܲܚܛܵܗܲܝ̈ܢ܀ ܥܲܝܢܹ̈ܐ ܕܟ݂ܠܢܵܫ ܠܘܵܬ݂ܵܟ݂ ܬܲܠ̈ܝܵܢ܀ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ܀ ܫܘܼܒ݂ܚܵܐ ܠܐܲܠܵܗܵܐ ܐܲܒ݂ܘܼܟ݂܀ ܘܠܵܟ݂ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ܀
ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܕ݂ܥܹܐܕܹ̈ܐ. ܠܵܟ݂ ܡܵܪܝ ܘܠܲܡܫܝܼܚܵܟ݂ ܘܲܠܪܘܼܚܵܟ݂ ܚܲܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܘܡܲܚܝܵܢܵܐ ܘܐܲܠܵܗܵܝܵܐ: ܡܲܣܩܝܼܢܲܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܒܩܵܠܵܐ. ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ.
ܘܥܵܢܹܝ̇ܢ ܒܩܵܠܵܐ. ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬܪܲܚܲܡ ܥܠܲܝܢ. ܫܲܒܲܚ. ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ...܀ ܥܵܠܲܡ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ...܀
ܘܐܵܡ̇ܪܝܼܢ. ܒܲܪܸܟ݂ ܡܵܪܝ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ... ܘܟܲܕ݂ ܡܫܲܒ݁ܚܝܼܢ ܒܵܗ̇ ܥܵܒ݂ܪܝܼܢ ܒܲܫܠܵܡܵܐ.
ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܸܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܸܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܘܲܡܨܲܠܸܐ ܪܲܒܵܐ: ܒܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܒ݂ܥܹܐܕܹ̈ܐ. ܚܲܢܵܢܵܐ ܕܩܲܕ݁ܝܼܫ ܫܡܹܗ: ܘܩܲܕ݁ܝܼܫ ܡܲܫܪܝܹܗ: ܘܩܲܕ݁ܝܼܫ ܐܲܬ݂ܪܵܐ ܕܲܫܟ݂ܝܼܢܬܹ݁ܗ: ܘܩܲܕ݁ܝܼܫܝܼܢ ܚܲܝ̈ܠܵܘܵܬ݂ܵܐ ܥܸܠܵܝܹ̈ܐ ܕܲܡܩܲܕ݁ܫܝܼܢ ܠܹܗ ܒܗܘܼܠܵܠܲܝ̈ܗܘܿܢ: ܘܡܢ ܪ̈ܘܼܚܵܢܹܐ ܘܦܲܓ݂ܪ̈ܵܢܹܐ ܡܸܬ݂ܩܲܕܲܫ ܒܩܵܠܹ̈ܐ ܩܲܕ݁ܝܼܫܹ̈ܐ ܕܠܵܐ ܫܲܠܘܵܐ: ܩܲܕܸ݁ܫ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܗܲܝܟܲܠ ܢܲܦ̮̈ܫܵܬܲܢ: ܘܕܲܟܵܐ ܬܹܐܪ̈ܵܬܲܢ: ܘܚܲܣܵܐ ܚܵܘ̈ܒܵܬܲܢ: ܘܚܵܘܲܪ ܟܘܼܬ݂̈ܡܵܬܲܢ: ܘܲܥܒܸܕܲܝܢ ܡܵܪܝ ܢܵܘܣܹ̈ܐ ܕܟܲܝ̈ܵܐ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܡܪܲܝܲܡܬܵܐ ܘܗܲܝܟ̈ܠܹܐ ܡܨܲܒ݁ܬܹ̈ܐ ܕܥܵܗܢ̇ܝܼܢ ܠܐܝܼܩܵܪ ܬܸܫܡܸܫܬܹ݁ܗ ܕܚܘܼܒܵܟ݂: ܡܩܲܕܸ݁ܫ ܟܠ ܒܚܲܝܠܵܐ ܕܡܸܠܬܹܗ ܘܲܕ݂ܪܘܼܚܹܗ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹ̇ܝܢ: ܐܵܡܹܝܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܚܵܬ݂ܵܗ̇. ܩܲܕ݁ܝܼܫܲܬ݁ ܡܵܪܝ ܐܝܼܬ݂ܝܵܐܝܼܬ݂: ܘܲܡܫܲܒܲܚ ܐܲܢ݇ܬ݁ ܡܬ݂ܘܿܡܵܐܝܼܬ݂: ܘܪܵܡ ܐܲܢ݇ܬ݁ ܘܲܡܥܲܠܲܝ ܐܲܢ݇ܬ݁ ܠܥܸܠ ܡܢ ܟܠ ܠܵܐ ܡܸܬ݁ܕܲܪܟܵܢܵܐܝܼܬ݂: ܘܡܸܣܬ݁ܓܸܕ݂ ܐܲܢ݇ܬ݁ ܡܢ ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܬܠܝܼܬ݂ܵܐܝܼܬ݂ ܘܲܚܕ݂ܵܢܵܐܝܼܬ݂: ܡܲܠܟܵܐ ܪܲܒܵܐ ܕܬܸܫܒ݁ܘܿܚܬܵܐ: ܐܝܼܬ݂ܝܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܡܢ ܥܵܠܲܡ: ܡܵܘܕܹ݁ܝܢܲܢ ܘܣܵܓ݂ܕ݁ܝܼܢܲܢ ܘܲܡܫܲܒ݁ܚܝܼܢܲܢ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܡܲܩܦܝܼܢ ܨܠܘܿܬ݂ܵܐ ܕܥܘܼܕ݂ܪܵܢܵܐ: ܕܲܒ݂ܥܹܕ݁ܬܵܐ ܒܲܠܚܘܿܕ݂ ܡܸܬ݂ܐܲܡܪܵܐ.
ܨܠܘܿܬ݂ܵܐ. ܥܘܼܕ݂ܪܵܢܵܐ ܡܵܪܝ ܕܪ̈ܲܚܡܲܝܟ ܘܣܘܼܝܵܥܵܐ ܪܲܒܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܚܲܝܠܵܐ ܟܲܣܝܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܘܝܲܡܝܼܢܵܟ݂ ܕܡܲܠܝܵܐ ܪ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ: ܐܲܓܸ݁ܢ ܘܐܲܠܘܵܐ ܠܲܡܚܝܼܠܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܡܸܢ ܒܲܝܬܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ ܡܠܸܐ ܟܠ ܥܘܼܕ݂ܪ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܒܲܨܠܘܿܬ݂ ܟܠܗܘܿܢ ܩܲܕ݁ܝܼܫܹ̈ܐ ܡܪ̈ܲܥܝܵܢܲܝܟ: ܡܵܪܵܐ ܕܟ݂ܠ...ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ ܒܲܪܸܟ݂ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ. ܒܒ݂ܘܼܪܟ݁ܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܢܸܬ݂ܒܲܪܟ݂ܘܼܢ ܥܲܒ݂ܕܲܝ̈ܟ: ܘܒܲܒ݂ܛܝܼܠܘܼܬܹܗ ܕܨܸܒ݂ܝܵܢܵܟ݂ ܢܸܬ݂ܢܲܛܪܘܼܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܘܫܲܝܢܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܲܫܠܵܡܵܐ ܢܲܓ݁ܝܼܪܵܐ ܕܡܵܪܘܼܬ݂ܵܟ݂: ܢܲܡܠܸܟ݂ ܒܥܲܡܵܟ݂ ܘܲܒ݂ܥܹܕ݁ܬܵܟ݂. ܟܠܗܘܿܢ ܝܵܘܡܵܬܹ̈ܗ ܕܥܵܠܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ...ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ ܒܲܪܸܟ݂ܡܵܪܝ.

ܐ݇ܚܪܹܬ݂ܵܐ. ܗܘܝܼ ܡܵܪܝ ܥܲܡܲܢ ܒܪ̈ܲܚܡܲܝܟ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܘܕܲܒܲܪ ܠܲܢ ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܘܼܪܟ݁ܬ݂ܵܐ ܡܵܪܝ ܕܪ̈ܲܚܡܹܐ: ܡܸܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܕܪ̈ܲܚܡܹܐ: ܒܝܲܕ݂ ܡܲܠܲܐܟ݂ܵܐ ܕܪ̈ܲܚܡܹܐ: ܡܸܢ ܠܘܵܬ݂ ܐܲܠܵܗܵܐ ܣܲܓ݁ܝܼ ܪ̈ܲܚܡܹܐ: ܬܸܫܬܲܕܲܪ ܘܬܸܫܬܲܦܲܥ ܥܲܠ ܟܸܢܫܹ̈ܐ ܕܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ: ܕܐܲܡܝܼܢܝܼܢ ܘܩܵܪܹܝ̇ܢ ܠܲܫܡܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ: ܘܚܵܝ̇ܪܝܼܢ ܘܲܡܣܲܟܹ݁ܝܢ ܠܪ̈ܲܚܡܹܐ: ܘܲܠܦܘܼܪܩܵܢܵܐ ܕܡܸܢ ܠܘܵܬ݂ܵܟ݂: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܡܲܩܦܝܼܢ. ܨܠܘܿܬ݂ܵܗ̇ ܡܵܪܝ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܗ̇ ܕܐܸܡܵܐ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܬܲܟ݂ܫܲܦ̮ܬܵܗ̇ ܘܬܲܚܢܲܢܬܵܗ̇ ܕܡܲܠܝܲܬ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܐ: ܡܵܪܬ݁ܝ ܡܲܪܝܲܡ ܛܘܼܒ݂ܵܢܝܼܬ݂ܵܐ: ܘܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ ܙܲܟܵܝܵܐ: ܘܥܘܼܕ݂ܪܵܢܵܐ ܐܲܠܵܗܵܝܵܐ: ܘܒ݂ܵܥܘܼܬܹܗ ܕܡܵܪܝ ܝܘܿܚܲܢܵܢ ܡܲܥܡܕ݂ܵܢܵܐ: ܘܕܲܫܠܝܼܚܹ̈ܐ ܛܘܼܒ݂ܵܢܹ̈ܐ: ܘܕܲܐܒ݂ܵܗ̈ܵܬ݂ܵܐ ܢܲܨܝܼܚܹ̈ܐ ܡܣܲܒ݁ܪ̈ܵܢܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܘܙܵܪ̈ܥܲܝ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܕܲܐܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܡܵܪܝ ܐܲܒ݂ܵܐ ܩܵܬ݂ܘܿܠܝܼܩܵܐ: ܘܡܵܪܝ ܐܸܣܛܲܦܵܢܘܿܣ ܒܘܼܟ݂ܪܵܐ ܕܣܵܗܕܹ̈ܐ: ܘܲܕ݂ܓܲܢ݇ܒܵܪ ܚܲܝܠܵܐ ܡܵܪܝ ܓܝܼܘܲܪܓܝܼܣ ܣܵܗܕܵܐ ܢܲܨܝܼܚܵܐ: ܘܲܕ݂ܡܵܪܝ ܦܠܵܢ: ܘܲܕ݂ܡܵܪܝ ܐܵܘܓܹ݁ܝܢ ܛܘܼܒ݂ܵܢܵܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܣܝܼܥܬܹ݁ܗ ܪܘܼܚܵܢܵܝܬܵܐ. ܘܕܲܫܡܘܿܢܝܼ ܘܕܲܒ݂ܢܹܝ̈ܗ̇: ܘܲܕ݂ܟ݂ܠܗܘܿܢ ܣܵܗܕܵܘ̈ܗܝ ܘܩܲܕ݁ܝܼ̈ܫܵܘܗܝ ܕܡܵܪܲܢ: ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܫܘܼܪܵܐ ܪܵܡܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܚܲܣܝܼܢܵܐ: ܕܬܸܦܪܘܿܩ ܘܲܬ݂ܦܲܨܸܐ ܘܲܬ݂ܫܵܘܙܸܒ݂ ܘܲܬ݂ܢܲܛܲܪ ܠܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ ܡܸܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹܗ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹ̇ܝܢ: ܐܵܡܹܝܢ.
ܘܟܸܢ ܚܵܬ݂ܡ̇ܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܪ̈ܵܝܹܐ. ܘܢܵܣܹ̇ܒ݂ ܟܵܗܢܵܐ ܨܠܝܼܒ݂ܵܐ ܒܐܝܼܕܹܗ ܕܝܲܡܝܼܢܵܐ: ܘܩܵܐܹ̇ܡ ܒܓܲܒܵܐ ܬܲܝܡܢܵܝܵܐ: ܘܡܲܗܦܸܟ݂ ܐܲܦܵܘ̈ܗܝ ܠܥܲܡܵܐ ܘܐܵܡܲܪ: ܒܲܪܸܟ݂ ܡܵܪܝ ܒܦܘܼܩܕܵܢܟ݂ܘܿܢ. ܘܥܵܢܹ̇ܝܢ: ܒܦܘܼܩܕܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܡܵܪܲܢ: ܘܫܘܼܒ݂ܚܵܐ ܠܲܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ. ܘܡܲܪܟ݁ܢܝܼܢ ܥܲܡܵܐ ܪܹ̈ܫܲܝܗܘܿܢ.
ܚܘܼܬܵܡܵܐ
ܫܲܟܸ݁ܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܒܥܸܕܵܢܵܐ ܗܵܢܵܐ ܕܨܲܦܪܵܐ: ܦܘܼܪܩܵܢܵܐ ܠܲܛܠܝܼܡܹ̈ܐ: ܫܪܵܝܵܐ ܠܐܲܣܝܼܪܹ̈ܐ: ܢܦܹܐܫܵܐ ܠܲܡܛܲܪ̈ܦܹܐ: ܐܵܣܝܘܼܬ݂ܵܐ ܠܲܟ݂ܪ̈ܝܼܗܹܐ: ܩܘܼܪܵܒ݂ܵܐ ܠܪ̈ܲܚܝܼܩܹܐ: ܢܘܼܛܵܪܵܐ ܠܩܲܪ̈ܝܼܒܹܐ: ܚܘܼܣܵܝܵܐ ܠܚܲܛܵܝܹ̈ܐ: ܩܘܼܒܵܠܵܐ ܠܬܲܝܵܒܹ̈ܐ: ܪܘܼܡܪܵܡܵܐ ܠܙܲܕ݁ܝܼܩܹ̈ܐ: ܦܘܼܪܢܵܣܵܐ ܠܡܸܣܟܹ݁ܢܹ̈ܐ: ܫܟ݂ܵܚܬܵܐ ܠܐܲܒ݁ܝܼܕܹ̈ܐ: ܦܘܼܢܵܝܵܐ ܠܲܛܪ̈ܝܼܕܹܐ: ܕܘܼܟ݂ܪܵܢܵܐ ܛܵܒ݂ܵܐ ܘܲܡܩܲܒ݁ܠܵܐ ܠܥܲܢܝܼܕܹ̈ܐ: ܪ̈ܲܚܡܹܐ ܘܲܚܢܲܢܵܐ ܠܟ݂ܠܗܘܿܢ ܒܪ̈ܲܝܵܐ ܘܲܥܒ݂ܝܼܕܹ̈ܐ: ܘܲܣܥܘܿܪ ܥܲܡܲܢ ܘܥܲܡ ܟܠܢܵܫ: ܐܲܝܠܹܝܢ ܕܠܲܢ ܡܚܝܼܠܹ̈ܐ ܘܚܲܛܵܝܹ̈ܐ ܥܵܕ݂ܪ̈ܵܢ ܘܲܠܡܵܪܘܼܬ݂ܵܟ݂ ܫܵܦܪ̈ܵܢ: ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܪ̈ܲܚܡܲܝܟ: ܗܵܫܵܐ ♰ ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܪܵܫܹ̇ܡ ܨܠܝܼܒ݂ܵܐ ܥܲܠ ܥܲܡܵܐ: ܟܲܕ݂ ܦܵܫܹ̇ܛ ܐܝܼܕܹܗ ܠܥܸܠ ܠܐܲܦܲܝ̈ ܡܲܥܪܒ݂ܵܐ: ܘܡܲܝܬܸ݁ܐ ܠܵܗ̇ ܠܡܲܕ݂ܢܚܵܐ: ܘܟܸܢ ܦܵܫܹ̇ܛ ܠܵܗ̇ ܠܓܲܒܵܐ ܕܬܲܝܡܢܵܝܵܐ: ܘܡܵܘܒܸ݁ܠ ܠܵܗ̇ ܠܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ. ܐܲܝܟܲܢܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܸܣܬ݁ܡܸܟ݂ ܥܲܠ ܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܓܲܒܵܐ ܣܸܡܵܠܵܝܵܐ ܕܟܸܢܫܵܐ.
ܒܲܫܪܵܪܵܐ ܓܹܝܪ ܕܠܵܐ ܦܘܼܠܵܓ݂ܵܐ: ܡܗܲܝܡܢܝܼܢܲܢ ܒܚܲܕ݂ ܐܲܠܵܗܵܐ: ܐܲܒ݂ܵܐ ܐܲܚܝܼܕ݂ ܟܠ. ܥܵܒ݂ܘܿܕ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܐܲܝܠܹܝܢ ܕܡܸܬ݂ܚܲܙ̈ܝܵܢ ܘܲܕ݂ܠܵܐ ܡܸܬ݂ܚܲܙ̈ܝܵܢ. ܘܲܒ݂ܚܲܕ݂ ܡܵܪܝܵܐ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ. ܒܪܹܗ ܕܐܲܠܵܗܵܐ ܝܼܚܝܼܕ݂ܵܝܵܐ: ܒܘܼܟ݂ܪܵܐ ܕܟ݂ܠܗܹܝܢ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܗܵܘ̇ ܕܡܢ ܐܲܒ݂ܘܼܗܝ ܐܸܬ݂ܝܼܠܸܕ݂ ܩܕ݂ܵܡ ܟܠܗܘܿܢ ܥܵܠܡܹ̈ܐ: ܘܠܵܐ ܐܸܬ݂ܥܒܸܕ݂. ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܕܡܢ ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܒܲܪ ܟܝܵܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܕܒ݂ܐܝܼ̈ܕ݂ܵܘܗܝ ܐܸܬ݁ܬܲܩܲܢܘ ܥܵܠܡܹ̈ܐ: ܘܐܸܬ݂ܒ݁ܪܝܼ ܟܠ ܡܸܕ݁ܡ. ܗܵܘ̇ ܕܡܸܛܠܵܬܲܢ ܒܢܲܝ̈ܢܵܫܵܐ: ܘܡܸܛܠ ܦܘܼܪܩܵܢܲܢ: ܢܚܸܬ݂ ܡܸܢ ܫܡܲܝܵܐ: ܘܐܸܬ݂ܓܲܫܲܡ ܡܸܢ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܘܲܗܘܵܐ ܒܲܪܢܵܫܵܐ: ܘܐܸܬ݂ܒ݁ܛܸܢ ܘܐܸܬ݂ܝܼܠܸܕ݂ ܡܢ ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܘܚܲܫ ܘܐܸܙܕ݁ܩܸܦ: ܒܝܵܘܡܲܝ̈ ܦܲܢܛܝܼܘܿܣ ܦܝܼܠܵܛܘܿܣ. ܘܐܸܬ݂ܩ̄ܒܲܪ: ܘܩܵܡ ܠܲܬ݂ܠܵܬ݂ܵܐ ܝܵܘܡܝܼ̈ܢ ܐܲܝܟ݂ ܕܲܟ݂ܬ݂ܝܼܒ݂. ܘܲܣܠܸܩ ܠܲܫܡܲܝܵܐ: ܘܝܼܬܸܒ݂ ܡܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܘܬ݂ܘܼܒ݂ ܥܬ݂ܝܼܕ݂ ܠܡܹܐܬ݂ܵܐ ܠܲܡܕ݂ܵܢ ܠܡܝܼܬܹ̈ܐ ܘܲܠܚܲܝܹ̈ܐ. ܘܲܒ݂ܚܲܕ݂ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܪܘܼܚܵܐ ܕܲܫܪܵܪܵܐ: ܗܵܘ̇ ܕܡܢ ܐܲܒ݂ܵܐ ܢܵܦܹ݁ܩ: ܪܘܼܚܵܐ ܡܲܚܝܵܢܵܐ. ܘܒܲܚܕ݂ܵܐ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܘܲܫܠܝܼܚܵܝܬܵܐ ܩܵܬ݂ܘܿܠܝܼܩܝܼ. ܘܡܵܘܕܹ݁ܝܢܲܢ ܒܲܚܕ݂ܵܐ ܡܲܥܡܘܿܕ݂ܝܼܬ݂ܵܐ: ܠܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ. ܘܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܲܒ݂ܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ: ܐܵܡܹܝܢ.
