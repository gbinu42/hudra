ܘܲܕ݂ܨܲܦܪܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܕܲܫܚܝܼܡܹ̈ܐ
ܕܲܥܪܘܼܒ݂ܬܵܐ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟ݂: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟ݂: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬ݂ܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

	ܡܵܘܬ݁ܒ݂ܵܐ܀
      ܡܵܘܬ݁ܒ݂ܵܐ܀ ܕܠܸܠܝܵܐ. ܒܐܲܠܵܗܵܐ ܡܸܠܬ݂ܵܐ. ܡܵܪܝܵܐ ܡܲܢܘܼ ܢܸܥܡܲܪ ܒܡܲܫܟ݁ܢܵܟ݂: ܡܲܢܘܼ ܐܲܟ݂ܘܵܬ݂ܵܟ݂ ܡܵܪܝܵܐ. ܡܲܢܘܼ ܢܸܬܸ݁ܠ ܠܝܼ ܐܲܬ݂ܪܵܐ ܠܲܢܝܵܚܵܐ ܡܵܪܝܵܐ. ܘܡܲܢܘܼ ܢܩܲܪܒܲܢܝ ܠܲܫܟ݂ܝܼܢܬܵܟ݂ ܠܡܸܫܡܲܥ ܩܵܠܹ̈ܐ ܕܚܝܼܠܹ̈ܐ. ܪܸܡܙܵܟ݂ ܡܵܪܝ ܒܲܪܦܵܦ ܥܲܝܢܵܐ ܐܲܕܪܟܲܢܝ. ܘܡܢ ܐܘܼܠܨܵܢܵܐ ܐܲܦ̮ܩܲܢܝ. ܘܒ݂ܐܘܿܪܸܫܠܸܡ ܐܲܬ݂ܪܵܐ ܕܲܫܟ݂ܝܢܼܬܵܟ݂. ܩܕ݂ܵܡ ܐܲܦܲܝ̈ܟ ܐܲܩܝܼܡܬܵܢܝ ܡܵܪܹܟ݁ܠ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂: ܬܢܝܼ. ܒܪܲܡܫܵܐ ܘܲܒ݂ܨܲܦܪܵܐ ܘܲܒ݂ܛܲܗܪܵܐ. ܠܟܲܪܡܵܐ ܪܘܼܚܵܢܵܐ ܕܲܡܠܸܐ ܛܘܼܒܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܩܪܲܝܬܵܢܝ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܒܲܚܕ݂ܵܥܸܣܪܹ̈ܐ ܫܵܥ̈ܝܼܢ. ܕܐܸܦܠܘܿܚ ܘܐܸܣܲܒ݂ ܡܸܢܵܟ݂ ܐܲܓ݂ܪܵܐ ܡܫܲܠܡܵܢܵܐ. ܘܟܲܕ݂ ܪ̈ܲܚܡܲܝܟ ܠܗܵܠܹܝܢ ܐܲܫܘܝܵܘܼܢܝ. ܐܸܢܵܐ ܒܨܸܒ݂ܝܵܢܝ ܒܪܲܦܝܘܼܬ݂ ܕܘܼܒܵܪ̈ܲܝ ܐܵܘܓ݁ܪܹܬ݂ ܠܲܩܢܘܿܡܝ ܠܥܵܘܠܵܐ ܘܠܲܚܛܝܼܬ݂ܵܐ. ܡܸܬܡܟܲܫܲܦܢܵܐ ܕܹܝܢ ܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܡܵܪܵܐ ܡܪܲܚܡܵܢܵܐ. ܕܡܵܐ ܕܐܵܬܹ݁ܝܬ݁ ܠܲܡܕ݂ܵܢ ܡܵܪܲܢ. ܘܝ̇ܵܗ݇ܒܲܬ݁ ܐܲܓ݂ܪܵܐ ܠܦܵܠܚܲܝ̈ܟ. ܠܵܐ ܡܵܪܝ ܐܸܬ݁ܬ݁ܒܲܥ ܕܝܼܢܵܐ ܕܒܸܛܠܵܢܝ. ܐܸܠܵܐ ܡܸܛܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܪܲܚܸܡ ܥܠܲܝ. ܡܸܛܠ ܕܲܓ݂ܡܲܪ ܚܲܝܲܝ̈ ܒܕ݂ܵܘܘܿܢܵܐ. ܟܝܵܢܵܐ ܡܵܝܘܿܬ݂ܵܐ ܕܲܓ݂ܒ݂ܝܼܠܬܹܗ ܕܐܵܕ݂ܵܡ ܗܵܐ ܓܵܥܹܐ ܟܲܕ݂ ܐܵܡܲܪ. ܕܕ݂ܵܘܝܵܐ ܐܝܼܬܲܝ ܒܪܲܢܵܫܵܐ. ܘܡܲܢܘܼ ܡܨܸܐ ܡܦܲܨܸܐ ܠܝܼ ܡܢ ܡܵܘܬܵܐ ܝܲܥܢܵܐ. ܕܗܵܐ ܣܲܚܦܲܢܝ ܘܲܫܕ݂ܵܢܝ ܘܒܲܙܹܗ ܠܩܸܢܝܵܢܝ ܘܫܲܒ݂ܩܲܢܝ ܥܲܪܛܸܠ. ܒܵܟ݂ ܡܵܪܝ ܡܸܬ݂ܓܵܘܲܣܢܵܐ. ܦܪܘܿܩܲܝܢܝ ܡܸܢܹܗ ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝ.
ܕܲܨܠܘܿܬ݂ܵܐ. ܐܲܒ݂ܵܐ ܚܲܢܵܢܵܐ. ܫܘܼܚܠܵܦܵܐ: ܓ. ܩܵܠܵܐ: ܐ. ܒܪܘܼܟܵܒ݂ ܝܵܕܲܥ ܚܘܼܫܵܒܹ̈ܐ.
ܗܵܘ̇ ܕܐܝܼܬ݂ܵܘܗܝ ܡܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ. ܐܲܒ݂ܵܐ ܚܲܢܵܢܵܐ: ܘܲܒ݂ܪܵܐ ܡܪܲܚܡܵܢܵܐ. ܘܪܘܼܚܵܐ ܡܪܲܚܦܵܢܵܐ: ܡܩܲܕܫܵܢܵܐ ܕܛܲܡ̈ܐܹܐ. ܩܲܕܸܫ ܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܦܬܲܚ ܠܝܼ ܬܲܪ̈ܥܹܐ ܕܙܲܕܝܼܩܘܼܬ݂ܵܐ. ܓܲܙܵܐ ܕܥܘܼܕ݂ܪ̈ܵܢܹܐ: ܬܲܪܥܵܐ ܗ݇ܘ ܕܲܚܢܵܢܵܟ݂. ܦܬܲܚ ܠܲܢ ܕܢܸܥܘܿܠ ܒܹܗ: ܪܵܚܹܡ ܬܲܝܵܒܹ̈ܐ. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܠܵܟ݂ ܒܲܠܚܘܿܕܲܝܟ ܚܛܹܝܬ݂. ܕܲܚܛܹܝܬ݂ ܝܵܕܲܥ ܐ݇ܢܵܐ: ܝܲܬ݁ܝܼܪ ܡܢ ܟܠܢܵܫ. ܘܩܲܠܝܼܠ ܡܢ ܣܘܿܓ݂ܐܵܐ: ܕܚܵܘܒܲܝ̈ ܠܵܐ ܦܸܪܥܹ̇ܬ݂. ܚܵܐܹܢ ܚܵܘܒܹ̈ܐ ܕܡܵܝܘܿܬܹ̈ܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝ. ܡܸܛܠ ܕܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܛܵܒ݂ ܐܲܢ݇ܬ݁. ܛܵܒ݂ܵܐ ܕܲܫܦܝܼܥܵܐ ܗ݇ܝ: ܪܲܒ݁ܘܼܬ݂ ܛܲܝܒ݁ܘܼܬܹܗ. ܥܲܠ ܟܠܗܘܿܢ ܐܲܝܠܹܝܢ: ܕܲܒ݂ܬܲܪܥܹܗ ܢܵܩܫܝܼܢ. ܦܬܲܚ ܠܝܼ ܬܲܪܥܹܗ ܕܲܚܢܵܢܵܟ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝ. ܘܚܲܛܵܝܹ̈ܐ ܠܘܵܬ݂ܵܟ݂ ܢܸܬ݂ܦܢܘܿܢ. ܡܵܟ݂ܣܹ̈ܐ ܘܙܵܢ̈ܝܵܬ݂ܵܐ: ܪܲܥܝܘܼܟ݂ ܦܵܪܘܿܩܲܢ. ܘܒܲܚܢܵܢܵܟ݂ ܐܲܓ݂ܙܲܪܬ݁: ܫܪܵܝܵܐ ܠܚܵܘ̈ܒܲܝܗܘܿܢ. ܐܵܦ ܠܝܼ ܚܘܼܢܲܝܢܝ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝ. ܥܘܼܕ݂ܪܵܢܲܢ ܒܲܫܡܹܗ ܕܡܵܪܝܵܐ. ܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܐܲܫܦܲܥ ܥܲܠ ܥܲܒ݂ܕܲܝ̈ܟ: ܕܠܵܟ݂ ܡܵܪܝ ܡܸܬ݂ܟܲܫܦܝܼܢ. ܕܐܲܢ݇ܬ݁ܘܼ ܐܝܼܬܲܝܟ ܐܝܼܵܠܲܢ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܡܸܛܠ ܕܫܵܡܲܥ ܡܵܪܝܵܐ ܠܡܸܣܟܹ݁ܢܹ̈ܐ. ܫܵܡܲܥ ܒܵܥܘܼܬ݂ܵܐ: ܕܐܲܝܠܹܝܢ ܕܩܵܪܹܝ̇ܢ ܠܹܗ. ܘܥܵܒܹ݁ܕ݂ ܨܸܒ݂ܝܵܢܵܐ: ܕܟ݂ܠ ܕܬ݂ܵܝܒ݁ܝܼܢ ܠܘܵܬܹܗ. ܫܡܲܥ ܡܵܪܝ ܒܵܥܘܼܬ݂ ܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܬܸܥܘܿܠ ܩܕ݂ܵܡܲܝܟ ܨܠܘܿܬ݂ܝ. ܨܠܘܿܬܲܢ ܬܸܒ݂ܣܲܡ ܠܵܟ݂: ܛܵܒ݂ܵܐ ܐܲܠܵܗܲܢ. ܘܐܲܝܟ݂ ܦܝܼܪܡܵܐ ܕܲܟ݂ܝܵܐ: ܩܕ݂ܵܡܲܝܟ ܬܸܬ݂ܩܲܒܲܠ. ܘܒ݂ܵܗ̇ ܢܸܬ݂ܦܲܢ̈ܝܵܢ ܫܹܐ̈ܠܵܬܲܢ: ܡܢ ܓܲܙܵܐ ܕܪ̈ܲܚܡܲܝܟ. ܩܲܕܝܼܫܘܼ ܘܲܕ݂ܚܝܼܠ ܫܡܹܗ. ܩܘܼܕܵܫܹ̈ܐ ܘܗܘܼܠܵܠܹ̈ܐ: ܕܟܸܢܫܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܘܒ݂ܵܥܘܼܬܹܗ ܕܲܒ݂ܪܵܐ: ܕܡܢ ܝܲܡܝܼܢ ܐܲܒ݂ܵܐ. ܢܦܝܼܣܘܼܢ ܚܠܵܦܲܝܢ ܠܟܹܐܢܘܼܬ݂ܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܫܡܲܥ ܐܲܠܵܗܵܐ ܩܵܠܝ ܟܲܕ݂ ܐܸܬ݂ܟܲܫܲܦ ܠܵܟ݂. ܫܡܲܥ ܩܵܠ ܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬܲܢ: ܘܦܲܢܵܐ ܫܹܐ̈ܠܵܬܲܢ. ܘܐܲܫܪܵܐ ܫܲܝܢܵܟ݂ ܒܲܝܢܵܬܲܢ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܐܸܬ݁ܕܲܟ݂ܪ ܥܹܕܬܵܟ݂ ܕܲܩܢܲܝܬ݁ ܡܢ ܩܕ݂ܝܼܡ. ܫܲܝܸܢ ܡܵܪܝ ܥܹܕܬܵܟ݂: ܦܪܝܼܩܲܬ݂ ܒܲܨܠܝܼܒܵܟ݂. ܕܗܵܐ ܫܵܓ݂ܫ̈ܵܢ ܠܵܗ̇: ܪ̈ܘܼܚܹܐ ܕܕܲܠܩܘܼܒ݂ܠܵܐ. ܢܲܛܲܪܹܝܗ̇ ܡܵܪܝ ܡܢ ܢܸܟ݂ܝܵܢܹ̈ܐ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܹܝܗ̇. ܒܲܕܲܪ ܥܲܡܡܹ̈ܐ ܕܨܵܒܹ݁ܝܢ ܩܪܵܒ݂ܵܐ. ܫܲܝܸܢ ܒܲܨܠܝܼܒ݂ܵܟ݂: ܠܥܹܕܬܵܟ݂ ܡܗܲܝܡܲܢܬܵܐ. ܘܒܲܛܸܠ ܡܢ ܓܵܘܵܗ̇ ܣܸܕ݂ܩܹ̈ܐ ܘܦܘܼܠܵܓܹ̈ܐ. ܘܢܲܛܲܪ ܝܲܠܕܹܝ̈ܗ̇ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܹܝܗ̇. ܐܸܬ݂ܦܲܢ ܥܠܲܝ ܘܪܲܚܸܡ ܥܠܲܝ. ܚܘܼܪ ܒܝܼ ܘܲܦܪܘܿܩܲܝܢܝ: ܡܢ ܒܝܼܫܵܐ ܕܣܲܩܪܲܢܝ. ܘܐܲܡܝܼܢ ܛܵܡܲܪ ܠܝܼ: ܦܲܚܹ̈ܐ ܕܲܚܛܝܼܬ݂ܵܐ. ܐܲܢ݇ܬ݁ ܡܵܪܝ ܦܲܨܵܢܝ ܡܢ ܢܸܫܒܵܘ̈ܗܝ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝ. ܐܲܠܵܗܵܐ ܦܲܨܵܢܝ ܡܢ ܐܝܼܕܹܗ ܕܪܲܫܝܼܥܵܐ. ܦܲܨܵܢܝ ܡܢ ܚܹܐܦܹܗ: ܕܒ݂ܝܼܫܵܐ ܢܵܟ݂ܘܿܠܵܐ. ܘܚܲܪܲܪ ܠܲܛܠܝܼܡܘܼܬ݂ܝ: ܡܸܢܹܗ ܕܡܵܪܘܿܕ݂ܵܐ. ܕܐܲܢ݇ܬ݁ܘܼ ܥܵܢܹ̇ܐ ܠܐܲܠܝܼܨܹ̈ܐ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝ. ܒܪܲܡܫܵܐ ܘܲܒ݂ܨܲܦܪܵܐ ܘܲܒ݂ܛܲܗܪܵܐ. ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡܸܚܛܵܐ ܚܵܛܹܐ ܐ݇ܢܵܐ. ܘܣܸܦ̈ܘܵܬ݂ ܡܸܫܬܵܘ̈ܕܝܵܢ: ܕܠܲܡܚܵܪ ܬܵܐܹܒ݂ ܐ݇ܢܵܐ. ܥܲܕܲܪ ܕܵܘܝܘܼܬ݂ܝ ܚܲܢܵܢܵܐ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝ. ܡܵܐ ܛܵܒ݂ ܘܡܵܐ ܫܲܦܝܼܪ. ܫܲܝܢܵܐ ܘܲܫܠܵܡܵܐ: ܘܚܘܼܒܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ. ܐܲܫܪܵܐ ܒܲܝܢܵܬܲܢ: ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܘܒܲܛܸܠ ܡܸܢܲܢ ܚܸܪ̈ܝܵܢܹܐ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܟܠܗܘܿܢ ܒܚܸܟ݂ܡܬ݂ܵܐ ܒܪܲܝܬ݁ ܐܸܢܘܿܢ. ܐܲܠܵܗܵܐ ܕܲܒ݂ܪܵܐ: ܐܝܼܡܵܡܵܐ ܘܠܸܠܝܵܐ. ܫܡܲܥ ܡܵܪܝ ܨ̈ܠܵܘܵܬܲܢ: ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬܲܢ. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܢܩܲܕܸܡ ܐܲܦܵܘ̈ܗܝ ܒܬ݂ܵܘܕܝܼܬ݂ܵܐ. ܬܵܘܕܝܼܬ݂ܵܐ ܠܪ̈ܲܚܡܲܝܟ: ܡܵܪܵܐ ܡܪܲܚܡܵܢܵܐ. ܕܲܦܬ݂ܝܼܚ ܒܹܝܬ݂ ܓܲܙܵܟ݂: ܠܐܲܝܠܹܝܢ ܕܩܵܪܹܝܢ ܠܵܟ݂. ܩܲܒܸ݁ܠ ܒܵܥܘܼܬ݂ ܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܡܢ ܦܘܼܡܵܐ ܕܲܥܠܲܝܡܹ̈ܐ ܘܕܲܛܠܵܝܹ̈ܐ. ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܐܲܒ݂ܵܐ: ܘܠܲܒ݂ܪܵܐ ܕܡܢ ܥܘܼܒܹ݁ܗ. ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܣܸܓ݂ܕܬ݂ܵܐ ܘܪܘܼܡܪܵܡܵܐ. ܕܗܘܝܘܼ ܡܚܲܣܸܐ ܚܵܘ̈ܒܵܬܲܢ: ܘܡܲܥܒܲܪ ܣܲܟ݂̈ܠܘܵܬܲܢ. ܕܐܘܼܪܚܵܐ܀ ܐܘܼܪ̈ܚܵܬ݂ܵܟ݂ ܡܵܪܝܵܐ ܚܵܘܵܢܝ. ܥܲܠ ܣܲܒ݂ܪܵܟ݂ ܡܵܪܲܢ: ܒܐܘܼܪܚܵܐ ܚܵܙ̇ܩܝܼܢܲܢ. ܘܠܲܝܬ݁ ܠܲܢ ܫܲܒܵܠܵܐ: ܐܸܠܵܐ ܐܸܢ ܪ̈ܲܚܡܲܝܟ. ܚܵܘܵܢ ܐܘܼܪܚܵܐ ܕܢܹܐܙܲܠ ܒܵܗ̇: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܕܲܒ݂ܨܝܼܪܘܼܬ݂ ܡܸܛܪܵܐ܀ ܬܸܥܘܿܠ ܩܕ݂ܵܡܲܝܟ ܨܠܘܿܬ݂. ܨܠܘܿܬܹܗ ܕܐܹܠܝܵܐ: ܕܒ݂ܵܗ̇ ܐܸܬ݂ܟ݁ܠܝܼ ܡܸܛܪܵܐ. ܗܝ ܬܦܝܼܣܵܟ݂ ܡܵܪܲܢ: ܕܲܬ݂ܫܲܕܲܪ ܡܸܛܪܵܐ. ܕܲܒ݂ܬܲܪ̈ܬܲܝܗܹܝܢ ܬܸܫܬܲܒܲܚ: ܐܲܢ݇ܬ݁ ܪܵܚܹ̇ܡ ܐ݇ܢܵܫܵܐ. ܕܲܢܚܵܘܸܐ ܓܲܢ݇ܒܵܪܘܼܬܹܗ. ܡܫܝܼܚܵܐ ܕܲܒ݂ܠܲܚܡܵܐ: ܐܲܫܠܸܡ ܠܲܢ ܦܲܓ݂ܪܹܗ: ܫܲܪܬܲܚ ܡܵܙܘܿܢܲܢ: ܡܸܛܠ ܐ݇ܪܵܙ ܦܲܓ݂ܪܵܟ݂. ܘܲܡܠܝܼ ܦܘܼܡܲܢ ܬܵܘܕܝܼܬ݂ܵܐ ܕܐܲܢ݇ܬ݁ܘܼ ܙܵܐܹ̇ܢ ܟܠ. ܕܡܲܪܝܲܡ܀ ܡܸܛܠ ܕܗܘܝܘܼ ܡܵܪܹܟ݂ܝ ܣܓ݂ܘܿܕ݂ܝ ܠܹܗ. ܐܘܿ ܡܲܪܝܲܡ ܐܸܡܹܗ: ܕܡܲܠܟܵܐ ܡܠܸܟ݂ ܡܲܠܟܹ̈ܐ. ܐܲܦܝܼܣܝ ܠܲܡܫܝܼܚܵܐ: ܕܲܕ݂ܢܲܚ ܡܢ ܥܘܼܒܹ݁ܟ݂ܝ. ܕܲܢܚܘܼܣ ܥܠܲܝܢ ܒܛܲܝܒ݁ܘܼܬܹܗ: ܘܢܲܫܘܹܝܢ ܠܡܲܠܟ݁ܘܼܬܹܗ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܛܲܪ ܪ̈ܲܚܡܲܝܟ ܠܩܲܪ̈ܝܼܒܲܝܟ. ܒܲܨܠܘܿܬ݂ ܩܲܕܝܼܫܲܝ̈ܟ: ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܢܒ݂ܝܼܹ̈ܐ ܘܲܫܠܝܼܚܹ̈ܐ: ܘܣܵܗܕܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ. ܢܲܛܲܪ ܠܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܘܒܹܗܘ ܢܸܚܕܸܐ ܠܸܒܲܢ. ܨܠܝܼܒ݂ܵܟ݂ ܩܲܕܝܼܫܵܐ: ܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ. ܕܒܹܗܘ ܐܸܬ݂ܚܲܪܲܪ: ܓܸܢܣܲܢ ܡܵܝܘܿܬ݂ܵܐ. ܢܸܗܘܸܐ ܫܘܼܪܵܐ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܡܢ ܟܠ ܢܸܟ݂ܝܵܢ̈ܝܼܢ. ܕܐܲܒ݂ܘܼܢ܀ ܒܥܝܼ ܡܢ ܡܵܪܝܵܐ ܘܨܲܠܵܐ ܩܕ݂ܵܡܵܘܗܝ. ܐܲܒ݂ܘܼܢ ܩܲܕܝܼܫܵܐ: ܗܘܝܼ ܠܲܢ ܗܲܕܝܼܐܵܐ. ܒܕ݂ܘܼܒܵܪܹ̈ܐ ܬܲܩܢܹ̈ܐ܆ ܕܲܡܪܲܥܹܝܢ ܠܡܵܪܵܟ݂. ܕܒܲܨ̈ܠܵܘܵܬ݂ܵܟ݂ ܢܸܬ݂ܥܲܕܲܪ: ܘܥܲܡܵܟ݂ ܢܸܬ݂ܒܲܣܲܡ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܠܵܐ ܬܸܫܕܹܝܢܝ ܡܢ ܩܕ݂ܵܡܲܝܟ. ܒܝܵܘܡܵܐ ܕܓܸܠܝܵܢܵܟ݂: ܝܼܫܘܿܥ ܦܵܪܘܩܲܢ. ܢܲܚܸܡ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܚܲܣܵܐ ܒܝܲܕ݂ ܪ̈ܲܚܡܲܝܟ. ܠܥܲܒ݂ܕܲܝ̈ܟ ܕܲܫܟܸܒ݂ܘ ܥܲܠ ܣܲܒ݂ܪܵܟ݂: ܘܐܵܘܕܝܼܘ ܒܲܨܠܝܼܒ݂ܵܟ݂. ܘܲܥܒܸܕ݂ ܥܲܡܝ ܐܵܬ݀ܵܐܛܵܒ݂ܬ݂ܵܐ. ܒܝܵܘܡܵܐ ܕܡܹܐܬ݂ܝܼܬ݂ܵܟ݂:ܡܢܲܚܡܵܢܵܐ ܕܡܝܼ̈ܬܹܐ. ܗܲܒ݂ܠܲܢ ܕܢܸܦ̮ܓܲܥ ܒܵܟ݂: ܒܐܲܦܹ̈ܐ ܓܲܠ̈ܝܵܬ݂ܵܐ. ܘܢܸܚܸܐ ܒܛܲܝܒ݁ܘܼ ܒܡܲܠܟ݁ܘܼܬ݂ܵܟ݂: ܡܸܛܠ ܫܘܼܘܕܵܝܵܟ݂܀
ܕܥܲܢܝܼܕܹ̈ܐ. ܢܸܫܬܲܥܘܿܢ ܕܲܒ݂ܩܲܒ݂ܪܹ̈ܐ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܥܲܢܝܼ̈ܕܹܐ ܕܡܢ ܩܲܒ݂ܪܹ̈ܐ. ܩܵܥܹܝܢ ܒܩܵܠܵܐ ܕܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܟܲܕ݂ ܐܵܡܪܝܼܢ. ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ. ܕܗܘ ܡܢܲܚܸܡ ܠܲܢ ܡܢ ܥܲܦܪܵܐ. ܫܲܒܲܚܘ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ ܚܕܲܬܵܐ. ܦܲܓ݂ܪܹܗ ܕܐܵܕ݂ܵܡ ܕܲܒ݂ܠܝܼ. ܘܨܘܼܪ̈ܵܬܹܗ ܚܲܒܸ݁ܠ ܡܵܘܬܵܐ ܒܝܲܕ݂ ܪܲܦܝܘܼܬܹܗ. ܥܬ݂ܝܼܕ݂ ܗ݇ܘܼ ܡܫܝܼܚܵܐ ܕܲܢܢܲܚܡܝܼܘܗܝ. ܘܲܢܩܝܼܡܝܼܘܗܝ ܡܢ ܝܲܡܝܼܢܹܗ.
ܕܲܨܠܝܼܒ݂ܵܐ. ܨܠܝܼܒ݂ܵܐ ܕܢܘܼܗܪܵܐ ܕܐܸܬ݂ܚܵܘܝܼ. ܠܩܘܼܣܛܲܢܛܝܼܢܘܿܣ ܒܲܫܡܲܝܵܐ. ܐܲܝܟ݂ ܪܲܒ݁ ܚܲܝܠܵܐ ܐܵܙܹܠ ܗ݇ܘܵܐ. ܒܪܹܫ ܡܲܫܪܝܼܬܹܗ ܠܲܩܪܵܒ݂ܵܐ. ܘܙܵܥ ܘܐܸܬ݂ܒܲܠܗܝܼ ܓܘܼܕܲܝ̈ܗܘܿܢ. ܕܚܲܢܦܹ̈ܐ ܣܵܓ݂ܕܲܝ̈ ܠܒܸܪ̈ܝܵܬ݂ܵܐ. ܘܐܲܪܦ̮ܝܼܘ ܛܘܼܥܝܲܝ ܕܚܲܢܦܘܼܬ݂ܵܐ. ܘܲܣܓܸܕ݂ܘ ܠܲܨܠܝܼܒ݂ܵܐ ܘܝܲܩܪܘܼܗܝ.
ܥܵܠܲܡ: ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܠܸܠܝܵܐ ܕܟ݂ܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܫܚܝܼܡܹ̈ܐ: ܠܒܲܪ ܡܢ ܨܵܘܡܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܐ:
ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܚܘܼܣ ܥܲܠ ܥܲܡܵܟ݂: ܘܠܵܐ ܬܸܬܸ݁ܠ ܝܵܪܬ݁ܘܼܬ݂ܵܟ݂ ܠܚܸܣܕܵܐ ܘܡܘܼܝܵܩܵܐ ܕܥܲܡܡܹ̈ܐ. ܚܘܼܪ ܡܢ ܡܸܕ݂ܝܵܪ̈ܲܝܟ ܘܲܚܙܝܼ ܩܛܝܼܪܵܐ: ܕܣܵܒ݂ܠ̇ܝܼܢܲܢ ܡܢ ܒܥܸܠܕ݁ܒ݂ܵܒܹ̈ܐ. ܘܬܸܕ݂ܢܲܚ ܠܲܢ ܒܲܥܓܲܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܕܠܵܐ ܢܹܐܡܪܘܼܢ ܥܲܡܡܹ̈ܐ ܐܲܝܟܵܘ ܐܲܠܵܗܗܘܿܢ

܀

 ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܢܲܛܲܪ ܠܥܲܒ݂ܕܲܝ̈ܟ: ܡܢ ܣܵܢܐܹ̈ܐ ܕܡܲܓܵܢ ܕܨܵܒܹܝ̇ܢ ܠܡܸܚܛܲܦ ܕܠܵܐ ܕܝܼܠܗܘܿܢ. ܣܘܼܓ݂ ܒܐܲܦܲܝ̈ܢ ܩܘܼܫܬܵܟ݂ ܚܲܣܝܼܢܵܐ: ܕܢܸܚܙܘܿܢ ܣܵܢܐܲܝ̈ܢ ܘܢܸܒ݂ܗܬ݂ܘܼܢ. ܘܐܲܟ݂ܪܸܟ݂ ܠܲܢ ܫܘܼܪܵܐ ܒܛܝܼܠܘܼܬ݂ܵܟ݂: ܕܢܸܣܬܲܬܲܪ ܒܓ݂ܵܘܹܗ ܘܢܵܘܕܸ݁ܐ ܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂

܀

ܒܲܨܠܘܿܬ݂ܵܗ̇ ܕܝܵܠܹܕ݂ܬ݂ܵܟ݂ ܡܵܪܲܢ ܝܼܫܘܿܥ: ܫܲܝܸܢ ܥܵܠܡܵܐ ܟܠܹܗ ܕܲܫܓ݂ܝܼܫ ܘܲܕ݂ܠܝܼܚ ܒܲܚܛܵܗܵܘ̈ܗܝ. ܘܒܲܛܸܠ ܡܢ ܐܲܪܥܵܐ ܩܪ̈ܵܒܹܐ ܘܩܹܐܪ̈ܣܹܐ: ܘܫܲܝܸܢ ܟܵܗܢܹ̈ܐ ܥܲܡ ܡܲܠܟܹ̈ܐ. ܕܒ݂ܵܐܘܝܘܼܬ݂ܵܐ ܘܚܘܼܒܵܐ ܢܸܙܕܲܝܲܚ: ܕܘܼܟ݂ܪܵܢ ܝܵܠܹܕ݂ܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ

܀

 ܙܲܕ݁ܝܼܩܹ̈ܐ ܕܲܫܦܲܪܘ ܠܵܟ݂ ܡܵܪܲܢ ܝܼܫܘܿܥ: ܢܒ݂ܝܼܹ̈ܐ ܫܠܝܼܚܹ̈ܐ ܘܣܵܗܕܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ ܕܲܒ݂ܟ݂ܠ ܐܲܬܲܪ. ܗܸܢܘܿܢ ܢܦܝܼܣܘܼܢܵܟ݂ ܚܠܵܦ ܢܲܦ̮̈ܫܵܬܲܢ: ܕܬܸܥܒܸ݁ܕ݂ ܪ̈ܲܚܡܹܐ ܥܲܠ ܟܠܲܢ. ܘܐܲܫܘܵܐ ܠܲܢ ܕܥܲܡܗܘܿܢ ܢܵܘܕܸ݁ܐ ܠܵܟ݂: ܒܝܵܘܡܵܐ ܕܒܹܗ ܢܵܣ̇ܒ݁ܝܼܢ ܐܲܓ݂ܪܵܐ ܕܥܲܡܠܲܝ̈ܗܘܿܢ

܀

ܛܘܼܒܲܝܟ ܠܵܟ݂ ܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ: ܕܫܲܡܠܝܼܬ݁ ܪܲܗܛܵܟ݂ ܒܙܵܟ݂ܘܼܬ݂ܵܐ. ܘܲܦܠܲܚܬ݁ ܒܟܲܪܡܹܗ ܕܲܡܫܝܼܚܵܐ: ܘܲܢܣܲܒ݂ܬ݁ ܐܲܓ݂ܪܵܐ ܚܠܵܦ ܥܲܡܠܲܝ̈ܟ: ܕܹܝܢܵܪ ܛܘܼܒܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܨ̈ܠܵܘܵܬ݂ܵܟ݂ ܢܸܗܘ̈ܝܵܢ ܥܲܠ ܟܠܲܢ: ܘܢܸܫܬ݁ܘܸܐ ܕܥܲܡܵܟ݂ ܢܹܐܪܲܬ݂ ܡܲܠܟ݁ܘܼܬ݂ܵܐ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܒܵܚܵܐ: ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܢ ܥܸܠܵܝܹ̈ܐ: ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܢ ܬܲܚܬܵܝܹ̈ܐ: ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܘܗܸܕ݂ܪܵܐ ܘܪܘܼܡܪܵܡܵܐ: ܡܢ ܟܠ ܕܲܒ݂ܪܲܝܬ݁ ܘܐܲܬ݂ܩܸܢܬ݁ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ. ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܫܘܼܒܵܚܹ̈ܐ ܘܬܸܫ̈ܒ݁ܚܵܬ݂ܵܐ ܕܡܸܬ݂ܐܲܡܪܝܼܢ ܒܟ݂ܠܵܗ̇ ܫܲܢ݇ܬܵܐ: ܒܟ݂ܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܫܚܝܼܡܹ̈ܐ ܕܫܲܒ݁ܬ݂ܵܐ.
   ܫܘܼܒܵܚܵܐ. ܡܵܪܝܵܐ ܡܲܢܘܼ ܢܸܥܡܲܪ ܒܡܲܫܟܢܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.ܓ: ܙܲܒ݂ܢܝܼ̈ܢ* ܘܡܲܢܘܼ ܢܸܫܪܸܐ ܒܛܘܼܪܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ* ܐܲܝܢܵܐ ܕܲܡܗܲܠܸܟ݂ ܕܠܵܐ ܡܘܼܡ ܘܥܵܒܸ݁ܕ ܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ* ܘܲܡܡܲܠܸܠ ܩܘܼܫܬܵܐ ܒܠܸܒܹܗ ܘܠܵܐ ܢܲܟ݁ܘܼܠܬ݂ܵܢ ܒܠܸܫܵܢܹܗ* ܘܠܵܐ ܥܵܒܹ݁ܕ݂ ܠܚܲܒ݂ܪܹܗ ܒܝܼܫܬܵܐ* ܘܫܘܼܚܕܵܐ ܥܲܠ ܩܲܪܝܼܒܹܗ ܠܵܐ ܡܩܲܒܸ݁ܠ* ܘܡܲܣܠܲܝ ܒܥܲܝ̈ܢܵܘܗܝ ܡܲܪܓ݁ܙܵܢܵܐ* ܘܲܠܕ݂ܵܚܠܵܘܗܝ ܕܡܵܪܝܵܐ ܡܝܲܩܲܪ* ܝܵܡܹܐ ܠܚܲܒ݂ܪܹܗ ܘܠܵܐ ܡܕܲܓܸܠ* ܘܟܸܣܦܹܗ ܒܪܸܒ݁ܝܼܬ݂ܵܐ ܠܵܐ ܝܵܗܹܒ݂* ܘܫܘܼܚܕܵܐ ܥܲܠ ܙܲܟܵܝܵܐ ܠܵܐ ܢܵܣܒ݂* ܡܢ ܕܥܵܒܹ݁ܕ݂ ܗܵܠܹܝܢ ܟܹܐܝܢ ܗ݇ܘ* ܘܠܵܐ ܡܸܬ݁ܬ݁ܙܝܼܥ ܠܥܵܠܲܡ*
ܫܲܒܲܚ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.ܓ: ܙܲܒ݂ܢܝܼ̈ܢ܀ ܥܵܠܲܡ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.ܓ: ܙܲܒ݂ܢܝܼ̈ܢ܀

ܟܵܪܘܿܙܘܼܬ݂ܵܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.܀
ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ݁ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡܠܲܟ݂ ܛܵܒ݂̈ܵܬ݂ܵܐ ܠܪ̈ܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕ݁ܬܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܠܵܐ ܨܵܒܹܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ: ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ܡܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܗܘܝܘ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܪ̈ܝܵܬ݂ܵܐ: ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܝܼܣ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܝܼܣ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܵܘ̇ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
