ܐܲܪܒܲܥܒ݁ܫܲܒܵܐ ܕܲܬ݂ܠܵܬܼܵܐ ܕܐܹܠܝܼܵܐ.
ܕܩܲܕ݂ܡܵܝܹ̈ܐ
ܩܲܕ݂ܡܵܝܑܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ. ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܖ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ. ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܢܵܘܕܸ݁ܐ ܡܵܪܝ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܡܵܪܘܼܬ݂ܵܟ݂ ܘܫܘܼܒ݂ܚܵܐ ܐܲܡܝܼܢܵܐ ܕܠܵܐ ܫܲܠܘܵܐ ܢܲܣܸܩ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܡܲܖ̈ܡܝܵܬ݂ܵܐ* ܠܐܲܠܵܗܵܐ ܡܣܲܟ̇ܝܵܐ ܢܲܦ̮ܫܝ ܕܡ̣ܢܗ ܗܘ݀ ܦܘܼܪܩܵܢܝ* ܠܵܟ݂ ܝܵܐ̇ܝܵܐ ܬܸܫܒ̇ܘܿܚܬܵܐ ܐܲܠܵܗܵܐ ܒܨܸܗܝܘܿܢ* ܚܙܝܼ: ܡܙܡܘܖ̈ܐ ܣܒ‐ܣܙ
ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܬܲܖ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܫܘܼܒܼܚܵܐ ܠܵܟܼ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܫܘܼܒܼܚܵܐ ܠܵܟܼ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ.
ܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ؛ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܘܐܵܡܲܪ ܟܵܗܢܵܐ ܨܠܘܿܬܼܵܐ: ܘܥܲܠ ܐܲܦܲܝ̈ ܟܠܗܘܿܢ ܥܘܼܕ݂ܖ̈ܵܢܲܝܟ ܘܛܲܝ̈ܒ݁ܘܵܬ݂ܵܟ݂ ܕܲܠܘܵܬܲܢ ܕܠܵܐ ܡܸܬ݂ܦܲܪܥܝܼܢ: ܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ ܒܥܹܕ݁ܬܵܟ݂ ܡܟܲܠܲܠܬܵܐ ܡܲܠܝܲܬ݂ ܟܠ ܥܘܼܕ݂ܖ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܵܐ ܘܒ݂ܵܪܘܿܝܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܬܲܖ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
(ܓܝܼܘܿܪܵܐ ܕܲܒ݂ܥܹܕ݇ܬܵܐ.( ܚܕ݂ܝܼܬ݂ ܟܲܕ݂ ܐܵܡ̇ܪܝܼܢ ܗ݇ܘ̣ܵܘ ܠܝܼ: ܠܒܲܝܬܹ݁ܗ ܕܡܵܪܝܵܐ ܐܵܙ̇ܠ݇ܝܼܢܲܢ(ܘܲܠܒܲܪ ܡ̣ܢ ܥܹܕ݁ܬܵܐ ܐܵܡ̇ܪܝܼܢ) ܒܟ݂ܠܕܘܼܟ ܐܝܼܬܲܝܟ ܐܲܠܵܗܵܐܩܲܒܸ݁ܠ ܬܲܟ݂ܫܲܦ̮ܬܲܢ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ:ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ:ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܟ.ܐܲܢ݇ܬ݁ܘܼ ܡܵܪܝ ܫܲܪܝܼܪܵܐܝܼܬ݂ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܘܲܡܢܲܛܪܵܢܵܐ ܐܲܡܝܼܢܵܐ ܕܚܲܝܲܝ̈ܢ: ܘܠܵܟ݂ ܡܵܪܝ ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܡ
(ܡܙ؛ ܟܓ؛ 1-8)
ܡܵܪܝܵܐ ܢܸܪܥܹܝܢܝ ܘܡܸܕܸ݁ܡ ܠܵܐ ܢܚܲܣܲܪ ܠܝܼ܀ ܗܲܠܸܠܘܼܝܲܗ ܗܲܠܸܠܘܼܝܲܗ ܐܹܗܲܠܸܠܘܼܝܲܗ. * ܘܥܲܠ ܡܲܖ̈ܓܹܐ ܕܥܘܼܫܢܵܐ ܢܲܫܪܹܝܢܝ. * ܘܥܲܠ ܡܲܝ̈ܵܐ ܢܝܼܚܹ̈ܐ ܢܕܲܒ̇ܪܲܢܝ.* ܢܲܦ̮ܫܝ ܐܲܦܢ̣ܝܼ ܘܕܲܒ̇ܪܲܢܝ ܒܲܫܒ̣ܝܼ̈ܠܲܝ ܩܘܼܫܬܵܐ.* ܡܸܛܠ ܫܡܵܟ݂ ܐܵܦܸܢ ܐܹܗܲܠܸܟ݂.* ܒܢܲܚ̈ܠܲܝ ܛܸܠܵܠܲܝ̈ ܡܵܘܬܵܐ.* ܠܵܐ ܐܸܕ݂ܚܲܠ ܡ̣ܢ ܒܝܼܫܬܵܐ ܡܸܛܠ ܕܐܲܢ݇ܬ ܥܲܡܝ.* ܫܲܒ̣ܛܵܟ݂ ܘܚܘܼܛܪܵܟ݂ ܗܸܢ̣ܘܿܢ ܒܲܝܐ̣ܘܼܢܝ.ܫܲܒܲܚ܀ ܥܵܠܲܡ܀
ܘܬ݂ܵܢܹܝܢ ܒܢܲܝ̈ ܓܘܼܕܵܐ ܐ݇ܚܪܹܬ݂ܵܐ: ܐܹܗܲܠܸܠܘܼܝܵܗ ܀
ܕܗܵܘ̇ ܕܲܡܫܲܪܸܐ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ . ܘܬ݂ܵܢܹܝܢ ܒܢܲܝ̈ ܓܘܼܕܵܐ ܐ݇ܚܪܹܬ݂ܵܐ: ܐܹܗܲܠܸܠܘܼܝܵܗ ܀
ܥܘܿܢ̈ܝܵܬ݂ܵܐ ܕܲܩܕ݂ܵܡ: ܒܚܲܢܵܢܵܐ ܘܲܡܠܸܐ.
ܒܐܝܼܡܵܡܵܐ ܘܲܒ݂ܠܸܠܝܵܐ ܀ ܒܟ݂ܠܙܒܲܢ ܘܲܒ݂ܟ݂ܠܥܸܕܵܢ. ܨܠܘܿܬ݂ܵܗ̇ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܡܲܪܝܲܡ: ܐܸܡܹܗ ܕܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܫܘܼܪܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ. ܬܢܝܼ. ܫܲܒܲܚ. ܢܒ݂ܝܼܹ̈ܐ ܨܲܠܵܘ ܥܲܠ ܫܲܝܢܵܐ: ܘܲܫܠܝܼܚܹ̈ܐ ܥܲܠ ܫܠܵܡܵܐ. ܘܣܵܗܕܹ̈ܐ ܘܟ݂ܵܗܢܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ: ܨܠܘܿܬ݂ܟ݂ܘܿܢ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܲܢ. ܥܵܠܲܡ. ܒܥܝܼ ܘܐܲܦܝܼܣ ܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ: ܠܲܡܫܝܼܚܵܐ ܕܲܪܚܸܡ̣ܬ݁ ܚܘܼܒܹ݁ܗ. ܕܟܸܢܫܵܐ ܕܙܲܝܲܚ̣ ܕܘܼܟ݂ܪܵܢܵܟ݂: ܢܸܬ݂ܥܲܕܲܪ ܒܲܨ̈ܠܵܘܵܬ݂ܵܟ݂. ܢܹܐܡܲܪ. ܫܵܡܲܥ̇ܨ̈ܠܵܘܵܬ݂ܵܐ ܕܥܲܒ݂ܕܵܘ̈ܗܝ: ܡܦܲܢܸܐ ܫܹܐ̈ܠܵܬ݂ ܣܵܓ݂ܘܿܕ݂ܵܘ̈ܗܝ. ܫܡܲܥ ܨܠܘܿܬܲܢ ܘܩܵܠ ܒܵܥܘܼܬܲܢ: ܘܦܲܢܵܐ ܒܖ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܡܵܪܝܵܐ ܩܪܹܝ̇ܬ݂ܵܟ݂ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܖ̈ܵܝܹܐ: ܠܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ: ܘܠܲܒ݂ܛܝܼܠܘܼܬܹܗ ܕܨܸܒ݂ܝܵܢܵܟ݂ ܕܲܥܠܲܝܢ: ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܘܐܵܡܪܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܖ̈ܵܝܹܐ ܡܵܪܝܵܐ ܩܪܹܝ݁ܬ݂ܵܟ̣. ܚܙܝܼ: ܡܙܡܘܖ̈ܐ ܕܡܪܝܐ ܕܩܪܝܬܟ

	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܒ݂ܵܬܲܪ: ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܖ̈ܵܝܹܐ. ܫܡܲܥ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܒܲܚܢܵܢܵܟ݂: ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܒܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂: ܘܚܘܼܢ ܠܚܲܛܵܝܘܼܬܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܖ̈ܲܚܡܲܝܟ: ܐܵܣܝܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܫܘܼܪܵܝܵܐ ܕܒ݂ܵܬܲܪ
(ܡܙ؛ ܟܕ؛ 1-10)
ܕܡܵܪܝܵܐ ܗ݇ܝ̣ ܐܲܪܥܵܐ ܒܲܡܠܹܐܵܗ̇* ܗܲܠܸܠܘܼܝܲܗ ܗܲܠܸܠܘܼܝܲܗ ܐܹܗܲܠܸܠܘܼܝܲܗ..* ܬܹܒܹܝܠ ܘܟ̣ܠܗܘܿܢ ܥܵܡܘܿܖܹ̈ܝܗ̇. * ܡܸܛܠ ܕܗܘ̤ ܒܝܲܡܵܐ ܣ̣ܵܡ ܫܲܬ̣ܐܣܹ̈ܝܗ̇.* ܘܲܒ̣ܢܲܗܖ̈ܵܘܵܬ̣ܵܐ ܐܲܬ̣ܩ̣ܢܵܗ̇.* ܡܲܢ̣ܘܼ ܢܸܣܲܩ ܠܛܘܼܪܹܗ ܕܡܵܪܝܵܐ.* ܘܡܲܢ̣ܘܼ ܢܩܘܼܡ ܒܛܘܼܪܹܗ ܩܲܕܝܼܫܵܐ.* ܐܲܝܢܵܐ ܕܕܲܟ̣̈ܝܵܢ ܐܝܼܕ݂ܵܘ̈ܗܝ ܘܲܓ̣ܒܸܐ ܒܠܸܒܹܗ.* ܘܠܵܐ ܝܵܡܹ̇ܐ ܒܢܲܦ̮ܫܹܗ ܒܕܲܓܵܠܘܼܬ̣ܵܐ ܘܠܵܐ ܝܵܡܹ̇ܐ ܒܢܸܟ̣ܠܵܐ* ܗܵܢܵܐ ܢܩܲܒܸ̇ܠ ܒܘܼܪܟܬ̣ܵܐ ܡ̣ܢ ܡܵܪܝܵܐ.* ܘܙܲܕܝܼܩܘܼܬ̣ܵܐ ܡ̣ܢ ܐܲܠܵܗܵܐ ܦܵܪܘܿܩܲܢ܀ ܫܲܒܲܚ܀ ܥܵܠܲܡ܀
ܕܗܵܘ̇ ܕܲܡܫܲܪܸܐ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ . ܘܬ݂ܵܢܹܝܢ ܒܢܲܝ̈ ܓܘܼܕܵܐ ܐ݇ܚܪܹܬ݂ܵܐ: ܐܹܗܲܠܸܠܘܼܝܵܗ ܀
ܥܘܿܢ̈ܝܵܬ݂ܵܐ ܕܒ݂ܵܬܲܪ: ܒܟ݂ܠ ܢܸܫܡܵܐ.
ܘܲܡܠܵܬ̤ ܐܲܪܥܵܐ ܡ̣ܢܗ̇܀ ܘܐܸܬ݂ܟܲܣܝܼܘ ܛܘܼܖܹ̈ܐ ܡ̣ܢ ܛܸܠܵܠܵܗ̇. ܡܲܪܝܲܡ ܕܝܸܠܕܲܬ̤ ܣܲܡ ܚܲܝܘܼܬ݂ܵܐ ܠܝܲܠܕܵܘ̈ܗܝ ܕܐܵܕ݂ܵܡ: ܒܝܲܕ݂ ܒܵܥܘܼܬܹܟ݂ܝ ܢܸܫܟܲܚ ܖ̈ܲܚܡܹܐ ܒܝܘܿܡ ܢܘܼܚܵܡܵܐ. ܬܢܝܼ. ܫܲܒܲܚ. ܢܸܒ݂ܥܵܐ ܕܚܲܝܹ̈ܐ ܣ̣ܵܡܬ݁ ܐܸܢܘܿܢ ܠܲܢ ܡܵܪܝܵܐ: ܠܟ݂ܠ ܫܲܪܝܼܖܹ̈ܐ ܒܢܲܝ̈ ܐ݇ܖ̈ܵܙܲܝܟ ܕܐܲܚܸܒ݂ܘ ܫܡܵܟ݂. ܥܵܠܲܡ. ܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܪܵܚܡܹܗ ܕܲܒ݂ܪܵܐ: ܒܥܝܼ ܡ̣ܢ ܡܵܪܵܟ݂ ܚܢܵܢܵܐ ܘܖ̈ܲܚܡܹܐ ܥܲܠ ܢܲܦ̮̈ܫܵܬܲܢ. ܢܹܐܡܲܪ. ܡܵܪܲܢ ܥܲܕܲܪ ܫܲܝܢܵܐ ܫܲܕܲܪ ܠܒ݂ܝܼܫܹ̈ܐ ܒܲܕܲܪ: ܠܥܹܕ݇ܬܵܟ݂ ܢܲܛܲܪ: ܠܝܲܠܕܹܝ̈ܗ̇ ܚܲܪܲܪ ܡ̣ܢ ܢܸܟ݂ܝܵܢܹ̈ܐ.
(ܘܪܵܡܹܐ ܡܫܲܡܫܵܢܵܐ ܐܘܼܪܵܪܵܐ: ܘܐܵܣܲܪ ܙܘܼܢܵܪܵܐ: ܘܩܵܐܹ̇ܡ ܩܕ݂ܵܡ ܫܟ݂ܝܼܢܬܵܐ ܡܛܲܟ݁ܣܵܐܝܼܬ݂(ܘܐܵܡܲܪ:
ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ: ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܫܵܐ ܘܒܲܚܦܝܼܛܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܐܲܒ݂ܵܐ ܕܖ̈ܲܚܡܹܐ ܘܐܲܠܵܗܵܐ ܕܟ݂ܠܹܗ ܒܘܼܝܵܐܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܦܵܪܘܿܩܲܢ ܘܝܲܨܘܼܦܬ݂ܵܢܲܢ ܘܲܡܦܲܪܢ̱ܣܵܢܵܐ ܕܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ ܘܩܘܼܝܵܡܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ ܘܲܕ݂ܟ݂ܠܗܹܝܢ ܥܹ̈ܕ݂ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܐܲܬ݂ܪܲܢ ܘܥܲܠ ܟܠ ܐܲܬ݂ܖ̈ܵܘܵܢ: ܘܥܲܠ ܐܲܝܠܹܝܢ ܕܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܥܵܡ̇ܪܝܼܢ ܒܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܡܘܼܙܵܓ݂ܵܘ̈ܗܝ ܕܐܵܐܲܪ ܘܟܲܗܝܼܢܘܼܬ݂ܵܐ ܕܫܲܢ݇ܬܵܐ ܘܥܲܠ̈ܠܵܬ݂ܵܐ ܕܦܹܐܖܹ̈ܐ ܘܥܲܠ ܬܘܼܩܵܢܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܡܲܠܟܹ̈ܐ ܐܲܚܝܼܕܲܝ̈ ܫܘܼܠܛܵܢܹܗ ܕܥܵܠܡܵܐ ܗܵܢܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܖ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܩܲܫܝܼܫܹ̈ܐ ܘܲܡܫܲܡܫܵܢܹ̈ܐ ܐܲܖ̈ܬܵܕ݁ܘܿܟ݁ܣܹܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ܗ̇ܘ ܕܥܲܬ݁ܝܼܪ ܒܖ̈ܲܚܡܵܘܗܝ ܘܲܫܦܝܼܥ ܒܲܚܢܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܡ̣ܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ ܐܝܼܬ݂ܵܘܗܝ ܘܲܠܕ݂ܵܪܕܵܖ̈ܝܼܢ ܩܲܝܵܡ ܫܘܼܠܛܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܒܲܟ݂ܝܵܢܹܗ ܐܝܼܬ݂ܵܘܗܝ ܛܵܒ݂ܵܐ ܘܝܵܗܘܿܒ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹܐ ܒܡܵܘܬܹ݁ܗ ܕܚܲܛܵܝܵܐ ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡ̣ܢ ܥܵܘ̣ܠܹܗ ܘܢܸܚܸܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܲܒ݂ܝܲܠܕܹ݁ܗ ܐܵܘ ܕܲܒ݂ܕܸܢܚܹܗ ܐܵܘ ܕܲܒ݂ܨܵܘܡܹܗ ܐܵܘ ܕܲܒ݂ܡܲܥܲܠܬܹ݁ܗ ܐܵܘ ܕܒܲܩܝܵܡܬܹ݁ܗ ܐܵܘ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܐܵܘ ܕܲܒ݂ܡܲܚܲܬ݁ܬܹ݁ܗ ܐܵܘ ܕܒܲܨܠܝܼܒܹܗ: ܩܲܕ݁ܝܼܫܵܐ (ܐܵܘ ܩܲܕܝܼܫܬܵܐ) ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܐܲܦܨܲܚ ܠܲܫܡܲܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܠܹܗ ܗܘ̤ ܟܝܵܢܵܐܝܼܬ݂ ܐܝܼܬ݂ ܠܵܐ ܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܘܝܲܬ݁ܝܼܪܵܐܝܼܬ݂ ܕܐܲܝܠܹܝܢ ܕܲܡܗܲܝܡܢܝܼܢ ܒܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܣܓܵܐ ܥܲܡܲܢ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀
 ܘܡܵܐ ܕܫܸܠܡܲܬ̤ ܡܲܩܸܦ: ܢܲܦ̮̈ܫܵܬܲܢ ܘܕܲܚܕ݂ܵܕܹ̈ܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܲܓ݂ܥܸܠ.
ܘܥܵܢܹܝ̇ܢ: ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ.
	ܘܲܡܨܲܠܸܐ ܟܵܗܢܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ. ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ ܡܲܓ݂ܥ̱ܠܝܼܢܲܢ ܦܲܓ݂ܖ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ: ܘܡܸܢܵܟܼ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܫܵܐ̇ܠܝܼܢܲܢ ܫܘܼܒ݂ܩܵܢܵܐ ܕܚܵܘܒܹ̈ܐ ܘܕܲܚܛܵܗܹ̈ܐ: ܗܲܒ݂ܠܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܖ̈ܲܚܡܲܝܟ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ ܟܵܪܘܿܙܘܼܬ݂ܵܐ: ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ.
ܘܐܵܡ̇ܪܝܼܢ ܒܩܵܠܵܐ ܦܫܝܼܛܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܫܲܒܲܚ܀ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܥܵܠܲܡ܀ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܒ݂ܵܬܲܪ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܖ̈ܵܝܹܐ. ܩܲܕ݁ܝܼܫܵܐ ܘܲܫܒ݂ܝܼܚܵܐ ܘܚܲܝܠ̱ܬ݂ܵܢܵܐ ܘܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܕܲܒ݂ܩܲܕ݁ܝܼܫܹ̈ܐ ܥܵܡܲܪ ܘܡܸܬ݁ܬ݁ܢܝܼܚ ܨܸܒ݂ܝܵܢܹܗ: ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ ܐܸܬ݂ܦܲܢ ܡܵܪܝ: ܘܚܘܼܣ ܘܪܲܚܸܡ ܥܠܲܝܢ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ. ܒܲܪܸܟ݂ ܡܵܪܝ: ܐܲܪܟܸ݁ܢܘ ܖܹ̈ܫܲܝܟ݁ܘܿܢ ܠܲܣܝܵܡܝܼܕ݂ܵܐ ܘܩܲܒܸ݁ܠܘ ܒܘܼܪܟ݁ܬ݂ܵܐ.
ܘܐܵܡܲܪ ܪܲܒܵܐ. ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܬܸܫܡܸܫܬܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
	ܘܡܲܩܸܦ ܨܠܘܿܬ݂ܵܐ. ܘܟܲܕ݂ ܫܵܠ̈ܡܵܢ ܢܲܦ̮̈ܫܵܬܲܢ ܠܲܚܕ݂ܵܐ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܓܡܝܼܪܬܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܟܠܲܢ ܒܲܚܕ݂ܵܐ ܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ: ܢܸܫܬ݁ܘܸܐ ܕܢܲܣܸܩ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ... ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܥܘܿܢ̈ܝܵܬ݂ܵܐ ܕܪܲܡܫܵܐ. ܒܠܵܐ ܒܵܗܬ݁ܝܼܢܲܢ.
ܩܲܕ݁ܝܼܫܘܼ ܒܹܝܬ݂ ܡܲܫܪ̱ܝܹܗ ܕܲܡܪܲܝܡܵܐ܀ ܠܡܸܠܬܹܗ ܕܐܲܠܵܗܵܐ ܐܸܫܲܒܲܚ. ܠܗܲܝܟ݁ܠܹܗ ܕܐܲܠܵܗܵܐ ܡܸܠܬ݂ܵܐ: ܒܫܘܼܒ݂ܚܵܐ ܪܲܒܵܐ ܙܲܝܚܲܬ̤ ܡܲܪܝܲܡ ܒܓ݂ܵܘ ܥܘܼܒܵܗ̇: ܘܲܗܘ̤ܵܬ̣ ܠܹܗ ܐܸܡܵܐ ܘܐܲܡܬ݂ܵܐ: ܠܝܼܫܘܿܥ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܵܐ. ܘܡܸܛܠ ܗܵܕܹܐ ܟܠ ܒܸܖ̈ܝܵܬ݂ܵܐ: ܗܵܐ ܚܵܕ݂̈ܝܵܢ ܒܝܘܿܡ ܥܲܕ݂ܥܹܕ݂ܵܗ̇: ܘܲܙܡܝܼ̈ܢܵܢ ܠܲܓ݂ܢܘܿܢ ܢܘܼܗܪܵܐ: ܠܒ݂ܘܼܣܵܡܵܐ ܕܠܵܐ ܫܘܼܠܵܡܵܐ. ܘܟ݂ܠܲܢ ܥܲܡ ܟܠ ܫܲܖ̈ܒ݂ܵܬ݂ܵܐ: ܛܘܼܒ݂ܵܐ ܪܓ݂ܝܼܓ݂ܵܐ ܢܸܬܸ݁ܠ ܠܵܗ̇: ܘܫܘܼܒ݂ܚܵܐ ܠܗ̇ܘ ܕܲܓ݂ܒ݂ܵܗ̇: ܥܘܼܡܪܵܐ ܠܨܲܠܡܹܗ ܡܫܲܒ݁ܚܵܐ. ܬܢܝܼ. ܫܲܒܲܚ. ܠܡܲܖ̈ܓܵܢܝܵܬ݂ܵܐ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܕܵܡܹܝ̇ܢ ܓܲܖ̈ܡܹܐ ܕܩܲܕ݁ܝܼܫܹ̈ܐ: ܕܲܩܒ݂ܝܼ̈ܥܵܢ ܒܲܟ݂ܠܝܼܠ ܡܲܠܟܵܐ: ܘܫܘܼܦܪ̱ܗܹܝܢ ܡܲܦܪܸܓ݂ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܬܵܘ ܟܠܲܢ ܒܚܘܼܒܵܐ ܫܲܠܡܵܐ: ܢܝܲܩܲܪ ܝܵܘܡܵܐ ܕܕ݂ܘܼܟ݂ܪܵܢܗܘܿܢ: ܒܲܙܡܝܼܖ̈ܵܬ݂ܵܐ ܕܪܘܼܚ ܩܘܼܕ݂ܫܵܐ: ܡ̣ܢ ܨܲܦܪܵܐ ܥܕܲܡܵܐ ܠܪܲܡܫܵܐ. ܘܢܸܒ݂ܥܸܐ ܖ̈ܲܚܡܹܐ ܡ̣ܢ ܡܵܪܗܘܿܢ: ܕܢܲܫܪܸܐ ܫܲܝܢܹܗ ܒܲܒ݂ܪܝܼܬ݂ܵܐ: ܘܲܒ݂ܝܲܕ݂ ܨܠܵܘܵܬ݂̈ܗܘܿܢ: ܥܹܕ݁ܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇ ܬܸܬ݂ܢ̱ܛܲܪ. ܥܵܠܲܡ. ܒܪܝܼܟ݂ ܕܘܼܟ݂ܪܵܢܵܟ݂ ܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ: ܕܡܸܛܠ ܫܪܵܪܵܐ ܐܸܬ݂ܪ̱ܕܸܦܬ݁: ܘܣܲܝܒܲܪܬ݁ ܚܲܫܹ̈ܐ ܘܐܘܼܠܨܵܢܹ̈ܐ: ܕܬܸܗܘܸܐ ܝܵܪܬܵܐ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ. ܡܲܢ̣ܘܼ ܡܸܫܟܲܚ ܕܲܢܬܲܢܸܐ: ܕܘܼܒܵܪ ܥܲܡܠܲܝ̈ܟ ܖ̈ܘܼܚܵܢܹܐ: ܕܲܒ݂ܢܲܟ݂ܦܘܼܬ݂ܵܐ ܡܨܲܒܲܬ݂ ܗ̣݇ܘܲܝܬ݁: ܒܫܲܗܪܵܐ ܘܨܵܘܡܵܐ ܘܲܨܠܘܿܬ݂ܵܐ. ܨܠܘܿܬ݂ܵܟ݂ ܬܸܗܘܸܐ ܒܹܝܬ݂ ܓܵܘܣܵܐ: ܠܚܲܛܵܝܹ̈ܐ ܕܒ݂ܵܟ݂ ܡܸܬ݂ܓܵܘܣܝܼܢ: ܘܢܸܫܬ݁ܘܸܐ ܠܡܲܣܵܩܘܼ: ܫܘܼܒ݂ܚܵܐ ܠܡܵܪܵܟ݂ ܕܐܵܘܪ̱ܒ݂ܵܟ݂. ܢܹܐܡܲܪ. ܐܲܝܟ݂ ܗ̇ܘ ܦܝܼܪܡܵܐ ܕܩܲܪܸܒ݂ ܐܲܗܪܘܿܢ: ܪܹܝܚܹܗ ܕܟܸܢܫܲܢ ܢܸܒ݂ܣܲܡ ܠܵܟ݂: ܘܐܲܝܟ݂ ܒܵܥܘܼܬ݂ܵܐ ܕܢܝܼܢܘܵܝܹ̈ܐ: ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܩܲܒܸ݁ܠ ܡܵܪܝ. ܘܦܲܢܵܐ ܒܖ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ: ܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܥܲܬ݁ܝܼܪܵܐ: ܘܐܲܝܟ݂ ܕܲܥ̣ܢܲܝܬܵܝܗܝ ܠܕ݂ܵܢܝܼܐܹܝܠ: ܡ̣ܢ ܓܘܼܒܵܐ ܕܐܲܖ̈ܝܵܘܵܬ݂ܵܐ. ܥܢܝܼ ܡܵܪܝ ܘܥܲܕܲܪ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܒܗܵܠܹܝܢ ܙܲܒ݂ܢܹ̈ܐ ܕܐܘܼܠܨܵܢܵܐ: ܕܲܥܠܲܝܟ ܐܝܼܬ݂ܵܘܗܝ ܬܘܼܟ݂ܠܵܢܲܢ: ܡܵܪܵܐ ܪܵܚܹ̇ܡ ܥܲܒ݂ܕܵܘ̈ܗܝ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܟ݂ܠܗܘܿܢ ܐܲܪܒܲܥܒ݁ܫܲܒܹ̈ܐ ܕܟ݂ܠܵܗ̇ ܫܲܢ݇ܬܵܐ. ܙܲܝܸܢ ܠܲܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܒܙܲܝܢܵܐ ܚܲܣܝܼܢܵܐ ܘܠܵܐ ܡܸܙܕܲܟ݂ܝܵܢܵܐ: ܒܲܨ̈ܠܵܘܵܬ݂ܵܗ̇ ܕܐܸܡܵܟ݂ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܲܥܒܸܕ݂ ܠܲܢ ܥܲܡܵܗ̇ ܡܢܵܬ݂ܵܐ ܘܫܵܘܬܵܦܘܼܬ݂ܵܐ ܒܲܓ݂ܢܘܿܢܵܟ݂ ܫܡܲܝܵܢܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ. ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.

ܒܲܪܸܟܼ ܡܵܪܝ
ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܫܡܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܩܲܕ݁ܝܼܫܵܐ ܢܸܫܬܲܒܲܚ: ܘܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܬܸܣܬ݁ܓܸܕ݂: ܘܡܵܪܘܼܬ݂ܵܟ݂ ܬܸܬ݂ܝܲܩܲܪ: ܘܪܲܒ݁ܘܼܬ݂ܵܟ݂ ܬܸܬ݂ܢܲܨܲܚ: ܘܐܝܼܬ݂ܘܼܬ݂ܵܟ݂ ܬܸܬ݂ܪܲܡܪܲܡ: ܘܥܲܠ ܥܲܡܵܟ݂ ܘܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬ݂ܵܟ݂: ܢܓ݂ܘܼܠܘܼܢ ܖ̈ܲܚܡܹܐ ܡܬ݂ܘܿܡܵܝܹ̈ܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
	ܚܵܬ݂ܵܗ̇. ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܒܪܝܼܟ݂ܵܐ ܡܵܪܝ ܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܲܣܓ݂ܝܼܕ݂ܵܐ ܡܵܪܘܼܬ݂ܵܟ݂: ܩܲܕ݁ܝܼܫ ܘܲܫܒ݂ܝܼܚ ܘܲܡܫܲܒܲܚ: ܘܪܵܡ ܘܲܡܥܲܠܲܝ ܫܡܵܐ ܣܓ݂ܝܼܕ݂ܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܣܵܗܕܹ̈ܐ.
ܡܸܛܠܵܬ݂ܵܟ݂ ܐܸܬ݂ܩܲܛܲܠ̄ܢ ܟܠܝܘܿܡ.ܣܵܗܕܹ̈ܐ ܩܲܕ݁ܝܼܫܹ̈ܐ ܕܐܸܬ݂ܩܲܛܲܠܘ: ܡܸܛܠ ܚܘܼܒܹ݁ܗ ܕܲܡܫܝܼܚܵܐ. ܡܦܝܼܣܝܼܢܲܢ ܥܲܠ ܟܠܲܢ: ܒܥܵܘ ܖ̈ܲܚܡܹܐ ܡ̣ܢ ܐܲܠܵܗܵܐ. ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ. ܢܵܨܚ̇ܝܼܢ ܥܲܡܠܲܝ̈ܟ݁ܘܿܢ ܘܲܡܚܲܕܸ݁ܐ ܠܲܢ: ܥܘܼܗܕܵܢܵܐ ܕܬܲܟ݂ܬ݁ܘܼܫܲܝ̈ܟ݁ܘܿܢ. ܕܲܚܠܵܦ ܡܫܝܼܚܵܐ ܐܸܬ݂ܩܲܛܲܠܬ݁ܘܿܢ: ܘܥܲܡܹܗ ܒܪܵܘܡܵܐ ܬܲܡܠ̱ܟ݂ܘܼܢ. ܒܥܵܘ ܠܡܵܪܝܵܐ ܘܐܸܬ݂ܚܲܝܲܠܘ.ܣܵܗܕܹ̈ܐ ܒܥܵܘ ܖ̈ܲܚܡܹܐ ܥܲܠ ܥܵܠܡܵܐ: ܕܐܲܚܝܼܕ݂ ܓܵܘܣܵܐ ܒܓܲܖ̈ܡܲܝܟ݁ܘܿܢ. ܕܲܒ݂ܝܲܕ݂ ܒܵܥܘܼܬ݂ܟ݂ܘܿܢ ܘܲܨ̈ܠܵܘܵܬ݂ܟ݂ܘܿܢ: ܢܸܫܟܲܚ ܖ̈ܲܚܡܹܐ ܒܝܘܿܡ ܕܝܼܢܵܐ. ܡܵܐ ܛܵܒ݂ ܘܡܵܐ ܫܲܦܝܼܪ. ܫܲܝܢܵܐ ܘܲܫܠܵܡܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ: ܡܠܸܐ ܐܲܬ݂ܪܵܐ ܕܲܫ̣ܪܹܝܬ݁ܘܿܢ ܒܹܗ. ܘܡ̣ܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܢܸܬ݂ܢ̱ܛܲܪ: ܒܲܨ̈ܠܵܘܵܬ݂ܟ݂ܘܿܢ ܛܘܼܒ݂ܵܢܹ̈ܐ. ܐܹܒܲܪܸܟ݂ ܠܡܵܪܝܵܐ ܒܟ݂ܠܙܒܲܢ. ܒܪܝܼܟ݂ܘܼ ܕܲܥ̣ܒܲܕ݂ܟ݂ܘܿܢ ܛܘܼܒ݂ܵܢܹ̈ܐ: ܐܵܣܵܘ̈ܵܬ݂ܵܐ ܫܲܪܝܼܖܹ̈ܐ. ܘܡܲܪܕܹ݁ܝܢ ܓܲܖ̈ܡܲܝܟ݁ܘܿܢ ܥܘܼܕ݂ܖ̈ܵܢܹܐ: ܠܐܲܝܢܵܐ ܕܲܒ݂ܗܘܿܢ ܡܸܬ݂ܓܵܘܲܣ. ܡܲܫܪܝܼܬ݂ܵܐ ܕܡܲܠܲܐܟ݂ܵܘ̈ܗܝ ܕܡܵܪܝܵܐ. ܓܘܼܕܝܼ̈ܢ ܥܲܡ ܓܘܼܕܝܼ̈ܢ ܥܵܐ̇ܠܝܼܢ ܗ݇ܘ̣ܵܘ: ܣܵܗܕܹ̈ܐ ܩܕ݂ܵܡ ܖ̈ܵܕ݂ܘܿܦܲܝܗܘܿܢ. ܟܲܕ݂ ܩܵܥܹܝ̇ܢ ܘܐܵܡ̇ܪܝܼܢ ܡܵܪܝܵܐ ܥܲܕܲܪ: ܠܥܲܒ݂ܕܲܝ̈ܟ ܒܙܲܒ݂ܢܹ̈ܐ ܕܐܘܼܠܨܵܢܵܐ. ܚܘܼܪܘ ܠܘܵܬܹܗ ܘܣܲܒܲܪܘ ܒܹܗ. ܚ̣ܙܵܘ ܠܲܡܫܝܼܚܵܐ ܘܚܘܼܒܹ݁ܗ ܪܚܸܡ̣ܘ: ܕܲܚܠܵܦ ܥܹܕ݁ܬܹܗ ܡܵܘܬܵܐ ܛ̣ܥܸܡ. ܘܲܪܗܸܛ̣ܘ ܘܐܲܫ̣ܠܸܡܘ ܢܲܦ̮̈ܫܵܬ݂ܗܘܿܢ: ܕܢܸܗܘܘܿܢ ܥܲܡܹܗ ܡܵܐ ܕܡܸܬ݂ܓܠܸܐ. ܕܲܒܲܚܘ ܠܹܗ ܕܸܒ݂ܚܹ̈ܐ ܕܫܘܼܒ݂ܚܵܐ. ܦܝܼܖ̈ܡܹܐ ܕܬܲܪܥܘܼܬ݂ܵܐ ܘܩܘܼܖ̈ܒܵܢܹܐ: ܕܟܲܝ̈ܵܐ ܗ̤ܘܲܝܬ݁ܘܿܢ ܛܘܼܒ݂ܵܢܹ̈ܐ. ܕܲܚ̣ܕ݂ܝܼ ܒܟ݂ܘܿܢ ܐܲܒ݂ܵܐ ܘܩܲܒܸ݁ܠܟ݂ܘܿܢ ܒܪܵܐ: ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܟܲܠܸܠ̣ܟ݂ܘܿܢ. ܡܫܲܒܲܚ ܒܲܡܖ̈ܵܘܡܹܐ ܡܵܪܝܵܐ. ܠܥܸܠ ܒܲܡܖ̈ܵܘܡܹܐ ܢܛܝܼܪ ܫܘܼܒ݂ܚܵܐ: ܕܠܵܒ݂ܫ̇ܝܼܢ ܣܵܗܕܹ̈ܐ ܒܢܘܼܚܵܡܵܐ. ܒܐܲܬ݂ܪܵܐ ܫܒ݂ܝܼܚܵܐ ܡܠܸܐ ܛܘܼܒܹ̈ܐ: ܕܪܵܡܝܼܢ ܚܲܝܵܘ̈ܗܝ ܡ̣ܢ ܩܸܢܛܵܐ. ܘܲܪܓ݂ܝܼܓ݂ܝܼܢ ܡ̣ܢ ܕܲܗܒ݂ܵܐ ܘܡ̣ܢ ܟܹܐܦܹ̈ܐ ܛܵܒ݂̈ܵܬ݂ܵܐ. ܟܠܝܼ̈ܠܹܐ ܕܲܡܝܲܬ݁ܪܝܼܢ ܡ̣ܢ ܕܲܗܒ݂ܵܐ: ܘܝܵܩܘܿܢܕܹ̈ܐ ܘܡܲܖ̈ܓܵܢܝܵܬ݂ܵܐ. ܣܵܐܹ̇ܡ ܒܖܹ̈ܫܲܝܟ݁ܘܿܢ ܡܫܝܼܚܵܐ: ܘܲܡܚܲܕܸ݁ܐ ܠܟ݂ܘܿܢ ܒܲܓ݂ܢܘܿܢܹܗ. ܐܸܬ݂ܚܲܝܲܠܘ ܘܢܸܬ݂ܥܲܫܲܢ ܠܸܒ݁ܟ݂ܘܿܢ. ܠܵܐ ܕܚܸܠ̣ܬ݁ܘܿܢ ܣܵܗܕܹ̈ܐ ܡ̣ܢ ܢܘܼܪܵܐ: ܘܣܲܝܦܹ̈ܐ ܕܚܝܼ̈ܠܹܐ ܕܕܲܝܵܢܹ̈ܐ. ܕܚܘܼܒܹ݁ܗ ܕܲܡܫܝܼܚܵܐ ܠܒ݂ܝܼܫܝܼܬ݁ܘܿܢ: ܘܥܲܠ ܟܠ ܫܸܢܕ݁ܝܼ̈ܢ ܒܲܣܲܪܬ݁ܘܿܢ. ܫܲܒܲܚܘ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ ܚܕܲܬܵܐ. ܬܸܫܒ݁ܘܿܚܬܵܐ ܚܕܲܬܵܐ ܡܲܠܒܸ݁ܫ ܠܗܘܿܢ: ܡܫܝܼܚܵܐ ܠܖ̈ܲܚܡܹܐ ܕܲܓ݂ܒ݂ܵܐ ܠܹܗ. ܘܲܚܠܵܦ ܐܘܼܠܨܵܢܹ̈ܐ ܕܣܲܝܒܲܪܘ ܗ݇ܘ̣ܵܘ: ܓܢܘܿܢܵܐ ܒܪܵܘܡܵܐ ܡܵܘܪܸܬ݂ ܠܗܘܿܢ. ܦܬܲܚܠܝܼ ܬܲܖ̈ܥܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ. ܦܬܲܚܘ ܓܲܙܲܝ̈ܟ݁ܘܿܢ ܛܘܼܒ݂ܵܢܹ̈ܐ: ܘܗܲܒ݂ܘ ܥܘܼܕ݂ܖ̈ܵܢܹܐ ܠܲܣܢܝܼܩܹ̈ܐ. ܕܚܵܝ̇ܪܝܼܢ ܘܲܡܣܲܟܹ݁ܝܢ ܠܲܨܠܘܿܬ݂ܟ݂ܘܿܢ: ܕܒ݂ܵܗ̇ ܢܸܣܬܲܬ݁ܪܘܼܢ ܡ̣ܢ ܒܝܼܫܵܐ. ܛܘܼܒܲܝܗܘܿܢ ܠܐܲܝܠܹܝܢ ܕܲܕ݂ܠܵܐ ܡܘܼܡ ܐܸܢܘܿܢ ܒܐܘܼܪܚܵܐ. ܛܘܼܒܲܝܟ݁ܘܿܢ ܣܵܗܕܹ̈ܐ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܖ̈ܵܚܡܵܘܗܝ ܕܚܲܬ݂ܢܵܐ ܫܡܲܝܵܢܵܐ. ܕܗܵܐ ܡܙܲܡܢܝܼܬ݁ܘܿܢ ܠܡܲܠܟ݁ܘܼܬ݂ܵܐ: ܘܚܲܝܹ̈ܐ ܘܛܘܼܒܹ̈ܐ ܐܲܡܝܼܢܹ̈ܐ. ܘܲܡܗܲܠܟ݂ܝܼܢ ܒܢܵܡܘܿܣܹܗ ܕܡܵܪܝܵܐ. ܛܘܼܒ݂ܵܐ ܕܠܵܐ ܥܵܒܲܪ ܘܡܲܠܟ݁ܘܼܬ݂ܵܐ: ܕܠܵܐ ܥܵܒ݂ܪܵܐ ܘܠܵܐ ܡܸܫܬܲܪܝܵܐ. ܡ̣ܠܲܟ݂ ܦܵܪܘܿܩܲܢ ܠܩܲܕ݁ܝܼܫܵܘ̈ܗܝ: ܕܪܲܚ̣ܡܘܼܗܝ ܘܲܢ̣ܛܲܪܘ ܦܘܼܩܕܵܢܵܘ̈ܗܝ. ܠܥܲܒ݂ܕ݁ܘܼܬ݂ܵܐ ܐܸܙܕܲܒܲܢ ܝܵܘܣܸܦ. ܓܲܖ̈ܡܵܘܗܝ ܕܝܵܘܣܸܦ ܢܲܨܝܼܚܵܐ: ܣ̣ܵܓ݂ܘ ܬܘܼܖ̈ܥܵܬ݂ܵܐ ܕܡܸܨܖ̈ܵܝܹܐ. ܘܓܲܖ̈ܡܲܝܟ݁ܘܿܢ ܣܵܗܕܹ̈ܐ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܗܘ̤ܵܘ ܚܢܵܢܵܐ ܠܟ݂ܠܵܗ̇ ܒܪܝܼܬ݂ܵܐ. ܣܵܡ̣ܬ݁ ܒܪܹܫܹܗ ܟܠܝܼܠܵܐ ܡܫܲܒ݁ܚܵܐ. ܟܠܝܼܠܵܐ ܫܒ݂ܝܼܚܵܐ ܘܝܲܩܝܼܪܵܐ: ܣ̣ܵܡ ܠܹܗ ܡܵܪܵܐ ܕܬܸܫܒ݁ܘܿܚܬܵܐ. ܠܡܵܪܝ ܩܘܿܪܝܵܩܘܿܣ ܢܲܨܝܼܚܵܐ: ܕܣܲܝܒܲܪ ܚܲܫܹ̈ܐ ܚܠܵܦ ܫܪܵܪܵܐ. ܐܹܡܲܠܸܠ ܥܠܲܝܟ݁ܝ ܫܠܵܡܵܐ. ܫܠܵܡ ܠܵܟ݂ ܢܲܨܝܼܚܵܐ ܡܵܪܝ ܦܸܬ݂ܝܘܿܢ: ܕܣܲܝܒܲܪܬ݁ ܚܲܫܹ̈ܐ ܘܐܘܼܠܨܵܢܹ̈ܐ. ܡ̣ܢ ܖ̈ܵܕ݂ܘܿܦܵܘܗܝ ܕܲܫܪܵܪܵܐ: ܡܸܛܠ ܚܘܼܒܹܗ ܕܲܡܫܝܼܚܵܐ. ܒܥܝܼ ܡ̣ܢ ܡܵܪܝܵܐ ܘܨܲܠܵܐ ܩܕ݂ܵܡܵܘܗܝ. ܐܲܦܝܼܣ ܡܵܪܝ ܝܼܫܘܿܥ ܢܲܨܝܼܚܵܐ: ܠܐܲܠܵܗܲܢ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ. ܕܡܵܐ ܕܲܡܩܝܼܡ ܒܢܲܝ̈ ܕܲܚܝܼܚܵܐ: ܢܲܥܠܲܢ ܠܲܓ݂ܢܘܿܢܹܗ ܫܒ݂ܝܼܚܵܐ.ܙܲܕ݁ܝܼܩܵܐ ܘܲܬ݂ܪܝܼܨܵܐ. ܣܵܗܕܵܐ ܡܵܪܝ ܣܲܪܓ̇ܝܼܣ ܕܲܒ݂ܚܲܝܠܹܗ: ܐܲܒ݂ܗܸܬ݂ ܠܒ݂ܝܼܫܵܐ ܘܲܠܦܵܠܚܵܘ̈ܗܝ. ܨܠܘܿܬܹܗ ܬܢܲܛܲܪ ܠܲܟ݂ܢܘܼܫܝܲܢ: ܡ̣ܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹ̈ܗ. ܡܵܪܝܵܐ ܢܸܬܸ݁ܠ ܚܲܝܠܵܐ ܠܥܲܡܹܗ. ܚܲܝܠܵܐ ܕܲܢ̣ܚܸܬ݂ ܠܐܵܓ݂ܘܿܢܵܐ: ܘܢܲܨ̣ܚܹܗ ܠܣܵܗܕܵܐ ܡܵܪܝ ܓܝܼܘܲܪܓ݁ܝܼܣ. ܗܘ̤ ܢܸܗܘܸܐ ܠܲܢ ܢܵܛܘܿܪܵܐ: ܡ̣ܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹ̈ܗ. ܣܹܐܡܵܐ ܓܲܒ݂ܝܵܐ ܕܲܒ݂ܚܝܼܪ ܒܐܲܪܥܵܐ. ܣܝܼܡܬ݂ܵܐ ܕܥܘܼܕ݂ܖ̈ܵܢܹܐ ܐܝܼܬ݂ܵܘܗܝ ܠܲܢ: ܦܲܓ݂ܪܹܟ݂ܝ ܢܲܟ݂ܦܬ݂ܵܐ ܐܘܿ ܡܲܪܝܲܡ. ܕܲܒ݂ܝܲܕ݂ ܥܘܼܕ݂ܖ̈ܵܢܵܘܗܝ ܣܲܓ̇ܝܼ̈ܐܹܐ: ܡܲܥܬܲܪ ܠܵܗ̇ ܠܲܣܢܝܼܩܘܼܬܲܢ. ܫܲܒܲܚ. ܫܘܼܒ݂ܚܵܐ ܠܡܵܪܝܵܐ ܕܐܵܘܪܸܒ݂ܟ݂ܘܿܢ: ܘܲܥ̣ܒܲܕ݂ܟ݂ܘܿܢ ܣܝܼܡܲܬ݂ ܚܲܝܹ̈ܐ. ܕܐܲܝܠܹܝܢ ܕܐܲܠܝܼܨܝܼܢ ܘܲܡܥܵܩܝܼܢ: ܒܟ݂ܘܿܢ ܡܸܬ݂ܓܵܘܣܝܼܢ ܘܡܸܬ݂ܦܲܪܩܝܼܢ. ܥܵܠܲܡ. ܐܲܬ݂ܠܹܝܛܵܐ ܙܪܝܼܙܵܐ ܘܢܲܨܝܼܚܵܐ: ܐܲܒ݂ܘܼܢ ܝܵܪܬܵܐ ܕܡܲܠܟ݁ܘܼܬ݂ܵܐ. ܐܲܦܝܼܣ ܠܲܡܫܝܼܚܵܐ ܕܲܒ݂ܖ̈ܲܚܡܵܘܗܝ: ܢܲܫܪܸܐ ܫܲܝܢܹܗ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܢܹܐܡܲܪ. ܬܸܫܒ݁ܘܿܚܬܵܐ ܕܪܲܡܫܵܐ ܡܲܣܩܝܼܢܲܢ: ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ. ܘܫܘܼܒ݂ܩܵܢܵܐ ܕܚܵܘܒܹ̈ܐ ܫܵܐ̇ܠܝܼܢܲܢ: ܡܸܢܵܟ݂ ܠܲܡܩܲܒܵܠܘܼ܀ ܚܲܝܹ̈ܐ ܕܠܵܐ ܚܛܵܗܹ̈ܐ ܗܲܒ݂ܠܲܢ ܡܵܪܝ: ܚܘܼܒܵܐ ܘܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ. ܘܟ݂ܠܗܹܝܢ ܫܹܐ̈ܠܵܬܲܢ ܢܸܬ݂ܦܲܢ̈ܝܵܢ: ܐܲܝܟ݂ ܕܫܵܦ̇ܪܵܐ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂܀ ܪܸܓ݁ܬ݂ܵܐ ܕܲܨܠܘܿܬ݂ܵܐ ܙܪܘܿܥ ܒܲܢ ܡܵܪܝ: ܘܒ݂ܵܥܘܼܬܲܢ ܨܹܐܕܲܝܟ ܬܸܣܲܩ. 	ܚܒܲܪܬ݂ܵܗ̇. ܒܲܨܠܘܿܬ݂ ܩܲܕ݁ܝܼܫܲܝ̈ܟ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܐܸܬ݂ܪܲܥܵܐ ܠܲܢ: ܘܲܒ݂ܒ݂ܵܥܘܼܬ݂ܗܘܿܢ ܕܫܲܪܝܼܖ̈ܲܝܟ ܐܲܥܒܲܪ ܚܛܵܗܲܝ̈ܢ: ܘܚܲܣܵܐ ܒܘܼܨܵܖ̈ܲܝܢ: ܘܫܲܝܸܢ ܒܥܸܠܕ݁ܒ݂ܵܒܲܝ̈ܢ: ܘܢܲܚܸܡ ܥܲܢܝܼܕܲܝ̈ܢ: ܘܐܲܫܘܵܐ ܠܲܢ ܠܫܘܼܒ݂ܚܵܐ ܦܲܐܝܵܐ ܕܡܲܠܟ݁ܘܼܬ݂ܵܟ݂: ܥܲܡ ܟܹܐܢܹ̈ܐ ܘܙܲܕ݁ܝܼܩܹ̈ܐ ܓܵܡܖ̈ܲܝ ܨܸܒ݂ܝܵܢܵܟ݂ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ: ܡܵܪܵܐ ܕܟ݂ܠ...ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ ܒܲܪܸܟ݂ܡܵܪܝ.

ܐ݇ܚܪܹܬ݂ܵܐ. ܒܝܲܡܝܼܢܵܟ݂ ܡܵܪܝ ܢܲܛܲܪ ܠܲܢ: ܘܲܬ݂ܚܹܝܬ݂ ܟܸܢܦܲܝ̈ܟ ܣܲܬܲܪ ܠܲܢ: ܘܥܘܼܕ݂ܪܵܢܵܟ݂ ܢܸܬ݂ܠܵܘܸܐ ܠܲܢ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܗܲܒ݂ܠܲܢ ܡܵܪܝ ܫܲܝܢܵܐ ܐܲܡܝܼܢܵܐ: ܘܚܘܼܒܵܐ ܘܪܸܚܡܲܬ݂ ܝܘܼܠܦܵܢܵܐ: ܘܚܲܝܹ̈ܐ ܘܛܘܼܒܹ̈ܐ ܘܚܲܕ݂̈ܘܵܬ݂ܵܐ. ܘܡܵܙܘܿܢܵܐ ܕܠܵܐ ܢܸܚܣܲܪ ܠܲܢ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܗܘ̤ܝܼ ܢܵܛܘܿܪܵܐ ܕܠܵܐ ܕܵܡܹܟ݂: ܠܟܲܪܟ݂ܵܐ ܕܥܵܢ̈ܵܟ݂ ܫܲܪܝܵܐ ܒܹܗ: ܕܠܵܐ ܬܸܬ݂ܢ̱ܟܸܐ ܡ̣ܢ ܕܹܐܒܹ̈ܐ ܨܲܗܝܲܝ̈ ܠܲܕ݂ܡܵܐ ܕܲܓ݂ܙܵܪܵܟ݂: ܕܐܲܢ݇ܬ݁ܘܼ ܝܲܡܵܐ ܕܠܵܐ ܚܵܣܲܪ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܒ݂ܘܼܖ̈ܟܵܬ݂ܵܟ݂ ܡܵܪܲܢ ܒܲܪܸܟܲܝܢ: ܘܫܘܼܪ ܒܛܝܼܠܘܼܬ݂ܵܟ݂ ܐܲܟ݂ܪܸܟܲܝܢ: ܘܡ̣ܢ ܛܵܒ݂̈ܵܢ ܠܵܐ ܬܲܨܪܸܟܲܝܢ: ܘܒܲܓ݂ܢܘܿܢ ܢܘܼܗܪܵܟ݂ ܐܲܣܡܸܟܲܝܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܬܵܐ ܡܵܪܝ ܠܐܝܼܵܠܲܢ ܒܲܚܢܵܢܵܟ݂: ܘܐܸܬ݂ܓܲܠܝ ܠܦܘܼܪܩܵܢܲܢ ܒܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂: ܘܲܬ݂ܪܘܿܨ ܗܲܠ̈ܟ݂ܵܬܲܢ ܒܲܫܒ݂ܝܼܠܹ̈ܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܬܸܕ݂ܢܲܚ ܠܲܢ ܡܵܪܝ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܡܵܐ ܕܕ݂ܵܝ̇ܢܵܐ ܠܲܢ ܟܹܐܢܘܼܬ݂ܵܟ݂: ܘܖ̈ܲܚܡܲܝܟ ܢܹܐܬ݂ܘܿܢ ܠܐܝܼܵܠܲܢ: ܒܝܵܘܡܵܐ ܕܕ݂ܵܢ̇ܚܵܐ ܪܲܒ݁ܘܼܬ݂ܵܟ݂: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܘܼܪܟ݁ܬ݂ܵܟ݂ ܘܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܝܲܡܝܼܢܵܟ݂ ܕܡܲܠܝܵܐ ܖ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ: ܐܲܓܸ݁ܢ ܘܐܲܠܘܵܐ ܥܲܠ ܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܕܩܵܪܹܝ̇ܢ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܵܟ݂ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܲܪܸܟ݂ ܡܵܪܝ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܢܲܛܲܪ ܒܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂ ܠܟܸܢܫܵܐ ܕܲܡܗܲܝܡܢܹ̈ܐ: ܕܲܬ݂ܟ݂ܝܼܠܝܼܢ ܥܠܲܝܟ ܘܩܵܪܹܝ̇ܢ ܠܲܫܡܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܵܟ݂ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܗܘ̤ܝܼ ܡܵܪܝ ܥܲܡܲܢ ܒܖ̈ܲܚܡܲܝܟ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܘܕܲܒܲܪ ܠܲܢ ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܘܼܪܟ݁ܬ݂ܵܐ ܡܵܪܝ ܕܖ̈ܲܚܡܹܐ: ܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܕܖ̈ܲܚܡܹܐ: ܒܝܲܕ݂ ܡܲܠܲܐܟ݂ܵܐ ܕܖ̈ܲܚܡܹܐ: ܡ̣ܢ ܠܘܵܬ݂ ܐܲܠܵܗܵܐ ܣܲܓ݁ܝܼ ܖ̈ܲܚܡܹܐ: ܬܸܫܬܲܕܲܪ ܘܬܸܫܬܲܦܲܥ ܥܲܠ ܟܸܢܫܹ̈ܐ ܕܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ: ܕܐܲܡܝܼܢܝܼܢ ܘܩܵܪܹܝ̇ܢ ܠܲܫܡܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ: ܘܚܵܝ̇ܪܝܼܢ ܘܲܡܣܲܟܹ݁ܝܢ ܠܖ̈ܲܚܡܹܐ: ܘܲܠܦܘܼܪܩܵܢܵܐ ܕܡ̣ܢ ܠܘܵܬ݂ܵܟ݂: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܘܡܲܩܦܝܼܢ. ܨܠܘܿܬ݂ܵܗ̇ ܡܵܪܝ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܗ̇ ܕܐܸܡܵܐ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܬܲܟ݂ܫܲܦ̮ܬܵܗ̇ ܘܬܲܚܢܲܢܬܵܗ̇ ܕܡܲܠܝܲܬ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܐ: ܡܵܪܬ݁ܝ ܡܲܪܝܲܡ ܛܘܼܒ݂ܵܢܝܼܬ݂ܵܐ: ܘܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ ܙܲܟܵܝܵܐ: ܘܥܘܼܕ݂ܪܵܢܵܐ ܐܲܠܵܗܵܝܵܐ: ܘܒ݂ܵܥܘܼܬܹܗ ܕܡܵܪܝ ܝܘܿܚܲܢܵܢ ܡܲܥ̱ܡܕ݂ܵܢܵܐ: ܘܕܲܫܠܝܼܚܹ̈ܐ ܛܘܼܒ݂ܵܢܹ̈ܐ: ܘܕܲܐܒ݂ܵܗ̈ܵܬ݂ܵܐ ܢܲܨܝܼܚܹ̈ܐ ܡܣܲܒ݁ܖ̈ܵܢܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܘܙܵܖ̈ܥܲܝ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܕܲܐܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܡܵܪܝ ܐܲܒ݂ܵܐ ܩܵܬ݂ܘܿܠܝܼܩܵܐ: ܘܡܵܪܝ ܐܸܣܛܲܦܵܢܘܿܣ ܒܘܼܟ݂ܪܵܐ ܕܣܵܗܕܹ̈ܐ: ܘܲܕ݂ܓܲܢ݇ܒܵܪ ܚܲܝܠܵܐ ܡܵܪܝ ܓܝܼܘܲܪܓܝܼܣ ܣܵܗܕܵܐ ܢܲܨܝܼܚܵܐ: ܘܲܕ݂ܡܵܪܝ ܦܠܵܢ: ܘܲܕ݂ܡܵܪܝ ܐܵܘܓܹ݁ܝܢ ܛܘܼܒ݂ܵܢܵܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܣܝܼܥܬܹ݁ܗ ܪܘܼܚܵܢܵܝܬܵܐ. ܘܕܲܫܡܘܿܢܝܼ ܘܕܲܒ݂ܢܹܝ̈ܗ̇: ܘܲܕ݂ܟ݂ܠܗܘܿܢ ܣܵܗܕܵܘ̈ܗܝ ܘܩܲܕ݁ܝܼ̈ܫܵܘܗܝ ܕܡܵܪܲܢ: ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܫܘܼܪܵܐ ܪܵܡܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܚܲܣܝܼܢܵܐ: ܕܬܸܦܪܘܿܩ ܘܲܬ݂ܦܲܨܸܐ ܘܲܬ݂ܫܵܘܙܸܒ݂ ܘܲܬ݂ܢܲܛܲܪ ܠܦܲܓ݂ܖ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ ܡ̣ܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹܗ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹ̇ܝܢ: ܐܵܡܹܝܢ.
ܘܟܸܢ ܚܵܬ݂ܡ̇ܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܖ̈ܵܝܹܐ. ܘܢܵܣܹ̇ܒ݂ ܟܵܗܢܵܐ ܨܠܝܼܒ݂ܵܐ ܒܐܝܼܕܹܗ ܕܝܲܡܝܼܢܵܐ: ܘܩܵܐܹ̇ܡ ܒܓܲܒܵܐ ܬܲܝܡ̱ܢܵܝܵܐ: ܘܡܲܗܦܸܟ݂ ܐܲܦܵܘ̈ܗܝ ܠܥܲܡܵܐ ܘܐܵܡܲܪ: ܒܲܪܸܟ݂ ܡܵܪܝ ܒܦܘܼܩܕܵܢܟ݂ܘܿܢ. ܘܥܵܢܹ̇ܝܢ: ܒܦܘܼܩܕܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܡܵܪܲܢ: ܘܫܘܼܒ݂ܚܵܐ ܠܲܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ. ܘܡܲܪܟ݁ܢܝܼܢ ܥܲܡܵܐ ܖܹ̈ܫܲܝܗܘܿܢ.
ܘܡܲܩܸܦ ܚܘܼܬܵܡܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ: ܨܸܡܚܹܗ ܕܐܲܒ݂ܵܐ ܡܬ݂ܘܿܡܵܝܵܐ: ܝܼܠܝܼܕ݂ܵܐ ܕܠܵܐ ܫܘܼܪܵܝܵܐ: ܠܥܸܠ ܡ̣ܢ ܙܲܒ݂ܢܹ̈ܐ ܘܲܗܘܲܝ̈ܵܐ: ܕܠܲܝܬ݁ ܠܲܢ ܣܲܒ݂ܪܵܐ ܘܣܘܼܟܵܝܵܐ: ܐܸܠܵܐ ܐܸܢ ܐܲܢ݇ܬ݁ ܒܵܪܘܿܝܵܐ. ܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܘܲܓ݂ܒܲܝ̈ܵܐ: ܕܲܫ̣ܦܲܪܘ ܠܵܟ݂ ܡ̣ܢ ܫܘܼܪܵܝܵܐ: ܥܒܸܕ݂ ܠܲܚܛܵܗܲܝ̈ܢ ܚܘܼܣܵܝܵܐ: ܘܲܠܣܲܟ݂ܠ̱̈ܘܵܬܲܢ ܫܪܵܝܵܐ: ܘܲܠܥܵܩ̈ܵܬܲܢ ܦܘܼܨܵܝܵܐ: ܘܲܠܫܸܐ̈ܠܵܬܲܢ ܦܘܼܢܵܝܵܐ: ܘܡܲܢܲܥܲܝܢ ܠܢܘܼܗܪܵܟ݂ ܓܲܐܝܵܐ: ܘܢܲܛܲܪܲܝܢ ܒܢܝܼܫܵܟ݂ ܚܲܝܵܐ: ܡ̣ܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܟܣܲܝ̈ܵܐ ܘܲܓ݂ܠܲܝ̈ܵܐ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ). ܗܵܫܵܐ ♰ ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܪܵܫܹ̇ܡ ܨܠܝܼܒ݂ܵܐ ܥܲܠ ܥܲܡܵܐ: ܟܲܕ݂ ܦܵܫܹ̇ܛ ܐܝܼܕܹܗ ܠܥܸܠ ܠܐܲܦܲܝ̈ ܡܲܥ̱ܪܒ݂ܵܐ: ܘܡܲܝܬܸ݁ܐ ܠܵܗ̇ ܠܡܲܕ݂ܢܚܵܐ: ܘܟܸܢ ܦܵܫܹ̇ܛ ܠܵܗ̇ ܠܓܲܒܵܐ ܕܬܲܝ̱ܡܢܵܝܵܐ: ܘܡܵܘܒܸ݁ܠ ܠܵܗ̇ ܠܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ. ܐܲܝܟܲܢܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܸܣܬ݁ܡܸܟ݂ ܥܲܠ ܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܓܲܒܵܐ ܣܸܡܵܠܵܝܵܐ ܕܟܸܢܫܵܐ.
ܘܥܵܢܹ̇ܝܢ: ܡܫܝܼܚܵܐ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬ݂ܵܟ݂: ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܟܵܗܢܘܼܬ݂ܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
ܐ݇ܚܪܹܢܵܐ: ܬܸܬ݂ܩܲܒܲܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܚܲܠܵܫܹ̈ܐ: ܩܕ݂ܵܡ ܒܹܝܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܬܸܗܘܸܐ ܟܢܝܼܫܘܼܬܲܢ ܗܵܕܹܐ ܠܲܢܝܵܚ ܨܸܒ݂ܝܵܢ ܡܵܪܘܼܬ݂ܵܟ݂: ܠܡܸܣܲܒ݂ ܡܸܢܵܟܼ ܒܛܲܝܒ݁ܘܼ ܚܘܼܠܡܵܢܵܐ ܕܦܲܓ݂ܪܵܐ: ܘܢܘܼܛܵܪܵܐ ܕܢܲܦ̮ܫܵܐ: ܘܫܘܼܪܬܵܚܵܐ ܕܡܵܙܘܿܢܵܐ:(ܘܚܘܼܣܵܝܵܐ ܕܚܵܘܒܹ̈ܐ ܘܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ):ܘܫܲܝܢܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ: ܘܲܫܠܵܡܵܐ ܢܲܓ̇ܝܼܪܵܐ: ܘܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ ܕܠܵܐ ܥܵܒ݂ܪܵܐ ܘܠܵܐ ܡܸܫܬܲܪܝܵܐ ܡ̣ܢ ܒܲܝܢܵܬܲܢ: ܒܟ݂ܠܹܗ ܙܲܒ݂ܢܵܐ ܕܩܵܐܹ̇ܡ ܒܹܗ ܥܵܠܡܵܐ ܗܵܢܵܐ: )ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܒܪܝܼܟ݂ ܐܲܠܵܗܵܐ ܠܥܵܠܡܝܼܢ: ܘܲܡܫܲܒܲܚ ܫܡܹܗ ܩܲܕ̇ܝܼܫܵܐ ܠܕ݂ܵܪܕܵܖ̈ܝܼܢ: ܕܡ̣ܢܗ ܒܵܥܹܝ̇ܢܲܢ: ܘܲܠܝܲܡܵܐ ܫܦܝܼܥܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ: ܕܗ̤ܘ ܢܲܫܘܸܐ ܠܲܢ ܠܫܘܼܒ݂ܚܵܐ ܦܲܐܝܵܐ ܕܡܲܠܟ̇ܘܼܬܹܗ: ܘܲܠܒ݂ܘܼܣܵܡܵܐ ܕܥܲܡ ܡܲܠܲܐܟ݂ܵܘ̈ܗܝ ܩܲܕ̇ܝܼܫܹ̈ܐ: ܘܲܠܓܲܠܝܘܼܬ݂ ܐܲܦܹ̈ܐ ܕܲܩܕ݂ܵܡܵܘܗܝ: ܘܲܠܩܵܘܡܵܐ ܕܡ̣ܢ ܝܲܡܝܼܢܹܗ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ: ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܖ̈ܲܚܡܵܘܗܝ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟ̇ܠ ܕܐܸܬ݂ܟܲܢܲܫܢܲܢ ܒܒܲܝܬܹ݁ܗ: ܘܨܲܠܝܼܢܲܢ ܩܕ݂ܵܡ ܪܲܒ݁ܘܼܬܹܗ: ܥܲܠ ܣܲܒ݂ܪܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܗ̤ܘ ܢܸܫܡܲܥ ܨܠܘܿܬܲܢ ܒܲܚܢܵܢܹܗ: ܘܲܢܩܲܒܸ݁ܠ ܒܵܥܘܼܬܲܢ ܒܚܵܘܣܵܢܹܗ: ܘܲܢܫܝܼܓ݂ ܘܢܸܡܪܘܿܩ ܨܵܐܬ݂ܵܐ ܕܚܵܘܒܲܝ̈ܢ ܘܕܲܚܛܵܗܲܝ̈ܢ ܒܙܘܿܦܵܐ ܫܦܝܼܥܵܐ ܕܚܲܝܘܼܣܬ݂ܵܢܘܼܬܹܗ: ܘܲܢܢܝܼܚ ܢܲܦ̮̈ܫܵܬ݂ܵܐ ܕܥܲܢܝܼܕܲܝ̈ܢ ܒܐܵܘܵܢܹ̈ܐ ܫܒ݂ܝܼܚܹ̈ܐ ܕܡܲܠܟ̇ܘܼܬܹܗ: ܘܢܸܪܘܿܣ ܥܲܠ ܟܠܲܢ ܛܲܠܵܐ ܕܒܲܣܝܼܡܘܼܬܹܗ: ܘܲܥܠܲܝܢ ܘܥܲܠ ܟܠ ܒܸܖ̈ܝܵܢ ܬܲܓܸ̇ܢ ܝܲܡܝܼܢܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬܹܗ: ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܖ̈ܲܚܡܵܘܗܝ:(ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟܠ ܕܝܲܗ݇ܒ݂ ܬܸܫܒ݁ܚܵܬܹ̈ܗ ܒܦܘܼܡܲܢ: ܘܲܙܡܝܼܖ̈ܵܬܹܗ ܒܠܸܫܵܢܲܢ: ܘܩܝܼܢܵܬܹ̈ܗ ܒܓܲܓ̇ܖ̈ܵܬܲܢ: ܘܬ݂ܵܘܕ̇ܝܼܬܹܗ ܒܣܸܦ̈ܘܵܬܲܢ: ܘܗܲܝܡܵܢܘܼܬܹܗ ܒܠܸܒܵܘ̈ܵܬܲܢ: ܗ̤ܘ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬܲܢ: ܘܲܢܩܲܒܸ݁ܠ ܒܵܥ̈ܘܵܬܲܢ: ܘܢܸܬ݂ܪܲܥܸܐ ܒܬܲܟ݂ܫ̈ܦܵܬܲܢ: (ܘܲܢܚܲܣܸܐ ܚܵܘ̈ܒܵܬܲܢ:)ܘܲܢܦܲܢܸܐ ܠܲܢ ܫܸܐ̈ܠܵܬܲܢ ܫܲܦܝܼܖ̈ܵܬ݂ܵܐ ܠܵܐ ܥܕ݂ܝܼ̈ܠܵܬ݂ܵܐ. ܘܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܢܲܫܦܲܥ ܖ̈ܲܚܡܵܘܗܝ ܘܲܚܢܵܢܹܗ: ܥܠܲܝܢ ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡ̣ܢ ܣܵܟܹܗ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܢܸܫܬܲܒܲܚ ܫܡܹܗ ܕܐܲܠܵܗܵܐ ܡܵܪܹܟܠ ܒܲܝܢܵܬܲܢ: ܡܛܲܟ̇ܣܵܢܗܘܿܢ ܕܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ. ܘܲܥܠܲܝܢ ܡܚܝܼ̈ܠܹܐ ܘܚܲܛܵܝܹ̈ܐ: ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡ̣ܢ ܣܵܟܹܗ. ܘܥܲܠ ܥܹܕ݁ܬܵܐ ܩܲܕ̇ܝܼܫܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇: ܘܥܲܠ ܐܲܒ݂ܵܗܲܝ̈ܢ ܘܖ̈ܲܒܵܢܲܝܢ ܘܡܲܠܦܵܢܲܝ̈ܢ ܘܥܲܢܝܼܕܲܝ̈ܢ: ܘܥܲܢܝܼܕܹ̈ܐ ܕܲܦܪܲܫ̣ܘ ܘܫܲܢ̣ܝܼܘ ܡ̣ܢ ܒܲܝܢܵܬܲܢ: ܘܥܲܠ ܟܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܬܲܓܸ̇ܢ ܝܲܡܝܼܢܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬܹܗ ܘܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܠܐܲܠܵܗܵܐ ܫܘܼܒ݂ܚܵܐ: ܘܲܠܡܲܠܲܐܟܹ̈ܐ ܐܝܼܩܵܪܵܐ: ܘܲܠܣܵܛܵܢܵܐ ܒܸܗܬ݁ܬ݂ܵܐ: ܘܠܲܨܠܝܼܒ݂ܵܐ ܣܸܓ݂ܕ̇ܬ݂ܵܐ: ܘܲܠܥܹܕ݁ܬܵܐ ܪܘܼܡܪܵܡܵܐ: ܘܲܠܥܲܢܝܼܕܹ̈ܐ ܢܘܼܚܵܡܵܐ: ܘܲܠܬܲܝܵܒܹ̈ܐ ܩܘܼܒܵܠܵܐ: ܘܠܲܐܣܝܼܖܹ̈ܐ ܫܪܵܝܵܐ: ܘܲܠܡܲܖ̈ܥܹܐ ܘܲܟ݂ܖ̈ܝܼܗܹܐܚܘܼܠܡܵܢܵܐ ܘܐܵܣܝܘܼܬ݂ܵܐ: ܘܠܲܐܖ̈ܒܲܥܦܸܢܝܵܬܹܗ ܕܥܵܠܡܵܐ ܫܲܝܢܵܐ ܪܲܒܵܐ ܘܲܫܠܵܡܵܐ. ܘܐܵܦ ܥܠܲܝܢ ܕܐܝܼܬܲܝܢ ܡܚܝܼ̈ܠܹܐ ܘܚܲܛܵܝܹ̈ܐ: ܚܢܵܢܹܗ ܘܖ̈ܲܚܡܵܘܗܝ ܕܐܲܠܵܗܲܢ ܣܓ݂ܝܼܕ݂ܵܐ ܢܹܐܬ݂ܘܿܢ ܘܢܲܓ̇ܢܘܼܢ ܘܢܸܫܬܲܦܥܘܼܢ ܘܲܢܩܵܘܘܿܢ ܘܢܲܡ̱ܠܟ݂ܘܼܢ ܒܐܲܡܝܼܢܘܼܬ݂ܵܐ: )ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: )ܒܝܲܡܝܼܢ ܡܵܪܘܼܬ݂ܵܟ݂ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ): ܒܲܪܸܟ݂ܡܵܪܝ ܠܟ݂ܠܲܢ: ܘܢܲܛܲܪ ܠܟ݂ܠܲܢ: ܘܥܲܕܲܪ ܠܟ݂ܠܲܢ: ܘܣܲܝܲܥ ܘܣܲܬܲܪ ܠܟ݂ܠܲܢ: ܘܢܲܚܸܡ ܥܲܢܝܼܕܹ̈ܐ ܕܟ݂ܠܲܢ: ܘܐܲܓܸ̇ܢ ܝܲܡܝܼܢܵܟ݂ ܥܲܠ ܟܠܲܢ: ܘܖ̈ܲܚܡܲܝܟ ܘܲܚܢܵܢܵܟ݂ ܢܸܫܬܲܦܥܘܼܢ ܥܲܠ ܟܠܲܢ:ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ̇ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ̇ܬ݂ܵܐ: ܘܩܘܼܒܵܠܛܲܝܒ݁ܘܼܬ݂ܵܐ ܐܲܡܝܼܢܬܵܐ ܬܸܣܲܩ ܠܵܟ݂ ܡ̣ܢ ܦܘܼܡ ܟܠܲܢ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟ̇ܠ ܢܒܲܪܸܟ݂ ܠܟܸܢܫܲܢ ܒܛܲܝܒ݁ܘܼܬܹܗ: ܘܲܢܢܲܛܲܪ ܠܲܩܝܵܡܲܢ ܒܣܘܿܓ݂ܐܵܐ ܫܦܝܼܥܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܘܲܢܦܲܢܸܐ ܫܸܐ̈ܠܵܬܲܢ ܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܹܗ: ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡ̣ܢ ܣܵܟܹܗ: ܘܥܲܠ ܥܹܕ݁ܬܵܐ ܩܲܕ̇ܝܼܫܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇: ܘܥܲܠ ܐܲܬ݂ܪܵܐ ܗܵܢܵܐ ܘܥܵܡܘܿܖ̈ܵܘܗܝ: ܘܥܲܠ ܥܘܼܡܪܵܐ ܗܵܢܵܐ ܘܝܲܬܵܒ݂ܵܘ̈ܗܝ: ܘܥܲܠ ܟܠܲܢ ܒܟ݂ܠܲܢ ܟܢܝܼܫܵܐܝܼܬ݂: ܚܢܵܢܹܗ ܘܖ̈ܲܚܡܵܘܗܝ ܕܐܲܠܵܗܲܢ ܛܵܒ݂ܵܐ: ܢܹܐܬ݂ܘܿܢ ܘܢܸܫܬܲܦܥܘܼܢ ܒܐܲܡܝܼܢܘܼ (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܡܲܩܦܝܼܢ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܲܬ݂ܠܵܬ݂ܡܐܵܐ ܘܲܬ݂ܡܲܢܬܲܥܣܲܪ ܐܲܒ݂ܵܗ̈ܵܬ݂ܵܐ ܐܲܦܸܣܩܘܿܦܹ̈ܐ: ܕܐܸܬ݂ܟܲܢܲܫܘ ܒܢܝܼܩܝܼܵܐ ܡܕ݂ܝܼܢ݇ܬܵܐ ܕܒܹܝܬ݂ܘܿܢܝܼܵܐ: ܒܙܲܒ݂ܢܵܐ ܕܩܘܿܣܛܲܢܛܝܼܢܘܿܣ ܡܲܠܟܵܐ ܙܲܟܵܝܵܐ. ܘܥܸܠܬ݂ܵܐ ܕܹܝܢ ܕܲܟ݂ܢܘܼܫܝܼܗܘܿܢ ܡܸܛܠ ܪܘܼܫܥܹܗ ܕܐܲܪܝܼܘܿܣ܀ ܘܐܲܝܟ݂ ܥܝܵܕ݂ܵܐ ܕܹܝܢ ܕܪܵܕܹܐ ܒܲܝܢܵܬ݂ ܒܢܹܝ̈ܗ̇ ܕܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܲܫܠܝܼܚܵܝܬܵܐ ܕܝܼܠܲܢ ܕܡܲܕ݂ܢ̱ܚܵܐ: ܘܲܟ݂ܒܲܪ ܕܡ̣ܢ ܪܵܘܡܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܪܕ݂ܵܐ: ܘܡ̣ܢ ܡܲܫܠ̱ܡܵܢܘܼܬ݂ܵܐ ܘܝܘܼܒܵܠܵܐ ܕܥܲܬ݁ܝܼܩܹ̈ܐ ܡܓ݂ܵܘܝܲܝ̈ ܒܬܹܐܘܿܖ̈ܝܼܵܣ ܕܲܫܪܵܪܵܐ ܐܲܠܵܗܵܝܵܐ ܡܬ݂ܝܼܚ ܗ݇ܘ̣ܵ ܥܕܲܡܵܐ ܠܝܵܘ̈ܡܵܬܲܢ. ܘܗܵܟܲܢܵܐ ܡܫܲܪܹܝܢ ܚ݇ܢܲܢ ܠܡܸܬ݂ܢܵܐ ܩܵܢܘܿܢܵܐ ܕܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܢܝܼܩܝܼܵܐ.

ܒܲܫܪܵܪܵܐ ܓܹܝܪ ܕܠܵܐ ܦܘܼܠܵܓ݂ܵܐ:

ܡܗܲܝܡܢܝܼܢܲܢ ܒܚܲܕ݂ ܐܲܠܵܗܵܐ: ܐܲܒ݂ܵܐ ܐܲܚܝܼܕ݂ ܟܠ. ܥܵܒ݂ܘܿܕ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܐܲܝܠܹܝܢ ܕܡܸܬ݂ܚܲܙ̈ܝܵܢ ܘܲܕ݂ܠܵܐ ܡܸܬ݂ܚܲܙ̈ܝܵܢ. ܘܲܒ݂ܚܲܕ݂ ܡܵܪܝܵܐ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ. ܒܪܹܗ ܕܐܲܠܵܗܵܐ ܝܼܚܝܼܕ݂ܵܝܵܐ: ܒܘܼܟ݂ܪܵܐ ܕܟ݂ܠܗܹܝܢ ܒܸܖ̈ܝܵܬ݂ܵܐ. ܗܵܘ̇ ܕܡ̣ܢ ܐܲܒ݂ܘܼܗܝ ܐܸܬ݂ܝܼܠܸܕ݂ ܩܕ݂ܵܡ ܟܠܗܘܿܢ ܥܵܠܡܹ̈ܐ: ܘܠܵܐ ܐܸܬ݂ܥ̱ܒܸܕ݂. ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܕܡ̣ܢ ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܒܲܪ ܟܝܵܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܕܒ݂ܐܝܼ̈ܕ݂ܵܘܗܝ ܐܸܬ݁ܬܲܩܲܢܘ ܥܵܠܡܹ̈ܐ: ܘܐܸܬ݂ܒ݁ܪܝܼ ܟܠ ܡܸܕ݁ܡ. ܗܵܘ̇ ܕܡܸܛܠܵܬܲܢ ܒܢܲܝ̈ܢܵܫܵܐ: ܘܡܸܛܠ ܦܘܼܪܩܵܢܲܢ: ܢ̣ܚܸܬ݂ ܡ̣ܢ ܫܡܲܝܵܐ: ܘܐܸܬ݂ܓܲܫܲܡ ܡ̣ܢ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܘܲܗܘ̤ܵܐ ܒܲܪܢܵܫܵܐ: ܘܐܸܬ݂ܒ݁ܛܸܢ ܘܐܸܬ݂ܝܼܠܸܕ݂ ܡ̣ܢ ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܘܚܲܫ̣ ܘܐܸܙܕ݁ܩܸܦ: ܒܝܵܘܡܲܝ̈ ܦܲܢܛܝܼܘܿܣ ܦܝܼܠܵܛܘܿܣ. ܘܐܸܬ݂ܩ̄ܒܲܪ: ܘܩ̣ܵܡ ܠܲܬ݂ܠܵܬ݂ܵܐ ܝܵܘܡܝܼ̈ܢ ܐܲܝܟ݂ ܕܲܟ݂ܬ݂ܝܼܒ݂. ܘܲܣ̣ܠܸܩ ܠܲܫܡܲܝܵܐ: ܘܝܼܬܸܒ݂ ܡ̣ܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܘܬ݂ܘܼܒ݂ ܥܬ݂ܝܼܕ݂ ܠܡܹܐܬ݂ܵܐ ܠܲܡܕ݂ܵܢ ܠܡܝܼܬܹ̈ܐ ܘܲܠܚܲܝܹ̈ܐ. ܘܲܒ݂ܚܲܕ݂ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܪܘܼܚܵܐ ܕܲܫܪܵܪܵܐ: ܗܵܘ̇ ܕܡ̣ܢ ܐܲܒ݂ܵܐ ܢܵܦܹ̇ܩ: ܪܘܼܚܵܐ ܡܲܚܝܵܢܵܐ. ܘܒܲܚܕ݂ܵܐ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܘܲܫܠܝܼܚܵܝܬܵܐ ܩܵܬ݂ܘܿܠܝܼܩܝܼ. ܘܡܵܘܕܹ݁ܝܢܲܢ ܒܲܚܕ݂ܵܐ ܡܲܥܡܘܿܕ݂ܝܼܬ݂ܵܐ: ܠܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ. ܘܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܲܒ݂ܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ: ܐܵܡܹܝܢ.
