ܒܠܠܝܐ ܕܹܝܢ ܗܵܢܵܐ ܠܵܐ ܐܵܡܪܝܼܢ ܡܵܘܬ݁ܒ݂ܵܐ. ܐܸܠܵܐ ܚܲܕ݂ ܗܘܼܠܵܠܵܐ. ܒܩܵܠܹ̈ܐ ܕܫܲܗܪܵܐ ܗܵܟܲܢܵܐ ܕܵܢܘܼ ܦܣܲܩ ܐܲܒ݂̈ܵܗܵܬ݂ܵܐ ܛܘܼܒ݂ܵܢܹ̈ܐ ܬܹܐܵܦܘܿܪ̈ܘܿ. ܘܡܲܠܦܵܢܹ̈ܐ ܐܲܪ̈ܬܵܕ݁ܘܿܟ݁ܣܘܿ.

      ܨ̈ܠܵܘܵܬ݂ܵܐ ܕܩܵܠܹ̈ܐ ܫܲܗܪܵܐ
ܨܠܘܿܬ݂ܵܐܕܩܲܕ݂ܡܵܝܵܐ. ܢܸܫܦܪܘܼܢ ܠܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܩܵܠܹ̈ܐ ܕܗܘܼܠܵܠܲܝ̈ܢ ܘܢܸܥ̈ܡܵܬ݂ܵܐ ܕܲܙܡܝܼܪ̈ܵܬܲܢ: ܘܩܲܒܸ݁ܠܡܢܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܦܹܐܪܹ̈ܐ ܡܠܝܼܠܹ̈ܐ ܕܣܸܦ̈ܘܵܬܲܢ: ܕܲܒ݂ܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܩܲܪܒ݂ܝܼܢܲܢ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܡܲܩܦܝܼܢ ܘܲܡܫܲܪܹܝܢ ܡܲܪܡܝܼܬ݂ܵܐ ܩܲܕ݂ܡܵܝܬܵܐ ܕܗܘܼܠܵܠܵܐ ܐܲܝܢܵܐ ܕܗܘܟܠܵܗ̇ ܒܩܵܠܵܗ̇: ܘܲܡܫܲܒ݁ܚܝܼܢ ܘܲܡܥܵܠܡܝܼܢ ܘܐܵܡܪܝܼܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܪܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܩܲܕܡܵܝܹ̈ܐ ܡܲܪܡܝܼܬ݂ܵܐ * ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܘܓܲܐܝܘܼܬ݂ܵܐ ܠܒܸܫ. ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܨܓ‐ܨܗ

	ܨܠܘܿܬ݂ܵܐ ܕܲܬ݂ܪܲܝܵܢܵܐ. ܐܲܫܘܵܐ ܠܲܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܕܥܲܡ ܥܝܼܪܹ̈ܐ ܘܟܸܢܫܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ: ܒܩܵܠܹ̈ܐ ܕܲܡܠܹܝܢ ܬܵܘܕ݁ܝܼܬ݂ܵܐ: ܢܸܙܡܲܪ ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ. ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܲܡܫܲܡܫܝܼܢ ܟܠܵܗ̇ ܡܲܪܡܝܼܬ݂ܵܐ ܬܪܲܝܵܢܝܼܬ݂ܵܐ ܕܗܘܼܠܵܠܵܐ ܕܐܲܝܢܵܐ ܕܗܘ ܒܩܵܠܵܗ̇. ܐܲܝܟ݂ ܩܲܕ݂ܡܵܝܬܵܐ.
ܘܐܵܡܪܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܪ̈ܵܝܹܐ ܡܲܪܡܝܼܬ݂ܵܐ. ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܬܸܫܒ݁ܘܿܚܬܵܐ ܚܕܲܬܵܐ. ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܨܘ‐ܨܚ
ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܵܐ: ܫܘܼܒ݂ܚܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ: ܘܗܘܼܠܵܠܹ̈ܐ ܕܠܵܐ ܫܲܠܘܵܐ: ܘܬܸܫܒ̈ܚܵܬ݂ܵܐ ܕܠܵܐ ܣܵܟ݂ܵܐ: ܘܩܵܠܹ̈ܐ ܕܲܡܠܹܝܢ ܬܵܘܕ݁ܝܼܬ݂ܵܐ: ܚܲܝܵܒ݂ ܟܠܹܗ ܟܝܵܢܵܐ ܕܲܡܠܝܼ̈ܠܹܐ ܕܲܒ݂ܪܲܝܬ݁: ܠܡܲܣܵܩܘܼ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܲܡܫܲܡܫܝܼܢ ܟܠܵܗ̇ ܡܲܪܡܝܼܬ݂ܵܐ ܬܠܝܼܬ݂ܵܝܬܵܐ ܕܗܘܼܠܵܠܵܐ ܐܲܝܢܵܐ ܕܗܘ ܒܩܵܠܵܗ̇. ܐܲܝܟ݂ ܩܲܕ݂ܡܵܝܬܵܐ.
ܘܐܵܡܪܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܩܲܕܡܵܝܹ̈ܐ ܡܲܪܡܝܼܬ݂ܵܐ. ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܢܙܘܼܥܘܼܢ ܥܲܡܡܹ̈ܐ. ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܨܛ‐ܩܐ

ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ. ܡܢ ܦܘܼܡܹ̈ܐ ܡܵܪܝ ܡܠܝܼܠܹ̈ܐ ܕܲܒ݂ܪܲܝܬ݁: ܘܡܢ ܠܸܫܵܢܹ̈ܐ ܡܫܲܒ݁ܚܵܢܹ̈ܐ ܕܪܲܟܸ݁ܒ݂ܬ݁: ܘܡܢ ܟܸܢܫܹ̈ܐ ܟܠܗܘܿܢ ܕܥܸܠܵܝܹ̈ܐ ܘܲܕ݂ܬܲܚܬܵܝܹ̈ܐ: ܫܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܲܕ݂ܡܵܪܘܼܬ݂ܵܟ݂ ܢܸܣܬ݁ܓܸܕ݂ ܘܢܸܫܬܲܒܲܚ ܘܢܸܬ݂ܝܲܩܲܪ ܘܢܸܬ݂ܪܲܡܪܲܡ ܘܢܸܬ݁ܬܵܘܕܸ݁ܐ ܘܢܸܬ݂ܒܲܪܲܟ݂ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
      ܕܠܸܠܝܵܐ. ܒܫܘܼܪܵܝ ܡܵܪܵܐ ܕܟ݂ܠܵܐ. ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܬܸܫܒ݁ܘܿܚܬܵܐ ܚܕܲܬܵܐ: ܬܹܒܹܝܠ ܘܟ݂ܠܗܘܿܡ ܥܵܡܘܿܪܹ̈ܝܗ. ܕܨܘܼܪܬܵܐ. ܘܟ݂ܠܡܸܕܸ݁ܡ ܡܢ ܕܪܹܫ ܐܸܬ݂ܚܲܕܲܬ݂ ܒܲܡܫܝܼܚܵܐ. ܥܲܬ݁ܝܼ̈ܩܵܬ݂ܵܐ ܥܒܲܪ̈ ܠܗܹܝܢ. ܟܠܗܹܝܢ ܒܸܪ̈ܝܵܬ݂ܵܐ ܗܵܐ ܐܸܬ݂ܚܲܕܲܬ݂ ܒܲܡܫܝܼܚܵܐ. ܕܗܘܝܘܼ ܪܹܫܵܐ ܕܚܲܝܹ̈ܐ ܚܲܕ݂ܬܹ̈ܐ. ܒܹܗܘ ܓܹܝܪ ܫܘܼܠܛܵܢܹܗ ܕܡܵܘܬܵܐ ܐܸܫܬ݁ܪܝܼ. ܘܲܠܡܝܼܬܹ̈ܐ ܒܩܵܠܹܗ ܐܲܩܝܼܡ. ܘܣܵܛܵܢܵܐ ܗܵܐ ܡܸܬ݁ܬܲܢܲܚ ܡܲܪܝܼܪܵܐܝܼܬ݂ ܘܐܵܡܲܪ. ܕܘܵܝܠܝܼ ܕܲܗܘܹܝܬ݂ ܓܘܼܚܟܵܐ ܠܐܵܕ݂ܵܡ ܘܲܠܝܲܠܕܵܘ̈ܗܝ. ܕܗܵܐ ܝܼܫܘܿܥ ܫܪܵܢܗ ܡܢ ܫܘܼܠܛܵܢܝ. ܘܒܲܙܹܗ ܠܩܸܢܝܵܢܝ ܡܢ ܐܝܼܕܲܝ̈. ܘܒܲܩܝܵܡܬܹ݁ܗ ܦܲܠܸܓ݂ ܚܲܝܹ̈ܐ ܠܟ݂ܠܹܗ ܓܸܢܣܹܗ: ܬܢܝܼ. ܫܲܒܲܚ. ܒܛܘܼܒ݂ܵܐ ܠܝܲܠܘܼܕܹ̈ܐ. ܗܘܵܬ݂ ܠܵܗ ܦܲܪܕܲܝܣܵܐ ܓܲܢܬܹܗ ܕܝܵܘܣܸܦ ܡܢ ܪܵܡܬ݂ܵܐ. ܕܲܡܫܝܼܚܵܐ ܒܓ݂ܵܘܵܗ ܐܸܬ݁ܬ݁ܣܝܼܡ. ܘܲܢܚܸܬ݂ܘ ܪ̈ܘܼܚܵܢܹܐ ܘܓ݂ܘܼܕܹ̈ܐ ܕܥܝܼܪܹ̈ܐ ܘܕܲܣܪ̈ܵܦܹܐ. ܘܲܠܩܲܒ݂ܪܹܗ ܕܡܵܪܲܢ ܡܙܲܝܚܝܼܢ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܡܵܪܲܢ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܒܪܵܐ ܕܐܲܠܵܗܵܐ. ܒܪܝܼܟ݂ܘܼ ܕܒܲܩܝܵܡܬܹ݁ܗ ܚܲܕ݂ܝܲܢ.

ܡܸܛܠ ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܝܵܐ ܡܪܲܝܡܵܐ ܥܲܠ ܟܠܵܗ̇ ܐܲܪܥܵܐ* ܘܛܵܒ݂ ܐܸܬ݁ܬ݁ܪܝܼܡܬ݁ ܥܲܠ ܟܠܗܘܿܢ ܐܲܠܵܗܹ̈ܐ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܪ̈ܵܚܡܵܘܗܝ ܕܡܵܪܝܵܐ ܣܵܢܹܝܢ ܠܒ݂ܝܼܫܬܵܐ* ܘܢ̇ܵܛܲܪ ܢܲܦ̮̈ܫܵܬ݂ܵܐ ܕܙܲܕ݁ܝܼܩܵܘ̈ܗܝ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܘܡܢ ܐܝܼܕ݂ܵܐ ܕܪ̈ܲܫܝܼܥܹܐ ܢܦܲܨܸܐ ܐܸܢܘܿܢ* ܢܘܼܗܪܵܐ ܕܲܕ݂ܢܲܚ ܠܙܲܕ݁ܝܼܩܹ̈ܐ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܘܠܲܬ݂ܪ̈ܝܼܨܲܝ ܠܸܒܵܐ ܚܲܕ݂ܘܼܬ݂ܵܐ* ܚܕ݂ܵܘ ܙܲܕ݁ܝܼܩܹ̈ܐ ܒܡܵܪܝܵܐ* ܘܐܵܘܕܵܘ ܠܕ݂ܘܼܟ݂ܪܵܢܵܐ ܕܩܘܼܕ݂ܫܹܗ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܫܲܒܲܚ܀ ܥܵܠܲܡ܀ ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܬܕ݂ܘܼܨ ܐܲܪܥܵܐ* ܘܢܸܚ̈ܕ݁ܝܵܢ ܓܵܙܪ̈ܵܬ݂ܵܐ ܣܲܓܝܼ̈ܐܵܬ݂ܵܐ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ.
	ܗܦܵܟ݂ܬ݂ܵܐ. ܡܲܠܟ݁ܘܼܬ݂ ܪܵܘܡܵܐ ܒܥܵܐ ܓܲܝܵܣܵܐ ܘܠܵܘ ܦܲܪܕܲܝܣܵܐ: ܘܛܘܼܒܹ̈ܐ ܪ̈ܓ݂ܝܼܓܹܐ ܫܐܸܠ ܗ݇ܘܵܐ ܡܸܢܹܗ ܘܠܵܘ ܐܝܼܠܵܢܹ̈ܐ܀ ܡܵܢܵܐ ܗ݇ܝܼ ܗܵܕܹܐ ܡܨܸܐ ܗܘܵܐ ܠܡܹܐܡܲܪ ܐܵܦ ܓܲܝܵܣܵܐ: ܐ݇ܢܵܐ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܫܹܐܠܹܬ݂ ܡܸܢܵܟ݂ ܘܠܵܘ ܦܲܪܕܲܝܣܵܐ. ܘܐܸܢ ܐܝܼܬ݂ ܐܲܦܸܣܩܘܿܦܵܐ. ܗܵܢ̇ܵܘ ܫܸܡܥܘܿܢ ܕܲܩܪܵܝܜ ܡܵܪܹܗ ܬܠܵܬ݂ ܙܲܒ݂̈ܢܵܬ݂ܵܐ: ܪܥܝܼ ܠܝܼ ܥܸܪ̈ܒܲܝ ܐܵܦ ܢܩܵܘ̈ܵܬ݂ܝ ܬܲܡܝܼ̈ܡܵܬ݂ܵܐ.
      ܬܸܫܒ݁ܘܿܚܬܵܐ. ܬܵܘܕ݁ܝܼ ܠܛܵܒ݂ܵܐ ܕܚܲܪܲܪ ܓܸܢܣܲܢ: ܡܢ ܥܲܒ݂ܕ݁ܘܼܬ݂ܵܐ ܕܒ݂ܝܼܫܵܐ ܘܡܵܘܬܵܐ܀ ܫܲܝܸܢ ܥܲܡܲܢ ܟܸܢܫܲܝ̈ ܪܵܘܡܵܐ: ܕܪܲܓ݁ܝܼܙܝܼܢ ܗ݇ܘܵܘ ܡܸܛܠ ܥܵܘܠܲܢ܀ ܒܪܝܼܟ݂ ܚܲܢܵܢܵܐ ܕܟܲܕ݂ ܠܵܐ ܒܥܲܝܢܵܝܗܝ: ܢܦܲܩ ܒܲܒ݂ܥܵܬܲܢ ܘܲܚܕ݂ܝܼ ܒܚܲܝܲܝ̈ܢ܀ ܘܨܵܪ ܕܸܡ̈ܘܵܬ݂ܵܐ ܕܐܲܒ݁ܝܼܕ݂ܘܼܬܲܢ: ܘܲܕ݂ܦܘܼܢܵܝܲܢ ܒܥܸܪܒܵܐ ܕܲܛܥܵܐ܀ ܝܵܪܬܵܐ ܘܲܒ݂ܪܵܐ ܩܪܵܝܗܝ ܠܲܟ݂ܝܵܢܲܢ: ܕܲܛܥܵܐ ܘܲܦܢܵܐ ܘܡܝܼܬ݂ ܘܐܸܬ݂ܢܲܚܲܡ܀ ܘܚܲܕ݁ܝܼ ܠܟܸܢܫܹ̈ܐ ܪ̈ܘܼܚܵܢܵܝܹܐ: ܒܲܬ݂ܝܵܒ݂ܘܼܬܲܢ ܘܲܒ݂ܢܘܼܚܵܡܲܢ܀ ܠܵܐ ܡܸܬ݂ܡܲܠܲܠ ܚܘܼܒܵܐ ܪܲܒܵܐ: ܕܚܵܘܝܼ ܨܹܐܕܲܝܢ ܪܵܚܹ̇ܡ ܓܸܢܣܲܢ܀ ܕܡܸܢܹܗ ܕܓܸܢܣܲܢ ܥܒܲܕ݂ ܡܸܨܥܵܝܵܐ: ܘܪܲܥܝܼ ܠܥܵܠܡܵܐ ܥܲܡ ܪܲܒܘܼܬܹܗ܀ ܪܲܒܵܐ ܗ݇ܝ ܡܸܢܲܢ ܘܡܢ ܟܠ ܒܸܪ̈ܝܵܢ: ܚܕܲܬܵܐ ܕܲܣܥܲܪ ܨܹܝܕ݂ ܐ݇ܢܵܫܘܼܬܲܢ܀ ܕܥܲܒ݂ܕܹ݁ܗ ܠܦܲܓ݂ܪܲܢ ܗܲܝܟܲܠ ܩܘܼܕ݂ܫܵܐ: ܕܲܢܡܲܠܸܐ ܒܹܗ ܣܸܓ݂ܕܲܬ݂ ܟܠܵܐ܀ ܬܵܘ ܐܲܪ̈ܥܵܢܹܐ ܘܲܫܡܲܝܵܢܹ̈ܐ: ܬܗܲܪܘ ܘܐܸܬ݁ܕܲܡܲܪܘ ܒܪܲܒ݁ܘܼܬ݂ ܕܲܪܓ݂ܵܐ܀ ܕܲܡܛܵܐ ܓܸܢܣܲܢ ܠܪܵܘܡܵܐ ܪܲܒܵܐ: ܕܐܲܠܵܗܘܼܬ݂ܵܐ ܕܠܵܐ ܡܸܬ݁ܕܲܪܟܵܐ܀ ܫܡܲܝܵܐ ܘܐܲܪܥܵܐ ܘܟ݂ܠܡܵܐ ܕܲܒ݂ܗܘܿܢ: ܢܵܘܕ݁ܘܿܢ ܥܲܡܲܢ ܠܡܵܘܪܸܒ݂ ܓܸܢܣܲܢ܀ ܕܚܲܕܸ݁ܬ݂ ܨܲܠܡܲܢ ܘܲܥܛܵܐ ܥܵܘܠܲܢ: ܘܲܩܪܵܢ ܒܲܫܡܹܗ ܘܫܲܥܒܸ݁ܕ݂ ܠܲܢ ܟܠ܀ ܫܵܘܹ̇ܐ ܠܫܘܼܒ݂ܚܵܐ ܡܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܗ̇ܘ ܕܐܲܪܝܼܡܲܢ ܠܥܸܠ ܡܢ ܟܠܵܐ܀ ܘܢܹܐܡܲܪ ܟܠܲܢ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ: ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀
      ܟܵܪܘܿܙܘܼܬ݂ܵܐ. ܢܩܘܼܡ ܫܲܦܝܼܪ. ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ. ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪܹܗ ܩܲܕ݁ܝܼܫܵܐ ܐܸܬ݁ܬܲܒܲܪܘ ܬܲܪ̈ܥܹܐ ܕܲܫܝܘܿܠ. ܘܲܫܪܵܗ ܠܲܚܛܝܼܬ݂ܵܐ ܘܲܠܡܵܘܬܵܐ ܒܲܩܝܵܡܬܹ݁ܗ. ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪܹܗ ܩܲܕ݁ܝܼܫܵܐ ܒܲܛܠܹܗ ܠܡܵܘܬܵܐ. ܘܝܲܗ݇ܒ݂ܠܲܢ ܚܲܝܹ̈ܐ ܚܲܕ݂ܬܹ̈ܐ. ܘܚܲܕܸ݁ܬ݂ ܠܟ݂ܠܗܹܝܢ ܒܸܪ̈ܝܵܬ݂ܵܐ ܒܲܩܝܵܡܬܹ݁ܗ. ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪܹܗ ܩܲܕ݁ܝܼܫܵܐ ܐܲܚܝܼ ܡܝܼܬܹ̈ܐ ܡܢ ܩܲܒ݂ܪ̈ܲܝܗܘܿܢ. ܘܐܲܩܝܼܡ ܠܟ݂ܠܹܗ ܟܝܵܢܵܐ ܐ݇ܢܵܫܵܝܵܐ ܡܢ ܡܵܝܘܼܬ݂ܘܼܬ݂ܵܐ ܒܲܩܝܵܡܬܹ݁ܗ. ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪܹܗ ܩܲܕ݁ܝܼܫܵܐ ܐܲܥܛܸܦ ܒܸܗܬ݁ܬܵܐ ܠܨܵܠܘܿܒܹ̈ܐ. ܘܐܲܒ݂ܗܸܬ݂ ܠܢ̇ܵܛܘܿܪܹ̈ܐ. ܘܚܲܕ݁ܝܼ ܠܟܸܢܫܹ̈ܐ ܕܬܲܠܡܝܼܕܹ̈ܐ ܒܲܩܝܵܡܬܹ݁ܗ. ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪܹܗ ܩܲܕ݁ܝܼܫܵܐ ܣܲܒܲܪ ܚܲܕ݂ܘܼܬ݂ܵܐ ܠܢܸܫܹ̈ܐ. ܘܢܸܫܹ̈ܐ ܠܲܫܠܝܼܚܹ̈ܐ ܘܲܫܠܝܼܚܹ̈ܐ ܚܲܕ݂ ܠܚܲܕ݂. ܘܦܘܼܪܩܵܢܵܐ ܠܥܵܠܡܵܐ ܟܠܹܗ ܡܢ ܣܵܟܹܗ ܗܘܵܐ ܒܲܩܝܵܡܬܹ݁ܗ. ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪܹܗ ܩܲܕ݁ܝܼܫܵܐ ܡܠܲܟ݂ ܠܲܢ ܩܝܵܡܬܵܐ ܓܵܘܵܢܵܝܬܵܐ. ܕܲܢܩܘܼܡ ܒܚܲܕ݂ܬ݂ܘܼܬ݂ܵܐ ܕܲܩܝܵܡܬܹ݁ܗ. ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪܹܗ ܩܲܕ݁ܝܼܫܵܐ ܐܸܬ݂ܚ݇ܙܝܼ ܙܲܒ݂̈ܢܵܬ݂ܵܐ ܣܲܓ݁ܝܼ̈ܐܵܬ݂ܵܐ ܠܬܲܠܡܝܼܕ݂ܵܘ̈ܜ. ܘܲܢܦܲܚ ܒܗܘܿܢ ܪܘܼܚܹܗ. ܘܩܲܒܸ݁ܠܘ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܚܲܕ݁ܝܼ ܐܲܦܨܲܚ ܠܟ݂ܠ ܒܲܩܝܵܡܬܹ݁ܗ. ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪܹܗ ܩܲܕ݁ܝܼܫܵܐ ܫܲܕܲܪ ܠܬܲܠܡܝܼܕ݂ܵܘ̈ܗܝ ܕܲܢܬܲܠܡܕ݂ܘܼܢ ܠܟ݂ܠܗܘܿܢ ܥܲܡܡܹ̈ܐ. ܘܢܲܥܡܕ݂ܘܼܢ ܐܸܢܘܿܢ ܒܫܸܡ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܐܸܫܬܵܘܕ݁ܝܼ ܕܢܸܗܘܸܐ ܥܲܡܗܘܿܢ ܥܕܲܡܵܐ ܠܫܘܼܠܵܡܹܗ ܕܥܵܠܡܵܐ. ܘܐܵܦܠܲܢ ܡܙܲܝܚܵܢܹ̈ܐ ܕܲܩܝܵܡܬܹ݁ܗ. ܡܫܝܼܚܵܐ ܕܪܲܡܪܡܲܢ ܒܲܩܝܵܡܬܹ݁ܗ. ܘܚܲܕ݂ܝܲܢ ܒܢܘܼܚܵܡܹܗ. ܘܐܲܦܨܲܚ ܠܲܢ ܒܦܘܼܪܩܵܢܹܗ. ܘܐܲܥܬܲܪ ܠܲܢ ܒܡܵܘ̈ܗܒ݂ܵܬܹܗ. ܘܐܲܪܝܼܡ ܪܹܫܲܢ ܒܪܘܼܡܪܵܡܹܗ. ܘܲܡܠܵܐ ܠܸܒܵܘ̈ܵܬܲܢ ܚܲܕ݂ܘܼܬܹܗ. ܘܚܲܕܸ݁ܬ݂ ܢܲܦ̮ܫ̈ܵܬܲܢ ܒܲܩܝܵܡܬܹ݁ܗ. ܥܲܠ ܚܘܼܠܡܵܢܵܐ. ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܕܐܲܩܝܼܡ ܠܚܲܒ݁ܝܼܒܹܗ ܘܥܲܒ݂ܕܹܗ ܠܵܐ ܡܵܝܘܼܬ݂ܵܐ ܘܠܵܐ ܚܵܫܘܿܫܵܐ. ܗܵܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܝܲܗ݇ܒܹ݁ܗ ܕܢܸܗܘܸܐ ܪܹܫܵܐ ܠܥܹܕ݁ܬܹ݁ܗ. ܘܲܡܕܲܒܲܪ ܠܟ݂ܠ ܒܛܲܝܒ݁ܘܼܬܹܗ. ܗܵܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܡܢ ܬܸܫܥܵܐ ܬܸܓ݂ܡܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ ܕܠܵܐ ܫܲܠܘܵܐ. ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂ ܡܢ ܬܲܚܬܵܝܹ̈ܐ ܒܥܹܐܕ݂ܵܐ ܫܒ݂ܝܼܚܵܐ ܕܲܩܝܵܡܬܹ݁ܗ. ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܘܐܲܓܸ݁ܢ ܝܲܡܝܼܢܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂ ܥܲܠ ܥܲܡܵܟ݂ ܡܣܲܟܸ݁ܐ ܠܲܩܝܵܡܬܵܟ݂. ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
