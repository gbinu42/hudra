ܛܲܟ݂ܣܵܐ ܕܬܸܫܡܸܫܬܵܐ ܕܠܸܠܝܵܐ
ܘܲܕ݂ܨܲܦܪܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܕܲܫܚܝܼܡܹ̈ܐ
ܕܫܲܒ݁ܬ݂ܵܐ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟ݂: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟ݂: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬ݂ܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܡܵܘܬ݁ܒ݂ܵܐ. ܬܸܒ݂ܣܲܡ ܠܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܚܲܠܵܫܘܼܬܲܢ: ܘܬܸܥܘܿܠ ܩܕ݂ܵܡܲܝܟ ܒܵܥܘܼܬ݂ܵܐ ܕܲܡܚܝܼܠܘܼܬܲܢ: ܘܪ̈ܲܚܡܲܝܟ ܢܸܗܘܘܿܢ ܣܢܹܐܓ݂ܪܹ̈ܐ ܛܵܒܹ̈ܐ ܠܚܲܝܵܒ݂ܘܼܬܲܢ: ܘܡܢ ܒܹܝܬ݂ ܓܲܙܹܗ ܪܲܒܵܐ ܕܲܚܢܵܢܵܟ݂ ܢܸܬ݂ܦܲܢ̈ܝܵܢ ܫܹܐ̈ܠܵܬ݂ܵܐ ܠܲܣܢܝܼܩܘܼܬܲܢ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܡܵܘܬ݁ܒ݂ܵܐ.
ܕܠܸܠܝܵܐ. ܒܢܸܬ݁ܛܲܝܲܒ݂. ܕܘܼܟ݂ܪܵܢܵܐ ܕܛܲܝܒ̈ܘܵܬ݂ܵܟ݂ ܣܲܓܝܼ̈ܐܵܬ݂ܵܐ: ܐܲܝܟ݂ ܕܦܲܩܸܕ݂ ܠܐܲܒ݂ܵܗܲܝ̈ܢ. ܕܘܼܟ݂ܪܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܕܚܲܫ ܥܲܠ ܐܲܦܲܝ̈ܢ. ܥܵܒ݂ܕܝܼܢܲܢ ܒܠܲܚܡܵܐ ܘܚܲܡܪܵܐ ܐܲܝܟ݂ ܕܲܟ݂ܬ݂ܝܼܒ݂. ܢܩܲܕܸܫ ܦܲܓ݂ܪ̈ܲܝܢ ܘܲܢܕܲܟܸܐ ܬܹܐܪ̈ܵܬܲܢ. ܡܢ ܚܘܼܫܵܒܹ̈ܐ ܦܠܝܼ̈ܓܹܐ ܘܲܢܟ݂ܝܼ̈ܠܹܐ. ܕܠܵܐ ܢܸܬ݂ܩܲܛܪܲܓ݂ ܡܵܐ ܕܐܵܬܹ݁ܐ. ܘܕ݂ܵܐܹ̇ܢ ܠܹܗ ܠܥܵܠܡܵܐ. ܬܢܝܼ܀ ܠܐܲܝܠܹܝܢ ܕܲܬ݂ܪܝܼܨܝܼܢ ܒܠܸܒܗܘܿܢ. ܗܵܐ ܐ݇ܪܵܙܵܐ ܫܡܲܝܵܢܵܐ ܕܛܲܝܸܒ݂ ܠܟ݂ܘܿܢ. ܗܘ ܪܵܥܝܵܐ ܫܲܪܝܼܪܵܐ ܕܚܲܫ ܥܲܠ ܐܲܦܲܝ̈ܢ. ܥܸܪ̈ܒܹ݁ܐ ܡܠܝܼ̈ܠܹܐ ܦܪ̈ܝܼܩܲܝ ܒܲܨܠܝܼܒ݂ܵܐ. ܩܪܘܿܒ݂ܘ ܘܐܸܬ݂ܒܲܣܲܡܘ ܒܐ݇ܪ̈ܵܙܹܐ ܡܲܚܝܵܢܹ̈ܐ. ܘܐܸܫܬܵܘ ܟܵܣܵܐ ܕܦܘܼܪܩܵܢܵܐ. ܕܗܘ ܡܫܝܼܚܵܐ ܡܙܲܓ݂ ܠܟ݂ܘܿܢ. ܘܡܲܠܟܘܼܬܹܗ ܒܟ݂ܠ ܫܲܠܝܼܛܵܐ. ܒܪܵܐ ܡܫܝܼܚܵܐ ܕܡܲܠܟܘܼܬܹܗ ܠܵܐ ܡܸܬ݁ܬ݁ܙܝܼܥܵܐ. ܘܟ݂ܘܼܡܪܘܼܬܹܗ ܠܵܐ ܥܵܒ݂ܪܵܐ ܘܠܵܐ ܡܸܫܬܲܪܝܵܐ. ܫܲܝܸܢ ܒܲܨܠܝܼܒ݂ܵܟ݂ ܟܵܗܢܘܼܬ݂ܵܐ ܘܡܲܠܟܘܼܬ݂ܵܐ. ܕܒ݂ܐܵܘܝܘܼܬ݂ܵܐ ܢܕܲܒܪ̈ܵܢ ܐܘܼܚܕܵܢܲܝ̈ܗܹܝܢ. ܟܲܕ݂ ܢܵܛ̇ܪ̈ܵܢ ܕܠܵܐ ܡܘܼܡܵܐ ܬܵܘܕܝܼܬ݂ܵܐ ܓܡܝܼܪܬܵܐ.
ܕܲܨܠܘܿܬ݂ܵܐ. ܒܨܸܒ݂ܝܵܢܲܢ ܟܲܪܟ݂ܘܼܢ. ܩܵܠܵܐ: ܚ ܒܪܘܼܟܵܒ݂ ܢܸܬ݁ܛܲܝܲܒ݂ ܒܕܸܚܠܬ݂ܵܐ
ܐܸܬ݂ܟܲܪܟ݁ܘܼܢܝ ܘܚܲܕ݂ܪܘܼܢܝ. ܒܨܸܒ݂ܝܵܢܲܢ ܟܲܪܟ݂ܘܼܢ ܚܲܫܹ̈ܐ ܕܲܚܛܝܼܬ݂ܵܐ. ܐܲܢ݇ܬ݁ ܛܵܒ݂ܵܐ ܦܬܲܚ ܠܵܗ̇ ܬܲܪܥܵܐ ܠܲܨܠܘܿܬܲܢ. ܟܝܵܢܲܢ ܚܲܠܵܫ ܘܲܢܩܸܐ ܠܲܚܛܝܼܬ݂ܵܐ. ܗܲܒ݂ ܠܹܗ ܐܝܼܕ݂ܐ ܗܵܫܵܐ ܕܐܸܬ݂ܡܲܟܲܟ݂. ܟܒܲܪ ܢܸܫܬܵܘܕܲܥ ܠܲܟ݂ܝܵܢܹܗ. ܒܪ̈ܲܚܡܹܐ ܕܛܲܝܒܘܼܬ݂ܵܟ݂. ܐܲܣܵܢܝ ܡܵܪܝܵܐ ܡܸܛܠ ܕܙܵܥܘ ܓܲܪ̈ܡܲܝ. ܠܵܟ݂ ܡܫܝܼܚܵܐ ܐܵܣܝܵܐ ܛܵܒ݂ܵܐ ܘܫܲܪܝܼܪܵܐ. ܡܩܲܪܒ݂ܝܼܢܲܢ ܡܘܼܡܹ̈ܐ ܕܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ. ܐܲܣܵܐ ܟܹܐܒܲܝ̈ܢ ܘܲܥܨܘܿܒ݂ ܟܘܼܪ̈ܗܵܢܲܝܢ. ܘܗܲܒ݂ܠܲܢ ܕܢܸܓ݂ܡܘܿܪ ܨܸܒ݂ܝܵܢ ܡܵܪܘܼܬ݂ܵܟ݂. ܐܲܝܟ݂ ܕܲܡܥܵܕ݂ܵܐ ܗ݇ܝ ܛܲܝܒܘܼܬ݂ܵܟ݂. ܚܵܐܹܢ ܟܠ ܒܪ̈ܲܚܡܵܘܗܝ. ܫܡܲܥ ܐܲܠܵܗܵܐ ܘܪܲܚܸܡ ܥܠܲܝ. ܫܡܲܥ ܡܵܪܲܢ ܒܵܥܘܼܬ݂ ܥܲܒ݂ܕܲܝ̈ܟ ܕܩܵܪܹܝ̇ܢ ܠܵܟ݂. ܘܢܸܬ݂ܦܲܢ̈ܝܵܢ ܡܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܫܸܐ̈ܠܵܬܲܢ. ܠܕܲܟ݂ܪܝܼܗ ܐܲܚܠܸܡ ܠܕܲܡܥܵܩ ܒܲܝܲܐܲܝܗܝ. ܘܗܲܒ݂ ܥܘܼܕ݂ܪܵܢܵܐ ܠܟܸܢܫܲܢ ܡܢ ܣܵܟܹܗ. ܕܠܵܟ݂ܘܼ ܡܣܲܟ̈ܝܵܢ ܢܲܦ̮̈ܫܵܬܲܢ. ܚܲܝܸܠ ܡܚܝܼܠܘܼܬܲܢ. ܨܘܼܬ݂ ܡܸܠܲܝ̈ ܘܩܲܒܸܠ. ܫܡܲܥ ܨܠܘܿܬܲܢ ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܸܐ̈ܠܵܬܲܢ. ܐܲܠܵܗܵܐ ܕܲܡܠܸܐ ܪ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ. ܪܕܲܝܬܵܢ ܒܟܹܐܢܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܪ̈ܲܚܡܹܐ ܕܛܲܝܒܘܼܬ݂ܵܟ݂. ܘܲܒ݂ܬܲܪ̈ܬܲܝܗܹܝܢ ܕܬܸܩܢܹܝܢ ܨܵܒܹܐ ܐܲܢ݇ܬ݁. ܠܵܟ݂ ܬܸܫܒܘܿܚܬܵܐ ܘܬ݂ܵܘܕܝܼܬ݂ܵܐ. ܕܐܲܢ݇ܬ݁ܘܼ ܝܵܨܘܿܦܲܢ. ܡܸܛܠ ܛܲܝܒܘܼܬ݂ܵܟ݂ ܐܲܠܵܗܵܐ. ܒܛܲܝܒܘܼܬ݂ܵܟ݂ ܦܲܨܵܢ ܡܵܪܲܢ ܡܢ ܒܝܼܫܵܐ. ܘܬܲܪܥܝܼܬ݂ܵܐ ܕܩܵܝ̇ܡܵܐ ܠܘܼܩܒܲܠ ܨܸܒ݂ܝܵܢܵܟ݂. ܘܲܒ݂ܢܵܡܘܿܣܵܟ݂ ܢܸܗܪܘܿܓ݂ ܒܐܲܡܝܼܢܘܼ. ܕܢܸܬܸ݁ܠ ܦܹܐܪܹ̈ܐ ܕܫܵܘܹܝ̇ܢ ܠܝܼܕܲܥܬ݂ܵܐ ܕܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܲܫܪܵܪܵܐ. ܕܠܵܗ̇ ܐܲܫܘܝܼܬܵܢ ܒܪ̈ܲܚܡܲܝܟ. ܠܵܐ ܠܥܵܠܲܡ ܬܸܪܓܲܙ ܥܠܲܝܢ. ܕܐܲܪܓܸܙܬܵܟ݂ ܒܝܲܕ݂ ܣܘܼܪ̈ܚܵܢܲܝ ܚܲܢܵܢܵܐ. ܠܵܐ ܬܸܫܒܩܲܢܝ ܘܠܵܐ ܬܲܪܦܸܐ ܒܝܼ ܐܝܼܕܲܝ̈ܵܐ. ܒܵܟ݂ ܡܵܪܝ ܣܲܒܪܹܬ݂ ܘܒ݂ܵܟ݂ ܣܲܒܲܪܘ ܐܲܒ݂ܵܗܲܝ̈. ܘܲܥܠܲܝܟ ܡܵܪܲܢ ܐܸܫܬ݁ܕ݂ܝܼܬ݂ ܡܢ ܡܲܪܒܥܵܐ. ܦܲܨܵܢܝ ܒܪ̈ܲܚܡܲܝܟ ܡܢ ܒܝܼܫܵܐ. ܕܟ݂ܠܫܵܥ ܣܵܩܲܪ ܠܝܼ. ܠܵܟ݂ ܐܹܫܲܒܲܚ ܡܵܪܝܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܡܫܝܼܚܵܐ ܪܵܚܹ̇ܡ ܬܲܝܵܒܹ̈ܐ. ܕܲܒ݂ܐ݇ܪ̈ܵܙܲܝܟ ܚܲܣܝܼܬ݁ ܚܵܘܒܲܝ̈ܢ ܘܲܚܛܵܗܲܝ̈ܢ. ܪܕܲܝܬܵܢ ܥܲܠ ܕܐܲܣܟܸܠܢ ܘܩܲܪܸܒ݂ܬܵܢ ܠܝܼܕܲܥܬ݂ܵܟ݂. ܢܲܓܝܼܪ ܪܘܼܚܵܐ ܘܣܲܓܝܼ ܛܲܝܒܘܼܬ݂ܵܐ. ܚܘܼܣ ܥܲܠ ܥܲܒ݂ܕܲܝ̈ܟ ܕܩܵܪܹܝ̇ܢ ܠܵܟ݂. ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈. ܗܲܒ݂ܠܲܢ ܡܵܪܝ ܚܘܼܒܵܐ ܘܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ. ܘܲܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܘܚܲܝܹ̈ܐ ܘܚܘܼܠܡܵܢܵܐ. ܕܐܲܢ݇ܬ݁ܘܼ ܐܝܼܬܲܝܟ ܡܵܪܵܐ ܡܪܲܚܡܵܢܵܐ. ܫܵܒܹܩ ܚܵܘܒܹ̈ܐ ܘܡܲܓ݂ܙܲܪ ܫܘܼܒ݂ܩܵܢܵܐ. ܘܲܡܩܲܒܠܵܢܵܐ ܕܬܲܝܵܒܹ̈ܐ. ܒܪ̈ܲܚܡܹܐ ܘܚܵܘܣܵܢܵܐ. ܐܲܝܟ݂ ܥܲܝܢܹ̈ܐ ܕܥܲܒ݂ܕܹ̈ܐ ܠܘܵܬ݂ ܡܵܪ̈ܲܝܗܘܿܢ. ܠܘܵܬ݂ܵܟ݂ ܡܵܪܝ ܚܵܝܪ̈ܵܢ ܥܲܝܢܲܝ̈ ܒܟ݂ܠܥܸܕܵܢ. ܕܐܲܢ݇ܬ݁ ܐܝܼܬܲܝܟ ܣܲܒ݂ܪܝ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܕܲܡܚܝܼܠܘܼܬ݂ܝ. ܗܲܒ݂ܠܝܼ ܕܐܸܫܦܲܪ ܠܵܟ݂ ܒܡܸܠܲܝ̈ ܘܚܘܼܫܵܒܲܝ̈. ܐܵܦ ܒܲܥܒ݂ܵܕܲܝ̈ ܬܸܬ݁ܢܝܼܚ ܡܵܪܘܼܬ݂ܵܟ݂. ܒܵܟ݂ ܡܵܪܝ ܣܲܒܪܹܬ݂ ܕܐܲܢ݇ܬ݁ܘܼ. ܬܘܼܟ݂ܠܵܢܝ ܥܲܫܝܼܢܵܐ. ܢܵܛܲܪ ܡܵܪܝܵܐ ܠܟ݂ܠܗܘܿܢ ܕܵܚܠܵܘܗ̈ܝ. ܢܲܛܲܪ ܡܵܪܝ ܠܟ݂ܠܗܘܿܢ ܬܸܓ݂ܡܹ̈ܐ ܕܒܹܝܬ݂ ܐܵܕ݂ܵܡ. ܒܲܨܠܝܼܒ݂ܵܟ݂ ܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ. ܕܢܸܙܡܲܪ ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܒܓܸܠܝܵܢܵܟ݂ ܬܡܝܼܗܵܐ. ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܲܢ ܘܐܲܠܵܗܵܐ ܕܟ݂ܠܲܢ. ܘܠܵܟ݂ܘܼ ܘܵܠܹܐ ܕܲܢܫܲܒܲܚ. ܓܸܢܣܲܢ ܥܲܦܪܵܢܵܐ. ܪܥܝܼ ܐܸܢܘܿܢ ܘܕܲܒܲܪ ܐܸܢܘܿܢ ܥܕܲܡܵܐ ܠܥܵܠܲܡ. ܒܪܵܐ ܡܫܝܼܚܵܐ ܕܫܲܝܸܢ ܥܵܠܡܹ̈ܐ ܒܡܹܐܬ݂ܝܼܬܹܗ. ܒܲܨܠܝܼܒ݂ܵܟ݂ ܢܲܛܲܪ ܠܟܸܢܫܲܢ ܡܢ ܒܝܼܫܵܐ. ܘܒܲܪܸܟ݂ ܠܝܵܪܬ݁ܘܼܬ݂ܵܟ݂ ܒܝܲܡܝܼܢ ܡܵܪܘܼܬ݂ܵܟ݂. ܘܫܲܝܢܵܐ ܕܡܸܢܵܟ݂ ܢܲܡܠܸܟ݂ ܒܥܵܡܲܪܬܵܐ. ܘܟ݂ܵܗܢܹ̈ܐ ܘܡܲܠܟܹ̈ܐ ܢܸܬ݂ܩܲܝܡܘܼܢ. ܒܲܚܕ݂ܵܐ ܐܵܘܝܘܼܬ݂ܵܐ. ܐܲܣܵܢܝ ܡܵܪܝܵܐ ܡܸܛܠ ܕܙܵܥܘ ܓܲܪ̈ܡܲܝ. ܦܵܪܘܿܩܲܢ ܐܵܣܝܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ. ܣܝܼܡ ܣܲܡܵܐ ܕܪ̈ܲܚܡܲܝܟ ܘܲܥܨܘܿܒ݂ ܟܪܝܼܗܘܼܬܲܢ. ܘܐܵܘܫܸܛ ܐܝܼܕ݂ܵܐ ܕܡܲܠܝܵܐ ܥܘܼܕ݂ܪ̈ܵܢܹܐ. ܠܲܡܚܝܼܠܘܼܬܹܗ ܕܓܸܢܣܲܢ ܡܵܝܘܿܬ݂ܵܐ. ܘܲܣܥܘܿܪ ܒܛܲܝܒܘܼܬ݂ܵܟ݂. ܘܗܲܒ݂ ܠܗܘܿܢ ܚܘܼܠܡܵܢܵܐ. ܡܲܐܣܸܐ ܠܲܬ݂ܒ݂ܝܼܪ̈ܲܝ ܠܸܒܵܐ ܘܥܵܨܹ̇ܒ݂ ܟܹܐܒܲܝ̈ܗܘܿܢ. ܡܵܪܹܐ ܟܠ ܫܲܕܲܪ ܚܲܝܠܵܐ ܕܛܲܝܒܘܼܬ݂ܵܟ݂. ܘܲܥܨܘܿܒܲܝܗܝ ܠܫܘܼܚܢܹܗ ܕܥܵܘܠܲܢ ܕܐܸܬ݂ܚܲܒܲܠ. ܐܲܟ݂ܡܵܐ ܕܐܲܣܝܼܬ݁ ܠܲܡܫܲܪܝܵܐ ܢܸܟ݂ܝܵܢܵܘ̈ܗܝܕܡܵܪܘܿܕ݂ܵܐ. ܣܵܢܐܹܗܕܲܟ݂ܝܵܢܲܢ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܒܥܵܘ ܠܡܵܪܝܵܐ ܘܐܸܬ݂ܚܲܝܲܠܘ. ܩܲܕܝܼܫܹ̈ܐ ܢܒ݂ܝܼܹ̈ܐ ܫܠܝܼܚܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ. ܘܡܵܘܕܝܵܢܹ̈ܐ ܘܣܵܗܕܹ̈ܐ ܘܟ݂ܵܗܢܹ̈ܐ ܘܥܵܢܘܵܝܹ̈ܐ. ܐܲܦܝܼܣܘ ܚܠܵܦ ܟܠܲܢ. ܠܲܡܫܝܼܚܵܐ ܘܚܲܝ̈ܠܵܘܵܬܹܗ. ܘܲܥܒܸܕܹܝܗ̇ ܠܡܹܐܢܵܐ ܠܟ݂ܠܗܘܿܢ ܚܲܛܵܝܹ̈ܐ. ܘܬܸܗܘܸܐ ܡܥܝܼܢܵܐ ܡܲܪܕܸܐ ܥܘܼܕ݂ܪ̈ܵܢܹܐ. ܘܒ݂ܵܗ̇ ܢܸܬ݂ܦܲܢ̈ܝܵܢ ܫܹܐ̈ܠܵܬܲܢ. ܡܢ ܓܲܙܵܐ ܕܪ̈ܲܚܡܲܝܟ. ܕܐܲܒ݂ܘܼܢ܀ܢܸܚܕܘܿܢ ܫܡܲܝܵܐ ܘܲܬ݁ܕ݂ܘܼܨ ܐܲܪܥܵܐ. ܒܕ݂ܘܼܟ݂ܪܵܢܹܗ ܕܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕܝܼܫܵܐ. ܐܲܬ݂ܠܹܝܛܵܐ ܪܵܚܡܹܗ ܕܚܲܬ݂ܢܵܐ ܫܡܲܝܵܢܵܐ. ܚܕ݂ܝܼܘ ܥܸܠܵܝܹ̈ܐ ܘܟܸܢܫܹ̈ܐ ܕܬܲܚܬܵܝܹ̈ܐ. ܘܲܚܢܲܢ ܥܲܡܗܘܿܢ ܢܦܝܼܣܝܼܘܗܝ ܠܲܡܫܝܼܚܵܐ ܕܢܲܫܘܹܝܢ ܠܲܓ݂ܢܘܿܢ ܡܲܠܟܘܼܬ݂ܵܐ. ܕܛܲܝܸܒ݂ ܠܩܲܕܝܼܫܵܘ̈ܗܝ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܨܠܝܼ ܡܵܪܝܵܐ ܐܸܕ݂ܢܵܟ݂ ܘܲܥܢܝܼܢܝ. ܒܪܵܐ ܡܫܝܼܚܵܐ ܕܐܸܬ݂ܵܐ ܡܸܛܠ ܦܘܼܪܩܵܢܲܢ. ܕܲܢܚܲܕܸܬ݂ ܨܲܠܡܹܗ ܕܐܵܕ݂ܵܡ ܕܐܸܬ݂ܚܲܒܲܠ. ܘܲܠܒܸܫ ܦܲܓ݂ܪܲܢ ܘܒܹܗ ܦܲܪܩܹܗ ܠܓܸܢܣܲܢ. ܘܝܲܗ݇ܒ݂ ܠܘܼܒܵܒ݂ܵܐ ܠܲܩܝܵܡܬܵܐ ܕܡܝܼ̈ܬܹܐ. ܚܲܣܵܐ ܒܛܲܝܒܘܼܬ݂ܵܟ݂ ܠܥܲܒ݂ܕܲܝ̈ܟ. ܒܝܵܘܡܵܐ ܕܡܹܐܬ݂ܝܼܬ݂ܵܟ݂. ܡܸܛܠ ܛܲܝܒܘܼܬ݂ܵܟ݂ ܐܲܠܵܗܵܐ. ܒܛܲܝܒܘܼܬ݂ܵܟ݂ ܒܪܲܝܬܵܝܗܝ ܠܓܸܢܣܲܢ ܡܢ ܠܘܼܩܕܲܡ. ܘܐܲܠܒܸܫܬܵܝܗܝ ܫܘܼܒ݂ܚܵܐ ܦܲܐܝܵܐ ܒܦܲܪܕܲܝܣܵܐ. ܘܕܲܐܡܪܲܚ ܘܲܚܛܵܐ ܘܲܢܦܲܠ ܡܢ ܫܘܼܒ݂ܚܹܗ. ܫܲܕܲܪܬ݁ ܠܲܒ݂ܪܵܟ݂ ܚܲܒܝܼܒ݂ܵܐ ܠܘܵܬܲܢ. ܘܝܲܗ݇ܒ݂ ܠܲܢ ܒܲܚܢܵܢܹܗ ܚܲܝܹ̈ܐ. ܡܠܝܼܟܹ̈ܐ ܕܲܠܥܵܠܲܡ܀
ܕܥܲܢܝܼ̈ܕܹܐ. ܩܲܪܝܼܒ݂ ܗܘ ܦܘܼܪܩܵܢܹܗ ܠܕ݂ܵܚܠܵܘ̈ܗܝ. ܩܪܸܒ݂ ܙܲܒ݂ܢܵܐ ܕܥܵܠܡܵܐ ܗܵܢܵܐ ܥܵܒܲܪ ܠܹܗ: ܘܡܸܫܬܲܪ̈ܝܵܢ ܘܒ݂ܵܛ̈ܠܵܢ ܟܠܗܹܝܢ ܪ̈ܓ݂ܝܼܓ݂ܵܬܹܗ. ܫܡܲܥܘ ܡܵܝܘܿܬܹ̈ܐ ܘܩܲܪܸܒ݂ܘ ܬܝܵܒ݂ܘܼܬ݂ܵܐ: ܥܲܕ݂ܠܵܐ ܢܹܐܬܸܐ ܝܵܘܡܹܗ ܕܥܘܼܢܕܵܢܵܐ: ܘܢܸܬ݂ܦܪܲܥ ܟܠܢܵܫ ܐܲܝܟ݂ ܕܲܣܥܲܪ ܒܹܝܬ݂ ܕܝܼܢܵܐ ܕܚܝܼܠܵܐ. ܐܲܝܟ݂ ܐܲܪܝܵܐ ܕܢܵܗܹ̇ܡ ܘܚܵܛܹ̇ܦ. ܟܲܕ݂ ܡܵܘܬܵܐ ܦܪܝܼܣ ܥܲܠ ܟܠܢܵܫ ܘܒ݂ܵܠܲܥ̇ ܠܹܗ: ܘܡܵܐ ܕܲܡܛܵܢܝ ܙܵܥܹ̇ܬ݂ ܣܲܓ݁ܝܼ ܘܐܸܬ݂ܪܲܗܒܹ݁ܬ݂. ܒܕܲܡܦܵܣ ܐ݇ܢܵܐ ܒܚܵܘܒܲܝ̈ ܕܠܵܘ ܐܲܝܟ݂ ܕܫܲܪܟܵܐ ܐܝܼܬܲܝܗܘܿܢ: ܐܸܠܵܐ ܕܲܚܙܵܢܝ ܕܲܡܢܵܬܹܗ ܐܝܼܬܲܝ: ܐܝܼܵܠܝ ܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܝ ܘܠܵܟ݂ܘܼ ܡܣܲܟܸ݁ܐ ܐ݇ܢܵܐ.
ܫܲܒܲܚ. ܕܘܼܟ݂ܪܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܕܚܲܫ ܥܲܠ ܐܲܦܲܝ̈ܢ. ܥܵܒ݂ܕܝܼܢܲܢ ܒܠܲܚܡܵܐ ܘܚܲܡܪܵܐ ܐܲܝܟ݂ ܕܲܟ݂ܬ݂ܝܼܒ݂. ܢܩܲܕܸܫ ܦܲܓ݂ܪ̈ܲܝܢ ܘܲܢܕܲܟܸܐ ܬܹܐܪ̈ܵܬܲܢ. ܡܢ ܚܘܼܫܵܒܹ̈ܐ ܦܠܝܼ̈ܓܹܐ ܘܲܢܟ݂ܝܼ̈ܠܹܐ. ܕܠܵܐ ܢܸܬ݂ܩܲܛܪܲܓ݂ ܡܵܐ ܕܐܵܬܹ݁ܐ. ܘܕ݂ܵܐܹ̇ܢ ܠܹܗ ܠܥܵܠܡܵܐ.
ܥܵܠܲܡ: ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܠܸܠܝܵܐ ܕܟ݂ܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܫܚܝܼܡܹ̈ܐ: ܠܒܲܪ ܡܢ ܨܵܘܡܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܐ:
ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܚܘܼܣ ܥܲܠ ܥܲܡܵܟ݂: ܘܠܵܐ ܬܸܬܸ݁ܠ ܝܵܪܬ݁ܘܼܬ݂ܵܟ݂ ܠܚܸܣܕܵܐ ܘܡܘܼܝܵܩܵܐ ܕܥܲܡܡܹ̈ܐ. ܚܘܼܪ ܡܢ ܡܸܕ݂ܝܵܪ̈ܲܝܟ ܘܲܚܙܝܼ ܩܛܝܼܪܵܐ: ܕܣܵܒ݂ܠ̇ܝܼܢܲܢ ܡܢ ܒܥܸܠܕ݁ܒ݂ܵܒܹ̈ܐ. ܘܬܸܕ݂ܢܲܚ ܠܲܢ ܒܲܥܓܲܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܕܠܵܐ ܢܹܐܡܪܘܼܢ ܥܲܡܡܹ̈ܐ ܐܲܝܟܵܘ ܐܲܠܵܗܗܘܿܢ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܢܲܛܲܪ ܠܥܲܒ݂ܕܲܝ̈ܟ: ܡܢ ܣܵܢܐܹ̈ܐ ܕܡܲܓܵܢ ܕܨܵܒܹܝ̇ܢ ܠܡܸܚܛܲܦ ܕܠܵܐ ܕܝܼܠܗܘܿܢ. ܣܘܼܓ݂ ܒܐܲܦܲܝ̈ܢ ܩܘܼܫܬܵܟ݂ ܚܲܣܝܼܢܵܐ: ܕܢܸܚܙܘܿܢ ܣܵܢܐܲܝ̈ܢ ܘܢܸܒ݂ܗܬ݂ܘܼܢ. ܘܐܲܟ݂ܪܸܟ݂ ܠܲܢ ܫܘܼܪܵܐ ܒܛܝܼܠܘܼܬ݂ܵܟ݂: ܕܢܸܣܬܲܬܲܪ ܒܓ݂ܵܘܹܗ ܘܢܵܘܕܸ݁ܐ ܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܒܲܨܠܘܿܬ݂ܵܗ̇ ܕܝܵܠܹܕ݂ܬ݂ܵܟ݂ ܡܵܪܲܢ ܝܼܫܘܿܥ: ܫܲܝܸܢ ܥܵܠܡܵܐ ܟܠܹܗ ܕܲܫܓ݂ܝܼܫ ܘܲܕ݂ܠܝܼܚ ܒܲܚܛܵܗܵܘ̈ܗܝ. ܘܒܲܛܸܠ ܡܢ ܐܲܪܥܵܐ ܩܪ̈ܵܒܹܐ ܘܩܹܐܪ̈ܣܹܐ: ܘܫܲܝܸܢ ܟܵܗܢܹ̈ܐ ܥܲܡ ܡܲܠܟܹ̈ܐ. ܕܒ݂ܵܐܘܝܘܼܬ݂ܵܐ ܘܚܘܼܒܵܐ ܢܸܙܕܲܝܲܚ: ܕܘܼܟ݂ܪܵܢ ܝܵܠܹܕ݂ܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ܀ ܙܲܕ݁ܝܼܩܹ̈ܐ ܕܲܫܦܲܪܘ ܠܵܟ݂ ܡܵܪܲܢ ܝܼܫܘܿܥ: ܢܒ݂ܝܼܹ̈ܐ ܫܠܝܼܚܹ̈ܐ ܘܣܵܗܕܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ ܕܲܒ݂ܟ݂ܠ ܐܲܬܲܪ. ܗܸܢܘܿܢ ܢܦܝܼܣܘܼܢܵܟ݂ ܚܠܵܦ ܢܲܦ̮̈ܫܵܬܲܢ: ܕܬܸܥܒܸ݁ܕ݂ ܪ̈ܲܚܡܹܐ ܥܲܠ ܟܠܲܢ. ܘܐܲܫܘܵܐ ܠܲܢ ܕܥܲܡܗܘܿܢ ܢܵܘܕܸ݁ܐ ܠܵܟ݂: ܒܝܵܘܡܵܐ ܕܒܹܗ ܢܵܣ̇ܒ݁ܝܼܢ ܐܲܓ݂ܪܵܐ ܕܥܲܡܠܲܝ̈ܗܘܿܢ܀ ܛܘܼܒܲܝܟ ܠܵܟ݂ ܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ: ܕܫܲܡܠܝܼܬ݁ ܪܲܗܛܵܟ݂ ܒܙܵܟ݂ܘܼܬ݂ܵܐ. ܘܲܦܠܲܚܬ݁ ܒܟܲܪܡܹܗ ܕܲܡܫܝܼܚܵܐ: ܘܲܢܣܲܒ݂ܬ݁ ܐܲܓ݂ܪܵܐ ܚܠܵܦ ܥܲܡܠܲܝ̈ܟ: ܕܹܝܢܵܪ ܛܘܼܒܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܨ̈ܠܵܘܵܬ݂ܵܟ݂ ܢܸܗܘ̈ܝܵܢ ܥܲܠ ܟܠܲܢ: ܘܢܸܫܬ݁ܘܸܐ ܕܥܲܡܵܟ݂ ܢܹܐܪܲܬ݂ ܡܲܠܟ݁ܘܼܬ݂ܵܐ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.

ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܒܵܚܵܐ: ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܢ ܥܸܠܵܝܹ̈ܐ: ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܢ ܬܲܚܬܵܝܹ̈ܐ: ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܘܗܸܕ݂ܪܵܐ ܘܪܘܼܡܪܵܡܵܐ: ܡܢ ܟܠ ܕܲܒ݂ܪܲܝܬ݁ ܘܐܲܬ݂ܩܸܢܬ݁ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ. ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.

ܬܸܫܒ݁ܘܿܚܬܵܐ ܕܫܲܒ݁ܬ݂ܵܐ
ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܩܲܕܝܼܫܵܐ ܡܵܪܝ ܐܲܦܪܹܝܡ ܪܲܒܵܐ
ܒܪܝܼܟ݂ ܐܝܼܬ݂ܝܵܐ ܕܲܒ݂ܪܵܢ ܘܦܲܪܩܲܢ ܒܲܡܫܝܼܚܹܗ: ܘܩܲܪܒܲܢ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܐ܀ ܐܲܠܵܗܵܐ ܡܸܠܬ݂ܵܐ ܝܼܚܝܼܕ݂ܵܝܹܗ ܕܐܲܒ݂ܵܐ: ܠܵܟ݂ ܐܸܫܦܲܪ ܒܲܠܚܘܿܕ݂ ܒܐ݇ܢܵܫܘܼܬܲܢ ܕܲܠܒܸܫܬ݁܀ ܘܠܵܐ ܐܹܦܘܼܫ ܘܐܸܫܠܸܐ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ: ܡܢ ܕܐܵܘܕܸ݁ܐ ܠܲܫܡܵܟ݂ ܒܟ݂ܠܗܘܿܢ ܥܸܕܵܢܹ̈ܐ܀
ܟܵܪܘܿܙܘܼܬ݂ܵܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.܀
ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

ܗܵܘ̇ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ݁ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

ܗܵܘ̇ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡܠܲܟ݂ ܛܵܒ݂̈ܵܬ݂ܵܐ ܠܪ̈ܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕ݁ܬܹܗ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

ܗܵܘ̇ ܕܠܵܐ ܨܵܒܹܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ: ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ܡܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

ܗܵܘ̇ ܕܗܘܝܘ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܪ̈ܝܵܬ݂ܵܐ: ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܝܼܣ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܝܼܣ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܵܘ̇ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

ܗܵܘ̇ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
܀܀܀
