ܛܲܟ݂ܣܵܐ ܕܬܸܫܡܸܫܬܵܐ ܕܠܸܠܝܵܐ
ܘܲܕ݂ܨܲܦܪܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܕܲܫܚܝܼܡܹ̈ܐ
ܕܲܥܪܘܼܒ݂ܬܵܐ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܖ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟ݂: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟ݂: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬ݂ܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܡܵܘܬ݁ܒ݂ܵܐ. ܬܸܒ݂ܣܲܡ ܠܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܚܲܠܵܫܘܼܬܲܢ: ܘܬܸܥܘܿܠ ܩܕ݂ܵܡܲܝܟ ܒܵܥܘܼܬ݂ܵܐ ܕܲܡܚܝܼܠܘܼܬܲܢ: ܘܖ̈ܲܚܡܲܝܟ ܢܸܗܘܘܿܢ ܣܢܹܐܓ݂ܖܹ̈ܐ ܛܵܒܹ̈ܐ ܠܚܲܝܵܒ݂ܘܼܬܲܢ: ܘܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܹܗ ܪܲܒܵܐ ܕܲܚܢܵܢܵܟ݂ ܢܸܬ݂ܦܲܢ̈ܝܵܢ ܫܹܐ̈ܠܵܬ݂ܵܐ ܠܲܣܢܝܼܩܘܼܬܲܢ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.

ܕܠܸܠܝܵܐ. ܒܲܫܠܝܼܚܵܘ̈ܗܝ ܕܡܲܠܟܵܐ. ܐܹܪܲܡܪ̱ܡܵܟ݂ ܡܵܪܝ ܡܲܠܟܵܐ: ܗܸܕ݂ܪܵܟ݂ ܘܫܘܼܒ݂ܚܵܟ݂ ܘܫܘܼܒ݂ܚܵܟ݂ ܙܵܟܹܐ. ܡ݂ܢ ܒܵܬܲܪ ܕܲܒ݂ܫܘܼܒ݂ܚܵܐ ܩܵܡ̣ܬ݁ ܡ݂ܢ ܩܲܒ݂ܪܵܐ. ܬܘܼܒ݂ ܟܲܕ݂ ܐܲܚܝܼܕ݂ܝܼܢ ܬܲܖ̈ܥܹܐ ܥܵܐܹ̇ܠ ܗ̣݇ܘܲܝܬ݁. ܘܠܲܫܠܝܼܚܹ̈ܐ ܥܲܠ ܩܝܵܡܬܵܟ݂ ܡܫܲܪܲܪ ܗ̣݇ܘܲܝܬ݁. ܫܠܵܡܵܐ ܝܲܗ̣݇ܒ݂ܬ݁ ܘܐܸܡܲܪܬ݁ ܠܗܘܿܢ. ܩܲܒܸܠܘ ܪܘܼܚܩܘܼܕ݂ܫܵܐ. ܘܟܲܕ݂ ܥܕܲܟܹܝܠ ܠܵܐ ܐܸܫ̣ܟܲܚܘ ܠܲܡܗܲܝܡܵܢܘܼ. ܕܩܵܡ̣ܬ݁ ܡ݂ܢ ܩܲܒ݂ܪܵܐ ܘܲܫ̣ܪܲܝܬ݁ ܦܟ݂ܵܖ̈ܵܘܗܝ ܕܡܵܘܬܵܐ. ܚ̣ܵܘܝܼܬ݁ ܐܸܢܘܿܢ ܒܦܲܓ݂ܪܵܟ݂ ܕܘܼܟܲܬ݂ ܨܸܨܹ̈ܐ. ܠܵܐ ܗ݇ܘ̣ܵܐ ܕܡܸܬ݂ܚܵܘܹܝܢ ܒܠܵܐ ܡܵܝ̈ܘܿܬܹܐ. ܐܸܠܵܐ ܠܫܘܼܪܵܪܗܘܿܢ. ܡܵܪܹܟܠ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܬܢܝܼ܀ ܥܲܡܵܐ ܣܲܟ݂ܠܵܐ ܘܠܵܐ ܗ݇ܘ̣ܵܐ ܚܲܟܝܼܡܵܐ: ܡ݂ܢ ܒܵܬܲܪ ܩܒ݂ܘܼܪܬܵܟ݂ ܡܫܲܒܲܚܬܵܐ. ܥܲܡܵܐ ܒܝܼܫܵܐ ܘܛܵܠܘܿܡܵܐ. ܐܲܩ̣ܝܼܡ ܗ݇ܘ̣ܵܐ ܐܸܣܛܖ̈ܵܛܝܼܘܿܛܹܐ ܕܢܸܛܪܘܼܢ ܩܲܒ݂ܪܵܟ݂. ܐܘܿ ܠܹܗ ܠܥܲܡܵܐ ܠܵܐ ܡܗܲܝܡܢܵܐ. ܐܸܢܗܘ̤ ܕܲܩ̣ܛܲܠܘ ܐܵܦ ܩ̣ܒܲܪܘ ܠܡܵܐ ܢܵܛ̇ܪܝܼܢ ܗ݇ܘ̣ܵܘ. ܘܐܸܢܗܘ̤ ܕܐܸܣܬܲܪܲܕ݂ܘ ܡܸܢܵܟ݂. ܠܡܵܢܵܐ ܐܲܡ̣ܪܲܚܘ ܘܙܲܩ̣ܦܘܼܟ݂. ܐܸܠܵܐ ܩܝܵܡܬܵܟ݂ ܕܠܲܬ݂ܠܵܬ݂ܵܐ ܝܵܘܡ̈ܝܼܢ. ܗܝ̤ ܐܲܒ݂ܗ̱ܬܲܬ̤ ܠܨܵܠܘܿܒܲܝ̈ܟ. ܘܚܲܕ݂ܝܲܬ̤ ܠܥܹܕܬܵܟ݂. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܘܲܦܪܲܩ̣ܬ݁ ܒܲܕ݂ܪܵܥܵܟ݂ ܠܥܲܡܵܟ݂. ܡ݂ܢ ܒܵܬܲܪ ܩܝܵܡܬܵܟ݂ ܡܫܲܒܲܚܬܵܐ. ܬܘܼܒ݂ ܦܲܛܪܘܿܣ ܢܘܼܢܹ̈ܐ ܨܵܐܹ̇ܕ݂ ܗ݇ܘ̣ܵܐ. ܘܲܠܘܵܬܹܗ ܦܵܪܘܿܩܲܢ ܡܡܲܠܸܠ ܗ̣݇ܘܲܝܬ݁. ܪܥܝܼ ܠܝܼ ܐܸܡܖܹ̈ܐ ܕܥܵܢܲܝ̈. ܪܥܝܼ ܠܝܼ ܥܸܖ̈ܒܲܝ ܘܲܢܩܵܘܵܬ݂̈ܝ. ܕܡܸܛܠܵܬ݂ܗܘܿܢ ܩܕ݂ܵܡ ܦܝܼܠܵܛܘܿܣ ܩܵܡܹ̇ܬ݂. ܘܡܸܛܠܵܬ݂ܗܘܿܢ ܪܘܼܩܵܐ ܒܐܲܦܲܝ̈ ܙܠܲܚ̣ܘ ܠܝܼ. ܘܡܸܛܠܵܬ݂ܗܘܿܢ ܙܩܝܼܦܵܐ ܕܚܸܣܕ݂ܵܐ ܣܲܝ̇ܒܪܹܬ݂. ܘܝܵܘܡܹ̈ܐ ܬܠܵܬ݂ܵܐ ܒܩܲܒ݂ܖܹ̈ܐ ܕܡܝܼ̈ܬܹܐ ܗ݇ܘܹܝ̇ܬ݂. ܐܸܢܵܐ ܕܐܝܼܬܲܝ ܦܵܪܘܿܩܵܐ ܘܪܵܚܹܡܢܵܫ̈ܵܐ.
ܕܲܨܠܘܿܬ݂ܵܐ. ܨܠܘܿܬܲܢ ܬܸܒ݂ܣܲܡ ܠܵܟ݂. ܩܵܠܵܐ: ܟܓ. ܒܪܘܼܟܵܒ݂ ܕܪܸܚܡܹ̇ܬ݂ ܢܵܡܘܿܣܵܟ݂.
ܬܸܥܘܿܠ ܩܕ݂ܵܡܲܝܟ ܨܠܘܿܬ݂ܝ. ܨܠܘܿܬܲܢ ܬܸܒ݂ܣܲܡ ܠܵܟ݂ ܛܵܒ݂ܵܐ ܐܲܠܵܗܲܢ. ܘܩܲܒܸܠ ܒܵܥܘܼܬܲܢ ܘܦܲܢܵܐ ܫܹܐ̈ܠܵܬܲܢ. ܘܫܵܘܙܸܒܲܝܢ ܡ݂ܢ ܐܘܼܠܨܵܢܹ̈ܐ ܕܐܲܡܝܼܢܝܼܢ ܠܹܗ ܠܲܟ݂ܝܵܢܲܢ. ܘܐܲܣܵܐ ܟܹܐܒܹ̈ܐ ܕܦܲܓ݂ܖ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ. ܒܣܲܡܵܐ ܪܲܒܵܐ ܕܲܚܢܵܢܵܟ݂. ܘܐܲܥܒܲܪ ܡܸܢܲܢ ܙܲܒ݂ܢܹ̈ܐ ܡܥܝܼܩܵܢܹ̈ܐ. ܕܐܲܥܝܼܪܘ ܥܠܲܝܢ ܚܛܵܗܲܝ̈ܢ ܘܠܵܐ ܗ݇ܘ̣ܵܐ ܐܲܝܟ݂ ܕܲܚܛܲܝܢ. ܬܸܪܕܹܝܢ ܒܪܘܼܓ݂ܙܵܟ݂ ܓܵܒ݂ܘܿܠܲܢ. ܐܸܠܵܐ ܚܘܼܢܲܝܢ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܪܲܚܸܡ ܥܠܲܝ ܡܵܪܝܵܐ ܡܸܛܠ ܕܲܟ݂ܪܝܼܗ ܐ݇ܢܵܐ. ܪܲܚܸܡ ܥܠܲܝ ܡܵܪܝܵܐ ܡܸܛܠ ܕܲܟ݂ܪܝܼܗ ܐ݇ܢܵܐ. ܘܐܲܣܵܗ̇ ܠܢܲܦ̮ܫܝ ܦܵܪܘܿܩܝ ܡܸܛܠ ܕܲܚ̇ܛܹܝܬ݂ ܠܵܟ݂. ܐܲܟ݂ܡܵܐ ܕܐܲܣܝܼܬ݁ ܠܲܡܫܲܪܝܵܐ. ܘܚܲܣܝܼܬ݁ ܚܵܘܒܹ̈ܐ ܕܚܲܛܵܝܹ̈ܐ. ܘܲܥ̣ܨܲܒ݂ܬ݁ ܚܒ݂ܵܖ̈ܵܬܹܗ ܕܓܸܢܣܲܢ ܫܝܼܛܵܐ. ܒܦܲܓ݂ܪܵܐ ܕܲܢ̣ܣܲܒ݂ܬ݁ ܡ݂ܢ ܒܬ݂ܘܼܠܬܵܐ. ܘܠܵܐ ܢܸܕܲܬ̤ ܠܵܟ݂ ܡܵܪܝ ܡ݂ܢ ܛܲܡܐܘܼܬܲܢ. ܒܲܕ݂ ܡܝܼܬ݂ܝܼܢ ܗ݇ܘܲܝܢ ܒܲܚܛܵܗܲܝ̈ܢ. ܒܚܲܝܠܵܟ݂ ܡܲܐܣܸܐ ܟܠ. ܐܲܣܵܐ ܟܹܐܒܲܝ̈ܢ ܘܟ݂ܘܼܖ̈ܗܵܢܲܝܢ. ܐܵܣܝܲܢ ܛܵܒ݂ܵܐ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܡܪܲܚܡܵܢܲܬ݁ ܡܵܪܝܵܐ ܘܙܲܕܝܼܩ. ܚܘܼܢܲܝܢܝ ܚܲܢܵܢܵܐ ܒܖ̈ܲܚܡܵܘܗܝ ܕܚܲܒܝܼܒ݂ܵܟ݂. ܝܼܫܘܿܥ ܒܪܹܗ ܕܕ݂ܵܘܝܼܕ݂ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝ. ܐܵܣܝܵܐ ܕܐܲܓ݂ܪܵܐ ܠܵܐ ܢܵܣܹܒ݂. ܖ̈ܲܚܡܲܝܟ ܣܢܹܐܓ݂ܖܹ̈ܐ ܢܸܗܘܘܿܢ ܠܝܼ. ܝܵܘܡܲܝ̈ ܟܠܗܘܿܢ ܓܡܲܪܘ ܒܲܣܪܝܼܩܘܼܬ݂ܵܐ. ܘܲܥܪܲܒ݂ ܫܸܡܫܹܗ ܕܗܵܢ ܥܵܠܡܵܐ. ܘܐܸܢ ܗܘ̤ ܕܲܚܢܵܢܵܟ݂ ܠܵܐ ܡܫܵܘܙܸܒ݂ ܠܝܼ. ܗ̤ܘܹܝ̇ܬ݂ ܠܝܼ ܐܘܼܟ݂ܠܵܐ ܠܲܚܛܝܼܬ݂ܵܐ. ܐܸܠܵܐ ܒܝܲܕ ܖ̈ܲܚܡܲܝܟ. ܚܘܼܢܲܝܢܝ ܒܝܵܘܡܵܐ ܕܡܹܐܬ݂ܝܼܬ݂ܵܟ݂. ܡܲܠܟܵܐ ܡܫܝܼܚܵܐ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝ. ܐܲܣܵܢܝ ܡܵܪܝܵܐ ܡܸܛܠ ܕܙܵܘܥܘ ܓܲܖ̈ܡܲܝ. ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ ܐܵܣܝܵܐ ܫܲܪܝܼܪܵܐ. ܐܲܣܵܐ ܠܝܼ ܟܹܐܒܲܝ̈ ܒܣܲܡܵܐ ܕܲܚܢܵܢܵܟ݂. ܐܲܟ݂ܡܵܐ ܕܐܲܣܝܼܬ݁ ܠܲܡܫܲܪܝܵܐ. ܗ̇ܘ ܕܡ݂ܢ ܢܘܼܓ݂ܪܵܐ ܡܢܵܘܲܠ ܗ݇ܘ̣ܵܐ. ܘܐܲܝܟܲܢ ܕܕܲܟܝܼܬ݁ ܠܓܲܪܒܵܐ ܒܦܘܼܩܕܵܢܵܟ݂. ܡ݂ܢ ܟܘܼܬ݂̈ܡܵܬ݂ܵܐ ܕܲܒ݂ܦܲܓ݂ܪܹܗ. ܘܐܲܝܟ݂ ܕܲܠܡܵܟ݂ܣܵܐ ܙܲܕܸܩܬ݁ ܒܛܲܝܒܘܼܬ݂ܵܟ݂ ܘܲܠܓܲܝܵܣܵܐ ܒܲܚܢܵܢܵܟ݂. ܐܵܦ ܠܝܼ ܐܲܚܠܸܡܲܝܢܝ. ܘܐܲܣܵܐ ܟܹܐܒܲܝ̈ ܘܟ݂ܘܼܖ̈ܗܵܢܲܝ. ܒܣܲܡܵܟ݂ ܛܵܒ݂ܵܐ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝ. ܚܢܲܢ ܥܲܡܵܟ݂ ܘܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬ݂ܵܟ݂. ܠܵܟ݂ܘܼ ܩܵܪܹܝܢܲܢ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܪܲܚܸܡ ܥܲܠ ܚܲܝܲܝ̈ܢ ܘܠܵܐ ܬܲܗܡܸܐ ܡܸܢܲܢ. ܘܐܲܫܦܲܥ ܚܢܵܢܵܐ ܕܛܲܝܒܘܼܬ݂ܵܟ݂ ܥܲܠ ܢܲܦ̮̈ܫܵܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܘܗܲܒ݂ܠܲܢ ܙܲܒ݂ܢܹ̈ܐ ܕܲܡܠܹܝܢ ܚܲܕ݂̈ܘܵܬ݂ܵܐ. ܕܲܒ݂ܗܘܿܢ ܢܵܘܕܸܐ ܠܛܲܝܒܘܼܬ݂ܵܟ݂. ܘܢܸܗܘܸܐ ܒܢܲܝ̈ܵܐ ܓܵܡܖ̈ܲܝ ܨܸܒ݂ܝܵܢܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ. ܛܵܒ݂ܵܐ ܕܲܒ݂ܛܝܼܠ ܠܹܗ. ܥܲܠ ܚܲܛܵܝܹ̈ܐ ܕܲܢܬ݂ܘܼܒ݂ܘܼܢ. ܐܲܦܢܵܢ ܨܹܐܕܲܝܟ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܦܬܲܚ ܠܝܼ ܬܲܖ̈ܥܹܐ ܕܙܲܕܝܼܩܘܼܬ݂ܵܐ. ܒܲܬ݂ܪܲܥ ܛܲܝܒܘܼܬ݂ܵܟ݂ ܡܵܪܵܐ ܡܪܲܚܡܵܢܵܐ. ܢܵܩܫܝܼܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܢܸܣܒ݂ܘܼܢ ܥܘܼܕ݂ܖ̈ܵܢܹܐ. ܦܬܲܚ ܠܵܗ̇ ܬܲܪܥܵܐ ܠܲܨܠܘܿܬܲܢ. ܘܩܲܒܸܠ ܒܖ̈ܲܚܡܲܝܟ ܒܵܥܘܼܬܲܢ. ܘܦܲܢܵܐ ܫܹܐ̈ܠܵܬܲܢ ܡ݂ܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂. ܘܲܥܒܸܕ݂ ܖ̈ܲܚܡܹܐ ܥܲܠ ܥܵܠܡܵܐ. ܘܒܲܛܸܠ ܡܸܢܲܢ ܟܠܗܘܿܢ ܢܸܟ݂ܝܵܢܹ̈ܐ. ܘܲܨܠܝܼܒ݂ܵܟ݂ ܢܸܗܘܸܐ ܫܘܼܪܵܐ ܠܥܹܕ݁ܬܵܟ݂ ܩܲܕܝܼܫܬܵܐ. ܕܲܒ݂ܦܘܼܡ ܝܲܠܕܹܝ̈ܗ̇ ܬܵܘܕܸܐ ܠܵܟ݂. ܫܘܼܒ݂ܚܵܐ ܚܲܕ݂ܬ݂ܵܐ ܒܟ݂ܠܗܘܿܢ ܥܸܕܵܢܹ̈ܐ. ܨܘܼܬ݂ ܡܸܠܲܝ̈ ܘܩܲܒܸܠ. ܡܫܝܼܚܵܐ ܡܠܸܐ ܖ̈ܲܚܡܹܐ ܩܲܒܸܠ ܒܵܥܘܼܬܲܢ. ܘܦܲܢܵܐ ܫܹܐ̈ܠܵܬܲܢ ܐܲܟ݂ܡܵܐ ܕܐܸܫܬܵܘܕܝܼܬ݁. ܕܟ݂ܠ ܐܲܝܢܵܐ ܕܫܵܐܹܠ ܢܵܣܹܒ݂. ܘܐܲܝܢܵܐ ܕܢܵܩܹܫ ܡܸܬ݂ܦܬܲܚ ܠܹܗ. ܠܵܟ݂ܘܼ ܩܵܪܹܝܢ ܐܲܝܠܹܝܢ ܕܐܲܠܝܼܨܝܼܢ. ܘܒ݂ܵܟ݂ ܡܸܬ݂ܓܵܘܣܝܼܢ ܕܲܡܥܵܩܝܼܢ. ܗܘ̤ܝܼ ܡܵܪܝ ܫܘܼܪܵܐ ܪܵܡܵܐ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܘܫܵܘܙܸܒ݂ ܐܸܢܘܿܢ ܡ݂ܢ ܒܝܼܫܵܐ. ܕܲܥܠܲܝܟܘܼ ܣܲܒ݂ܪܲܢ. ܘܐܲܢ݇ܬ݁ܘܼ ܐܝܼܬܲܝܟ ܒܹܝܬ݂ ܓܵܘܣܲܢ. ܚܘܼܢܲܝܢ ܒܖ̈ܲܚܡܲܝܟ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܐܲܣܵܐ ܬܒ݂ܵܪܹܗ ܕܠܹܒܝ ܘܲܥܨܘܿܒ݂. ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ ܐܵܣܝܵܐ ܕܡ݂ܢ ܪܵܘܡܵܐ. ܬܵܐ ܡܵܪܝ ܠܥܘܼܕ݂ܪܵܢܲܢ ܘܲܦܪܘܿܩ ܢܲܦ̮̈ܫܵܬܲܢ. ܡ݂ܢ ܒܝܼܫܵܐ ܕܡܲܩܪܸܒ݂ ܥܲܡܲܢ. ܘܨܵܒܹܐ ܕܲܢܨܘܼܕܲܢ ܘܢܵܘܒܕܲܢ. ܗܲܒ݂ܠܲܢ ܚܲܝܠܵܐ ܕܢܸܓ݂ܡܘܿܪ ܨܸܒ݂ܝܵܢܵܟ݂ ܒܲܥܒ݂ܵܕܹ̈ܐ ܕܲܡܝܲܬ݁ܪܘܼܬ݂ܵܐ. ܘܢܸܬ݂ܦܢܸܐ ܡ݂ܢ ܐܘܼܖ̈ܚܵܬܲܢ ܒܝܼ̈ܫܵܬ݂ܵܐ. ܘܢܸܗܘܸܐ ܝܵܖ̈ܬܹ݁ܐ ܒܡܲܠܟܘܼܬ݂ܵܟ݂. ܕܒ݂ܵܟ݂ ܡܸܬ݂ܓܵܘܣܝܼܢܲܢ ܘܡܸܢܵܟ݂ ܖ̈ܲܚܡܹܐ ܫܵܐܠܝܼܢܲܢ. ܚܲܣܵܐ ܚܵܘܒܲܝ̈ܢ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܗ̇ܘ ܕܐܝܼܬ݂ܵܘܗܝ ܡ݂ܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ. ܐܝܼܬ݂ܝܵܐ ܕܠܵܐ ܫܘܼܪܵܝ ܕܥܵܡܲܪ ܒܲܡܖ̈ܵܘܡܹܐ. ܠܵܟ݂ܘܼ ܩܵܪܹܝܢܲܢ ܪܲܚܸܡ ܥܲܠ ܚܲܝܲܝ̈ܢ. ܘܐܲܫܪܵܐ ܫܲܝܢܵܟ݂ ܒܥܵܡܲܪܬܵܐ. ܘܐܲܣܓܵܐ ܫܠܵܡܵܟ݂ ܒܓ݂ܵܘ ܥܹܕܬܵܐ. ܘܩܲܝܸܡ ܟܵܗܢܹ̈ܐ ܒܪܹܫܵܐ ܕܡܲܖ̈ܥܝܵܬ݂ܗܘܿܢ. ܘܛܲܪ ܡܲܠܟܹ̈ܐ ܒܐܘܼܚܕܵܢܲܝ̈ܗܘܿܢ. ܘܒܲܪܸܟ݂ ܟܠܝܼܠܵܐ ܕܫܲܢ݇ܬܵܐ ܒܛܲܝܒܘܼܬ݂ܵܟ݂. ܘܐܲܪܥܵܐ ܬܸܬܸ݁ܠ ܥܲܠ̈ܠܵܬ݂ܵܗ̇. ܕܒܲܚܕ݂ܵܐ ܐܵܘܝܘܼܬ݂ܵܐ. ܢܸܦܪܘܿܥ ܫܘܼܒ݂ܚܵܐ ܕܙܵܕ݂ܹܩ ܠܵܟ݂. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܣܓ݂ܝܼܕ݂ ܡ݂ܢ ܟܠ. ܩܲܕܝܼܫܘܼ ܘܲܕ݂ܚܝܼܠ ܫܡܹܗ. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ ܢܸܬ݂ܩܲܕܲܫ ܒܲܢ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂. ܐܲܟ݂ܡܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܒܲܫܡܲܝܵܐ. ܘܡ݂ܢ ܖ̈ܘܼܚܵܢܹܐ ܡܸܫܬܲܡܲܫ. ܗܵܟܲܢܵܐ ܢܸܬ݁ܬ݁ܢܝܼܚ ܐܵܦ ܒܥܲܒ݂ܕܲܝ̈ܟ. ܘܢܸܦܪܘܿܥ ܠܲܫܡܵܟ݂ ܬܵܘܕܝܼܬ݂ܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܠܡܘܼܠܵܝ ܣܢܝܼܩܘܼܬܲܢ. ܘܠܵܐ ܬܲܥܸܠܲܝܢ ܠܢܸܣܝܘܿܢܵܐ. ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟܘܼܬ݂ܵܐ ܘܓܲܢ݇ܒܵܪܘܼܬ݂ܵܐ. ܐܲܫܘܵܢ ܕܒܲܓ݂ܢܘܿܢܵܟ݂ ܢܸܙܡܲܪ ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ. ܐܸܬ݂ܟܲܪܗܲܬ̤ ܘܐܲܢ݇ܬ݁ ܬܲܩܸܢܬܵܗ̇. ܠܩܲܪܢܵܐ ܕܡܲܪܛܘܼܛܵܟ݂ ܩܸܪܒܲܬ̤ ܡܡܲܚܲܝܬܵܐ. ܘܐܸܬ݂ܐܲܣܝܲܬ̤ ܡ݂ܢ ܟܹܐܒ݂ܵܐ ܥܲܣܩܵܐ ܕܩܲܢܝܵܐ ܗ݇ܘ̣ܵܬ݂. ܐܵܦ ܠܲܢ ܡܵܪܲܢ ܒܲܚܢܵܢܵܟ݂. ܐܲܣܵܐ ܒܪܸܡܙܵܟ݂ ܒܲܣܝܼܡܵܐ. ܘܗܲܒ݂ܠܲܢ ܚܘܼܠܡܵܢ ܦܲܓ݂ܖ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ. ܡ݂ܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܥܲܬ݁ܝܼܪܵܐ. ܘܚܲܣܵܐ ܘܲܥܛܝܼ ܘܐܲܥܒܲܪ ܣܲܟ݂̈ܠ̱ܘܵܬܲܢ. ܒܖ̈ܲܚܡܹܐ ܫܦܝܼܥܹ̈ܐ ܕܛܲܝܒܘܼܬ݂ܵܟ݂. ܕܢܸܦܪܘܿܥ ܬܵܘܕܝܼܬ݂ܵܐ ܠܲܫܡܵܟ݂ ܪܲܒܵܐ ܘܩܲܕܝܼܫܵܐ. ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܲܢ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܬܸܬ݂ܟܬܸܒ݂ ܗܵܕܹܐ ܠܕ݂ܵܪܵܐ ܐ݇ܚܪܹܢܵܐ. ܐܲܝܕܵܐ ܕܐܸܫܬܵܘܝܲܬ̤ ܗ݇ܘ̣ܵܬ݂ ܕܬܸܩܪܘܿܒ݂ ܘܲܕ݂ܬܸܡܫܘܿܚ. ܠܵܟ݂ ܪܹܫܵܐ ܕܚܲܝܲܝ̈ܢ ܡܫܝܼܚܵܐ ܡܸܫܚܵܐ ܕܒ݂ܘܼܣܵܡܹ̈ܐ. ܘܐܸܬ݂ܡܲܠܲܠ ܡܸܕܸܡ ܕܥܸܒ݂ܕܲܬ̤. ܐܲܝܟ݂ ܕܲܠܕ݂ܘܼܟ݂ܪܵܢܵܗ̇ ܒܥܵܠܡܵܐ. ܘܲܒ݂ܣܸܡ ܠܵܟ݂ ܪܹܝܚ ܣܘܵܬ݂ܵܐ ܕܗܲܝܡܵܢܘܼܬ݂ܵܗ̇. ܡ݂ܢ ܓܵܘ ܦܝܼܪܡܵܐ ܕܪܸܥܝܵܢܵܗ̇. ܘܩܲܒܸܠܬܵܗ̇ ܐܲܝܟ݂ ܐܸܠܦܵܐ ܛܥܝܼܢܲܬ݂ ܥܘܼܬ݂ܪܵܐ. ܡܵܪܝܵܐ ܡܙܲܕܸܩ ܠܚܲܛܵܝܹ̈ܐ. ܘܥܲܡܵܗ̇ ܕܚܲܛܵܝܬܵܐ. ܐܵܦ ܚܢܲܢ ܗܵܐ ܩܵܥܹܝܢܲܢ ܚܘܼܢܲܝܢ ܒܖ̈ܲܚܡܲܝܟ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܡܸܛܠ ܕܲܡ̣ܛܵܐ ܙܲܒ݂ܢܵܐ ܠܲܡܪܲܚܵܡܘܼ ܥܠܹܝܗ̇. ܐܲܝܕܵܐ ܕܐܸܬ݂ܟܲܪܗܲܬ̤ ܗ݇ܘ̣ܵܬ݂ ܒܣܘܼܥܖ̈ܵܢܹܐ ܣܢܲܝ̈ܵܐ. ܘܐܸܬ݂ܩܲܒ݁ܠܲܬ̤ ܗ݇ܘ̣ܵܬ݂ ܬܘܼܒ݂ ܡܸܛܠ ܡܸܬ݂ܦܲܢܝܵܢܘܼܬ݂ܵܗ̇. ܫܵܛܝܼܦ̮ܬܵܐ ܕܒܸܣܡܹ̈ܐ ܫܸܩܠܲܬ̤. ܘܩܸܪܒܲܬ̤ ܨܹܝܕ ܡܵܪܲܢ ܘܐܸܡܪܲܬ̤. ܠܵܐ ܡܵܪܝ ܬܲܣܠܹܝܗ̇ ܠܚܲܛܵܝܬܵܐ ܘܬܸܫܕܹ݁ܝܗ̇. ܐܲܢ݇ܬ݁ ܕܐܸܬ݂ܝܼܠܸܕ݁ܬ݁ ܡܸܢ ܒܬ݂ܘܼܠܬܵܐ. ܘܠܵܐ ܬܲܗܡܸܐ ܡܸܢ ܕܸܡܥܹ̈ܐ ܕܐܲܫܦܲܥܝ̈ ܥܲܝܢܲܝ̈. ܐܲܢ݇ܬ݁ ܚܲܕ݂ܘܼܬ݂ܵܐ ܕܡܲܠܲܐܟܲܝ̈ܟ. ܐܸܠܵܐ ܩܲܒܸ݁ܠܲܝܢܝ ܕܬ݂ܵܒܹܬ݂ ܡܸܢ ܚܵܘܒܲܝ̈. ܡܸܛܠ ܣܘܿܓ݂ܐܵܐ ܕܖ̈ܲܚܡܲܝܟ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝ. ܡܸܛܠ ܕܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܛܵܒ݂ ܐܲܢ݇ܬ݁. ܛܵܒ݂ܵܐ ܘܒܲܪ ܛܵܒ݂ܵܐ ܩܲܒܸܠ ܒܵܥܘܼܬܲܢ. ܘܦܲܢܵܐ ܫܸܐ̈ܠܵܬܲܢ ܒܖ̈ܲܚܡܹܐ ܕܛܲܝܒܘܼܬ݂ܵܟ݂. ܕܲܥܠܲܝܟ ܐܝܼܬ݂ܵܘܗܝ ܬܘܼܟ݂ܠܵܢܲܢ. ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ. ܠܵܐ ܬܲܗܡܸܐ ܚܝܵܪܵܟ݂ ܡ݂ܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܡܫܝܼܚܵܐ ܚܵܐܹܢ ܠܚܲܛܵܝܹ̈ܐ. ܕܠܵܟ݂ܘܼ ܩܵܪܹܝܢ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ. ܘܫܵܐܠܝܼܢ ܡܸܢܵܟ݂ ܫܘܼܒ݂ܩܵܢܵܐ. ܐܘܿ ܕܐܸܬ݂ܵܐ ܒܚܘܼܒܹܗ. ܕܢܸܦܪܘܿܩ ܠܓܸܢܣܲܢ ܡ݂ܢ ܒܝܼܫܵܐ. ܦܪܘܿܩܲܝܢ ܒܖ̈ܲܚܡܲܝܟ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܕܓ݂ܵܘܵܐ܀ ܐܹܒܲܪܸܟ݂ ܠܡܵܪܝܵܐ ܒܟ݂ܠܙܒܲܢ. ܒܪܝܼܟ݂ܘܼ ܦܵܪܘܿܩܲܢ ܥܸܠܬ݂ܵܐ ܕܟ݂ܠ ܛܵܒ݂̈ܵܢ. ܕܐܵܘܪܸܒ݂ ܒܲܚܢܵܢܹܗ ܕܘܼܟ݂ܪܵܢ ܝܵܠܹܕܬܹ݁ܗ. ܢܸܗܘܸܐ ܫܡܹܗ ܕܡܵܪܝܵܐ ܡܵܪܝ. ܕܐܲܢ݇ܬ݁ܘܼ ܣܵܘܩܵܐ ܕܚܲܝܲܝ̈ܗܘܿܢ. ܕܐܲܝܠܹܝܢ ܕܲܥܢܲܕ݂ܘ ܘܐܲܝܠܹܝܢ ܕܩܲܝܵܡܝܼܢ. ܕܡܲܪܝܲܡ܀ ܡܸܛܠ ܕܡ݂ܢ ܠܘܵܬܹܗ ܐܸܢܘܿܢ ܖ̈ܲܚܡܹܐ. ܒܬ݂ܘܼܠܬܵܐ ܩܲܕܝܼܫܬܵܐ ܐܸܡܵܐ ܡܒܲܪܲܟ݂ܬܵܐ. ܐܲܦܝܼܣܝ ܠܲܡܫܝܼܚܵܐ ܕܲܕ݂ܢܲܚ ܡ݂ܢ ܥܘܼܒܹܟ݂ܝ. ܕܒܲܨܠܵܘܵܬܹ̈ܟ݂ܝ ܢܸܫܬܲܝܲܢ ܥܵܠܡܵܐ ܕܲܕ݂ܠܝܼܚ ܒܲܚܛܵܗܵܘ̈ܗܝ. ܘܢܸܫܠܘܿܢ ܩܖ̈ܵܒܹܐ ܘܩܹܐܖ̈ܣܹܐ ܡ݂ܢ ܐܲܪܥܵܐ. ܘܟ݂ܵܗܢܹ̈ܐ ܘܡܲܠܟܹ̈ܐ ܢܸܬ݂ܩܲܝܡܘܼܢ. ܟܖ̈ܝܼܗܹܐ ܘܡܲܖ̈ܥܹܐ ܢܩܲܒܠܘܼܢ ܚܘܼܠܡܵܢܵܐ. ܒܝܲܕ݂ ܥܘܼܕ݂ܪܵܢܵܐ ܕܲܨܠܘܿܬܹܟ݂ܝ. ܘܡܵܐ ܕܕ݂ܵܢܲܚ ܝܲܠܕܹܟ݂ܝ. ܠܦܘܼܪܩܵܢ ܟܠܗܹܝܢ ܒܸܖ̈ܝܵܬ݂ܵܐ. ܢܸܫܬ݁ܘܸܐ ܕܥܲܡܹܟ݂ܝ ܢܸܥܘܿܠ ܠܲܓ݂ܢܘܿܢܹܗ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܛܲܪ ܖ̈ܲܚܡܲܝܟ ܠܩܲܖ̈ܝܼܒܲܝܟ. ܒܲܨܠܘܿܬ݂ ܩܲܕܝܼܫܲܝ̈ܟ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܢܒ݂ܝܼܹ̈ܐ ܘܲܫܠܝܼܚܹ̈ܐ ܘܣܵܗܕܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ. ܕܲܦܠܲܚܘ ܒܟܲܪܡܵܐ ܕܲܣܒܲܪܬ݂ܵܟ݂. ܡ݂ܢ ܨܲܦܪܵܐ ܥܕܲܡܵܐ ܠܪܲܡܫܵܐ. ܘܲܢܣܲܒ݂ܘ ܐܲܓ݂ܪܵܐ ܛܵܒ݂ܵܐ ܕܥܲܡܠܲܝ̈ܗܘܿܢ. ܕܹܝܢܵܪ ܛܘܼܒܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܐܵܦ ܠܲܢ ܒܛܲܝܒܘܼ ܫܵܘܬܸ݁ܦ ܒܲܝܢܵܬ݂ܗܘܿܢ. ܕܢܸܗܘܸܐ ܝܵܖ̈ܬܹ݁ܐ ܕܚܘܼܡܣܵܢܗܘܿܢ. ܘܥܲܡܗܘܿܢ ܢܸܬ݂ܒܲܣܲܡ ܒܲܓ݂ܢܘܿܢ ܢܘܼܗܪܵܐ. ܘܛܘܼܒܹ̈ܐ ܕܣܵܟ݂ܵܐ ܠܵܐ ܢܵܣܒܝܼܢ. ܡܵܪܵܐ ܕܟܲܫܝܼܖܹ̈ܐ. ܘܐܲܒ݂ܵܐ ܕܟ݂ܠ ܖ̈ܵܚܡܲܝ ܚܘܼܒܹܗ. ܢܲܛܲܪ ܠܟܸܢܫܲܢ ܒܲܨܠܘܿܬ݂ ܩܲܕܝܼܫܲܝ̈ܟ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܒܵܟ݂ ܢܕܲܩܲܪ ܠܲܥܸܠܕܒ݂ܵܒܲܝ̈ܢ. ܨܠܝܼܒ݂ܵܟ݂ ܙܲܟܵܝܵܐ ܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ. ܕܒܹܗܘ̤ ܐܸܬ݂ܚܲܪܲܪ ܓܸܢܣܲܢ ܡ݂ܢ ܛܘܼܥܝܲܝ. ܗܘ̤ ܢܸܗܘܸܐ ܫܘܼܪܵܐ ܪܵܡܵܐ. ܠܟܸܢܫܹ̈ܐ ܦܖ̈ܝܼܩܲܝ ܒܝܲܕ݂ ܢܝܼܫܵܟ݂. ܘܲܬ݂ܚܹܝܬ݂ ܛܸܠܵܠܵܐ ܕܒܲܣܝܼܡܘܼܬ݂ܵܟ݂. ܣܲܬܲܪ ܚܲܝܲܝ̈ܢ ܡ݂ܢ ܒܝܼܫܵܐ. ܘܲܒ݂ܝܲܕ݂ ܚܲܝܠܵܟ݂ ܪܲܒܵܐ ܘܬܲܩܝܼܦܵܐ. ܒܲܛܸܠ ܩܖ̈ܵܒܹܐ ܘܬܲܟ݂ܬ݁ܘܼܫܹ̈ܐ. ܘܫܲܝܸܢ ܒܛܲܝܒܘܼܬ݂ܵܟ݂. ܟܵܗܢܹ̈ܐ ܘܡܲܠܟܹ̈ܐ ܡܕܲܒܖ̈ܵܢܲܝܢ. ܕܒܲܚܕ݂ܵܐ ܫܲܘܝܘܼ ܢܸܣܓܕ݂ܘܼܢ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܕܐܲܒ݂ܘܼܢ܀ ܢܸܬ݂ܩܲܝܲܡ ܠܥܵܠܲܡ ܩܕ݂ܵܡ ܐܲܠܵܗܵܐ. ܕܘܼܟ݂ܪܵܢܹܗ ܕܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕܝܼܫܵܐ. ܪܫܝܼܡܘܼ ܒܲܫܡܲܝܵܐ ܩܕ݂ܵܡ ܒܹܝܡ ܕܲܡܫܝܼܚܵܐ. ܕܡ݂ܢ ܛܲܠܝܘܼܬܹܗ ܐܸܬ݂ܟܲܫܲܪ. ܘܲܦܠܲܚ ܒܟܲܪܡܵܐ ܪܘܼܚܵܢܵܐ. ܘܐܲܣܠܝܼ ܠܪܵܡܘܼܬ݂ܵܐ ܘܲܠܝܲܥܢܘܼܬ݂ܵܐ. ܘܲܪܚܸܡ ܨܵܘܡܵܐ ܘܲܨܠܘܿܬ݂ܵܐ. ܘܲܒ݂ܝܲܕ݂ ܗܸܪܓܵܐ ܕܣܸܦܖܹ̈ܐ ܖ̈ܘܼܚܵܢܹܐ. ܫܲܠܸܡ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܵܘ̈ܗܝ. ܘܕܲܚܙܵܝܗܝ ܦܵܪܘܿܩܲܢ. ܕܫܲܡܠܝܼ ܘܲܓ݂ܡܲܪ ܐܵܓ݂ܘܿܢܹܗ. ܥܲܒ݂ܕܹ݁ܗ ܒܥܹܕܬܹ݁ܗ ܢܸܒ݂ܥܵܐ ܕܥܘܼܕ݂ܖ̈ܵܢܹܐ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܡܵܪܝܵܐ ܪܸܚ̇ܡܹܬ݂ ܬܸܫܡܸܫܬܹ݁ܗ ܕܒܲܝܬܵܟ݂. ܕܪܸܚܡܹܬ݂ ܢܵܡܘܿܣܵܟ݂ ܘܢܸܛܪܹܬ݂ ܦܘܼܩܕܵܢܲܝ̈ܟ. ܐܸܬ݂ܟܲܬܲܫ ܥܲܡܝ ܒܝܼܫܵܐ ܒܨܸܢ̈ܥܵܬܹܗ ܘܲܙܟ݂ܵܢܝ. ܒܚܵܘܒܲܝ̈ ܗܸܦܟܹܬ݂ ܠܥܲܦܪܵܐ. ܩܵܠܵܟ݂ ܩܵܪܹܐ ܘܲܡܩܝܼܡ ܠܝܼ. ܦܹܐܪܵܐ ܕܲܩܛܲܦ ܗ݇ܘ̣ܵܐ ܐܵܕ݂ܵܡ ܒܲܥܕܸܢ. ܚܲܝܒܲܢܝ ܕܐܹܡܘܼܬ݂ ܒܲܚܛܵܗܲܝ̈. ܘܝܲܠܕܵܐ ܕܡ݂ܢ ܒܬ݂ܘܼܠܬܵܐ ܡܲܪܝܲܡ ܕܢܲܚ. ܙܲܕܩܲܢܝ ܕܐܸܚܸܐ ܒܛܲܝܒܘܼܬ݂ܵܟ݂. ܘܲܠܥܲܒ݂ܕܵܟ݂ ܕܲܫܟܸܒ݂. ܐܲܚܵܝܗܝ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂ ܡܫܝܼܚܵܐ. ܕܐܲܢ݇ܬ݁ܘܼ ܒܲܠܚܘܿܕ݂ ܚܵܐܹܢ ܠܚܲܛܵܝܹ̈ܐ. ܐܸܬ݁ܕܲܟ݂ܪܹܬ݂ ܐܲܠܵܗܵܐ ܘܐܸܬ݁ܕܲܠܚܹܬ݂. ܝܵܘܡܵܐ ܕܥܘܼܢܕܵܢܵܐ ܥܸܗܕܹܬ݂ ܘܐܸܬ݂ܓܲܢܚܹܬ݂. ܕܲܚܙܹܝܬ݂ ܕܲܣܓ݂ܝܼܘ ܚܵܘܒܲܝ̈ ܘܲܥܫܸܢܝ̈ ܣܲܟ݂ܠ̱ܘܵܬ݂̈ܝ. ܘܲܒ݂ܣܘܼܖ̈ܚܵܢܲܝ ܪܵܢܹܐ ܐ݇ܢܵܐ. ܕܲܠܥܹܝܢ ܟܠܢܵܫ ܡܸܬ݂ܦܲܪܣܹܝܢ. ܕܠܵܐ ܐܝܼܬ݂ ܬܲܡܵܢ ܐܲܬ݂ܪܵܐ ܠܲܬ݂ܘܵܬ݂ܵܐ. ܐܵܦ ܠܵܐ ܕܸܡܥܹ̈ܐ ܠܲܦܝܵܣܵܐ. ܕܟ݂ܠܢܵܫ ܐܲܝܟ݂ ܥܲܡܠܹܗ ܢܵܣܹ̇ܒ݂ ܐܲܓ݂ܪܹܗ. ܘܐܲܝܟ݂ ܦܘܼܠܚܵܢܹܗ ܦܘܼܪܥܵܢܹܗ. ܐܸܠܵܐ ܒܵܥܹܐ ܐ݇ܢܵܐ ܒܗܵܘ ܒܹܝܬ݂ ܕܝܼܢܵܐ ܕܟܹܐܢܘܼܬ݂ܵܟ݂. ܪܲܚܸܡ ܥܠܲܝ ܕܠܵܟ݂ܘܼ ܡܣܲܟܸ݁ܐ ܐ݇ܢܵܐ܀
ܫܲܒܲܚ. . ܡ݂ܢ ܒܵܬܲܪ ܕܲܒ݂ܫܘܼܒ݂ܚܵܐ ܩܵܡ̣ܬ݁ ܡ݂ܢ ܩܲܒ݂ܪܵܐ. ܬܘܼܒ݂ ܟܲܕ݂ ܐܲܚܝܼܕ݂ܝܼܢ ܬܲܖ̈ܥܹܐ ܥܵܐܹ̇ܠ ܗ̣݇ܘܲܝܬ݁. ܘܠܲܫܠܝܼܚܹ̈ܐ ܥܲܠ ܩܝܵܡܬܵܟ݂ ܡܫܲܪܲܪ ܗ̣݇ܘܲܝܬ݁. ܫܠܵܡܵܐ ܝܲܗ̣݇ܒ݂ܬ݁ ܘܐܸܡܲܪܬ݁ ܠܗܘܿܢ. ܩܲܒܸܠܘ ܪܘܼܚܩܘܼܕ݂ܫܵܐ. ܘܟܲܕ݂ ܥܕܲܟܹܝܠ ܠܵܐ ܐܸܫ̣ܟܲܚܘ ܠܲܡܗܲܝܡܵܢܘܼ. ܕܩܵܡ̣ܬ݁ ܡ݂ܢ ܩܲܒ݂ܪܵܐ ܘܲܫ̣ܪܲܝܬ݁ ܦܟ݂ܵܖ̈ܵܘܗܝ ܕܡܵܘܬܵܐ. ܚ̣ܵܘܝܼܬ݁ ܐܸܢܘܿܢ ܒܦܲܓ݂ܪܵܟ݂ ܕܘܼܟܲܬ݂ ܨܸܨܹ̈ܐ. ܠܵܐ ܗ݇ܘ̣ܵܐ ܕܡܸܬ݂ܚܵܘܹܝܢ ܒܠܵܐ ܡܵܝ̈ܘܿܬܹܐ. ܐܸܠܵܐ ܠܫܘܼܪܵܪܗܘܿܢ. ܡܵܪܹܟܠ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂.
ܥܵܠܲܡ: ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܠܸܠܝܵܐ ܕܟ݂ܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܫܚܝܼܡܹ̈ܐ: ܠܒܲܪ ܡ̣ܢ ܨܵܘܡܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܐ:
ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܚܘܼܣ ܥܲܠ ܥܲܡܵܟ݂: ܘܠܵܐ ܬܸܬܸ݁ܠ ܝܵܪܬ݁ܘܼܬ݂ܵܟ݂ ܠܚܸܣܕܵܐ ܘܡܘܼܝܵܩܵܐ ܕܥܲܡܡܹ̈ܐ. ܚܘܼܪ ܡ̣ܢ ܡܸܕ݂ܝܵܖ̈ܲܝܟ ܘܲܚܙܝܼ ܩܛܝܼܪܵܐ: ܕܣܵܒ݂ܠ̇ܝܼܢܲܢ ܡ̣ܢ ܒܥܸܠܕ݁ܒ݂ܵܒܹ̈ܐ. ܘܬܸܕ݂ܢܲܚ ܠܲܢ ܒܲܥܓܲܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܕܠܵܐ ܢܹܐܡܪܘܼܢ ܥܲܡܡܹ̈ܐ ܐܲܝܟܵܘ ܐܲܠܵܗܗܘܿܢ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܢܲܛܲܪ ܠܥܲܒ݂ܕܲܝ̈ܟ: ܡ̣ܢ ܣܵܢܐܹ̈ܐ ܕܡܲܓܵܢ ܕܨܵܒܹܝ̇ܢ ܠܡܸܚܛܲܦ ܕܠܵܐ ܕܝܼܠܗܘܿܢ. ܣܘܼܓ݂ ܒܐܲܦܲܝ̈ܢ ܩܘܼܫܬܵܟ݂ ܚܲܣܝܼܢܵܐ: ܕܢܸܚܙܘܿܢ ܣܵܢܐܲܝ̈ܢ ܘܢܸܒ݂ܗ̱ܬ݂ܘܼܢ. ܘܐܲܟ݂ܪܸܟ݂ ܠܲܢ ܫܘܼܪܵܐ ܒܛܝܼܠܘܼܬ݂ܵܟ݂: ܕܢܸܣܬܲܬܲܪ ܒܓ݂ܵܘܹܗ ܘܢܵܘܕܸ݁ܐ ܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܒܲܨܠܘܿܬ݂ܵܗ̇ ܕܝܵܠܹܕ݂ܬ݂ܵܟ݂ ܡܵܪܲܢ ܝܼܫܘܿܥ: ܫܲܝܸܢ ܥܵܠܡܵܐ ܟܠܹܗ ܕܲܫܓ݂ܝܼܫ ܘܲܕ݂ܠܝܼܚ ܒܲܚܛܵܗܵܘ̈ܗܝ. ܘܒܲܛܸܠ ܡ̣ܢ ܐܲܪܥܵܐ ܩܖ̈ܵܒܹܐ ܘܩܹܐܖ̈ܣܹܐ: ܘܫܲܝܸܢ ܟܵܗܢܹ̈ܐ ܥܲܡ ܡܲܠܟܹ̈ܐ. ܕܒ݂ܵܐܘܝܘܼܬ݂ܵܐ ܘܚܘܼܒܵܐ ܢܸܙܕܲܝܲܚ: ܕܘܼܟ݂ܪܵܢ ܝܵܠܹܕ݂ܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ܀ ܙܲܕ݁ܝܼܩܹ̈ܐ ܕܲܫ̣ܦܲܪܘ ܠܵܟ݂ ܡܵܪܲܢ ܝܼܫܘܿܥ: ܢܒ݂ܝܼܹ̈ܐ ܫܠܝܼܚܹ̈ܐ ܘܣܵܗܕܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ ܕܲܒ݂ܟ݂ܠ ܐܲܬܲܪ. ܗܸܢ̣ܘܿܢ ܢܦܝܼܣܘܼܢܵܟ݂ ܚܠܵܦ ܢܲܦ̮̈ܫܵܬܲܢ: ܕܬܸܥܒܸ݁ܕ݂ ܖ̈ܲܚܡܹܐ ܥܲܠ ܟܠܲܢ. ܘܐܲܫܘܵܐ ܠܲܢ ܕܥܲܡܗܘܿܢ ܢܵܘܕܸ݁ܐ ܠܵܟ݂: ܒܝܵܘܡܵܐ ܕܒܹܗ ܢܵܣ̇ܒ݁ܝܼܢ ܐܲܓ݂ܪܵܐ ܕܥܲܡܠܲܝ̈ܗܘܿܢ܀ ܛܘܼܒܲܝܟ ܠܵܟ݂ ܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ: ܕܫܲܡ̣ܠܝܼܬ݁ ܪܲܗܛܵܟ݂ ܒܙܵܟ݂ܘܼܬ݂ܵܐ. ܘܲܦܠܲܚ̣ܬ݁ ܒܟܲܪܡܹܗ ܕܲܡܫܝܼܚܵܐ: ܘܲܢ̣ܣܲܒ݂ܬ݁ ܐܲܓ݂ܪܵܐ ܚܠܵܦ ܥܲܡܠܲܝ̈ܟ: ܕܹܝܢܵܪ ܛܘܼܒܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܨ̈ܠܵܘܵܬ݂ܵܟ݂ ܢܸܗܘ̈ܝܵܢ ܥܲܠ ܟܠܲܢ: ܘܢܸܫܬ݁ܘܸܐ ܕܥܲܡܵܟ݂ ܢܹܐܪܲܬ݂ ܡܲܠܟ݁ܘܼܬ݂ܵܐ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܒܵܚܵܐ: ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡ̣ܢ ܥܸܠܵܝܹ̈ܐ: ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡ̣ܢ ܬܲܚܬܵܝܹ̈ܐ: ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܘܗܸܕ݂ܪܵܐ ܘܪܘܼܡܪܵܡܵܐ: ܡ̣ܢ ܟܠ ܕܲܒ݂ܪܲܝܬ݁ ܘܐܲܬ݂ܩܸܢ̣ܬ݁ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ. ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܫܘܼܒܵܚܹ̈ܐ ܘܬܸܫ̈ܒ݁ܚܵܬ݂ܵܐ ܕܡܸܬ݂ܐܲܡܪܝܼܢ ܒܟ݂ܠܵܗ̇ ܫܲܢ݇ܬܵܐ: ܒܟ݂ܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܫܚܝܼܡܹ̈ܐ ܕܫܲܒ݁ܬ݂ܵܐ.
ܫܘܼܒܵܚܵܐ ܕܲܥܪܘܼܒ݂ܬܵܐ

 ܘܐܸܡܵܐ ܕܲܒ̣ܢܲܝ̈ܵܐ ܟܲܕ݂ ܚܵܕܼ݁ܝܵܐ. ܫܲܒܲܚ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.ܓ: ܙܲܒ݂ܢܝܼ̈ܢ܀ ܥܵܠܲܡ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.ܓ: ܙܲܒ݂ܢܝܼ̈ܢ܀ ܫܲܒܲܚܘ ܥܲܒ̣ܕܵܘ̈ܗܝ ܕܡܵܪܝܵܐ* ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܀
ܬܸܫܒ݁ܘܿܚܬܵܐ ܕܲܥܪܘܼܒ݂ܬܵܐ. ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܡܵܪܝ ܐܲܒ݂ܪܵܗܵܡ ܢܸܬ݂ܦ̮ܪܵܝܵܐ:
ܘܐ݇ܢܵܫ̈ܝܼܢ ܒܫܸܡ ܡܵܪܝ ܝܘܿܚܲܢܵܢ ܕܒܹܝܬ݂ ܪܲܒܲܢ ܟܲܬ݂ܒ݁ܘܼܗ̇
ܫܘܼܒ݂ܚܵܐ ܠܛܵܒ݂ܵܐ ܕܲܒ݂ܝܲܕ݂ ܚܘܼܒܹ݁ܗ: ܓܠ̣ܵܐ ܬܸܫܒ݁ܘܿܚܬܹ݁ܗ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ

܀

ܒ̣ܪܵܐ ܡ̣ܢ ܥܲܦܪܵܐ ܟܝܵܢܵܐ ܚܲܪܫܵܐ: ܘܨܲܒ݁ܬܹܗ ܒܢܲܦ̮ܫܵܐ ܡܵܪܲܬ݂ ܓܲܙܹ̈ܐ

܀

ܘܣ̣ܵܡ ܬܵܘܕ݁ܝܼܬܹܗ ܒܦܲܓ݂ܪܵܐ ܫܝܼܛܵܐ: ܕܟ݂ܠܵܗ̇ ܒܪܝܼܬ݂ܵܐ ܫܘܼܒ݂ܚܹܗ ܬܸܙܡܲܪ

܀

ܬܵܘ ܡܲܠܵܠܹ̈ܐ ܙܡܲܪܘ ܠܹܗ ܫܘܼܒ݂ܚܵܐ: ܥܲܕ݂ܠܵܐ ܢܸܕ݂ܡܲܟ݂ ܒܫܸܢܲܬ݂ ܡܵܘܬܵܐ

܀

ܒܠܸܠܝܵܐ ܕܐܲܪܝܼܟ݂ ܢܸܬ݁ܕ݁ܟܲܪ ܡܵܘܬܲܢ: ܕܒ݂ܵܠܹ̇ܡ ܦܘܼܡܲܢ ܘܲܡܫܲܬܸ݁ܩܠܲܢ

܀

ܟܹܐܢܹ̈ܐ ܕܫܲܒ݁ܚܘܼܗܝ ܒܠܲܝ̈ܠܵܘܵܬ݂ܵܐ: ܟܲܕ݂ ܡܲܝܝܼܬ݂ܝܼܢ ܚܲܝܹ̈ܐ ܐܸܢܘܿܢ

܀

ܘܒ݂ܝܼܫܹ̈ܐ ܕܲܛܠܲܡ̣ܘ ܫܘܼܒ݂ܚܹܗ ܪܲܒܵܐ: ܐܵܦ ܟܲܕ݂ ܚܲܝܝܼ̈ܢ ܡܝܼܬܹ̈ܐ ܐܸܢܘܿܢ

܀

ܢܥܝܼܪܝܼܘܗܝ ܠܦܲܓ݂ܪܲܢ ܒܲܨ̈ܠܵܘܵܬ݂ܵܐ: ܘܲܒ݂ܗܘܼܠܵܠܹ̈ܐ ܕܚܲܝܠܵܐ ܟܲܣܝܵܐ

܀

ܕܢܸܗܘܸܐ ܚܲܒ݂ܖܹ̈ܐ ܠܲܒ݂ܬ݂ܘܼ̈ܠܵܬ݂ܵܐ: ܚܲܟ݁ܝܼ̈ܡܵܬ݂ܵܐ ܕܩܲܠܸܣ ܡܵܪܲܢ

܀

ܘܲܒ݂ܗܵܘ̇ ܠܸܠܝܵܐ ܕܡܲܪܗܸܒ݂ ܥܵܠܡܹ̈ܐ: ܟܲܕ݂ ܥܝܼܪܝܼܢܲܢ ܢܸܚܙܹܝܘܗܝ ܠܲܒ݂ܪܵܐ

܀

ܕܠܵܐ ܢܸܬ݁ܛܲܒܲܥ ܒܲܖ̈ܓ݂ܝܼܓ݂ܵܬ݂ܵܐ: ܕܢܸܚܙܸܐ ܫܘܼܒ݂ܚܹܗ ܒܝܵܘܡܵܐ ܕܕ݂ܵܢܲܚ̇

܀

ܘܢܸܗܘܸܐ ܩܕ݂ܵܡܵܘܗܝ ܥܲܒ݂̈ܕܹ݁ܐ ܥܝܼܖܹ̈ܐ: ܒܫܵܥܬ݂ܵܐ ܕܕ݂ܵܒܲܪ ܠܲܒ݂ܢܲܝ̈ ܓܢܘܿܢܹܗ

܀

ܘܦܵܝܫ̇ܝܼܢ ܒܝܼܫܹ̈ܐ ܒܓ݂ܵܘ ܫܘܼܢܵܩܵܐ: ܘܡܸܬ݁ܬ݁ܚܸܕ݂ ܡ̣ܢ ܫܹܠܝ ܬܲܪܥܵܐ ܕܖ̈ܲܚܡܹܐ

܀

ܥܲܕ݂ ܚܲܝܝܼܢܲܢ ܢܸܥܡܲܠ ܩܲܠܝܼܠ: ܕܒ݂ܵܬܲܪ ܡܵܘܬܵܐ ܝܘܿܡ ܦܘܼܪܥܵܢܵܐ ܗܘ̣

܀

ܦܲܓ݂ܪܵܐ ܕܠܵܐܹ̇ܐ ܒܲܨ̈ܠܵܘܵܬ݂ܵܐ: ܒܝܘܿܡ ܢܘܼܚܵܡܵܐ ܒܐܵܐܲܪ ܦܵܪܲܚ̇

܀

ܘܚܵܙܹ̇ܐ ܠܡܵܪܲܢ ܕܠܵܐ ܟܘܼܚܵܕ݂ܵܐ: ܘܥܵܐܹ̇ܠ ܥܲܡܹܗ ܠܒܹܝܬ݂ ܡܲܠܟ݁ܘܼܬ݂ܵܐ

܀

ܘܲܡܚܲܒ݁ܒ݂ܝܼܢ ܠܹܗ ܥܝܼܖܹ̈ܐ ܘܟܹܐܢܹ̈ܐ: ܕܲܫ̣ܗܲܪܘ ܘܲܥ̣ܡܲܠܘ ܒܲܨ̈ܠܵܘܵܬ݂ܵܐ

܀

ܒܪܝܼܟ݂ܘܼ ܕܥܲܒ݂ܕܲܢ ܡܵܐܢܹ̈ܐ ܕܫܘܼܒ݂ܚܹܗ: ܘܣ̣ܵܡ ܬܵܘܕ݁ܝܼܬܹܗ ܒܦܘܼܡ ܕܲܚܝܼܚܵܐ

܀

ܘܫܘܼܒ݂ܚܵܐ ܠܖ̈ܲܚܡܵܘܗܝ ܕܲܠܥܲܦܖ̈ܵܢܹܐ: ܫܵܘܬܸ݁ܦ ܐܸܢܘܿܢ ܥܲܡ ܖ̈ܘܼܚܵܢܹܐ

܀

ܕܢܸܗܘܘܿܢ ܙܵܡ̇ܪܝܼܢ ܒܟ݂ܠ ܠܲܝ̈ܠܵܘܵܢ: ܘܲܒ݂ܟ݂ܠܥܸܕܵܢ ܩܲܕ݁ܝܼܫ ܠܲܫܡܹܗ

܀

ܘܢܹܐܡܲܪ ܟܠܲܢ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ: ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀
ܟܵܪܘܿܙܘܼܬ݂ܵܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.܀
ܡܵܪܝܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܖ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀

ܗܵܘ̇ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ̇ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀

ܗܵܘ̇ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡ̣ܠܲܟ݂ ܛܵܒ̣̈ܵܬ݂ܵܐ ܠܖ̈ܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕ݁ܬܹܗ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀

ܗܵܘ̇ ܕܠܵܐ ܨܵܒܹܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ: ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ܡ̣ܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀

ܗܵܘ̇ ܕܗܘ̤ܝܘ̣ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܖ̈ܝܵܬ݂ܵܐ: ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀

ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܝܼܣ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܝܼܣ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀

ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܵܘ̇ ܕܲܒ݂ܖ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀

ܗܵܘ̇ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀

ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
