ܛܲܟ݂ܣܵܐ ܕܬܸܫܡܸܫܬܵܐ ܕܠܸܠܝܵܐ
ܘܲܕ݂ܨܲܦܪܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܕܲܫܚܝܼܡܹ̈ܐ
ܕܚܲܡܫܵܒ݁ܫܲܒܵܐ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܖ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟ݂: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟ݂: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬ݂ܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܡܵܘܬ݁ܒ݂ܵܐ. ܬܸܒ݂ܣܲܡ ܠܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܚܲܠܵܫܘܼܬܲܢ: ܘܬܸܥܘܿܠ ܩܕ݂ܵܡܲܝܟ ܒܵܥܘܼܬ݂ܵܐ ܕܲܡܚܝܼܠܘܼܬܲܢ: ܘܖ̈ܲܚܡܲܝܟ ܢܸܗܘܘܿܢ ܣܢܹܐܓ݂ܖܹ̈ܐ ܛܵܒܹ̈ܐ ܠܚܲܝܵܒ݂ܘܼܬܲܢ: ܘܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܹܗ ܪܲܒܵܐ ܕܲܚܢܵܢܵܟ݂ ܢܸܬ݂ܦܲܢ̈ܝܵܢ ܫܹܐ̈ܠܵܬ݂ܵܐ ܠܲܣܢܝܼܩܘܼܬܲܢ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܡܵܘܬ݁ܒ݂ܵܐ.
ܕܠܸܠܝܵܐ. ܒܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ. ܡܸܟܹܝܠ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ: ܢܸܒ݂ܗ̱ܬ݂ܘܼܢ ܟܠ ܥܵܒ݂ܕܲܝ̈ ܥܵܘ̣ܠܵܐ. ܐܲܒ݂ܝܼܠ ܡܵܘܬܵܐ ܘܲܟ݂ܡܝܼܪ ܣܵܛܵܢܵܐ. ܘܚܵ̇ܕ݂ܝܵܐ ܥܹܕ݁ܬܵܐ ܒܲܩܝܵܡܬܹܗ ܕܲܒ݂ܪܵܐ. ܬܢܝܼ܀ ܚܲܕ݂ܘܼܬ݂ܵܐ ܒܟ݂ܠܵܗ̇ ܐܲܪܥܵܐ. ܒܲܩܝܵܡܬܹܗ ܕܲܒ݂ܪܵܐ ܖ̈ܵܘܙܵܢ ܒܸܖ̈ܝܵܬ݂ܵܐ. ܕܲܗܘ̤ܵܬ݂ ܬܲܪܥܘܼܬ݂ܵܐ ܘܐܲܡ̣ܠܸܟ݂ ܢܘܼܚܵܡܵܐ. ܢܸܒ݂ܗ̱ܬ݂ܘܼܢ ܟܠ ܕܥܵܒ݂ܕܝܼ̇ܢ ܦܬܲܟ݂ܖܹ̈ܐ. ܐܸܒ݂ܠܵܐ ܠܚܲܢܦܹ̈ܐ. ܘܒܸܗܬ݁ܬ݂ܵܐ ܠܟ݂ܵܦܘܿܖܹ̈ܐ. ܘܛܘܼܒ݂ܵܐ ܠܥܹܕ݁ܬܵܐ ܘܘܵܝܵܐ ܠܙܵܩܘܿܦܹ̈ܐ.

ܠܸܒܵܐ ܕܲܟ݂ܝܵܐ ܒܪܝܼ ܒܝܼ ܐܲܠܵܗܵܐ. ܗܲܒ݂ܠܲܢ ܡܫܝܼܚܵܐ: ܕܲܒ݂ܠܸܒܵܐ ܕܲܟ݂ܝܵܐ. ܘܒܲܥܒ݂ܵܕܹ̈ܐ ܛܵܒܹ̈ܐ: ܢܸܓ݂ܡܘܿܪ ܨܸܒ݂ܝܵܢܵܟ݂. ܚܵܘܵܢ ܡܵܪܝܵܐ ܛܲܝܒܘܼܬ݂ܵܟ݂ ܘܦܘܼܪܩܵܢܵܟ݂ ܗܲܒ݂ܠܲܢ. ܗܲܒ݂ܠܲܢ ܡܫܝܼܚܵܐ: ܕܲܒ݂ܓܲܠܝܘܼܬ݂ ܐܲܦܹ̈ܐ. ܢܲܩܒܸܠ ܓܸܠܝܵܢܵܟ݂: ܒܝܵܘܡܵܐ ܕܡܹܐܬ݂ܝܼܬ݂ܵܟ݂. ܦܬܲܚ ܠܝܼ ܬܲܖ̈ܥܹܐ ܕܙܲܕܝܼܩܘܼܬ݂ܵܐ. ܬܲܪܥܵܐ ܕܩܘܼܫܬܵܐ: ܕܲܦܬ݂ܝܼܚ ܠܐܲܒܝܼ̈ܕܹܐ. ܩܪܝܼ ܠܲܢ ܕܢܸܥܘܿܠ: ܠܒܹܝܬ݂ ܓܲܙܵܟ݂ ܕܲܠܥܸܠ. ܕܐܸܥܘܿܠ ܒܗܘܿܢ ܘܐܵܘܕܸܐ ܠܡܵܪܝܵܐ. ܢܸܬ݂ܦܲܬ݂ܚܘܼܢ ܬܲܖ̈ܥܲܝܟܝ: ܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ. ܘܬܸܥܘܿܠ ܒܵܥܘܼܬܲܢ: ܩܕ݂ܵܡ ܒܹܝܡ ܕܲܡܫܝܼܚܵܐ. ܠܵܐ ܬܲܗܦܸܟ݂ ܐܲܦܲܝ̈ܟ ܡܸܢܲܢ. ܠܵܐ ܡܵܪܝ ܬܲܗܦܸܟ݂: ܐܲܦܲܝ̈ܟ ܡ݂ܢ ܥܲܒ݂ܕܲܝ̈ܟ. ܕܲܠܘܵܬ݂ܵܟ݂ ܚܵܝܪܝܼܢ: ܕܢܸܣܒ݂ܘܼܢ ܥܘܼܕ݂ܖ̈ܵܢܹܐ. ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈. ܠܵܟ݂ܘܼ ܩܵܪܹܝܢܲܢ: ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܪܲܚܸܡ ܥܲܠ ܚܲܝܲܝ̈ܢ: ܘܠܵܐ ܬܲܗܡܸܐ ܡܸܢܲܢ. ܥܲܠ ܛܲܝܒܘܼܬ݂ܵܟ݂ ܘܥܲܠ ܩܘܼܫܬܵܟ݂. ܠܵܟ݂ ܢܵܘܕܸܐ ܘܢܸܣܓܘܿܕ݂: ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܕܲܦܬ݂ܝܼܚܘܼ ܬܲܪܥܵܟ݂: ܠܟ݂ܠ ܕܬ݂ܵܝܒܝܼܢ ܠܘܵܬ݂ܵܟ݂. ܡܪܲܡ̱ܪܡܵܢܝ ܡ݂ܢ ܬܲܖ̈ܥܵܘܗܝ ܕܡܵܘܬܵܐ. ܡܫܝܼܚܵܐ ܕܐܲܪܝܼܡ: ܫܸܦܠܲܢ ܘܐܲܩܝܼܡܲܢ. ܟܐܝܼ ܒܹܗ ܒܒ݂ܝܼܫܵܐ: ܕܕ݂ܵܠܲܚ ܡܲܫܪܝܼܬܲܢ. ܐܸܬ݂ܦܲܢ ܥܠܲܝ ܘܪܲܚܸܡ ܥܠܲܝ. ܡܫܝܼܚܵܐ ܕܠܵܐ ܡܲܗܡܸܐ: ܡ݂ܢ ܟܠ ܕܩܵܪܹܝܢ ܠܹܗ. ܒܖ̈ܲܚܡܲܝܟ ܠܵܐ ܬܲܣܠܸܐ ܒܵܥܘܼܬ݂ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܠܵܐ ܬܸܬܸ݁ܠ ܡܵܪܝܵܐ ܪܸܓܬܹܗ ܠܥܵܘܵܠܵܐ. ܡܵܪܲܢ ܒܝܲܡܝܼܢܵܟ݂: ܣܚܘܿܦܲܝܗܝ ܠܣܵܛܵܢܵܐ. ܕܡܲܪܘܸܐ ܕܠܵܐ ܚܲܡܪܵܐ: ܘܡܲܫܪܲܥ ܕܠܵܐ ܛܝܼܢܵܐ. ܡܵܪܝܵܐ ܖ̈ܲܚܡܲܝܟ ܠܥܵܠܲܡ. ܢܹܐܬ݂ܘܿܢ ܠܐܝܼܵܠܲܢ: ܖ̈ܲܚܡܲܝܟ ܕܲܒ݂ܪܵܐܘܼܢ. ܘܢܲܐܣܘܿܢ ܟܪܝܼܗܘܼܬܲܢ: ܒܣܲܡܵܐ ܕܲܚܢܵܢܵܟ݂. ܐܸܬ݂ܦܲܢ ܡܵܪܝܵܐ ܥܕܲܡܵܐ ܠܐܸܡܲܬ݂ܝ. ܥܲܡܹܗ ܕܗ̇ܘ ܡܲܟ݂ܣܵܐ: ܫܵܐܠܝܼܢܲܢ ܖ̈ܲܚܡܹܐ. ܚܘܼܢܲܝܢ ܐܲܠܵܗܵܐ: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܡܸܛܠ ܕܛܵܒ݂ܝܼܢ ܖ̈ܲܚܡܲܝܟ ܡ݂ܢ ܚܲܝܹ̈ܐ. ܒܐܝܼܕܲܝ̈ܟ ܐܲܠܵܗܵܐ: ܕܣܲܓܝܼܐܝܼܢ ܖ̈ܲܚܡܲܝܟ. ܬܸܗܘܸܐ ܡܲܪܕܘܼܬܲܢ: ܘܠܵܐ ܒܐܝܼܕܲܝ̈ ܐ݇ܢܵܫ̈ܵܐ. ܥܲܕܲܪܲܝܢ ܐܲܠܵܗܵܐ ܦܵܪܘܿܩܲܢ. ܬܵܐ ܡܵܪܝ ܠܥܘܼܕ݂ܪܵܢܲܢ: ܘܚܲܝܸܠ ܡܚܝܼܠܘܼܬܲܢ. ܕܲܥܠܲܝܟܘܼ ܣܲܒ݂ܪܲܢ: ܒܠܸܠܝܵܐ ܘܐܝܼܡܵܡܵܐ. ܡܪܲܚܡܵܢܲܬ݁ ܡܵܪܝܵܐ ܘܙܲܕܝܼܩ. ܖ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ: ܡܸܢܵܟ݂ ܫܵܐܠܝܼܢܲܢ. ܠܵܐ ܬܸܟ݂ܠܸܐ ܡܸܢܲܢ: ܐܲܢ݇ܬ݁ ܪܵܚܹܡ ܐ݇ܢܵܫܵܐ. ܡܒܲܛܸܠ ܩܖ̈ܵܒܹܐ ܡ݂ܢ ܣܵܘܦܹܝ̈ܗ݁ ܕܐܲܪܥܵܐ. ܩܖ̈ܵܒܹܐ ܡܵܪܝ ܢܸܫܠܘܿܢ: ܘܚܸܖ̈ܝܵܢܹܐ ܢܢܘܼܚܘܼܢ. ܘܫܲܝܢܵܟ݂ ܢܲܡܠܸܟ݂: ܒܐܲܖ̈ܒܲܥܦܸܢܝܵܬ݂ܵܐ. ܥܢܝܼܢ ܐܲܠܵܗܵܐ ܦܵܪܘܿܩܲܢ. ܐܹܝܠ ܐܲܠܵܗܵܐ: ܕܥܵܢܹܐ ܠܙܲܕܝܼܩܹ̈ܐ. ܬܵܐ ܠܐܝܼܵܠܵܐ: ܕܥܲܒ݂ܕܲܝ̈ܟ ܕܩܵܪܹܝܢ ܠܵܟ݂. ܢܲܛܲܪܲܝܢܝ ܐܲܝܟ݂ ܒܵܒ݂ܬ݂ܵܐ ܕܥܲܝܢܵܐ. ܡܵܪܲܢ ܢܲܛܲܪܲܝܢ: ܕܐܲܝܟ݂ ܐܸܡܖܹ̈ܐ ܐܝܼܬܲܝܢ. ܒܲܝܢܵܬ݂ ܚܲܖ̈ܡܵܢܹܐ: ܕܒ݂ܝܼܫܝܼܢ ܡ݂ܢ ܕܹܐܒܹ̈ܐ. ܨܘܼܬ݂ ܡܸܠܲܝ̈ ܘܩܲܒܸܠ. ܡܫܝܼܚܵܐ ܡܲܚܝܵܢܲܢ: ܨܘܼܬ݂ ܒܚܘܼܒܵܟ݂ ܨܠܘܿܬܲܢ. ܘܲܥܒܸܕ݂ ܫܲܝܢܵܐ: ܒܐܲܖ̈ܒܲܥܦܸܢܝܵܬ݂ܵܐ. ܡܵܪܝܵܐ ܠܢܘܼܓ݂ܪܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ. ܫܠܵܡܵܐ ܘܫܲܝܢܵܐ: ܐܲܣܓܵܐ ܒܝܵܘ̈ܡܵܬܲܢ. ܘܒܲܛܸܠ ܡܸܢܲܢ: ܩܖ̈ܵܒ݂ܵܘܗܝ ܕܲܢܟ݂ܝܼܠܵܐ. ܡܵܪܝܵܐ ܢܒܲܪܸܟ݂ ܠܥܲܡܹܗ ܒܲܫܠܵܡܵܐ. ܠܟܸܢܫܲܢ ܡܵܪܲܢ: ܒܲܪܸܟ݂ ܘܢܲܛܲܪ. ܘܐܲܫܪܵܐ ܒܓ݂ܵܘܹܗ: ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂. ܡܸܟܹܝܠ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ. ܢܲܛܲܪ ܡܵܪܲܢ: ܠܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܡ݂ܢ ܨܸܢܥܵܬܹ̈ܗ: ܕܒ݂ܝܼܫܵܐ ܒܥܸܠܕܲܪܵܐ. ܢܫܲܒܲܚ ܠܡܵܪܝܵܐ ܓܲܐܝܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ: ܘܬ݂ܵܘܕܝܼܬ݂ܵܐ ܠܲܒ݂ܪܵܐ. ܘܲܠܪܘܼܚ ܩܘܼܕ݂ܫܵܐ: ܣܸܓ݂ܕܬ݂ܵܐ ܘܪܘܼܡܪܵܡܵܐ. ܠܥܵܠܲܡ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ. ܥܕܲܡܵܐ ܠܥܵܠܲܡ: ܠܵܟ݂ ܝܵܐܹܐ ܫܘܼܒ݂ܚܵܐ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ: ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܕܡܲܪܝܲܡ܀ ܥܕܲܡܵܐ ܠܡܘܼܠܵܝܵܐ ܕܙܲܒ݂ܢܹ̈ܐ. ܨܠܘܿܬ݂ܵܗ̇ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ: ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܲܢ. ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡ݂ܢ ܒܝܼܫܵܐ ܘܚܲܝܠܹܗ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܐܲܢ݇ܬ݁ܩܲܕܝܼܫ ܘܝܵܬܹܒ݂ ܒܫܘܼܒ݂ܚܵܟ݂ ܝܼܣܪܵܝܹܠ. ܨܠܘܿܬ݂ܗܘܿܢ ܕܩܲܕܝܼܫܲܝ̈ܟ: ܕܦܸܬ݂ܚܲܬ̤ ܬܪܲܥ ܪܵܘܡܵܐ. ܗܝ̤ ܡܵܪܝ ܬܸܦ̮ܬܲܚ ܠܵܗ̇: ܬܲܪܥܵܐ ܠܲܨܠܘܿܬܲܢ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܐܲܚܘܿܕ݂ ܙܲܝܢܵܐ ܘܣܲܟܪܵܐ ܘܩܘܼܡ ܒܥܘܼܕ݂ܪܵܢܝ. ܨܠܝܼܒ݂ܵܟ݂ ܦܵܪܘܿܩܲܢ: ܙܲܝܢܵܐ ܢܸܗܘܸܐ ܠܲܢ. ܘܒܹܗ ܢܸܙܟܹ݁ܝܘܗܝ ܠܒ݂ܝܼܫܵܐ: ܘܟ݂ܠܗܹܝܢ ܨܸܢܥܵܬܹ̈ܗ. ܕܐܲܒ݂ܘܼܢ܀ ܒܥܝܼ ܡ݂ܢ ܡܵܪܝܵܐ ܘܨܲܠܵܐ ܩܕ݂ܵܡܵܘܗܝ. ܐܲܒ݂ܘܼܢ ܩܲܕܝܼܫܵܐ: ܐܲܦܝܼܣ ܠܲܡܫܝܼܚܵܐ. ܕܲܒ݂ܝܲܕ݂ ܨ̈ܠܵܘܵܬ݂ܵܟ݂: ܢܸܗܘܘܿܢ ܥܠܲܝܢ ܖ̈ܲܚܡܹܐ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܙܥܘܿܖܹ̈ܐ ܥܲܡ ܪܵܘܖ̈ܒܹܐ. ܗܵܐ ܫ̣ܟܸܒ݂ܘ ܥܲܠ ܣܲܒ݂ܪܵܟ݂: ܟܠܗܘܿܢ ܥܲܢܝܼܕܲܝ̈ܢ. ܕܒܲܩܝܵܡܬܵܟ݂ ܫܒ݂ܝܼܚܬܵܐ: ܒܫܘܼܒ݂ܚܵܐ ܬܩܝܼܡ ܐܸܢܘܿܢ. ܐܸܬ݂ܚܲܝܲܠܘ ܘܢܸܬ݂ܥܲܫܲܢ ܠܸܒܟ݂ܘܿܢ. ܐܸܬ݂ܠܲܒܲܒ݂ܘ ܐܲܚܲܝ̈: ܕܐܸܫܬ݁ܪܝܼ ܓܙܵܪ ܕܝܼܢܵܐ. ܒܐܵܕ݂ܵܡ ܕܲܬ݂ܪܹܝܢ: ܕܲܙܟ݂ܵܗ̇ ܠܲܚܛܝܼܬ݂ܵܐ܀
ܕܥܲܢܝܼ̈ܕܹܐ. ܫܲܒܲܚܘ ܙܲܕ݁ܝܼܩܹ̈ܐ ܠܡܵܪܝܵܐ. ܢܲܦ̮ܫܗܘܿܢ ܕܙܲܕ݁ܝܼܩܹ̈ܐ ܒܐܝܼܕ݂ܵܘ̈ܗܝ ܕܡܵܪܝܵܐ: ܕܪܲܚ̣ܡܘܼܗܝ ܘܗܲܝܡܸܢ̣ܘ ܒܹܗ ܘܲܢ̣ܛܲܪܘ ܦܘܼܩܕܵܢܵܘ̈ܗܝ. ܫܲܒܲܚܘ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ ܚܕܲܬܵܐ: ܕܲܫ̣ܦܲܪܬ݁ ܠܲܡܫܝܼܚܵܐ ܕܲܒ݂ܪ̣ܵܟ݂ ܡ̣ܢ ܥܵܠܡܵܐ: ܕܬܹܐܙܲܠ ܘܬܸܬ݂ܒܲܣܲܡ ܒܲܓ݂ܢܘܿܢ ܡܲܠܟ݁ܘܼܬ݂ܵܐ.
ܫܲܒܲܚ. ܐܲܒ݂ܝܼܠ ܡܵܘܬܵܐ ܘܲܟ݂ܡܝܼܪ ܣܵܛܵܢܵܐ. ܘܚܵ̇ܕ݂ܝܵܐ ܥܹܕ݁ܬܵܐ ܒܲܩܝܵܡܬܹܗ ܕܲܒ݂ܪܵܐ.
ܥܵܠܲܡ: ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܠܸܠܝܵܐ ܕܟ݂ܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܫܚܝܼܡܹ̈ܐ: ܠܒܲܪ ܡ̣ܢ ܨܵܘܡܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܐ:
ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܚܘܼܣ ܥܲܠ ܥܲܡܵܟ݂: ܘܠܵܐ ܬܸܬܸ݁ܠ ܝܵܪܬ݁ܘܼܬ݂ܵܟ݂ ܠܚܸܣܕܵܐ ܘܡܘܼܝܵܩܵܐ ܕܥܲܡܡܹ̈ܐ. ܚܘܼܪ ܡ̣ܢ ܡܸܕ݂ܝܵܖ̈ܲܝܟ ܘܲܚܙܝܼ ܩܛܝܼܪܵܐ: ܕܣܵܒ݂ܠ̇ܝܼܢܲܢ ܡ̣ܢ ܒܥܸܠܕ݁ܒ݂ܵܒܹ̈ܐ. ܘܬܸܕ݂ܢܲܚ ܠܲܢ ܒܲܥܓܲܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܕܠܵܐ ܢܹܐܡܪܘܼܢ ܥܲܡܡܹ̈ܐ ܐܲܝܟܵܘ ܐܲܠܵܗܗܘܿܢ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܢܲܛܲܪ ܠܥܲܒ݂ܕܲܝ̈ܟ: ܡ̣ܢ ܣܵܢܐܹ̈ܐ ܕܡܲܓܵܢ ܕܨܵܒܹܝ̇ܢ ܠܡܸܚܛܲܦ ܕܠܵܐ ܕܝܼܠܗܘܿܢ. ܣܘܼܓ݂ ܒܐܲܦܲܝ̈ܢ ܩܘܼܫܬܵܟ݂ ܚܲܣܝܼܢܵܐ: ܕܢܸܚܙܘܿܢ ܣܵܢܐܲܝ̈ܢ ܘܢܸܒ݂ܗ̱ܬ݂ܘܼܢ. ܘܐܲܟ݂ܪܸܟ݂ ܠܲܢ ܫܘܼܪܵܐ ܒܛܝܼܠܘܼܬ݂ܵܟ݂: ܕܢܸܣܬܲܬܲܪ ܒܓ݂ܵܘܹܗ ܘܢܵܘܕܸ݁ܐ ܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܒܲܨܠܘܿܬ݂ܵܗ̇ ܕܝܵܠܹܕ݂ܬ݂ܵܟ݂ ܡܵܪܲܢ ܝܼܫܘܿܥ: ܫܲܝܸܢ ܥܵܠܡܵܐ ܟܠܹܗ ܕܲܫܓ݂ܝܼܫ ܘܲܕ݂ܠܝܼܚ ܒܲܚܛܵܗܵܘ̈ܗܝ. ܘܒܲܛܸܠ ܡ̣ܢ ܐܲܪܥܵܐ ܩܖ̈ܵܒܹܐ ܘܩܹܐܖ̈ܣܹܐ: ܘܫܲܝܸܢ ܟܵܗܢܹ̈ܐ ܥܲܡ ܡܲܠܟܹ̈ܐ. ܕܒ݂ܵܐܘܝܘܼܬ݂ܵܐ ܘܚܘܼܒܵܐ ܢܸܙܕܲܝܲܚ: ܕܘܼܟ݂ܪܵܢ ܝܵܠܹܕ݂ܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ܀ ܙܲܕ݁ܝܼܩܹ̈ܐ ܕܲܫ̣ܦܲܪܘ ܠܵܟ݂ ܡܵܪܲܢ ܝܼܫܘܿܥ: ܢܒ݂ܝܼܹ̈ܐ ܫܠܝܼܚܹ̈ܐ ܘܣܵܗܕܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ ܕܲܒ݂ܟ݂ܠ ܐܲܬܲܪ. ܗܸܢ̣ܘܿܢ ܢܦܝܼܣܘܼܢܵܟ݂ ܚܠܵܦ ܢܲܦ̮̈ܫܵܬܲܢ: ܕܬܸܥܒܸ݁ܕ݂ ܖ̈ܲܚܡܹܐ ܥܲܠ ܟܠܲܢ. ܘܐܲܫܘܵܐ ܠܲܢ ܕܥܲܡܗܘܿܢ ܢܵܘܕܸ݁ܐ ܠܵܟ݂: ܒܝܵܘܡܵܐ ܕܒܹܗ ܢܵܣ̇ܒ݁ܝܼܢ ܐܲܓ݂ܪܵܐ ܕܥܲܡܠܲܝ̈ܗܘܿܢ܀ ܛܘܼܒܲܝܟ ܠܵܟ݂ ܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ: ܕܫܲܡ̣ܠܝܼܬ݁ ܪܲܗܛܵܟ݂ ܒܙܵܟ݂ܘܼܬ݂ܵܐ. ܘܲܦܠܲܚ̣ܬ݁ ܒܟܲܪܡܹܗ ܕܲܡܫܝܼܚܵܐ: ܘܲܢ̣ܣܲܒ݂ܬ݁ ܐܲܓ݂ܪܵܐ ܚܠܵܦ ܥܲܡܠܲܝ̈ܟ: ܕܹܝܢܵܪ ܛܘܼܒܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܨ̈ܠܵܘܵܬ݂ܵܟ݂ ܢܸܗܘ̈ܝܵܢ ܥܲܠ ܟܠܲܢ: ܘܢܸܫܬ݁ܘܸܐ ܕܥܲܡܵܟ݂ ܢܹܐܪܲܬ݂ ܡܲܠܟ݁ܘܼܬ݂ܵܐ܀

ܗܵܘ̇ ܕܗܘ̤ܝܘ̣ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܖ̈ܝܵܬ݂ܵܐ: ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀

ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܝܼܣ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܝܼܣ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀

ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܵܘ̇ ܕܲܒ݂ܖ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀

ܗܵܘ̇ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀

ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.

܀܀܀
