ܛܲܟܼܣܵܐ ܕܬܸܫܡܸܫܬܵܐ
ܕܠܸܠܝܵܐ ܘܕܨܲܦܪܵܐ
 ܒܫܸܡ ܐܲܒܼܵܐ ܘܲܒܼܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕܫܵܐ ܠܥܵܠܡܝܼܢ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒܿܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܖ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒܿܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟܼ: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟܼ: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬܼܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟܼ: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟܼ: ܠܡܲܣܵܩܘܼ ܠܵܟܼ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬܼܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓܼܕ݁ܬ݂ܵܐ ܒܟܼܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟܼܠ.
ܘܥܵܢܹܝ̇ܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.

ܨܠܘܿܬ݂ܵܐ. ܚܲܝܸܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܠܲܡܚܝܼܠܘܼܬܲܢ ܒܲܚܢܵܢܵܟ݂: ܘܠܲܒܸ݁ܒ݂ ܘܥܲܕܲܪ ܠܲܙܥܘܿܪܘܼܬ݂ ܢܲܦ̮ܫܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܥܝܼܪ ܕܲܡܟ݁ܘܼܬ݂ ܪܸܥܝܵܢܲܢ: ܘܐܲܩܸܠ ܝܘܼܩܪܵܐ ܕܗܲܕܵܡܲܝ̈ܢ: ܘܐܲܫܝܼܓ݂ ܘܲܡܪܘܿܩ ܨܵܐܬ݂ܵܐ ܕܚܵܘܒܲܝ̈ܢ ܘܕܲܚܛܵܗܲܝ̈ܢ: ܘܐܲܢܗܲܪ ܠܚܸܫܘܿܟ݂ܘܼܬܹܗ ܕܡܲܕ݁ܥܲܢ: ܘܐܵܘܫܸܛ ܘܫܲܪܲܪ ܘܗܲܒ݂ ܒܲܢ ܚܲܝܠܵܐ ܘܐܝܼܕ݂ܵܐ ܕܥܘܼܕ݂ܖ̈ܵܢܹܐ: ܕܒ݂ܵܗ̇ ܢܩܘܼܡ ܘܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܡܵܘܬ݁ܒ݂ܵܐ
	ܡܘܬ݁ܒ݂ܐ. ܒܥܲܕ݂ܠܵܐ ܬܲܝܬܹ݁ܝܢ. ܢܹܐ̈ܙܵܠ݇ܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܒ݂ܒܲܣܝܼܡܘܼܬ݂ܵܐ: ܒܪܲܡܫܵܐ ܬܒ݂ܘܼܬ݂ ܒܟ݂ܵܬ݂ܵܐ ܘܲܒ݂ܨܲܦܪܵܐ ܚܲܕ݂ܘܼܬ݂ܵܐ. ܢܸܫܹ̈ܐ ܕܐܸܬܲܝ̈ ܗ̣݇ܘܲܝ̈ ܕܢܸܚ̈ܙܝܵܢ ܩܲܒ݂ܪܵܐ. ܬܲܗܪܵܐ ܘܕ݂ܘܼܡܵܪܵܐ ܐܸܚܲܕ݂ ܗ݇ܘ̣ܵܐ ܐܸܢܹܝܢ. ܓܵܘܢܵܐ ܚܸܫܘܿܟ݂ܵܐ ܐܲܠ̣ܒܸ݁ܫ ܗ݇ܘ̣ܵܐ. ܗܸܢܹܝܢ ܕܹܝܢ ܒܵܟ̣̈ܝܵܢ ܗ̣݇ܘܲܝ̈. ܟܲܕ݂ ܠܵܐ ܝܵܕ݂̈ܥܵܢ ܡܵܢܵܐ ܗܘ̤ܵܐ. ܝܼܫܘܿܥ ܕܹܝܢ ܦܓܲܥ̣ ܒܗܹܝܢ ܘܲܒ݂ܓܲܢܵܢܵܐ ܬܘܼܒ݂ ܕܲܡ̣ܝܵܝ̈ܗܝ. ܡܵܪܝ ܐܸܢ ܐܲܢ݇ܬ݁ ܫ̣ܩܲܠܬܵܝܗܝ ܐܸܡܲܪܘ ܠܲܢ ܕܐܲܝܟܵܘ ܕܢܹܐܙܲܠ ܘܢܸܫܩܠܝܼܘܗܝ. ܐܸܠܵܐ ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ ܚܘܼܣܥܠܲܝܢ. ܬܢܝܼ܀ ܐܸܬ݁ܕܲܠܚܲܬ̤ ܘܙܵܥܲܬ̤ ܐܲܪܥܵܐ. ܟܲܕ݂ ܐܸܬܲܒܲܙܲܚܬ݁ ܒܲܙܩܝܼܦܵܐ. ܡܫܝܼܚܵܐ ܡܵܪܵܗ̇ ܕܬܸܫܒܘܿܚܬܵܐ. ܒܨܸܨܹ̈ܐ ܘܪܘܼܡܚܵܐ ܣܲܬ݂ܪܘܼܗܝ ܒܝܼܫܹ̈ܐ ܠܗܲܝܟܲܠ ܘܐܸܬ݂ܒܲܠܗܝܼ̈ ܕܲܚܙܲܝ̈ ܨܲܥܪܹܗ. ܫܸܡܫܵܐ ܦܐܸܐ ܒܙܲܠܝܼܩܵܘ̈ܗܝ ܐܲܚ̣ܫܸܟ݂ ܢܘܼܗܪܹܗ ܘܚܲܦ̣ܝܼ ܨܸܡܚܹܗ. ܕܲܚ̣ܙܵܐ ܠܡܵܪܹܗ ܕܲܬ݂ܠܸܐ ܒܩܲܝܣܵܐ ܘܚܵܐܹ̇ܫ ܡ̣ܢ ܡܲܖ̈ܵܚܹܐ. ܘܣܲܗܪܵܐ ܠܲܕ݂ܡܵܐ ܐܸܫܬܲܚܠܲܦ ܗ݇ܘ̣ܵܐ. ܠܡܲܟܣܵܢܘܼܬܹܗ ܕܥܲܡܵܐ ܣܲܟ݂ܠܵܐ ܕܐܲܡ̣ܪܲܚ ܘܲܣ̣ܥܵܐ ܘܲܩ̣ܛܲܠ ܡܵܪܹܗ. ܡܲܢ̣ܘܼ ܗܵܢܵܐ ܕܒܹܝܬ݂ ܩܘܼܕ݂ܫܵܐ ܒܩܸܛܠܹܗ ܐܸܣܬ݁ܕܸܩ. ܘܡܝܼ̈ܬܹܐ ܬܘܼܒ݂ ܒܩܵܠܹܗ ܡܸܬ݂ܢܲܚܡܝܼܢ. ܘܲܠܓܲܝܵܣܵܐ ܡܲܠܟܘܼܬ݂ܵܐ ܡܸܫܬܵܘܕܸܐ. ܟܒܲܪ ܗܘ̤ܝܘܼ ܡܫܝܼܚܵܐ ܡܲܚܝܵܢܵܐ ܕܡܝܼܬ݂ܘܼܬܲܢ. ܬܵܘ ܐܲܚܲܝ̈ ܢܸܪܗܲܛ ܠܣܸܓ݂ܕܬܹܗ. ܕܲܚܠܵܦ ܟܠܹܗ ܟܝܵܢܲܢ ܡܵܘܬܵܐ ܛܥܸܡ. ܘܢܸܩܥܸܐ ܠܹܗ ܟܠܲܢ ܕܠܵܐ ܫܲܠܘܵܐ ܡܵܪܹܟܠ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܡ̣ܚܵܐ ܠܲܒ݂ܥܸܠܕܒ݂ܵܒ݂ܵܘ̈ܗܝ ܠܒܸܣܬ݁ܪܹܗ. ܟܲܕ݂ ܣ̣ܠܸܩ ܗ݇ܘ̣ܵܐ ܦܵܪܘܿܩܲܢ ܒܬܸܫܒܘܿܚܬܵܐ ܡ̣ܢ ܓܵܘ ܫܝܘܿܠ. ܐܸܬ݂ܡܲܪܡܲܪ ܒܝܼܫܵܐ ܕܲܚ̣ܙܵܐ ܚܕܲܬܵܐ ܘܫܲܪܝܼ ܕܢܹܐܡܲܪ. ܕܐܸܢܵܐ ܒܝܲܕ݂ ܓܲܢ݇ܒܵܪܘܼܬ݂ܝ ܠܕ݂ܵܖܹ̈ܐ ܘܲܠܫܲܖ̈ܒ݂ܵܬ݂ܵܐ ܠܥܲܒ݂ܕܘܼܬ݂ܝ ܫܲܥ̇ܒܕܹܬ݂. ܘܐܸܢܵܐ ܒܝܲܕ݂ ܚܲܣܝܼܢܘܼܬ݂ܝ ܠܡܲܓ݂ܕܠܵܐ ܥܓ݂ܝܼܪ ܒܲܚܙܵܬܹܗ ܓܘܼܠܝܵܕ݂ ܫܲܦ̇ܠܹܬ݂. ܘܦܲܓ݂ܖܹ̈ܐ ܡܝܲܩܖܹ̈ܐ ܕܡܲܠܟܹ̈ܐ ܚܬ݂ܝܼܖܹ̈ܐ ܠܲܫܝܘܿܠ ܐܲܚ̇ܬܹܬ݂. ܘܲܠܓܲܢ݇ܒܵܖܹ̈ܐ ܘܲܠܚܲܣܝܼܢܹ̈ܐ ܐܸܢܵܐ ܡܲܟ݁ܟܹܬ݂. ܡܲܢ̣ܘܼ ܗܵܢܵܐ ܕܗܵܟܲܢܵܐ ܪܲܒ ܫܘܼܠܛܵܢܹܗ. ܕܗܵܐ ܒܵܥܹ̇ܐ ܕܢܸܣܲܒ݂ ܐܝܼܩܵܪܝ ܡܹܢܝ. ܘܗܵܐ ܦܵܐܲܚ̇ ܪܹܝܚܵܐ ܕܚܲܝܘܼܬܹܗ ܡ̣ܢ ܒܹܝܬ݂ ܡܝܼ̈ܬܹܐ ܟܠܗܘܿܢ. ܟܒܲܪ ܗܘ̤ܝܘܼ ܕܬ݂ܵܒܲܥ̇ ܚܘܼܫܒܵܢܵܐ ܕܥܲܢܝܼ̈ܕܹܐ ܀ ܬܵܘ ܐܲܚܲܝ̈ ܢܲܣܸܩ ܬܸܫܒܘܿܚܬܵܐ. ܠܕܲܒ݂ܡܵܘܬܹ݁ܗ ܓܲܢ݇ܒ݁ܪܲܢ ܥܲܠ ܡܵܘܬܵܐ. ܘܢܸܩܥܸܐ ܠܹܗ ܟܠܲܢ ܕܠܵܐ ܫܲܠܘܵܐ ܡܵܪܹܟ̇ܠ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂.
ܕܲܨܠܘܿܬ݂ܵܐ. ܫܒ݂ܝܼܚܘܼ ܬܘܼܟ݂ܠܵܢܵܟ݂. ܫܘܼܚܠܵܦܹܗ ܕܩܵܠܵܐ: ܝ.
ܐܲܠܵܗܲܢ ܒܹܝܼܬ݂ ܓܵܘܣܲܢ ܥܲܫܝܼܢܵܐ. ܫܒ݂ܝܼܚܘܼ ܬܘܼܟ̣ܠܵܢܵܟ݂ ܒܹܝܬ݂ ܓܵܘܣܲܢ ܥܲܫܝܼܢܵܐ: ܘܲܪܓ݂ܝܼܓ݂ ܘܲܡܝܲܬܲܪ ܚܢܵܢܵܟ݂ ܪܲܒܵܐ ܡܠܸܐ ܥܘܼܕ݂ܖ̈ܵܢܹܐ. ܕܲܡܨܸܐ ܦܵܪܹܩ ܠܲܕ݂ܡܸܬ݂ܓܵܘܣܝܼܢ ܒܹܗ ܡ݂ܢ ܟܠ ܒܝܼ̈ܫܵܢ: ܘܲܡܫܵܘܙܸܒ݂ ܬܘܼܒ݂ ܠܲܕ݂ܛܵܦܣܝܼܢ ܒܹܗ ܡ̣ܢ ܟܠ ܥܵܩ̈ܵܢ. ܘܐܲܝܢܵܐ ܕܐܲܡܝܼܢ ܘܢܵܩܹܫ ܒܬܲܪܥܹܗ ܠܵܐ ܡܲܗܡܸܐ ܡܸܢܹܗ: ܘܠܲܕ݂ܡܲܚܨܸܦ ܬܘܼܒ݂ ܘܫܵܐܹܠ ܖ̈ܲܚܡܹܐ ܡܲܣܓܸܐ ܥܵܢܹܐ ܠܹܗ. ܡܲܢ̣ܘܼ ܗܵܢܵܐ ܕܠܵܐ ܢܸܪܗܲܛ ܘܢܸܬ݂ܓܵܘܲܣ ܒܹܗ: ܒܲܚܢܵܢܵܟ݂ ܡܠܸܐ ܟܠ ܥܘܼܕ݂ܖ̈ܵܢܝܼܢ. ܘܢܸܫܐܲܠ ܡܸܢܹܗ ܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ: ܟܲܕ݂ ܩܵܥܹܐ ܒܲܚܢܲܓ݂ܬ݂ܵܐ ܘܐܵܡܲܪ. ܚܘܼܣ ܥܠܲܝ ܐܲܢ݇ܬ݁ ܚܲܢܵܢܵܐ: ܘܲܦܪܘܿܩܲܝܢܝ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁. ܛܵܒ݂ܵܐ ܕܠܵܐ ܟܵܠܹܐ ܖ̈ܲܚܡܵܘܗܝ ܡܵܪܹܐ ܟܠ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܐܸܢܵܐ ܕܹܝܢ ܥܠܲܝܟ ܡܵܪܝܵܐ ܐܸܬ݁ܬܲܟ݂ܠܹܬ݂. ܐܲܢ݇ܬ݁ܘܼ ܬܘܼܟ̣ܠܵܢܲܢ ܘܒܹܝܬ݂ ܓܵܘܣܲܢ ܦܵܪܘܿܩܲܢ: ܘܠܵܟ̣ܘܼ ܡܩܲܪܒ݂ܝܼܢܲܢ ܫܸܐ̈ܠܵܬ݂ܵܐ ܠܲܣܢܝܼܩܘܼܬܲܢ. ܦܬܲܚ ܠܲܦܝܵܣܲܢ ܬܪܲܥ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܡܠܸܐ ܥܘܼܕ݂ܖ̈ܵܢܹܐ: ܘܦܲܢܵܐ ܒܖ̈ܲܚܡܲܝܟ ܫܸܐܠܲܬ݂ ܢܲܦ̮ܫܲܢ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁. ܘܚܲܣܵܐ ܚܵܘܒܹ̈ܐ ܘܣܲܟ̣̈ܠ̱ܘܵܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܡܵܪܝܵܐ: ܘܗܲܒ݂ ܥܘܼܕ݂ܪܵܢܵܐ ܠܲܡܚܝܼܠܘܼܬܲܢ ܕܢܸܓ݂ܡܘܿܪ ܨܸܒ݂ܝܵܢܵܟ݂. ܨܠܝܼܒ݂ܵܟ݂ ܡܵܪܲܢ ܫܘܼܪܵܐ ܪܵܡܵܐ ܢܸܗܘܸܐ ܠܲܢ: ܒܟ̣ܠ ܥܸܕܵܢ̈ܝܼܢ ܡ̣ܢ ܟܠ ܢܸܟ̣ܝܵܢ̈ܝܼܢ. ܘܒܹܗ ܢܸܣܬܲܬܲܪ ܡ̣ܢ ܒܝܼܫܵܐ ܒܥܸܠܕܲܪܲܢ: ܘܠܵܟ݂ ܢܵܘܕܸܐ ܘܲܢܫܲܒܲܚ ܘܢܹܐܡܲܪ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܪܵܚܹܡ ܐ݇ܢܵܫܵܐ: ܕܠܵܐ ܒܵܗܬ݁ܝܼܢ ܕܲܡܣܲܒܪܝܼܢ ܒܵܟ݂. ܡܵܪܵܐ ܕܠܵܐ ܡܲܗܡܸܐ ܡ̣ܢ ܟܠ ܕܩܵܪܹܝܢ ܠܹܗ ܡܵܪܹܐ ܟܠ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܕܡܲܪܝܲܡ܀ ܢܫܲܒܲܚ ܘܢܸܚܕܸܐ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬܲܢ. ܫܘܼܒ݂ܚܵܐ ܠܚܲܝܠܵܐ ܬܠܝܼܬ݂ܵܝܵܐ: ܕܲܢܚܸܬ݂ ܘܲܫܪܵܐ ܥܲܠ ܒܬ݂ܘܼܠܬܵܐ. ܘܲܓܒܲܠ ܡܸܢܵܗ̇ ܒܲܪܢܵܫܵܐ: ܘܥܲܒ݂ܕܹܗ ܗܲܝܟ̇ܠܵܐ ܠܲܥܡܘܼܪܝܹܗ. ܘܒܹܗ ܐܸܫܬܵܘܕܝܼ ܠܡܝܼܬ݂ܘܼܬܲܢ: ܚܲܝܹ̈ܐ ܚܲܕ݂ܬܹ̈ܐ ܘܫܲܪܝܼܖܹ̈ܐ. ܘܗܘ̤ܝܘܼ ܢܘܼܗܪܵܐ ܫܲܪܝܼܪܵܐ: ܠܡܲܠܲܐܟܹ̈ܐ ܘܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ. ܡܫܝܼܚܵܐ ܕܲܕ݂ܢܲܚ ܡ̣ܢ ܡܲܪܝܲܡ: ܠܦܘܼܪܩܵܢ ܟܠܗܹܝܢ ܒܸܖ̈ܝܵܬ݂ܵܐ. ܐܲܫܪܵܐ ܫܲܝܢܵܟ݂ ܒܲܒ݂ܪܝܼܬ݂ܵܐ: ܘܛܲܪ ܥܵܡܘܿܖܹ̈ܝܗ̇ ܒܲܚܢܵܢܵܟ݂. ܘܫܲܝܸܢ ܟܵܗܢܹ̈ܐ ܥܲܡ ܡܲܠܟܹ̈ܐ: ܕܥܘܼܡܪܵܐ ܫܲܠܝܵܐ ܘܢܝܼܚܵܐ ܢܸܥܡܲܪ. ܒܲܨܠܘܿܬ݂ ܡܲܪܝܲܡ ܕܝܼܠܸܕ݂ܬ݂ܵܟ݂: ܛܲܪ ܥܹܕ݁ܬܵܟ݂ ܡ̣ܢ ܢܸܟ݂ܝܵܢܹ̈ܐ ܘܚܘܼܣ ܥܠܹܝܗ̇. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܛܲܪ ܖ̈ܲܚܡܲܝܟ ܠܩܲܖ̈ܝܼܒܲܝܟ. ܡܫܝܼܚܵܐ ܡܵܪܵܗ̇ ܕܬܸܫܒܘܿܚܬܵܐ: ܕܐܵܘܪܸܒ݂ ܒܚܘܼܒܹܗ ܠܩܲܕܝܼܫܵܘ̈ܗܝ. ܒܲܨܠܘܿܬ݂ ܢܒ݂ܝܼܹ̈ܐ ܚܘܼܣ ܥܠܲܝܢ: ܘܒܲܨܠܘܿܬ݂ ܫܠܝܼܚܹ̈ܐ ܐܸܬ݂ܪܲܥܵܐ ܠܲܢ. ܘܒܲܨܠܘܿܬ݂ ܣܵܗܕܹ̈ܐ ܚܲܝܸܠ ܠܲܢ: ܕܢܸܙܟܸܐ ܠܚܲܫܹ̈ܐ ܕܲܚܛܝܼܬ݂ܵܐ. ܘܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܘܥܵܢܘܵܝܹ̈ܐ: ܕܚܘܿܩ ܡܸܢܲܢ ܚܲܝܠܹܗ ܕܒ݂ܝܼܫܵܐ. ܘܒܲܨܠܘܿܬ݂ ܟܵܗܢܹ̈ܐ ܒܲܪܸܟ݂ ܡܵܪܝ: ܟܠܝܼܠܵܐ ܕܫܲܢ݇ܬܵܐ ܒܛܲܝܒܘܼܬ݂ܵܟ݂. ܘܥܲܡ ܡܲܠܦܵܢܹ̈ܐ ܢܸܗܪܘܿܓ݂ ܒܣܸܦܖܹ̈ܐ ܕܝܘܼܠܦܵܢܵܟ݂: ܘܢܸܬ݂ܩܲܒܠܘܼܢ ܡܵܪܝ ܩܘܼܖ̈ܒܵܢܲܝܢ. ܩܕ݂ܵܡ ܒܹܝܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܢܸܗܘܸܐ ܫܪܵܝܵܐ ܠܥܵܩ̈ܵܬܲܢ. ܘܚܘܼܣܵܝܵܐ ܠܣܲܟ݂̈ܠ̱ܘܵܬܲܢ: ܘܥܲܡ ܟܹܐܢܹ̈ܐ ܢܸܚܙܸܐ ܐܲܦܲܝ̈ܟ. ܒܝܵܘܡܵܐ ܕܕ݂ܵܢܚܵܐ ܪܲܒܘܼܬ݂ܵܟ݂: ܘܢܸܙܡܲܪ ܫܘܼܒ݂ܚܵܐ ܠܡܵܪܘܼܬ݂ܵܟ݂ ܒܟ̣ܠ ܥܸܕܵܢ̈ܝܼܢ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܕܢܲܚ ܢܘܼܗܪܵܐ ܒܚܸܫܘܿܟ݂ܵܐ ܠܲܬ݂ܖ̈ܝܼܨܹܐ. ܢܘܼܗܪܵܐ ܘܚܲܝܹ̈ܐ ܘܲܫܪܵܪܵܐ: ܕܡܲܢܗܲܪ ܠܣܵܘܦܲܝ̈ ܒܸܖ̈ܝܵܬ݂ܵܐ. ܨܠܝܼܒ݂ܵܐ ܡܲܒܘܼܥ ܛܵܒ݂̈ܵܬܲܢ: ܙܲܝܢܵܐ ܕܙܲܟ̇ܝܹܗ ܠܲܟ̣ܝܵܢܲܢ. ܟܐܝܼ ܒܠܸܓ݂ܝܘܿܢܵܘ̈ܗܝ ܕܲܢܟ̣ܝܼܠܵܐ: ܕܚܵܪܹܒ݂ ܣܸܕ݂ܖܹ̈ܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܘܢܲܦܸܣ ܪܸܡܙܵܟ݂ ܠܖ̈ܘܼܚܵܢܹܐ: ܕܲܢܓܲܒܘܿܢ ܡܵܪܝ ܙܝܼܙܵܢܹ̈ܐ. ܘܐܲܪܝܼܡ ܩܲܪܢܵܗ̇ ܕܲܪܚܘܼܡܬܵܟ݂: ܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ ܡܗܲܝܡܲܢܬܵܐ. ܕܲܨܕܲܝ̈ ܕܵܖܹ̈ܝܗ̇ ܡ̣ܢ ܫܘܼܒ݂ܚܵܟ݂: ܒܝܲܕ݂ ܪܲܦܝܘܼܬ݂ܵܐ ܕܥܲܠܠܵܢܹܝ̈ܗ̇. ܘܝܲܠܕܹܝ̈ܗ̇ ܦܵܗܹ̇ܝܢ: ܒܹܝܬ݂ ܩܸܩ̈ܠܵܬ݂ܵܐ ܘܡܲܪܝܼܪ ܠܵܗ̇. ܚܘܼܒܵܟ݂ ܥܲܒ݂ܕܵܟ݂ ܒܲܪܢܵܫܵܐ: ܟܲܕ݂ ܐܲܠܵܗܵܐ ܐܲܢ݇ܬ݁ ܒܲܟ̣ܝܵܢܵܟ݂. ܒܚܘܼܒܵܟ݂ ܫܲܪܝܼܬ݁ ܒܬ݂ܘܼܩܵܢܲܢ: ܒܚܘܼܒܵܟ݂ ܫܲܠܸܡ ܥܲܡ ܓܸܢܣܲܢ. ܘܐܲܫܪܵܐ ܫܲܝܢܵܟ݂ ܒܲܒ݂ܪܝܼܬ݂ܵܐ: ܫܠܵܡܵܐ ܕܐܲܝܬ݁ܝܼܘ ܖ̈ܘܼܚܵܢܹܐ. ܐܘܿ ܨܠܝܼܒ݂ܵܐ ܡܵܘܬܹ݁ܗ ܕܡܵܘܬܵܐ ܚܘܼܣ ܥܠܲܝܢ. ܕܐܲܒ݂ܘܼܢ܀ ܣܹܐܡܵܐ ܓܲܒ݂ܝܵܐ ܕܲܒ݂ܚܝܼܪ ܒܐܲܪܥܵܐ. ܣܝܼܡܬ݂ܵܐ ܕܓܲܙܵܐ ܫܡܲܝܵܢܵܐ: ܐܝܼܬ݂ܵܘܗܝ ܦܲܓ݂ܪܵܟ݂ ܐܘܿ ܐܲܒ݂ܘܼܢ. ܡܵܪܝ ܦܠܢ ܢܲܨܝܼܚܵܐ: ܡܵܐܢܵܐ ܓܲܒ݂ܝܵܐ ܘܲܡܝܲܬ݁ܪܵܐ. ܕܲܗ̤ܘܲܝܬ݁ ܚܲܒ݂ܪܵܐ ܘܒܲܪ ܣܝܼܥܬܵܐ: ܠܐܸܣܛܲܦܵܢܘܿܣ ܩܲܕ݂ܡܵܝܵܐ. ܘܒܹܝܬ݂ ܡܵܩܵܪܝܼܣ ܘܐܲܪܣܵܢܝܼܣ ܘܦܵܘܠܘܿܣ ܪܹܫܵܐ ܕܐܲܒ݂ܝܼ̈ܠܹܐ. ܛܘܼܒܲܝܟ ܕܲܢܛܝܼܪܘܼ ܛܘܼܒ݂ܵܟ݂: ܥܲܡ ܗ̇ܝ ܣܝܼܥܬܵܐ ܕܲܒ݂ܢܲܝ̈ ܐܲܦܸܣܘܿܣ. ܕܲܦܠܲܚܬ݁ ܒܟܲܪܡܵܐ ܪܘܼܚܵܢܵܐ: ܘܲܢܣܲܒ݂ܬ݁ ܐܲܓ݂ܪܵܐ ܡܫܲܠܡܵܢܵܐ. ܨܠܘܿܬ݂ܵܟ݂ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܹܗ: ܠܟܸܢܫܵܐ ܕܙܲܝܲܚ ܕܘܼܟ݂ܪܵܢܵܟ݂. ܣܥܘܿܪ ܡܲܖ̈ܥܹܐ ܘܐܲܚܠܸܡܟܖ̈ܝܼܗܹܐ ܒܲܨ̈ܠܵܘܵܬ݂ܵܟ݂. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܡܸܛܠ ܕܩܲܠܝܼܠܘܼ ܙܲܒ݂ܢܵܐ ܛܵܒ݂ ܘܲܙܥܘܿܪ. ܩܲܠܝܼܠ ܘܲܙܥܘܿܪ ܙܲܒ݂ܢܵܐ. ܕܪܵܕܹܝܢܲܢ ܒܥܵܠܡܵܐ ܗܵܢܵܐ. ܚܲܛܵܝܹ̈ܐ ܬܘܼܒ݂ܘܼܢ ܘܲܚܝܵܘ. ܕܡܵܘܬܵܐ ܫܘܼܚܕܵܐ ܠܵܐ ܢܵܣܹܒ݂. ܘܲܠܥܲܬ݁ܝܼܖܹ̈ܐ ܘܡܸܣܟܹܢܹ̈ܐ ܐܲܟܲܚ݇ܕ݂ ܕܵܒܲܪ ܫܲܘܝܵܐܝܼܬ݂. ܫܲܒܲܚܘ ܠܹܗ ܬܸܫܒܘܿܚܬܵܐ ܚܕܲܬܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܕܓܲܒ݂ܠܲܢ. ܘܠܲܒ݂ܪܵܐ ܕܦܲܪܩܲܢ ܒܲܨܠܝܼܒܹܗ. ܘܲܠܪܘܼܚܵܐ ܦܵܪܲܩܠܹܝܛܵܐ. ܐ݇ܪܵܙܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܐ. ܕܗܘ̤ܝܘܼ ܡܢܲܚܸܡ ܥܲܢܝܼܕܲܝ̈ܢ. ܘܡܲܠܒܸ݁ܫ ܫܘܼܒ݂ܚܵܐ ܦܲܓ݂ܖ̈ܲܝܗܘܿܢ܀
ܫܲܒܲܚ. ܒܩܵܠܵܐ ܪܲܒܵܐ. ܢܸܫܹ̈ܐ ܕܐܸܬܲܝ̈ ܠܒܹܝܬ݂ ܩܒ݂ܘܼܪܵܟ݂ ܡܵܪܝܵܐ. ܚܠܵܦ ܟܲܪܝܘܼܬ݂ܵܐ ܚܲܕ݂ܘܼܬ݂ܵܐ ܐܸܬ݂ܡ̣ܠܝܼ̈. ܘܠܲܫܠܝܼܚܹ̈ܐ ܖ̈ܗܸܛ̣ ܗ̣݇ܘܲܝ̈ ܕܲܢܣܲܒܖ̈ܵܢ. ܡܸܠܹ̈ܐ ܕܡ̣ܢ ܡܲܠܲܐܟ̣ܵܐ ܫ̣ܡܲܥ̈ ܗ̣݇ܘܲܝ̈. ܕܐܸܡܲܪ ܠܗܹܝܢ ܕܐܲܢ݇ܬܹ݁ܝܢ ܠܵܐ ܬܸܕ݂ܚ̈ܠܵܢ. ܠܡ݁ܢ ܒܵܟ̣ܝܵܬܹ݁ܝ̈ܢ. ܠܝܼܫܘܿܥ ܒܵܥܝܵܬܹ݁ܝ̈ܢ. ܩ̣ܵܡ ܠܹܗ ܘܲܙ̣ܟ̣ܵܝܗܝ ܠܡܵܘܬܵܐ ܒܡܵܘܬܹ݁ܗ. ܚܢܲܢ ܕܹܝܢ ܡܵܪܝ ܒܬ݂ܵܘܕܝܼܬ݂ܵܐ ܢܸܩܥܸܐ ܘܢܹܐܡܲܪ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂.
ܥܵܠܲܡ. ܡ̣ܢ ܒܵܬܲܪ ܩܝܵܡܬܵܟ݂ ܡܫܲܒܲܚܬܵܐ. ܥܲܡܵܐ ܕܲܝܗܘܼܕ݂ܝܹ̈ܐ ܠܵܐ ܗܲܝܡ̣ܢܘ ܒܵܟ݂. ܟܲܕ݂ ܩܵܡ̣ܬ݁ ܗ̣݇ܘܲܝܬ݁ ܡ̣ܢ ܒܹܝܬ݂ ܡܝܼ̈ܬܹܐ. ܐܸܠܵܐ ܡܸܛܠ ܟܸܣܦܵܐ ܕܲܢ̣ܣܲܒ݂ܘ ܢܵܛܘܿܖܹ̈ܐ. ܐܲܟ̣ܪܸܙ̣ܘ ܘܐܸܡܲܪܘ ܕܲܓܵܠܵܐܝܼܬ݂. ܕܬܲܠܡܝܼܕ݂ܵܘ̈ܗܝ ܓܲܢ̣ܒܿܘܼܗܝ ܟܲܕ݂ ܕܲܡܟܿܝܼܢܲܢ. ܐܘܿ ܠܹܗ ܠܥܲܡܵܐ ܕܠܵܐ ܗܲܝܡܸܢܘ ܠܲܫܪܵܪܵܐ. ܐܸܢܗܘ̤ ܕܦܲܓ݂ܪܵܐ ܬܲܠܡܝܼܕܹ̈ܐ ܓܢܲܒ݂ܘ. ܢܲܚ̈ܬܹܐ ܘܟܸܬܵܢܹ̈ܐ ܠܡܵܢ ܒܒܹܝܬ݂ ܩܲܒ݂ܪܵܐ ܫ̣ܒܲܩܘ. ܘܲܠܡܵܢ ܟܲܕ݂ ܣܲܒ݁ܪܘܼܗܝ ܐܲܚܵܘ̈ܗܝ ܠܲܬ݂ܐܘܿܡܵܐ ܕܩ̣ܵܡ ܠܹܗ ܡܵܪܲܢ ܐܲܚ̣ܪܸܡ ܗ݇ܘ̣ܵܐ ܘܐܸܡܲܪ. ܕܐܸܢ ܠܵܐ ܚܵܙܹ̇ܐ ܐ݇ܢܵܐ ܒܐܝܼܕ݂ܵܘ̈ܗܝ ܕܘܼܟ̈ܝܵܬ݂ܵܐ ܕܨܸܨܹ̈ܐ ܗܵܢܹ̇ܝܢ ܕܲܒ݂ܠܲܥ̣ ܡ̣ܢ ܒܝܼܫܹ̈ܐ. ܠܵܐ ܡܗܲܝܡܸܢ ܐ݇ܢܵܐ ܕܩ̣ܵܡ ܦܵܪܘܿܩܵܐ ܕܒܸܖ̈ܝܵܬ݂ܵܐ. ܕܲܩܕ݂ܵܡܲܝ ܠܲܒ݂ܟ݁ܘܼܗܝ ܘܐܵܘܒ݁ܠܘܼܗܝ ܠܘܵܬ݂ ܦܝܼܠܵܛܘܿܣ. ܘܲܩܕ݂ܵܡܲܝ ܙܲܩ̣ܦܘܼܗܝ ܥܲܠ ܪܹܫ ܩܲܝܣܵܐ ܒܓ݂ܵܓ݂ܘܼܠܬܵܐ. ܐܵܦ ܡܲܪܝܲܡ ܙܸܒ݂ܢܲܬ̤ ܗܹܖ̈ܘܿܡܹܐ. ܘܐܸܬܲܬ̤ ܠܬܲܪܥܵܐ ܕܒܹܝܬ݂ ܩܒ݂ܘܼܪܵܐ. ܘܟܲܕ݂ ܝܼܕܲܥ̣ ܗ݇ܘ̣ܵܐ ܦܵܪܘܿܩܲܢ. ܕܲܫܠܝܼܚܹ̈ܐ ܐܸܬ݂ܦܲܠܲܓ݂ܘ ܗ݇ܘ̣ܵܘ. ܒܲܫܪܵܪܹܗ ܠܗܘܿܢ ܫܲܪܲܪ. ܢܲܓܝܼܪ ܪܘܼܚܵܐ ܪܲܚܸܡܥܠܲܝܢ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܟ݂ܠܗܘܿܢ ܥܹܐܕܹ̈ܐ ܡܵܖ̈ܵܢܵܝܹܐ. ܠܲܡܕܲܒ݁ܪܵܢܘܼܬ݂ܵܟ݂ ܡܵܪܝ ܬܡܝܼܗܬܵܐ ܘܠܵܐ ܡܸܬ݂ܡܲܠܠܵܢܝܼܬ݂ܵܐ: ܕܲܒ݂ܖ̈ܲܚܡܹܐ ܘܒܲܚܢܵܢܵܐ ܐܸܬ݂ܓܲܡܪܲܬ̤ ܘܐܸܣܬܲܥܪܲܬ̤ ܘܐܸܫܬܲܡܠ̱ܝܲܬ̤ ܠܚܘܼܕܵܬܹܗ ܘܲܠܦܘܼܪܩܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ ܚܲܠܵܫܵܐ: ܒܪܹܫܝܼܬ݂ܵܐ ܕܡ̣ܢܢ: ܡܲܣܩܝܼܢܲܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܩܵܢܘܿܢܵܐ.(ܡܲܙ.ܩܡܚ.)
 ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡ̣ܢ ܫܡܲܝܵܐ* ܫܲܒ̇ܚܘܼܗܝ ܒܲܡܖ̈ܵܘܡܹܐ܀ ܡ݂ܢ ܫܡܲܝܵܐ ܢ̣ܚܸܬ݂ܘ ܥܝܼܖܹ̈ܐ ܒܐܝܼܩܵܪܵܐ ܘܫܲܡܸܫ̣ܘ ܒܕܸܚܠ̱ܬ݂ܵܐ. ܠܗܲܝܟܲܠ ܦܲܓ݂ܪܵܐ ܕܲܒ݂ܪܵܐ ܡܫܝܼܚܵܐ ܕܚܲܫ̣ ܥܲܠ ܐܲܦܲܝ̈ܢ. ܘܲܚ̣ܙܵܘ ܫܘܼܒ݂ܚܵܐ ܕܲܩܝܵܡܬܹ݁ܗ ܒܬܸܕ݂ܡܘܼܪܬܵܐ ܘܣܲܒ݁ܪܘܼܗܝ ܠܐܲܖ̈ܥܵܢܹܐ܀ ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡ̣ܢ ܫܡܲܝܵܐ* ܫܲܒ̇ܚܘܼܗܝ ܒܲܡܖ̈ܵܘܡܹܐ܀ ܫܲܒ̇ܚܘܼܗܝ ܟܠܗܘܿܢ ܡܲܠܲܐܟ̣ܵܘ̈ܗܝ* ܫܲܒ̇ܚܘܼܗܝ ܟܠܗܘܿܢ ܚܲܝ̈ܠܵܘܵܬܹܗ܀ ܫܲܒ̇ܚܘܼܗܝ ܫܸܡܫܵܐ ܘܣܲܗܪܵܐ* ܫܲܒ̇ܚܘܼܗܝ ܟܠ ܟܵܘܟ̇ܒܹ̈ܐ ܘܢܘܼܗܪܵܐ܀ ܫܲܒ̇ܚܘܼܗܝ ܫܡܲܝ̈ ܫܡܲܝܵܐ* ܡܲܝ̈ܵܐ ܕܲܠܥܸܠ ܡ̣ܢ ܫܡܲܝ̈ܵܐ ܢܫܲܒ̇ܚܘܼܢ ܠܲܫܡܹܗ ܕܡܵܪܝܵܐ܀ ܡܸܛܠ ܕܗܘ̤ ܐܸܡܲܪ ܘܲܗܘ̤ܵܘ ܘܗܘ̤ ܦܩܲܕ݂ ܘܐܸܬ̣ܒ̇ܪܝܼܘ* ܘܐܲܩܝܼܡ ܐܸܢܘܿܢ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ܀ ܢܵܡܘܿܣܵܐ ܝܲܗ̣݇ܒ̣ ܘܠܵܐ ܥܵܒܲܪ* ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡ̣ܢ ܐܲܪܥܵܐ܀ ܬܲܢܝܼܢܹ̈ܐ ܘܟ̣ܠܗܘܿܢ ܬܗܘܿܡܹ̈ܐ* ܢܘܼܪܵܐ ܘܒܲܪܕ݂ܵܐ ܘܬܲܠܓܵܐ ܘܲܓ̣ܠܝܼܕ݂ܵܐ܀ ܖ̈ܘܼܚܹܐ ܘܥܲܠܥܵܠܹ̈ܐ ܥܒ݂ܵܕ݂ܵܐ ܕܡܸܠܬܹܗ* ܛܘܼܖܹ̈ܐ ܘܟ̣ܠܗܹܝܢ ܖ̈ܵܡܵܬ̣ܵܐ܀ ܐܝܼܠܵܢܹ̈ܐ ܕܦܹܐܖܹ̈ܐ ܘܟ̣ܠܗܘܿܢ ܐܲܖ̈ܙܹܐ* ܚܲܝܘ̱ܬ̣ܵܐ ܘܟ̣ܠܵܗ̇ ܒܥܝܼܪܵܐ܀ ܖ̈ܲܚܫܵܐ ܘܛܲܝܪܵܐ ܘܦܵܪܲܚܬ̣ܵܐ* ܡܲܠܟܹ̈ܐ ܕܐܲܪܥܵܐ ܘܟ̣ܠܗܹܝܢ ܐܸܡ̈ܘܵܬ̣ܵܐ܀ ܪܵܘ̱ܖ̈ܒ̣ܵܢܹܐ ܘܟ̣ܠܗܘܿܢ ܕܲܝܵܢܹ̈ܐ ܕܐܲܪܥܵܐ* ܓܲܕ̇ܘܼܕܹ̈ܐ ܘܲܒ̣ܬ̣ܘܼ̈ܠܵܬ̣ܵܐ܀ ܣܵܒܹ̈ܐ ܘܲܥܠܲܝܡܹ̈ܐ ܢܫܲܒ̇ܚܘܼܢ ܠܲܫܡܹܗ ܕܡܵܪܝܵܐ* ܡܸܛܠ ܕܪܲܒ̇ܘܼ ܫܡܹܗ ܒܲܠܚܘܿܕ݂ܵܘܗܝ܀ ܘܫܘܼܒ̣ܚܹܗ ܒܐܲܪܥܵܐ ܘܒܲܫܡܲܝܵܐ* ܡܪܝܼܡ ܩܲܪܢܵܐ ܕܥܲܡܹܗ ܘܬܸܫܒ̇ܘܿܚܬܵܐ ܕܟ̣ܠܗܘܿܢ ܙܲܕ̇ܝܼ̈ܩܵܘܗܝ܀ ܠܲܒ̣ܢܲܝ̈ ܝܑܼܣܪܵܝܹܠ ܥܲܡܵܐ ܕܩܲܪܝܼܒ̣ ܠܹܗ܀ ܡ݂ܢ ܫܡܲܝܵܐ ܢ̣ܚܸܬ݂ܘ ܥܝܼܖܹ̈ܐ ܒܐܝܼܩܵܪܵܐ ܘܫܲܡܸܫ̣ܘ ܒܕܸܚܠ̱ܬ݂ܵܐ. ܠܗܲܝܟܲܠ ܦܲܓ݂ܪܵܐ ܕܲܒ݂ܪܵܐ ܡܫܝܼܚܵܐ ܕܚܲܫ̣ ܥܲܠ ܐܲܦܲܝ̈ܢ. ܘܲܚ̣ܙܵܘ ܫܘܼܒ݂ܚܵܐ ܕܲܩܝܵܡܬܹ݁ܗ ܒܬܸܕ݂ܡܘܼܪܬܵܐ ܘܣܲܒ݁ܪܘܼܗܝ ܠܐܲܖ̈ܥܵܢܹܐ܀
ܫܲܒܲܚ܀ ܥܵܠܲܡ܀ ܡ݂ܢ ܒܵܬܲܪ ܩܒ݂ܘܼܪܬܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܐܸܬ݂ܟܲܢܲܫܘ ܥܲܡܵܐ ܕܲܝܗܘܼܕ݂ܵܝܹ̈ܐ. ܘܲܠܢܵܛܘܿܖܹ̈ܐ ܟܸܣܦܵܐ ܝܲܗ̣݇ܒ݂ܘ ܠܗܘܿܢ ܕܢܸܛܪܘܼܢ ܩܲܒ݂ܪܵܟ݂. ܘܡܸܢܫܹܠܝ ܐܲܒ݂ܗܸܬ݁ܬ݁ ܠܨܵܠܘܿܒܲܝ̈ܟ ܘܚܲܕܝܼܬ݁ ܠܥܹ̈ܕ݂ܵܬ݂ܵܟ݂܀
ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡ̣ܢ ܫܡܲܝܵܐ* ܫܲܒ̇ܚܘܼܗܝ ܒܲܡܖ̈ܵܘܡܹܐ. ܡ݂ܢ ܒܵܬܲܪ ܩܒ݂ܘܼܪܬܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܐܸܬ݂ܟܲܢܲܫܘ ܥܲܡܵܐ ܕܲܝܗܘܼܕ݂ܵܝܹ̈ܐ. ܘܲܠܢܵܛܘܿܖܹ̈ܐ ܟܸܣܦܵܐ ܝܲܗ̣݇ܒ݂ܘ ܠܗܘܿܢ ܕܢܸܛܪܘܼܢ ܩܲܒ݂ܪܵܟ݂. ܘܡܸܢܫܹܠܝ ܐܲܒ݂ܗܸܬ݁ܬ݁ ܠܨܵܠܘܿܒܲܝ̈ܟ ܘܚܲܕܝܼܬ݁ ܠܥܹ̈ܕ݂ܵܬ݂ܵܟ݂܀
ܬܸܫܒܘܿܚܬܵܐ (ܒܟ݂ܠܹܗ ܫܵܒ݂ܘܿܥܵܐ ܕܲܩܝܵܡܬܵܐ). ܒܲܩܝܵܡܬܹ݁ܗ ܕܲܒ݂ܪܵܐ ܖ̈ܵܘܙܵܢ ܒܸܖ̈ܝܵܬ݂ܵܐ. ܕܲܗܘ̤ܵܬ݂ ܬܲܪܥܘܼܬ݂ܵܐ ܘܐܲܡ̣ܠܸܟ݂ ܢܘܼܚܵܡܵܐ܀ ܐܲܒ݂ܝܼܠ ܡܵܘܬܵܐ ܘܲܟ݂ܡܝܼܪ ܣܵܛܵܢܵܐ. ܘܚܵܕ݂ܝܵܐ ܥܹܕ݁ܬܵܐ ܒܲܩܝܵܡܬܹܗ ܕܲܒ݂ܪܵܐ܀ ܐܸܒ݂ܠܵܐ ܠܚܲܢܦܹ̈ܐ ܘܒܸܗܬ݁ܬ݂ܵܐ ܠܟ݂ܵܦܘܿܖܹ̈ܐ. ܘܛܘܼܒ݂ܵܐ ܠܥܹܕ݁ܬܵܐ ܘܘܵܝܵܐ ܠܙܵܩܘܿܦܹܐ܀ ܫܪܵܝܵܗ̇ ܕܲܚܛܝܼܬ݂ܵܐ ܘܒ݂ܘܼܛܵܠܹܗ ܕܡܵܘܬܵܐ. ܡ̣ܠܲܟ݂ ܠܹܗ ܠܲܟ݂ܝܵܢܲܢ ܡܵܪܲܢ ܒܲܩܝܵܡܬܹ݁ܗ܀ ܚܵܕܹܝ̇ܢ ܝܵܘܡܵܢܵܐ ܟܸܢܫܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ. ܘܥܲܡܲܢ ܫܲܘܝܵܐܝܼܬ݂ ܡܲܣܩܝܼܢ ܬܸܫܒܘܿܚܬܵܐ܀
ܟܵܪܘܿܙܘܼܬ݂ܵܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.܀
ܡܵܪܝܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܖ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗܵܘ̇ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ̇ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗܵܘ̇ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡ̣ܠܲܟ݂ ܛܵܒ̣̈ܵܬ݂ܵܐ ܠܖ̈ܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕ݁ܬܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗܵܘ̇ ܕܠܵܐ ܨܵܒܹܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ: ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ܡ̣ܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗܵܘ̇ ܕܗܘ̤ܝܘ̣ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܖ̈ܝܵܬ݂ܵܐ: ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܝܼܣ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܝܼܣ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܵܘ̇ ܕܲܒ݂ܖ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗܵܘ̇ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.

ܨܠܘܿܬ݂ܵܐܕܩܲܕ݂ܡܵܝܵܐ. ܢܸܫܦܪܘܼܢ ܠܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܩܵܠܹ̈ܐ ܕܗܘܼܠܵܠܲܝ̈ܢ ܘܢܸܥ̈ܡܵܬ݂ܵܐ ܕܲܙܡܝܼܖ̈ܵܬܲܢ: ܘܩܲܒܸ݁ܠܡ̣ܢܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܦܹܐܖܹ̈ܐ ܡܠܝܼܠܹ̈ܐ ܕܣܸܦ̈ܘܵܬܲܢ: ܕܲܒ݂ܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܩܲܪܒ݂ܝܼܢܲܢ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܬܵܘܕ݁ܝܼ ܠܛܵܒ݂ܵܐ ܕܚܲܪܲܪ ܓܸܢܣܲܢ: ܡ̣ܢ ܥܲܒ݂ܕ݁ܘܼܬ݂ܵܐ ܕܒ݂ܝܼܫܵܐ ܘܡܵܘܬܵܐ܀ ܫܲܝܸܢ ܥܲܡܲܢ ܟܸܢܫܲܝ̈ ܪܵܘܡܵܐ: ܕܪܲܓ݁ܝܼܙܝܼܢ ܗ݇ܘ̣ܵܘ ܡܸܛܠ ܥܵܘ̣ܠܲܢ܀ ܒܪܝܼܟ݂ ܚܲܢܵܢܵܐ ܕܟܲܕ݂ ܠܵܐ ܒܥܲܝܢ̣ܵܝܗܝ: ܢ̣ܦܲܩ ܒܲܒ݂ܥܵܬܲܢ ܘܲܚ̣ܕ݂ܝܼ ܒܚܲܝܲܝ̈ܢ܀ ܘܨ̣ܵܪ ܕܸܡ̈ܘܵܬ݂ܵܐ ܕܐܲܒ݁ܝܼܕ݂ܘܼܬܲܢ: ܘܲܕ݂ܦܘܼܢܵܝܲܢ ܒܥܸܪܒܵܐ ܕܲܛ̣ܥܵܐ܀ ܝܵܪܬܵܐ ܘܲܒ݂ܪܵܐ ܩ̣ܪܵܝܗܝ ܠܲܟ݂ܝܵܢܲܢ: ܕܲܛ̣ܥܵܐ ܘܲܦܢ̣ܵܐ ܘܡ̣ܝܼܬ݂ ܘܐܸܬ݂ܢܲܚܲܡ܀ ܘܚܲܕ݁ܝܼ ܠܟܸܢܫܹ̈ܐ ܖ̈ܘܼܚܵܢܵܝܹܐ: ܒܲܬ݂ܝܵܒ݂ܘܼܬܲܢ ܘܲܒ݂ܢܘܼܚܵܡܲܢ܀ ܠܵܐ ܡܸܬ݂ܡܲܠܲܠ ܚܘܼܒܵܐ ܪܲܒܵܐ: ܕܚ̣ܵܘܝܼ ܨܹܐܕܲܝܢ ܪܵܚܹ̇ܡ ܓܸܢܣܲܢ܀ ܕܡܸܢܹܗ ܕܓܸܢܣܲܢ ܥ̣ܒܲܕ݂ ܡܸܨܥܵܝܵܐ: ܘܪܲܥ̣ܝܼ ܠܥܵܠܡܵܐ ܥܲܡ ܪܲܒܘܼܬܹܗ܀ ܪܲܒܵܐ ܗ݇ܝ̣ ܡܸܢܲܢ ܘܡ̣ܢ ܟܠ ܒܸܖ̈ܝܵܢ: ܚܕܲܬܵܐ ܕܲܣ̣ܥܲܪ ܨܹܝܕ݂ ܐ݇ܢܵܫܘܼܬܲܢ܀ ܕܥܲܒ݂ܕܹ݁ܗ ܠܦܲܓ݂ܪܲܢ ܗܲܝܟܲܠ ܩܘܼܕ݂ܫܵܐ: ܕܲܢܡܲܠܸܐ ܒܹܗ ܣܸܓ݂ܕܲܬ݂ ܟܠܵܐ܀ ܬܵܘ ܐܲܖ̈ܥܵܢܹܐ ܘܲܫܡܲܝܵܢܹ̈ܐ: ܬܗܲܪܘ ܘܐܸܬ݁ܕܲܡܲܪܘ ܒܪܲܒ݁ܘܼܬ݂ ܕܲܪܓ݂ܵܐ܀ ܕܲܡ̣ܛܵܐ ܓܸܢܣܲܢ ܠܪܵܘܡܵܐ ܪܲܒܵܐ: ܕܐܲܠܵܗܘܼܬ݂ܵܐ ܕܠܵܐ ܡܸܬ݁ܕܲܪܟܵܐ܀ ܫܡܲܝܵܐ ܘܐܲܪܥܵܐ ܘܟ݂ܠܡܵܐ ܕܲܒ݂ܗܘܿܢ: ܢܵܘܕ݁ܘܿܢ ܥܲܡܲܢ ܠܡܵܘܪܸܒ݂ ܓܸܢܣܲܢ܀ ܕܚܲܕܸ݁ܬ݂ ܨܲܠܡܲܢ ܘܲܥ̣ܛܵܐ ܥܵܘ̣ܠܲܢ: ܘܲܩ̣ܪܵܢ ܒܲܫܡܹܗ ܘܫܲܥܒܸ݁ܕ݂ ܠܲܢ ܟܠ܀ ܫܵܘܹ̇ܐ ܠܫܘܼܒ݂ܚܵܐ ܡ̣ܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܗ̇ܘ ܕܐܲܪܝܼܡܲܢ ܠܥܸܠ ܡ̣ܢ ܟܠܵܐ܀ ܘܢܹܐܡܲܪ ܟܠܲܢ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ: ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀

(ܘܪܵܡܹܐ ܡܫܲܡܫܵܢܵܐ ܐܘܼܪܵܪܵܐ: ܘܐܵܣܲܪ ܙܘܼܢܵܪܵܐ: ܘܩܵܐܹ̇ܡ ܩܕ݂ܵܡ ܫܟ݂ܝܼܢܬܵܐ ܡܛܲܟ݁ܣܵܐܝܼܬ݂(ܘܐܵܡܲܪ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ: ܒܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܒ݂ܥܹܐܕܹ̈ܐ ܘܲܒ݂ܕ݂ܘܼܟ݂ܖ̈ܵܢܹܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܚܲܕܝܼ ܘܐܲܦ̣ܨܲܚ ܠܐ݇ܢܵܫ̈ܵܐ ܘܡܲܠܲܐܟܹ̈ܐ. ܘܐܲܢܗܲܪܘ ܠܣܵܘܦܹܝ̈ܗ݁ ܕܥܵܡܲܪܬܵܐ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܕܵܨ̣ܘ ܥܸܠܵܝܹ̈ܐ. ܘܲܪܘܲܙ̣ܘ ܬܲܚܬܵܝܹ̈ܐ. ܘܲܩ̣ܥܵܘ ܫܘܼܒ݂ܚܵܐ ܟܠܗܘܿܢ ܥܒ݂ܝܼܕܹ̈ܐ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܪܲܡ̣ܪܸܡ ܘܥܲܠ̣ܝܼ ܐܝܼܩܵܪܹܗ ܕܲܟ̣ܝܵܢܲܢ. ܘܐܲܩ̣ܢܝܼ ܦܘܼܪܩܵܢܵܐ ܠܓܸܢܣܲܢ ܕܒܲܚܛܝܼܬ݂ܵܐ ܐܸܬ݂ܚܲܒܲܠ̣. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܚܲܪܲܪ ܠܲܟ̣ܝܵܢܲܢ ܡ̣ܢ ܥܲܒ݂ܕܘܼܬ݂ܵܐ ܕܫܹܐܕܹ̈ܐ ܒܝܼܫܹ̈ܐ. ܘܲܥܒ݂ܵܕܹ̈ܐ ܕܲܚܛܝܼܧ̣ܵ. ܘܝܲܗ̣݇ܒ݂ ܠܘܼܒܵܒ݂ܵܐ ܠܣܵܓ݂ܘܿܕ݂ܵܘ̈ܗܝ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܚܵܒ݂ ܣܵܛܵܢܵܐ. ܘܐܲܠ̣ܒ݁ܫܹܗ ܒܸܗܬ݁ܬ݂ܵܐ. ܘܲܗܘ̤ܵܐ ܚܸܣܕܵܐ ܗ̤ܘ ܘܚܲܝܠܵܘܵܬܹ̈ܗ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܣܠܸܩ̣ ܡ̣ܢ ܩܲܒ݂ܪܵܐ ܘܒܲܕܸܩ̣ ܥܲܠ ܣܘܼܠܵܩܹܗ ܕܠܲܫܡܲܝܵܐ. ܘܲܡ̣ܠܲܟ̣ ܠܲܢ ܕܥܲܡܹܗ ܢܲܣܩܲܢ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܠܩܲܣܛܘܿܢܵܖܹ̈ܐ ܘܲܠܣܵܦ݁ܖܹ̈ܐ ܐܲܠܒܸ݁ܫ̣ ܟܡܝܼܪܘܼܬ݂ܵܐ. ܘܐܲܩ̣ܢܝܼ ܚܲܫܵܐ ܘܟܲܪܝܘܼܬ݂ܵܐ ܘܐܲܚ̣ܦܲܪ ܐܲܦܲܝ̈ܗܘܿܢ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܐܸܬ݂ܚ̱ܙܝܼ ܠܡܲܓ݂ܕܠܵܝܧ̣ܵ ܘܲܚܒ݂ܵܖ̈ܵܬ݂ܵܗ݁ ܘܗܸܢܹܝܢ ܣܲܒܲܖ̈ܝ ܠܬܲܠܡܝܼܕ݂ܵܘ̈ܗܝ ܕܒܲܓ݂ܠܝܼܠܵܐ ܢܸܚܙܘܿܢܵܝܗܝ. ܣܒܲܪܬ݂ܵܐ ܡܲܠܝܲܬ݂ ܫܘܼܟܵܢܹ̈ܐ ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܖ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܦܪܘܿܩ ܘܚܲܕܵܐ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ ܒܥܹܐܕ݂ܵܐ ܦܨܝܼܚܵܐ ܕܲܩܝܵܡܬܵܟ̣. ܡܫܝܼܚܵܐ ܡܢܲܚܵܡܢܵܐ ܕܥܲܦܖ̈ܵܢܹܐ. ܘܐܸܬܼܪܲܚܲܡ ܥܠܲܝܢ.
ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
