ܛܲܟ݂ܣܵܐ ܕܬܸܫܡܸܫܬܵܐ
ܕܠܸܠܝܵܐ ܘܕܨܲܦܪܵܐ
 ܒܫܸܡ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕܫܵܐ ܠܥܵܠܡܝܼܢ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟ݂: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟ݂: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬ݂ܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

	ܐܲܩܸܦ ܬܵܘܫܸܛ ܐܝܼܕ݂ܵܟ݂ ܘܬܸܦܪܩܲܢܝ* ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܝܲܡܝܼܢܵܟ݂ ܡܵܪܝܵܐ ܐܲܓܸ݁ܢ ܥܠܲܝ*

ܡܵܘܬ݁ܒ݂ܵܐ
	ܡܘܬ݁ܒ݂ܐ. ܒܥܲܕ݂ܠܵܐ ܬܲܝܬܹ݁ܝܢ. ܢܹܐ̈ܙܵܠ݇ܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܒ݂ܒܲܣܝܼܡܘܼܬ݂ܵܐ: ܒܪܲܡܫܵܐ ܬܒ݂ܘܼܬ݂ ܒܟ݂ܵܬ݂ܵܐ ܘܲܒ݂ܨܲܦܪܵܐ ܚܲܕ݂ܘܼܬ݂ܵܐ. ܢܸܫܹ̈ܐ ܕܐܸܬܲܝ̈ ܗ݇ܘܲܝ̈ ܕܢܸܚ̈ܙܝܵܢ ܩܲܒ݂ܪܵܐ. ܬܲܗܪܵܐ ܘܕ݂ܘܼܡܵܪܵܐ ܐܸܚܲܕ݂ ܗ݇ܘܵܐ ܐܸܢܹܝܢ. ܟܲܕ݂ ܚܙܲܝ̈ ܗ݇ܘܲܝ̈ ܠܲܡܫܝܼܚܵܐ ܕܲܒ݂ܩܲܒ݂ܪܵܐ ܐܸܬ݁ܬ݁ܣܝܼܡ ܗ݇ܘܵܐ. ܘܲܚܙܲܝ̈ ܗ݇ܘܲܝ̈ ܠܢܵܛܘܿܪܹ̈ܐ ܕܲܠܩܲܒ݂ܪܵܐ ܢܵܛ̇ܪܝܼܢ ܗ݇ܘܵܘ. ܘܐܸܬ݂ܒܲܝܲܐ ܕܲܚܙܲܝ̈ ܡܲܠܲܐܟ݂ܵܐ ܕܲܢܚܸܬ݂ ܘܥܲܓܸܠ ܟܹܐܦܵܐ ܡܢ ܩܲܒ݂ܪܵܐ. ܘܲܠܢܵܛܘܿܪܹ̈ܐ ܓܵܘܢܵܐ ܚܸܫܘܿܟ݂ܵܐ ܐܲܠܒܸ݁ܫ ܗ݇ܘܵܐ. ܗܸܢܹܝܢ ܕܹܝܢ ܒܵܟ݂̈ܝܵܢ ܗ݇ܘܲܝ̈. ܟܲܕ݂ ܠܵܐ ܝܵܕ݂̈ܥܵܢ ܡܵܢܵܐ ܗܘܵܐ. ܝܼܫܘܿܥ ܕܹܝܢ ܦܓܲܥ ܒܗܹܝܢ ܘܲܒ݂ܓܲܢܵܢܵܐ ܬܘܼܒ݂ ܕܲܡܝܵܝ̈ܗܝ. ܡܵܪܝ ܐܸܢ ܐܲܢ݇ܬ݁ ܫܩܲܠܬܵܝܗܝ ܐܸܡܲܪܘ ܠܲܢ ܕܐܲܝܟܵܘ ܕܢܹܐܙܲܠ ܘܢܸܫܩܠܝܼܘܗܝ. ܐܸܠܵܐ ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂ ܚܘܼܣܥܠܲܝܢ. ܬܢܝܼ܀ ܐܸܬ݁ܕܲܠܚܲܬ݀ ܘܙܵܥܲܬ݀ ܐܲܪܥܵܐ. ܟܲܕ݂ ܐܸܬܲܒܲܙܲܚܬ݁ ܒܲܙܩܝܼܦܵܐ. ܡܫܝܼܚܵܐ ܡܵܪܵܗ̇ ܕܬܸܫܒܘܿܚܬܵܐ. ܒܨܸܨܹ̈ܐ ܘܪܘܼܡܚܵܐ ܣܲܬ݂ܪܘܼܗܝ ܒܝܼܫܹ̈ܐ ܠܗܲܝܟܲܠ ܐ݇ܢܵܫܘܼܬ݂ܵܟ݂. ܘܕܲܐܪܟܸܢܬ݁ ܪܹܫܵܟ݂ ܒܲܙܩܝܼܦܵܐ ܚܠܵܦ ܦܘܼܪܩܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ. ܫܸܡܫܵܐ ܚܫܸܟ݂ܘ ܗ݇ܘܵܘ ܙܲܠܝܼܩܵܘ̈ܗܝ ܕܐܲܫܠܸܡܬ݁ ܪܘܼܚܵܟ݂ ܥܲܠ ܩܲܝܣܵܐ. ܘܫܘܿܥܹ̈ܐ ܘܩܲܒ݂ܪܹ̈ܐ ܐܸܨܛܲܪܝܼܘ. ܘܡܝܼ̈ܬܹܐ ܕܲܒ݂ܗܘܿܢ ܐܸܬ݂ܢܲܚܲܡܘ. ܘܝܵܘܣܸܦ ܗܘ ܕܐܸܬ݁ܬܲܠܡܲܕ݂ ܠܵܟ݂. ܫܐܸܠ ܗ݇ܘܵܐ ܦܲܓ݂ܪܵܟ݂ ܩܲܕܝܼܫܵܐ. ܘܲܒ݂ܗܹܪ̈ܘܿܡܹܐ ܘܢܲܚ̈ܬܹܐ ܥܲܦ̮ܝܼ ܠܲܩܒ݂ܘܼܪܬܵܟ݂. ܘܲܒ݂ܩܲܒ݂ܪܵܐ ܕܝܼܠܹܗ ܣܵܡܵܟ݂. ܟܹܐܢܵܐ ܕܐܲܚܸܒ݂ ܓܸܠܝܵܢܵܟ݂. ܙܵܥܲܬ݀ ܫܝܘܿܠ ܕܐܲܓܸܣܬ݁ ܒܵܗ̇. ܕܒܸܠܥܲܬ݀ ܠܓܸܢܣܲܢ ܘܟ݂ܵܡܲܬ݀ ܥܠܵܘܗܝ. ܘܲܚܬܲܡܘ ܟܹܐܦܵܐ ܥܲܠ ܩܲܒ݂ܪܵܟ݂. ܙܵܒܹ݁ܢ ܥܵܠܡܹ̈ܐ ܡܢ ܡܵܘܬܵܐ ܝܲܥܢܵܐ. ܡܢܲܚܸܡ ܦܲܓ݂ܪܹ̈ܐ ܕܥܲܢܝܼ̈ܕܹܐ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܐܸܫܬ݁ܓܸܫܘ ܥܲܡܡܹ̈ܐ ܘܙܵܥܝ̈ ܡܲܠܟ̈ܘܵܬ݂ܵܐ. ܟܲܕ݂ ܩܵܡ ܦܵܪܘܿܩܲܢ ܡܢ ܩܲܒ݂ܪܵܐ. ܬܲܗܪܵܐ ܠܢܵܛܘܿܪܹ̈ܐ ܐܸܚܲܕ݂ ܗ݇ܘܵܐ ܐܸܢܘܿܢ. ܟܲܕ݂ ܚܵܝ̇ܪܝܼܢ ܗ݇ܘܵܘ ܒܲܡܫܝܼܚܵܐ. ܕܫܘܼܒ݂ܚܵܐ ܘܙܝܼܘܵܐ ܥܛܝܼܦ ܗ݇ܘܵܐ. ܘܚܵܙܹܝ̇ܢ ܗ݇ܘܵܘ ܬܘܼܒ݂ ܠܡܲܠܲܐܟܹ̈ܐ ܥܲܠ ܩܲܒ݂ܪܵܐ ܝܵܬ݂ܒ݁ܝܼܢ ܗ݇ܘܵܘ. ܘܲܒ݂ܕܸܚܠܵܐ ܪܲܒܵܐ ܡܣܲܒܪܝܼܢ ܗ݇ܘܵܘ. ܕܩܵܡ ܠܹܗ ܘܲܙܟ݂ܵܝܗܝ ܠܡܵܘܬܵܐ ܒܡܵܘܬܹ݁ܗ. ܘܲܠܢܵܛܘܿܪܹ̈ܐ ܕܝܼܢܵܐ ܘܫܘܼܢܵܩܵܐ ܛܲܝܸܒ݂ ܠܗܘܿܢ. ܘܲܠܚܸܫܘܿܟ݂ܵܐ ܕܛܘܼܥܝܲܝ ܒܢܘܼܚܵܡܹܗ ܛܲܠܸܩ ܗ݇ܘܵܐ. ܘܲܠܟ݂ܠܵܗ̇ ܒܪܝܼܬ݂ܵܐ ܢܘܼܗܪܵܐ ܡܢ ܓܵܘ ܩܲܒ݂ܪܵܐ ܕܢܲܚ. ܘܲܥܒܲܕ݂ ܣܲܒ݂ܪܵܐ ܘܦܘܼܪܩܵܢܵܐ ܘܢܘܼܚܵܡܵܐ ܘܚܲܝܹ̈ܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ. ܘܥܲܠ ܐܲܦܲܝ̈ ܗܵܕܹܐ ܢܸܩܥܸܐ ܘܢܹܐܡܲܪ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܫܲܬ݂ܐܣܹ̈ܐ ܕܛܘܼܪܹ̈ܐ ܙܵܥ̈ ܘܐܸܬ݁ܬܲܠܲܚ̈. ܟܲܕ݂ ܣܠܸܩ ܗ݇ܘܵܐ ܦܵܪܘܿܩܲܢ ܠܲܙܩܝܼܦܵܐ ܕܲܢܚܲܪܪܲܢ. ܒܸܪ̈ܝܵܬ݂ܵܐ ܟܠܗܹܝܢ ܙܵܥ̈ ܘܐܸܬ݂ܒܲܠܗܝܼ̈ ܕܲܚܙܲܝ̈ ܨܲܥܪܹܗ. ܫܸܡܫܵܐ ܦܐܸܐ ܒܙܲܠܝܼܩܵܘ̈ܗܝ ܐܲܚܫܸܟ݂ ܢܘܼܗܪܹܗ ܘܚܲܦ̮ܝܼ ܨܸܡܚܹܗ. ܕܲܚܙܵܐ ܠܡܵܪܹܗ ܕܲܬ݂ܠܸܐ ܒܩܲܝܣܵܐ ܘܚܵܐܹ̇ܫ ܡܢ ܡܲܪ̈ܵܚܹܐ. ܘܣܲܗܪܵܐ ܠܲܕ݂ܡܵܐ ܐܸܫܬܲܚܠܲܦ ܗ݇ܘܵܐ. ܠܡܲܟܣܵܢܘܼܬܹܗ ܕܥܲܡܵܐ ܣܲܟ݂ܠܵܐ ܕܐܲܡܪܲܚ ܘܲܣܥܵܐ ܘܲܩܛܲܠ ܡܵܪܹܗ. ܡܲܢܘܼ ܗܵܢܵܐ ܕܒܹܝܬ݂ ܩܘܼܕ݂ܫܵܐ ܒܩܸܛܠܹܗ ܐܸܣܬ݁ܕܸܩ. ܘܡܝܼ̈ܬܹܐ ܬܘܼܒ݂ ܒܩܵܠܹܗ ܡܸܬ݂ܢܲܚܡܝܼܢ. ܘܲܠܓܲܝܵܣܵܐ ܡܲܠܟܘܼܬ݂ܵܐ ܡܸܫܬܵܘܕܸܐ. ܟܒܲܪ ܗܘܝܘܼ ܡܫܝܼܚܵܐ ܡܲܚܝܵܢܵܐ ܕܡܝܼܬ݂ܘܼܬܲܢ. ܬܵܘ ܐܲܚܲܝ̈ ܢܸܪܗܲܛ ܠܣܸܓ݂ܕܬܹܗ. ܕܲܚܠܵܦ ܟܠܹܗ ܟܝܵܢܲܢ ܡܵܘܬܵܐ ܛܥܸܡ. ܘܢܸܩܥܸܐ ܠܹܗ ܟܠܲܢ ܕܠܵܐ ܫܲܠܘܵܐ ܡܵܪܹܟܠ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܡܚܵܐ ܠܲܒ݂ܥܸܠܕܒ݂ܵܒ݂ܵܘ̈ܗܝ ܠܒܸܣܬ݁ܪܹܗ. ܟܲܕ݂ ܣܠܸܩ ܗ݇ܘܵܐ ܦܵܪܘܿܩܲܢ ܒܬܸܫܒܘܿܚܬܵܐ ܡܢ ܓܵܘ ܫܝܘܿܠ. ܐܸܬ݂ܡܲܪܡܲܪ ܒܝܼܫܵܐ ܕܲܚܙܵܐ ܚܕܲܬܵܐ ܘܫܲܪܝܼ ܕܢܹܐܡܲܪ. ܕܐܸܢܵܐ ܒܝܲܕ݂ ܓܲܢ݇ܒܵܪܘܼܬ݂ܝ ܠܕ݂ܵܪܹ̈ܐ ܘܲܠܫܲܪ̈ܒ݂ܵܬ݂ܵܐ ܠܥܲܒ݂ܕܘܼܬ݂ܝ ܫܲܥ̇ܒܕܹܬ݂. ܘܐܸܢܵܐ ܒܝܲܕ݂ ܚܲܣܝܼܢܘܼܬ݂ܝ ܠܡܲܓ݂ܕܠܵܐ ܥܓ݂ܝܼܪ ܒܲܚܙܵܬܹܗ ܓܘܼܠܝܵܕ݂ ܫܲܦ݁ܠܹܬ݂. ܘܦܲܓ݂ܪܹ̈ܐ ܡܝܲܩܪܹ̈ܐ ܕܡܲܠܟܹ̈ܐ ܚܬ݂ܝܼܪܹ̈ܐ ܠܲܫܝܘܿܠ ܐܲܚ̇ܬܹܬ݂. ܘܲܠܓܲܢ݇ܒܵܪܹ̈ܐ ܘܲܠܚܲܣܝܼܢܹ̈ܐ ܐܸܢܵܐ ܡܲܟ݁ܟܹܬ݂. ܡܲܢܘܼ ܗܵܢܵܐ ܕܗܵܟܲܢܵܐ ܪܲܒ ܫܘܼܠܛܵܢܹܗ. ܕܗܵܐ ܒܵܥܹ̇ܐ ܕܢܸܣܲܒ݂ ܐܝܼܩܵܪܝ ܡܹܢܝ. ܘܗܵܐ ܦܵܐܲܚ̇ ܪܹܝܚܵܐ ܕܚܲܝܘܼܬܹܗ ܡܢ ܒܹܝܬ݂ ܡܝܼ̈ܬܹܐ ܟܠܗܘܿܢ. ܟܒܲܪ ܗܘܝܘܼ ܕܬ݂ܵܒܲܥ̇ ܚܘܼܫܒܵܢܵܐ ܕܥܲܢܝܼ̈ܕܹܐ ܀ ܬܵܘ ܐܲܚܲܝ̈ ܢܲܣܸܩ ܬܸܫܒܘܿܚܬܵܐ. ܠܕܲܒ݂ܡܵܘܬܹ݁ܗ ܓܲܢ݇ܒ݁ܪܲܢ ܥܲܠ ܡܵܘܬܵܐ. ܘܢܸܩܥܸܐ ܠܹܗ ܟܠܲܢ ܕܠܵܐ ܫܲܠܘܵܐ ܡܵܪܹܟ݁ܠ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂.
ܕܲܨܠܘܿܬ݂ܵܐ. ܫܒ݂ܝܼܚܘܼ ܬܘܼܟ݂ܠܵܢܵܟ݂. ܫܘܼܚܠܵܦܹܗ ܕܩܵܠܵܐ: ܝ.
ܐܲܠܵܗܲܢ ܒܹܝܼܬ݂ ܓܵܘܣܲܢ ܥܲܫܝܼܢܵܐ. ܫܒ݂ܝܼܚܘܼ ܬܘܼܟ݂ܠܵܢܵܟ݂ ܒܹܝܬ݂ ܓܵܘܣܲܢ ܥܲܫܝܼܢܵܐ: ܘܲܪܓ݂ܝܼܓ݂ ܘܲܡܝܲܬܲܪ ܚܢܵܢܵܟ݂ ܪܲܒܵܐ ܡܠܸܐ ܥܘܼܕ݂ܪ̈ܵܢܹܐ. ܕܲܡܨܸܐ ܦܵܪܹܩ ܠܲܕ݂ܡܸܬ݂ܓܵܘܣܝܼܢ ܒܹܗ ܡܢ ܟܠ ܒܝܼ̈ܫܵܢ: ܘܲܡܫܵܘܙܸܒ݂ ܬܘܼܒ݂ ܠܲܕ݂ܛܵܦܣܝܼܢ ܒܹܗ ܡܢ ܟܠ ܥܵܩ̈ܵܢ. ܘܐܲܝܢܵܐ ܕܐܲܡܝܼܢ ܘܢܵܩܹܫ ܒܬܲܪܥܹܗ ܠܵܐ ܡܲܗܡܸܐ ܡܸܢܹܗ: ܘܠܲܕ݂ܡܲܚܨܸܦ ܬܘܼܒ݂ ܘܫܵܐܹܠ ܪ̈ܲܚܡܹܐ ܡܲܣܓܸܐ ܥܵܢܹܐ ܠܹܗ. ܡܲܢܘܼ ܗܵܢܵܐ ܕܠܵܐ ܢܸܪܗܲܛ ܘܢܸܬ݂ܓܵܘܲܣ ܒܹܗ: ܒܲܚܢܵܢܵܟ݂ ܡܠܸܐ ܟܠ ܥܘܼܕ݂ܪ̈ܵܢܝܼܢ. ܘܢܸܫܐܲܠ ܡܸܢܹܗ ܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ: ܟܲܕ݂ ܩܵܥܹܐ ܒܲܚܢܲܓ݂ܬ݂ܵܐ ܘܐܵܡܲܪ. ܚܘܼܣ ܥܠܲܝ ܐܲܢ݇ܬ݁ ܚܲܢܵܢܵܐ: ܘܲܦܪܘܿܩܲܝܢܝ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁. ܛܵܒ݂ܵܐ ܕܠܵܐ ܟܵܠܹܐ ܪ̈ܲܚܡܵܘܗܝ ܡܵܪܹܐ ܟܠ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܐܸܢܵܐ ܕܹܝܢ ܥܠܲܝܟ ܡܵܪܝܵܐ ܐܸܬ݁ܬܲܟ݂ܠܹܬ݂. ܐܲܢ݇ܬ݁ܘܼ ܬܘܼܟ݂ܠܵܢܲܢ ܘܒܹܝܬ݂ ܓܵܘܣܲܢ ܦܵܪܘܿܩܲܢ: ܘܠܵܟ݂ܘܼ ܡܩܲܪܒ݂ܝܼܢܲܢ ܫܸܐ̈ܠܵܬ݂ܵܐ ܠܲܣܢܝܼܩܘܼܬܲܢ. ܦܬܲܚ ܠܲܦܝܵܣܲܢ ܬܪܲܥ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܡܠܸܐ ܥܘܼܕ݂ܪ̈ܵܢܹܐ: ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܸܐܠܲܬ݂ ܢܲܦ̮ܫܲܢ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁. ܘܚܲܣܵܐ ܚܵܘܒܹ̈ܐ ܘܣܲܟ݂̈ܠܘܵܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܡܵܪܝܵܐ: ܘܗܲܒ݂ ܥܘܼܕ݂ܪܵܢܵܐ ܠܲܡܚܝܼܠܘܼܬܲܢ ܕܢܸܓ݂ܡܘܿܪ ܨܸܒ݂ܝܵܢܵܟ݂. ܨܠܝܼܒ݂ܵܟ݂ ܡܵܪܲܢ ܫܘܼܪܵܐ ܪܵܡܵܐ ܢܸܗܘܸܐ ܠܲܢ: ܒܟ݂ܠ ܥܸܕܵܢ̈ܝܼܢ ܡܢ ܟܠ ܢܸܟ݂ܝܵܢ̈ܝܼܢ. ܘܒܹܗ ܢܸܣܬܲܬܲܪ ܡܢ ܒܝܼܫܵܐ ܒܥܸܠܕܲܪܲܢ: ܘܠܵܟ݂ ܢܵܘܕܸܐ ܘܲܢܫܲܒܲܚ ܘܢܹܐܡܲܪ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܪܵܚܹܡ ܐ݇ܢܵܫܵܐ: ܕܠܵܐ ܒܵܗܬ݁ܝܼܢ ܕܲܡܣܲܒܪܝܼܢ ܒܵܟ݂. ܡܵܪܵܐ ܕܠܵܐ ܡܲܗܡܸܐ ܡܢ ܟܠ ܕܩܵܪܹܝܢ ܠܹܗ ܡܵܪܹܐ ܟܠ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܕܡܲܪܝܲܡ܀ ܢܫܲܒܲܚ ܘܢܸܚܕܸܐ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬܲܢ. ܫܘܼܒ݂ܚܵܐ ܠܚܲܝܠܵܐ ܬܠܝܼܬ݂ܵܝܵܐ: ܕܲܢܚܸܬ݂ ܘܲܫܪܵܐ ܥܲܠ ܒܬ݂ܘܼܠܬܵܐ. ܘܲܓܒܲܠ ܡܸܢܵܗ̇ ܒܲܪܢܵܫܵܐ: ܘܥܲܒ݂ܕܹܗ ܗܲܝܟ݁ܠܵܐ ܠܲܥܡܘܼܪܝܹܗ. ܘܒܹܗ ܐܸܫܬܵܘܕܝܼ ܠܡܝܼܬ݂ܘܼܬܲܢ: ܚܲܝܹ̈ܐ ܚܲܕ݂ܬܹ̈ܐ ܘܫܲܪܝܼܪܹ̈ܐ. ܘܗܘܝܘܼ ܢܘܼܗܪܵܐ ܫܲܪܝܼܪܵܐ: ܠܡܲܠܲܐܟܹ̈ܐ ܘܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ. ܡܫܝܼܚܵܐ ܕܲܕ݂ܢܲܚ ܡܢ ܡܲܪܝܲܡ: ܠܦܘܼܪܩܵܢ ܟܠܗܹܝܢ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܐܲܫܪܵܐ ܫܲܝܢܵܟ݂ ܒܲܒ݂ܪܝܼܬ݂ܵܐ: ܘܛܲܪ ܥܵܡܘܿܪܹ̈ܝܗ̇ ܒܲܚܢܵܢܵܟ݂. ܘܫܲܝܸܢ ܟܵܗܢܹ̈ܐ ܥܲܡ ܡܲܠܟܹ̈ܐ: ܕܥܘܼܡܪܵܐ ܫܲܠܝܵܐ ܘܢܝܼܚܵܐ ܢܸܥܡܲܪ. ܒܲܨܠܘܿܬ݂ ܡܲܪܝܲܡ ܕܝܼܠܸܕ݂ܬ݂ܵܟ݂: ܛܲܪ ܥܹܕ݁ܬܵܟ݂ ܡܢ ܢܸܟ݂ܝܵܢܹ̈ܐ ܘܚܘܼܣ ܥܠܹܝܗ̇. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܛܲܪ ܪ̈ܲܚܡܲܝܟ ܠܩܲܪ̈ܝܼܒܲܝܟ. ܡܫܝܼܚܵܐ ܡܵܪܵܗ̇ ܕܬܸܫܒܘܿܚܬܵܐ: ܕܐܵܘܪܸܒ݂ ܒܚܘܼܒܹܗ ܠܩܲܕܝܼܫܵܘ̈ܗܝ. ܒܲܨܠܘܿܬ݂ ܢܒ݂ܝܼܹ̈ܐ ܚܘܼܣ ܥܠܲܝܢ: ܘܒܲܨܠܘܿܬ݂ ܫܠܝܼܚܹ̈ܐ ܐܸܬ݂ܪܲܥܵܐ ܠܲܢ. ܘܒܲܨܠܘܿܬ݂ ܣܵܗܕܹ̈ܐ ܚܲܝܸܠ ܠܲܢ: ܕܢܸܙܟܸܐ ܠܚܲܫܹ̈ܐ ܕܲܚܛܝܼܬ݂ܵܐ. ܘܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܘܥܵܢܘܵܝܹ̈ܐ: ܕܚܘܿܩ ܡܸܢܲܢ ܚܲܝܠܹܗ ܕܒ݂ܝܼܫܵܐ. ܘܒܲܨܠܘܿܬ݂ ܟܵܗܢܹ̈ܐ ܒܲܪܸܟ݂ ܡܵܪܝ: ܟܠܝܼܠܵܐ ܕܫܲܢ݇ܬܵܐ ܒܛܲܝܒܘܼܬ݂ܵܟ݂. ܘܥܲܡ ܡܲܠܦܵܢܹ̈ܐ ܢܸܗܪܘܿܓ݂ ܒܣܸܦܪܹ̈ܐ ܕܝܘܼܠܦܵܢܵܟ݂: ܘܢܸܬ݂ܩܲܒܠܘܼܢ ܡܵܪܝ ܩܘܼܪ̈ܒܵܢܲܝܢ. ܩܕ݂ܵܡ ܒܹܝܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܢܸܗܘܸܐ ܫܪܵܝܵܐ ܠܥܵܩ̈ܵܬܲܢ. ܘܚܘܼܣܵܝܵܐ ܠܣܲܟ݂̈ܠܘܵܬܲܢ: ܘܥܲܡ ܟܹܐܢܹ̈ܐ ܢܸܚܙܸܐ ܐܲܦܲܝ̈ܟ. ܒܝܵܘܡܵܐ ܕܕ݂ܵܢܚܵܐ ܪܲܒܘܼܬ݂ܵܟ݂: ܘܢܸܙܡܲܪ ܫܘܼܒ݂ܚܵܐ ܠܡܵܪܘܼܬ݂ܵܟ݂ ܒܟ݂ܠ ܥܸܕܵܢ̈ܝܼܢ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܕܢܲܚ ܢܘܼܗܪܵܐ ܒܚܸܫܘܿܟ݂ܵܐ ܠܲܬ݂ܪ̈ܝܼܨܹܐ. ܢܘܼܗܪܵܐ ܘܚܲܝܹ̈ܐ ܘܲܫܪܵܪܵܐ: ܕܡܲܢܗܲܪ ܠܣܵܘܦܲܝ̈ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܨܠܝܼܒ݂ܵܐ ܡܲܒܘܼܥ ܛܵܒ݂̈ܵܬܲܢ: ܙܲܝܢܵܐ ܕܙܲܟ݁ܝܹܗ ܠܲܟ݂ܝܵܢܲܢ. ܟܐܝܼ ܒܠܸܓ݂ܝܘܿܢܵܘ̈ܗܝ ܕܲܢܟ݂ܝܼܠܵܐ: ܕܚܵܪܹܒ݂ ܣܸܕ݂ܪܹ̈ܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܘܢܲܦܸܣ ܪܸܡܙܵܟ݂ ܠܪ̈ܘܼܚܵܢܹܐ: ܕܲܢܓܲܒܘܿܢ ܡܵܪܝ ܙܝܼܙܵܢܹ̈ܐ. ܘܐܲܪܝܼܡ ܩܲܪܢܵܗ̇ ܕܲܪܚܘܼܡܬܵܟ݂: ܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ ܡܗܲܝܡܲܢܬܵܐ. ܕܲܨܕܲܝ̈ ܕܵܪܹ̈ܝܗ̇ ܡܢ ܫܘܼܒ݂ܚܵܟ݂: ܒܝܲܕ݂ ܪܲܦܝܘܼܬ݂ܵܐ ܕܥܲܠܠܵܢܹܝ̈ܗ̇. ܘܝܲܠܕܹܝ̈ܗ̇ ܦܵܗܹ̇ܝܢ: ܒܹܝܬ݂ ܩܸܩ̈ܠܵܬ݂ܵܐ ܘܡܲܪܝܼܪ ܠܵܗ̇. ܚܘܼܒܵܟ݂ ܥܲܒ݂ܕܵܟ݂ ܒܲܪܢܵܫܵܐ: ܟܲܕ݂ ܐܲܠܵܗܵܐ ܐܲܢ݇ܬ݁ ܒܲܟ݂ܝܵܢܵܟ݂. ܒܚܘܼܒܵܟ݂ ܫܲܪܝܼܬ݁ ܒܬ݂ܘܼܩܵܢܲܢ: ܒܚܘܼܒܵܟ݂ ܫܲܠܸܡ ܥܲܡ ܓܸܢܣܲܢ. ܘܐܲܫܪܵܐ ܫܲܝܢܵܟ݂ ܒܲܒ݂ܪܝܼܬ݂ܵܐ: ܫܠܵܡܵܐ ܕܐܲܝܬ݁ܝܼܘ ܪ̈ܘܼܚܵܢܹܐ. ܐܘܿ ܨܠܝܼܒ݂ܵܐ ܡܵܘܬܹ݁ܗ ܕܡܵܘܬܵܐ ܚܘܼܣ ܥܠܲܝܢ. ܕܐܲܒ݂ܘܼܢ܀ ܣܹܐܡܵܐ ܓܲܒ݂ܝܵܐ ܕܲܒ݂ܚܝܼܪ ܒܐܲܪܥܵܐ. ܣܝܼܡܬ݂ܵܐ ܕܓܲܙܵܐ ܫܡܲܝܵܢܵܐ: ܐܝܼܬ݂ܵܘܗܝ ܦܲܓ݂ܪܵܟ݂ ܐܘܿ ܐܲܒ݂ܘܼܢ. ܡܵܪܝ ܦܠܢ ܢܲܨܝܼܚܵܐ: ܡܵܐܢܵܐ ܓܲܒ݂ܝܵܐ ܘܲܡܝܲܬ݁ܪܵܐ. ܕܲܗܘܲܝܬ݁ ܚܲܒ݂ܪܵܐ ܘܒܲܪ ܣܝܼܥܬܵܐ: ܠܐܸܣܛܲܦܵܢܘܿܣ ܩܲܕ݂ܡܵܝܵܐ. ܘܒܹܝܬ݂ ܡܵܩܵܪܝܼܣ ܘܐܲܪܣܵܢܝܼܣ ܘܦܵܘܠܘܿܣ ܪܹܫܵܐ ܕܐܲܒ݂ܝܼ̈ܠܹܐ. ܛܘܼܒܲܝܟ ܕܲܢܛܝܼܪܘܼ ܛܘܼܒ݂ܵܟ݂: ܥܲܡ ܗ̇ܝ ܣܝܼܥܬܵܐ ܕܲܒ݂ܢܲܝ̈ ܐܲܦܸܣܘܿܣ. ܕܲܦܠܲܚܬ݁ ܒܟܲܪܡܵܐ ܪܘܼܚܵܢܵܐ: ܘܲܢܣܲܒ݂ܬ݁ ܐܲܓ݂ܪܵܐ ܡܫܲܠܡܵܢܵܐ. ܨܠܘܿܬ݂ܵܟ݂ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܹܗ: ܠܟܸܢܫܵܐ ܕܙܲܝܲܚ ܕܘܼܟ݂ܪܵܢܵܟ݂. ܣܥܘܿܪ ܡܲܪ̈ܥܹܐ ܘܐܲܚܠܸܡܟܪ̈ܝܼܗܹܐ ܒܲܨ̈ܠܵܘܵܬ݂ܵܟ݂. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܡܸܛܠ ܕܩܲܠܝܼܠܘܼ ܙܲܒ݂ܢܵܐ ܛܵܒ݂ ܘܲܙܥܘܿܪ. ܩܲܠܝܼܠ ܘܲܙܥܘܿܪ ܙܲܒ݂ܢܵܐ. ܕܪܵܕܹܝܢܲܢ ܒܥܵܠܡܵܐ ܗܵܢܵܐ. ܚܲܛܵܝܹ̈ܐ ܬܘܼܒ݂ܘܼܢ ܘܲܚܝܵܘ. ܕܡܵܘܬܵܐ ܫܘܼܚܕܵܐ ܠܵܐ ܢܵܣܹܒ݂. ܘܲܠܥܲܬ݁ܝܼܪܹ̈ܐ ܘܡܸܣܟܹܢܹ̈ܐ ܐܲܟܲܚ݇ܕ݂ ܕܵܒܲܪ ܫܲܘܝܵܐܝܼܬ݂. ܫܲܒܲܚܘ ܠܹܗ ܬܸܫܒܘܿܚܬܵܐ ܚܕܲܬܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܕܓܲܒ݂ܠܲܢ. ܘܠܲܒ݂ܪܵܐ ܕܦܲܪܩܲܢ ܒܲܨܠܝܼܒܹܗ. ܘܲܠܪܘܼܚܵܐ ܦܵܪܲܩܠܹܝܛܵܐ. ܐ݇ܪܵܙܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܐ. ܕܗܘܝܘܼ ܡܢܲܚܸܡ ܥܲܢܝܼܕܲܝ̈ܢ. ܘܡܲܠܒܸ݁ܫ ܫܘܼܒ݂ܚܵܐ ܦܲܓ݂ܪ̈ܲܝܗܘܿܢ܀
ܫܲܒܲܚ. ܒܩܵܠܵܐ ܪܲܒܵܐ. ܢܸܫܹ̈ܐ ܕܐܸܬܲܝ̈ ܠܒܹܝܬ݂ ܩܒ݂ܘܼܪܵܟ݂ ܡܵܪܝܵܐ. ܚܠܵܦ ܟܲܪܝܘܼܬ݂ܵܐ ܚܲܕ݂ܘܼܬ݂ܵܐ ܐܸܬ݂ܡܠܝܼ̈. ܘܠܲܫܠܝܼܚܹ̈ܐ ܪ̈ܗܸܛ ܗ݇ܘܲܝ̈ ܕܲܢܣܲܒܪ̈ܵܢ. ܡܸܠܹ̈ܐ ܕܡܢ ܡܲܠܲܐܟ݂ܵܐ ܫܡܲܥ̈ ܗ݇ܘܲܝ̈. ܕܐܸܡܲܪ ܠܗܹܝܢ ܕܐܲܢ݇ܬܹ݁ܝܢ ܠܵܐ ܬܸܕ݂ܚ̈ܠܵܢ. ܠܡܢ ܒܵܟ݂ܝܵܬܹ݁ܝ̈ܢ. ܠܝܼܫܘܿܥ ܒܵܥܝܵܬܹ݁ܝ̈ܢ. ܩܵܡ ܠܹܗ ܘܲܙܟ݂ܵܝܗܝ ܠܡܵܘܬܵܐ ܒܡܵܘܬܹ݁ܗ. ܚܢܲܢ ܕܹܝܢ ܡܵܪܝ ܒܬ݂ܵܘܕܝܼܬ݂ܵܐ ܢܸܩܥܸܐ ܘܢܹܐܡܲܪ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂.
ܥܵܠܲܡ. ܡܢ ܒܵܬܲܪ ܩܝܵܡܬܵܟ݂ ܡܫܲܒܲܚܬܵܐ. ܥܲܡܵܐ ܕܲܝܗܘܼܕ݂ܝܹ̈ܐ ܠܵܐ ܗܲܝܡܢܘ ܒܵܟ݂. ܟܲܕ݂ ܩܵܡܬ݁ ܗ݇ܘܲܝܬ݁ ܡܢ ܒܹܝܬ݂ ܡܝܼ̈ܬܹܐ. ܐܸܠܵܐ ܡܸܛܠ ܟܸܣܦܵܐ ܕܲܢܣܲܒ݂ܘ ܢܵܛܘܿܪܹ̈ܐ. ܐܲܟ݂ܪܸܙܘ ܘܐܸܡܲܪܘ ܕܲܓܵܠܵܐܝܼܬ݂. ܕܬܲܠܡܝܼܕ݂ܵܘ̈ܗܝ ܓܲܢܒ݁ܘܼܗܝ ܟܲܕ݂ ܕܲܡܟ݁ܝܼܢܲܢ. ܐܘܿ ܠܹܗ ܠܥܲܡܵܐ ܕܠܵܐ ܗܲܝܡܸܢܘ ܠܲܫܪܵܪܵܐ. ܐܸܢܗܘ ܕܦܲܓ݂ܪܵܐ ܬܲܠܡܝܼܕܹ̈ܐ ܓܢܲܒ݂ܘ. ܢܲܚ̈ܬܹܐ ܘܟܸܬܵܢܹ̈ܐ ܠܡܵܢ ܒܒܹܝܬ݂ ܩܲܒ݂ܪܵܐ ܫܒܲܩܘ. ܘܲܠܡܵܢ ܟܲܕ݂ ܣܲܒ݁ܪܘܼܗܝ ܐܲܚܵܘ̈ܗܝ ܠܲܬ݂ܐܘܿܡܵܐ ܕܩܵܡ ܠܹܗ ܡܵܪܲܢ ܐܲܚܪܸܡ ܗ݇ܘܵܐ ܘܐܸܡܲܪ. ܕܐܸܢ ܠܵܐ ܚܵܙܹ̇ܐ ܐ݇ܢܵܐ ܒܐܝܼܕ݂ܵܘ̈ܗܝ ܕܘܼܟ̈ܝܵܬ݂ܵܐ ܕܨܸܨܹ̈ܐ ܗܵܢܹ̇ܝܢ ܕܲܒ݂ܠܲܥ ܡܢ ܒܝܼܫܹ̈ܐ. ܠܵܐ ܡܗܲܝܡܸܢ ܐ݇ܢܵܐ ܕܩܵܡ ܦܵܪܘܿܩܵܐ ܕܒܸܪ̈ܝܵܬ݂ܵܐ. ܕܲܩܕ݂ܵܡܲܝ ܠܲܒ݂ܟ݁ܘܼܗܝ ܘܐܵܘܒ݁ܠܘܼܗܝ ܠܘܵܬ݂ ܦܝܼܠܵܛܘܿܣ. ܘܲܩܕ݂ܵܡܲܝ ܙܲܩܦܘܼܗܝ ܥܲܠ ܪܹܫ ܩܲܝܣܵܐ ܒܓ݂ܵܓ݂ܘܼܠܬܵܐ. ܐܵܦ ܡܲܪܝܲܡ ܙܸܒ݂ܢܲܬ݀ ܗܹܪ̈ܘܿܡܹܐ. ܘܐܸܬܲܬ݀ ܠܬܲܪܥܵܐ ܕܒܹܝܬ݂ ܩܒ݂ܘܼܪܵܐ. ܘܟܲܕ݂ ܝܼܕܲܥ ܗ݇ܘܵܐ ܦܵܪܘܿܩܲܢ. ܕܲܫܠܝܼܚܹ̈ܐ ܐܸܬ݂ܦܲܠܲܓ݂ܘ ܗ݇ܘܵܘ. ܒܲܫܪܵܪܹܗ ܠܗܘܿܢ ܫܲܪܲܪ. ܢܲܓܝܼܪ ܪܘܼܚܵܐ ܪܲܚܸܡܥܠܲܝܢ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܟ݂ܠܗܘܿܢ ܥܹܐܕܹ̈ܐ ܡܵܪ̈ܵܢܵܝܹܐ. ܠܲܡܕܲܒ݁ܪܵܢܘܼܬ݂ܵܟ݂ ܡܵܪܝ ܬܡܝܼܗܬܵܐ ܘܠܵܐ ܡܸܬ݂ܡܲܠܠܵܢܝܼܬ݂ܵܐ: ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܘܒܲܚܢܵܢܵܐ ܐܸܬ݂ܓܲܡܪܲܬ݀ ܘܐܸܣܬܲܥܪܲܬ݀ ܘܐܸܫܬܲܡܠܝܲܬ݀ ܠܚܘܼܕܵܬܹܗ ܘܲܠܦܘܼܪܩܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ ܚܲܠܵܫܵܐ: ܒܪܹܫܝܼܬ݂ܵܐ ܕܡܢܢ: ܡܲܣܩܝܼܢܲܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܩܵܢܘܿܢܵܐ.(ܡܲܙ.ܩܡܚ.)
 ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡܢ ܫܡܲܝܵܐ* ܫܲܒ݁ܚܘܼܗܝ ܒܲܡܪ̈ܵܘܡܹܐ܀ ܡܢ ܫܡܲܝܵܐ ܢܚܸܬ݂ܘ ܥܝܼܪܹ̈ܐ ܒܐܝܼܩܵܪܵܐ ܘܫܲܡܸܫܘ ܒܕܸܚܠܬ݂ܵܐ. ܠܗܲܝܟܲܠ ܦܲܓ݂ܪܵܐ ܕܲܒ݂ܪܵܐ ܡܫܝܼܚܵܐ ܕܚܲܫ ܥܲܠ ܐܲܦܲܝ̈ܢ. ܘܲܚܙܵܘ ܫܘܼܒ݂ܚܵܐ ܕܲܩܝܵܡܬܹ݁ܗ ܒܬܸܕ݂ܡܘܼܪܬܵܐ ܘܣܲܒ݁ܪܘܼܗܝ ܠܐܲܪ̈ܥܵܢܹܐ܀ ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡܢ ܫܡܲܝܵܐ* ܫܲܒ݁ܚܘܼܗܝ ܒܲܡܪ̈ܵܘܡܹܐ܀ ܫܲܒ݁ܚܘܼܗܝ ܟܠܗܘܿܢ ܡܲܠܲܐܟ݂ܵܘ̈ܗܝ* ܫܲܒ݁ܚܘܼܗܝ ܟܠܗܘܿܢ ܚܲܝ̈ܠܵܘܵܬܹܗ܀ ܫܲܒ݁ܚܘܼܗܝ ܫܸܡܫܵܐ ܘܣܲܗܪܵܐ* ܫܲܒ݁ܚܘܼܗܝ ܟܠ ܟܵܘܟ݁ܒܹ̈ܐ ܘܢܘܼܗܪܵܐ܀ ܫܲܒ݁ܚܘܼܗܝ ܫܡܲܝ̈ ܫܡܲܝܵܐ* ܡܲܝ̈ܵܐ ܕܲܠܥܸܠ ܡܢ ܫܡܲܝ̈ܵܐ ܢܫܲܒ݁ܚܘܼܢ ܠܲܫܡܹܗ ܕܡܵܪܝܵܐ܀ ܡܸܛܠ ܕܗܘ ܐܸܡܲܪ ܘܲܗܘܵܘ ܘܗܘ ܦܩܲܕ݂ ܘܐܸܬ݂ܒ݁ܪܝܼܘ* ܘܐܲܩܝܼܡ ܐܸܢܘܿܢ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ܀ ܢܵܡܘܿܣܵܐ ܝܲܗ݇ܒ݂ ܘܠܵܐ ܥܵܒܲܪ* ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡܢ ܐܲܪܥܵܐ܀ ܬܲܢܝܼܢܹ̈ܐ ܘܟ݂ܠܗܘܿܢ ܬܗܘܿܡܹ̈ܐ* ܢܘܼܪܵܐ ܘܒܲܪܕ݂ܵܐ ܘܬܲܠܓܵܐ ܘܲܓ݂ܠܝܼܕ݂ܵܐ܀ ܪ̈ܘܼܚܹܐ ܘܥܲܠܥܵܠܹ̈ܐ ܥܒ݂ܵܕ݂ܵܐ ܕܡܸܠܬܹܗ* ܛܘܼܪܹ̈ܐ ܘܟ݂ܠܗܹܝܢ ܪ̈ܵܡܵܬ݂ܵܐ܀ ܐܝܼܠܵܢܹ̈ܐ ܕܦܹܐܪܹ̈ܐ ܘܟ݂ܠܗܘܿܢ ܐܲܪ̈ܙܹܐ* ܚܲܝܘܬ݂ܵܐ ܘܟ݂ܠܵܗ̇ ܒܥܝܼܪܵܐ܀ ܪ̈ܲܚܫܵܐ ܘܛܲܝܪܵܐ ܘܦܵܪܲܚܬ݂ܵܐ* ܡܲܠܟܹ̈ܐ ܕܐܲܪܥܵܐ ܘܟ݂ܠܗܹܝܢ ܐܸܡ̈ܘܵܬ݂ܵܐ܀ ܪܵܘܪ̈ܒ݂ܵܢܹܐ ܘܟ݂ܠܗܘܿܢ ܕܲܝܵܢܹ̈ܐ ܕܐܲܪܥܵܐ* ܓܲܕ݁ܘܼܕܹ̈ܐ ܘܲܒ݂ܬ݂ܘܼ̈ܠܵܬ݂ܵܐ܀ ܣܵܒܹ̈ܐ ܘܲܥܠܲܝܡܹ̈ܐ ܢܫܲܒ݁ܚܘܼܢ ܠܲܫܡܹܗ ܕܡܵܪܝܵܐ* ܡܸܛܠ ܕܪܲܒ݁ܘܼ ܫܡܹܗ ܒܲܠܚܘܿܕ݂ܵܘܗܝ܀ ܘܫܘܼܒ݂ܚܹܗ ܒܐܲܪܥܵܐ ܘܒܲܫܡܲܝܵܐ* ܡܪܝܼܡ ܩܲܪܢܵܐ ܕܥܲܡܹܗ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܕܟ݂ܠܗܘܿܢ ܙܲܕ݁ܝܼ̈ܩܵܘܗܝ܀ ܠܲܒ݂ܢܲܝ̈ ܝܑܼܣܪܵܝܹܠ ܥܲܡܵܐ ܕܩܲܪܝܼܒ݂ ܠܹܗ܀ ܡܢ ܫܡܲܝܵܐ ܢܚܸܬ݂ܘ ܥܝܼܪܹ̈ܐ ܒܐܝܼܩܵܪܵܐ ܘܫܲܡܸܫܘ ܒܕܸܚܠܬ݂ܵܐ. ܠܗܲܝܟܲܠ ܦܲܓ݂ܪܵܐ ܕܲܒ݂ܪܵܐ ܡܫܝܼܚܵܐ ܕܚܲܫ ܥܲܠ ܐܲܦܲܝ̈ܢ. ܘܲܚܙܵܘ ܫܘܼܒ݂ܚܵܐ ܕܲܩܝܵܡܬܹ݁ܗ ܒܬܸܕ݂ܡܘܼܪܬܵܐ ܘܣܲܒ݁ܪܘܼܗܝ ܠܐܲܪ̈ܥܵܢܹܐ܀
ܫܲܒܲܚ܀ ܥܵܠܲܡ܀ ܡܢ ܒܵܬܲܪ ܩܒ݂ܘܼܪܬܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܐܸܬ݂ܟܲܢܲܫܘ ܥܲܡܵܐ ܕܲܝܗܘܼܕ݂ܵܝܹ̈ܐ. ܘܲܠܢܵܛܘܿܪܹ̈ܐ ܟܸܣܦܵܐ ܝܲܗ݇ܒ݂ܘ ܠܗܘܿܢ ܕܢܸܛܪܘܼܢ ܩܲܒ݂ܪܵܟ݂. ܘܡܸܢܫܹܠܝ ܐܲܒ݂ܗܸܬ݁ܬ݁ ܠܨܵܠܘܿܒܲܝ̈ܟ ܘܚܲܕܝܼܬ݁ ܠܥܹ̈ܕ݂ܵܬ݂ܵܟ݂܀
ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡܢ ܫܡܲܝܵܐ* ܫܲܒ݁ܚܘܼܗܝ ܒܲܡܪ̈ܵܘܡܹܐ. ܡܢ ܒܵܬܲܪ ܩܒ݂ܘܼܪܬܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܐܸܬ݂ܟܲܢܲܫܘ ܥܲܡܵܐ ܕܲܝܗܘܼܕ݂ܵܝܹ̈ܐ. ܘܲܠܢܵܛܘܿܪܹ̈ܐ ܟܸܣܦܵܐ ܝܲܗ݇ܒ݂ܘ ܠܗܘܿܢ ܕܢܸܛܪܘܼܢ ܩܲܒ݂ܪܵܟ݂. ܘܡܸܢܫܹܠܝ ܐܲܒ݂ܗܸܬ݁ܬ݁ ܠܨܵܠܘܿܒܲܝ̈ܟ ܘܚܲܕܝܼܬ݁ ܠܥܹ̈ܕ݂ܵܬ݂ܵܟ݂܀

ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.܀
ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ݁ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡܠܲܟ݂ ܛܵܒ݂̈ܵܬ݂ܵܐ ܠܪ̈ܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕ݁ܬܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܠܵܐ ܨܵܒܹܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ: ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ܡܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܗܘܝܘ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܪ̈ܝܵܬ݂ܵܐ: ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܝܼܣ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܝܼܣ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܵܘ̇ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.

ܨܠܘܿܬ݂ܵܐܕܩܲܕ݂ܡܵܝܵܐ. ܢܸܫܦܪܘܼܢ ܠܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܩܵܠܹ̈ܐ ܕܗܘܼܠܵܠܲܝ̈ܢ ܘܢܸܥ̈ܡܵܬ݂ܵܐ ܕܲܙܡܝܼܪ̈ܵܬܲܢ: ܘܩܲܒܸ݁ܠܡܢܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܦܹܐܪܹ̈ܐ ܡܠܝܼܠܹ̈ܐ ܕܣܸܦ̈ܘܵܬܲܢ: ܕܲܒ݂ܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܩܲܪܒ݂ܝܼܢܲܢ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

	ܨܠܘܿܬ݂ܵܐ ܕܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ. ܡܢ ܦܘܼܡܹ̈ܐ ܡܵܪܝ ܡܠܝܼܠܹ̈ܐ ܕܲܒ݂ܪܲܝܬ݁: ܘܡܢ ܠܸܫܵܢܹ̈ܐ ܡܫܲܒ݁ܚܵܢܹ̈ܐ ܕܪܲܟܸ݁ܒ݂ܬ݁: ܘܡܢ ܟܸܢܫܹ̈ܐ ܟܠܗܘܿܢ ܕܥܸܠܵܝܹ̈ܐ ܘܲܕ݂ܬܲܚܬܵܝܹ̈ܐ: ܫܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܲܕ݂ܡܵܪܘܼܬ݂ܵܟ݂ ܢܸܣܬ݁ܓܸܕ݂ ܘܢܸܫܬܲܒܲܚ ܘܢܸܬ݂ܝܲܩܲܪ ܘܢܸܬ݂ܪܲܡܪܲܡ ܘܢܸܬ݁ܬܵܘܕܸ݁ܐ ܘܢܸܬ݂ܒܲܪܲܟ݂ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ. 	ܒܲܫܠܝܼܚܵܘ̈ܗܝ ܕܡܲܠܟܵܐ. ܐܹܪܲܡܪܡܵܟ݂ ܡܵܪܝ ܡܲܠܟܵܐ: ܘܐܲܫܠܸܛܬܵܝܗܝ ܒܲܥܒ݂ܵܕ݂ܵܐ ܕܐܝܼܕܲܝ̈ܟ. ܕܨܘܼܪܬܵܐ. ܐܵܡܲܪ ܠܗܘܿܢ ܫܸܡܥܘܿܢ ܟܹܐܦܵܐ ܐܵܙܹܠܢܵܐ ܐܹܨܘܼܕ݂ ܢܘܼܢܹ̈ܐ. ܘܲܣܠܸܩ ܫܸܡܥܘܿܢ ܟܹܐܦܵܐ ܘܢܲܓ݂ܕ݂ܵܗ ܠܲܡܨܝܼܕ݂ܬܵܐ ܠܐܲܪܥܵܐ. ܡܢ ܒܵܬܲܪ ܩܝܵܡܬܵܟ݂ ܡܫܲܒܲܚܬܵܐ. ܬܘܼܒ݂ ܦܲܛܪܘܿܣ ܢܘܼܢܹ̈ܐ ܨܵܐܹܕ݂ ܗ݇ܘܵܐ. ܘܲܠܘܵܬܹܗ ܦܵܪܘܿܩܲܢ ܡܡܲܠܸܠ ܗ݇ܘܲܝܬ݁. ܪܥܝܼ ܠܝܼ ܐܸܡܪܹ̈ܐ ܕܥܵܢܲܝ̈. ܪܥܝܼ ܠܝܼ ܥܸܪ̈ܒܲܝ ܘܲܢܩܵܘ̈ܵܬ݂ܝ. ܕܡܸܛܠܵܬ݂ܗܘܿܢ ܩܕ݂ܵܡ ܦܝܼܠܵܛܘܿܣ ܩܵܡܹܬ݂. ܘܡܸܛܠܵܬ݂ܗܘܿܢ ܪܘܼܩܵܐ ܒܐܲܦܲܝ̈ ܙܠܲܚܘ ܠܝܼ. ܘܡܸܛܠܵܬ݂ܗܘܿܢ ܙܩܝܼܦܵܐ ܕܚܸܣܕ݂ܵܐ ܣܲܝܒܲܪܬ݁. ܘܝܵܘܡܹ̈ܐ ܬܠܵܬܵܐ ܒܩܲܒ݂ܪܵܐ ܕܡܝܼܬܹ̈ܐ ܗܘܹܝܬ݂. ܐܸܢܵܐ ܕܐܝܼܬܲܝ ܦܵܪܘܿܩܵܐ ܘܪܵܚܹܡ ܐ݇ܢܵܫܵܐ. ܬܢܝܼ܀
ܘܫܲܒܲܚ. ܡܢ ܒܵܬܲܪ ܩܝܵܡܬܵܟ݂ ܡܫܲܒܲܚܬܵܐ. ܬܘܼܒ݂ ܦܲܛܪܘܿܣ ܢܘܼܢܹ̈ܐ ܨܵܐܹܕ݂ ܗ݇ܘܵܐ. ܘܲܠܘܵܬܹܗ ܦܵܪܘܿܩܲܢ ܡܡܲܠܸܠ ܗ݇ܘܲܝܬ݁. ܪܥܝܼ ܠܝܼ ܐܸܡܪܹ̈ܐ ܕܥܵܢܲܝ̈. ܪܥܝܼ ܠܝܼ ܥܸܪ̈ܒܲܝ ܘܲܢܩܵܘ̈ܵܬ݂ܝ. ܕܡܸܛܠܵܬ݂ܗܘܿܢ ܩܕ݂ܵܡ ܦܝܼܠܵܛܘܿܣ ܩܵܡܹܬ݂. ܘܡܸܛܠܵܬ݂ܗܘܿܢ ܪܘܼܩܵܐ ܒܐܲܦܲܝ̈ ܙܠܲܚܘ ܠܝܼ. ܘܡܸܛܠܵܬ݂ܗܘܿܢ ܙܩܝܼܦܵܐ ܕܚܸܣܕ݂ܵܐ ܣܲܝܒܲܪܬ݁. ܘܝܵܘܡܹ̈ܐ ܬܠܵܬܵܐ ܒܩܲܒ݂ܪܵܐ ܕܡܝܼܬܹ̈ܐ ܗܘܹܝܬ݂. ܐܸܢܵܐ ܕܐܝܼܬܲܝ ܦܵܪܘܿܩܵܐ ܘܪܵܚܹܡ ܐ݇ܢܵܫܵܐ.

ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܒܵܚܵܐ. ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܢ ܟܠ ܠܸܫܵܢܝܼ̈ܢ: ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܘܗܸܕ݂ܪܵܐ ܘܪܘܼܡܪܵܡܵܐ ܡܢ ܟܠ ܒܸܪ̈ܝܵܢ: ܐܝܼܬ݂ܝܵܐ ܓܢܝܼܙܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܫܘܼܒܵܚܵܐ.
	ܐܹܪܲܡܪܡܵܟ݂ ܡܵܪܝܵܐ ܕܐܲܪܝܼܡܬܵܢܝ* ܘܠܵܐ ܚܲܕܝܼܬ݁ ܒܝܼ ܠܲܒ݂ܥܸܠܕܒ݂ܵܒܲܝ̈܀ ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ. ܐܹܪܲܡܪܡܵܟ݂ ܡܵܪܝܵܐ ܕܐܲܪܝܼܡܬܵܢܝ* ܘܠܵܐ ܚܲܕܝܼܬ݁ ܒܝܼ ܠܲܒ݂ܥܸܠܕܒ݂ܵܒܲܝ̈܀ ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ.

ܘܐܵܘܕܵܘ ܠܕ݂ܘܼܟ݂ܪܵܢܵܐ ܕܩܘܼܕ݂ܫܹܗ* ܡܸܛܠ ܕܲܟ݂ܐܵܬ݂ܵܐ ܒܪܘܼܓ݂ܙܹܗ ܘܚܲܝܹ̈ܐ ܒܨܸܒ݂ܝܵܢܹܗ܀

ܗܦܵܟ݂ܬܵܐ. ܐܲܝܟ݂ ܕܲܠܚܸܘܝܵܐ ܕܲܙܩܲܦ̮ ܡܘܼܫܹܐ ܙܲܩܦܹܗ ܐܵܦ ܠܹܗ: ܕܢܲܐܣܸܐ ܠܐ݇ܢܵܫ̈ܵܐ ܡܢ ܢܘܼ̈ܟ݁ܬܵܬ݂ܵܐ ܕܫܹܐܕܹ̈ܐ ܛܪ̈ܘܼܢܹܐ܀ ܫܹܐܕܹ̈ܐ ܢܲܟ݁ܬ݂ܘܼܗܝ ܠܓܸܢܣܵܐ ܕܐ݇ܢܵܫ̈ܵܐ ܚܲܪܡܵܢܵܐܝܼܬ݂: ܘܲܬ݂ܠܵܐ ܐ݇ܢܵܫܵܐ ܒܪܹܫ ܓܵܓ݂ܘܼܠܬܵܐ ܕܢܲܚܸܐ ܠܐ݇ܢܵܫ̈ܵܐ.
ܬܸܫܒܘܿܚܬܵܐ. ܘܲܒ݂ܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܕܹܝܢ ܐܵܡ̇ܪܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܗܵܕܹܐ ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܡܵܪܝ ܢܲܪܣܲܝ ܡܲܠܦܵܢܵܐ.
ܬܵܘܕ݁ܝܼ ܠܛܵܒ݂ܵܐ ܕܚܲܪܲܪ ܓܸܢܣܲܢ: ܡܢ ܥܲܒ݂ܕ݁ܘܼܬ݂ܵܐ ܕܒ݂ܝܼܫܵܐ ܘܡܵܘܬܵܐ܀ ܫܲܝܸܢ ܥܲܡܲܢ ܟܸܢܫܲܝ̈ ܪܵܘܡܵܐ: ܕܪܲܓ݁ܝܼܙܝܼܢ ܗ݇ܘܵܘ ܡܸܛܠ ܥܵܘܠܲܢ܀ ܒܪܝܼܟ݂ ܚܲܢܵܢܵܐ ܕܟܲܕ݂ ܠܵܐ ܒܥܲܝܢܵܝܗܝ: ܢܦܲܩ ܒܲܒ݂ܥܵܬܲܢ ܘܲܚܕ݂ܝܼ ܒܚܲܝܲܝ̈ܢ܀ ܘܨܵܪ ܕܸܡ̈ܘܵܬ݂ܵܐ ܕܐܲܒ݁ܝܼܕ݂ܘܼܬܲܢ: ܘܲܕ݂ܦܘܼܢܵܝܲܢ ܒܥܸܪܒܵܐ ܕܲܛܥܵܐ܀ ܝܵܪܬܵܐ ܘܲܒ݂ܪܵܐ ܩܪܵܝܗܝ ܠܲܟ݂ܝܵܢܲܢ: ܕܲܛܥܵܐ ܘܲܦܢܵܐ ܘܡܝܼܬ݂ ܘܐܸܬ݂ܢܲܚܲܡ܀ ܘܚܲܕ݁ܝܼ ܠܟܸܢܫܹ̈ܐ ܪ̈ܘܼܚܵܢܵܝܹܐ: ܒܲܬ݂ܝܵܒ݂ܘܼܬܲܢ ܘܲܒ݂ܢܘܼܚܵܡܲܢ܀ ܠܵܐ ܡܸܬ݂ܡܲܠܲܠ ܚܘܼܒܵܐ ܪܲܒܵܐ: ܕܚܵܘܝܼ ܨܹܐܕܲܝܢ ܪܵܚܹ̇ܡ ܓܸܢܣܲܢ܀ ܕܡܸܢܹܗ ܕܓܸܢܣܲܢ ܥܒܲܕ݂ ܡܸܨܥܵܝܵܐ: ܘܪܲܥܝܼ ܠܥܵܠܡܵܐ ܥܲܡ ܪܲܒܘܼܬܹܗ܀ ܪܲܒܵܐ ܗ݇ܝ ܡܸܢܲܢ ܘܡܢ ܟܠ ܒܸܪ̈ܝܵܢ: ܚܕܲܬܵܐ ܕܲܣܥܲܪ ܨܹܝܕ݂ ܐ݇ܢܵܫܘܼܬܲܢ܀ ܕܥܲܒ݂ܕܹ݁ܗ ܠܦܲܓ݂ܪܲܢ ܗܲܝܟܲܠ ܩܘܼܕ݂ܫܵܐ: ܕܲܢܡܲܠܸܐ ܒܹܗ ܣܸܓ݂ܕܲܬ݂ ܟܠܵܐ܀ ܬܵܘ ܐܲܪ̈ܥܵܢܹܐ ܘܲܫܡܲܝܵܢܹ̈ܐ: ܬܗܲܪܘ ܘܐܸܬ݁ܕܲܡܲܪܘ ܒܪܲܒ݁ܘܼܬ݂ ܕܲܪܓ݂ܵܐ܀ ܕܲܡܛܵܐ ܓܸܢܣܲܢ ܠܪܵܘܡܵܐ ܪܲܒܵܐ: ܕܐܲܠܵܗܘܼܬ݂ܵܐ ܕܠܵܐ ܡܸܬ݁ܕܲܪܟܵܐ܀ ܫܡܲܝܵܐ ܘܐܲܪܥܵܐ ܘܟ݂ܠܡܵܐ ܕܲܒ݂ܗܘܿܢ: ܢܵܘܕ݁ܘܿܢ ܥܲܡܲܢ ܠܡܵܘܪܸܒ݂ ܓܸܢܣܲܢ܀ ܕܚܲܕܸ݁ܬ݂ ܨܲܠܡܲܢ ܘܲܥܛܵܐ ܥܵܘܠܲܢ: ܘܲܩܪܵܢ ܒܲܫܡܹܗ ܘܫܲܥܒܸ݁ܕ݂ ܠܲܢ ܟܠ܀ ܫܵܘܹ̇ܐ ܠܫܘܼܒ݂ܚܵܐ ܡܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܗ̇ܘ ܕܐܲܪܝܼܡܲܢ ܠܥܸܠ ܡܢ ܟܠܵܐ܀ ܘܢܹܐܡܲܪ ܟܠܲܢ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ: ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀

(ܘܪܵܡܹܐ ܡܫܲܡܫܵܢܵܐ ܐܘܼܪܵܪܵܐ: ܘܐܵܣܲܪ ܙܘܼܢܵܪܵܐ: ܘܩܵܐܹ̇ܡ ܩܕ݂ܵܡ ܫܟ݂ܝܼܢܬܵܐ ܡܛܲܟ݁ܣܵܐܝܼܬ݂(ܘܐܵܡܲܪ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ: ܒܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܒ݂ܥܹܐܕܹ̈ܐ ܘܲܒ݂ܕ݂ܘܼܟ݂ܪ̈ܵܢܹܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܚܲܕܝܼ ܘܐܲܦ̮ܨܲܚ ܠܐ݇ܢܵܫ̈ܵܐ ܘܡܲܠܲܐܟܹ̈ܐ. ܘܐܲܢܗܲܪܘ ܠܣܵܘܦܹܝ̈ܗ ܕܥܵܡܲܪܬܵܐ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܕܵܨܘ ܥܸܠܵܝܹ̈ܐ. ܘܲܪܘܲܙܘ ܬܲܚܬܵܝܹ̈ܐ. ܘܲܩܥܵܘ ܫܘܼܒ݂ܚܵܐ ܟܠܗܘܿܢ ܥܒ݂ܝܼܕܹ̈ܐ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܪܲܡܪܸܡ ܘܥܲܠܝܼ ܐܝܼܩܵܪܹܗ ܕܲܟ݂ܝܵܢܲܢ. ܘܐܲܩܢܝܼ ܦܘܼܪܩܵܢܵܐ ܠܓܸܢܣܲܢ ܕܒܲܚܛܝܼܬ݂ܵܐ ܐܸܬ݂ܚܲܒܲܠ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܚܲܪܲܪ ܠܲܟ݂ܝܵܢܲܢ ܡܢ ܥܲܒ݂ܕܘܼܬ݂ܵܐ ܕܫܹܐܕܹ̈ܐ ܒܝܼܫܹ̈ܐ. ܘܲܥܒ݂ܵܕܹ̈ܐ ܕܲܚܛܝܼܧܵ. ܘܝܲܗ݇ܒ݂ ܠܘܼܒܵܒ݂ܵܐ ܠܣܵܓ݂ܘܿܕ݂ܵܘ̈ܗܝ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܚܵܒ݂ ܣܵܛܵܢܵܐ. ܘܐܲܠܒ݁ܫܹܗ ܒܸܗܬ݁ܬ݂ܵܐ. ܘܲܗܘܵܐ ܚܸܣܕܵܐ ܗܘ ܘܚܲܝܠܵܘܵܬܹ̈ܗ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܣܠܸܩ ܡܢ ܩܲܒ݂ܪܵܐ ܘܒܲܕܸܩ ܥܲܠ ܣܘܼܠܵܩܹܗ ܕܠܲܫܡܲܝܵܐ. ܘܲܡܠܲܟ݂ ܠܲܢ ܕܥܲܡܹܗ ܢܲܣܩܲܢ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܠܩܲܣܛܘܿܢܵܪܹ̈ܐ ܘܲܠܣܵܦ݁ܪܹ̈ܐ ܐܲܠܒܸ݁ܫ ܟܡܝܼܪܘܼܬ݂ܵܐ. ܘܐܲܩܢܝܼ ܚܲܫܵܐ ܘܟܲܪܝܘܼܬ݂ܵܐ ܘܐܲܚܦܲܪ ܐܲܦܲܝ̈ܗܘܿܢ. ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܡܫܝܼܚܵܐ ܕܒܲܩܝܵܡܬܹ݁ܗ ܐܸܬ݂ܚܙܝܼ ܠܡܲܓ݂ܕܠܵܝܧܵ ܘܲܚܒ݂ܵܪ̈ܵܬ݂ܵܗ ܘܗܸܢܹܝܢ ܣܲܒܲܪ̈ܝ ܠܬܲܠܡܝܼܕ݂ܵܘ̈ܗܝ ܕܒܲܓ݂ܠܝܼܠܵܐ ܢܸܚܙܘܿܢܵܝܗܝ. ܣܒܲܪܬ݂ܵܐ ܡܲܠܝܲܬ݂ ܫܘܼܟܵܢܹ̈ܐ ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܪܘܿܩ ܘܚܲܕܵܐ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ ܒܥܹܐܕ݂ܵܐ ܦܨܝܼܚܵܐ ܕܲܩܝܵܡܬܵܟ݂. ܡܫܝܼܚܵܐ ܡܢܲܚܵܡܢܵܐ ܕܥܲܦܪ̈ܵܢܹܐ. ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
