ܛܲܟ݂ܣܵܐ ܕܬܸܫܡܸܫܬܵܐ
ܕܠܸܠܝܵܐ ܘܕܨܲܦܪܵܐ
 ܒܫܸܡ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕܫܵܐ ܠܥܵܠܡܝܼܢ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܐ ܟܲܣܝܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܐ ܬܡܝܼܗܵܐ ܕܡܵܪܘܼܬ݂ܵܟ݂: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬ݂ܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܒܠܸܠܝܵܐ ܕܹܝܢ ܘܲܡܫܲܪܹܝܢ ܡܢ ܐܲܠܵܗܵܐ ܩܵܡ. ܓ: ܗܘܼܠܵܠܹ̈ܐ. ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܦܒ‐ܩܐ
ܘܥܵܢܹܝܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.
ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.
ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܡܵܪܝܵܐ ܪܲܚܸܡ ܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܗܦܵܟ݂ܬܵܐ ܕܣܘܼܠܵܩܹܗ ܕܡܵܪܲܢ. ܒܫܝܼܛܘܼܬ݂ ܡܸܕ݂ܪܲܢ ܬܸܗܪܹܬ݂ ܣܲܓ݁ܝܼ ܕܲܟ݂ܡܵܐ ܝܼܪܸܒ݂. ܕܟܲܕ݂ ܕܲܚܝܼܚܵܐ ܗ݇ܘ ܩܢܵܐ ܫܘܼܠܛܵܢܵܐ ܘܐܲܡܠܸܟ݂ ܥܲܠ ܟܠ܀ ܒܚܸܨܦܵܐ ܫܝܼܛܵܐ ܦ̮ܵܗܘ ܚܘܼܫܵܒܲܝ̈ ܟܡܵܐ ܐܸܬ݂ܪܲܡܪܲܡ ܕܲܗܘܵܐ ܡܵܐܢܵܐ ܕܚܵܫܲܚ ܣܲܓ݁ܝܼ ܠܲܫܒ݂ܝܼ̈ܚܵܬܵܐ.
( ܬܸܫܡܸܫܬܵܐ ܕܠܸܠܝܵܐ ܐܲܝܟ݂ ܥܝܵܕܵܐ ܕܫܲܗܪܵܐ ).
ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܡܵܘܬ݁ܒ݂ܵܐ ܕܥܹܐܕܹ̈ܐ ܘܲܕ݁ܕ݂ܘܼܟ݂ܪ̈ܵܢܹܐ.
	ܠܵܟ݂ ܓܲܙܵܐ ܕܥܘܼܕ݂ܪ̈ܵܢܹܐ ܘܲܡܥܝܼܢܵܐ ܕܟ݂ܠܗܹܝܢ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܘܝܲܡܵܐ ܫܦܝܼܥܵܐ ܕܲܚܢܵܢܵܐ ܘܕܲܡܪܲܚܡܵܢܘܼܬ݂ܵܐ: ܘܲܬ݂ܗܘܿܡܵܐ ܪܲܒܵܐ ܕܚܘܼܣܵܝܵܐ ܘܲܕ݂ܚܵܘܣܵܢܵܐ: ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ ܐܸܬ݂ܦܲܢ ܡܵܪܝ ܘܚܘܼܣ ܘܪܲܚܸܡ ܥܠܲܝܢ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
             ܡܘܬ݁ܒ݂ܐ. ܒܢܵܘܣܵܐ ܕܪܘܼܚܩܘܼܕ݂ܫܵܐ. ܕܲܢܚܵܘܸܐ ܓܲܢ݇ܒܵܪܘܼܬܹܗ: ܚܵܘܵܐ ܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂ ܘܬ݂ܵܐ ܠܦܘܼܪܩܵܢܲܢ. ܡܫܝܼܚܵܐ ܐܸܨܛܠܸܒ݂ ܘܡܝܼܬ݂ ܘܐܸܬ݂ܩ̄ܒܲܪ. ܘܩܵܡ ܘܐܸܬ݂ܢܲܚܲܡ ܡܲܠܟܵܐ ܪܲܒܵܐ ܕܬܸܫܒ݁ܘܿܚܬܵܐ. ܘܲܣܠܸܩ ܠܲܫܡܲܝܵܐ ܘܐܲܡܠܸܟ݂ ܠܥܸܠ ܡܢ ܟܠ. ܘܲܥܬ݂ܝܼܕ݂ ܕܢܹܐܬܸܐ ܕܲܢܕ݂ܘܼܢ ܠܚܲܝܹ̈ܐ ܐܵܦ ܠܡܝܼܬܹ̈ܐ: ܬܢܝܼ. ܣܠܸܩ ܐܲܠܵܗܵܐ ܒܫܘܼܒ݂ܚܵܐ ܡܵܪܝܵܐ ܒܩܵܠܵܐ ܕܩܲܪܢܵܐ. ܒܲܥܢܵܢܹ̈ܐ ܕܢܘܼܗܪܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܕܪܘܼܚܵܐ. ܐܸܣܬܲܠܲܩ ܡܵܪܲܢ ܝܼܫܘܿܥ ܠܲܫܡܲܝܵܐ. ܘܲܡܠܲܟ݂ ܘܐܸܫܬܵܘܕ݁ܝܼ ܪܘܼܚܵܐ ܕܲܡܫܲܕܲܪ. ܠܬܲܠܡܝܼܕ݂ܵܘ̈ܜ ܫܠܝܼܚܹ̈ܐ ܘܒܹܗܘ ܡܢܲܨܲܚ ܠܗܘܿܢ. ܒܲܪܸܟ݂ܘ ܠܡܵܪܝܵܐ ܡܲܠܲܐܟ݂ܵܘ̈ܗܝ. ܟܸܢܫܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ ܟܪ̈ܘܿܒܹܐ ܘܲܣܪ̈ܵܦܹܐ. ܒܩܵܠܹ̈ܐ ܕܬ݂ܵܘܕ݁ܝܼܬܵܐ ܙܵܡܪܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ. ܒܪܝܼܟ݂ ܛܵܒ݂ܵܐ ܕܫܲܝܸܢ ܠܥܝܼܪܹ̈ܐ ܘܲܠܐ݇ܢܵܫ̈ܵܐ. ܘܐܲܫܘܝܼ ܠܡܵܝܘܿܬܹ̈ܐ ܕܲܢܩܲܕ݁ܫܘܼܢ ܠܲܫܡܹܗ. ܙܥܘܿܪܹ̈ܐ ܥܲܡ ܪܵܘܪ̈ܒܹܐ. ܒܕܸܚܠܵܐ ܘܲܪܬܹܝܬܵܐ ܩܵܡܘ ܗ݇ܘܵܘ ܬܲܠܡܝܼܕܹ̈ܐ. ܒܥܸܕܵܢܵܐ ܕܲܣܠܸܩ ܡܵܪܗܘܿܢ ܠܲܫܡܲܝܵܐ. ܘܟܸܢܫܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ ܘܓ݂ܘܼܕܹ̈ܐ ܕܲܣܪ̈ܵܦܹܐ. ܙܵܡܪܝܼܢ ܠܹܗ ܫܘܼܒ݂ܚܵܐ ܡܢ ܝܲܡܝܼܢ ܐܲܒ݂ܵܐ. ܡܢ ܦܘܼܡܵܐ ܕܲܥܠܲܝܡܹ̈ܐ ܘܕܲܛܠܵܝܹ̈ܐ. ܠܵܟ݂ ܡܵܪܝ ܬܸܫܒ݁ܘܿܚܬܵܐ ܡܲܣ̈ܩܵܢ ܒܸܪ̈ܝܵܬܵܐ. ܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ ܕܪܲܡܪܸܡܬ݁ ܠܲܟ݂ܝܵܢܲܢ. ܘܚܲܪܲܪܬ݁ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܠܓܸܢܣܵܐ ܕܡܵܝܘܿܬܹ̈ܐ. ܘܲܓ݂ܠܲܝܬ݁ ܒܲܚܢܵܢܵܟ݂ ܐ݇ܪ̈ܵܙܲܝܟ ܠܬܲܠܡܝܼܕܲܝ̈ܟ. ܫܘܼܚܠܵܦܵܐ. ܒܲܒ݂ܝܲܕ݂ ܫܠܵܡܵܐ. ܫܡܲܥܘ ܗܵܕܹܐ ܟܠܟ݂ܘܿܢ ܥܲܡܡܹ̈ܐ. ܥܲܠ ܛܘܼܪ ܙܲܝ̈ܬܹ݁ܐ ܨܲܠܝܼ ܗ݇ܘܵܐ ܡܵܪܲܢ ܩܕ݂ܵܡ ܡܲܣܩܹܗ. ܘܦܲܩܸܕ݂ ܐܸܢܘܿܢ ܠܬܲܠܡܝܼܕ݂ܵܘ̈ܗܝ ܕܒ݂ܐܘܿܪܸܫܠܸܡ ܢܟܲܬ݁ܪܘܼܢ. ܥܕܲܡܵܐ ܕܢܸܠܒ݁ܫܘܼܢ ܚܲܝܠܵܐ ܡܢ ܪܵܘܡܵܐ. ܘܲܢܩܲܒ݁ܠܘܼܢܵܝܗܝ ܠܢܘܼܗܪܵܐ ܕܝܼܕܲܥܬܹܗ. ܘܢܸܦܩܘܼܢ ܘܢܲܟ݂ܪܙܘܼܢ ܒܹܝܬ݂ ܥܲܡܡܹ̈ܐ ܬܸܕ݂ܡܪ̈ܵܬܹܗ ܕܡܵܪܝܵܐ. ܒܲܓ݂ܒܲܪ̈ܘܵܬ݂ܵܐ ܕܣܵܥܲܪ ܒܐܝܼܕܲܝ̈ܗܘܿܢ. ܘܲܢܫܲܒ݁ܚܘܼܢ ܠܲܫܡܵܟ݂ ܠܥܵܠܡܝܼܢ. ܫܘܼܒ݂ܚܵܐ ܠܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܐܲܠܵܗܵܐ ܕܟ݂ܠ ܒܪܵܐ. ܕܲܬ݂ܪܹܝܢ ܥܵܠܡܹ̈ܐ ܛܲܝܸܒ݂ ܠܲܢ ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܚܸܟ݂ܡܬܹܗ. ܗܵܢܵܐ ܕܗܵܪܟܵܐ ܐܲܝܟ݂ ܕܲܠܕ݂ܘܼܪܵܫܲܢ. ܘܲܠܗܵܘ ܕܲܥܬ݂ܝܼܕ݂ ܠܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ. ܘܒܲܡܫܝܼܚܵܐ ܕܚܲܫ ܥܲܠ ܐܲܦܲܝ̈ ܐܘܼܪܚܵܐ ܐܸܬ݁ܕܲܪܫܲܬ݀. ܕܩܵܡ ܡܢ ܩܲܒ݂ܪܵܐ ܘܲܣܠܸܩ ܠܲܫܡܲܝܵܐ. ܪܟܸܒ݂ ܥܲܠ ܟܪ̈ܘܿܒܹܐ ܘܲܦܪܲܚ. ܡܫܝܼܚܵܐ ܡܲܠܟܵܐ ܕܬܸܫܒ݁ܘܿܚܬܵܐ ܒܫܘܼܒ݂ܚܵܐ ܐܸܬ݂ܥܲܠܝܼ. ܠܲܩܕ݂ܘܿܫ ܩܘܼܕ݂ܫܹ̈ܐ ܥܸܠܵܝܹ̈ܐ ܠܪܵܘܡܵܐ ܕܲܫܡܲܝܵܐ. ܘܝܼܬܸܒ݂ ܬܲܡܢ ܡܢ ܝܲܡܝܼܢ ܐܲܒ݂ܵܐ. ܒ݂ܵܥܹܐ ܚܠܵܦܲܝܢ ܪ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ. ܢܝܲܩܲܪ ܟܠܲܢ ܣܘܼܠܵܩܹܗ ܒܚܘܼܒܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ. ܕܲܒ݂ܝܘܿܡ ܡܹܐܬ݂ܝܹܗ ܢܐܹܪܥܝܼܘܗܝ ܒܚܲܕ݂ܘܼܬ݂ܵܐ. ܡܵܐ ܛܵܒ݂ ܘܡܵܐ ܫܲܦܝܼܪ. ܡܵܐ ܫܲܦܝܼܪ ܠܵܗ ܠܲܡܟ݂ܝܼܪܬܵܟ݂ ܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ. ܟܲܕ݂ ܡܙܲܝܚܵܐ ܠܝܘܿܡ ܣܘܼܠܵܩܵܟ݂ ܕܲܠܘܵܬ݂ ܫܵܠܘܿܚܵܟ݂. ܘܙܵܡܪܝܼܢ ܝܲܠܕܹܝ̈ܗ ܥܲܡ ܟܸܢܫܲܝ̈ ܪܵܘܡܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܲܫܡܵܟ݂ ܕܐܸܬ݂ܓ݁ܠܝܼܬ݁ ܠܦܘܼܪܩܵܢܲܢ. ܘܚܵܕܹ݁ܝܢ ܥܲܡܹܗ ܥܸܠܵܝܹ̈ܐ ܕܙܲܝܚܘܼܗ ܠܡܲܣܲܩܬܵܟ݂. ܟܲܕ݂ ܐܸܣܬܲܠܲܩܬ݁ ܥܲܠ ܥܢܵܢܹ̈ܐ ܕܢܘܼܗܪܵܐ. ܩܵܠܵܐ. ܒܣܵܗܕܹ̈ܐ ܒܪ̈ܝܼܟܹܐ. ܪܟܸܒ݂ ܥܲܠ ܟܪ̈ܘܿܒܹܐ ܘܲܦܪܲܚ. ܠܥܸܠ ܡܢ ܟܠܗܘܿܢ ܪܹ̈ܫܹܐ ܘܫܘܼܠܛܵܢܹ̈ܐ ܘܟܸܢܫܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܣܠܸܩ ܘܝܼܬܸ݁ܒ݂ ܡܢ ܝܲܡܝܼܢܵܐ. ܝܼܫܘܿܥ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܵܐ. ܘܲܠܓܸܢܣܵܐ ܕܲܒ݂ܢܲܝ̈ܢܵܫܵܐ. ܝܲܗ݇ܒ݂ ܚܲܝܹ̈ܐ ܠܵܐ ܡܵܝܘܿܬܹ̈ܐ. ܘܦܘܼܪܩܵܢܵܐ ܕܲܠܥܵܠܲܡ ܩܲܝܵܡ. ܬܵܘ ܒܢܲܝ̈ܵܐ ܫܘܼܡܥܘܼܢܝ. ܓܲܒ݂ܪܹ̈ܐ ܓܠܝܼܠܵܝܹ̈ܐ ܐܸܡܲܪܘ ܪ̈ܘܼܚܵܢܹܐ ܠܟܸܢܫܹ̈ܐ ܕܬܲܠܡܝܼܕܹ̈ܐ. ܠܡܵܢ ܚܵܝܪܝܼܢ ܐܢ݇ܬ݁ܘܿܢ ܒܲܫܡܲܝܵܐ. ܘܟܲܪܝܵܐ ܠܟ݂ܘܿܢ ܥܲܠ ܡܵܐ ܕܲܗܘܵܐ. ܗܵܢܵܐ ܝܼܫܘܿܥ ܕܐܸܣܬܲܠܲܩ. ܥܬ݂ܝܼܕ݂ܘܼ ܕܢܹܐܬܸܐ ܒܬܸܫܒ݁ܘܿܚܬܵܐ. ܘܢܲܣܸܩܟ݂ܘܿܢ ܥܲܡܹܗ ܠܲܫܡܲܝܵܐ. ܫܡܲܥܘ ܗܵܕܹܐ ܟܠܟ݂ܘܿܢ ܥܲܡܡܹ̈ܐ. ܬܪܹܝܢ ܕܹܝܢ ܡܲܠܲܐܟܹ̈ܐ ܒܲܠܒ݂ܘܼܫܹ̈ܐ ܚܸܘܵܪܹ̈ܐ ܢܚܸܬ݂ܘ ܠܘܵܬ݂ ܬܲܠܡܝܼܕܹ̈ܐ. ܘܒܲܝܲܐܘ ܐܸܢܘܿܢ ܟܲܕ݂ ܐܵܡܪܝܼܢ. ܕܗܵܢܵܐ ܝܼܫܘܿܥ ܕܐܸܣܬܲܠܲܩ. ܗܵܟܲܢܵܐ ܐܵܬܹ݁ܐ ܒܫܘܼܒ݂ܚܵܐ. ܘܚܲܝܠܵܐ ܪܲܒܵܐ ܕܡܲܠܲܐܟ݂ܵܘ̈ܗܝ. ܘܡܸܬ݂ܚܵܘܸܐ ܠܥܵܠܡܵܐ ܡܢ ܣܵܟܹܗ. ܩܲܕ݁ܝܼܫܘܼ ܘܲܕ݂ܚܝܼܠ ܫܡܹܗ. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܵܥܹܝܢ ܗ݇ܘܵܘ ܟܸܢܫܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܒܝܵܘܡܵܐ ܕܲܣܠܸܩ ܦܵܪܘܿܩܲܢ. ܠܲܩܕ݂ܘܿܫ ܩܘܼܕ݂ܫܹ̈ܐ ܥܸܠܵܝܹ̈ܐ. ܘܲܟ݂ܢܲܫܘ ܓܘܼܕܹ̈ܐ ܥܲܡ ܣܸܕ݂ܪܹ̈ܐ. ܘܩܵܡܘ ܥܲܠ ܬܲܪܥܹܗ ܕܲܪܩܝܼܥܵܐ. ܟܲܕ݂ ܙܵܡܪܝܼܢ ܫܘܼܒ݂ܚܵܐ ܒܩܵܠܲܝ̈ܗܘܿܢ. ܩܵܠܵܐ. ܒܬ݂ܵܘ ܢܸܬ݁ܕܲܡܲܪ. ܕܐܲܢ݇ܬ݁ܘܼ ܟܘܼܡܪܵܐ ܠܥܵܠܲܡ. ܪܲܒܵܐ ܕܟ݂ܘܼܡܪܹ̈ܐ ܦܵܪܘܿܩܲܢ. ܫܲܡܠܝܼ ܘܲܓ݂ܡܲܪ ܚܘܼܕܵܬܲܢ. ܘܲܫܪܵܝܗܝ ܘܒܲܟ݂ܠܹܗ ܠܡܵܘܬܵܐ ܝܲܥܢܵܐ. ܕܒܲܠܥܹܗ ܠܓܸܢܣܲܢ ܡܢ ܠܘܼܩܕ݂ܵܡ. ܘܲܣܠܸܩ ܒܫܘܼܒ݂ܚܵܐ ܠܗܲܝܟܲܠ ܡܲܠܟ݁ܘܼܬܹܗ. ܘܐܲܣܩܲܢ ܘܐܵܘܬ݁ܒܲܢ ܥܲܡܹܗ ܒܲܫܡܲܝܵܐ. ܐܸܬ݁ܬ݁ܪܝܼܡܘ ܬܲܪ̈ܥܹܐ ܕܪܵܘܡܵܐ. ܕܢܸܥܘܿܠ ܡܲܠܟܵܐ ܕܐܝܼܩܵܪܹ̈ܐ. ܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ ܡܵܪܵܗ ܕܬܸܫܒ݁ܘܿܚܬܵܐ. ܚܲܕܵܐ ܡܵܪܝ ܠܥܹܕ݁ܬܵܐ ܘܲܠܝܲܠܕܹܝ̈ܗ. ܘܫܵܘܙܸܒܹܝܗ ܡܢ ܒܝܼܫܵܐ ܘܚܲܝܠܹܗ. ܘܚܘܼܣ ܥܠܹܝܗ. ܡܫܲܡܫܵܢܵܘ̈ܗܝ ܕܥܵܒ݂ܕܝܼܢ ܨܸܒ݂ܝܵܢܹܗ. ܚܵܕܹ݁ܝܢ ܟܸܢܫܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܘܪ̈ܵܘܙܵܢ ܓܘܼܕܹ̈ܐ ܕܬܲܚܬܵܝܹ̈ܐ. ܒܝܵܘܡܵܐ ܪܲܒܵܐ ܕܣܘܼܠܵܩܵܟ݂. ܡܲܠܟܵܐ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܕܲܚܙܵܘ ܕܐܲܡܠܸܟ݂ ܫܲܝܢܵܐ ܘܲܫܠܵܡܵܐ. ܒܪܵܘܡܵܐ ܘܥܘܼܡܩܵܐ ܐܲܟܲܚ݇ܕ݂ ܫܲܘܝܵܐܝܼܬ݂. ܘܲܩܢܵܘ ܣܲܒ݂ܪܵܐ ܘܦܘܼܪܩܵܢܵܐ. ܓܸܢܣܵܐ ܟܠܹܗ ܕܡܵܝܘܿܬܹ̈ܐ. ܒܝܲܕ݂ ܣܘܼܠܵܩܵܟ݂ ܡܵܪܲܢ ܕܠܲܫܡܲܝܵܐ. ܕܲܚܙܵܘ ܡܸܟܹ݁ܝܠ ܕܐܸܫܬ݁ܪܝܼ ܡܵܘܬܵܐ. ܘܲܕ݂ܢܲܚܘ ܠܗܘܿܢ ܚܲܝܹ̈ܐ ܫܲܪܝܼܪܹ̈ܐ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܫܘܼܚܠܵܦܵܐ. ܒܬ݂ܵܘ ܒܲܝܬܵܝܹ̈ܐ. ܒܲܕ݂ܡܘܼܬܹܗ ܕܡܲܠܟ݁ܝܼܙܕܸܩ. ܪܲܒܵܐ ܕܟ݂ܘܼܡܪܹ̈ܐ ܦܵܪܘܿܩܲܢ. ܫܲܡܠܝܼ ܛܲܟ݂ܣܹܗ ܠܘܵܬ݂ ܬܲܠܡܝܼܕ݂ܵܘ̈ܗܝ. ܘܫܲܪܝܼ ܕܢܸܪܕܸ݁ܐ ܒܐܘܼܪܚܵܐ ܚܕܲܬܵܐ ܠܐܲܦܲܝ̈ ܪܵܘܡܵܐ ܠܲܩܕ݂ܘܿܫ ܩܘܼܕ݂ܫܹ̈ܐ: ܘܲܟ݂ܢܲܫܘ ܪ̈ܘܼܚܵܢܹܐ ܘܦܲܓ݂ܪ̈ܵܢܹܐ ܠܲܚܙܵܬ݂ܵܐ ܕܠܵܐ ܒܲܥܝܵܕ݂ܵܐ. ܟܲܕ݂ ܚܵܙܹܩ ܘܲܪܗܝܼܒ݂ ܕܢܸܣܲܒ݂ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܡܛܲܝܒ݂ܵܐ ܩܕ݂ܵܡ ܕܵܪܹ̈ܐ. ܪܸܡܙܵܐ ܐܲܠܵܗܵܝܵܐ ܒܥܲܪ̈ܩܲܝ ܢܘܼܪܵܐ. ܡܲܪܟܲܒ݂ܬܵܐ ܟܕܲܢ ܠܐܝܼܩܵܪܹܗ܀ ܥܢܵܢܹ̈ܐ ܕܢܘܼܗܪܵܐ ܟܸܬܵܢܹ̈ܐ ܦܪܲܣ ܥܲܠ ܐܵܐܲܪ. ܢܲܚܬܵܐ ܕܡܲܝ̈ܵܐ ܦܬܲܚ ܬܲܪ̈ܥܹܐ ܥܝܼܪܹ̈ܐ ܨܹܝܕ݂ ܬܲܠܡܝܼܕܹ̈ܐ. ܟܲܕ݂ ܐܸܣܬܲܠܲܩ ܡܲܠܟܵܐ ܡܫܝܼܚܵܐ ܠܲܩܕ݂ܘܿܫ ܩܘܼܕ݂ܫܹ̈ܐ: ܘܐܲܠܸܦ̮ ܒܣܘܼܠܵܩܹܗ ܕܗܵܟܲܢܵܐ ܥܬ݂ܝܼܕܘܼ ܕܢܹܐܬܸܐ. ܘܚܵܘܝܼ ܕܐ݇ܢܵܫܘܼܬܹܗ ܠܵܐ ܡܸܫܬܲܪܝܵܐ ܒܲܫܡܲܝܵܐ. ܘܒܲܕܸ݁ܩ ܕܲܒ݂ܦܲܓ݂ܪܵܐ ܐܵܬܹ݁ܐ ܒܚܲܪܬ݂ܵܐ. ܐܲܟ݂ܡܵܐ ܕܲܣܠܸܩ ܠܲܫܡܲܝܵܐ܀ ܗܦܘܿܟ݂ܘ ܠܐܘܿܪܸܫܠܸܡ ܐܲܝܟܲܢܵܐ ܕܐܸܬ݂ܦܲܩܲܕ݂ܬ݁ܘܿܢ. ܘܩܵܘܵܘ ܥܸܣܪܵܐ ܝܵܘ̈ܡܵܬܵܐ. ܥܕܲܡܵܐ ܕܢܹܐܬܸܐ ܫܘܼܘܕ݂ܵܝܹܗ. ܕܦܵܪܲܩܠܹܝܛܵܐ ܡܫܲܕܲܪ ܠܟ݂ܘܿܢ. ܘܡܢ ܡܵܘܗܲܒ݂ܬܹܗ ܡܵܠܹܐ ܠܟ݂ܘܿܢ. ܘܲܣܓܸܕ݂ܘ ܘܲܗܦܲܟ݂ܘ ܡܲܣܩܹܗ. ܕܗܲܒ݂ ܠܗܘܿܢ ܐܲܒ݂ܵܐ ܕܢܸܬ݂ܚܲܕܬ݂ܘܼܢ. ܥܲܡܵܟ݂ ܒܫܘܼܒ݂ܚܵܐ ܘܚܲܕܵܐ ܠܗܘܿܢ. ܫܘܼܚܠܵܦܵܐ. ܒܲܒ݂ܕ݂ܘܼܡܝܵܐ ܕܥܝܼܪܹ̈ܐ. ܒܟ݂ܠܵܗ ܐܲܪܥܵܐ ܢܸܦܩܲܬ݀ ܣܒܲܪܬ݂ܗܘܿܢ. ܫܠܝܼܚܹ̈ܐ ܩܲܠܣܘܼܝ ܠܲܒ݂ܪܵܐ ܒܣܘܼܠܵܩܹܗ. ܥܢܵܢܵܐ ܩܲܒܸ݁ܠܬܹܗ ܫܡܲܝܵܐ ܓܢܘܿܢܵܐ ܩܛܲܪ ܠܹܗ. ܒܪܝܼܟ݂ ܡܲܠܟܵܐ ܕܲܣܠܸܩ ܘܝܼܬܸܒ݂. ܡܢ ܝܲܡܝܼܢܵܐ ܕܫܵܠܘܿܚܹܗ. ܬܵܘ ܢܕ݂ܘܼ ܘܢܸܚܕܸ݁ܐ ܒܹܗ. ܒܗܵܢܵܐ ܝܵܘܡܵܐ ܫܒ݂ܝܼܚܵܐ ܕܣܘܼܠܵܩܹܗ. ܕܝܼܫܘܿܥ ܡܵܪܲܢ ܘܦܵܪܘܿܩܲܢ ܠܘܵܬ݂ ܐܲܒ݂ܵܐ ܫܵܠܘܿܚܹܗ. ܢܕ݂ܘܼܨܘܼܢ ܐܲܟܲܚ݇ܕ݂ ܥܸܠܵܝܹ̈ܐ. ܥܲܡ ܬܲܚܬܵܝܹ̈ܐ ܒܬܸܫܒ݁ܘܿܚܬܵܐ. ܫܘܼܚܠܵܦܵܐ. ܒܙܲܝܢܵܐ ܕܪܘܼܚܵܐ. ܢܸܚܕ݁ܘܿܢ ܫܡܲܝܵܐ ܘܲܬ݁ܕ݂ܘܼܨ ܐܲܪܥܵܐ. ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܕܲܟ݂ܡܝܼܪܝܼܢ ܗ݇ܘܵܘ. ܩܢܵܘ ܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܪܘܵܙܵܐ. ܒܝܼܫܘܿܥ: ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܕܚܲܫ ܒܲܙܩܝܼܦܵܐ. ܘܩܲܛܠܹܗ ܠܡܵܘܬܵܐ ܒܝܲܕ݂ ܡܵܘܬܹ݁ܗ܀ ܘܩܵܡ ܡܢ ܩܲܒ݂ܪܵܐ ܒܙܵܟ݂ܘܼܬ݂ܵܐ ܕܠܵܐ ܡܸܬ݂ܡܲܠܠܵܐ. ܘܲܣܠܸܩ ܘܝܼܬܸܒ݂ ܒܲܫܡܲܝܵܐ. ܠܥܸܠ ܡܢ ܐܲܪ̈ܟ݁ܘܿܣ ܘܫܘܼܠܛܵܢܹ̈ܐ܀ ܘܡܸܛܠ ܗܵܕܹܐ ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܫܲܘܝܵܐܝܼܬ݂. ܩܵܥܹܝܢ ܘܐܵܡܪܝܼܢ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ. ܩܲܕ݁ܝܼܫ ܐܲܒ݂ܵܐ ܡܬ݂ܘܿܡܵܝܵܐ. ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܩܲܕ݁ܝܼܫܵܐ܀ ܡܫܝܼܚܵܐ: ܕܲܣܠܸܩ ܒܫܘܼܒ݂ܚܵܐ ܠܘܵܬ݂ ܐܲܒ݂ܘܼܗܝ. ܐܲܫܘܵܐ ܠܟ݂ܠܲܢ ܠܚܲܝܹ̈ܐ ܚܲܕ݂ܬܹ̈ܐ. ܘܕܢܸܙܡܲܪ ܫܘܼܒ݂ܚܵܐ ܒܡܲܠܟ݁ܘܼܬ݂ܵܟ݂ ܘܪܲܚܸܡ ܥܠܲܝܢ. ܣܠܸܩ ܐܲܠܵܗܵܐ ܒܫܘܼܒ݂ܚܵܐ ܡܵܪܝܵܐ ܒܩܵܠܵܐ ܕܩܲܪܢܵܐ. ܒܫܘܼܒ݂ܚܵܐ ܪܲܒܵܐ ܐܸܬ݂ܥܲܠܝܼ. ܠܲܫܡܲܝ̈ ܫܡܲܝܵܐ ܦܵܪܘܿܩܲܢ. ܘܝܼܪܸܬ݂: ܕܲܪܓ݂ܵܐ ܪܵܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܐ. ܘܡܵܘܬ݁ܒ݂ܵܐ ܫܒ݂ܝܼܚܵܐ ܕܥܲܡ ܐܲܒ݂ܵܐ܀ ܬܵܘ ܢܵܘܕܸ݁ܐ ܠܹܗ ܕܪܲܡܪܸܡ ܠܓܸܢܣܲܢ ܒܣܘܼܠܵܩܹܗ. ܘܪܲܥܝܼ ܠܲܢ ܥܲܡ ܫܵܠܘܿܚܹܗ. ܕܛܲܝܸܒ݂ ܐܵܘܵܢܹ̈ܐ ܨܹܐܕ݂ܵܘ̈ܗܝ. ܠܟ݂ܠܗܘܿܢ ܓܵܡܪ̈ܲܝ ܨܸܒ݂ܝܵܢܹܗ܀ ܡܫܝܼܚܵܐ: ܕܫܲܡܠܝܼ ܚܘܼܒܹ݁ܗ ܕܲܠܘܵܬܲܢ. ܘܲܣܠܸܩ ܒܫܘܼܒ݂ܚܵܐ ܠܲܫܡܲܝܵܐ ܘܪܲܚܸܡ ܥܠܲܝܢ. ܫܘܼܚܠܵܦܵܐ. ܒܨܵܒܹ݁ܐ ܒܚܲܝܹ̈ܐ. ܐܹܒܲܪܸܟ݂ ܠܡܵܪܝܵܐ ܒܟ݂ܠܙܒܲܢ. ܒܪܝܼܟ݂ ܡܲܠܟܵܐ ܕܲܣܠܸܩ ܘܚܲܕ݁ܝܼ ܒܣܘܼܠܵܩܹܗ. ܠܐܲܪ̈ܒܲܥܦܸܢܝܵܬܵܐ ܘܬܸܓ݂ܡܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ: ܘܥܝܼܪܹ̈ܐ ܥܲܡ ܐ݇ܢܵܫ̈ܵܐ ܒܹܗܘ ܐܸܫܬܲܝܲܢܘ. ܘܥܲܡܡܹ̈ܐ ܘܐܸܡ̈ܘܵܬ݂ܵܐ ܠܹܗܘ ܗܵܐ ܡܝܲܩܪܝܼܢ: ܟܲܕ݂ ܩܵܥܹܝܢ ܒܚܘܼܒܵܐ ܒܪܝܼܟ݂ ܗ݇ܘ ܣܘܼܠܵܩܵܟ݂ ܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ. ܕܲܣܠܸܩܬ݁ ܒܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܲܣܸܩܬܵܢ ܥܲܡܵܟ݂ ܒܫܘܼܒ݂ܚܵܐ ܠܘܵܬ݂ ܐܲܒ݂ܘܼܟ݂: ܘܠܵܟ݂ ܘܵܠܹܐ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂. ܕܐܲܢ݇ܬ݁ܘܼ ܦܪܲܩܬܵܢ ܡܢ ܣܵܢܐܲܝ̈ܢ ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ. ܚܲܕ݂ܘܼܬ݂ܵܐ ܒܟ݂ܠܵܗ ܐܲܪܥܵܐ. ܒܝܵܘܡܵܐ ܕܣܘܼܠܵܩܵܟ݂ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܒܗܸܕ݂ܪܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܪܵܘܡܵܐ ܕܲܫܡܲܝܵܐ: ܟܸܢܫܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ ܟܪ̈ܘܿܒܹܐ ܘܲܣܪ̈ܵܦܹܐ. ܒܕܸܚܠܵܐ ܘܒܲܪܬܹܝܬܵܐ ܡܩܲܕ݁ܫܝܼܢ ܠܐܝܼܩܵܪܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁ ܡܲܠܟܵܐ ܓܲܐܝܵܐ ܕܠܵܐ ܥܵܒܲܪ. ܕܲܦܢܵܐ ܒܬܸܫܒ݁ܘܿܚܬܵܐ ܠܟ݂ܘܼܪܣܝܵܐ ܫܒ݂ܝܼܚܵܐ ܕܡܢ ܝܲܡܝܼܢ ܐܲܒ݂ܵܐ: ܒܪܘܿܟ݂ܘ ܐܲܚܲܝ̈ ܘܲܣܓ݂ܘܿܕ݂ܘ ܠܹܗ. ܠܗܵܢܵܐ ܡܲܠܟܵܐ ܫܒ݂ܝܼܚ ܒܐܝܼܬ݂ܘܼܬܹܗ ܠܹܗ ܫܘܼܒ݂ܚܵܐ. ܩܵܠܵܐ ܒܪܝܼܟ݂ ܚܲܝܠܵܐ. ܒܲܪܸܟ݂ ܐܸܢܘܿܢ ܘܲܣܓ݂ܝܼܘ ܛܵܒ݂. ܥܲܠ ܛܘܼܪ ܙܲܝ̈ܬܹ݁ܐ ܒܲܪܸܟ݂ ܡܵܪܲܢ ܠܬܲܠܡܝܼܕ݂ܵܘ̈ܗܝ ܟܲܕ݂ ܐܸܣܬܲܠܲܩ. ܘܲܥܢܵܢܵܐ ܕܢܘܼܗܪܵܐ ܛܸܥܢܲܬ݀ ܦܲܓ݂ܪܹܗ ܙܲܗܝܵܐ ܡܢ ܨܹܐܕܲܝܗܘܿܢ. ܘܲܣܠܸܩ ܘܝܼܬܸܒ݂ ܡܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܵܐ ܕܫܲܠܚܹܗ. ܒܪܝܼܟ݂ ܕܲܒ݂ܚܘܼܒܹ݁ܗ. ܐܵܘܬ݁ܒܲܢ ܥܲܡܹܗ. ܒܲܩܕ݂ܘܿܫ ܩܘܼܕ݂ܫܹ̈ܐ ܥܸܠܵܝܹ̈ܐ. ܢܸܚܕ݁ܘܿܢ ܫܡܲܝ̈ܵܐ ܘܲܬ݁ܕ݂ܘܼܨ ܐܲܪܥܵܐ. ܒܝܘܿܡ ܣܘܼܠܵܩܵܟ݂ ܕܡܢ ܐܲܪܥܵܐ ܠܪܵܘܡܵܐ ܕܲܠܥܸܠ ܡܫܝܼܚܵܐ ܡܲܠܟܲܢ. ܩܢܵܘ ܚܘܼܕܵܬ݂ܵܐ ܥܸܠܵܝܹ̈ܐ ܘܓܸܢܣܵܐ ܕܐ݇ܢܵܫ̈ܵܐ ܠܬܲܚܬ݁ ܥܲܠ ܐܲܪܥܵܐ. ܕܐܸܬ݂ܦܬܲܚܘ ܬܲܪ̈ܥܲܝ ܪܵܘܡܵܐ ܩܕ݂ܵܡ ܪܲܒ݁ܘܼܬ݂ܵܟ݂. ܕܡܢ ܥܵܠܲܡ ܐܲܚܝܼܕܝܼܢ ܗ݇ܘܵܘ. ܘܐܲܡܠܸܟ݂ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬܵܐ. ܫܘܼܚܠܵܦܵܐ. ܒܡܲܕ݂ܢܵܚܲܝ̈ ܨܲܦܪܵܐ. ܫܡܲܥܘ ܗܵܕܹܐ ܟܠܟ݂ܘܿܢ ܥܲܡܡܹ̈ܐ. ܠܒܲܪ ܡܢ ܒܹܝܬ݂ ܥܲܢܝܵܐ ܢܦܲܩ ܡܵܪܲܢ ܥܲܡ ܬܲܠܡܝܼܕ݂ܵܘ̈ܗܝ. ܠܛܘܼܪܵܐ ܕܙܲܝ̈ܬܹ݁ܐ ܒܝܵܘܡܵܐ ܕܣܘܼܠܵܩܹܗ. ܘܐܲܪܝܼܡ ܗ݇ܘܵܐ ܥܠܲܝܗܘܿܢ ܐܝܼܕ݂ܵܘ̈ܗܝ ܩܲܕ݂ܝܼ̈ܫܵܬܵܐ. ܘܐܸܬ݂ܦܪܸܫ ܡܸܢܗܘܿܢ ܘܲܣܠܸܩ ܠܲܫܡܲܝܵܐ. ܢܸܚܕ݁ܘܿܢ ܙܲܕ݁ܝܼܩܹ̈ܐ ܒܡܵܪܝܵܐ ܘܢܸܬ݁ܬܲܟ݂ܠܘܼܢ ܥܠܵܘܗܝ. ܩܵܘܵܘ ܒܐܘܿܪܸܫܠܸܡ ܐܸܡܲܪ ܡܵܪܲܢ ܠܬܲܠܡܝܼܕ݂ܵܘ̈ܗܝ. ܥܕܲܡܵܐ ܕܬܸܠܒ݁ܫܘܼܢ ܚܲܝܠܵܐ ܡܢ ܪܵܘܡܵܐ. ܡܫܲܕܲܪ ܐ݇ܢܵܐ ܪܘܼܚ ܩܘܼܕ݂ܫܵܐ ܦܵܪܲܩܠܹܝܛܵܐ. ܕܢܸܗܘܘܿܢ ܥܲܡܟ݂ܘܿܢ ܥܕܲܡܵܐ ܠܫܘܼܠܵܡܵܐ. ܣܠܸܩ ܐܲܠܵܗܵܐ ܒܫܘܼܒ݂ܚܵܐ ܡܵܪܝܵܐ ܒܩܵܠܵܐ ܕܩܲܪܢܵܐ. ܣܠܸܩ ܠܹܗ ܒܬܸܫܒ݁ܘܿܚܬܵܐ ܪܲܒ݁ܟܘܼܡܪܹ̈ܐ ܝܼܫܘܿܥ ܡܲܠܟܲܢ. ܘܝܼܬܸܒ݂ ܒܪܵܘܡܵܐ ܡܢ ܝܲܡܝܼܢ ܐܲܒ݂ܵܐ. ܘܲܥܬ݂ܝܼܕ݂ ܬܘܼܒ݂ ܕܢܹܐܬܸܐ ܒܫܘܼܒ݂ܚܵܐ ܪܲܒܵܐ ܕܡܲܠܲܐܟ݂ܵܘ̈ܗܝ. ܒܝܵܘܡܵܐ ܐ݇ܚܪܵܝܵܐ ܕܕ݂ܵܐܹܢ ܠܟ݂ܠܒܸ݁ܪ̈ܝܵܢ. ܚܲܕ݂ܘܼܬ݂ܵܐ ܒܟ݂ܠܵܗ ܐܲܪܥܵܐ. ܒܝܵܘܡܵܐ ܕܣܘܼܠܵܩܹܗ ܕܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ. ܐܲܪܝܼܡ ܐܝܼܕ݂ܵܘ̈ܗܝ ܘܒܲܪܸܟ݂ ܠܬܲܠܡܝܼܕ݂ܵܘ̈ܗܝ. ܘܲܣܠܸܩ ܒܬܸܫܒ݁ܘܿܚܬܵܐܒܡܲܪܟܲܒ݂ܬܵܐ ܪܘܼܚܵܢܵܝܬܵܐ. ܘܟܸܢܫܲܝ̈ ܪܵܘܡܵܐ ܚܕ݂ܵܪ̈ܵܘܗܝ ܡܝܲܒ݁ܒ݂ܝܼܢ ܗ݇ܘܵܘ. ܫܘܼܚܠܵܦܵܐ. ܒܟ݂ܠ ܢܸܫܡܵܐ. ܣܠܸܩ ܐܲܠܵܗܵܐ ܒܫܘܼܒ݂ܚܵܐ ܡܵܪܝܵܐ ܒܩܵܠܵܐ ܕܩܲܪܢܵܐ. ܣܠܸܩ ܒܲܪ ܡܲܠܟܵܐ ܘܲܪܟܸܒ݂ ܥܘܼܫܢܵܐ ܘܓܲܢ݇ܒܵܪܘܼܬ݂ܵܐ. ܘܲܢܦܲܩܘ ܠܐܘܼܪܥܹܗ ܟܠܗܘܿܢ ܛܲܟ݂ܣܹ̈ܐ ܕܲܫܡܲܝܵܢܹ̈ܐ. ܒܲܪܸܟ݂ ܐܸܢܘܿܢ ܘܲܣܓ݂ܝܼܘܼ ܛܵܒ݂. ܦܫܲܛ ܗ݇ܘܵܐ ܐܝܼܕܹܗ ܘܲܒ݂ܪܘܼܚܵܦܹܗ ܒܲܪܸܟ݂ ܐܸܢܘܿܢ. ܘܒܲܣܝܵܡ ܐܝܼܕܹܗ ܐܲܪܥܵܐ ܕܠܝܼܛܲܬ݀ ܐܸܬ݂ܒܲܪܟܲܬ݀ ܗ݇ܘܵܬ݂. ܕܲܢܚܵܘܸܐ ܓܲܢ݇ܒܵܪܘܼܬܹܗ. ܩܵܘܵܘ ܬܲܡܢ ܥܸܣܪܵܐ ܝܵܘܡ̈ܝܼܢ ܕܠܵܐ ܡܲܪܢܝܼܬ݂ܵܐ. ܥܕܲܡܵܐ ܕܬܸܠܒ݁ܫܘܼܢ ܚܲܝܠܵܐ ܕܪܘܼܚܵܐ ܕܲܡܫܲܕܲܪ ܐ݇ܢܵܐ. ܕܲܨܠܘܿܬ݂ܵܐ܀ ܐܹܝܠ ܐܹܝܠ ܠܥܘܼܕ݂ܪܵܢܝ ܟܲܬܲܪ. ܐܹܝܠ ܐܲܠܵܗܵܐ ܫܡܲܥ ܒܵܥܘܼܬܲܢ ܕܲܫܡܝܼܥܵܐ ܠܵܟ݂. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹܐܠܲܬ݂ ܢܲܦ̮ܫܲܢ ܕܲܡܦܵܣܲܬ݁ ܒܵܗ. ܩܲܕ݁ܝܼܫܘܼ ܘܲܕ݂ܚܝܼܠ ܫܡܹܗ. ܐܹܝܠ ܐܲܠܵܗܵܐ ܕܐܲܫܪܝܼ ܫܟ݂ܝܼܢܬܹ݁ܗ ܥܲܠ ܛܘܼܪ ܣܝܼܢܲܝ. ܐܲܫܪܵܐ ܚܢܵܢܵܐ ܕܒܲܣܝܼܡܘܼܬ݂ܵܟ݂ ܥܲܠ ܢܲܦ̮ܫ̈ܵܬܲܢ. ܘܲܠܡܸܫܪܵܐ ܒܢܲܝ̈ܵܐ ܡܢ ܡܵܘܬܵܐ. ܡܲܪܝܲܡ ܕܝܸܠܕܲܬ݀ ܣܲܡ ܚܲܝܘܼܬ݂ܵܐ ܠܝܲܠܕܵܘ̈ܗܝ ܕܐܵܕ݂ܵܡ. ܒܝܲܕ݂ ܒܵܥܘܼܬܹܟ݂ܝ ܢܸܫܟܲܚ ܪ̈ܲܚܡܹܐ ܒܝܘܿܡ ܢܘܼܚܵܡܵܐ. ܡܢ ܕܠܘܼܚܝܵܐ ܕܲܒ݂ܢܲܝ̈ܢܵܫܵܐ. ܨܠܝܼܒ݂ܵܟ݂ ܡܵܪܲܢ ܢܸܗܘܸܐ ܫܘܼܪܵܐ ܠܲܟ݂ܢܘܼ̈ܫܵܬܲܢ. ܘܒܹܗ ܢܸܣܬܲܬܲܪ ܡܢ ܢܸܟ݂ܝܵܢܵܘ̈ܗܝ ܕܐܵܟܹ݁ܠܩܲܪܨܵܐ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܠܵܐ ܬܸܫܕܹܝܢܝ ܡܢ ܩܕ݂ܵܡܲܝܟ. ܒܗܵܝ ܫܵܥܬܵܐ ܕܙܵܥܩܵܐ ܩܲܪܢܵܐ ܐܲܫܘܵܢ ܡܵܪܲܢ. ܕܥܲܡ ܩܲܕ݁ܝܼܫܹ̈ܐ ܕܲܫܦܲܪܘ ܠܵܟ݂ ܢܸܚܙܸܐ ܚܢܵܢܵܟ݂. ܡܲܬ݂ܩܲܢܘܼ ܟܘܼܪܣܝܵܟ݂ ܡܢ ܩܕ݂ܝܼܡ. ܒܥܸܕܵܢܵܐ ܕܬ݂ܵܩܹܢ ܟܘܼܪܣܝܵܟ݂ ܠܒ݂ܘܼܚܪܵܢ ܟܠ. ܠܵܐ ܬܸܥܘܿܠ ܡܵܪܝܵܐ ܒܕܝܼܢܵܐ ܥܲܡ ܥܲܒ݂ܕܲܝ̈ܟ.
       ܫܒܚ. ܕܓ݂ܝܼܘܲܪܓ݁ܝܼܣ ܘܲܪܕ݂ܵܐ. ܒܐܹܝܘ ܠܵܟ݂ ܥܘܼܒܵܐ. ܐܘܼܪܚܹܗ ܕܲܒ݂ܪܵܐ ܝܼܚܝܼܕ݂ܵܝܵܐ. ܠܵܐ ܡܸܬ݁ܕܲܪܟܵܐ ܠܟ݂ܠܗܘܿܢ ܒܪ̈ܲܝܵܐ. ܘܣܵܗܕܵܐ ܡܸܠܬܹܗ ܕܚܲܟ݁ܝܼܡܵܐ. ܕܢܸܫܪܵܐ ܒܐܵܐܲܪ ܘܐܸܠܦܵܐ ܒܝܲܡܵܐ. ܘܚܸܘܝܵܐ ܕܪܵܫܹܦ ܥܲܠ ܫܘܿܥܵܐ. ܐܘܼܪܚܵܐ ܕܗܵܠܹܝܢ ܠܵܐ ܡܸܬ݂ܝܲܕ݂ܥܵܐ. ܘܐܸܢܗܘ ܕܗܵܠܹܝܢ ܗܵܟܲܢܵܐ. ܠܵܐ ܡܸܬ݁ܕܲܪܟ݁ܝܼܢ ܡܢ ܪܸܥܝܵܢܵܐ. ܟܡܵܐ ܟܲܝ ܐܘܼܪܚܹܗ ܕܲܡܪܲܝܡܵܐ. ܥܲܣܩܵܐ ܘܟܲܣܝܵܐ ܓܢܝܼܙܵܐ ܘܪܵܡܵܐ܀ ܡܸܠܬܹܗ ܕܐܲܒ݂ܵܐ ܝ݇ܚܝܼܕ݂ܵܝܵܐ. ܗܵܘ ܕܥܲܡܹܗ ܫܘܸܐ ܒܐܘܼܣܝܼܵܐ. ܐܸܨܛܒ݂ܝܼ ܒܚܘܼܒܹ݁ܗ ܡܬ݂ܘܿܡܵܝܵܐ. ܘܐܸܬ݂ܓ݁ܠܝܼ ܒܦܲܓ݂ܪܵܐ ܐ݇ܢܵܫܵܝܵܐ܀ ܬܲܗܪܵܐ ܥܲܒ݂ܕܵܗ ܠܡܹܐܬ݂ܝܼܬܹܗ. ܕܟܲܕ݂ ܡܲܠܝܵܐ ܡܸܢܹܗ ܒܪܝܼܬܹܗ. ܘܟܲܕ݂ ܣܵܟ݂ܵܐ ܠܲܝܬ݁ ܒܐܝܼܬ݂ܘܼܬܹܗ. ܗܘܵܐ ܒܲܪܢܵܫܵܐ ܒܛܲܝܒ݁ܘܼܬܹܗ܀ ܩܢܘܿܡܹܗ ܟܣܸܐ ܡܢ ܟܠ ܚܙܵܬ݂ܵܐ. ܘܠܵܐ ܡܸܬ݁ܕܪܸܟ݂ ܒܟ݂ܠ ܝܲܕ݂̈ܥܵܬܵܐ. ܘܠܲܝܬ݁ ܠܹܗ ܣܵܟ݂ܵܐ ܘܲܡܫܘܼܚܬܵܐ. ܘܠܵܐ ܐܸܡܲܬ݂ܝ ܘܲܟ݂ܡܵܝܘܼܬ݂ܵܐ܀ ܗܵܢܵܐ ܕܗܵܕܹܐ ܪܲܒ݁ܘܼܬ݂ܵܐ. ܩܢܸܐ ܐܲܝܟ݂ ܐܲܒ݂ܘܼܗܝ ܒܐܝܼܬ݂ܘܼܬ݂ܵܐ. ܨܒ݂ܵܐ ܒܚܘܼܒܹ݁ܗ ܫܪܵܐ ܒܲܒ݂ܬ݂ܘܼܠܬܵܐ. ܘܲܠܒܸܫ ܦܲܓ݂ܪܵܐ ܕܐ݇ܢܵܫܘܼܬ݂ܵܐ܀ ܠܵܐ ܒܲܛܢܹܗ ܗܘܵܐ ܡܢ ܓܲܒ݂ܪܵܐ. ܐܵܦܠܵܐ ܝܲܠܕܹܗ ܒܬ݂ܘܼ̈ܠܹܐ ܫܪܵܐ. ܘܥܲܠ ܗܵܕܹܐ ܩܪܵܝܗܝ ܕܘܼܡܵܪܵܐ. ܢܒ݂ܝܼܵܐ ܘܡܸܠܬܹܗ ܫܲܪܝܼܪܵܐ܀ ܒܝܲܠܕܹܗ ܐܲܙܝܼܥ ܠܡܲܠܟ݁ܘ̈ܵܬ݂ܵܐ. ܒܲܥܡܵܕܹܗ ܫܒܲܩ ܚܵܘ̈ܒܵܬܵܐ. ܘܲܒ݂ܨܵܘܡܹܗ ܥܒܲܕ݂ ܙܵܟ݂ܘܼܬ݂ܵܐ. ܠܟ݂ܠܹܗ ܟܝܵܢܵܐ ܕܐ݇ܢܵܫܘܼܬ݂ܵܐ܀ ܡܝܼܬ݂ ܘܩܲܛܠܹܗ ܒܡܵܘܬܹ݁ܗ ܠܡܵܘܬܵܐ. ܘܚܲܝܒ݂ܵܗ ܒܒܸܣܪܹܗ ܠܲܚܛܝܼܬܵܐ. ܘܩܵܡ ܘܲܥܒܲܕ݂ ܠܓܸܢܣܲܢ ܩܝܵܡܬܵܐ. ܘܲܠܪܵܘܡܵܐ ܣܠܸܩ ܒܬܸܫܒ݁ܘܿܚܬܵܐ܀ ܟܲܕ݂ ܫܲܪܝܼ ܒܲܪ ܥܸܠܵܝܵܐ. ܕܢܸܦܪܲܚ ܠܐܲܬ݂ܪܹܗ ܥܸܠܵܝܵܐ. ܕܒܲܪ ܗ݇ܘܵܐ ܠܟܸܢܫܵܐ ܫܠܝܼܚܵܝܵܐ. ܘܲܡܛܵܐ ܥܕܲܡܵܐ ܠܒܹܝܬ݂ ܥܲܢܝܵܐ܀ ܘܟܲܕ݂ ܫܲܪܝܼ ܠܡܸܦܪܲܫ ܡܸܢܗܘܿܢ. ܕܲܫܠܝܼܚܵܘ̈ܗܝ ܦܲܩܸܕ݂ ܐܸܢܘܿܢ. ܘܗܵܟܲܢܵܐ ܐܸܡܲܪ ܗ݇ܘܵܐ ܠܗܘܿܢ. ܗܦܘܿܟ݂ܘ ܠܨܸܗܝܘܿܢ ܘܠܵܐ ܬܸܟ݂ܪܸܐ ܠܟ݂ܘܿܢ܀ ܠܵܐ ܬܸܬ݁ܥܝܼܩܘܼܢ ܕܐܵܙܹܠ ܐ݇ܢܵܐ. ܘܡܸܢܟ݂ܘܿܢ ܒܲܦܓܲܪ ܡܫܲܢܸܐ ܐ݇ܢܵܐ. ܕܐܸܢ ܐܸܢܵܐ ܠܵܐ ܐܵܙܹܠ ܐ݇ܢܵܐ. ܪܘܼܚܩܘܼܕ݂ܫܵܐ ܠܵܐ ܡܫܲܕܲܪ ܐ݇ܢܵܐ܀ ܩܵܘܵܘ ܒܐܘܿܪܸܫܠܸܡ ܡܕ݂ܝܼܢ݇ܬܵܐ. ܥܕܲܡܵܐ ܕܬܹܐܬܸܐ ܡܵܘܗܲܒ݂ܬܵܐ. ܘܬܸܬ݂ܢܲܨܚܘܼܢ ܒܲܓ݂ܒܲܪ̈ܘܵܬ݂ܵܐ. ܘܒ݂ܵܐܬ݂ܪ̈ܵܘܵܬ݂ܵܐ ܘܲܒ݂ܬܸܕ݂ܡܪ̈ܵܬ݂ܵܐ܀ ܒܵܗ ܓܵܠܹܐ ܐ݇ܢܵܐ ܚܹܝܠ ܪܲܒ݁ܘܼܬ݂ܝ. ܘܲܡܚܵܘܸܐ ܐ݇ܢܵܐ ܓܲܢ݇ܒܵܪܘܼܬ݂ܝ. ܘܡܵܘܕܲܥ ܐ݇ܢܵܐ ܫܘܼܠܛܵܢܝ ܘܡܵܪܘܼܬ݂ܝ. ܘܡܸܬ݂ܟ݁ܪܸܙ ܒܟ݂ܠ ܫܸܡ ܐܲܠܵܗܘܼܬ݂ܝ܀ ܘܟܲܕ݂ ܫܲܪܝܼ ܠܡܸܦܪܲܫ ܡܸܢܗܘܿܢ. ܒܕ݂ܘܼܡܵܪܵܐ ܐܲܩܝܼܡ ܐܸܢܘܿܢ. ܓܲܫܸܡ ܠܐܵܐܲܪ ܠܥܹܢܲܝܗܘܿܢ. ܘܐܸܬ݂ܟܲܣܝܼ ܗ݇ܘܵܐ ܡܢ ܚܙܵܬ݂ܗܘܿܢ܀ ܪܗܸܛܘ ܠܐܘܼܪܥܹܗ ܚܲܝ̈ܠܵܘܵܬ݂ܵܐ. ܘܐܸܬ݂ܟܲܢܲܫܘ ܡܲܫܪ̈ܝܵܬܵܐ. ܘܲܩܥܵܘ ܘܲܩܪܵܘ ܒܩܲܪ̈ܢܵܬܵܐ. ܟܠܗܘܿܢ ܦܵܠܚܹ̈ܐ ܕܡܲܠܟ݁ܘܼܬ݂ܵܐ܀ ܘܡܢ ܫܸܠܝܵܐ ܒܲܪܗܝܼܒ݂ܘܼܬ݂ܵܐ. ܚܠܵܦ ܡܵܐܢܹ̈ܐ ܕܲܙܚܘܿܪ̈ܝܵܬܵܐ. ܒܲܥܢܵܢܹ̈ܐ ܚܸܘܵܪ̈ܵܬ݂ܵܐ. ܐܸܨܛܲܒ݁ܬܲܬ݀ ܠܹܗ ܡܲܪܟܲܒ݂ܬܵܐ܀ ܪܗܸܛܘ ܠܐܘܼܪܥܹܗ ܩܲܠܝܼܠܵܐܝܼܬ݂. ܪ̈ܘܼܚܵܢܹܐ ܚܲܪܝܼܦܵܐܝܼܬ݂. ܦܲܟ݂ܘ ܪܹ̈ܫܲܝܗܘܿܢ ܫܲܘܝܵܐܝܼܬ݂. ܘܲܣܓܸܕ݂ܘ ܠܹܗ ܚܘܼܒܵܢܵܐܝܼܬ݂܀ ܡܲܠܲܐܟܹ̈ܐ ܠܹܗ ܡܫܲܡܫܝܼܢ ܗ݇ܘܵܘ. ܘܪ̈ܲܒܲܝ ܚܲܝܠܹ̈ܐ ܡܫܲܒ݁ܚܝܼܢ ܗ݇ܘܵܘ. ܘܥܝܼܪܹ̈ܐ ܥܲܡܗܘܿܢ ܡܵܘܕܹ݁ܝܢ ܗ݇ܘܵܘ. ܘܲܕ݂ܒܹܝܬ݂ ܐܲܪ̈ܟ݁ܘܿܣ ܪܗܝܼܒ݂ܝܼܢ ܗ݇ܘܵܘ܀ ܘܫܲܠܝܼܛܵܢܹ̈ܐ ܡܝܲܒ݁ܒ݂ܝܼܢ ܗ݇ܘܵܘ. ܘܡܵܪ̈ܵܘܵܬ݂ܵܐ ܡܙܲܡܪܝܼܢ ܗ݇ܘܵܘ. ܘܡܵܘܬ݁ܒܹ̈ܐ ܕܹܝܢ ܡܪܲܡܪܡܝܼܢ ܗ݇ܘܵܘ. ܣܪ̈ܵܦܹܐ ܩܘܼܕ݂ܫܹܗ ܬܵܢܹܝܢ ܗ݇ܘܵܘ܀ ܟܪ̈ܘܿܒܹܐ ܗܘܵܘ ܠܹܗ ܛܵܥܘܿܢܹ̈ܐ. ܠܗܵܘ ܒܲܪ ܓܸܢܣܵܐ ܕܥܲܦܪ̈ܵܢܹܐ. ܘܲܩܥܵܘ ܣܸܕ݂ܪܹ̈ܐ ܕܢܘܼܪ̈ܵܢܹܐ. ܘܚܲܝ̈ܠܵܘܵܬ݂ܵܐ ܕܪ̈ܘܼܚܵܢܹܐ܀ ܡܲܢܘܼ ܗܵܢ ܕܐܵܬܹ݁ܐ ܒܫܘܼܒ݂ܚܵܐ. ܥܲܠ ܥܢܵܢܹ̈ܐ ܕܢܘܼܪܵܐ ܘܪܘܼܚܵܐ. ܘܐܲܒ݂ܵܐ ܩ̇ܵܪܹܐ ܠܹܗ ܒܙܵܘܚܵܐ. ܕܬ݂ܵܐ ܬܸܒ݂ ܥܲܠ ܟܘܼܪܣܝܵܐ ܡܫܲܒ݁ܚܵܐ܀ ܡܲܢܘܼ ܗܵܢ ܕܐܲܝܟ݂ ܒܲܪ ܐ݇ܢܵܫ̈ܝܼܢ. ܡܸܬ݂ܚܙܸܐ ܘܡܢ ܩܕ݂ܵܡܵܘܗܝ ܙܵܝܥܝܼܢ. ܚܲܝܠܹ̈ܐ ܘܣܸܕ݂ܪܹ̈ܐ ܘܠܹܗ ܡܫܲܒ݁ܚܝܼܢ. ܘܕܲܢܚܘܼܪܘܼܢ ܒܹܗ ܠܵܐ ܡܸܫܟ݁ܚܝܼܢ܀ ܡܲܢܘܼ ܗܵܢ ܕܐܵܬܹ݁ܐ ܒܦܲܓ݂ܪܵܐ. ܘܲܡܠܸܐ ܟܠܹܗ ܕܘܼܡܵܪܵܐ. ܘܲܠܪ̈ܘܼܚܵܢܹܐ ܡܩܝܼܡ ܒܬܲܗܪܵܐ. ܒܗܵܕܹܐ ܕܨܹܐܕ݂ܵܘܗܝ ܡܸܣܬܲܥܪܵܐ܀ ܡܲܢܘܼ ܕܡܢ ܒܹܝܬ݂ ܬܲܚܬܵܝܹ̈ܐ. ܣܵܠܹܩ ܠܐܲܬ݂ܪܵܐ ܕܥܸܠܵܝܹ̈ܐ. ܘܗܵܐ ܡܲܪܗܸܒ݂ ܠܪ̈ܘܼܚܵܢܵܝܹܐ. ܘܲܡܒܲܠܗܸܐ ܠܢܘܼܪ̈ܵܢܵܝܹܐ܀ ܡܲܢܘܼ ܕܲܚܫܸܟ݂ܘ ܢܲܗܝܼܪܹ̈ܐ. ܒܦܸܚܡܵܐ ܕܢܲܚܬܵܘ̈ܗܝ ܫܲܦܝܼܪܹ̈ܐ. ܘܲܦܨܝܼܚܝܼܢ ܠܩܘܼܒ݂ܠܹܗ ܥܝܼܪܹ̈ܐ. ܘܠܹܗ ܡܝܲܩܪܝܼܢ ܒܟ݂ܠ ܐܝܼܩܵܪܹ̈ܐ܀ ܡܲܢܘܼ ܗܵܢ ܦܹܐܪܵܐ ܕܝܼܥܵܐ. ܘܲܫܘܲܚ ܘܲܣܠܸܩ ܡܢ ܐܲܪܥܵܐ. ܘܐܸܬ݀ܵܐ ܘܲܥܒܲܪ ܒܲܪܩܝܼܥܵܐ. ܟܲܪ ܕܠܲܝܬ݁ ܨܸܪܝܵܐ ܘܠܵܐ ܬܲܪܥܵܐ܀ ܡܲܢܘܼ ܗܵܢܵܐ ܕܓ݂ܵܘܙܲܠܬܵܐ. ܘܫܲܠܗܹܒ݂ܝܼܬܵܐ ܝܵܩܹܕܬܵܐ. ܗܘܵܬ݂ ܠܹܗ ܝܵܘܡܵܢ ܬܸܫܘܝܼܬܵܐ. ܘܲܥܠܹܝܗ ܝܼܬܸܒ݂ ܒܬܸܫܒ݁ܘܿܚܬܵܐ܀ ܡܲܢܘܼ ܗܵܢܵܐ ܕܥܲܣܩܵܐ ܠܲܢ. ܒܥܵܬܹܗ ܘܠܵܐ ܡܸܬ݂ܦܲܫܩܵܐ ܠܲܢ. ܘܐܲܝܟ݂ ܒܲܪ ܐ݇ܢܵܫ̈ܝܼܢ ܡܸܬ݂ܚܙܸܐ ܠܲܢ. ܘܐܲܝܟ݂ ܡܵܪܝܵܐ ܘܠܵܐ ܡܸܬ݁ܕܪܸܟ݂ ܠܲܢ܀ ܐܸܢ ܟܲܣܝܵܐ ܗ݇ܘ ܡܢ ܐܲܝܟܵܐ ܠܹܗ. ܓܲܠܝܘܼܬ݂ܵܐ ܕܡܸܬ݂ܚܲܙܝܵܐ ܠܹܗ. ܘܐܸܢ ܓܲܠܝܵܐ ܗ݇ܘ ܡܲܢܘܼ ܝܲܗ݇ܒ݂ ܠܹܗ. ܕܡܲܪܟܲܒ݂ܬܵܐ ܬܸܫܬܲܥܒܲܕ݂ ܠܹܗ܀ ܣܸܕ݂ܪܹ̈ܐ ܡܸܢܹܗ ܕܲܚܝܼܠܝܼܢ. ܟܸܢܫܹ̈ܐ ܡܸܢܹܗ ܪܲܥܝܼܠܝܼܢ. ܘܟ݂ܠܗܘܿܢ ܠܦܘܼܩܕܵܢܹܗ ܚܵܝܪܝܼܢ. ܘܲܠܨܸܒ݂ܝܵܢܹܗ ܡܸܫܬܲܥܒ݁ܕܝܼܢ܀ ܡܲܢ ܡܲܠܸܦ ܠܲܢ ܥܲܠ ܗܵܢܵܐ. ܕܠܵܐ ܡܲܨܝܵܐ ܕܬܸܚܙܹܝܘܜ ܥܲܝܢܵܐ. ܘܠܵܐ ܡܲܕ݂ܪܸܟ݂ ܠܹܗ ܪܸܥܝܵܢܵܐ. ܘܠܵܐ ܡܲܕ݂ܥܵܐ ܘܐܵܦܠܵܐ ܗܵܘܢܵܐ܀ ܐܲܝܟ݂ ܡܸܠܬܵܐ ܕܒܲܢܒ݂ܝܼܘܼܬ݂ܵܐ. ܐܲܠܦܲܬ݀ ܐܸܢܘܿܢ ܟܲܣܝܘܼܬ݂ܵܐ. ܕܗܵܢ̇ܵܘ ܝܲܠܕܵܐ ܕܐܝܼܬ݂ܘܼܬ݂ܵܐ. ܕܲܠܒܸܫ ܦܲܓ݂ܪܵܐ ܕܐ݇ܢܵܫܘܼܬ݂ܵܐ܀ ܗܵܢ̇ܵܘ ܡܵܪܝܵܐ ܒܲܫܪܵܪܵܐ. ܘܟ݂ܠܡܸܕܸ݁ܡ ܗܘ ܥܒܲܕ݂ ܘܲܒ݂ܪܵܐ. ܘܲܒ݂ܛܲܝܒ݁ܘܼܬܹܗ ܠܒܸܫ ܦܲܓ݂ܪܵܐ. ܕܢܲܚܸܐ ܠܟ݂ܠ ܠܒ݂ܝܼܫܲܝ̈ ܒܸܣܪܵܐ܀ ܡܢ ܟܲܪ ܕܲܣܠܸܩ ܠܲܫܡܲܝܵܐ. ܫܠܲܚ ܠܘܵܬ݂ ܬܲܠܡܝܼܕ݂ܵܘ̈ܗܝ ܓܒܲܝ̈ܵܐ. ܬܪܹܝܢ ܥܝܼܪܹ̈ܐ ܒܢܲܚ̈ܬܹ݁ܐ ܦܐܲܝ̈ܵܐ. ܘܐܸܡܲܪܘ ܠܗܘܿܢ ܒܩܵܠܹ̈ܐ ܚܠܲܝ̈ܵܐ܀ ܓܠܝܼܠܵܝܹ̈ܐ ܠܡܵܢ ܩܵܝܡܝܼܬ݁ܘܿܢ. ܘܒܲܫܡܲܝܵܐ ܡܘܿܢ ܚܵܝܪܝܼܬ݁ܘܿܢ. ܠܝܼܫܘܿܥ ܕܲܣܠܸܩ ܒܵܥܹܝܬ݁ܘܿܢ. ܗܦܘܿܟ݂ܘ ܠܐܲܝܟܵܐ ܕܐܸܬ݂ܦܲܩܲܕ݂ܬ݁ܘܿܢ܀ ܐܲܟ݂ܡܵܐ ܕܲܣܠܸܩ ܒܐܝܼܩܵܪܵܐ. ܗܵܟܲܢܵܐ ܐܵܬܹ݁ܐ ܒܗܸܕ݂ܪܵܐ. ܘܲܡܚܲܕܸ݁ܬ݂ ܟܠܡܵܐ ܕܲܒ݂ܪܵܐ. ܘܡܸܬ݁ܬܵܘܕܸܐ ܡܵܪܵܐ ܘܲܒ݂ܪܵܐ܀ ܐܘܿ ܕܐܸܬ݂ܚܙܸܝܼ ܙܒܲܢ ܐ݇ܢܵܫܵܐܝܼܬ݂. ܘܬ݂ܘܼܒ݂ ܡܸܬ݂ܚܙܸܐ ܐܲܠܵܗܵܐܝܼܬ݂. ܚܘܼܪ ܘܪܲܚܸܡ ܚܲܢܵܢܵܐܝܼܬ݂. ܠܕܸܐܡܲܪܘ ܗܵܠܹܝܢ ܡܚܝܼܠܵܐܝܼܬ݂܀ ܒܪܝܼܟ݂ ܣܘܼܠܵܩܵܟ݂. ܫܒ݂ܝܼܚ ܪܘܼܡܪܵܡܵܟ݂. ܒܪܵܐ ܕܦܲܪܩܲܢ ܒܝܲܕ݂ ܡܲܚܲܬ݁ܬܹ݁ܗ. ܘܪܲܡܪܸܡ ܠܓܸܢܣܲܢ ܒܝܲܕ݂ ܡܲܣܲܩܬܹ݁ܗ. ܘܐܸܫܬܵܘܕ݁ܝܼ ܠܲܢ ܒܫܲܪܝܼܪܘܼܬ݂ܵܐ. ܕܲܥܬ݂ܝܼܕ݂ ܕܢܹܐܬܸܐ ܐܲܟ݂ܡܵܐ ܕܐܸܬ݀ܵܐ. ܘܢܸܦܪܩܲܢܡܢ ܝܲܕ݂ ܡܵܘܬܵܐ. ܘܢܲܡܠܸܟ݂ ܥܲܡܹܗ ܒܹܝܬ݂ ܡܲܠܟ݁ܘܼܬ݂ܵܐ. ܘܲܢܫܲܒܲܚ ܠܹܗ ܟܠ ܝܵܘ̈ܡܵܬܵܐ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܟ݂ܠܗܘܿܢ ܥܹܐܕܹ̈ܐ ܡܵܪ̈ܵܢܵܝܹܐ. ܠܲܡܕܲܒ݁ܪܵܢܘܼܬ݂ܵܟ݂ ܡܵܪܝ ܬܡܝܼܗܬܵܐ ܘܠܵܐ ܡܸܬ݂ܡܲܠܠܵܢܝܼܬ݂ܵܐ: ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܘܒܲܚܢܵܢܵܐ ܐܸܬ݂ܓܲܡܪܲܬ݀ ܘܐܸܣܬܲܥܪܲܬ݀ ܘܐܸܫܬܲܡܠܝܲܬ݀ ܠܚܘܼܕܵܬܹܗ ܘܲܠܦܘܼܪܩܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ ܚܲܠܵܫܵܐ: ܒܪܹܫܝܼܬ݂ܵܐ ܕܡܸܢܲܢ: ܡܲܣܩܝܼܢܲܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕܬ݂ܵܐ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
      ܩܵܢܘܿܢܵܐ. ܒܲܒ݂ܪܵܐ ܕܩܲܠܣܘܼܗܝ. ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܬܕ݂ܘܼܨ ܐܲܪܥܵܐ* ܘܢܸܚ̈ܕ݁ܝܵܢ ܓܵܙܪ̈ܵܬ݂ܵܐ ܣܲܓܝܼ̈ܐܵܬ݂ܵܐ*  ܒܣܘܼܠܵܩܵܟ݂ ܕܡܢ ܐܲܪܥܵܐ ܠܲܫܡܲܝܵܐ ܡܵܪܲܢܝܼܫܘܿܥ. ܟܠ ܛܵܒ݂̈ܵܢ ܥܲܬܸ݁ܕܬ݁ ܠܲܢ܀
ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܬܕ݂ܘܼܨ ܐܲܪܥܵܐ* ܘܢܸܚ̈ܕ݁ܝܵܢ ܓܵܙܪ̈ܵܬ݂ܵܐ ܣܲܓܝܼ̈ܐܵܬ݂ܵܐ* ܥܢܵܢܹ̈ܐ ܘܥܲܪ̈ܦܸܠܹܐ ܚܕ݂ܵܪ̈ܵܘܗܝ* ܒܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ ܘܲܒ݂ܕ݂ܝܼܢܵܐ ܡܲܬ݂ܩܲܢ ܟܘܼܪܣܝܹܗ* ܢܘܼܪܵܐ ܬܹܐܟ݂ܘܿܠ ܩܕ݂ܵܡܵܘܗܝ* ܘܬ݂ܵܘܩܸܕ݂ ܠܲܒ݂ܥܸܠܕܒ݂ܵܒ݂ܵܘ̈ܗܝ* ܐܲܢܗܲܪܘ ܒܲܪ̈ܩܵܘܗܝ ܠܬܹܒܹܝܠ* ܚܙܵܬ݀ ܘܐܸܬ݁ܕܲܠܚܲܬ݀ ܐܲܪܥܵܐ* ܘܛܘܼܪܹ̈ܐ ܐܲܝܟ݂ ܫܥܘܿܬ݂ܵܐ ܐܸܬ݂ܦܲܫܲܪܘ* ܡܢ ܩܕ݂ܵܡ ܡܵܪܝܵܐ ܡܵܪܵܐ ܕܟ݂ܠܵܗ̇ ܐܲܪܥܵܐ* ܚܵܘܝܼܘ ܫܡܲܝ̈ܵܐ ܙܲܕ݁ܝܼܩܘܼܬܹܗ* ܘܲܚܙܵܘ ܟܠܗܘܿܢ ܥܲܡܡܹ̈ܐ ܐܝܼܩܵܪܹܗ* ܢܸܒ݂ܗܬ݂ܘܼܢ ܟܠ ܕܥܵܒ݂ܕܝܼܢ ܦܬܲܟ݂ܪܹ̈ܐ* ܘܡܸܫܬܲܒ݂ܗܪܝܼܢ ܒܲܓ݂ܠܝܼܦܹ̈ܐ* ܣܓ݂ܘܿܕ݂ܘ ܠܹܗ ܟܠܗܘܿܢ ܡܲܠܲܐܟ݂ܵܘ̈ܗܝ* ܬܸܫܡܲܥ ܘܬܸܚܕܸܐ ܨܸܗܝܘܿܢ* ܘܲܢ̈ܕ݂ܘܼܨܵܢ ܒ̈ܢܵܬ݂ ܝܼܗܘܼܕ݂ܵܐ* ܡܸܛܠ ܕܝܼܢܲܝ̈ܟ ܡܵܪܝܵܐ* ܡܸܛܠ ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܝܵܐ ܡܪܲܝܡܵܐ ܥܲܠ ܟܠܵܗ̇ ܐܲܪܥܵܐ* ܘܛܵܒ݂ ܐܸܬ݁ܬ݁ܪܝܼܡܬ݁ ܥܲܠ ܟܠܗܘܿܢ ܐܲܠܵܗܹ̈ܐ* ܪ̈ܵܚܡܵܘܗܝ ܕܡܵܪܝܵܐ ܣܵܢܹܝܢ ܠܒ݂ܝܼܫܬܵܐ* ܘܢ̇ܵܛܲܪ ܢܲܦ̮̈ܫܵܬ݂ܵܐ ܕܙܲܕ݁ܝܼܩܵܘ̈ܗܝ* ܘܡܢ ܐܝܼܕ݂ܵܐ ܕܪ̈ܲܫܝܼܥܹܐ ܢܦܲܨܸܐ ܐܸܢܘܿܢ* ܢܘܼܗܪܵܐ ܕܲܕ݂ܢܲܚ ܠܙܲܕ݁ܝܼܩܹ̈ܐ* ܘܠܲܬ݂ܪ̈ܝܼܨܲܝ ܠܸܒܵܐ ܚܲܕ݂ܘܼܬ݂ܵܐ* ܚܕ݂ܵܘ ܙܲܕ݁ܝܼܩܹ̈ܐ ܒܡܵܪܝܵܐ* ܘܐܵܘܕܵܘ ܠܕ݂ܘܼܟ݂ܪܵܢܵܐ ܕܩܘܼܕ݂ܫܹܗ* ܒܣܘܼܠܵܩܵܟ݂ ܕܡܢ ܐܲܪܥܵܐ ܠܲܫܡܲܝܵܐ ܡܵܪܲܢܝܼܫܘܿܥ. ܟܠ ܛܵܒ݂̈ܵܢ ܥܲܬܸ݁ܕܬ݁ ܠܲܢ܀
ܫܲܒܲܚ. ܥܵܠܲܡ* ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܦܵܪܘܿܩܲܢ ܕܝܲܩܲܪܬܵܝܗܝ ܠܟ݂ܠܹܗ ܟܝܵܢܲܢ. ܒܣܘܼܠܵܩܵܟ݂ ܕܠܲܫܡܲܝܵܐ܀
ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܬܕ݂ܘܼܨ ܐܲܪܥܵܐ* ܘܢܸܚ̈ܕ݁ܝܵܢ ܓܵܙܪ̈ܵܬ݂ܵܐ ܣܲܓܝܼ̈ܐܵܬ݂ܵܐ* ܒܪܵܐ ܕܫܲܒ݁ܚܘܼܗܝ ܡܲܠܲܐܟܹ̈ܐ ܒܣܘܼܠܵܩܹܗ ܒܗܘܼܠܵܠܲܝ̈ܗܘܿܢ. ܦܪܘܿܩ ܠܟܸܢܫܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀
	ܬܸܫܒ݁ܘܿܚܬܵܐ. ܠܵܐ ܣܵܦ݁ܩܝܼܢ ܪܵܘܡܵܐ ܘܥܘܼܡܩܵܐ ܘܟ݂ܠ ܕܲܒ݂ܗܘܿܢ: ܕܢܵܘܕ݁ܘܿܢ ܠܐܝܼܬ݂ܘܼܬ݂ܵܟ݂ ܐܝܼܬ݂ܝܵܐ ܡܲܬ݂ܩܸܢ ܟܠ܀ ܘܲܙܥܘܿܪܝܼܢ ܕܢܸܬ݂ܢܘܿܢ ܚܘܼܒܵܟ݂ ܕܲܠܘܵܬܲܢ: ܘܪܲܒ݁ܘܼܬ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܪ̈ܲܚܡܲܝܟ ܣܲܓ݁ܝܼ̈ܐܹܐ܀ ܕܲܣܥܲܪܬ݁ ܨܹܝܕ݂ ܓܸܢܣܲܢ ܟܲܕ݂ ܠܵܐ ܫ̇ܵܘܹܝܢ ܗ݇ܘܲܝܢ: ܛܵܒ݂ܵܐ ܘܒܲܣܝܼܡܵܐ ܕܠܲܒ݂ܫܹܗ ܠܲܟ݂ܝܵܢܲܢ܀ ܘܦܲܪܩܹܗ ܡܢ ܡܵܘܬܵܐ ܘܐܲܣܩܹܗ ܠܲܫܡܲܝܵܐ: ܘܥܲܒ݂ܕܹ݁ܗ ܠܥܸܠ ܡܢ ܟܠ ܡܵܪܵܐ ܘܫܲܠܝܼܛܵܐ܀ ܘܗܵܐ ܣܵܓ݂ܕ݁ܝܼܢ ܩܕ݂ܵܡܵܘܗܝ ܟܸܢܫܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ: ܘܩܵܥܹܝܢ ܕܠܵܐ ܫܲܠܘܵܐ ܟܠܗܘܿܢ ܫܲܘܝܵܐܝܼܬ݂܀ ܩܲܕ݁ܝܼܫܲܬ݁ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܩܘܼܕ݂ܫܵܐ: ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܢ ܟܠ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ܀
  ܟܵܪܘܿܙܘܼܬ݂ܵܐ. ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܒܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ. ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ݁ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡܠܲܟ݂ ܛܵܒ݂̈ܵܬ݂ܵܐ ܠܪ̈ܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕܬܹܗ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܵܢ: ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܗܘܝܘܼ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܪ̈ܝܵܬ݂ܵܐ: ܘܥܲܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟ݁ܝܼܣ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܦܵܐ ܡܝܼܛܪܵܦܘܠܹܝܛܝܼܣ: ( ܌ ) ܘܥܲܠ ܟܠܗܘܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܢ ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒ݂ܐܲܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܘܫܲܝܸܢ ܥܹܕܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܢ.
ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܡܸܢܵܟ݂ ܡܠܸܐ ܪ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ: ܘܡܢ ܥܘܼܬ݂ܪܵܐ ܪܲܒܵܐ ܕܒܲܣܝܼܡܘܼܬܹܗ ܕܚܘܼܒܵܟ݂: ܘܓܲܙܵܐ ܫܦܝܼܥܵܐ ܕܲܚܢܵܢܵܟ݂: ܫܵܐܠܝܼܢܲܢ ܥܘܼܕ݂ܪܵܢܵܐ ܘܚܘܼܝܵܠܵܐ ܘܫܘܼܘܙܵܒ݂ܵܐ ܘܢܘܼܛܵܪܵܐ ܘܐܵܣܝܘܼܬ݂ܵܐ ܠܟܹܐܒܹ̈ܐ ܕܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮ܫ̈ܵܬܲܢ: ܗܲܒ݂ܠܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܪ̈ܲܚܡܲܝܟ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܚܵܬ݂ܵܗ̇. ܒܪܝܼܟ݂ܝܼܢ ܘܲܣܓ݂ܝܼܕ݂ܝܼܢ: ܘܪܵܡܝܼܢ ܘܲܡܥܲܠܝܼܢ: ܘܠܵܐ ܡܸܬ݁ܕܲܪܟ݁ܝܼܢ ܪ̈ܲܚܡܹܐ ܬ݂ܘܿܡܵܝܹ̈ܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܕܚܵܢܝܼܢ ܡܲܓܵܢ ܠܚܲܛܵܝܹ̈ܐ: ܣܲܒ݂ܪܲܢ ܛܵܒ݂ܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܲܢ ܡܠܸܐ ܪ̈ܲܚܡܹܐ ܘܫܵܒܹ݁ܩ ܚܵܘܒܹ̈ܐ ܘܲܚܛܵܗܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
      ܡܲܕ݂ܪܵܫܵܐ. ܒܠܲܐܪ̈ܒܲܥܦܸܢܝܵܢ. ܥܘܼܢܵܝܵܐ. ܬܵܘܕ݁ܝܼ ܠܙܲܟܵܝܵܐ. ܕܲܙܟ݂ܵܐ ܘܙܲܟ݁ܝܼ ܠܟ݂ܠ. ܘܲܣܠܸܩ ܒܬܸܫܒ݁ܘܿܚܬܵܐ ܠܲܫܡܲܝܵܐ. ܒܵܬܹ̈ܐ. ܚܲܕ݂ ܪܲܒ݁ܟܘܼܡܪܹ̈ܐ ܐܸܬ݁ܕܒܲܪ ܡܢ ܡܵܝܘܿܬܹ̈ܐ ܚܠܵܦ ܡܵܝܘܿܬܹ̈ܐ. ܘܥܲܠ ܕܲܢܟܲܗܸܢ ܒܡܲܫܟ݁ܢܵܐ ܐܲܠܵܗܵܝܵܐ ܚܠܵܦ ܡܲܪܥܝܼܬܹܗ. ܠܲܩܕ݂ܘܿܫ ܩܘܼܕ݂ܫܹ̈ܐ ܕܪܵܡ ܡܢ ܩܘܼܕ݂ܫܵܐ ܐܲܪܥܵܢܵܝܵܐ ܐܸܫܬ݁ܘܝܼ ܕܢܸܥܘܿܠ: ܒܲܕ݂ܡܵܐ ܕܲܩܢܘܿܡܹܗ. ܥܲܠ ܘܐܸܬ݂ܚܙܝܼ ܠܟܲܣܝܵܐ. ܘܐܸܫܟܲܚ ܦܘܼܪܩܵܢܵܐ ܕܲܠܥܵܠܲܡ܀ ܒܲܕ݂ܡܵܐ ܕܢܲܦ̮ܫܹܗ ܚܲܣܝܹܗ ܠܥܵܘܠܵܐ ܣܲܢܝܵܐ ܕܲܒ݂ܢܲܝ̈ ܓܸܢܣܹܗ. ܘܲܒ݂ܕܲܟ݂ܝܘܼܬܹܗ ܕܲܟ݂ܝܵܗ ܠܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܡܲܠܝܲܬ݂ ܚܲܫܹ̈ܐ. ܐܲܝܟ݂ ܕܲܠܢܲܚܬܵܐ ܐܲܫܝܼܓ݂ ܐܸܢܘܿܢ ܘܲܥܒܲܕ݂ ܐܸܢܘܿܢ ܡܵܐܢܹ̈ܐ ܕܟܲܝ̈ܵܐ: ܒܦܲܓ݂ܪܹܗ ܠܒܸܫ ܐܸܢܘܿܢ. ܐܲܝܟ݂ ܟܵܗܢܵܐ ܠܢܲܚܬܵܐ. ܘܥܲܠ ܠܡܲܫܟ݁ܢܵܐ ܪܲܒܵܐ ܕܲܫܡܲܝܵܐ܀ ܦܸܕ݂ܬܵܐ ܕܦܲܓ݂ܪܵܐ ܠܒܸܫ ܗ݇ܘܵܐ ܘܥܲܠ ܦܲܓ݂ܪܵܢܵܐ ܠܲܩܕ݂ܘܿܫ ܩܘܼܕ݂ܫܹ̈ܐ. ܘܐܲܝܟ݂ ܡܲܨܢܲܦܬܵܐ ܙܲܝܚܵܗ ܠܡܵܝܘܿܬ݁ܘܼܬ݂ܵܐ ܒܪܹܫ ܪܸܥܝܵܢܹܗ. ܟܠܹܗ ܓܸܢܣܵܐ ܕܒܲܪ ܡܵܝܘܿܬ݂ܵܐ. ܘܐܲܥܸܠ ܥܲܡܹܗ ܠܐܲܬ݂ܪܵܐ ܕܥܲܠ ܠܹܗ: ܕܘܼܨ ܘܐܸܬ݂ܦܲܨܚ ܡܸܟܹ݁ܝܠ. ܟܝܵܢܵܐ ܡܠܸܐ ܚܲܫܹ̈ܐ. ܕܲܗܘܲܝܬ݁ ܠܒ݂ܘܼܫ ܩܘܼܕ݂ܫܵܐ ܠܪܲܒ݁ܟܘܼܡܪܹ̈ܐ.
      ܐ݇ܚܪܹܢܵܐ. ܒܲܡܫܝܼܚܵܐ ܕܲܒ݂ܪ̈ܲܚܡܵܘܜ. ܥܘܼܢܵܝܵܐ. ܡܫܝܼܚܵܐ ܕܲܒ݂ܝܘܿܡ ܣܘܼܠܵܩܹܗ. ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܠܲܫܡܲܝܵܐ. ܘܚܲܕ݁ܝܼ ܠܟܸܢܫܹ̈ܐ ܕܪ̈ܘܼܚܵܢܹܐ. ܘܐܲܦ̮ܨܲܚ ܠܓ݂ܘܼܕܹ̈ܐ ܕܐܲܪ̈ܥܵܢܹܐ. ܒܵܬܹ̈ܐ. ܐܘܿ ܕܪܲܡܪܸܡ ܠܓܸܢܣܲܢ ܫܝܼܛܵܐ. ܒܣܘܼܠܵܩܹܗ ܕܲܠܘܵܬ݂ ܐܲܒ݂ܘܼܜ. ܘܙܲܡܢܹܗ ܠܲܓ݂ܢܘܿܢ ܡܲܠܟ݁ܘܼܬܹܗ. ܒܡܲܣܲܩܬܹܗ ܠܲܫܡܲܝ̈ ܫܡܲܝܵܐ. ܘܫܵܘܬ݁ܦܹܗ ܒܐܝܼܩܵܪ ܫܘܼܠܛܵܢܹܗ. ܘܟ݂ܘܼܪܣܝܵܐ ܫܒ݂ܝܼܚܵܐ ܕܪܲܒ݁ܘܼܬܹܗ. ܘܥܲܒ݂ܕܹܗ ܪܹܫܵܐ ܘܕܲܝ̇ܵܢܵܐ. ܒܪܝܼܟ݂ܘܼ ܕܪܵܡ ܗ݇ܘ ܘܲܡܥܲܠܲܝ܀ ܥܢܵܢܹ̈ܐ ܕܢܘܼܗܪܵܐ ܘܲܕ݂ܪܘܼܚܵܐ. ܛܥܲܢ ܗ݇ܘܲܝ̈ ܦܲܓ݂ܪܹܗ ܩܲܕ݁ܝܼܫܵܐ. ܥܸܠܵܝܹ̈ܐ ܣܘܲܚܘ ܠܐܝܼܩܵܪܹܗ. ܘܬܲܚܬܵܝܹ̈ܐ ܬܡܲܗܘ ܒܲܚܙܵܬܹܗ. ܘܬܲܠܡܝܼܕܹ̈ܐ ܟܡܲܪܘ ܒܡܹܐܙܲܠܬܹܗ. ܕܐܸܬ݂ܦܪܸܫ ܠܹܗ ܡܢ ܨܹܐܕܲܝܗܘܿܢ. ܬܵܘ ܐܲܚܲܝ̈ ܢܸܙܡܲܪ ܟܠܲܢ. ܒܪܝܼܟ݂ܘܼ ܘܲܣܓ݂ܝܼܕ݂ ܐܝܼܩܵܪܹܗ܀ ܓܘܼܕ̈ܝܼܢ ܓܘܼܕ̈ܝܼܢ ܕܲܟ݂ܪ̈ܘܿܒܹܐ. ܘܣܸܕ݂ܪ̈ܝܼܢ ܣܸܕ݂ܪ̈ܝܼܢ ܕܲܣܪ̈ܵܦܹܐ. ܘܣܝܼ̈ܥܵܢ ܣܝܼ̈ܥܵܢ ܕܡܲܠܲܐܟܹ̈ܐ. ܙܲܝܚܘܼܜ ܠܲܒ݂ܪܵܐ ܒܡܲܣܲܩܬܹܗ. ܐܲܒ݂ܵܐ ܒܚܘܼܒܹ݁ܗ ܓܕܲܠ ܘܣܵܡ ܠܹܗ. ܬܵܓ݂ܵܐ ܕܫܘܼܠܛܵܢ ܡܲܠܟ݁ܘܼܬܹܗ. ܕܗܵܢ̇ܵܘ ܡܲܠܟܵܐ ܕܐܝܼܩܵܪܹܗ. ܕܡܢ ܟܠ ܙܵܕܹ݁ܩ ܘܢܸܫܬܲܒܲܚ.
      ܕܐܵܦܸܢܵܐ. ܒܪܝܼܟ݂ ܕܐܲܬ݂ܩܸܢ ܡܲܪܟܲܒ݂ܬܵܐ. ܡܢ ܢܘܼܪܵܐ ܕܓ݂ܵܘܙܲܠܬܵܐ. ܘܐܸܣܬܲܠܲܩ ܒܬܸܫܒ݁ܘܿܚܬܵܐ. ܥܲܠ ܓܝܼܓ݂ܠܲܝ̈ ܡܵܘܙܲܠܬܵܐ܀ ܒܪܝܼܟ݂ ܕܲܒ݂ܛܘܼܪܵܐ ܕܙܲܝ̈ܬܹ݁ܐ. ܒܲܪܸܟ݂ ܠܲܬ݂ܪܸܥܣܲܪܬܹܗ. ܘܲܥܢܵܢܵܐ ܩܲܒܸ݁ܠܬܹܗ. ܘܠܲܫܡܲܝܵܐ ܐܲܣܸܩܬܹܗ.
ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܩܠ̈ܐ ܕܫܗܪܐ.
	ܨܠܘܿܬ݂ܵܐ ܩܲܕ݂ܡܵܝܬܵܐ. ܝܲܡܵܐ ܠܵܐ ܡܸܬ݁ܬ݁ܟ݂ܝܼܠܵܢܵܐ ܕܚܲܝܹ̈ܐ: ܗ̇ܘ ܕܓܲܠܠܵܘ̈ܗܝ ܓܝܼܓ݂ܠܲܝ̈ ܢܘܼܗܪܵܐ: ܘܫܸܦܥܵܐ ܕܲܚܢܵܢܹܗ ܡܫܲܒ݁ܚܵܐ ܡܠܸܐ ܛܘܼܒܹ̈ܐ. ܘܡܢ ܛܲܠܵܐ ܡܵܪܝ ܕܚܲܝܹ̈ܐ ܣܵܝܩܝܼܢ ܚܲܝܘܼܬ݂ܵܐ: ܘܪܵܘܙܝܼܢ ܒܵܟ݂ ܛܲܟ݂ܣܹ̈ܐ ܟܠܗܘܿܢ ܕܥܸܠܵܝܹ̈ܐ ܡܫܲܡܫܵܢܹ̈ܐ ܕܙܝܼܘܹܗ ܕܫܘܼܒ݂ܚܵܟ݂ . ܣܥܘܿܪ ܡܵܪܝ ܒܪ̈ܲܚܡܹܐ ܠܬܸܓ݂ܡܵܐ ܡܚܝܼܠܵܐ ܕܝܼܠܲܢ ܥܲܦܪ̈ܵܢܹܐ ܛܒ݂ܝܼܥܲܝ̈ ܒܝܘܼܩܢܵܟ݂: ܘܚܲܕܸ݁ܬ݂ ܢܲܦ̮ܫ̈ܵܬܲܢ ܒܡܵܘܗܲܒ݂ܬ݂ܵܟ݂: ܘܩܲܕܸ݁ܫ ܣܸܦܘ̈ܵܬܲܢ ܒܬܸܫ̈ܒ݁ܚܵܬܵܐ ܕܦܲܐ̈ܝܵـܢ ܠܲܫܡܵܟ݂ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܩܠ̈ܐ ܕܫܗܪܐ ܩܲܕ݂ܡܵܝܵܐ. (ܡܲܙܡܘܿܪܵܐ. ܡܙ). ܟܠܟ݂ܘܿܢ ܥܲܡܡܹ̈ܐ ܩܘܿܫܘ ܟܲܦܵܐ* ܗܲܠܸܠܘܿܝܵܗ ܗܲܠܸܠܘܿܝܵܗ ܘܫܲܒܲܚܘ ܠܐܲܠܵܗܵܐ ܒܩܵܠܵܐ ܕܫܘܼܒ݂ܚܵܐ* ܐܹܗܲܠܸܠܘܿܝܵܗ ܡܸܛܠ ܕܡܵܪܝܵܐ ܡܪܲܝܡܵܐ ܘܲܕ݂ܚܝܼܠܵܐ* ܗܘܝܘܼ ܡܲܠܟܵܐ ܪܲܒܵܐ ܥܲܠ ܟܠܵܗ̇ ܐܲܪܥܵܐ* ܕܫܲܥܒܸܕ݂ ܠܲܢ ܥܲܡܡܹ̈ܐ ܬܚܘܿܬܲܝܢ* ܘܐܸܡ̈ܘܵܬ݂ܵܐ ܬܚܘܿܬ ܪܸ̈ܓ݂ܠܲܝܢ* ܐܲܓ݂ܒܝܲܢ ܝܵܪܬܘܼܬܹܗ ܘܐܝܼܩܵܪܹܗ ܕܝܲܥܩܘܿܒ݂ ܕܲܪܚܸܡ* ܣܠܸܩ ܐܲܠܵܗܵܐ ܒܫܘܼܒ݂ܚܵܐ ܡܵܪܝܵܐ ܒܩܵܠܵܐ ܕܩܲܪܢܵܐ* ܙܡܲܪܘ ܠܐܲܠܵܗܵܐ ܒܫܘܼܒ݂ܚܵܐ ܙܡܲܪܘ ܠܡܲܠܟܲܢ* ܡܸܛܠ ܡܲܠܟܵܐ ܕܟ݂ܠܵܗ̇ ܐܲܪܥܵܐ ܐܲܠܵܗܵܐ ܗ݇ܘ ܙܡܲܪܘ ܠܹܗ ܫܘܼܒ݂ܚܵܐ* ܐܲܡܠܸܟ݂ ܐܲܠܵܗܵܐ ܥܲܠ ܥܲܡܡܹ̈ܐ* ܐܲܠܵܗܵܐ ܝܼܬܸܒ݂ ܥܲܠ ܟܘܼܪܣܝܹܗ ܩܲܕܝܼܫܵܐ* ܫܲܠܝܼܛܵܢܹ̈ܐ ܕܥܲܡܡܹ̈ܐ ܐܸܬ݂ܦܢܵܘ ܠܘܵܬ݂ ܐܲܠܵܗܹܗ ܕܐܲܒ݂ܪܵܗܵܡ* ܡܸܛܠ ܕܐܲܠܵܗܵܐ ܐܸܢܘܿܢ ܐܘܼܚܕܵܢܹܝ̈ܗ̇ ܕܐܲܪܥܵܐ ܘܛܵܒ݂ ܐܸܬܬܪܝܼܡ*
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܲܬ݂ܪܲܝܵܢܵܐ. ܒܩ̈ܝܼܢܵܬܵܐ ܡܵܪܝ ܩܲܕ݁ܝܼ̈ܫܵܬܵܐ: ܘܲܒ݂ܡܸܠܹ̈ܐ ܚܲܠ̈ܝܵܬܵܐ ܘܒܲܣܝܼ̈ܡܬܵܐ ܡܸܬ݂ܒܲܪܟ݂ܵܐ ܟܠܫܵܥ ܫܟ݂ܝܼܢܬܵܐ ܕܪܲܒ݁ܘܼܬ݂ܵܟ݂: ܡܢ ܠܸܓ݂ܝܘܿܢܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ ܡܫܲܡܫܲܝ̈ ܩܘܼܕ݂ܫܵܟ݂ ܕܲܩܕ݂ܵܡ ܬܪܘܿܢܘܿܣ ܕܝܼܠܵܟ݂ ܩܵܝܡܝܼܢ: ܥܲܡܗܹܝܢ ܡܵܪܝ ܘܲܒ݂ܗܹܝܢ ܩܲܒܸ݁ܠ ܘܚܲܠܸܛ ܩܘܼܪܒܵܢ ܬܸܫ̈ܒ݁ܚܵܬܵܐ ܗܵܢܹܝܢ ܕܩܲܪܸܒ݂ܘ ܘܲܡܩܲܪܒ݂ܝܼܢ ܥܲܒ݂ܕܲܝ̈ܟ ܡܥܲܗܕܵܢܹ̈ܐ ܕܐ݇ܪ̈ܵܙܹܐ ܕܲܡܕܲܒ݁ܪܵܢܘܼܬ݂ܵܟ݂: ܘܩܲܕܸ݁ܫ ܢܲܦ̮ܫ̈ܵܬܲܢ ܒܫܘܼܟܵܢܹ̈ܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܬܪܲܝܵܢܵܐ. (ܡܲܙܡܘܿܪܵܐ. ܨܓ). ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܘܓܲܐܝܘܼܬ݂ܵܐ ܠܒܸܫ* ܗܲܠܸܠܘܿܝܵܗ ܗܲܠܸܠܘܿܝܵܗ ܠܒܸܫ ܡܵܪܝܵܐ ܥܘܼܫܢܵܐ ܘܐܸܬ݂ܥܲܫܲܢ* ܐܹܗܲܠܸܠܘܿܝܵܗ ܘܐܲܬ݂ܩܢܵܗ̇ ܠܬܹܒܹܝܠ ܕܠܵܐ ܬܙܘܼܥ* ܡܲܬ݂ܩܲܢܘܼ ܟܘܼܪܣܝܵܟ݂ ܡܢ ܩܕ݂ܝܼܡ* ܘܡܢ ܥܵܠܲܡ ܐܲܢ݇ܬܘܼ* ܐܸܬܬܪܝܼܡܘ ܢܲܗܪ̈ܵܘܵܬ݂ܵܐ ܡܵܪܝܵܐ* ܘܐܲܪܝܼܡܘ ܢܲܗܪ̈ܵܘܵܬ݂ܵܐ ܩܵܠܗܘܿܢ* ܐܸܬܬܪܝܼܡܘ ܢܲܗܪ̈ܘܵܬ݂ܵܐ ܒܕܲܟ݂ܝܘܼܬ݂ܵܐ* ܡܢ ܩܵܠܵܐ ܕܡܲܝ̈ܵܐ ܣܲܓܝܼܐܹ̈ܐ* ܥܲܫܝܼܢܝܼܢ ܡܲܚܫܘܼܠܵܘ̈ܗܝ ܕܝܲܡܵܐ* ܡܫܲܒܲܚ ܒܲܡܪ̈ܵܘܡܹܐ ܡܵܪܝܵܐ* ܘܣܵܗ̈ܕܘܵܬ݂ܵܟ݂ ܛܵܒ݂ ܡܗܲܝ̈ܡܢܵܢ* ܘܲܠܒܲܝܬܵܟ݂ ܝܵܐܝܵܐ ܩܲܕܝܼܫܘܼܬ݂ܵܐ* ܡܵܪܝܵܐ ܠܢܘܼܓ݂ܪܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ*
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܵܐ: ܥܘܼܗܕܵܢܵܐ ܡܵܪܝ ܬܗܝܼܪܵܐ ܕܓܸܠܝܵܢܵܟ݂ ܗܕ݂ܝܼܪܵܐ: ܘܕܲܙܥܵܩܵܬܵܐ ܬܠܵܬ݂ ܕܚܝܼܠܵܬܵܐ ܗܵܢܹܝܢ ܕܩܲܪ̈ܢܵܬܵܐ ܡܸܛܠ ܕܚܝܼܠܘܼܬ݂ܗܹܝܢ ܐܸܫܬܲܡܲܗ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܬܠܝܼܬ݂ܵܝܲܬ݂ ܩܘܼܕ݂ܫܹ̈ܐ ܩܲܪܒܲܬ݀ ܩܕ݂ܵܡ ܪܲܒ݁ܘܼܬ݂ܵܟ݂ ܥܹܕ݁ܬܵܟ݂ ܒܟ݂ܠ ܬܸܓ݂ܡܹ̈ܐ ܕܝܲܠܕܹܝ̈ܗ̇. ܢܫܵܘܙܸܒ݂ ܐܸܢܘܿܢ ܡܵܪܝ ܐܘܼܟ݂ܦܵܢ ܚܢܵܢܵܟ݂ ܡܢ ܟܠ ܦܸܓ݂ܥܹ̈ܐ ܕܚܝܼܠܹ̈ܐ ܘܢܸܒ݂ܣܲܡ ܟܠܫܵܥ ܒܩܝܼ̈ܢܵܬ݂ܵܐ ܕܦܲܐ̈ܝܵܢ ܠܲܫܡܵܟ݂: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܬܠܝܼܬ݂ܵܝܵܐ. (ܡܲܙܡܘܿܪܵܐ. ܩܝ). ܐܸܡܲܪ ܡܵܪܝܵܐ ܠܡܵܪܝ ܕܬܸܒ݂ ܠܵܟ݂ ܡܢ ܝܲܡܝܼܢܝ* ܗܲܠܸܠܘܿܝܵܗ ܗܲܠܸܠܘܿܝܵܗ ܥܕܲܡܵܐ ܕܐܹܣܝܼܡ ܒܥܸܠܕܒ݂ܵܒܲܝ̈ܟ ܟܘܼܒ݂ܫܵܐ ܠܪܸ̈ܓ݂ܠܲܝܟ* ܐܹܗܲܠܸܠܘܿܝܵܗ ܚܘܼܛܪܵܐ ܕܥܘܼܫܢܵܐ ܢܫܲܕܲܪ ܡܵܪܝܵܐ ܡܢ ܨܸܗܝܘܿܢ* ܘܢܸܫܬܲܠܲܛ ܥܲܠ ܒܥܸܠܕܒ݂ܵܒܲܝ̈ܟ* ܥܲܡܵܟ݂ ܡܫܲܒܚܵܐ ܒܝܵܘܡܵܐ ܕܚܲܝܠܵܐ* ܒܗܸܕ݂ܪ̈ܲܝ ܩܘܼܕ݂ܫܵܐ ܡܢ ܡܲܪܒܥܵܐ* ܡܢ ܩܕ݂ܝܼܡ ܠܵܟ݂ ܛܲܠܝܵܐ ܝܼܠܸܕܬܵܟ݂* ܝܼܡܵܐ ܡܵܪܝܵܐ ܘܠܵܐ ܢܟܲܕܸܒ݂* ܕܐܲܢ݉ܬܘܼ ܟܘܼܡܪܵܐ ܠܥܵܠܲܡ* ܒܲܕ݂ܡܘܼܬܹܗ ܕܡܲܠܟܝܼܙܕܸܩ* ܡܵܪܝܵܐ ܥܲܠ ܝܲܡܝܼܢܵܟ݂* ܬܒܲܪ ܒܝܵܘܡܵܐ ܕܪܘܼܓܹܙܗ ܡܲܠܟܹ̈ܐ* ܢܕ݂ܘܼܢ ܠܥܲܡܡܹ̈ܐ ܘܢܸܡܠܸܐ ܫܠܲܕܹ̈ܐ* ܘܢܸܦܣܘܿܩ ܪܹܫܵܐ ܕܣܲܓܝܼ̈ܐܹܐ ܒܐܲܪܥܵܐ* ܘܡܢ ܢܲܚܠܵܐ ܒܐܘܼܪܚܵܐ ܢܸܫܬܸܐ* ܡܸܛܠ ܗܵܢܵܐ ܢܸܬܬܪܝܼܡ ܪܹܫܹܗ*
	ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܵܘ̈ܵܬ݂ܵܐ ܕܥܘܿܢ̈ܝܵܬ݂ܵܐ ܕܠܸܠܝܵܐ ܕܥܹܐܕܹ̈ܐ ܡܵܪ̈ܵܢܵܝܹܐ.
	ܨܠܘܿܬ݂ܵܐ ܩܲܕ݂ܡܵܝܬܵܐ. ܡܢ ܟܠ ܒܸܪ̈ܝܵܢ ܕܨܲܒܸ݁ܬ݂ ܪܸܡܙܵܟ݂: ܬܵܘܕ݁ܝܵܬܵܐ ܐܲܡܝܼ̈ܢܵܬܵܐ ܒܝܲܕ݂ ܙܵܘܥܹ̈ܐ ܕܠܵܐ ܩܵܠܵܐ ܡܸܬ݂ܦܲܢ̈ܝܵܢ ܠܪܲܒ݁ܘܼܬ݂ ܐܝܼܬ݂ܘܼܬ݂ܵܟ݂: ܒܛܲܟ݂ܣܵܐ ܠܵܐ ܡܸܬ݁ܕܲܪܟܵܢܵܐ ܗ̇ܘ ܕܪܲܟ݁ܒܲܬ݀ ܚܸܟ݂ܡܬ݂ܵܟ݂: ܢܸܒ݂ܣܲܡ ܠܵܟ݂ ܡܵܪܝ ܐܵܦ ܗܵܫܵܐ ܩܘܼܪܒܵܢ ܩܵܠܹ̈ܐ ܕܬܸܫ̈ܒ݁ܚܵܬܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܐܲܠܦܲܬ݀ ܠܥܹܕ݁ܬܵܟ݂ ܠܲܡܩܲܪܵܒ݂ܘܼ ܩܕ݂ܵܡ ܫܡܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ: ܘܐܲܢܗܲܪ ܠܲܡܩܲܪ̈ܒ݂ܵܢܲܝܗܘܿܢ ܒܢܘܼܗܪܹܗ ܪܲܒܵܐ ܕܲܫܡܵܟ݂: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܕܠܸܠܝܵܐ. ܒܲܫܠܝܼܚܹ̈ܐ ܓܒܲܝ̈ܵܐ. ܡܵܢܵܘ ܓܲܒ݂ܪܵܐ ܕܐܸܬܕܟܲܪܬܵܝܗܝ: ܐܲܢ݇ܬܘܼ ܦܪܲܩܬܵܢ ܡܢ ܣܵܢܐܲܝ̈ܢ. ܕܨܘܼܪܬܵܐ. ܐܸܬ݂ܒܠܲܥ ܡܵܘܬܵܐ ܒܙܵܟ݂ܘܼ ܠܥܵܠܡܝܼܢ. ܕܝܼܠܵܟ݂ ܡܵܪܝܵܐ ܙܵܟ݂ܘܼܬ݂ܵܐ. ܒܚܲܫܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܐ. ܙܟܲܝܬܵܝܗܝ ܠܡܵܘܬܵܐ ܡܫܝܼܚܵܐ ܡܲܚܝܵܢܲܢ. ܘܲܗܘܲܝܬ ܢܘܼܚܵܡܵܐ ܘܪܹܫܝܼܬ݂ܵܐ ܠܲܩܝܵܡܬܵܐ ܕܡܝܼܬܹ̈ܐ. ܘܐܸܣܬܲܠܲܩܬ ܒܬܸܫܒܘܿܚܬܵܐ. ܘܥܲܠ ܟܘܼܪܣܲܝ ܫܘܒ݂ܚܵܐ ܡܙܲܝܚܝܼܢ ܠܵܟ݂ ܟܪ̈ܘܼܒܹܐ ܘܲܣܪ̈ܵܦܹܐ. ܘܥܲܡ ܡܲܠܲܐܟܹ̈ܐ ܐܵܦ ܚܢܲܢ ܡܫܲܒܚܝܼܢܲܢ ܠܵܟ݂ ܡܵܪܝܵܐ ܘܐܵܡܪܝܼܢܲܢ. ܠܵܟ݂ ܢܸܒ݂ܪܘܿܟ݂ ܘܠܵܟ݂ ܢܸܣܓܘܿܕ݂. ܟܠ ܡܵܐ ܕܒܲܫܡܝܵܐ ܘܡܵܐ ܕܒܲܐܪܥܵܐ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ: ܬܢܝܼ. ܘܫܲܪܵܐ. ܒܣܵܗܕܹ̈ܐ ܒܪ̈ܝܼܟܹܐ. ܛܵܒ݂ ܡܢ ܟܠܗܘܿܢ ܡܲܫܟܢܵܘ̈ܗܝ ܕܝܲܥܩܘܿܒ݂: ܒܢܵܐ ܥܲܠ ܪܵܘܡܵܐ ܡܲܩܕܫܹܗ. ܕܨܘܼܪܬܵܐ. ܘܲܗܘܵܐ ܡܫܲܡܫܵܢܵܐ ܕܒܹܝܬ݂ ܡܲܩܕܫܵܐ ܘܲܕ݂ܡܲܫܟܢܵܐ ܕܲܫܪܵܪܵܐ: ܘܥܲܠ ܠܡܲܫܟܢܵܐ ܪܲܒܵܐ ܘܲܡܫܲܠܡܵܢܵܐ ܕܠܵܐ ܥܒ݂ܝܼܕ݂ ܒܐܝܼܕܲܝ̈ܵܐ. ܠܡܲܫܟܢܵܐ ܕܲܫܪܵܪܵܐ ܗܵܘ ܕܲܩܒܲܥ ܐܲܠܵܗܵܐ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ. ܠܬܲܡܵܢ ܐܸܬܬܪܝܼܡ ܦܵܪܘܿܩܲܢ. ܕܥܲܠ ܝܲܡܝܼܢܵܐ ܢܸܬ݂ܓܲܐܸܐ. ܘܢܸܬ݂ܩܲܒܲܠ ܐܲܝܟ݂ ܩܘܼܪܒܵܢܵܐ. ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܚܠܵܦ ܟܠܲܢ. ܕܲܢܚܲܪܲܪ ܠܓܸܢܣܵܐ ܕܡܵܝܘܿܬܹ̈ܐ: ܬܢܝܼ. ܫܲܒܲܚ. ܒܗܵܘ ܒܵܪܘܿܝܵܐ. ܣܘܼܠܵܩܹܗ ܕܲܡܫܝܼܚܵܐ ܡܲܠܟܲܢ ܫܒ݂ܝܼܚ. ܘܲܡܪܲܡܪܲܡ ܒܬܸܫܒܘܿܚܬܵܐ. ܘܪܵܡ ܗ݇ܘܼ ܘܲܡܥܲܠܲܝ ܘܪܹܫܵܐ ܐܝܼܬ݂ܵܘܗܝ ܕܟ݂ܠܗܘܿܢ ܥܹܐܕܹ̈ܐ. ܗܘ ܓܹܝܪ ܦܲܪܩܲܢ ܡܢ ܡܵܘܬܵܐ ܘܲܚܒ݂ܵܠܵܐ. ܘܫܲܝܸܢ ܥܲܡܲܢ ܥܸܠܵܝܹ̈ܐ ܘܬܲܚܬܵܝܹ̈ܐ. ܘܟ݂ܠܹܗ ܣܵܟ݂ܵܐ ܕܙܲܕܝܼܩܘܼܬ݂ܵܐ ܫܲܡܠܝܼ. ܘܦܘܼܪܩܵܢܵܐ ܪܲܒܵܐ ܗܘ ܒܲܩܢܘܿܡܹܗ ܥܒܲܕ݂ ܠܓܸܢܣܲܢ. ܘܲܠܥܸܠ ܠܲܫܡܲܝܵܐ ܐܸܣܬܲܠܲܩ. ܘܗܵܟܲܢܵܐ ܬܘܼܒ݂ ܐܝܼܬ݂ܠܹܗ ܕܢܹܐܬܸܐ. ܘܡܢ ܡܗܲܝܡܢܹ̈ܐ ܒܕܸܚܠܵܐ ܡܸܬܬܵܘܕܸܐ. ܕܲܢܕ݂ܘܼܢ ܠܚܲܝܹ̈ܐ ܐܵܦ ܠܡܝܼܬܹ̈ܐ. ܠܘܵܬܹܗ ܢܹܐܡܲܪ ܟܠܲܢ. ܗܵܘ ܕܐܸܬ݂ܵܐ ܘܲܥܬ݂ܝܼܕ݂ ܕܢܹܐܬܸܐ ܡܵܪܹܟܠ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂.
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܟ݂ܠܗܘܿܢ ܥܹܐܕܹ̈ܐ ܡܵܪ̈ܵܢܵܝܹܐ. ܠܲܡܕܲܒ݁ܪܵܢܘܼܬ݂ܵܟ݂ ܡܵܪܝ ܬܡܝܼܗܬܵܐ ܘܠܵܐ ܡܸܬ݂ܡܲܠܠܵܢܝܼܬ݂ܵܐ: ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܘܒܲܚܢܵܢܵܐ ܐܸܬ݂ܓܲܡܪܲܬ݀ ܘܐܸܣܬܲܥܪܲܬ݀ ܘܐܸܫܬܲܡܠܝܲܬ݀ ܠܚܘܼܕܵܬܹܗ ܘܲܠܦܘܼܪܩܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ ܚܲܠܵܫܵܐ: ܒܪܹܫܝܼܬ݂ܵܐ ܕܡܸܢܲܢ: ܡܲܣܩܝܼܢܲܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬܵܐ ܘܣܸܓ݂ܕܬ݂ܵܐ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܩܵܢܘܿܢܵܐ. ܒܣܘܿܓ݂ܵܐܐ. ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡܢ ܫܡܲܝܵܐ* ܫܲܒܚܘܼܗܝ ܒܲܡܪ̈ܵܘܡܹܐ* ܒܝܵܘܡܵܐ ܕܐܸܣܬܲܠܲܩ ܪܹܫܝܼܬܲܢ ܒܫܘܼܒ݂ܚܵܐ ܠܲܫܡܲܝܵܐ. ܗܘܵܬ݂ ܬܲܪܥܘܼܬ݂ܵܐ ܒܹܝܬ݂ ܐܲܪ̈ܥܵܢܹܐ ܠܲܫܡܲܝܵܢܹ̈ܐ. ܢܸܚܕܸܐ ܟܝܵܢܲܢ ܚܲܠܵܫܵܐ ܕܙܲܝܚܘܼܗܝ ܪ̈ܘܼܚܵܢܹܐ.
	ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡܢ ܫܡܲܝܵܐ* ܫܲܒܚܘܼܗܝ ܒܲܡܪ̈ܵܘܡܹܐ* ܫܲܒܚܘܼܗܝ ܟܠܗܘܿܢ ܡܲܠܲܐܟ݂ܵܘ̈ܗܝ* ܫܲܒܚܘܼܗܝ ܟܠܗܘܿܢ ܚܲܝ̈ܠܵܘܵܬܹܗ* ܫܲܒܚܘܼܗܝ ܫܸܡܫܵܐ ܘܣܲܗܪܵܐ* ܫܲܒܚܘܼܗܝ ܟܠ ܟܵܘܟܒܹ̈ܐ ܘܢܘܼܗܪܵܐ* ܫܲܒܚܘܼܗܝ ܫܡܲܝ̈ ܫܡܲܝܵܐ* ܡܲܝ̈ܵܐ ܕܲܠܥܸܠ ܡܢ ܫܡܲܝ̈ܵܐ ܢܫܲܒܚܘܼܢ ܠܲܫܡܹܗ ܕܡܵܪܝܵܐ* ܡܸܛܠ ܕܗܘ ܐܸܡܲܪ ܘܲܗܘܵܘ ܘܗܘ ܦܩܲܕ݂ ܘܐܸܬ݂ܒܪܝܼܘ* ܘܐܲܩܝܼܡ ܐܸܢܘܿܢ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ* ܢܵܡܘܿܣܵܐ ܝܲܗ݉ܒ݂ ܘܠܵܐ ܥܵܒܲܪ* ܒܝܵܘܡܵܐ ܕܐܸܣܬܲܠܲܩ ܪܹܫܝܼܬܲܢ ܒܫܘܼܒ݂ܚܵܐ ܠܲܫܡܲܝܵܐ. ܗܘܵܬ݂ ܬܲܪܥܘܼܬ݂ܵܐ ܒܹܝܬ݂ ܐܲܪ̈ܥܵܢܹܐ ܠܲܫܡܲܝܵܢܹ̈ܐ. ܢܸܚܕܸܐ ܟܝܵܢܲܢ ܚܲܠܵܫܵܐ ܕܙܲܝܚܘܼܗܝ ܪ̈ܘܼܚܵܢܹܐ.
	ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡܢ ܐܲܪܥܵܐ* ܬܲܢܝܼܢܹ̈ܐ ܘܟ݂ܠܗܘܿܢ ܬܗܘܿܡܹ̈ܐ* ܒܣܘܼܠܵܩܵܟ݂ ܡܢ ܐܲܪܥܵܐ ܠܲܫܡܲܝܵܐ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܐܸܚܲܕ݂ ܢܘܼܡܹܐ ܓܸܢܣܲܢ ܫܝܼܛܵܐ ܒܪܵܘܡܵܐ ܕܲܠܥܸܠ. ܘܗܵܐ ܡܸܙܕܲܝܲܚ ܡܢ ܥܝܼܪܹ̈ܐ ܘܟܸܢܫܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ.
	ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡܢ ܐܲܪܥܵܐ* ܬܲܢܝܼܢܹ̈ܐ ܘܟ݂ܠܗܘܿܢ ܬܗܘܿܡܹ̈ܐ* ܢܘܼܪܵܐ ܘܒܲܪܕ݂ܵܐ ܘܬܲܠܓܵܐ ܘܲܓ݂ܠܝܼܕ݂ܵܐ* ܪ̈ܘܼܚܹܐ ܘܥܲܠܥܵܠܹ̈ܐ ܥܒ݂ܵܕ݂ܵܐ ܕܡܸܠܬܹܗ* ܛܘܼܪܹ̈ܐ ܘܟ݂ܠܗܹܝܢ ܪ̈ܵܡܵܬ݂ܵܐ* ܐܝܼܠܵܢܹ̈ܐ ܕܦܹܐܪܹ̈ܐ ܘܟ݂ܠܗܘܿܢ ܐܲܪ̈ܙܹܐ* ܚܲܝܘܬ݂ܵܐ ܘܟ݂ܠܵܗ̇ ܒܥܝܼܪܵܐ* ܪ̈ܲܚܫܵܐ ܘܛܲܝܪܵܐ ܘܦܵܪܲܚܬ݂ܵܐ* ܡܲܠ̈ܟܹܐ ܕܐܲܪܥܵܐ ܘܟ݂ܠܗܹܝܢ ܐܸܡ̈ܘܵܬ݂ܵܐ* ܪܵܘܪ̈ܒ݂ܵܢܹܐ ܘܟ݂ܠܗܘܿܢ ܕܲܝܵܢܹ̈ܐ ܕܐܲܪܥܵܐ* ܓܲܕܘܼܕܹ̈ܐ ܘܲܒ݂ܬ݂ܘܼ̈ܠܵܬ݂ܵܐ* ܣܵܒܹ̈ܐ ܘܲܥܠܲܝܡܹ̈ܐ ܢܫܲܒܚܘܼܢ ܠܲܫܡܹܗ ܕܡܵܪܝܵܐ* ܡܸܛܠ ܕܪܲܒܘܼ ܫܡܹܗ ܒܲܠܚܘܿܕ݂ܵܘܗܝ* ܘܫܘܼܒ݂ܚܹܗ ܒܐܲܪܥܵܐ ܘܒܲܫܡܲܝܵܐ* ܡܪܝܼܡ ܩܲܪܢܵܐ ܕܥܲܡܹܗ ܘܬܸܫܒܘܿܚܬܵܐ ܕܟ݂ܠܗܘܿܢ ܙܲܕܝܼ̈ܩܵܘܗܝ* ܠܲܒ݂ܢܲܝ̈ ܝܼܣܪܵܝܹܠ ܥܲܡܵܐ ܕܩܲܪܝܼܒ݂ ܠܹܗ* ܒܣܘܼܠܵܩܵܟ݂ ܡܢ ܐܲܪܥܵܐ ܠܲܫܡܲܝܵܐ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܐܸܚܲܕ݂ ܢܘܼܡܹܐ ܓܸܢܣܲܢ ܫܝܼܛܵܐ ܒܪܵܘܡܵܐ ܕܲܠܥܸܠ. ܘܗܵܐ ܡܸܙܕܲܝܲܚ ܡܢ ܥܝܼܪܹ̈ܐ ܘܟܸܢܫܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ.
	ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܒܩܘܼܕ݂ܫܹܗ* ܫܲܒܚܘܼܗܝ ܒܲܪܩܝܼܥܵܐ ܕܥܘܼܫܢܹܗ* ܝܵܘܡܵܢܵܐ ܐܸܣܬܲܠܲܩ ܪܹܫܝܼܬܲܢ ܒܫܘܼܒ݂ܚܵܐ ܠܲܫܡܲܝܵܐ. ܘܝܲܒܸܒ݂ܘ ܘܲܩܥܵܘ ܟܸܢܫܹ̈ܐ ܟܠܗܘܿܢ ܕܲܫܡܲܝܵܢܹ̈ܐ. ܒܪܝܼܟ݂ܘܼ ܡܲܠܟܵܐ ܕܬܸܫܒܘܿܚܬܵܐ ܕܲܣܠܸܩ ܘܚܲܕܝܼ ܠܟ݂ܠ.
	ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܒܩܘܼܕ݂ܫܹܗ* ܫܲܒܚܘܼܗܝ ܒܲܪܩܝܼܥܵܐ ܕܥܘܼܫܢܹܗ* ܫܲܒܚܘܼܗܝ ܒܚܲܝܠܬ݂ܵܢܘܼܬܹܗ* ܫܲܒܚܘܼܗܝ ܒܣܘܿܓ݂ܐܵܐ ܕܪܲܒܘܼܬܹܗ* ܫܲܒܚܘܼܗܝ ܒܩܵܠܵܐ ܕܩܲܪܢܵܐ* ܫܲܒܚܘܼܗܝ ܒܩܝܼܬ݂ܵܪܹ̈ܐ ܘܲܒ݂ܟܸܢܵܪܹ̈ܐ* ܫܲܒܚܘܼܗܝ ܒܲܦܠܲܓܹ̈ܐ ܘܒܲܪ̈ܒ݂ܝܼܥܹܐ* ܫܲܒܚܘܼܗܝ ܒܡܸܢܹ̈ܐ ܚܲܠ̈ܝܵܬ݂ܵܐ* ܫܲܒܚܘܼܗܝ ܒܨܸܨ̈ܠܲܝ ܫܸܡܥܵܐ ܫܲܒܚܘܼܗܝ ܒܩܵܠܵܐ ܘܒܲܩܥܵܬ݂ܵܐ* ܒܟ݂ܠ ܢܸܫܡܵܐ ܢܫܲܒܲܚ ܠܡܵܪܝܵܐ* ܝܵܘܡܵܢܵܐ ܐܸܣܬܲܠܲܩ ܪܹܫܝܼܬܲܢ ܒܫܘܼܒ݂ܚܵܐ ܠܲܫܡܲܝܵܐ. ܘܝܲܒܸܒ݂ܘ ܘܲܩܥܵܘ ܟܸܢܫܹ̈ܐ ܟܠܗܘܿܢ ܕܲܫܡܲܝܵܢܹ̈ܐ. ܒܪܝܼܟ݂ܘܼ ܡܲܠܟܵܐ ܕܬܸܫܒܘܿܚܬܵܐ ܕܲܣܠܸܩ ܘܚܲܕܝܼ ܠܟ݂ܠ.

	ܫܲܒܲܚ* ܥܵܠܲܡ* ܒܥܹܐܕ݂ܵܐ ܫܒ݂ܝܼܚܵܐ ܕܲܣܠܸܩ ܒܹܗ ܪܲܒ ܟܘܼܡܪܹ̈ܐ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܢܸܚܠܘܿܛ ܩܵܠܲܝ̈ܢ ܥܲܡ ܪ̈ܘܼܚܵܢܹܐ ܘܢܸܙܡܲܪ ܫܘܼܒ݂ܚܵܐ. ܠܐܝܼܬ݂ܝܵܐ ܕܪܲܡܪܸܡ ܠܲܟ݂ܝܵܢܲܢ ܘܐܵܘܬܒܹܗ ܒܲܫܡܲܝܵܐ*
	ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܡܢ ܫܡܲܝܵܐ* ܫܲܒܚܘܼܗܝ ܒܲܡܪ̈ܵܘܡܹܐ* ܥܠܲܝܟܘܼ ܥܵܕ݂ܘܿܪܲܢ ܣܵܝܡܝܼܢܲܢ ܟܠܫܵܥ ܬܘܼܟ݂ܠܵܢܲܢ. ܘܡܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܗܵܐ ܫܵܐܠܝܼܢܲܢ ܚܢܵܢܵܐ ܘܪ̈ܲܚܡܹܐ ܘܫܘܼܒ݂ܩܵܢ ܚܵܘܒܹ̈ܐ. ܐܲܫܦܲܥ ܒܐܝܼܕܲܝ̈ܢ ܡܵܘܗܲܒ݂ܬ݂ܵܟ݂ ܘܚܘܼܢ ܠܚܲܛܵܝܘܼܬܲܢ.
	ܗܦܵܟ݂ܬܵܐ. ܐܲܝܟ݂ ܣܘܼܕܵܪܹ̈ܐ ܦܪܲܣܘ ܓܸܦܲܝ̈ܗܘܿܢ ܥܲܠ ܐܲܦܲܝ̈ܗܘܿܢ: ܕܠܵܐ ܢܸܣܬܲܝ̈ܛܵܢ ܡܢ ܦܲܐܝܘܼܬ݂ܵܐ ܕܗܸܕ݂ܪܵܐ ܕܫܘܼܒ݂ܚܹܗ* ܦܪܲܚܘ ܪ̈ܘܼܚܵܢܹܐ ܘܛܵܣܘ ܢܘܼܪ̈ܵܢܹܐ ܪܗܸܛܘ ܩܲܠܝܼܠܹ̈ܐ: ܘܲܣܒܲܪܘ ܕܲܡܛܵܘ ܘܩܲܕܸܡ ܐܸܢܘܿܢ ܪܸܡܙܹܗ ܟܲܣܝܵܐ.
	ܘܐܵܡܪܝܼܢ ܬܸܫܒܘܿܚܬܵܐ ܗܵܕܹܐ ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܡܵܪܝ ܢܲܪܣܲܝ ܡܲܠܦܵܢܵܐ:
	ܬܵܘܕܝܼ ܠܛܵܒ݂ܵܐ ܕܚܲܪܲܪ ܓܸܢܣܲܢ: ܡܢ ܥܲܒ݂ܕܘܼܬ݂ܵܐ ܕܒ݂ܝܼܫܵܐ ܘܡܵܘܬܵܐ* ܫܲܝܸܢ ܥܲܡܲܢ ܟܸܢܫܲܝ̈ ܪܵܘܡܵܐ: ܕܪܲܓܝܼܙܝܼܢ ܗ݇ܘܵܘ ܡܸܛܠ ܥܵܘܠܲܢ* ܒܪܝܼܟ݂ ܚܲܢܵܢܵܐ ܕܟܲܕ݂ ܠܵܐ ܒܥܲܝܢܵܝܗܝ: ܢܦܲܩ ܒܲܒ݂ܥܵܬܲܢ ܘܲܚܕ݂ܝܼ ܒܚܲܝܲܝ̈ܢ* ܘܨܵܪ ܕܸܡ̈ܘܵܬ݂ܵܐ ܕܐܲܒܝܼܕ݂ܘܼܬܲܢ: ܘܲܕ݂ܦܘܼܢܵܝܲܢ ܒܥܸܪܒܵܐ ܕܲܛܥܵܐ* ܝܵܪܬܵܐ ܘܲܒ݂ܪܵܐ ܩܪܵܝܗܝ ܠܲܟ݂ܝܵܢܲܢ: ܕܲܛܥܵܐ ܘܲܦܢܵܐ ܘܡܝܼܬ݂ ܘܐܸܬ݂ܢܲܚܲܡ* ܘܚܲܕܝܼ ܠܟܸܢܫܹ̈ܐ ܪ̈ܘܼܚܵܢܵܝܹܐ: ܒܲܬ݂ܝܵܒ݂ܘܼܬܲܢ ܘܲܒ݂ܢܘܼܚܵܡܲܢ* ܠܵܐ ܡܸܬ݂ܡܲܠܲܠ ܚܘܼܒܵܐ ܪܲܒܵܐ: ܕܚܵܘܝܼ ܨܹܐܕܲܝܢ ܪܵܚܹܡ ܓܸܢܣܲܢ* ܕܡܸܢܹܗ ܕܓܸܢܣܲܢ ܥܒܲܕ݂ ܡܸܨܥܵܝܵܐ: ܘܪܲܥܝܼ ܠܥܵܠܡܵܐ ܥܲܡ ܪܲܒܘܼܬܹܗ* ܪܲܒܵܐ ܗ̄ܝܼ ܡܸܢܲܢ ܘܡܢ ܟܠ ܒܸܪ̈ܝܵܢ: ܚܕܲܬܵܐ ܕܲܣܥܲܪ ܨܹܝܕ݂ ܐ݇ܢܵܫܘܼܬܲܢ* ܕܥܲܒ݂ܕܹܗ ܠܦܲܓ݂ܪܲܢ ܗܲܝܟܲܠ ܩܘܼܕ݂ܫܵܐ: ܕܲܢܡܲܠܸܐ ܒܹܗ ܣܸܓ݂ܕܲܬ݂ ܟܠܵܐ* ܬܵܘ ܐܲܪ̈ܥܵܢܹܐ ܘܲܫܡܲܝܵܢܹ̈ܐ: ܬܗܲܪܘ ܘܐܸܬܕܲܡܲܪܘ ܒܪܲܒܘܼܬ݂ ܕܲܪܓ݂ܵܐ* ܕܲܡܛܵܐ ܓܸܢܣܲܢ ܠܪܵܘܡܵܐ ܪܲܒܵܐ: ܕܐܲܠܵܗܘܼܬ݂ܵܐ ܕܠܵܐ ܡܸܬܕܲܪܟܵܐ* ܫܡܲܝܵܐ ܘܐܲܪܥܵܐ ܘܟ݂ܠܡܵܐ ܕܲܒ݂ܗܘܿܢ: ܢܵܘܕܘܿܢ ܥܲܡܲܢ ܠܡܵܘܪܸܒ݂ ܓܸܢܣܲܢ* ܕܚܲܕܸܬ݂ ܨܲܠܡܲܢ ܘܲܥܛܵܐ ܥܵܘܠܲܢ: ܘܲܩܪܵܢ ܒܲܫܡܹܗ ܘܫܲܥܒܸܕ݂ ܠܲܢ ܟܠ* ܫܵܘܹܐ ܠܫܘܼܒ݂ܚܵܐ ܡܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܗ̇ܘ ܕܐܲܪܝܼܡܲܢ ܠܥܸܠ ܡܢ ܟܠܵܐ* ܘܢܹܐܡܲܪ ܟܠܲܢ ܠܹܗ ܬܸܫܒܘܿܚܬܵܐ: ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ*
ܟܵܪܘܿܙܘܼܬ݂ܵܐ. ܢܩܘܼܡ ܫܲܦܝܼܪ. ܡܫܝܼܚܵܐ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܫܲܡܠܝܼ ܟܠܵܗ̇ ܡܕܲܒܪܵܢܘܼܬ݂ܵܐ ܕܲܥܠܵܘܗܝ ܐܲܟ݂ܪܸܙܘ ܢܒ݂ܝܼܹ̈ܐ. ܡܫܝܼܚܵܐ ܕܲܣܠܸܩ ܒܫܘܼܒ݂ܚܵܐ ܘܗܸܕ݂ܪܵܐ ܠܲܫܡܲܝܵܐ. ܘܝܼܬܸܒ݂ ܡܢ ܝܲܡܝܼܢ ܐܲܒ݂ܘܼܗܝ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ. ܡܫܝܼܚܵܐ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܬܡܝܼܗܵܐ ܐܲܬ݂ܗܲܪ ܠܲܫܠܝܼܚܵܘ̈ܗܝ. ܘܩܵܡܘ ܟܲܕ݂ ܚܵܝܪܝܼܢ ܒܹܗ. ܘܡܸܚܕ݂ܵܐ ܐܸܬ݂ܟܲܣܝܼ ܡܢ ܥܲܝܢܲܝ̈ܗܘܿܢ. ܡܫܝܼܚܵܐ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܫܲܥܒܸܕ݂ ܠܟ݂ܠܗܘܿܢ ܒܥܸܠܕܒ݂ܵܒ݂ܵܘ̈ܗܝ ܬܚܹܝܬ݂ ܟܘܼܒ݂ܫܵܐ ܕܪܸ̈ܓ݂ܠܵܘܗܝ. ܡܫܝܼܚܵܐ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܐܲܪܝܼܡܹܗ ܠܲܟ݂ܝܵܢܲܢ ܠܥܸܠ ܡܢ ܟܠܗܘܿܢ ܐܲܪ̈ܟܘܿܣ ܘܫܘܼܠܛܵܢܹ̈ܐ. ܘܐܲܫܠܛܹܗ ܥܲܠ ܟܠ ܡܵܐ ܕܒܲܫܡܲܝܵܐ ܘܡܵܐ ܕܒܲܐܪܥܵܐ. ܡܫܝܼܚܵܐ ܕܲܚܠܵܦ ܐܲܪܥܵܐ ܕܡܲܦܩܵܐ ܟܘܼܒܹ̈ܐ. ܠܲܢ ܡܵܝܘܿܬܹ̈ܐ ܥܒܲܕ݂ ܕܘܼܝܵܪܵܐ ܒܲܫܡܲܝܵܐ. ܕܲܢܕܲܝܲܪ ܒܚܲܝܹ̈ܐ ܠܵܐ ܡܵܝܘܿܬܹ̈ܐ. ܡܫܝܼܚܵܐ ܕܲܚܠܵܦ ܥܘܼܡܪܵܐ ܡܸܬ݂ܚܲܙܝܵܢܵܐ. ܥܲܠܝܼ ܠܲܟ݂ܝܵܢܲܢ ܡܸܬ݂ܚܲܒܠܵܢܵܐ ܠܥܘܼܡܪܵܐ ܠܵܐ ܡܸܬ݂ܚܲܙܝܵܢܵܐ ܒܣܘܼܠܵܩܹܗ. ܡܫܝܼܚܵܐ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܫܒ݂ܵܐ ܫܒ݂ܝܬ݂ܵܐ ܡܢ ܫܲܒܵܝܹܗ ܕܓܸܢܣܲܢ. ܘܝܲܗ݇ܒ݂ ܡܵܘܗ̈ܒ݂ܵܬ݂ܵܐ ܠܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܕܲܡܗܲܝܡܢܝܼܢ ܒܹܗ. ܡܫܝܼܚܵܐ ܕܐܸܣܬܲܠܲܩ ܒܲܥܢܵܢܹ̈ܐ ܠܲܫܡܲܝܵܐ. ܘܲܥܠܲܝܗܹܝܢ ܥܬ݂ܝܼܕ݂ ܕܢܹܐܬܸܐ ܒܬܸܫܒܘܿܚܬܵܐ. ܡܫܝܼܚܵܐ ܕܝܼܬܸܒ݂ ܡܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܵܐ ܕܫܲܠܚܹܗ ܘܒ݂ܵܥܹܐ ܚܠܵܦܲܝܢ. ܡܫܝܼܚܵܐ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܦܲܩܸܕ݂ ܠܲܫܠܝܼܚܵܘ̈ܗܝ ܕܠܵܐ ܢܸܦܩܘܿܢ ܡܢ ܐܘܿܪܸܫܠܸܡ. ܥܕܲܡܵܐ ܕܢܸܒܠܫܘܼܢ ܚܲܝܠܵܐ ܡܢ ܪܵܘܡܵܐ. ܡܫܝܼܚܵܐ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܕܠܲܫܡܲܝܵܐ ܪܲܡܪܸܡ ܕܲܚܝܼܚܲܢ ܡܢ ܐܲܪܥܵܐ ܠܲܫܡܲܝܵܐ. ܘܝܲܗ݇ܒ݂ܠܲܢ ܕܲܪܓ݂ܵܐ ܪܲܒܵܐ ܕܟ݂ܵܗܢܘܼܬ݂ܵܐ. ܥܲܠ ܚܘܼܠܡܵܢܵܐ. ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܕܲܒ݂ܪ̈ܲܚܡܵܘܗܝ ܪܲܚܸܡ ܥܠܲܝܢ. ܘܲܠܓܸܢܣܲܢ ܚܲܢ ܒܲܚܢܵܢܹܗ. ܘܐܲܗܦܟܹܗ ܒܪ̈ܲܚܡܵܘܗܝ ܠܬܲܪܥܘܼܬ݂ܵܐ ܒܲܡܫܝܼܚܹܗ. ܗܵܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܡܢ ܬܸܓ݂ܡܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ. ܘܒܲܐܪܥܵܐ ܡܸܣܬܓܸܕ݂ ܡܢ ܟܸܢܫܹ̈ܐ ܕܐܲܪ̈ܥܵܢܹܐ. ܦܪܘܿܩ ܡܵܪܝ ܠܩܵܪ̈ܝܲܝ ܫܡܵܟ݂. ܘܦܲܢܵܐ ܫܹܐ̈ܠܵܬ݂ܵܐ ܕܐܲܝܠܹܝܢ ܕܫܵܐܠܝܼܢ ܠܵܟ݂. ܘܗܲܒ݂ܠܲܢ ܫܲܝܢܵܐ ܘܲܫܠܵܡܵܐ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ. ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
