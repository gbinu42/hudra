ܛܲܟ݂ܣܵܐ ܕܬܸܫܡܸܫܬܵܐ
ܕܠܸܠܝܵܐ ܘܕܨܲܦܪܵܐ
 ܒܫܸܡ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕܫܵܐ ܠܥܵܠܡܝܼܢ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟ݂: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟ݂: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬ݂ܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹܝ̇ܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.

ܨܠܘܿܬ݂ܵܐ. ܚܲܝܸܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܠܲܡܚܝܼܠܘܼܬܲܢ ܒܲܚܢܵܢܵܟ݂: ܘܠܲܒܸ݁ܒ݂ ܘܥܲܕܲܪ ܠܲܙܥܘܿܪܘܼܬ݂ ܢܲܦ̮ܫܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܥܝܼܪ ܕܲܡܟ݁ܘܼܬ݂ ܪܸܥܝܵܢܲܢ: ܘܐܲܩܸܠ ܝܘܼܩܪܵܐ ܕܗܲܕܵܡܲܝ̈ܢ: ܘܐܲܫܝܼܓ݂ ܘܲܡܪܘܿܩ ܨܵܐܬ݂ܵܐ ܕܚܵܘܒܲܝ̈ܢ ܘܕܲܚܛܵܗܲܝ̈ܢ: ܘܐܲܢܗܲܪ ܠܚܸܫܘܿܟ݂ܘܼܬܹܗ ܕܡܲܕ݁ܥܲܢ: ܘܐܵܘܫܸܛ ܘܫܲܪܲܪ ܘܗܲܒ݂ ܒܲܢ ܚܲܝܠܵܐ ܘܐܝܼܕ݂ܵܐ ܕܥܘܼܕ݂ܪ̈ܵܢܹܐ: ܕܒ݂ܵܗ̇ ܢܩܘܼܡ ܘܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܨܠܘܿܬ݂ܵܐ. ܚܲܝܠܵܐ ܡܵܪܝ ܟܲܣܝܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܥܘܼܕ݂ܪܵܢܵܐ ܬܡܝܼܗܵܐ ܕܡܵܪܘܼܬ݂ܵܟ݂: ܘܣܘܼܝܵܥܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂: ܗܘ ܢܚܲܝܸܠ ܠܲܡܚܝܼܠܘܼܬܹܗ ܕܲܟ݂ܝܵܢܲܢ ܚܲܠܵܫܵܐ: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܘܡܢ ܒܵܬܲܪ ܕܲܫܠܸܡܘ ܗܘܼܠܵܠܹ̈ܐ: ܐܸܢ ܡܸܬ݂ܥܒܸܕ݂ ܩܘܼܪܒܵܢܵܐ ܒܝܵܘܡܵܐ ܗ̇ܘ. ܐܵܡ̇ܪܝܼܢ ܨܠܘܿܬ݂ܵܐ ܕܩܵܠ̈ܝܵܬ݂ܵܐ.
ܨܠܘܿܬ݂ܵܐ. ܠܲܟ݂ܝܵܢ ܟܲܣܝܘܼܬ݂ܵܟ݂ ܕܠܵܐ ܡܸܬ݁ܕ݁ܪܸܟ݂ ܘܠܵܐ ܡܸܣܬܲܝܲܟ݂ ܡܢ ܚܘܼܫܵܒ݂ܵܐ ܘܣܘܼܟܵܠܵܐ ܕܟ݂ܠܗܘܿܢ ܥܒ݂ܝܼܕܹ̈ܐ: ܝܵܐܹ̇ܐ ܡܵܪܝ ܫܘܼܒ݂ܚܵܐ ܘܙܵܕ݂ܩ̇ܝܼܢ ܩܵܠܹ̈ܐ ܕܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ: ܘܦܲܐܝܵܐ ܘܡܸܬ݁ܬ݁ܚܝܼܒ݂ܵܐ ܣܸܓ݂ܕ݁ܬ݂ܵܐ: )ܡܢ ܟܠ ܕܲܒ݂ܪܲܝܬ݁ ܘܐܲܬ݂ܩܸܢܬ݁) ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	ܩܲܠܬ݂ܵܐ. ܐܵܡܪܵܐ ܪܗ݇ܘܿܡܹܐ ܠܕ݂ܵܪܕܵܪ̈ܝܼܢ ܒܲܢܝܵܐ ܥܹܕ݁ܬܵܐ: ܘܠܵܐ ܡܸܬ݂ܚܲܣܢܵܐ ܠܵܐ ܠܡܲܠܟܹ̈ܐ ܘܠܵܐ ܠܫܘܼܠܛܵܢܹ̈ܐ܀ ܐܲܫܘܵܢ ܡܵܪܲܢ ܠܗܵܘ ܟܸܢܵܪܵܐ ܕܲܙܡܲܪ ܕܵܘܝܼܕ݂: ܘܲܩܥܵܘ ܥܝܼܪܹ̈ܐ ܒܗܘܼܠܵܠܲܝ̈ܗܘܿܢ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ܀ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܡܵܪܝܵܐ ܕܲܫܪܸܐ ܒܢܘܼܗܪܵܐ: ܬܠܵܬ݂ܵܐ ܩܘܼܕ݂ܫܹ̈ܐ ܩܥܵܘ ܫܲܘܝܵܐܝܼܬ݂ ܚܲܕ݂ ܫܘܼܠܛܵܢܵܐ.
ܡܲܙܡܘܿܪܵܐ. ܐܲܒܲܥܠܹܒ݁ܝ ܦܸܬ݂ܓ݂ܵܡܹ̈ܐ ܛܵܒܹ̈ܐ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܘܐܹܡܲܪ ܥܒ݂ܵܕܲܝ̈ ܠܡܲܠܟܵܐ*
ܠܸܫܵܢܝ ܩܲܢܝܹܗ ܕܣܵܦܪܵܐ ܡܗܝܼܪܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܫܲܦܝܼܪ ܒܚܸܙܘܹܗ ܡܢ ܒܢܲܝ̈ܢܵܫܵܐ*
ܐܸܬ݂ܢܣܸܟ݂ܘ ܪ̈ܲܚܡܹܐ ܥܲܠ ܣܸܦܘ̈ܵܬ݂ܵܟ݂. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܡܸܛܠ ܗܵܢܵܐ ܒܲܪܟ݂ܵܟ݂ ܐܲܠܵܗܵܐ ܠܥܵܠܲܡ*
ܐܲܪܡܵܐ ܣܲܝܦܵܐ ܒܚܲܨܲܝ̈ܟ ܓܲܢ݇ܒܵܪܵܐ ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܗܸܕ݂ܪܵܟ݂ ܘܫܘܼܒ݂ܚܵܟ݂ ܘܫܘܼܒ݂ܚܵܟ݂ ܙܵܟܹ݁ܐ*
ܪܟܲܒ݂ ܥܲܠ ܡܸܠܬ݂ܵܐ ܕܩܘܼܫܬܵܐ ܘܡܘܼܟܵܟ݂ܵܐ ܕܙܲܕܝܼܩܘܼܬ݂ܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܢܵܡܘܿܣܵܟ݂ ܒܕܸܚܠܬ݂ܵܐ ܕܝܲܡܝܼܢܵܟ݂*

ܡܵܘܬ݁ܒ݂ܵܐ
      ܡܵܘܬ݁ܒ݂ܵܐ. ܒܡܲܒ݁ܘܼܥܚܲܝܹ̈ܐ. ܬܸܥܫܲܢ ܐܝܼܕ݂ܵܟ݂ ܘܬܸܦܪܩܲܢܝ: ܘܲܦܪܲܩܬ݁ ܒܲܕ݂ܪܵܥܵܟ݂ ܠܥܲܡܵܟ݂. ܒܝܲܕ݂ ܥܘܼܫܢܵܐ ܘܬ݂ܘܼܩܦܵܐ ܕܝܲܡܝܼܢܵܟ݂ ܐܲܢ݇ܬ݁ ܐܸܬ݂ܓ݁ܠܝܼܬ݁ ܘܲܦܪܲܩܬ݁ ܠܲܢ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܠܥܹܕ݁ܬܵܟ݂ ܩܲܝܸܡ ܒܲܚܢܵܢܵܟ݂ ܘܫܲܝܸܢ ܒܪ̈ܲܚܡܲܝܟ. ܘܲܠܟ݂ܠܗܘܿܢ ܐܲܝܠܹܝܢ ܕܥܵܡܪܝܼܢ ܒܵܗ ܒܲܪܸܟ݂ ܘܢܲܛܲܪ܀ ܚܲܝܠܵܟ݂ ܪܲܒܵܐ ܦܵܪܘܿܩܲܢ. ܗܲܒ݂ ܘܐܲܫܪܵܐ ܒܟ݂ܠܗܹܝܢ ܥܹܕܵܬ݂̈ܵܐ. ܘܟ݂ܵܗܢܹ̈ܐ ܕܲܒ݂ܗܹܝܢ ܒܕܸܚܠܵܐ ܢܫܲܡܫܘܼܢ. ܟܲܕ݂ ܡܵܘܕܹ݁ܝܢ ܠܵܟ݂. ܘܐܲܢ݇ܬ݁ ܗܲܒ݂ܠܲܢ ܕܲܕ݂ܠܵܐ ܫܲܠܘܵܐ. ܒܐܵܘܝܘܼܬ݂ܵܐ ܘܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ. ܕܟ݂ܠܲܢ ܒܝܲܕ݂ ܚܘܼܒܵܐ ܫܲܠܡܵܐ. ܢܸܬ݂ܟܲܫܲܦ ܟܲܕ݂ ܡܵܘܕܹ݁ܝܢܲܢ ܠܵܟ݂܀ ܘܢܸܩܥܸܐ ܟܠܲܢ ܡܵܪܹܟ݁ܠ. ܐܲܢ݇ܬ݁ ܥܢܝܼܢ ܒܲܚܢܵܢܵܟ݂ ܘܪܲܚܸܡܥܠܲܝܢ: ܬܢܝܼ. ܒܲܪ݇ܬ݂ ܡܲܠܟܵܐ ܒܫܘܼܒ݂ܚܵܐ ܩܵܡܲܬ݀. ܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ ܙܡܲܪܝ ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܚܲܬ݂ܢܵܐ ܕܡܲܟ݂ܪܹܟ݂ܝ ܘܐܲܥܠܹܟ݂ܝ ܠܩܲܝܛܘܿܢܹܗ. ܘܝܲܗ݇ܒ݂ܠܹܟ݂ܝ ܐ݇ܪ̈ܵܙܵܘܗܝ ܪܲܗܒ݂ܘܿܢܵܐ ܚܘܼܣܵܝܵܐ ܠܝܲܠܕܲܝ̈ܟ݁ܝ. ܘܚܲܣܝܹܟ݂ܝ ܘܩܲܕ݂ܫܹܟ݂ܝ ܒܛܲܝܒ݁ܘܼܬܹܗ ܡܢ ܛܲܡܐܘܼܬ݂ܵܐ ܕܕܲܝܘܹ̈ܐ܀ ܚܘܼܒܹ݁ܗ ܙܲܡܸܢ ܘܲܩܪܵܟ݂ܝ ܠܡܲܥܡܘܿܕܝܼܬܹܗ. ܘܝܲܗ݇ܒ݂ ܠܹܟ݂ܝ ܢܝܼܫܵܐ ܕܙܵܟ݂ܘܼܬ݂ܵܐ. ܨܠܝܼܒܹܗ ܚܲܝܵܐ ܕܒܹܗ ܙܵܟ݂ܝܲܬ݁ܝ. ܠܒ݂ܝܼܫܵܐ ܘܟ݂ܠܗܹܝܢ ܨܸܢܥܵܬܹ̈ܗ. ܒܟܸܢܫܹ̈ܐ ܪ̈ܚܝܼܡܹܐ ܕܲܒ݂ܢܲܝ̈ܟ݁ܝ ܦܘܿܩܝ ܠܐܘܼܪܥܹܗ. ܘܲܩܥܵܝ ܠܹܗ ܐܘܿܫܲܥܢܵܐ. ܒܪܝܼܟ݂ܘܼ ܡܲܠܟܵܐ ܕܡܢ ܕܵܘܝܼܕ݂ ܕܲܙܟ݂ܵܐ ܘܙܵܟ݁ܝܹܟ݂ܝ. ܘܲܥܬ݂ܝܼܕ݂ ܕܢܹܐܬܸܐ. ܒܲܫܡܹܗ ܕܡܵܪܝܵܐ. ܒܫܘܼܒ݂ܚܵܐ ܪܲܒܵܐ. ܘܢܲܣܩܹܟ݂ܝ ܥܲܡܹܗ ܠܲܫܡܲܝܵܐ. ܒܫܘܼܪܵܝܵܐ ܘܲܡܫܲܚܠܦܵܐ. ܡܵܪܝܵܐ ܡܲܢܘܼ ܢܸܥܡܲܪ ܒܡܲܫܟ݁ܢܵܟ݂. ܐܲܒ݂ܵܐ ܕܐܲܫܪܝܼ ܫܟ݂ܝܼܢܬܹ݁ܗ ܒܛܘܼܪ ܣܝܼܢܲܝ. ܘܲܡܠܵܝܗܝ ܩܘܼܕ݂ܫܵܐ ܕܐܝܼܩܵܪܹܗ. ܡܠܝܼܘܜ ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܗܵܢ ܗܲܝܟ݁ܠܵܐ ܕܲܦܪܝܼܫ ܠܐܝܼܩܵܪܵܟ݂. ܠܲܚܠܘܼܠ ܚܘܼܒܵܟ݂ ܘܠܐܝܼܩܵܪ ܫܘܼܒ݂ܚܵܟ݂. ܕܢܸܗܘܸܐ ܠܡܹܐܢܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܠܟ݂ܠ ܕܐܲܠܝܼܨܝܼܢ. ܘܒܹܗ ܬܸܫܬܲܒܲܚ ܐܲܢ݇ܬ݁ ܘܲܒ݂ܪܵܟ݂ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܡܢ ܫܡܲܝܵܢܹ̈ܐ ܘܐܲܪ̈ܥܵܢܵܝܹܐ ܒܟ݂ܠ ܥܸܕܵܢ̈ܝܼܢ. ܡܸܢܵܗ ܘܠܵܗ. ܒܲܪ݇ܬ݂ ܡܲܠܟܵܐ ܒܫܘܼܒ݂ܚܵܐ ܩܵܡܲܬ݀. ܥܹܕ݁ܬܵܐ ܒܦܘܼܡ ܝܲܠܕܹܝ̈ܗ ܬܸܙܡܲܪ ܬܸܫܒ݁ܘܿܚܬܵܐ. ܠܚܲܬ݂ܢܵܐ ܫܒ݂ܝܼܚܵܐ ܕܡܲܟ݂ܪܵܗ ܘܐܲܥܠܵܗ ܠܩܲܝܛܘܿܢܹܗ. ܘܝܲܗ݇ܒ݂ ܠܵܗ ܚܪܘܼܪܹ̈ܐ ܡܢ ܥܲܒ݂ܕܘܼܬ݂ܵܐ ܕܲܦܬܲܟ݂ܪܹ̈ܐ. ܕܲܡܫܲܥܒ݁ܕ݂ܵܐ ܗ݇ܘܵܬ݂ ܠܛܵܥܝܘܼܬ݂ܵܐ. ܘܲܕ݂ܪܲܫ ܠܵܗ ܐܘܼܪܚܵܐ ܕܚܲܝܹ̈ܐ ܥܬ݂ܝܼܕܹ̈ܐ. ܘܛܘܼܒܹ̈ܐ ܕܠܵܐ ܦܵܛܪܝܼܢ. ܘܚܲܙܩܵܗ ܒܙܲܝܢܵܐ ܕܪܘܼܚ ܩܘܼܕ݂ܫܵܐ. ܕܒܹܗ ܚܵܣܢܵܐ ܘܙܵܟ݂ܝܵܐ ܠܚܲܝܠܹܗ ܕܲܛܪܘܼܢܵܐ. ܘܗܵܐ ܚܵܕ݂ܝܵܐ ܘܕ݂ܵܝܨܵܐ ܟܲܠܬ݂ܵܐ ܡܨܲܒܲܬ݁ܬܵܐ. ܒܲܡܫܝܼܚܵܐ ܕܚܲܫ ܥܲܠ ܐܲܦܹܝ̈ܗ ܘܦܲܪܩܵܗ ܒܲܨܠܝܼܒܹܗ. ܕܡܵܐ ܕܐܵܬܹ݁ܐ ܒܫܘܼܒ݂ܚܹܗ ܪܲܒܵܐ. ܡܲܠܒܸ݁ܫ ܠܵܗ ܐܸܣܛܲܠ ܬܸܫܒ݁ܘܿܚܬܵܐ.
ܕܲܨܠܘܿܬ݂ܵܐ. ܥܲܠ ܬܲܪܥܹܗ. ܩܵܠܵܐ: ܝܚ. ܒܪܘܼܟܵܒ݂ ܟܲܕ݂ ܫܲܒ݂ܪܹ̈ܐ.
ܦܬܲܚܠܝܼ ܬܲܪ̈ܥܹܐ ܕܙܲܕܝܼܩܘܼܬ݂ܵܐ. ܥܲܠ ܬܲܪܥܹܗ ܕܲܚܢܵܢܵܟ݂ ܢܵܩܫ̇ܝܼܢ ܥܲܒ݂ܕܲܝ̈ܟ ܡܵܪܝܵܐ ܟܲܕ݂ ܡܸܬ݂ܟܲܫܦܝܼܢ ܘܫܵܐ̇ܠܝܼܢ ܡܸܢܵܟ݂. ܚܢܵܢܵܐ ܘܪ̈ܲܚܡܹܐ. ܕܬܸܥܢܸܐ ܐܸܢܘܿܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܘܐܲܣܵܐ ܟܹܐܒܹ̈ܐ ܕܗܲܕܵܡܲܝ̈ܗܘܿܢ. ܒܣܲܡܵܐ ܪܲܒܵܐ ܕܲܚܢܵܢܵܟ݂. ܫܡܲܥ ܡܵܪܝ ܒܵܥܘܼܬ݂ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܟ ܫܹܐ̈ܠܵܬܲܢ. ܕܐܲܢ݇ܬ݁ܘܼ ܐܝܼܬܲܝܟ ܐܝܼܵܠܵܐ ܕܟ݂ܠ ܕܐܲܠܝܼܨܝܼܢ. ܬܵܐ ܠܥܘܼܕ݂ܪܵܢܲܢ ܪܵܚܹܡ ܐ݇ܢܵܫܵܐ. ܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܕܒܲܝܬܵܝܵܘ̈ܗܝ. ܥܲܕܲܪܲܝܢ ܐܲܠܵܗܵܐ ܦܵܪܘܿܩܲܢ. ܥܘܼܕ݂ܪܵܢܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܡܵܪܵܐ ܛܵܒ݂ܵܐ ܡܠܸܐ ܪ̈ܲܚܡܹܐ. ܢܸܬ݂ܠܵܘܸܐ ܠܲܢ ܒܟ݂ܠ ܥܸܕܵܢܝܼ̈ܢ ܡܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂. ܘܢܸܒ݂ܗܲܬ݂ ܒܝܼܫܵܐ ܡܵܪܘܿܕ݂ܵܐ. ܘܬܸܬ݁ܪܝܼܡ ܩܲܪܢܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܒܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ. ܫܡܲܥ ܚܲܢܵܢܵܐ ܩܵܠ ܬܲܟ݂ܫܲܦ̮ܬܲܢ. ܘܬܸܥܘܿܠ ܩܕ݂ܵܡܲܝܟ ܒܵܥܘܼܬܲܢ. ܘܗܲܒ݂ܠܲܢ ܚܲܝܠܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܕܢܵܘܕܸܐ ܠܲܫܡܵܟ݂. ܘܲܢܙܲܡܲܪ ܠܵܟ݂ ܒܟ݂ܠ ܥܸܕܵܢܝܼ̈ܢ. ܡܵܪܵܐ ܪܵܚܹܡ ܬܲܝܵܒܹ̈ܐ. ܣܲܬܲܪܲܝܢܝ ܡܢ ܥܵܘܠܹܗ ܕܒ݂ܝܼܫܵܐ. ܬܚܹܝܬ݂ ܟܸܢܦܹ̈ܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܡܵܪܵܐ ܛܵܒ݂ܵܐ ܡܠܸܐ ܪ̈ܲܚܡܹܐ. ܣܲܬܲܪ ܐܸܢܘܿܢ ܠܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ ܡܢ ܢܸܟ݂ܝܵܢܹ̈ܐ. ܘܡܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܕܠܵܐ ܚܵܣܲܪ. ܙܘܼܢ ܡܵܙܘܿܢܵܐ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܠܵܟ݂ܘܼ ܩܵܪܹܝܢ ܒܟ݂ܠ ܥܸܕܵܢܝܼ̈ܢ. ܘܚܘܼܪ ܒܵܗ̇ ܒܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬ݂ܵܟ݂. ܕܚܵܝܪܵܐ ܠܲܫܡܵܟ݂ ܘܲܡܣܲܟ݁ܝܵܐ. ܘܲܣܥܘܿܪ ܒܪ̈ܲܚܡܹܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܠܝܲܠܕܹܝ̈ܗ̇ ܕܥܹܕܬܵܐ. ܕܐܲܢ݇ܬ݁ ܚܲܢܵܢܵܐ ܐܲܢ݇ܬ݁ ܘܣܲܓ݁ܝܼ ܪ̈ܲܚܡܹܐ. ܘܠܵܟ݂ ܡܸܬ݁ܬ݁ܚܝܼܒ݂ܵܐ ܬܵܘܕܝܼܬ݂ܵܐ. ܚܛܲܝܢ ܥܲܡ ܐܲܒ݂ܵܗܲܝ̈ܢ ܘܐܲܣܟܸ݁ܠܢ ܕܵܪ̈ܝܼܢ.  ܬܸܟ݂ܪܸܐ ܠܵܟ݂ ܡܵܪܝ ܥܲܠ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܲܫܩܲܠܘ ܦܲܓ݂ܪܵܟ݂ ܘܲܕ݂ܡܵܟ݂ ܚܲܝܵܐ. ܐܲܝܟ݂ ܕܐܸܫܬܵܘܕܝܼܬ݁ ܒܲܣܒܲܪܬ݂ܵܟ݂ ܠܘܵܬ݂ ܬܲܠܡܝܼܕܲܝ̈ܟ. ܕܐܲܝܢܵܐ ܕܫܵܐܹܠ ܒܗܲܝܡܵܢܘܼܬ݂ܵܐ ܐ݇ܪܵܙ ܦܘܼܪܩܵܢܵܐ. ܢܵܣܹܒ݂ ܫܪܵܝܵܐ ܚܛܵܗܹ̈ܐ ܢܵܛܲܪ ܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܡܲܢܘܼ ܡܸܫܟܲܚ ܠܲܡܩܵܡ. ܚܘܼܢܲܝܢ ܡܵܪܝ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܡܫܝܼܚܵܐ ܡܵܪܵܗ̇ ܕܬܸܫܒ݁ܘܿܚܬܵܐ. ܘܠܵܐ ܢܸܬ݁ܕܲܓܲܠ ܫܘܼܘܕܵܝ ܡܸܠܲܝ̈ܟ ܗܵܘ̇ ܕܐܸܫܬܵܘܕܝܼܬ݁. ܕܟ݂ܠ ܕܩܵܪܹܐ  ܥܵܢܹܐ ܐ݇ܢܵܐ ܠܹܗ. ܘܢܵܣܹܒ݂ ܫܪܵܝܵܐ ܠܣܲܟ݂ܠܘܵܬܹ̈ܗ. ܘܫܘܼܒ݂ܩܵܢ ܚܵܘܒܵܘ̈ܗܝ ܘܲܚܛܵܗܵܘ̈ܗܝ. ܘܲܚܢܲܢ ܡܵܪܲܢ ܕܩܵܪܹܝܢܲܢ ܠܵܟ݂. ܫܡܲܥ ܒܵܥܘܼܬܲܢ ܘܲܨܠܘܿܬܲܢ. ܘܲܦܬܲܚ ܬܲܪܥܹܗ ܕܲܚܢܵܢܵܟ݂ ܠܲܣܢܝܼܩܘܼܬܲܢ. ܕܢܵܘܕܸܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܲܫܡܵܟ݂ ܪܲܒܵܐ. ܘܢܸܙܡܲܪ ܫܘܼܒ݂ܚܵܐ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܕܡܲܪܝܲܡ܀ ܗܵܝܕܹܝܢ ܢܹܐܡܪܘܼܢ ܒܹܝܬ݂ ܥܲܡܡܹ̈ܐ. ܛܘܼܒܲܝܟ݁ܝ ܠܹܟ݂ܝ ܒܲܪ݇ܬ݂ ܕܵܘܝܼܕ݂. ܒܬ݂ܘܼܠܬܵܐ ܕܟ݂ܝܼܬ݂ܵܐ ܘܩܲܕܝܼܫܬܵܐ. ܕܲܗܘܲܝܬ݁ܝ ܐܸܡܵܐ ܒܝܲܕ݂ ܕܲܟ݂ܝܘܼܬܹܟ݂ܝ ܠܲܡܠܸܟ݂ ܡܲܠܟܹ̈ܐ. ܘܒܹܟ݂ܝ ܐܸܬ݂ܚܲܬܲܡܘ ܓܸܠܝܵܢܹ̈ܐ. ܕܲܢܒ݂ܝܼܘܼܬ݂ܵܐ ܕܲܡܒܲܕܩܵܐ ܗ݇ܘܵܬ݂. ܒܝܲܕ݂ ܦܸܠܐܬ݂ܵܐ ܕܐ݇ܪܵܙ ܡܹܐܬ݂ܝܼܬܹܗ. ܘܡܸܛܠ ܗܵܕܹܐ ܛܘܼܒ݂ܵܐ ܪܓ݂ܝܼܓ݂ܵܐ. ܠܹܟ݂ܝ ܝܵܗܒ̈ܵܢ ܟܠ ܫܲܪ̈ܒ݂ܵܬ݂ܵܐ. ܕܲܗܘܲܝܬ݁ܝ ܠܡܹܐܢܵܐ ܡܢܝܼܚܵܢܝܼܬ݂ܵܐ ܠܛܘܼܦ̮ܣܵܐ ܕܟܹܐܢܹ̈ܐ. ܒܪܝܼܟ݂ܘܼ ܐܲܒ݂ܵܐ ܕܫܲܕܲܪ ܠܲܒ݂ܪܹܗ. ܘܦܲܪܩܹܗ ܠܓܸܢܣܲܢ ܡܢ ܛܘܼܥܝܲܝ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܡܵܪܝܵܐ ܒܚܲܝܠܵܟ݂ ܢܸܚܕܸܐ ܡܲܠܟܵܐ. ܒܝܲܕ݂ ܚܲܝܠܵܐ ܕܡܢ ܪܵܘܡܵܐ ܫܠܝܼܚܹ̈ܐ ܩܲܕܝܼܫܹ̈ܐ ܐܸܬ݂ܚܲܝܲܠܘ. ܘܲܠܟ݂ܠܗܘܿܢ ܛܪ̈ܘܼܢܹܐ ܫܲܥܒܸ݁ܕ݂ܘ ܘܲܙܟ݂ܵܘ ܒܚܹܝܠ ܥܘܼܕ݂ܪܵܢܵܟ݂. ܘܠܲܡܗܲܝܡܢܹ̈ܐ ܕܲܒ݂ܟ݂ܠ ܦܸܢ̈ܝܵܬ݂ܵܐ. ܒܝܲܕ݂ ܥܘܼܕ݂ܪܵܢܵܐ ܕܡܢ ܓܲܪ̈ܡܲܝܗܘܿܢ. ܐܲܚܠܸܡܘ ܐܸܢܘܿܢ ܒܗܲܝܡܵܢܘܼܬ݂ܗܘܿܢ. ܘܗܵܐ ܥܒ݂ܝܼܕ݂ܝܼܢ ܚܠܵܦܲܝܢ ܡܵܪܝ ܒܲܥܵܝܹ̈ܐ. ܕܲܢܦܝܼܣܘܼܢ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܕܲܒ݂ܝܲܕ݂ ܪ̈ܲܚܡܲܝܟ ܣܲܓ݁ܝܼ̈ܐܹܐ ܬܸܫܒ݁ܘܿܩ ܚܵܘܒܲܝ̈ܢ. ܘܐܲܫܘܵܐ ܠܟ݂ܠܲܢ ܕܒܲܨ̈ܠܵܘܵܬ݂ܗܘܿܢ. ܢܸܗܘܸܐ ܝܵܪ̈ܬܹ݁ܐ ܒܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܡܸܛܠ ܐܝܼܩܵܪܵܐ ܕܲܫܡܵܟ݂. ܠܲܨܠܝܼܒ݂ܵܟ݂ ܦܵܪܘܿܩܲܢ ܣܵܓ݂ܕܝܼܢ ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫܵܐ. ܘܲܡܙܲܝܚܝܼܢ ܠܹܗ ܥܲܠ ܐܝܼܕܲܝ̈ܗܘܿܢ ܘܙܵܡܪܝܼܢ ܫܘܼܒ݂ܚܵܐ. ܕܲܥܒܲܕ݂ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܒܲܛܸܠ ܩܪ̈ܵܒܹܐ ܘܬܲܟ݂ܬ݁ܘܼܫܹ̈ܐ. ܘܫܲܝܸܢ ܥܲܡܵܐ ܥܲܡ ܥܲܡܡܹ̈ܐ. ܕܒܲܚܕ݂ܵܐ ܥܹܕܬܵܐ ܠܹܗ ܢܸܬ݂ܟܲܫܦܘܼܢ. ܘܢܵܘܕܘܿܢ ܘܢܸܣܓ݁ܕ݂ܘܼܢ ܠܡܵܪܘܼܬܹܗ. ܘܢܸܩܥܘܿܢ ܒܩܵܠܹ̈ܐ ܩܲܕܝܼܫܹ̈ܐ ܩܲܕܝܼܫ ܩܲܕܝܼܫ. ܩܲܕܝܼܫܲܬ݁ ܡܵܪܝ ܕܲܒ݂ܝܲܕ݂ ܨܠܝܼܒ݂ܵܟ݂. ܦܪܲܩܬܵܝܗܝ ܠܓܸܢܣܲܢ ܡܢ ܛܘܼܥܝܲܝ. ܕܐܲܒ݂ܘܼܢ܀ ܢܸܬ݂ܩܲܝܲܡ ܠܥܵܠܲܡ ܩܕ݂ܵܡ ܐܲܠܵܗܵܐ. ܕܘܼܟ݂ܪܵܢܵܟ݂ ܐܘܿ ܐܲܒ݂ܘܼܢ. ܠܥܸܠ ܒܲܫܡܲܝܵܐ ܡܸܬ݂ܢܲܨܲܚ. ܘܲܠܬܲܚܬ݁ ܒܐܲܪܥܵܐ ܙܡܝܼ̈ܢܵܢ ܓܘܼܕܹ̈ܐ ܕܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ. ܕܲܒ݂ܥܲܡܠܵܐ ܪܲܒܵܐ ܕܐܵܓ݂ܘܿܢܵܟ݂. ܙܒܲܢܬ݁ ܠܵܟ݂ ܚܲܝܹ̈ܐ ܫܲܪܝܼܪܹ̈ܐ. ܒܥܘܼܡܪܵܐ ܦܲܐܝܵܐ ܕܲܫܡܲܝܵܐ. ܘܗܵܐ ܥܒ݂ܝܼܕܲܬ݁ ܣܢܹܐܓ݂ܪܵܐ ܚܠܵܦ ܢܲܦ̮̈ܫܵܬܲܢ. ܕܬܸܒ݂ܥܸܐ ܘܲܬ݂ܦܝܼܣ ܠܲܡܫܝܼܚܵܐ. ܕܲܒ݂ܝܲܕ݂ ܚܘܼܒܹ݁ܗ ܡܬ݂ܘܿܡܵܝܵܐ ܕܠܵܐ ܡܸܣܬܲܝܲܟ݂. ܢܲܫܘܸܐ ܠܟ݂ܠܲܢ ܠܲܓ݂ܢܘܿܢ ܢܘܼܗܪܵܐ. ܠܡܸܙܡܲܪ ܫܘܼܒ݂ܚܵܐ ܕܠܵܐ ܦܵܛܲܪ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܠܵܐ ܬܸܡܢܹܝܢܝ ܥܲܡ ܪ̈ܲܫܝܼܥܹܐ ܘܥܲܡ ܥܵܒ݂ܕܲܝ̈ ܥܵܘܠܵܐ. ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂ ܦܵܪܘܿܩܲܢ ܡܵܐ ܕܡܸܙܕܲܝܚܲܬ݁ ܡܢ ܥܝܼܪܹ̈ܐ. ܘܦܵܩܹܕ݂ ܪܸܡܙܵܟ݂ ܘܩܵܪܝܵܐ ܩܲܪܢܵܐ ܥܲܡ ܫܝܼܦܘܿܪܵܐ. ܘܩܵܝܡܝܼܢ ܡܝܼܬܹ̈ܐ ܡܢ ܩܲܒ݂ܪܹ̈ܐ. ܘܠܲܩܕ݂ܵܡ ܬܪܘܿܢܘܿܣ ܕܪܲܒ݁ܘܼܬ݂ܵܟ݂. ܛܵܒܹ̈ܐ ܘܒ݂ܝܼܫܹ̈ܐ ܡܸܬ݂ܟܲܢܫܝܼܢ. ܛܵܒܹ̈ܐ ܠܪܵܘܡܵܐ ܡܸܬ݂ܥܲܠܹܝܢ. ܘܒ݂ܝܼܫܹ̈ܐ ܦܵܝܫܝܼܢ ܥܲܠ ܐܲܪܥܵܐ. ܚܘܼܢ ܡܵܪܝ ܒܪ̈ܲܚܡܹܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܠܐܲܝܠܹܝܢ ܕܐܵܘܕܝܼܘ ܒܲܨܠܝܼܒ݂ܵܟ݂ ܘܐܲܫܘܵܐ ܐܸܢܘܿܢ ܕܥܲܡ ܩܲܕܝܼܫܲܝ̈ܟ ܢܸܬ݂ܒܲܣܡܘܼܢ ܒܐܘܿܪܸܫܠܸܡ. ܟܠܗܘܿܢ ܠܵܟ݂ܘܼ ܡܣܲܟܹ݁ܝܢ. ܠܓܸܠܝܵܢܵܟ݂ ܦܵܪܘܿܩܲܢ ܡܣܲܟܹ݁ܝܢ ܟܠܗܘܿܢ ܡܵܝܘܿܬܹ̈ܐ. ܕܐܸܡܲܬ݂ܝ ܬܸܕ݂ܢܲܚ ܘܬܸܩܪܸܐ ܩܲܪܢܵܐ ܥܲܡ ܫܝܼܦܘܿܪܵܐ. ܘܬܸܦܩܘܿܕ݂ ܠܐܲܪܥܵܐ ܒܝܲܕ݂ ܡܸܠܬ݂ܵܟ݂. ܘܬ݂ܵܘܠܸܕ ܟܠܗܘܿܢ ܦܲܓ݂ܪ̈ܲܝܗܘܿܢ. ܒܲܪܦܵܦ ܥܲܝܢܵܐ ܡܢ ܓܵܘ ܥܘܼܒܵܗ̇. ܛܵܒܹ̈ܐ ܠܪܵܘܡܵܐ ܡܥܲܠܸܐ ܐܲܢ݇ܬ݁. ܘܒ݂ܝܼܫܹ̈ܐ ܦܵܝܫܝܼܢ ܒܓ݂ܵܘ ܚܸܫܟܵܐ. ܒܗܵܘ ܥܸܕܵܢܵܐ ܦܵܪܘܿܩܲܢ ܚܘܼܣ ܥܲܠ ܥܲܒ݂ܕܲܝ̈ܟ. ܕܲܫܩܲܠܘ ܦܲܓ݂ܪܵܟ݂ ܘܐܸܫܬ݁ܝܼܘ ܠܲܕ݂ܡܵܟ݂. ܘܒܲܣܸܡ ܐܸܢܘܿܢ ܒܡܲܠܟ݁ܘܼܬ݂ܵܟ݂܀
ܕܥܲܢܝܼܕܹ̈ܐ. ܕܐܵܬܹ݁ܐ ܠܲܡܕ݂ܵܢܵܗ ܠܐܲܪܥܵܐ. ܡܲܠܟܵܐ ܡܢܲܚܡܵܢܵܐ ܒܫܘܼܒ݂ܚܹܗ ܕܵܢܲܚ ܡܢ ܪܵܘܡܵܐ. ܘܡܲܚܸܐ ܡܝܼ̈ܬܹܐ ܘܲܡܩܝܼܡ ܠܲܩܒ݂ܝܼܪܹ̈ܐ. ܘܩܵܝܡܝܼܢ ܫܲܘܝܵܐܝܼܬ݂. ܥܲܢܝܼ̈ܕܹܐ ܡܢ ܓܵܘ ܩܲܒ݂ܪܹ̈ܐ. ܘܡܲܣܩܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܡܲܚܸܐ ܡܝܼ̈ܬܹܐ. ܟܠܟ݂ܘܿܢ ܥܲܒ݂ܕܵܘ̈ܗܝ ܕܐܲܠܵܗܵܐ. ܐܲܥܒܲܪܘ ܟܲܪܝܘܼܬ݂ܵܐ ܡܢ ܠܸܒ݁ܟ݂ܘܿܢ ܐܘܿ ܡܵܝܘܿܬܹ̈ܐ. ܕܝܵܘܡܹܗ ܕܡܵܪܲܢ ܐܵܬܹ݁ܐ ܘܡܲܦܨܲܚ ܠܲܢ. ܘܲܡܥܝܼܪ ܕܲܡܟ݁ܘܼܬܲܢ. ܡܵܐ ܕܙܵܝܚܝܼܢ ܥܝܼܪܹ̈ܐ ܩܕ݂ܵܡܵܘܗܝ. ܚ̇ܵܕܹܝܢ ܡܲܠܲܐܟܹ̈ܐ ܒܝܘܼܡ ܢܘܼܚܵܡܵܐ.
ܫܲܒܲܚ. ܒܨܵܒܹ݁ܐ ܒܚܲܝܹ̈ܐ. ܠܘܵܬ݂ܵܟ݂ ܟܠܲܢ ܡܲܠܟܵܐ ܡܫܝܼܚܵܐ ܩ̇ܵܪܹܝܢܲܢ. ܫܲܝܸܢ ܒܲܨܠܝܼܒ݂ܵܟ݂ ܠܥܹܕ݁ܬܵܟ݂ ܡܗܲܝܡܲܢܬܵܐ: ܕܗܵܐ ܡܲܠܝܵܐ ܚܸܪ̈ܝܵܢܹܐ ܘܣܸܕ݂ܩܹ̈ܐ ܘܦܘܼܠܵܓܹ̈ܐ. ܕܐܲܒܲܥ ܒܵܗ ܒܝܼܫܵܐ ܘܕܲܠܚܵܗ ܒܟ݂ܠܦܸܢ̈ܝܵܢ: ܘܗܵܐ ܨܵܠܹܐ ܟܠܫܵܥ ܦܲܚܹ̈ܐ ܘܠܵܐ ܫܵܠܹܐ ܘܨܵܒܹ݁ܐ ܕܫܵܒܹ݁ܐ ܠܵܗ. ܦܲܨܵܗ ܡܢ ܐܝܼܕ݂ܵܘ̈ܗܝ ܡܵܪܝܵܐ ܡܪܲܚܡܵܢܵܐ. ܕܒܲܕ݂ܡܵܟ݂ ܐܸܙܕܲܒ݂ܢܲܬ݀: ܠܵܐ ܬܲܪܦܹܝܘܗܝ ܕܢܸܫܬܲܥܠܸܐ. ܫܘܼܪܵܐ ܗܘܝܼ ܠܵܗ ܚܲܣܝܼܢܵܐ ܘܚܘܼܣ ܥܠܹܝܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܫܲܪܲܪ ܡܵܪܝ ܫܲܬ݂ܐܣܹܝ̈ܗ̇ ܕܥܹܕ݁ܬܵܟ݂ ܒܲܚܢܵܢܵܟ݂: ܘܚܲܣܸܢ ܡܘܼܟ݂ܠܹܝ̈ܗ̇ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܫܪܵܐ ܫܘܼܒ݂ܚܵܟ݂ ܒܗܲܝܟ݁ܠܵܐ ܕܲܦܪܝܼܫ ܠܐܝܼܩܵܪ ܬܸܫܡܸܫܬܵܟ݂: ܟܠܗܘܿܢ ܝܵܘܡܵܬܹ̈ܗ ܕܥܵܠܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
      ܩܵܢܘܿܢܵܐ. ܒܚܘܼܣ ܘܚܲܣܵܐ. ܐܸܬ݁ܕܲܟ݂ܪ ܡܵܪܝܵܐ ܠܕ݂ܵܘܝܼܕ݂ ܘܲܠܟ݂ܠܹܗ ܡܘܼܟܵܟܹܗ* ܕܝܼܡܵܐ ܡܵܪܝܵܐ ܘܲܢܕܲܪ ܠܐܲܠܵܗܹܗ ܕܝܲܥܩܘܿܒ݂܀

ܩܲܕ݁ܝܼܫܵܐ ܕܐܲܫܪܝܼ ܫܟ݂ܝܼܢܬܹ݁ܗ ܥܲܠ ܛܘܼܪ ܣܝܼܢܲܝ ܘܩܲܕ݂ܫܹܗ ܒܕܸܚܠܵܐ. ܐܲܫܪܵܐ ܫܲܝܢܵܟ݂ ܒܓ݂ܵܘ ܥܹܕ݁ܬܵܟ݂ ܘܩܲܕܸ݁ܫܹܝܗ܀

ܕܠܵܐ ܐܸܥܘܿܠ ܡܲܛܠܠܹܗ ܕܒܲܝܬ݁ܝ* ܘܠܵܐ ܐܸܣܲܩ ܠܬܸܫܘܝܼܬ݂ܵܗ ܕܥܲܪܣܝ܀ ܘܠܵܐ ܐܸܬܸ݁ܠ ܫܸܢܬܵܐ ܠܥܲܝ̈ܢܲܝ* ܘܐܵܦܠܵܐ ܢܵܘܡܬܵܐ ܠܲܓ݂ܒ݂ܝܼܢܲܝ̈܀ ܥܕܲܡܵܐ ܕܐܸܫܟܲܚ ܐܲܬ݂ܪܵܐ ܠܡܵܪܝܵܐ* ܘܡܲܫܟ݁ܢܵܐ ܠܐܲܠܵܗܹܗ ܕܝܲܥܩܘܿܒ݂܀ ܗܵܐ ܫܡܲܥܢܵܗ ܒܐܵܦܲܪܬ݂ܵܐ ܘܐܸܫܟܲܚܢܵܗ ܒܚܲܩ̈ܠܵܬܵܐ* ܢܸܥܘܿܠ ܠܡܲܫܟ݁ܢܹܗ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܟ݂ܘܼܒ݂ܫܵܐ ܕܪܸ̈ܓ݂ܠܵܘܗܝ܀ ܩܘܼܡ ܡܵܪܝܵܐ ܠܲܢܝܵܚܬܵܟ݂* ܐܲܢ݇ܬ݁ ܘܩܹܒ݂ܘܿܬ݂ܵܐ ܕܥܘܼܫܢܵܟ݂܀ ܟܵܗܢܲܝ̈ܟ ܢܸܠܒ݁ܫܘܼܢ ܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ ܘܙܲܕ݁ܝܼܩܲܝ̈ܟ ܫܘܼܒ݂ܚܵܐ* ܡܸܛܠ ܕܵܘܝܼܕ݂ ܥܲܒ݂ܕܵܟ݂ ܠܵܐ ܬܲܗܦܸܟ݂ ܐܲܦܵܘ̈ܗܝ ܕܲܡܫܝܼܚܵܟ݂܀ ܝܼܡܵܐ ܡܵܪܝܵܐ ܠܕ݂ܵܘܝܼܕ݂ ܒܲܫܪܵܪܵܐ ܘܠܵܐ ܢܸܗܦܘܿܟ݂ ܡܸܢܹܗ* ܕܡܢ ܦܹܐܪ̈ܲܝ ܟܲܪܣܵܟ݂ ܐܵܘܬܸ݁ܒ݂ ܥܲܠ ܟܘܼܪܣܝܵܟ݂܀ ܐܸܢ ܢܸܛܪܘܼܢ ܒܢܲܝ̈ܟ ܩܝܵܡܝ* ܣܵܗܕܘܼܬ݂ܵܐ ܗܵܕܹܐ ܕܡܲܠܸܦ ܐ݇ܢܵܐ ܠܗܘܿܢ܀ ܘܐܵܦ ܡܢ ܒܢܲܝ̈ܗܘܿܢ ܢܸܬ݁ܒ݂ܘܼܢ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܥܲܠ ܟܘܼܪܣܝܵܟ݂* ܡܸܛܠ ܕܲܨܒ݂ܵܐ ܡܵܪܝܵܐ ܒܨܸܗܝܘܿܢ܀ ܘܲܓ݂ܒ݂ܵܗ ܠܹܗ ܠܒܹܝܬ݂ ܡܲܥܡܪܵܐ* ܗܵܕܲܐ ܗ݇ܝܼ ܢܝܵܚܬ݁ܝ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ܀ ܗܵܪܟܵܐ ܐܸܬܸ݁ܒ݂ ܡܸܛܠ ܕܪܲܓ݂ܬܵܗ* ܘܲܠܨܲܝܵܕܹܝܗ ܐܹܒܲܪܸܟ݂܀ ܘܲܠܡܸܣܟܹ݁ܢܹܝ̈ܗ ܐܹܣܲܒܲܥ ܠܲܚܡܵܐ* ܠܟ݂ܘܼܡܪܹ̈ܝܗ ܐܲܠܒܸ݁ܫ ܦܘܼܪܩܵܢܵܐ ܘܲܠܙܲܕ݁ܝܼ̈ܩܹܝܗ ܫܘܼܒ݂ܚܵܐ܀ ܬܲܡܵܢ ܐܲܕ݂ܢܲܚ ܩܲܪܢܵܐ ܠܕ݂ܵܘܝܼܕ݂* ܘܐܲܢܗܲܪ ܫܪܵܓ݂ܵܐ ܠܲܡܫܝܼܚܹܗ܀ ܘܠܲܒ݂ܥܸܠܕܒ݂ܵܒ݂ܵܘ̈ܗܝ ܐܲܠܒܸ݁ܫ ܒܸܗܬ݁ܬ݂ܵܐ* ܘܲܥܠܵܘܗܝ ܢܸܥܦܸܐ ܩܘܼܕ݂ܫܝ܀

ܩܲܕ݁ܝܼܫܵܐ ܕܐܲܫܪܝܼ ܫܟ݂ܝܼܢܬܹ݁ܗ ܥܲܠ ܛܘܼܪ ܣܝܼܢܲܝ ܘܩܲܕ݂ܫܹܗ ܒܕܸܚܠܵܐ. ܐܲܫܪܵܐ ܫܲܝܢܵܟ݂ ܒܓ݂ܵܘ ܥܹܕ݁ܬܵܟ݂ ܘܩܲܕܸ݁ܫܹܝܗ܀

ܫܲܒܲܚ. ܥܵܠܲܡ.

ܩܲܕ݁ܝܼܫܵܐ ܕܐܲܫܪܝܼ ܫܟ݂ܝܼܢܬܹ݁ܗ ܒܢܵܘܣܹ̈ܐ ܕܟܲܝ̈ܵܐ ܕܲܓ݂ܒ݂ܵܐ ܠܲܫܡܹܗ. ܐܲܫܪܵܐ ܫܲܝܢܵܟ݂ ܒܓ݂ܵܘ ܥܹܕ݁ܬܵܟ݂ ܘܢܲܛܲܪܹܝܗ܀

ܐܸܬ݁ܕܲܟ݂ܪ ܡܵܪܝܵܐ ܠܕ݂ܵܘܝܼܕ݂ ܘܲܠܟ݂ܠܹܗ ܡܘܼܟܵܟܹܗ* ܕܝܼܡܵܐ ܡܵܪܝܵܐ ܘܲܢܕܲܪ ܠܐܲܠܵܗܹܗ ܕܝܲܥܩܘܿܒ݂܀

ܫܲܝܢܵܟ݂ ܡܵܪܲܢ ܘܲܫܠܵܡܵܟ݂ ܐܲܫܪܵܐ ܒܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂. ܘܢܲܛܲܪ ܝܲܠܕܹܝ̈ܗ ܡܢ ܒܝܼܫܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀

ܐܸܬ݁ܕܲܟ݂ܪ ܡܵܪܲܢ ܫܘܼܘܕܵܝܵܟ݂ ܕܲܠܘܵܬ݂ ܦܲܛܪܘܿܣ ܘܕܲܠܘܵܬ݂ ܥܹܕ݁ܬܵܟ݂. ܕܬܲܪ̈ܥܹܝܗ ܕܲܫܝܘܿܠ ܘܲܛܪ̈ܘܼܢܹܝܗ ܠܵܐ ܢܸܚܣܢܘܼܢܵܗ܀

ܚܘܼܣ ܘܚܲܣܵܐ ܦܵܪܘܿܩܲܢ. ܘܩܲܝܸܡ ܥܹܕ݁ܬܵܟ݂ ܘܢܲܛܲܪ ܝܲܠܕܹܝ̈ܗ. ܒܲܨܠܘܿܬ݂ ܟܠܗܘܿܢ ܩܲܕ݁ܝܼܫܲܝ̈ܟ.

ܬܸܫܒ݁ܘܿܚܬܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܪ̈ܲܚܡܲܝܟ ܡܫܝܼܚܵܐ ܡܲܠܟܲܢ. ܒܪܵܐ ܕܐܲܠܵܗܵܐ ܣܓ݂ܝܼܕ݂ ܡܢ ܟܠܵܐ܀ ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܲܢ ܘܐܲܢ݇ܬ݁ ܐܲܠܵܗܲܢ. ܘܪܹܫܵܐ ܕܚܲܝܲܝ̈ܢ ܘܣܲܒ݂ܪܲܢ ܒܪܝܼܟ݂ܵܐ܀ ܘܠܵܟ݂ܘܼ ܡܫܲܒ݁ܚܝܼܢ ܣܸܕ݂ܪܹ̈ܐ ܕܲܠܥܸܠ. ܘܟܸܢܫܹ̈ܐ ܕܲܠܬܲܚܬ݁ ܒܲܚܕ݂ܵܐ ܫܲܘܝܘܼ܀ ܟܲܕ݂ ܡܵܘܕܹ݁ܝܢ ܒܵܟ݂ ܕܐܲܢ݇ܬ݁ܘܼ ܟܲܣܝܵܐ. ܕܐܸܬ݂ܓ݁ܠܝܼܬ݁ ܒܦܲܓ݂ܪܲܢ ܒܫܘܼܠܵܡ ܙܲܒ݂ܢܹ̈ܐ܀ ܟܲܕ݂ ܓܠܵܘ ܪ̈ܲܚܡܲܝܟ ܐܸܨܛܒ݂ܝܼܬ݁ ܒܚܘܼܒܵܟ݂. ܐ݇ܬܹܝܬ݁ ܠܦܘܼܪܩܵܢܲܢ ܘܚܲܪܲܪܬ݁ ܠܓܸܢܣܲܢ܀ ܟܹܐܒܲܝ̈ܢ ܐܲܣܝܼܬ݁ ܘܚܲܘܒܲܝ̈ܢ ܚܲܣܝܼܬ݁. ܘܲܠܡܝܼܬ݂ܘܼܬܲܢ ܢܲܚܸܡܬ݁ ܒܪ̈ܲܚܡܲܝܟ܀ ܘܐܲܩܝܼܡܬ݁ ܒܐܲܪܥܵܐ ܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ. ܒܛܘܼܦ̮ܣܵܐ ܕܲܠܥܸܠ ܗܵܝ ܕܒܲܫܡܲܝܵܐ܀ ܛܘܼܦ̮ܣܵܐ ܪܫܲܡܬܵܗ̇ ܒܚܘܼܒܵܐ ܡܟܲܪܬܵܗ̇. ܒܪ̈ܲܚܡܹܐ ܢܣܲܒ݂ܬܵܗ̇ ܒܚܲܫܵܐ ܓܡܲܪܬܵܗ̇܀ ܘܣܵܢܐܵܐ ܕܐ݇ܢܵܫܵܐ ܗܵܐ ܫܵܓܹ݁ܫ ܠܵܗ̇. ܒܚܘܼܨܦܵܐ ܕܓ݂ܘܼܡܕܹܗ ܒܝܲܕ݂ ܫܲܡܵܫܵܘ̈ܗܝ܀ ܠܵܐ ܡܵܪܝ ܬܲܗܡܸܐ ܒܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ. ܘܠܵܐ ܢܸܬ݁ܕܲܓܲܠ ܫܘܼܘܕܵܝ ܡܸܠܲܝ̈ܟ܀ ܘܠܵܐ ܢܸܫܬܲܚܲܡ ܫܘܼܦܪ̈ܵܗ̇ ܪܓ݂ܝܼܓ݂ܵܐ. ܘܥܘܼܬ݂ܪܵܗ̇ ܪܲܒܵܐ ܠܵܐ ܢܸܬ݂ܡܲܣܟܲܢ܀ ܥܗܲܕ݂ ܫܘܼܘܕܵܝܵܟ݂ ܕܲܠܘܵܬ݂ ܦܵܛܪܘܿܣ. ܓܡܘܿܪ ܒܲܥܒ݂ܵܕ݂ܵܐ ܡܸܕܸ݁ܡ ܕܐܸܡܲܪܬ݁܀ ܫܲܪܲܪ ܬܲܪ̈ܥܝܹܗ̇ ܘܚܲܣܸܢ ܡܘܼܟ݂ܠܹܝ̈ܗ̇. ܐܲܪܝܼܡ ܩܲܪܢܵܗ̇ ܘܪܲܡܪܸܡ ܫܘܼܪܹ̈ܝܗ̇܀ ܒܲܪܸܟ݂ ܠܲܒ݂ܢܹܝ̈ܗ̇ ܘܢܲܛܲܪ ܝܲܠܕܹܝ̈ܗ̇. ܩܲܝܸܡ ܟܵܗܢܹܝ̈ܗ̇ ܘܐܲܒ݂ܗܸܬ݂ ܠܣܵܢܐܹܝ̈ܗ̇܀ ܘܫܲܝܢܵܐ ܕܡܸܢܵܟ݂ ܐܲܫܪܵܐ ܒܓ݂ܵܘܵܗ̇. ܘܒܲܛܸܠ ܡܸܢܵܗ̇ ܣܸܕ݂ܩܹ̈ܐ ܦܠܝܼܓܹ̈ܐ܀ ܘܗܲܒ݂ܠܲܢ ܕܢܸܥܡܲܪ ܥܘܼܡܪܵܐ ܫܲܠܝܵܐ. ܕܠܵܐ ܕܘܼܘܵܕ݂ܵܐ ܒܕܸܚܠܲܬ݂ ܩܘܼܫܬܵܐ܀ ܟܲܕ݂ ܢ̇ܵܛܪܝܼܢܲܢ ܗܲܝܡܵܢܘܼܬܲܢ. ܒܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܘܚܘܼܒܵܐ ܫܲܠܡܵܐ܀ ܐܵܦ ܕܘܼܒܵܪ̈ܲܝܢ ܢܸܫܦܪܘܼܢ ܩܘܼܕ݂ܡܵܝܟ. ܘܢܸܫܟܲܚ ܪ̈ܲܚܡܹܐ ܒܝܘܿܡ ܦܘܼܪܥܵܢܵܐ܀ ܘܲܕ݂ܠܵܐ ܫܲܠܘܵܐ ܢܲܣܸܩ ܫܘܼܒ݂ܚܵܐ. ܠܐܲܒ݂ܘܼܟ݂ ܒܐܝܼܕܲܝ̈ܟ ܘܲܠܪܘܼܚܩܘܼܕ݂ܫܵܐ܀ ܕܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ ܒܟ݂ܠܗܘܿܢ ܕܵܪܹ̈ܐ. ܕܥܵܠܡܲܝ̈ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ. ܘܡܸܬ݂ܐܲܡܪܵܐ ܒܟ݂ܠܹܗ ܫܵܒ݂ܘܿܥܵܐ ܕܡܲܥܲܠܬܵܐ.
  ܟܵܪܘܿܙܘܼܬ݂ܵܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.܀
ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ݁ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡܠܲܟ݂ ܛܵܒ݂̈ܵܬ݂ܵܐ ܠܪ̈ܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕ݁ܬܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܠܵܐ ܨܵܒܹܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ: ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ܡܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܗܘܝܘ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܪ̈ܝܵܬ݂ܵܐ: ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܝܼܣ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܝܼܣ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܵܘ̇ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
	ܘܟܸܢ ܐܵܡܪܝܼܢ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.

ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ. ܡܢ ܦܘܼܡܹ̈ܐ ܡܵܪܝ ܡܠܝܼܠܹ̈ܐ ܕܲܒ݂ܪܲܝܬ݁: ܘܡܢ ܠܸܫܵܢܹ̈ܐ ܡܫܲܒ݁ܚܵܢܹ̈ܐ ܕܪܲܟܸ݁ܒ݂ܬ݁: ܘܡܢ ܟܸܢܫܹ̈ܐ ܟܠܗܘܿܢ ܕܥܸܠܵܝܹ̈ܐ ܘܲܕ݂ܬܲܚܬܵܝܹ̈ܐ: ܫܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܲܕ݂ܡܵܪܘܼܬ݂ܵܟ݂ ܢܸܣܬ݁ܓܸܕ݂ ܘܢܸܫܬܲܒܲܚ ܘܢܸܬ݂ܝܲܩܲܪ ܘܢܸܬ݂ܪܲܡܪܲܡ ܘܢܸܬ݁ܬܵܘܕܸ݁ܐ ܘܢܸܬ݂ܒܲܪܲܟ݂ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
  ܕܠܸܠܝܵܐ. ܒܗܵܝ ܕܪܲܡܫܵܐ. ܡܫܲܒܲܚ ܒܲܡܪ̈ܵܘܡܹܐ ܡܵܪܝܵܐ: ܕܒܲܡܪܵܘܡܵܐ ܣܵܡܬ݁ ܡܲܥܡܪܵܟ݂. ܕܨܘܼܪܬܵܐ. ܘܲܠܥܸܠ ܡܢ ܟܠ ܫܸܡ ܕܡܸܫܬܲܡܲܗ. ܠܥܸܠ ܒܲܡܪ̈ܵܘܡܹܐ ܒܲܩܕ݂ܘܿܫ ܩܘܼܕ݂ܫܹ̈ܐ ܐܝܼܬܲܝܟ ܡܵܪܲܢ. ܡܢ ܝܲܡܝܼܢܵܐ ܥܲܡ ܐܲܒ݂ܘܼܟ݂. ܘܲܠܬܲܚܬ݁ ܒܐܲܪܥܵܐ ܟܸܢܫܹ̈ܐ ܦܪ̈ܝܼܩܹܐ ܕܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ. ܗܲܝܟ݁ܠܹ̈ܐ ܫܒ݂ܝܼܚܹ̈ܐ ܒܵܢܹܝܢ ܠܵܟ݂. ܕܲܒ݂ܗܘܿܢ ܡܸܬ݂ܟܲܫܦܝܼܢ ܘܫܵܐܠܝܼܢ ܪ̈ܲܚܡܹܐ. ܘܫܘܼܒ݂ܩܵܢ ܚܵܘܒܹ̈ܐ ܒܟ݂ܠ ܥܸܕܵܢ. ܕܬܸܛܲܪ ܥܹܕ݁ܬܵܟ݂ ܡܢ ܒܝܼܫܵܐ. ܘܥܸܪ̈ܒܹ݁ܐ ܕܥܵܢ̈ܵܟ݂ ܡܢ ܕܹܐܒܹ̈ܐ. ܡܫܝܼܚܵܐ ܕܲܠܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ. ܒܲܕ݂ܡܹܗ ܦܲܪܩܵܗ ܡܢ ܛܘܼܥܝܲܝ. ܫܘܼܪܵܐ ܗܘܝܼ ܠܵܗ ܚܲܣܝܼܢܵܐ. ܘܢܸܒ݂ܗܲܬ݂ ܒܝܼܫܵܐ ܒܪܘܼܡܪܵܡܵܗ. ܘܫܲܝܢܵܟ݂ ܢܲܡܠܸܟ݂ ܒܵܗ:

ܬܢܝܼ.

ܫܲܒܲܚ. ܠܥܸܠ ܒܲܡܪ̈ܵܘܡܹܐ ܒܲܩܕ݂ܘܿܫ ܩܘܼܕ݂ܫܹ̈ܐ ܐܝܼܬܲܝܟ ܡܵܪܲܢ. ܡܢ ܝܲܡܝܼܢܵܐ ܥܲܡ ܐܲܒ݂ܘܼܟ݂. ܘܲܠܬܲܚܬ݁ ܒܐܲܪܥܵܐ ܟܸܢܫܹ̈ܐ ܦܪ̈ܝܼܩܹܐ ܕܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ. ܗܲܝܟ݁ܠܹ̈ܐ ܫܒ݂ܝܼܚܹ̈ܐ ܒܵܢܹܝܢ ܠܵܟ݂. ܕܲܒ݂ܗܘܿܢ ܡܸܬ݂ܟܲܫܦܝܼܢ ܘܫܵܐܠܝܼܢ ܪ̈ܲܚܡܹܐ. ܘܫܘܼܒ݂ܩܵܢ ܚܵܘܒܹ̈ܐ ܒܟ݂ܠ ܥܸܕܵܢ. ܕܬܸܛܲܪ ܥܹܕ݁ܬܵܟ݂ ܡܢ ܒܝܼܫܵܐ. ܘܥܸܪ̈ܒܹ݁ܐ ܕܥܵܢ̈ܵܟ݂ ܡܢ ܕܹܐܒܹ̈ܐ. ܡܫܝܼܚܵܐ ܕܲܠܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ. ܒܲܕ݂ܡܹܗ ܦܲܪܩܵܗ ܡܢ ܛܘܼܥܝܲܝ. ܫܘܼܪܵܐ ܗܘܝܼ ܠܵܗ ܚܲܣܝܼܢܵܐ. ܘܢܸܒ݂ܗܲܬ݂ ܒܝܼܫܵܐ ܒܪܘܼܡܪܵܡܵܗ. ܘܫܲܝܢܵܟ݂ ܢܲܡܠܸܟ݂ ܒܵܗ.
ܥܵܠܲܡ. ܡܫܝܼܚܵܐ ܠܵܐ ܬܲܗܡܸܐ ܡܸܢܲܢ ܘܠܵܐ ܬܲܪܚܸܩ ܡܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܕܒ݂ܵܟ݂ܘܼ ܡܵܪܝ ܐܸܬ݂ܓܵܘܲܣܢܲܢ: ܕܲܒܲܪܲܝܢ ܒܐܘܼܪܚܵܟ݂ ܕܚܲܝܹ̈ܐ: ܕܲܢܙܲܡܲܪ ܠܵܟ݂ ܟܠܲܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ.
ܢܹܐܡܲܪ. ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܐܸܡܹܗ ܕܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ: ܐܲܦܝܼܣܝ ܥܲܡܲܢ ܠܲܡܫܝܼܚܵܐ ܕܢܲܫܪܸܐ ܫܲܝܢܹܗ ܒܲܝܢܵܬܲܢ: ܘܲܢܢܲܛܲܪ ܠܲܢ ܡܢ ܟܠ ܢܸܟ݂ܝܵܢ̈ܝܼܢ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ.
ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܒܵܚܵܐ. ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܢ ܟܠ ܠܸܫܵܢܝܼ̈ܢ: ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܘܗܸܕ݂ܪܵܐ ܘܪܘܼܡܪܵܡܵܐ ܡܢ ܟܠ ܒܸܪ̈ܝܵܢ: ܐܝܼܬ݂ܝܵܐ ܓܢܝܼܙܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
  ܫܘܼܒܵܚܵܐ. ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܘܓܲܐܝܘܼܬ݂ܵܐ ܠܒܸܫ

ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܘܓܲܐܝܘܼܬ݂ܵܐ ܠܒܸܫ

 ܠܒܸܫ ܡܵܪܝܵܐ ܥܘܼܫܢܵܐ ܘܐܸܬ݂ܥܲܫܲܢ

ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
 ܘܐܲܬ݂ܩܢܵܗ̇ ܠܬܹܒܹܝܠ ܕܠܵܐ ܬܙܘܼܥ

 ܡܲܬ݂ܩܲܢܘܼ ܟܘܼܪܣܝܵܟ݂ ܡܢ ܩܕ݂ܝܼܡ

 ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܘܡܢ ܥܵܠܲܡ ܐܲܢ݇ܬ݁ܘܼ

 ܐܸܬ݁ܬ݁ܪܝܼܡܘ ܢܲܗܪ̈ܵܘܵܬ݂ܵܐ ܡܵܪܝܵܐ

 ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܘܐܲܪܝܼܡܘ ܢܲܗܪ̈ܵܘܵܬ݂ܵܐ ܩܵܠܗܘܿܢ

 ܐܸܬ݁ܬ݁ܪܝܼܡܘ ܢܲܗܪ̈ܵܘܵܬ݂ܵܐ ܒܕܲܟ݂ܝܼܘܼܬ݂ܵܐ

 ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܡܢ ܩܵܠܵܐ ܕܡܲܝ̈ܵܐ ܣܲܓ݁ܝܼ̈ܐܹܐ

 ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܡܫܲܒܲܚ ܒܲܡܪ̈ܵܘܡܹܐ ܡܵܪܝܵܐ

 ܘܣܵܗ̈ܕ݁ܘܵܬ݂ܵܟ݂ ܛܵܒ݂ ܡܗܲܝܡ̈ܢܵܢ

 ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܘܲܠܒܲܝܬܵܟ݂ ܝܵܐ̇ܝܵܐ ܩܲܕ݁ܝܼܫܘܼܬ݂ܵܐ

 ܡܵܪܝܵܐ ܠܢܘܼܓ݂ܪܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܫܲܒܲܚ܀ ܥܵܠܲܡ܀ ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
ܡܵܪܝܵܐ ܐܲܡܠܸܟ݂ ܘܓܲܐܝܘܼܬ݂ܵܐ ܠܒܸܫ

 ܠܒܸܫ ܡܵܪܝܵܐ ܥܘܼܫܢܵܐ ܘܐܸܬ݂ܥܲܫܲܢ

ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܐܵܗܝܼܢܓܵܗ: ܠܵܟ݂ ܡܵܪܝܵܐ*
      ܗܦܵܟ݂ܬܵܐ. ܡܲܠܟܵܐ ܡܫܝܼܚܵܐ ܐܲܫܪܵܐ ܫܲܝܢܵܟ݂ ܒܒܲܝܬܵܐ ܗܵܢܵܐ ܥܕܲܡܵܐ ܠܥܵܠܲܡ: ܘܛܲܪ ܥܵܡܘܿܪ̈ܵܘܗܝ ܡܢ ܢܸܟ݂ܝܵܢܵܘ̈ܗܝ ܕܐܵܟܹ݁ܠܩܲܪܨܵܐ ܣܵܢܐܹܗ ܕܓܸܢܣܲܢ܀ ܒܲܪܸܟ݂ ܡܵܪܲܢ ܢܲܛܲܪ ܡܵܪܲܢ ܠܒܲܝܬܵܐ ܗܵܢܵܐ: ܘܐܲܫܪܵܐ ܒܓ݂ܵܘܹܗ ܫܲܝܢܵܟ݂ ܪܲܒܵܐ ܥܕܲܡܵܐ ܠܥܵܠܲܡ܀ ܡܫܝܼܚܵܐ ܪܹܫܵܐ ܕܗܲܕܵܡܹ̈ܐ: ܚܲܕ݂ ܓܘܼܫܡܵܐ ܥܲܒ݂ܕܵܗ ܠܥܹܕ݁ܬܹ݁ܗ. ܘܒܲܣܲܪ ܥܲܠ ܚܲܫܹ̈ܐ ܕܙܲܒ݂ܢܹ̈ܐ: ܡܸܛܠ ܚܘܼܒܵܐ ܕܢܵܣܘܿܒܹܗ܀ ܗܵܐ ܐܵܣܝܵܐ ܝ̇ܵܕܲܥ ܚܲܫܹ̈ܐ. ܕܟܲܣܝ̈ܵܬ݂ܵܐ ܘܲܕ݂ܓܲܠ̈ܝܵܬ݂ܵܐ. ܟܪ̈ܝܼܗܹܐ ܬܵܘ ܣܲܒ݂ܘ ܚܘܼܠܡܵܢܵܐ: ܕܠܵܐ ܬܡܘܼܬ݂ܘܼܢ ܒܲܚܛܵܗܲܝ̈ܟ݁ܘܿܢ.
ܬܸܫܒܘܿܚܬܵܐ. ܘܲܒ݂ܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܕܹܝܢ ܐܵܡ̇ܪܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܗܵܕܹܐ ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܡܵܪܝ ܢܲܪܣܲܝ ܡܲܠܦܵܢܵܐ.
ܬܵܘܕ݁ܝܼ ܠܛܵܒ݂ܵܐ ܕܚܲܪܲܪ ܓܸܢܣܲܢ: ܡܢ ܥܲܒ݂ܕ݁ܘܼܬ݂ܵܐ ܕܒ݂ܝܼܫܵܐ ܘܡܵܘܬܵܐ܀ ܫܲܝܸܢ ܥܲܡܲܢ ܟܸܢܫܲܝ̈ ܪܵܘܡܵܐ: ܕܪܲܓ݁ܝܼܙܝܼܢ ܗ݇ܘܵܘ ܡܸܛܠ ܥܵܘܠܲܢ܀ ܒܪܝܼܟ݂ ܚܲܢܵܢܵܐ ܕܟܲܕ݂ ܠܵܐ ܒܥܲܝܢܵܝܗܝ: ܢܦܲܩ ܒܲܒ݂ܥܵܬܲܢ ܘܲܚܕ݂ܝܼ ܒܚܲܝܲܝ̈ܢ܀ ܘܨܵܪ ܕܸܡ̈ܘܵܬ݂ܵܐ ܕܐܲܒ݁ܝܼܕ݂ܘܼܬܲܢ: ܘܲܕ݂ܦܘܼܢܵܝܲܢ ܒܥܸܪܒܵܐ ܕܲܛܥܵܐ܀ ܝܵܪܬܵܐ ܘܲܒ݂ܪܵܐ ܩܪܵܝܗܝ ܠܲܟ݂ܝܵܢܲܢ: ܕܲܛܥܵܐ ܘܲܦܢܵܐ ܘܡܝܼܬ݂ ܘܐܸܬ݂ܢܲܚܲܡ܀ ܘܚܲܕ݁ܝܼ ܠܟܸܢܫܹ̈ܐ ܪ̈ܘܼܚܵܢܵܝܹܐ: ܒܲܬ݂ܝܵܒ݂ܘܼܬܲܢ ܘܲܒ݂ܢܘܼܚܵܡܲܢ܀ ܠܵܐ ܡܸܬ݂ܡܲܠܲܠ ܚܘܼܒܵܐ ܪܲܒܵܐ: ܕܚܵܘܝܼ ܨܹܐܕܲܝܢ ܪܵܚܹ̇ܡ ܓܸܢܣܲܢ܀ ܕܡܸܢܹܗ ܕܓܸܢܣܲܢ ܥܒܲܕ݂ ܡܸܨܥܵܝܵܐ: ܘܪܲܥܝܼ ܠܥܵܠܡܵܐ ܥܲܡ ܪܲܒܘܼܬܹܗ܀ ܪܲܒܵܐ ܗ݇ܝ ܡܸܢܲܢ ܘܡܢ ܟܠ ܒܸܪ̈ܝܵܢ: ܚܕܲܬܵܐ ܕܲܣܥܲܪ ܨܹܝܕ݂ ܐ݇ܢܵܫܘܼܬܲܢ܀ ܕܥܲܒ݂ܕܹ݁ܗ ܠܦܲܓ݂ܪܲܢ ܗܲܝܟܲܠ ܩܘܼܕ݂ܫܵܐ: ܕܲܢܡܲܠܸܐ ܒܹܗ ܣܸܓ݂ܕܲܬ݂ ܟܠܵܐ܀ ܬܵܘ ܐܲܪ̈ܥܵܢܹܐ ܘܲܫܡܲܝܵܢܹ̈ܐ: ܬܗܲܪܘ ܘܐܸܬ݁ܕܲܡܲܪܘ ܒܪܲܒ݁ܘܼܬ݂ ܕܲܪܓ݂ܵܐ܀ ܕܲܡܛܵܐ ܓܸܢܣܲܢ ܠܪܵܘܡܵܐ ܪܲܒܵܐ: ܕܐܲܠܵܗܘܼܬ݂ܵܐ ܕܠܵܐ ܡܸܬ݁ܕܲܪܟܵܐ܀ ܫܡܲܝܵܐ ܘܐܲܪܥܵܐ ܘܟ݂ܠܡܵܐ ܕܲܒ݂ܗܘܿܢ: ܢܵܘܕ݁ܘܿܢ ܥܲܡܲܢ ܠܡܵܘܪܸܒ݂ ܓܸܢܣܲܢ܀ ܕܚܲܕܸ݁ܬ݂ ܨܲܠܡܲܢ ܘܲܥܛܵܐ ܥܵܘܠܲܢ: ܘܲܩܪܵܢ ܒܲܫܡܹܗ ܘܫܲܥܒܸ݁ܕ݂ ܠܲܢ ܟܠ܀ ܫܵܘܹ̇ܐ ܠܫܘܼܒ݂ܚܵܐ ܡܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܗ̇ܘ ܕܐܲܪܝܼܡܲܢ ܠܥܸܠ ܡܢ ܟܠܵܐ܀ ܘܢܹܐܡܲܪ ܟܠܲܢ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ: ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀
ܟܵܪܘܿܙܘܼܬ݂ܵܐ
(ܘܪܵܡܹܐ ܡܫܲܡܫܵܢܵܐ ܐܘܼܪܵܪܵܐ: ܘܐܵܣܲܪ ܙܘܼܢܵܪܵܐ: ܘܩܵܐܹ̇ܡ ܩܕ݂ܵܡ ܫܟ݂ܝܼܢܬܵܐ ܡܛܲܟ݁ܣܵܐܝܼܬ݂(ܘܐܵܡܲܪ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ: ܒܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܒ݂ܥܹܐܕܹ̈ܐ ܘܲܒ݂ܕ݂ܘܼܟ݂ܪ̈ܵܢܹܐ.
ܘܡܲܩܸܦ:ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
 ܡܫܝܼܚܵܐ ܕܡܢ ܛܵܥܝܘܼܬ݂ܵܐ ܘܣܸܓ݂ܕܲܬ݂ ܦܬܲܟ݂ܪܹ̈ܐ ܐܲܦܢܝܼ ܠܣܸܓ݂ܕܬܹܗ ( ܠܥܲܡܡܹ̈ܐ ) ܘܐܲܣܠܝܼ ܠܨܸܗܝܘܿܢ ܕܛܸܠܡܲܬ݀ ܫܘܼܒ݂ܚܹܗ.

 ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

 ܡܫܝܼܚܵܐ ܕܐܲܫܪܝܼ ܝܵܘܡܵܢܵܐ ܚܲܝܠܹܗ ܪܲܒܵܐ ܒܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ. ܕܐܸܬ݂ܩܲܕܫܲܬ݀ ܐܲܝܟ݂ ܪܲܒ݁ܘܼܬ݂ܵܐ ܕܫܘܼܒ݂ܚܹܗ ܘܗܸܕ݂ܪܵܐ ܕܲܫܡܹܗ.

 ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

 ܡܫܝܼܚܵܐ ܕܐܸܡܲܪ ܠܥܹܕ݁ܬܹ݁ܗ ܕܢܹܐܬ݂ܘܿܢ ܡܲܠܟܹ̈ܐ ܠܕܸܢܚܹܟ݂ܝ. ܘܫܲܡܠܝܼ ܠܵܗ ܩܝܵܡܹ̈ܐ ܕܐܸܡܲܪ ܕܢܹܐܬ݂ܘܿܢ ܠܘܵܬܹܟ݂ܝ. ܟܲܕ݂ ܕܒ݂ܝܼܪܝܼܢ ܒܢܲܝ̈ܗܘܿܢ ܘܲܒ݂ܢ̈ܵܬ݂ܗܘܿܢ ܘܢܸܣܓ݁ܕ݂ܘܿܢ ܠܟ݂ܘܼܒ݂ܫܵܐ ܕܪܸ̈ܓ݂ܠܲܝܟ݁ܝ.

 ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

 ܡܫܝܼܚܵܐ ܕܲܡܠܲܟ݂ ܠܥܹܕ݁ܬܹ݁ܗ ܕܢܸܗܘܘܿܢ ܫܘܼܪܹ̈ܝܗ ܦܘܼܪܩܵܢܵܐ ܘܬܲܪ̈ܥܹܝܗ ܬܸܫܒ݁ܘܿܚܬܵܐ.

 ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

 ܡܫܝܼܚܵܐ ܕܩܲܕܸ݁ܡ ܘܨܵܪ ܒܸܢܝܵܢܵܗ ܕܥܹܕ݁ܬܹ݁ܗ ܒܟܹܐܦܹ̈ܐ ܛܵܒ݂̈ܵܬ݂ܵܐ. ܒܐܝܼܵܣܦܘܿܢ ܘܲܩܪܘܿܣܛܸܠܘܿܣ ܘܲܒ݂ܩܲܪ̈ܟܲܕ݂ܢܹܐ ܘܲܙܡܲܪ̈ܓ݁ܕܹܐ.

 ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

 ܡܫܝܼܚܵܐ ܕܝܲܗ݇ܒ݂ ܠܥܹܕ݁ܬܹ݁ܗ ܫܘܼܠܛܵܢܵܐ ܕܟ݂ܵܗܢܘܼܬ݂ܵܐ ܠܡܹܐܣܲܪ ܘܲܠܡܸܫܪܵܐ ܒܪܵܘܡܵܐ ܘܲܒ݂ܥܘܼܡܩܵܐ.

 ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

 ܡܫܝܼܚܵܐ ܕܣܵܡܵܗ ܠܥܹܕ݁ܬܹ݁ܗ ܥܲܠ ܥܲܡܘܼܕ݂ܵܐ ܕܠܵܐ ܡܸܬ݁ܬ݁ܙܝܼܥ. ܘܫܲܪܪܵܗ ܒܗܲܝܡܵܢܘܼܬ݂ܵܐ ܬܪܝܼܨܲܬ݂ ܫܘܼܒ݂ܚܵܐ. ܘܥܲܒ݂ܕܵܗ ܫܘܼܪܵܐ ܚܲܣܝܼܢܵܐ ܠܲܒ݂ܢܹܝ̈ܗ.

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

 ܡܫܝܼܚܵܐ ܛܲܪ ܠܵܗ ܫܘܼܘܕܵܝܵܟ݂ ܘܲܗܘܝܼ ܥܲܡܵܗ ܐܲܝܟ݂ ܡܸܠܬ݂ܵܟ݂ ܥܕܲܡܵܐ ܠܡܘܼܠܵܝܵܐ ܕܙܲܒ݂ܢܹ̈ܐ

 ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

ܥܲܠ

 ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ

ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ

 ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂:

ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀

 ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܘܩܲܝܸܡ ܥܹܕ݁ܬܵܟ݂ ܒܲܨܠܝܼܒ݂ܵܟ݂. ܘܦܲܨܵܗ ܡܢ ܣܵܢܐܹ̈ܐ ܘܣܵܩܘܿܪܹ̈ܐ ܕܡܲܩܪܒ݂ܝܼܢ ܥܲܡܵܗ. ܡܫܝܼܚܵܐ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
