ܛܲܟ݂ܣܵܐ ܕܬܸܫܡܸܫܬܵܐ ܕܠܸܠܝܵܐ
ܘܲܕ݂ܨܲܦܪܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܕܲܫܚܝܼܡܹ̈ܐ
ܕܲܬ݂ܠܵܬ݂ܒ݁ܫܲܒܵܐ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟ݂: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟ݂: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬ݂ܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	ܡܵܘܬ݁ܒ݂ܵܐ܀
      ܡܵܘܬ݁ܒ݂ܵܐ܀ ܕܠܸܠܝܵܐ. ܒܡܲܒ݁ܘܼܥ ܚܲܝܹ̈ܐ. ܬܸܥܫܲܢ ܐܝܼܕ݂ܵܟ݂ ܘܬܸܬ݁ܪܝܼܡ ܝܲܡܝܼܢܵܟ݂: ܒܚܲܝܠܵܐ ܕܦܘܼܪܩܵܢܵܐ ܕܝܲܡܝܼܢܹܗ. ܒܝܲܕ݂ ܥܘܼܫܢܵܐ ܘܬ݂ܘܼܩܦܵܐ ܕܝܲܡܝܼܢܵܟ݂ ܐܲܢ݇ܬ݁ ܐܸܬ݂ܓ݁ܠܝܼܬ݁ ܘܲܦܪܲܩܬ݁ ܠܲܢ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܠܥܹܕ݁ܬܵܟ݂ ܩܲܝܸܡ ܒܲܚܢܵܢܵܟ݂ ܘܫܲܝܸܢ ܒܪ̈ܲܚܡܲܝܟ. ܘܲܠܟ݂ܠܗܘܿܢ ܐܲܝܠܹܝܢ ܕܥܵܡܪܝܼܢ ܒܵܗ ܒܲܪܸܟ݂ ܘܢܲܛܲܪ܀ ܚܲܝܠܵܟ݂ ܪܲܒܵܐ ܦܵܪܘܿܩܲܢ. ܗܲܒ݂ ܘܐܲܫܪܵܐ ܒܟ݂ܠܗܹܝܢ ܥܹܕܵܬ݂̈ܵܐ. ܘܟ݂ܵܗܢܹ̈ܐ ܕܲܒ݂ܗܹܝܢ ܒܕܸܚܠܵܐ ܢܫܲܡܫܘܼܢ. ܟܲܕ݂ ܡܵܘܕܹ݁ܝܢ ܠܵܟ݂. ܘܐܲܢ݇ܬ݁ ܗܲܒ݂ܠܲܢ ܕܲܕ݂ܠܵܐ ܫܲܠܘܵܐ. ܒܐܵܘܝܘܼܬ݂ܵܐ ܘܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ. ܕܟ݂ܠܲܢ ܒܝܲܕ݂ ܚܘܼܒܵܐ ܫܲܠܡܵܐ. ܢܸܬ݂ܟܲܫܲܦ ܟܲܕ݂ ܡܵܘܕܹ݁ܝܢܲܢ ܠܵܟ݂܀ ܘܢܸܩܥܸܐ ܟܠܲܢ ܡܵܪܹܟ݁ܠ. ܐܲܢ݇ܬ݁ ܥܢܝܼܢ ܒܲܚܢܵܢܵܟ݂ ܘܪܲܚܸܡܥܠܲܝܢ: ܬܢܝܼ. ܙܡܲܪܘ ܠܐܲܠܵܗܵܐ ܒܫܘܼܒ݂ܚܵܐ ܙܡܲܪܘ ܠܡܲܠܟܵܐ. ܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ ܙܡܲܪܝ ܬܸܫܒ݁ܘܿܚܬܵܐ. ܠܚܲܬ݂ܢܵܐ ܕܡܲܟ݂ܪܹܟ݂ܝ ܘܐܲܥܠܹܟ݂ܝ ܠܩܲܝܛܘܿܢܹܗ. ܘܝܲܗ݇ܒ݂ܠܹܟ݂ܝ ܐ݇ܪ̈ܵܙܵܘܜ ܪܲܗܒ݂ܘܿܢܵܐ ܚܘܼܣܵܝܵܐ ܠܝܲܠܕܲܝ̈ܟ݁ܝ. ܘܚܲܣܝܹܟ݂ܝ ܘܩܲܕ݂ܫܹܟ݂ܝ ܒܛܲܝܒ݁ܘܼܬܹܗ ܡܢ ܛܲܡܐܘܼܬ݂ܵܐ ܕܕܲܝܘܹ̈ܐ܀ ܘܫܲܟ݂ܠܸܠ ܘܐܵܘܪܸܒ݂ ܒܸܢܝܵܢܲܝ̈ܟ݁ܝ. ܒܚܹܝܠ ܦܘܼܪܩܵܢܵܐ ܕܐܲܠܵܗܘܼܬܹܗ. ܘܲܩܪܵܐ ܠܥܲܡܡܹ̈ܐ ܘܐܸܡ̈ܘܵܬ݂ܵܐ. ܕܲܢܩܲܪܒ݂ܘܼܢ ܒܹܟ݂ܝ ܕܸܒ݂ܚܹ̈ܐ ܠܲܫܡܹܗ. ܘܗܵܐ ܣܵܓ݂ܕܝܼܢ ܒܹܟ݂ܝ ܒܲܚܕ݂ܵܐ ܫܲܘܝܘܼ. ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܠܐܲܠܵܗܘܼܬܹܗ܀ ܘܩܵܥܹܝܢ ܟܠܗܘܿܢ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ. ܩܲܕ݁ܝܼܫ ܡܵܪܝܵܐ ܫܪܸܐ ܒܢܘܼܪܵܐ ܠܹܗ ܫܘܼܒ݂ܚܵܐ. ܒܲܪ݇ܬ݂ ܡܲܠܟܵܐ ܒܫܘܼܒ݂ܚܵܐ ܩܵܡܲܬ݀. ܙܡܲܪܝ ܥܹܕ݁ܬܵܐ ܒܦܘܼܡ ܝܲܠܕܲܝ̈ܟ݁ܝ. ܫܘܼܒ݂ܚܵܐ ܠܲܡܫܝܼܚܵܐ ܡܲܠܟܹܟ݂ܝ. ܕܲܒ݂ܝܲܕ݂ ܚܘܼܒܹ݁ܗ ܙܲܡܸܢ ܘܲܩܪܵܟ݂ܝ ܠܡܲܥܡܘܿܕܝܼܬܹܗ. ܘܝܲܗ݇ܒ݂ ܠܹܟ݂ܝ ܢܝܼܫܵܐ ܕܙܵܟ݂ܘܼܬ݂ܵܐ. ܨܠܝܼܒܹܗ ܚܲܝܵܐ ܕܒܹܗ ܙܵܟ݂ܝܲܬ݁ܝ. ܠܒ݂ܝܼܫܵܐ ܘܟ݂ܠܗܹܝܢ ܨܸܢܥܵܬܹ̈ܗ. ܒܟܸܢܫܹ̈ܐ ܪ̈ܚܝܼܡܹܐ ܕܲܒ݂ܢܲܝ̈ܟ݁ܝ ܦܘܿܩܝ ܠܐܘܼܪܥܹܗ. ܘܲܩܥܵܝ ܠܹܗ ܐܘܿܫܲܥܢܵܐ. ܒܪܝܼܟ݂ܘܼ ܡܲܠܟܵܐ ܕܡܢ ܕܵܘܝܼܕ݂ ܕܲܙܟ݂ܵܐ ܘܙܵܟ݁ܝܹܟ݂ܝ. ܘܲܥܬ݂ܝܼܕ݂ ܕܢܹܐܬܸܐ. ܒܲܫܡܹܗ ܕܡܵܪܝܵܐ. ܒܫܘܼܒ݂ܚܵܐ ܪܲܒܵܐ. ܘܢܲܣܩܹܟ݂ܝ ܥܲܡܹܗ ܠܲܫܡܲܝܵܐ.
ܕܲܨܠܘܿܬ݂ܵܐ. ܒܪܘܼܟܵܒ݂ ܟܲܕ݂ ܫܲܒ݂ܪܹ̈ܐ.
ܦܬܲܚܠܝܼ ܬܲܪ̈ܥܹܐ ܕܙܲܕܝܼܩܘܼܬ݂ܵܐ. ܥܲܠ ܬܲܪܥܹܗ ܕܲܚܢܵܢܵܟ݂ ܢܵܩܫ̇ܝܼܢ ܥܲܒ݂ܕܲܝ̈ܟ ܡܵܪܝܵܐ ܟܲܕ݂ ܡܸܬ݂ܟܲܫܦܝܼܢ ܘܫܵܐ̇ܠܝܼܢ ܡܸܢܵܟ݂. ܚܢܵܢܵܐ ܘܪ̈ܲܚܡܹܐ. ܕܬܸܥܢܸܐ ܐܸܢܘܿܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܘܐܲܣܵܐ ܟܹܐܒܹ̈ܐ ܕܗܲܕܵܡܲܝ̈ܗܘܿܢ. ܒܣܲܡܵܐ ܪܲܒܵܐ ܕܲܚܢܵܢܵܟ݂. ܫܡܲܥ ܡܵܪܝ ܒܵܥܘܼܬ݂ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܟ ܫܹܐ̈ܠܵܬܲܢ. ܕܐܲܢ݇ܬ݁ܘܼ ܐܝܼܬܲܝܟ ܐܝܼܵܠܵܐ ܕܟ݂ܠ ܕܐܲܠܝܼܨܝܼܢ. ܬܵܐ ܠܥܘܼܕ݂ܪܵܢܲܢ ܪܵܚܹܡ ܐ݇ܢܵܫܵܐ. ܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܕܒܲܝܬܵܝܵܘ̈ܗܝ. ܥܲܕܲܪܲܝܢ ܐܲܠܵܗܵܐ ܦܵܪܘܿܩܲܢ. ܥܘܼܕ݂ܪܵܢܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܡܵܪܵܐ ܛܵܒ݂ܵܐ ܡܠܸܐ ܪ̈ܲܚܡܹܐ. ܢܸܬ݂ܠܵܘܸܐ ܠܲܢ ܒܟ݂ܠ ܥܸܕܵܢܝܼ̈ܢ ܡܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂. ܘܢܸܒ݂ܗܲܬ݂ ܒܝܼܫܵܐ ܡܵܪܘܿܕ݂ܵܐ. ܘܬܸܬ݁ܪܝܼܡ ܩܲܪܢܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܒܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ. ܫܡܲܥ ܚܲܢܵܢܵܐ ܩܵܠ ܬܲܟ݂ܫܲܦ̮ܬܲܢ. ܘܬܸܥܘܿܠ ܩܕ݂ܵܡܲܝܟ ܒܵܥܘܼܬܲܢ. ܘܗܲܒ݂ܠܲܢ ܚܲܝܠܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܕܢܵܘܕܸܐ ܠܲܫܡܵܟ݂. ܘܲܢܙܲܡܲܪ ܠܵܟ݂ ܒܟ݂ܠ ܥܸܕܵܢܝܼ̈ܢ. ܡܵܪܵܐ ܪܵܚܹܡ ܬܲܝܵܒܹ̈ܐ. ܣܲܬܲܪܲܝܢܝ ܡܢ ܥܵܘܠܹܗ ܕܒ݂ܝܼܫܵܐ. ܬܚܹܝܬ݂ ܟܸܢܦܹ̈ܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܡܵܪܵܐ ܛܵܒ݂ܵܐ ܡܠܸܐ ܪ̈ܲܚܡܹܐ. ܣܲܬܲܪ ܐܸܢܘܿܢ ܠܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ ܡܢ ܢܸܟ݂ܝܵܢܹ̈ܐ. ܘܡܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܕܠܵܐ ܚܵܣܲܪ. ܙܘܼܢ ܡܵܙܘܿܢܵܐ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܠܵܟ݂ܘܼ ܩܵܪܹܝܢ ܒܟ݂ܠ ܥܸܕܵܢܝܼ̈ܢ. ܘܚܘܼܪ ܒܵܗ̇ ܒܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬ݂ܵܟ݂. ܕܚܵܝܪܵܐ ܠܲܫܡܵܟ݂ ܘܲܡܣܲܟ݁ܝܵܐ. ܘܲܣܥܘܿܪ ܒܪ̈ܲܚܡܹܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܠܝܲܠܕܹܝ̈ܗ̇ ܕܥܹܕܬܵܐ. ܕܐܲܢ݇ܬ݁ ܚܲܢܵܢܵܐ ܐܲܢ݇ܬ݁ ܘܣܲܓ݁ܝܼ ܪ̈ܲܚܡܹܐ. ܘܠܵܟ݂ ܡܸܬ݁ܬ݁ܚܝܼܒ݂ܵܐ ܬܵܘܕܝܼܬ݂ܵܐ. ܚܛܲܝܢ ܥܲܡ ܐܲܒ݂ܵܗܲܝ̈ܢ ܘܐܲܣܟܸ݁ܠܢ ܘܐܲܪܫܲܥܢ. ܥܲܠ ܕܲܚܛܲܝܢ ܡܵܪܹܟ݁ܠ ܐܸܫܬܲܠܲܛܝ̈ ܒܲܢ ܟܠ ܡܲܪ̈ܕܘܵܢ. ܘܗܵܐ ܚܕ݂ܝܼܪ̈ܵܢ ܠܲܢ ܟܠ ܒܝܼ̈ܫܵܬ݂ܵܐ ܕܲܢܓܲܡܪ̈ܵܢܲܢ. ܘܐܸܢ ܥܘܼܕ݂ܪܵܢܵܟ݂ ܠܵܐ ܡܲܕ݂ܪܸܟ݂ ܠܲܢ. ܡܸܢܫܹܠܝ ܐܵܒܹܕ݂ ܕܵܪܲܢ ܒܝܼܫܵܐ. ܒܥܵܘܠܲܢ ܕܲܣܓ݂ܝܼ ܡܢ ܟܠ ܕܵܪ̈ܝܼܢ.  ܬܸܟ݂ܪܸܐ ܠܵܟ݂ ܡܵܪܝ ܥܲܠ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܲܫܩܲܠܘ ܦܲܓ݂ܪܵܟ݂ ܘܲܕ݂ܡܵܟ݂ ܚܲܝܵܐ. ܐܲܝܟ݂ ܕܐܸܫܬܵܘܕܝܼܬ݁ ܒܲܣܒܲܪܬ݂ܵܟ݂ ܠܘܵܬ݂ ܬܲܠܡܝܼܕܲܝ̈ܟ. ܕܐܲܝܢܵܐ ܕܫܵܐܹܠ ܒܗܲܝܡܵܢܘܼܬ݂ܵܐ ܐ݇ܪܵܙ ܦܘܼܪܩܵܢܵܐ. ܢܵܣܹܒ݂ ܫܪܵܝܵܐ ܠܣܲܟ݂ܠܘܵܬܹ̈ܗ. ܚܲܣܵܢ ܘܦܲܨܵܢ ܡܢ ܚܛܵܗܲܝ̈ܢ ܡܸܛܠ ܫܡܵܟ݂. ܒܲܚܢܵܢܵܟ݂ ܦܵܪܘܿܩܲܢ ܚܲܣܵܐ ܚܵܘܒܲܝ̈ܢ ܘܲܚܛܵܗܲܝ̈ܢ. ܕܲܚܛܲܝܢ ܘܐܲܣܟܸ݁ܠܢ ܘܐܲܪܓܸ݁ܙܢ ܠܲܫܡܵܟ݂ ܒܲܥܒ݂ܵܕܲܝ̈ܢ. ܘܐܸܢ ܥܘܼܕ݂ܪܵܢܵܟ݂ ܠܵܐ ܡܲܕ݂ܪܸܟ݂ ܠܲܢ. ܐܵܟܹܠܩܲܪܨܵܐ ܒܲܚܣܵܡܹܗ. ܗܵܐ ܠܚܝܼܡ ܥܠܲܝܢ ܠܡܵܘܒܵܕ݂ܘܼܬܲܢ. ܣܚܘܿܦܲܝܗܝ ܘܒܲܛܸܠ ܫܘܼܠܛܵܢܹܗ. ܕܠܵܐ ܢܸܬ݂ܓܲܢ݇ܒܲܪ ܥܲܠ ܫܸܦܠܲܢ. ܕܐܲܢ݇ܬ݁ܘܼ ܣܲܒ݂ܪܲܢ ܘܒܹܝܬ݂ ܓܵܘܣܲܢ ܒܟ݂ܠ ܥܸܕܵܢܝܼ̈ܢ. ܢܲܛܲܪ ܠܥܲܒ݂ܕܲܝ̈ܟ ܡܢ ܟܠ ܢܸܟ݂ܝܵܢ. ܒܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ. ܐܸܬ݂ܦܲܢ ܥܠܲܝ ܘܪܲܚܸܡܥܠܲܝ. ܒܲܚܢܵܢܵܟ݂ ܦܵܪܘܿܩܲܢ ܚܲܣܵܐ ܚܵܘܒܲܝ̈ܢ ܘܲܚܛܵܗܲܝ̈ܢ. ܕܠܵܐ ܢܸܬ݁ܕܲܓܲܠ ܫܘܼܘܕܵܝ ܡܸܠܲܝ̈ܟ ܒܲܦܣܵܩ ܣܲܒ݂ܪܵܐ. ܕܲܚܛܲܝܢ ܘܐܲܪܓܸ݁ܙܢ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܗܵܐ ܟܪ̈ܝܼܟ݂ܵܢ ܠܲܢ ܟܠ ܡܲܪ̈ܕܘܵܬ݂ܵܐ ܕܲܢܓܲܡܪ̈ܵܢܲܢ ܡܸܛܠ ܕܲܚܛܲܝܢ. ܠܵܐ ܡܵܪܝ ܬܸܦܣܘܿܩ ܣܲܒ݂ܪܵܐ ܕܚܲܝܲܝ̈ܢ. ܕܐܲܢ݇ܬ݁ܘܼ ܐܝܼܬܲܝܟ ܒܹܝܬ݂ ܓܵܘܣܲܢ. ܐܸܠܵܐ ܒܪ̈ܲܚܡܲܝܟ ܚܘܼܣܥܠܲܝܢ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁. ܚܘܼܢ ܠܚܲܛܵܝܹ̈ܐ ܕܩܵܪܹܝܢ ܠܲܫܡܵܟ݂. ܡܵܪܵܐ ܚܵܐܹܢ ܠܣܵܓ݂ܘܿܕܵܘ̈ܗܝ. ܡܸܛܠ ܕܡܢ ܠܘܵܬ݂ܵܟ݂ ܗ݇ܘ ܫܘܼܒ݂ܩܵܢܵܐ. ܫܒ݂ܘܿܩ ܚܵܘܒܲܝ̈ܢ ܒܝܲܕ݂ ܪ̈ܲܚܡܲܝܟ. ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܵܐ ܕܣܵܓ݂ܘܿܕ݂ܵܘ̈ܗܝ. ܕܲܚܛܲܝܢ ܟܠܫܵܥ ܩܕ݂ܵܡ ܡܵܪܘܼܬ݂ܵܟ݂ ܒܲܥܪܝܼܪܵܐܝܼܬ݂. ܘܲܛܥܲܝܢ ܟܠܗܘܿܢ ܦܘܼܩܕܵܢܲܝ̈ܟ. ܘܲܗܘܲܝܢ ܥܲܒ݂ܕܹ̈ܐ ܠܲܚܛܝܼܬ݂ܵܐ. ܘܠܵܐ ܨܒܲܝܢ ܕܢܸܬ݂ܦܢܸܐ ܡܢ ܥܵܘܠܲܢ. ܘܗܵܫܵܐ ܕܲܩܪܸܒ݂ܢ ܠܲܬ݂ܝܵܒ݂ܘܼܬ݂ܵܐ. ܚܲܝܸܠ ܢܲܦ̮ܫܲܢ ܕܢܸܬ݂ܓܲܢ݇ܒܲܪ. ܘܗܲܒ݂ܠܲܢ ܕܲܢܩܘܼܡ ܡܢ ܡܲܦܘܼܠܬܲܢ ܘܢܸܫܦܲܪ ܩܘܼܕ݂ܡܲܝܟ. ܪܵܥܝܵܐ ܕܐܲܦܢܝܼ ܠܥܸܪܒܵܐ ܕܲܛܥܵܐ. ܐܲܦܢܵܢ ܨܹܐܕܲܝܟ ܒܲܚܢܵܢܵܟ݂. ܐܸܢ ܚܛܵܗܹ̈ܐ ܢܵܛܲܪ ܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܡܲܢܘܼ ܡܸܫܟܲܚ ܠܲܡܩܵܡ. ܚܘܼܢܲܝܢ ܡܵܪܝ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܡܫܝܼܚܵܐ ܡܵܪܵܗ̇ ܕܬܸܫܒ݁ܘܿܚܬܵܐ. ܘܠܵܐ ܢܸܬ݁ܕܲܓܲܠ ܫܘܼܘܕܵܝ ܡܸܠܲܝ̈ܟ ܗܵܘ̇ ܕܐܸܫܬܵܘܕܝܼܬ݁. ܕܟ݂ܠ ܕܩܵܪܹܐ  ܥܵܢܹܐ ܐ݇ܢܵܐ ܠܹܗ. ܘܢܵܣܹܒ݂ ܫܪܵܝܵܐ ܠܣܲܟ݂ܠܘܵܬܹ̈ܗ. ܘܫܘܼܒ݂ܩܵܢ ܚܵܘܒܵܘ̈ܗܝ ܘܲܚܛܵܗܵܘ̈ܗܝ. ܘܲܚܢܲܢ ܡܵܪܲܢ ܕܩܵܪܹܝܢܲܢ ܠܵܟ݂. ܫܡܲܥ ܒܵܥܘܼܬܲܢ ܘܲܨܠܘܿܬܲܢ. ܘܲܦܬܲܚ ܬܲܪܥܹܗ ܕܲܚܢܵܢܵܟ݂ ܠܲܣܢܝܼܩܘܼܬܲܢ. ܕܢܵܘܕܸܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܲܫܡܵܟ݂ ܪܲܒܵܐ. ܘܢܸܙܡܲܪ ܫܘܼܒ݂ܚܵܐ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܕܡܲܪܝܲܡ܀ ܗܵܝܕܹܝܢ ܢܹܐܡܪܘܼܢ ܒܹܝܬ݂ ܥܲܡܡܹ̈ܐ. ܛܘܼܒܲܝܟ݁ܝ ܠܹܟ݂ܝ ܒܲܪ݇ܬ݂ ܕܵܘܝܼܕ݂. ܒܬ݂ܘܼܠܬܵܐ ܕܟ݂ܝܼܬ݂ܵܐ ܘܩܲܕܝܼܫܬܵܐ. ܕܲܗܘܲܝܬ݁ܝ ܐܸܡܵܐ ܒܝܲܕ݂ ܕܲܟ݂ܝܘܼܬܹܟ݂ܝ ܠܲܡܠܸܟ݂ ܡܲܠܟܹ̈ܐ. ܘܒܹܟ݂ܝ ܐܸܬ݂ܚܲܬܲܡܘ ܓܸܠܝܵܢܹ̈ܐ. ܕܲܢܒ݂ܝܼܘܼܬ݂ܵܐ ܕܲܡܒܲܕܩܵܐ ܗ݇ܘܵܬ݂. ܒܝܲܕ݂ ܦܸܠܐܬ݂ܵܐ ܕܐ݇ܪܵܙ ܡܹܐܬ݂ܝܼܬܹܗ. ܘܡܸܛܠ ܗܵܕܹܐ ܛܘܼܒ݂ܵܐ ܪܓ݂ܝܼܓ݂ܵܐ. ܠܹܟ݂ܝ ܝܵܗܒ̈ܵܢ ܟܠ ܫܲܪ̈ܒ݂ܵܬ݂ܵܐ. ܕܲܗܘܲܝܬ݁ܝ ܠܡܹܐܢܵܐ ܡܢܝܼܚܵܢܝܼܬ݂ܵܐ ܠܛܘܼܦ̮ܣܵܐ ܕܟܹܐܢܹ̈ܐ. ܒܪܝܼܟ݂ܘܼ ܐܲܒ݂ܵܐ ܕܫܲܕܲܪ ܠܲܒ݂ܪܹܗ. ܘܦܲܪܩܹܗ ܠܓܸܢܣܲܢ ܡܢ ܛܘܼܥܝܲܝ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܡܵܪܝܵܐ ܒܚܲܝܠܵܟ݂ ܢܸܚܕܸܐ ܡܲܠܟܵܐ. ܒܝܲܕ݂ ܚܲܝܠܵܐ ܕܡܢ ܪܵܘܡܵܐ ܫܠܝܼܚܹ̈ܐ ܩܲܕܝܼܫܹ̈ܐ ܐܸܬ݂ܚܲܝܲܠܘ. ܘܲܠܟ݂ܠܗܘܿܢ ܛܪ̈ܘܼܢܹܐ ܫܲܥܒܸ݁ܕ݂ܘ ܘܲܙܟ݂ܵܘ ܒܚܹܝܠ ܥܘܼܕ݂ܪܵܢܵܟ݂. ܘܠܲܡܗܲܝܡܢܹ̈ܐ ܕܲܒ݂ܟ݂ܠ ܦܸܢ̈ܝܵܬ݂ܵܐ. ܒܝܲܕ݂ ܥܘܼܕ݂ܪܵܢܵܐ ܕܡܢ ܓܲܪ̈ܡܲܝܗܘܿܢ. ܐܲܚܠܸܡܘ ܐܸܢܘܿܢ ܒܗܲܝܡܵܢܘܼܬ݂ܗܘܿܢ. ܘܗܵܐ ܥܒ݂ܝܼܕ݂ܝܼܢ ܚܠܵܦܲܝܢ ܡܵܪܝ ܒܲܥܵܝܹ̈ܐ. ܕܲܢܦܝܼܣܘܼܢ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܕܲܒ݂ܝܲܕ݂ ܪ̈ܲܚܡܲܝܟ ܣܲܓ݁ܝܼ̈ܐܹܐ ܬܸܫܒ݁ܘܿܩ ܚܵܘܒܲܝ̈ܢ. ܘܐܲܫܘܵܐ ܠܟ݂ܠܲܢ ܕܒܲܨ̈ܠܵܘܵܬ݂ܗܘܿܢ. ܢܸܗܘܸܐ ܝܵܪ̈ܬܹ݁ܐ ܒܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܡܸܛܠ ܐܝܼܩܵܪܵܐ ܕܲܫܡܵܟ݂. ܠܲܨܠܝܼܒ݂ܵܟ݂ ܦܵܪܘܿܩܲܢ ܣܵܓ݂ܕܝܼܢ ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫܵܐ. ܘܲܡܙܲܝܚܝܼܢ ܠܹܗ ܥܲܠ ܐܝܼܕܲܝ̈ܗܘܿܢ ܘܙܵܡܪܝܼܢ ܫܘܼܒ݂ܚܵܐ. ܕܲܥܒܲܕ݂ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܒܲܛܸܠ ܩܪ̈ܵܒܹܐ ܘܬܲܟ݂ܬ݁ܘܼܫܹ̈ܐ. ܘܫܲܝܸܢ ܥܲܡܵܐ ܥܲܡ ܥܲܡܡܹ̈ܐ. ܕܒܲܚܕ݂ܵܐ ܥܹܕܬܵܐ ܠܹܗ ܢܸܬ݂ܟܲܫܦܘܼܢ. ܘܢܵܘܕܘܿܢ ܘܢܸܣܓ݁ܕ݂ܘܼܢ ܠܡܵܪܘܼܬܹܗ. ܘܢܸܩܥܘܿܢ ܒܩܵܠܹ̈ܐ ܩܲܕܝܼܫܹ̈ܐ ܩܲܕܝܼܫ ܩܲܕܝܼܫ. ܩܲܕܝܼܫܲܬ݁ ܡܵܪܝ ܕܲܒ݂ܝܲܕ݂ ܨܠܝܼܒ݂ܵܟ݂. ܦܪܲܩܬܵܝܗܝ ܠܓܸܢܣܲܢ ܡܢ ܛܘܼܥܝܲܝ. ܕܐܲܒ݂ܘܼܢ܀ ܢܸܬ݂ܩܲܝܲܡ ܠܥܵܠܲܡ ܩܕ݂ܵܡ ܐܲܠܵܗܵܐ. ܕܘܼܟ݂ܪܵܢܵܟ݂ ܐܘܿ ܐܲܒ݂ܘܼܢ. ܠܥܸܠ ܒܲܫܡܲܝܵܐ ܡܸܬ݂ܢܲܨܲܚ. ܘܲܠܬܲܚܬ݁ ܒܐܲܪܥܵܐ ܙܡܝܼ̈ܢܵܢ ܓܘܼܕܹ̈ܐ ܕܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ. ܕܲܒ݂ܥܲܡܠܵܐ ܪܲܒܵܐ ܕܐܵܓ݂ܘܿܢܵܟ݂. ܙܒܲܢܬ݁ ܠܵܟ݂ ܚܲܝܹ̈ܐ ܫܲܪܝܼܪܹ̈ܐ. ܒܥܘܼܡܪܵܐ ܦܲܐܝܵܐ ܕܲܫܡܲܝܵܐ. ܘܗܵܐ ܥܒ݂ܝܼܕܲܬ݁ ܣܢܹܐܓ݂ܪܵܐ ܚܠܵܦ ܢܲܦ̮̈ܫܵܬܲܢ. ܕܬܸܒ݂ܥܸܐ ܘܲܬ݂ܦܝܼܣ ܠܲܡܫܝܼܚܵܐ. ܕܲܒ݂ܝܲܕ݂ ܚܘܼܒܹ݁ܗ ܡܬ݂ܘܿܡܵܝܵܐ ܕܠܵܐ ܡܸܣܬܲܝܲܟ݂. ܢܲܫܘܸܐ ܠܟ݂ܠܲܢ ܠܲܓ݂ܢܘܿܢ ܢܘܼܗܪܵܐ. ܠܡܸܙܡܲܪ ܫܘܼܒ݂ܚܵܐ ܕܠܵܐ ܦܵܛܲܪ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܠܵܐ ܬܸܡܢܹܝܢܝ ܥܲܡ ܪ̈ܲܫܝܼܥܹܐ ܘܥܲܡ ܥܵܒ݂ܕܲܝ̈ ܥܵܘܠܵܐ. ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂ ܦܵܪܘܿܩܲܢ ܡܵܐ ܕܡܸܙܕܲܝܚܲܬ݁ ܡܢ ܥܝܼܪܹ̈ܐ. ܘܦܵܩܹܕ݂ ܪܸܡܙܵܟ݂ ܘܩܵܪܝܵܐ ܩܲܪܢܵܐ ܥܲܡ ܫܝܼܦܘܿܪܵܐ. ܘܩܵܝܡܝܼܢ ܡܝܼܬܹ̈ܐ ܡܢ ܩܲܒ݂ܪܹ̈ܐ. ܘܠܲܩܕ݂ܵܡ ܬܪܘܿܢܘܿܣ ܕܪܲܒ݁ܘܼܬ݂ܵܟ݂. ܛܵܒܹ̈ܐ ܘܒ݂ܝܼܫܹ̈ܐ ܡܸܬ݂ܟܲܢܫܝܼܢ. ܛܵܒܹ̈ܐ ܠܪܵܘܡܵܐ ܡܸܬ݂ܥܲܠܹܝܢ. ܘܒ݂ܝܼܫܹ̈ܐ ܦܵܝܫܝܼܢ ܥܲܠ ܐܲܪܥܵܐ. ܚܘܼܢ ܡܵܪܝ ܒܪ̈ܲܚܡܹܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܠܐܲܝܠܹܝܢ ܕܐܵܘܕܝܼܘ ܒܲܨܠܝܼܒ݂ܵܟ݂ ܘܐܲܫܘܵܐ ܐܸܢܘܿܢ ܕܥܲܡ ܩܲܕܝܼܫܲܝ̈ܟ ܢܸܬ݂ܒܲܣܡܘܼܢ ܒܐܘܿܪܸܫܠܸܡ. ܟܠܗܘܿܢ ܠܵܟ݂ܘܼ ܡܣܲܟܹ݁ܝܢ. ܠܓܸܠܝܵܢܵܟ݂ ܦܵܪܘܿܩܲܢ ܡܣܲܟܹ݁ܝܢ ܟܠܗܘܿܢ ܡܵܝܘܿܬܹ̈ܐ. ܕܐܸܡܲܬ݂ܝ ܬܸܕ݂ܢܲܚ ܘܬܸܩܪܸܐ ܩܲܪܢܵܐ ܥܲܡ ܫܝܼܦܘܿܪܵܐ. ܘܬܸܦܩܘܿܕ݂ ܠܐܲܪܥܵܐ ܒܝܲܕ݂ ܡܸܠܬ݂ܵܟ݂. ܘܬ݂ܵܘܠܸܕ ܟܠܗܘܿܢ ܦܲܓ݂ܪ̈ܲܝܗܘܿܢ. ܒܲܪܦܵܦ ܥܲܝܢܵܐ ܡܢ ܓܵܘ ܥܘܼܒܵܗ̇. ܛܵܒܹ̈ܐ ܠܪܵܘܡܵܐ ܡܥܲܠܸܐ ܐܲܢ݇ܬ݁. ܘܒ݂ܝܼܫܹ̈ܐ ܦܵܝܫܝܼܢ ܒܓ݂ܵܘ ܚܸܫܟܵܐ. ܒܗܵܘ ܥܸܕܵܢܵܐ ܦܵܪܘܿܩܲܢ ܚܘܼܣ ܥܲܠ ܥܲܒ݂ܕܲܝ̈ܟ. ܕܲܫܩܲܠܘ ܦܲܓ݂ܪܵܟ݂ ܘܐܸܫܬ݁ܝܼܘ ܠܲܕ݂ܡܵܟ݂. ܘܒܲܣܸܡ ܐܸܢܘܿܢ ܒܡܲܠܟ݁ܘܼܬ݂ܵܟ݂܀
ܕܥܲܢܝܼܕܹ̈ܐ. ܕܐܵܬܹ݁ܐ ܠܲܡܕ݂ܵܢܵܗ ܠܐܲܪܥܵܐ. ܡܲܠܟܵܐ ܡܢܲܚܡܵܢܵܐ ܒܫܘܼܒ݂ܚܹܗ ܕܵܢܲܚ ܡܢ ܪܵܘܡܵܐ. ܘܡܲܚܸܐ ܡܝܼ̈ܬܹܐ ܘܲܡܩܝܼܡ ܠܲܩܒ݂ܝܼܪܹ̈ܐ. ܘܩܵܝܡܝܼܢ ܫܲܘܝܵܐܝܼܬ݂. ܥܲܢܝܼ̈ܕܹܐ ܡܢ ܓܵܘ ܩܲܒ݂ܪܹ̈ܐ. ܘܡܲܣܩܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܡܲܚܸܐ ܡܝܼ̈ܬܹܐ. ܟܠܟ݂ܘܿܢ ܥܲܒ݂ܕܵܘ̈ܗܝ ܕܐܲܠܵܗܵܐ. ܐܲܥܒܲܪܘ ܟܲܪܝܘܼܬ݂ܵܐ ܡܢ ܠܸܒ݁ܟ݂ܘܿܢ ܐܘܿ ܡܵܝܘܿܬܹ̈ܐ. ܕܝܵܘܡܹܗ ܕܡܵܪܲܢ ܐܵܬܹ݁ܐ ܘܡܲܦܨܲܚ ܠܲܢ. ܘܲܡܥܝܼܪ ܕܲܡܟ݁ܘܼܬܲܢ. ܡܵܐ ܕܙܵܝܚܝܼܢ ܥܝܼܪܹ̈ܐ ܩܕ݂ܵܡܵܘܗܝ. ܚ̇ܵܕܹܝܢ ܡܲܠܲܐܟܹ̈ܐ ܒܝܘܼܡ ܢܘܼܚܵܡܵܐ.
	ܫܲܒܲܚ. ܒܝܲܕ݂ ܥܘܼܫܢܵܐ ܘܬ݂ܘܼܩܦܵܐ ܕܝܲܡܝܼܢܵܟ݂ ܐܲܢ݇ܬ݁ ܐܸܬ݂ܓ݁ܠܝܼܬ݁ ܘܲܦܪܲܩܬ݁ ܠܲܢ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܠܥܹܕ݁ܬܵܟ݂ ܩܲܝܸܡ ܒܲܚܢܵܢܵܟ݂ ܘܫܲܝܸܢ ܒܪ̈ܲܚܡܲܝܟ. ܘܲܠܟ݂ܠܗܘܿܢ ܐܲܝܠܹܝܢ ܕܥܵܡܪܝܼܢ ܒܵܗ ܒܲܪܸܟ݂ ܘܢܲܛܲܪ܀ ܚܲܝܠܵܟ݂ ܪܲܒܵܐ ܦܵܪܘܿܩܲܢ. ܗܲܒ݂ ܘܐܲܫܪܵܐ ܒܟ݂ܠܗܹܝܢ ܥܹܕܵܬ݂̈ܵܐ. ܘܟ݂ܵܗܢܹ̈ܐ ܕܲܒ݂ܗܹܝܢ ܒܕܸܚܠܵܐ ܢܫܲܡܫܘܼܢ. ܟܲܕ݂ ܡܵܘܕܹ݁ܝܢ ܠܵܟ݂. ܘܐܲܢ݇ܬ݁ ܗܲܒ݂ܠܲܢ ܕܲܕ݂ܠܵܐ ܫܲܠܘܵܐ. ܒܐܵܘܝܘܼܬ݂ܵܐ ܘܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ. ܕܟ݂ܠܲܢ ܒܝܲܕ݂ ܚܘܼܒܵܐ ܫܲܠܡܵܐ. ܢܸܬ݂ܟܲܫܲܦ ܟܲܕ݂ ܡܵܘܕܹ݁ܝܢܲܢ ܠܵܟ݂܀ ܘܢܸܩܥܸܐ ܟܠܲܢ ܡܵܪܹܟ݁ܠ. ܐܲܢ݇ܬ݁ ܥܢܝܼܢ ܒܲܚܢܵܢܵܟ݂ ܘܪܲܚܸܡܥܠܲܝܢ.
ܥܵܠܲܡ: ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܠܸܠܝܵܐ ܕܟ݂ܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܫܚܝܼܡܹ̈ܐ: ܠܒܲܪ ܡܢ ܨܵܘܡܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܐ:
ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܚܘܼܣ ܥܲܠ ܥܲܡܵܟ݂: ܘܠܵܐ ܬܸܬܸ݁ܠ ܝܵܪܬ݁ܘܼܬ݂ܵܟ݂ ܠܚܸܣܕܵܐ ܘܡܘܼܝܵܩܵܐ ܕܥܲܡܡܹ̈ܐ. ܚܘܼܪ ܡܢ ܡܸܕ݂ܝܵܪ̈ܲܝܟ ܘܲܚܙܝܼ ܩܛܝܼܪܵܐ: ܕܣܵܒ݂ܠ̇ܝܼܢܲܢ ܡܢ ܒܥܸܠܕ݁ܒ݂ܵܒܹ̈ܐ. ܘܬܸܕ݂ܢܲܚ ܠܲܢ ܒܲܥܓܲܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܕܠܵܐ ܢܹܐܡܪܘܼܢ ܥܲܡܡܹ̈ܐ ܐܲܝܟܵܘ ܐܲܠܵܗܗܘܿܢ

܀

 ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܢܲܛܲܪ ܠܥܲܒ݂ܕܲܝ̈ܟ: ܡܢ ܣܵܢܐܹ̈ܐ ܕܡܲܓܵܢ ܕܨܵܒܹܝ̇ܢ ܠܡܸܚܛܲܦ ܕܠܵܐ ܕܝܼܠܗܘܿܢ. ܣܘܼܓ݂ ܒܐܲܦܲܝ̈ܢ ܩܘܼܫܬܵܟ݂ ܚܲܣܝܼܢܵܐ: ܕܢܸܚܙܘܿܢ ܣܵܢܐܲܝ̈ܢ ܘܢܸܒ݂ܗܬ݂ܘܼܢ. ܘܐܲܟ݂ܪܸܟ݂ ܠܲܢ ܫܘܼܪܵܐ ܒܛܝܼܠܘܼܬ݂ܵܟ݂: ܕܢܸܣܬܲܬܲܪ ܒܓ݂ܵܘܹܗ ܘܢܵܘܕܸ݁ܐ ܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂

܀

ܒܲܨܠܘܿܬ݂ܵܗ̇ ܕܝܵܠܹܕ݂ܬ݂ܵܟ݂ ܡܵܪܲܢ ܝܼܫܘܿܥ: ܫܲܝܸܢ ܥܵܠܡܵܐ ܟܠܹܗ ܕܲܫܓ݂ܝܼܫ ܘܲܕ݂ܠܝܼܚ ܒܲܚܛܵܗܵܘ̈ܗܝ. ܘܒܲܛܸܠ ܡܢ ܐܲܪܥܵܐ ܩܪ̈ܵܒܹܐ ܘܩܹܐܪ̈ܣܹܐ: ܘܫܲܝܸܢ ܟܵܗܢܹ̈ܐ ܥܲܡ ܡܲܠܟܹ̈ܐ. ܕܒ݂ܵܐܘܝܘܼܬ݂ܵܐ ܘܚܘܼܒܵܐ ܢܸܙܕܲܝܲܚ: ܕܘܼܟ݂ܪܵܢ ܝܵܠܹܕ݂ܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ

܀

 ܙܲܕ݁ܝܼܩܹ̈ܐ ܕܲܫܦܲܪܘ ܠܵܟ݂ ܡܵܪܲܢ ܝܼܫܘܿܥ: ܢܒ݂ܝܼܹ̈ܐ ܫܠܝܼܚܹ̈ܐ ܘܣܵܗܕܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ ܕܲܒ݂ܟ݂ܠ ܐܲܬܲܪ. ܗܸܢܘܿܢ ܢܦܝܼܣܘܼܢܵܟ݂ ܚܠܵܦ ܢܲܦ̮̈ܫܵܬܲܢ: ܕܬܸܥܒܸ݁ܕ݂ ܪ̈ܲܚܡܹܐ ܥܲܠ ܟܠܲܢ. ܘܐܲܫܘܵܐ ܠܲܢ ܕܥܲܡܗܘܿܢ ܢܵܘܕܸ݁ܐ ܠܵܟ݂: ܒܝܵܘܡܵܐ ܕܒܹܗ ܢܵܣ̇ܒ݁ܝܼܢ ܐܲܓ݂ܪܵܐ ܕܥܲܡܠܲܝ̈ܗܘܿܢ

܀

ܛܘܼܒܲܝܟ ܠܵܟ݂ ܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ: ܕܫܲܡܠܝܼܬ݁ ܪܲܗܛܵܟ݂ ܒܙܵܟ݂ܘܼܬ݂ܵܐ. ܘܲܦܠܲܚܬ݁ ܒܟܲܪܡܹܗ ܕܲܡܫܝܼܚܵܐ: ܘܲܢܣܲܒ݂ܬ݁ ܐܲܓ݂ܪܵܐ ܚܠܵܦ ܥܲܡܠܲܝ̈ܟ: ܕܹܝܢܵܪ ܛܘܼܒܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܨ̈ܠܵܘܵܬ݂ܵܟ݂ ܢܸܗܘ̈ܝܵܢ ܥܲܠ ܟܠܲܢ: ܘܢܸܫܬ݁ܘܸܐ ܕܥܲܡܵܟ݂ ܢܹܐܪܲܬ݂ ܡܲܠܟ݁ܘܼܬ݂ܵܐ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܒܵܚܵܐ: ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܢ ܥܸܠܵܝܹ̈ܐ: ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܢ ܬܲܚܬܵܝܹ̈ܐ: ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܘܗܸܕ݂ܪܵܐ ܘܪܘܼܡܪܵܡܵܐ: ܡܢ ܟܠ ܕܲܒ݂ܪܲܝܬ݁ ܘܐܲܬ݂ܩܸܢܬ݁ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ. ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܫܘܼܒܵܚܹ̈ܐ ܘܬܸܫ̈ܒ݁ܚܵܬ݂ܵܐ ܕܡܸܬ݂ܐܲܡܪܝܼܢ ܒܟ݂ܠܵܗ̇ ܫܲܢ݇ܬܵܐ: ܒܟ݂ܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܫܚܝܼܡܹ̈ܐ ܕܫܲܒ݁ܬ݂ܵܐ.
ܫܘܼܒܵܚܵܐ ܕܲܬ݂ܪܹܝܢܒ݁ܫܲܒܵܐ
(ܡܙ: ܝܓ:1‐12)
ܥܕܲܡܵܐ ܠܐܸܡܲܬ݂ܝ ܬܸܛܥܹܝܢܝ ܡܵܪܝܵܐ ܠܥܵܠܲܡ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܥܕܲܡܵܐ ܠܐܸܡܲܬ݂ܝ ܬܸܛܥܹܝܢܝ ܡܵܪܝܵܐ ܠܥܵܠܲܡ* ܥܕܲܡܵܐ ܠܐܸܡܲܬ݂ܝ ܬܲܗܦܸܟ݂ ܐܲܦܲܝ̈ܟ ܡܹܢܝ* ܥܕܲܡܵܐ ܠܐܸܡܲܬ݂ܝ ܬܣܝܼܡ ܬܲܟ݂ܪܝܼܬ݂ܵܐ ܒܢܲܦ̮ܫܝ* ܘܕ݂ܵܘܘܿܢܵܐ ܒܠܹܒ݁ܝ ܟܠܝܘܿܡ* ܥܕܲܡܵܐ ܠܐܸܡܲܬ݂ܝ ܢܸܬ݁ܬ݁ܪܝܼܡ ܒܥܸܠܕ݁ܒ݂ܵܒ݂ܝ ܥܠܲܝ* ܚܘܼܪ ܘܲܥܢܝܼܢܝ ܡܵܪܝܵܐ ܐܲܠܵܗܝ* ܘܐܲܢܗܲܪ ܥܲܝܢܲܝ̈ ܕܠܵܐ ܐܸܕ݂ܡܲܟ݂ ܠܡܵܘܬܵܐ* ܘܠܵܐ ܢܹܐܡܲܪ ܒܥܸܠܕ݁ܒ݂ܵܒ݂ܝ ܕܲܙܟܹܝ̇ܬܹܗ* ܘܐܵܠܘܿܨܲܝ̈ ܢܸܚܕ݁ܘܿܢ ܠܝܼ ܟܲܕ݂ ܐܹܙܘܼܥ* ܐܸܢܵܐ ܕܹܝܢ ܥܲܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܐܸܬ݁ܬܲܟ݂ܠܹܬ݂* ܢܸܪܘܲܙ ܠܹܒ݁ܝ ܒܦܘܼܪܩܵܢܵܟ݂* ܘܐܹܫܲܒܲܚ ܠܡܵܪܝܵܐ ܕܦܲܪܩܲܢܝ* ܫܲܒܲܚ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.ܓ: ܙܲܒ݂ܢܝܼ̈ܢ܀ ܥܵܠܲܡ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.ܓ: ܙܲܒ݂ܢܝܼ̈ܢ܀ ܥܕܲܡܵܐ ܠܐܸܡܲܬ݂ܝ ܬܸܛܥܹܝܢܝ ܡܵܪܝܵܐ ܠܥܵܠܲܡ.ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.ܓ: ܙܲܒ݂ܢܝܼ̈ܢ܀
ܬܸܫܒ݁ܘܿܚܬܵܐ ܕܠܸܠܝܵܐ ܕܲܬ݂ܪܹܝܢܒ݁ܫܲܒܵܐ
ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܡܵܪܝ ܐܲܒ݂ܪܵܗܵܡ ܡܲܠܦܵܢܵܐ.
ܐܸܬ݂ܦܲܢ ܥܲܠ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܦܵܪܘܿܩܲܢ. ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬܲܢ ܘܦܲܢܵܐ ܫܹܐ̈ܠܵܬܲܢ܀ ܕܐܲܢ݇ܬ݁ܘܼ ܒܹܝܬ݂ ܓܵܘܣܵܐ ܘܣܲܒ݂ܪܵܐ ܕܲܡܚܝܼܠܹ̈ܐ. ܒܥܘܼܕ݂ܪܵܢܵܟ݂ ܢܸܙܟܸ݁ܐ ܠܨܸܢܥܵܬܹ̈ܗ ܕܒ݂ܝܼܫܵܐ܀ ܕܗܵܐ ܨܵܠܹ̇ܐ ܟܠܫܵܥ ܦܲܚܹ̈ܐ ܠܢܲܦ̮̈ܫܵܬܲܢ. ܘܨܵܒܹܐ ܢܵܟ݂ܘܿܠܵܐ ܕܲܢܟܲܬ݁ܡܲܢ ܒܥܵܘܠܹܗ܀ ܫܲܕܲܪ ܡܵܪܝ ܚܲܝܠܵܟ݂ ܘܲܢܦܲܣܸܩ ܢܸܫܒܵܘ̈ܗܝ. ܕܠܵܐ ܢܨܘܼܕܲܢ ܘܢܸܦܸܠ ܒܗܵܘܬ݂ܵܐ ܕܲܚܛܝܼܬ݂ܵܐ܀ ܚܲܝܠܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܢܚܲܝܸܠ ܡܚܝܼܠܘܼܬܲܢ. ܕܢܸܫܦܲܪ ܒܲܥܒ݂ܵܕܲܝ̈ܢ ܩܕ݂ܵܡܲܝܟ ܚܲܢܵܢܵܐ܀ ܘܐܲܫܘܵܢ ܕܢܸܙܡܲܪ ܠܵܟ݂ ܒܝܵܘܡܵܐ ܕܡܹܐܬ݂ܝܼܬ݂ܵܟ݂ . ܫܘܼܒ݂ܚܵܐ ܕܠܵܐ ܦܵܛܲܪ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܟܵܪܘܿܙܘܼܬ݂ܵܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.܀
ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ݁ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡܠܲܟ݂ ܛܵܒ݂̈ܵܬ݂ܵܐ ܠܪ̈ܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕ݁ܬܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܠܵܐ ܨܵܒܹܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ: ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ܡܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܗܘܝܘ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܪ̈ܝܵܬ݂ܵܐ: ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܝܼܣ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܝܼܣ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܵܘ̇ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ̇ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
