ܬܠܵܬ݂ܒ݁ܫܲܒܵܐ ܕܐܲܪܒ݁ܥܵܐ ܕܡܵܪܝ ܡܘܼܫܹܐ.
ܕܲܐ݇ܚܪ̈ܵܝܹܐ
ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܖ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܢܵܘܕܸ݁ܐ ܡܵܪܝ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܡܵܪܘܼܬ݂ܵܟ݂ ܘܫܘܼܒ݂ܚܵܐ ܐܲܡܝܼܢܵܐ ܕܠܵܐ ܫܲܠܘܵܐ ܢܲܣܸܩ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܡܲܖ̈ܡܝܵܬ݂ܵܐ. ܠܘܵܬ̣ܵܟ̣ ܡܵܪܝܵܐ ܢܲܦ̮ܫܝ ܐܲܪܝܼܡܹ݁ܬ̣* ܠܵܟ̣ ܡܵܪܝܵܐ ܩܪܹܝ݁ܬ̣ ܐܲܠܵܗܝ ܠܵܐ ܬܸܫܬ̇ܘܿܩ ܡܹܢܝ ܚܙܝܼ؛ ܡܙܡܘܖ̈ܐ ܟܗ-ܠ.
ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܬܲܖ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܫܘܼܒܼܚܵܐ ܠܵܟܼ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܫܘܼܒܼܚܵܐ ܠܵܟܼ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ.
ܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܘܐܵܡܲܪ ܟܵܗܢܵܐ ܨܠܘܿܬܼܵܐ: ܘܥܲܠ ܐܲܦܲܝ̈ ܟܠܗܘܿܢ ܥܘܼܕ݂ܖ̈ܵܢܲܝܟ ܘܛܲܝ̈ܒ݁ܘܵܬ݂ܵܟ݂ ܕܲܠܘܵܬܲܢ ܕܠܵܐ ܡܸܬ݂ܦܲܪܥܝܼܢ: ܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ ܒܥܹܕ݁ܬܵܟ݂ ܡܟܲܠܲܠܬܵܐ ܡܲܠܝܲܬ݂ ܟܠ ܥܘܼܕ݂ܖ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܵܐ ܘܒ݂ܵܪܘܿܝܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܬܲܖ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
(ܓܝܼܘܿܪܵܐ ܕܲܒ݂ܥܹܕ݇ܬܵܐ.( ܚܕ݂ܝܼܬ݂ ܟܲܕ݂ ܐܵܡ̇ܪܝܼܢ ܗ݇ܘ̣ܵܘ ܠܝܼ: ܠܒܲܝܬܹ݁ܗ ܕܡܵܪܝܵܐ ܐܵܙ̇ܠ݇ܝܼܢܲܢ(ܘܲܠܒܲܪ ܡ̣ܢ ܥܹܕ݁ܬܵܐ ܐܵܡ̇ܪܝܼܢ) ܒܟ݂ܠܕܘܼܟ ܐܝܼܬܲܝܟ ܐܲܠܵܗܵܐܩܲܒܸ݁ܠ ܬܲܟ݂ܫܲܦ̮ܬܲܢ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ:ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ:ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܟ. ܐܲܢ݇ܬ݁ܘܼ ܡܵܪܝ ܫܲܪܝܼܪܵܐܝܼܬ݂ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܘܲܡܢܲܛܪܵܢܵܐ ܐܲܡܝܼܢܵܐ ܕܚܲܝܲܝ̈ܢ: ܘܠܵܟ݂ ܡܵܪܝ ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
(ܘܐܵܡܪܝܼܢ ܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܡ ܒܓ݂ܘܼܕܵܐ ܕܩܲܕܡܵܝܹ̈ܐ)

 ܫܲܒܲܚ܀ ܥܵܠܲܡ܀
ܥܘܿܢ̈ܝܵܬ݂ܵܐ ܕܲܩܕ݂ܵܡ. ܪܹܫܩܵܠܵܐ.
ܐܲܢܗܲܪ ܐܲܦܲܝ̈ܟ ܘܢܸܬ݂ܦܪܸܩ. ܚܲܢܵܢܵܐ ܘܲܡܠܸܐ ܖ̈ܲܚܡܹܐ: ܠܵܐ ܬܲܗܡܸܐ ܚܝܵܪܵܟ݂ ܡ̣ܢܢ. ܫܲܕܲܪ ܠܲܢ ܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂: ܚܢܵܢܵܐ ܘܖ̈ܲܚܡܹܐ ܘܦܘܼܪܩܵܢܵܐ. ܥܕܲܡܵܐ ܕܲܬ݂ܪܲܚܸܡ ܥܠܲܝܢ. ܒܬܲܪܥܵܟ݂ ܡܵܪܲܢ ܢܵܩ̇ܫܝܼܢܲܢ: ܘܡܸܢܵܟܼ ܖ̈ܲܚܡܹܐ ܫܵܐܠܝܼܢܲܢ. ܦܬܲܚ ܠܲܢ ܘܦܲܢܵܐ ܫܹܐ̈ܠܵܬܲܢ: ܛܵܒ݂ܵܐ ܕܠܵܐ ܟܵܠܹ̇ܐ ܖ̈ܲܚܡܵܘܗܝ. ܫܲܒܲܚ. ܨܠܘܿܬ݂ܵܗ̇ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܡܲܪܝܲܡ: ܐܸܡܹܗ ܕܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ. ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܫܘܼܪܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ. ܥܵܠܲܡ. ܢܒ݂ܝܼܹ̈ܐ ܨܲܠܵܘ ܥܲܠ ܫܲܝܢܵܐ: ܘܲܫܠܝܼܚܹ̈ܐ ܥܲܠ ܫܠܵܡܵܐ: ܘܣܵܗܕܹ̈ܐ ܘܟ݂ܵܗܢܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ. ܨܠܘܿܬ݂ܟ݂ܘܿܢ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܲܢ. ܢܹܐܡܲܪ. ܒܥܝܼ ܘܐܲܦܝܼܣ ܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ: ܠܲܡܫܝܼܚܵܐ ܕܲܪܚܸܡ̣ܬ݁ ܚܘܼܒܹ݁ܗ. ܕܟܸܢܫܵܐ ܕܙܲܝܲܚ̣ ܕܘܼܟ݂ܪܵܢܵܟ݂: ܢܸܬ݂ܥܲܕܲܪ ܒܲܨ̈ܠܵܘܵܬ݂ܵܟ݂.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܡܵܪܝܵܐ ܩܪܹܝ̇ܬ݂ܵܟ݂ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܖ̈ܵܝܹܐ: ܠܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ: ܘܠܲܒ݂ܛܝܼܠܘܼܬܹܗ ܕܨܸܒ݂ܝܵܢܵܟ݂ ܕܲܥܠܲܝܢ: ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܫܲܒܲܚ܀ ܥܵܠܲܡ܀
ܥܘܿܢ̈ܝܵܬ݂ܵܐ ܕܒ݂ܵܬܲܪ. ܒܚܲܢܵܢܵܐ ܕܲܦܬ݂ܝܼܚ
ܫܵܘܙܸܒܲܝܢܝ ܘܦܲܨܵܢܝ ܡ̣ܢ ܕܵܪܵܐ ܗܵܢܵܐ ܠܥܵܠܲܡ. ܐܲܠܵܗܵܐ ܕܫ̣ܵܘܙܸܒ݂ ܠܲܕ݂ܒܹܝܬ݂ ܚܹܙܲܩܝܵܐ: ܘܦܲܨ̣ܝܵܗ̇ ܠܐܘܿܪܸܫܠܸܡ ܡ̣ܢ ܐܵܬ݂ܘܿܪܵܝܵܐ. ܐܲܓܸ݁ܢ ܝܲܡܝܼܢܵܟ݂ ܘܲܦܪܘܿܩ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܘܠܵܐ ܡܵܪܝ ܬܲܝܬܸ݁ܐ ܥܠܲܝܢ ܪܸܓ݂ܠܵܐ ܥܲܫܝܼܢܬܵܐ. ܡ̣ܢ ܠܸܒܹ݁ܗ ܥܲܡܝܼܩܵܐ. ܟܲܕ݂ ܩ̣ܪܵܟ݂ ܒܲܪ ܐܝܼܫܲܝ ܡܵܪܝܵܐ ܘܲܥ̣ܢܲܝܬܵܝܗܝ: ܠܐܘܼܠܨܵܢܵܘ̈ܗܝ ܐܲܪܘܲܚ̣ܬ݁ ܘܠܲܒ݂ܥܸܠܕ݁ܒ݂ܵܒ݂ܵܘ̈ܗܝ ܣ̣ܚܲܦܬ݁. ܘܲܚܢܲܢ ܕܩܵܪܹܝ̇ܢܲܢ ܒܲܛܸܠ ܖ̈ܵܕ݂ܘܿܦܲܝܢ: ܘܩܲܝܸܡ ܥܹ̈ܕ݂ܵܬ݂ܵܐ ܕܠܲܨܠܝܼܒ݂ܵܟ݂ ܣܵܓ݂̈ܕܵܢ. ܫܲܒܲܚ. ܒܹܝܬ݂ ܓܵܘܣܵܐ ܗ݇ܝ̣ ܪܲܒܵܐ ܒܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܠܲܡܗܲܝܡܢܹ̈ܐ ܕܟ݂ܠܫܵܥ ܫܵܐ̇ܠܝܼܢ ܨ̈ܠܵܘܵܬ݂ܵܗ̇. ܒܚܲܝܠܵܐ ܕܲܨܠܘܿܬ݂ܵܗ̇ ܟܸܢܫܲܢ ܢܸܬ݂ܒܲܪܲܟ݂: ܘܥܹܕ݇ܬܵܐ ܬܸܬ݂ܟܲܗܲܢ ܒܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ. ܥܵܠܲܡ. ܢܒ݂ܝܼܹ̈ܐ ܘܲܫܠܝܼܚܹ̈ܐ ܘܣܵܗܕܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ: ܒܥܵܘ ܡ̣ܢ ܐܲܠܵܗܵܐ ܖ̈ܲܚܡܹܐ ܥܲܠ ܥܵܠܡܵܐ. ܠܟ݂ܵܗܢܹ̈ܐ ܐܵܘܝܘܼܬ݂ܵܐ ܠܡܲܠܟܹ̈ܐ ܬܲܪܥܘܼܬ݂ܵܐ: ܠܥܹܕ݇ܬܵܐ ܥܲܡ ܝܲܠܕܹ݁ܝ̈ܗ̇ ܫܘܼܒ݂ܩܵܢܵܐ ܕܚܵܘܒܹ̈ܐ. ܢܹܐܡܲܪ. ܐܲܒ݂ܘܼܢ ܩܲܕ݁ܝܼܫܵܐ ܪܵܚܡܹܗ ܕܲܡܫܝܼܚܵܐ: ܕܲܙܒܲܢ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܒܥܲܡܠܹ̈ܐ ܕܕ݂ܘܼܒܵܖ̈ܵܘܗܝ. ܒܥܝܼ ܚܠܵܦܲܝܢ ܖ̈ܲܚܡܹܐ ܡ̱ܢ ܡܵܪܵܟ݂ ܕܲܓ݂ܒ݂ܵܟ݂: ܕܲܢܪܲܚܸܡ ܥܠܲܝܢ ܘܢܸܦܪܘܿܩ ܢܲܦ̮̈ܫܵܬܲܢ.
(ܘܪܵܡܹܐ ܡܫܲܡܫܵܢܵܐ ܐܘܼܪܵܪܵܐ: ܘܐܵܣܲܪ ܙܘܼܢܵܪܵܐ: ܘܩܵܐܹ̇ܡ ܩܕ݂ܵܡ ܫܟ݂ܝܼܢܬܵܐ ܡܛܲܟ݁ܣܵܐܝܼܬ݂(ܘܐܵܡܲܪ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ: ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܫܵܐ ܘܒܲܚܦܝܼܛܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܐܲܒ݂ܵܐ ܕܖ̈ܲܚܡܹܐ ܘܐܲܠܵܗܵܐ ܕܟ݂ܠܹܗ ܒܘܼܝܵܐܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܦܵܪܘܿܩܲܢ ܘܝܲܨܘܼܦܬ݂ܵܢܲܢ ܘܲܡܦܲܪܢ̱ܣܵܢܵܐ ܕܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ ܘܩܘܼܝܵܡܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ ܘܲܕ݂ܟ݂ܠܗܹܝܢ ܥܹ̈ܕ݂ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܐܲܬ݂ܪܲܢ ܘܥܲܠ ܟܠ ܐܲܬ݂ܖ̈ܵܘܵܢ: ܘܥܲܠ ܐܲܝܠܹܝܢ ܕܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܥܵܡ̇ܪܝܼܢ ܒܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܡܘܼܙܵܓ݂ܵܘ̈ܗܝ ܕܐܵܐܲܪ ܘܟܲܗܝܼܢܘܼܬ݂ܵܐ ܕܫܲܢ݇ܬܵܐ ܘܥܲܠ̈ܠܵܬ݂ܵܐ ܕܦܹܐܖܹ̈ܐ ܘܥܲܠ ܬܘܼܩܵܢܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܡܲܠܟܹ̈ܐ ܐܲܚܝܼܕܲܝ̈ ܫܘܼܠܛܵܢܹܗ ܕܥܵܠܡܵܐ ܗܵܢܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܖ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܩܲܫܝܼܫܹ̈ܐ ܘܲܡܫܲܡܫܵܢܹ̈ܐ ܐܲܖ̈ܬܵܕ݁ܘܿܟ݁ܣܹܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ܗ̇ܘ ܕܥܲܬ݁ܝܼܪ ܒܖ̈ܲܚܡܵܘܗܝ ܘܲܫܦܝܼܥ ܒܲܚܢܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܡ̣ܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ ܐܝܼܬ݂ܵܘܗܝ ܘܲܠܕ݂ܵܪܕܵܖ̈ܝܼܢ ܩܲܝܵܡ ܫܘܼܠܛܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܒܲܟ݂ܝܵܢܹܗ ܐܝܼܬ݂ܵܘܗܝ ܛܵܒ݂ܵܐ ܘܝܵܗܘܿܒ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹܐ ܒܡܵܘܬܹ݁ܗ ܕܚܲܛܵܝܵܐ ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡ̣ܢ ܥܵܘ̣ܠܹܗ ܘܢܸܚܸܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܲܒ݂ܝܲܠܕܹ݁ܗ ܐܵܘ ܕܲܒ݂ܕܸܢܚܹܗ ܐܵܘ ܕܲܒ݂ܨܵܘܡܹܗ ܐܵܘ ܕܲܒ݂ܡܲܥܲܠܬܹ݁ܗ ܐܵܘ ܕܒܲܩܝܵܡܬܹ݁ܗ ܐܵܘ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܐܵܘ ܕܲܒ݂ܡܲܚܲܬ݁ܬܹ݁ܗ ܐܵܘ ܕܒܲܨܠܝܼܒܹܗ: ܩܲܕ݁ܝܼܫܵܐ (ܐܵܘ ܩܲܕܝܼܫܬܵܐ) ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܐܲܦܨܲܚ ܠܲܫܡܲܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܠܹܗ ܗܘ̤ ܟܝܵܢܵܐܝܼܬ݂ ܐܝܼܬ݂ ܠܵܐ ܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܘܝܲܬ݁ܝܼܪܵܐܝܼܬ݂ ܕܐܲܝܠܹܝܢ ܕܲܡܗܲܝܡܢܝܼܢ ܒܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܣܓܵܐ ܥܲܡܲܢ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀

	ܘܲܡܨܲܠܸܐ ܟܵܗܢܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ. ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ ܡܲܓ݂ܥ̱ܠܝܼܢܲܢ ܦܲܓ݂ܖ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ: ܘܡܸܢܵܟܼ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܫܵܐ̇ܠܝܼܢܲܢ ܫܘܼܒ݂ܩܵܢܵܐ ܕܚܵܘܒܹ̈ܐ ܘܕܲܚܛܵܗܹ̈ܐ: ܗܲܒ݂ܠܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܖ̈ܲܚܡܲܝܟ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ ܟܵܪܘܿܙܘܼܬ݂ܵܐ: ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ.
ܘܐܵܡ̇ܪܝܼܢ ܒܩܵܠܵܐ ܦܫܝܼܛܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܫܲܒܲܚ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܥܵܠܲܡ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܒ݂ܵܬܲܪ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܖ̈ܵܝܹܐ. ܩܲܕ݁ܝܼܫܵܐ ܘܲܫܒ݂ܝܼܚܵܐ ܘܚܲܝܠ̱ܬ݂ܵܢܵܐ ܘܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܕܲܒ݂ܩܲܕ݁ܝܼܫܹ̈ܐ ܥܵܡܲܪ ܘܡܸܬ݁ܬ݁ܢܝܼܚ ܨܸܒ݂ܝܵܢܹܗ: ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ ܐܸܬ݂ܦܲܢ ܡܵܪܝ: ܘܚܘܼܣ ܘܪܲܚܸܡ ܥܠܲܝܢ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ. ܒܲܪܸܟ݂ ܡܵܪܝ: ܐܲܪܟܸ݁ܢܘ ܖܹ̈ܫܲܝܟ݁ܘܿܢ ܠܲܣܝܵܡܝܼܕ݂ܵܐ ܘܩܲܒܸ݁ܠܘ ܒܘܼܪܟ݁ܬ݂ܵܐ.
ܘܐܵܡܲܪ ܪܲܒܵܐ. ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܬܸܫܡܸܫܬܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
	ܘܡܲܩܸܦ ܨܠܘܿܬ݂ܵܐ. ܘܟܲܕ݂ ܫܵܠ̈ܡܵܢ ܢܲܦ̮̈ܫܵܬܲܢ ܠܲܚܕ݂ܵܐ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܓܡܝܼܪܬܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܟܠܲܢ ܒܲܚܕ݂ܵܐ ܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ: ܢܸܫܬ݁ܘܸܐ ܕܢܲܣܸܩ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ...
ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܥܘܿܢܝܼܬܼܵܐ ܕܪܲܡܫܵܐ.
      ܥܘܿܢܝܼܬܼܵܐ ܕܪܲܡܫܵܐ. ܒܗܵܐ ܫ̣ܒ݂ܵܢܝ. ܛܘܼܒܲܝܗܘܿܢ ܠܐܲܝܠܹܝܢ ܕܲܕ݂ܠܵܐ ܡܘܼܡ ܐܸܢܘܿܢ ܒܐܘܼܪܚܵܐ: ܐܲܛܐܸܒ݂ ܡܵܪܝܵܐ ܠܛܵܒܹ̈ܐ. ܛܘܼܒ݂ܵܐ ܪܲܒܵܐ ܕܲܡܠܝܼܟ̣ ܠܐܲܝܠܹܝܢ ܕܲܒ݂ܦܘܼܩܕܵܢܲܝ̈ܟ ܡܸܬ݁ܕܲܒ݁ܪܝܼܢ ܥ̣ܒܲܪ ܡ̣ܢ ܨܹܐܕܲܝ. ܘܲܚܠܵܦ ܛܘܼܒ݂ܵܐ ܗܵܐ ܩܢܹ݁ܝܬ݂ ܘܵܝܵܐ ܥܲܡ ܕܵܘܘܿܢܵܐ ܥܲܠ ܕܐܲܪܦܝܼܬ݂ܵܟ̣ ܘܲܫܒܲܩ̣ܬ݂ܵܟ̣. ܠܵܐ ܡܵܪܝ ܬܲܪܦܹܝܢܝ ܘܠܵܐ ܬܲܗܡܸܐ ܡܹܢܝ. ܐܸܠܵܐ ܒܥܝܼܵܢܝ ܒܲܚܢܵܢܵܟ̣. ܕܠܵܘ ܒܲܦܝܵܣܵܐ ܐܸܬ݂ܓܲܒ݂ܠܹ݁ܬ݂. ܐܵܦܠܵܐ ܒܪܘܼܓ݂ܙܵܟ̣ ܐܸܬ݂ܚܲܒܲܠ. ܛܵܒ݂ܵܐ ܕܓܲܒ݂ܠܲܢ̣ܝ ܚܘܼܣ ܥܠܲܝ. ܡܵܪܝܵܐ ܕܲܒ݂ܪܵܢ̣ܝ ܚܲܠܸܠܲܝܢܝ. ܘܗܲܒ݂ ܠܝܼ ܐܝܼܕ݂ܵܐ ܕܥܘܼܕ݂ܖ̈ܵܢܹܐ ܕܐܸܦܘܿܩ ܘܐܸܬ݂ܟܲܬܲܫ ܥܲܡ ܒܝܼܫܵܐ ܕܲܫ̣ܒ݂ܵܢܝ ܡܸܢܵܟ̣: ܬܢܝܼ. ܫܲܒܲܚ. ܫܠܵܡܵܐ ܘܫܲܝܢܵܐ ܢܸܣܓܸ݁ܐ ܠܥܲܡܵܟ̣ ܫܲܝܢܹܗ ܕܥܵܠܡܵܐ. ܡܫܝܼܚܵܐ ܕܐܸܬ݂ܵܐ ܠܦܘܼܪܩܵܢܲܢ. ܙܪܘܿܥ ܐܵܘܝܘܼܬ݂ܵܐ ܒܲܒ݂ܢܹ̈ܝܗ݁ ܕܥܹܕܬܵܐ ܕܢܸܗܘܘܿܢ ܙܵܡ݁ܪܝܼܢ ܥܲܡ ܥܝܼܖܹ̈ܐ ܫܘܼܒ݂ܚܵܐ ܠܲܫܡܵܟ̣. ܘܩܲܝܸܡ ܟܵܗܢܹ̈ܐ. ܘܫܲܝܸܢ ܡܲܠܟܹ̈ܐ. ܘܒܲܛܸܠ ܩܖ̈ܵܒܹܐ ܡ̣ܢ ܣܵܘ̈ܦܹܝܗ݁ ܕܐܲܪܥܵܐ. ܘܢܲܛܲܪ ܠܟܸܢܫܵܐ ܕܣܵܓ݂݁ܘܿܕܲܝ̈ܟ. ܕܠܵܟ̣ܘܼ ܩ̇ܵܪܹܝܢ ܒܟ̣ܠܥܸܕܵܢ. ܘܲܣܥܘܿܪ ܡܲܖ̈ܥܹܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ̣. ܘܐܲܚܠܸܡ ܟܖ̈ܝܼܗܹܐ ܒܲܚܢܵܢܵܟ̣. ܘܐܵܘܫܸܛ ܐܝܼܕ݂ܵܐ ܕܥܘܼܕ݂ܖ̈ܵܢܹܐ. ܠܟ̣ܠܗܘܿܢ ܐܲܝܠܹܝܢ ܕܐܲܠܝܼܨܝܼܢ. ܡܵܪܲܢ ܚܘܼܣܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܐܵܬ݂̈ܘܵܬ݂ܵܐ: ܒܓ݂ܘܼܕܵܐ ܕܒ݂ܵܬܲܪ. ܚܘܼܣ ܥܠܲܝܢ ܐܲܢ݇ܬ݁ ܚܲܢܵܢܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܸܬ݂ܦܲܢ ܥܠܲܝܢ ܡܠܸܐ ܖ̈ܲܚܡܹܐ: ܘܠܵܐ ܡܵܪܝ ܬܲܗܡܸܐ ܚܝܵܪܵܟ݂ ܘܲܒ݂ܛܝܼܠܘܼܬ݂ܵܟ݂ ܡ̣ܢܢ: ܕܲܥܠܲܝܟ݁ܘܼ ܣܲܒ݂ܪܲܢ ܘܬ݂ܘܼܟ݂ܠܵܢܲܢ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ: ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
(ܘܐܵܡܪܝܼܢ ܐܵܬ݂ܘ̈ܵܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܖ̈ܵܝܹܐ)
(ܡܙ؛ ܩܝܚ؛ ܣ ؛ 1-8).
ܣܢܹ̇ܝܬ̣ ܠܥܵܘܵܠܹ̈ܐ ܘܢܵܡܘܿܣܵܟ݂ ܪܸܚܡܹ̇ܬ̣. ܗܲܠܸܠܘܼܝܲܗ ܗܲܠܸܠܘܼܝܲܗ ܐܹܗܲܠܸܠܘܼܝܲܗ* ܣܸܬܵܪܝ ܘܒܹܝܬ̣ ܓܵܘܣܝ ܐܲܢ݇ܬ̇ܘܼ ܘܲܠܡܸܠܬ̣ܵܟ݂ ܣܲܒ̇ܪܹܬ̣

*

ܥܒܲܪܘ ܡܹܢܝ ܥܵܘܵܠܹ̈ܐ ܕܐܸܛܲܪ ܦܘܼܩܕܵܢܵܘ̈ܗܝ ܕܐܲܠܵܗܝ

*

ܐܲܫܲܪܲܝܢܝ ܒܡܸܠܬ̣ܵܟ݂ ܘܐܸܚܸܐ ܘܠܵܐ ܬܲܒ̣ܗܸܬܲܝܢܝ ܡ̣ܢ ܣܲܒ̣ܪ̱ܝ

*

ܣܲܝܲܥܲܝܢܝ ܘܐܸܬ̣ܦܪܸܩ ܘܐܸܬ̣ܝܲܠܲܦ ܒܦܘܼܩܕܵܢܲܝ̈ܟ

*

ܒܟ̣ܠܙܒܲܢ ܐܲܣܠܝܼ݁ܬ̣ ܠܟ̣ܠ ܕܛܵܥܹ̇ܝܢ ܡܸܢܵܟܼ ܡܸܛܠ ܕܥܵܘܠܵܐ ܗ݇ܘ̣ ܪܸܢܝܼܗܘܿܢ

*

ܣܲܡܸܟܲܝܢܝ ܘܐܸܬ̣ܦܪܸܩ ܘܐܸܪܢܸܐ ܒܟ̣ܠܙܒܲܢ ܒܦܘܼܩܕܵܢܲܝ̈ܟ

*

ܩܦܘܿܕ݂ ܒܸܣܪ̱ܝ ܡ̣ܢ ܕܸܚܠ̱ܬ̣ܵܟ݂ ܘܡ̣ܢ ܕܝܼܢܲܝ̈ܟ ܕܸܚܠܹ̇ܬ̣. (ܡܙ؛ ܩܝܚ؛ ܥ ؛ 1-8).* ܥܵܒܹ̇ܕ݂ ܕܝܼܢܵܐ ܘܙܲܕ̇ܝܼܩܘܼܬ̣ܵܐ ܠܵܐ ܬܸܫܒ̇ܘܿܩܲܝܢܝ ܒܐܝܼܕ݂ܵܐ ܕܛܵܠܘܿܡܲܝ̈

*

ܒܲܣܸܡܲܝܗܝ ܠܥܲܒ̣ܕܵܟ݂ ܒܛܵـܒ̣̈ܵܬ̣ܵܐ ܘܠܵܐ ܢܸܥܫܩܘܼܢܵܢܝ ܫܲܒ̣ܗ̱ܖ̈ܵܢܹܐ

*

ܥܲܝܢܲܝ̈ ܣܲܟ̇ܝܼ ܠܦܘܼܪܩܵܢܵܟ݂ ܘܲܠܡܹܐܡܪܵܐ ܕܙܲܕ̇ܝܼܩܘܼܬ̣ܵܟ

݂*

ܥܒܸܕ݂ ܠܥܲܒ̣ܕܵܟ݂ ܐܲܝܟ݂ ܖ̈ܲܚܡܲܝܟ ܘܐܲܠܸܦܲܝܢܝ ܢܵܡܘܿܣܵܟ

݂*

ܥܲܒ̣ܕܵܟ݂ ܐ݇ܢܵܐ ܣܲܟܸ̇ܠܲܝܢܝ ܘܐܸܕܲܥ ܣܵܗܕ̇ܘܼܬ̣ܵܟ

݂*

ܙܲܒ̣ܢܵܐ ܗ݇ܘ̣ ܠܡܸܦܠ̱ܚܹܗ ܠܡܵܪܝܵܐ ܘܗܵܐ ܒܲܛܸܠܘ ܢܵܡܘܿܣܵܟ

݂*

ܡܸܛܠ ܗܵܢܵܐ ܪܸܚܡܹ̇ܬ̣ ܦܘܼܩܕܵܢܲܝ̈ܟ ܛܵܒ̣ ܡ̣ܢ ܕܲܗܒ̣ܵܐ ܘܛܵܒ̣ ܡ̣ܢ ܟܹܐܦܹ̈ܐ ܛܵܒ̣̈ܵܬ̣ܵܐ

*

ܟܠܗܘܿܢ ܦܘܼܩܕܵܢܲܝ̈ܟ ܪܸܚܡܹ̇ܬ̣ ܘܟ̣ܠܵܗ̇ ܐܘܼܪܚܵܐ ܕܥܵܘܵܠܹ̈ܐ ܣܢܹ̇ܝܬ̣

܀

ܫܲܒܲܚ܀ ܥܵܠܲܡ܀
ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܬܲܖ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܫܡܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܩܲܕ݁ܝܼܫܵܐ ܢܸܫܬܲܒܲܚ: ܘܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܬܸܣܬ݁ܓܸܕ݂: ܘܡܵܪܘܼܬ݂ܵܟ݂ ܬܸܬ݂ܝܲܩܲܪ: ܘܪܲܒ݁ܘܼܬ݂ܵܟ݂ ܬܸܬ݂ܢܲܨܲܚ: ܘܐܝܼܬ݂ܘܼܬ݂ܵܟ݂ ܬܸܬ݂ܪܲܡܪܲܡ: ܘܥܲܠ ܥܲܡܵܟ݂ ܘܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬ݂ܵܟ݂: ܢܓ݂ܘܼܠܘܼܢ ܖ̈ܲܚܡܹܐ ܡܬ݂ܘܿܡܵܝܹ̈ܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
	ܚܵܬ݂ܵܗ̇. ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܒܪܝܼܟ݂ܵܐ ܡܵܪܝ ܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܲܣܓ݂ܝܼܕ݂ܵܐ ܡܵܪܘܼܬ݂ܵܟ݂: ܩܲܕ݁ܝܼܫ ܘܲܫܒ݂ܝܼܚ ܘܲܡܫܲܒܲܚ: ܘܪܵܡ ܘܲܡܥܲܠܲܝ ܫܡܵܐ ܣܓ݂ܝܼܕ݂ܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܣܵܗ̈ܕܹ݁ܐ.
ܕܘܼܟ݂ܪܵܢܵܐ ܠܥܵܠܲܡ ܢܸܗܘܸܐ ܠܙܲܕ݁ܝܼܩܵܐ. ܐܸܣܛܲܦܵܢܘܿܣ ܐܘܼܪܚܵܐ ܕܪܲܫ̣: ܘܲܒ݂ܥܸܩܒ݂ܵܬܹ̈ܗ ܪܕ݂ܵܘ ܣܵܗܕܹ̈ܐ. ܘܥܲܡܹܗ ܕܚܲܬ݂ܢܵܐ ܡܸܬ݂ܒܲܣܡܝܼܢ: ܒܲܓ݂ܢܘܿܢ ܢܘܼܗܪܵܐ ܕܠܵܐ ܡܸܫܬ݁ܪܸܐ. ܘܡ̣ܢ ܛܸܒܵܐ ܒܝܼܫܵܐ ܠܵܐ ܢܸܕ݂ܚܲܠ. ܐܸܣܛܲܦܵܢܘܿܣ ܟܲܕ݂ ܐܸܬ݂ܪ̱ܓܸܡ: ܙܝܼܘܵܐ ܕܡܵܪܹܗ ܒܪܵܘܡܵܐ ܚ̣ܙܵܐ: ܘܲܠܪܘܼܚܩܘܼܕ݂ܫܵܐ ܟܲܕ݂ ܓܵܕ݂ܠܵܐ: ܟܠܝܼܠܵܐ ܠܪܹܫܵܐ ܕܲܡܗܲܝܡܢܹ̈ܐ. ܚܲܝܹ̈ܐ ܫܲܐ̣ܠܵܟ݂ ܘܝܲܗ̣݇ܒ݂ܬ݁ ܠܹܗ. ܐܸܣܛܲܦܵܢܘܿܣ ܖ̈ܲܚܡܹܐ ܫ̣ܐܸܠ: ܠܥܲܡܵܐ ܕܲܩ̣ܪܸܒ݂ ܠܲܪܓ݂ܘܼܡܝܹܗ. ܚܲܣܵܐ ܡܵܪܲܢ ܘܲܫܒ݂ܘܿܩ ܠܗܘܿܢ: ܠܵܐ ܓܹܝܪ ܝܵܕ݂ܥ̇ܝܼܢ ܡܵܢ ܥܵܒ݂ܕ݁ܝܼܢ. ܡ̣ܢ ܥܲܡܵܐ ܕܠܵܐ ܡܪܲܚܡܵܢ. ܐܸܣܛܲܦܵܢܘܿܣ ܟܲܕ݂ ܐܸܬ݂ܩܛܸܠ: ܥܲܠ ܩܵܛܘܿܠܵܘ̈ܗܝ ܖ̈ܲܚܡܹܐ ܫ̣ܐܸܠ. ܒܲܕ݂ܡܘܼܬ݂ ܡܵܪܹܗ ܟܲܕ݂ ܐܸܙܕ݁ܩܸܦ: ܡ̣ܢ ܝܗ݇ܘܼܕ݂ܵܝܹ̈ܐ ܛܵܠܘܿܡܹ̈ܐ. ܘܐܸܡܲܪܘ ܫܲܦܝܼܪ ܒܩܵܠܵܐ. ܒܩܵܠܵܐ ܪܵܡܵܐ ܩ̣ܥܵܘ ܣܵܗܕܹ̈ܐ: ܩܕ݂ܵܡ ܕܲܝܵܢܹ̈ܐ ܟܲܕ݂ ܐܵܡ̇ܪܝܼܢ. ܕܠܵܐ ܟܵܦ̇ܪܝܼܢܲܢ ܒܲܡܫܝܼܚܵܐ: ܕܡܸܛܠܵܬܲܢ ܡܵܘܬܵܐ ܛ̣ܥܸܡ. ܥܲܕܲܪܲܝܢ ܐܲܠܵܗܵܐ ܦܵܪܘܿܩܲܢ. ܡܫܝܼܚܵܐ ܕܲܢܚܸܬ݂ ܥܲܡ ܣܵܗܕܹ̈ܐ: ܘܥܲܕܲܪ ܐܸܢܘܿܢ ܒܐܵܓ݂ܘܿܢܗܘܿܢ. ܗܘ̤ܝܼ ܠܲܢ ܙܲܝܢܵܐ ܕܠܵܐ ܡܸܙܕ݁ܟܸܐ: ܘܫܘܼܪܵܐ ܪܵܡܵܐ ܕܠܵܐ ܡܸܬ݂ܚܣܸܢ. ܠܲܝܬ݁ ܐܲܟ݂ܘܵܬ݂ܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܝ ܘܠܲܝܬ݁ ܐܲܝܟ݂ ܥܒ݂ܵܕܲܝ̈ܟ. ܠܲܝܬ݁ ܐܘܼܠܨܵܢܵܐ ܕܲܡܥܝܼܩ ܠܲܢ: ܕܢܸܟ݂ܦܘܿܪ ܘܢܸܥܪܘܿܩ ܡ̣ܢ ܣܸܓ݂ܕ݁ܬ݂ܵܟ݂ . ܕܐܲܠܵܗܹ̈ܐ ܕܥܲܡܡܹ̈ܐ ܒܵܛ̇ܠܝܼܢ: ܘܐܲܢ݇ܬ݁ ܒܲܠܚܘܿܕܲܝܟ ܠܵܐ ܡܸܫܬ݁ܪܹܝܬ݁. ܕܲܢܨܝܼܒ݂ܝܼܢ ܒܒܲܝܬܹ݁ܗ ܕܡܵܪܝܵܐ. ܐܲܝܟ݂ ܐܝܼܠܵܢܹ̈ܐ ܒܓ݂ܵܘ ܓܲܢܬ݂ܵܐ: ܣܝܼܡܝܼܢ ܣܵܗܕܹ̈ܐ ܒܓ݂ܵܘ ܗܲܝܟ݁ܠܵܐ: ܘܲܠܥܸܠ ܡܸܢܗܘܿܢ ܣܝܼܡ ܡܲܕ݂ܒ݁ܚܵܐ: ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܡܫܲܡܫܵܐ ܠܗܘܿܢ. ܡܫܲܡܫܵܢܵܘ̈ܗܝ ܕܥܵܒ݂ܕ݁ܝܼܢ ܨܸܒ݂ܝܵܢܹܗ. ܢܵܚ̇ܬ݁ܝܼܢ ܥܝܼܖܹ̈ܐ ܡܸܢ ܐܲܬ݂ܪ̱ܗܘܿܢ: ܘܙܵܡ̇ܪܝܼܢ ܫܘܼܒ݂ܚܵܐ ܒܩܝܼ̈ܢܵܬ݂ܗܘܿܢ: ܥܲܠ ܓܲܖ̈ܡܲܝܗܘܿܢ ܕܲܡܗܲܝܡܢܹ̈ܐ: ܕܲܥ̣ܒܲܕ݂ܘ ܨܸܒ݂ܝܵܢ ܒܵܪܘܿܝܗܘܿܢ. ܕܲܒܲܚܘ ܠܹܗ ܕܸܒ݂ܚܹ̈ܐ ܕܫܘܼܒ݂ܚܵܐ.ܣܵܗܕܹ̈ܐ ܕܝܼܩܸܕ݂ܘ ܒܸܣܡܹ̈ܐ ܗܘ̤ܵܘ: ܘܐܲܝܟ݂ ܥܸܩܵܖܹ̈ܐ ܒܢܘܼܪܵܐ ܢ̣ܦܲܠܘ: ܘܲܒ݂ܣܸܡ ܪܹܝܚܵܐ ܕܝܲܩܕܵܢܗܘܿܢ: ܐܲܝܟ݂ ܗܹܖ̈ܘܿܡܹܐ ܒܓ݂ܵܘ ܗܲܝܟ݁ܠܵܐ. ܘܐܲܖ̈ܙܹܐ ܕܠܸܒ݂ܢܵܢ ܕܲܢܨܲܒ݂ܬ݁.ܣܵܗܕܹ̈ܐ ܐܲܖ̈ܙܹܐ ܕܠܵܐ ܐܸܬ݂ܪ̱ܟܸܢܘ: ܩܕ݂ܵܡ ܕܲܝܵܢܹ̈ܐ ܩܵܘ̈ܡܵܬ݂ܗܘܿܢ. ܘܥܲܠ ܩܲܒ݂ܖ̈ܲܝܗܘܿܢ ܡܲܠܟܹ̈ܐ ܣܓܸܕ݂ܘ: ܕܡ̣ܢ ܓܲܖ̈ܡܲܝܗܘܿܢ ܢܸܬ݂ܥܲܕ݁ܪܘܼܢ. ܐܲܝܟ݂ ܡܕ݂ܝܼܢ݇ܬܵܐ ܕܲܟ݂ܪܝܼܟ݂ ܠܵܗ̇ ܫܘܼܪܵܐ.ܣܵܗܕܹ̈ܐ ܫܘܼܖܹ̈ܐ ܕܠܵܐ ܢܵܦ̇ܠܝܼܢ: ܘܢܸܒ݂ܥܹ̈ܐ ܒܖ̈ܝܼܟܹܐ ܕܠܵܐ ܓܵܝ̇ܙܝܼܢ. ܒܲܨ̈ܠܵܘܵܬ݂ܟ݂ܘܿܢ ܖ̈ܲܚܡܹܐ ܒܥܵܘ: ܡ̣ܢ ܚܲܢܵܢܵܐ ܥܲܠ ܥܵܠܡܵܐ. ܨܠܝܼ ܡܵܪܝܵܐ ܐܸܕ݂ܢܵܟ݂ ܘܲܥܢܝܼܢܝ. ܨܲܠ̣ܝܼ ܡܘܼܫܹܐ ܝܲܡܵܐ ܦ̣ܠܲܓ݂: ܨܲܠ̣ܝܼ ܫܸܡܥܘܿܢ ܠܒ݂ܝܼܫܵܐ ܙ̣ܟ݂ܵܐ: ܨܲܠ̣ܝܼ ܡܵܪܲܢ ܡܵܘܬܵܐ ܛ̣ܥܸܡ: ܘܐܲܚ̣ܝܼ ܠܐܵܕ݂ܵܡ ܕܐܲܒ݁ܝܼܕ݂ ܗ݇ܘ̣ܵܐ. ܡܲܢ̣ܘܼ ܕܦܵܚܹ̇ܡ ܠܡܵܪܝܵܐ ܒܲܫܡܲܝ̈ ܫܡܲܝܵܐ. ܚܲܫܹ̈ܐ ܕܙܲܒ݂ܢܵܐ ܠܵܐ ܦܵܚ̇ܡܝܼܢ: ܠܗ̇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܡܛܲܝܒ݂ܵܐ. ܡ̣ܢ ܪܹܫܝܼܬ݂ܵܐ ܠܲܡܗܲܝܡܢܹ̈ܐ: ܐܲܝܠܹܝܢ ܕܲܪܕ݂ܵܘ ܒܵܬܲܪ ܫܡܵܟ݂. ܡܵܐ ܛܵܒ݂ ܘܡܵܐ ܫܲܦܝܼܪ. ܡܵܐ ܚܠܸܐ ܪܹܝܚܵܐ ܕܗ̇ܝ ܓܲܢܬ݂ܵܐ: ܕܣܝܼܡܵܐ ܒܗܲܝܟ݁ܠܵܐ ܕܐܘܿܪܸܫܠܸܡ: ܘܥܵܐ̇ܠܝܼܢ ܣܵܗܕܹ̈ܐ ܘܡܸܬ݂ܒܲܣܡܝܼܢ: ܬܚܹܝܬ݂ ܛܸܠܵܠܵܐ ܕܣܵܘܟܲܝ̈ܗܘܿܢ. ܛܘܼܒܲܝܗܘܿܢ ܠܐܲܝܠܹܝܢ ܕܲܕ݂ܠܵܐ ܡܘܼܡ ܐܸܢܘܿܢ ܒܐܘܼܪܚܵܐ. ܛܘܼܒ݂ܵܐ ܠܣܵܗܕܹ̈ܐ ܡܵܐ ܕܫܵܡ̇ܥܝܼܢ: ܩܵܠܹܗ ܕܐܲܒ݂ܵܐ ܕܐܵܡܲܪ ܠܗܘܿܢ. ܕܬ݂ܵܘ ܥܘܿܠܘ ܘܝܼܪܲܬ݂ܘ ܡܲܠܟ݁ܘܼܬ݂ܵܐ: ܕܡ̣ܢ ܪܹܫܝܼܬ݂ܵܐ ܠܟ݂ܘܿܢ ܚܵܝ̇ܪܵܐ. ܘܲܡܗܲܠܟ݂ܝܼܢ ܒܢܵܡܘܿܣܹܗ ܕܡܵܪܝܵܐ. ܛܘܼܒ݂ܵܐ ܪܲܒܵܐ ܕܠܵܐ ܥܵܒܲܪ: ܐܸܫܬܵܘܕ݁ܝܼ ܡܵܪܲܢ ܝܼܫܘܿܥ. ܠܐܲܝܠܹܝܢ ܕܖ̈ܲܚ̣ܡܘܼܗܝ ܘܗܲܝܡܸܢ̣ܘ ܒܹܗ: ܘܲܢ̣ܛܲܪܘ ܟܠܗܘܿܢ ܦܘܼܩܕܵܢܵܘ̈ܗܝ. ܬܵܘ ܢܸܒ݂ܪܘܿܟ݂ ܘܢܸܣܓ̇ܘܿܕ݂ ܠܹܗ. ܬܵܘ ܢܸܬ݂ܒܲܪܲܟ݂ ܡ̣ܢ ܣܵܗܕܹ̈ܐ: ܬܵܘ ܢܸܬ݂ܒܲܪܲܟ݂ ܡ̣ܢ ܟܵܗܢܹ̈ܐ. ܨܠܘܿܬ݂ܗܘܿܢ ܕܣܵܗܕܹ̈ܐ ܘܲܕ݂ܟ݂ܵܗܢܹ̈ܐ: ܬܸܗܘܸܐ ܫܘܼܪܵܐ ܠܢܲܦ̮̈ܫܵܬܲܢ. ܐܲܠܵܗܵܐ ܡܲܢ̣ܘܼ ܕܕ݂ܵܡܹ̇ܐ ܠܵܟ݂. ܕܵܡܹ̇ܐ ܡܵܘܬ݁ܗܘܿܢ ܕܐܲܬ݂ܠܹܝܛܹ̈ܐ: ܠܗ̇ܘ ܡܵܘܬܵܐ ܕܡ̣ܝܼܬ݂ ܒܹܗ ܡܵܪܲܢ. ܕܝܼܠܹܗ ܕܐܸܬ݁ܬ݁ܠܝܼ ܒܲܙܩܝܼܦܵܐ: ܘܕ݂ܝܼܠܗܘܿܢ ܣܲܝܦܵܐ ܘܲܪܓ݂ܘܼܡܝܵܐ. ܠܐܲܝܠܹܝܢ ܕܲܬ݂ܪܝܼܨܝܼܢ ܒܠܸܒ݁ܗܘܿܢ. ܠܛܘܼܒܹ̈ܐ ܖ̈ܓ݂ܝܼܓܹܐ ܕܠܵܐ ܦܵܛ̇ܪܝܼܢ: ܙܡܝܼܢܝܼܢ ܣܵܗܕܹ̈ܐ ܩܲܕ݁ܝܼܫܹ̈ܐ. ܘܲܚܠܵܦ ܚܲܫܹ̈ܐ ܕܗܵܪܟܵܐ ܣ̣ܒܲܠܘ: ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܒܪܵܘܡܵܐ ܝܵܪܬ݁ܝܼܢ. ܛܵܣ̣ ܥܲܠ ܟܸܢܦܹ̈ܐ ܕܪܘܼܚܵܐ. ܙܵܚܲܬ̤ ܘܢܸܚܬܲܬ̤ ܡܲܪܟܲܒ݂ܬ݂ܵܐ: ܗ̇ܝ ܕܲܛܥܲܢܬ݂ܹܗ ܠܐܹܠܝܼܵܐ. ܘܒ݂ܵܗ̇ ܗ݇ܘ̣ ܣܵܠ̇ܩܝܼܢ ܩܲܕܝܼܫܹ̈ܐ: ܠܐܘܼܪܥܹܗ ܕܡܵܪܲܢ ܡܵܐ ܕܐܵܬܹܐ. ܡܵܐ ܛܵܒ݂ ܘܡܵܐ ܫܲܦܝܼܪ. ܡܵܐ ܫܲܦܝܼܪܝܼܢ ܣܸܕ݂ܖ̈ܲܝܗܘܿܢ: ܕܣܵܗܕܹ̈ܐ ܕܩܵܝ̇ܡܝܼܢ ܒܲܨܠܘܿܬ݂ܵܐ. ܘܲܚܠܸܐ ܩܵܠܵܐ ܕܢܸܥ̈ܡܵܬ݂ܗܘܿܢ: ܕܫܵܐ̇ܠܝܼܢ ܖ̈ܲܚܡܹܐ ܥܲܠ ܥܵܠܡܵܐ. ܙܲܕܝܼܩܵܐ ܘܲܬ݂ܪܝܼܨܵܐ. ܣܵܗܕܵܐ ܙܲܗܝܵܐ ܘܩܲܕܝܼܫܵܐ. ܡܵܪܝ ܩܘܼܪܝܵܩܘܿܣ ܢܲܨܝܼܚܵܐ. ܒܥܝܼ ܠܲܢ ܖ̈ܲܚܡܹܐ ܡ̣ܢ ܡܵܪܵܟ݂: ܕܢܸܫܬ݁ܘܸܐ ܠܫܘܼܒ݂ܩܵܢܵܐ ܕܚܵܘܒܹ̈ܐ. ܡ̣ܢ ܡܲܕ݂ܢܵܚܵܘ̈ܗܝ ܕܫܸܡܫܵܐ ܘܲܥܕܲܡܵܐ ܠܡܲܥܖ̈ܵܒ݂ܵܘܗܝ. ܗܵܐ ܒܟ݂ܠ ܦܸܢ̈ܝܵܢ ܡܸܙܕܲܝܲܚ: ܝܵܘܡܵܐ ܫܒ݂ܝܼܚܵܐ ܕܕ݂ܘܼܟ݂ܪܵܢܗܘܿܢ. ܕܡܵܪܝ ܣܲܪܓܝܼܣ ܘܲܕ݂ܡܵܪܝ ܒܵܟ݁ܘܿܣ: ܨܠܘܿܬ݂ܗܘܿܢ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܲܢ. ܘܲܨ̈ܠܵܘܵܬ݂ܵܟ݂ ܥܲܠ ܟܠܲܢ. ܫܠܵܡ ܠܵܟ݂ ܣܵܗܕܵܐ ܡܵܪܝ ܦܸܬ݂ܝܘܿܢ: ܓܹܐܙܲܒ݂ܪܵܐ ܪܘܼܚܵܢܵܝܵܐ. ܦܲܪܢܸܣ ܥܘܼܬ݂ܪܵܟ݂ ܠܲܣܢܝܼܩܹ̈ܐ: ܕܡܸܬ݂ܓܵܘܣܝܼܢ ܒܲܨ̈ܠܵܘܵܬ݂ܵܟ݂. ܙܥܘܿܖܹ̈ܐ ܥܲܡ ܪܵܘܖ̈ܒܹܐ. ܒܡܵܪܝ ܓܝܼܘܲܪܓ݁ܝܼܣ ܢܸܬ݂ܓܵܘܲܣ: ܕܲܒ݂ܚܲܝܠܵܐ ܕܲܨܠܵܘܵܬܹ̈ܗ. ܢܫܲܦܸܐ ܡܵܪܲܢ ܐܘܼܖ̈ܚܵܬܲܢ: ܘܢܲܩܸܠ ܝܘܼܩܪܵܐ ܕܗܲܕܵܡܲܝ̈ܢ. ܘܐܹܡܲܠܸܠ ܥܠܲܝܟ݁ܝ ܫܠܵܡܵܐ. ܫܠܵܡ ܠܹܟ݂ܝ ܡܲܠܝܲܬ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܐ: ܥܝܼܪܵܐ ܠܡܲܪܝܲܡ ܐܸܡܲܪ ܗ݇ܘ̣ܵܐ. ܡܵܪܝ ܥܲܡܹܟ݂ܝ ܒܲܪ݇ܬ݂ ܡܵܝܘܿܬܹ̈ܐ: ܕܡܸܢܹܟ݂ܝ ܕܵܢܲܚ̇ ܦܵܪܘܿܩܵܐ. ܫܲܒܲܚ. ܫܠܵܡܵܐ ܥܲܡܗܘܿܢ ܕܐܲܖ̈ܕܸܟ݂ܠܹܐ: ܕܲܒ݂ܢ̣ܵܘ ܒܝܼܪܬܵܐ ܕܠܵܐ ܢܵܦ̇ܠܵܐ. ܘܒܲܫܡܹܗ ܕܝܼܫܘܿܥ ܨܲܒܸ݁ܬ݂ܘ ܒܢ̣ܵܘ: ܓܢܘܿܢܵܐ ܒܪܵܘܡܵܐ ܕܠܵܐ ܡܸܫܬ݁ܪܸܐ. ܥܵܠܲܡ. ܟܠܝܼܠ ܙܵܟ݂ܘܼܬ݂ܵܐ ܕܐܵܓ݂ܘܿܢܵܟ݂: ܢܛܝܼܪܘܼ ܒܪܵܘܡܵܐ ܐܘܿ ܐܲܒ݂ܘܼܢ. ܕܲܒ݂ܝܵܘܡܵܐ ܐ݇ܚܪܵܝܵܐ ܬܸܣܒ݂ܝܼܘܗܝ: ܡ̣ܢ ܝܲܡܝܼܢ ܡܵܪܵܟ݂ ܕܐܵܘܪܒ݂ܵܟ݂. ܢܹܐܡܲܪ. ܡܫܲܠܸܡ ܪܲܗܛܹܗ ܕܐܝܼܡܵܡܵܐ: ܘܝܵܗܹ̇ܒ݂ ܠܸܠܝܵܐ ܠܲܢܝܵܚܵܐ. ܫܲܡܠܵܐ ܥܲܡܲܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܡܵܪܝܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ܀ ܡܫܝܼܚܵܐ ܕܢܲܛ̣ܪܲܢ ܒܐܝܼܡܵܡܵܐ: ܘܡܲܢ̣ܥܲܢ ܠܪܲܡܫܵܐ ܒܛܲܝܒ݁ܘܼܬܹܗ. ܗܲܒ݂ܠܲܢ ܠܸܠܝܵܐ ܕܲܢܝܵܚܵܐ. ܕܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܥܲܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܡܵܪܲܢ ܗܲܒ݂ ܫܲܝܢܵܐ ܒܐܲܬ݂ܪܲܢ: ܘܒܲܪܸܟ݂ ܠܟ݂ܠܹܗ ܦܘܼܠܚܵܢܲܢ. ܕܢܸܫܬ݁ܘܸܐ ܠܖ̈ܲܚܡܹܐ ܘܠܲܚܢܵܢܵܐ: ܘܓܲܠܝܘܼܬ݂ ܐܲܦܹ̈ܐ ܒܹܝܬ݂ ܕܝܼܢܵܐ܀ ܫܲܟܸ݁ܢ ܒܖ̈ܲܚܡܹܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܠܸܠܝܵܐ ܘܫܸܢܬ݂ܵܐ ܕܲܢܝܵܚܵܐ. ܠܟ݂ܠܗܘܿܢ ܟܖ̈ܝܼܗܹܐ ܘܐܲܠܝܼܨܹ̈ܐ: ܕܩܵܪܹܝ̇ܢ ܠܵܟ݂ ܒܐܘܼܠܨܵܢܲܝ̈ܗܘܿܢ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.

	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ ܒܲܪܸܟ݂ܡܵܪܝ.
	ܚܒܲܪܬ݂ܵܗ̇. ܒܲܨܠܘܿܬ݂ ܩܲܕ݁ܝܼܫܲܝ̈ܟ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܐܸܬ݂ܪܲܥܵܐ ܠܲܢ: ܘܲܒ݂ܒ݂ܵܥܘܼܬ݂ܗܘܿܢ ܕܫܲܪܝܼܖ̈ܲܝܟ ܐܲܥܒܲܪ ܚܛܵܗܲܝ̈ܢ: ܘܚܲܣܵܐ ܒܘܼܨܵܖ̈ܲܝܢ: ܘܫܲܝܸܢ ܒܥܸܠܕ݁ܒ݂ܵܒܲܝ̈ܢ: ܘܢܲܚܸܡ ܥܲܢܝܼܕܲܝ̈ܢ: ܘܐܲܫܘܵܐ ܠܲܢ ܠܫܘܼܒ݂ܚܵܐ ܦܲܐܝܵܐ ܕܡܲܠܟ݁ܘܼܬ݂ܵܟ݂: ܥܲܡ ܟܹܐܢܹ̈ܐ ܘܙܲܕ݁ܝܼܩܹ̈ܐ ܓܵܡܖ̈ܲܝ ܨܸܒ݂ܝܵܢܵܟ݂ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ: ܡܵܪܵܐ ܕܟ݂ܠ...

	ܘܡܲܩܦܝܼܢ. ܨܠܘܿܬ݂ܵܗ̇ ܡܵܪܝ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܗ̇ ܕܐܸܡܵܐ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܬܲܟ݂ܫܲܦ̮ܬܵܗ̇ ܘܬܲܚܢܲܢܬܵܗ̇ ܕܡܲܠܝܲܬ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܐ: ܡܵܪܬ݁ܝ ܡܲܪܝܲܡ ܛܘܼܒ݂ܵܢܝܼܬ݂ܵܐ: ܘܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ ܙܲܟܵܝܵܐ: ܘܥܘܼܕ݂ܪܵܢܵܐ ܐܲܠܵܗܵܝܵܐ: ܘܒ݂ܵܥܘܼܬܹܗ ܕܡܵܪܝ ܝܘܿܚܲܢܵܢ ܡܲܥ̱ܡܕ݂ܵܢܵܐ: ܘܕܲܫܠܝܼܚܹ̈ܐ ܛܘܼܒ݂ܵܢܹ̈ܐ: ܘܕܲܐܒ݂ܵܗ̈ܵܬ݂ܵܐ ܢܲܨܝܼܚܹ̈ܐ ܡܣܲܒ݁ܖ̈ܵܢܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܘܙܵܖ̈ܥܲܝ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܕܲܐܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܡܵܪܝ ܐܲܒ݂ܵܐ ܩܵܬ݂ܘܿܠܝܼܩܵܐ: ܘܡܵܪܝ ܐܸܣܛܲܦܵܢܘܿܣ ܒܘܼܟ݂ܪܵܐ ܕܣܵܗܕܹ̈ܐ: ܘܲܕ݂ܓܲܢ݇ܒܵܪ ܚܲܝܠܵܐ ܡܵܪܝ ܓܝܼܘܲܪܓܝܼܣ ܣܵܗܕܵܐ ܢܲܨܝܼܚܵܐ: ܘܲܕ݂ܡܵܪܝ ܦܠܵܢ: ܘܲܕ݂ܡܵܪܝ ܐܵܘܓܹ݁ܝܢ ܛܘܼܒ݂ܵܢܵܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܣܝܼܥܬܹ݁ܗ ܪܘܼܚܵܢܵܝܬܵܐ. ܘܕܲܫܡܘܿܢܝܼ ܘܕܲܒ݂ܢܹܝ̈ܗ̇: ܘܲܕ݂ܟ݂ܠܗܘܿܢ ܣܵܗܕܵܘ̈ܗܝ ܘܩܲܕ݁ܝܼ̈ܫܵܘܗܝ ܕܡܵܪܲܢ: ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܫܘܼܪܵܐ ܪܵܡܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܚܲܣܝܼܢܵܐ: ܕܬܸܦܪܘܿܩ ܘܲܬ݂ܦܲܨܸܐ ܘܲܬ݂ܫܵܘܙܸܒ݂ ܘܲܬ݂ܢܲܛܲܪ ܠܦܲܓ݂ܖ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ ܡ̣ܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹܗ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹ̇ܝܢ: ܐܵܡܹܝܢ.
ܘܟܸܢ ܚܵܬ݂ܡ̇ܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܖ̈ܵܝܹܐ. ܘܢܵܣܹ̇ܒ݂ ܟܵܗܢܵܐ ܨܠܝܼܒ݂ܵܐ ܒܐܝܼܕܹܗ ܕܝܲܡܝܼܢܵܐ: ܘܩܵܐܹ̇ܡ ܒܓܲܒܵܐ ܬܲܝܡ̱ܢܵܝܵܐ: ܘܡܲܗܦܸܟ݂ ܐܲܦܵܘ̈ܗܝ ܠܥܲܡܵܐ ܘܐܵܡܲܪ: ܒܲܪܸܟ݂ ܡܵܪܝ ܒܦܘܼܩܕܵܢܟ݂ܘܿܢ. ܘܥܵܢܹ̇ܝܢ: ܒܦܘܼܩܕܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܡܵܪܲܢ: ܘܫܘܼܒ݂ܚܵܐ ܠܲܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ. ܘܡܲܪܟ݁ܢܝܼܢ ܥܲܡܵܐ ܖܹ̈ܫܲܝܗܘܿܢ.
ܘܡܲܩܸܦ ܚܘܼܬܵܡܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ: ܨܸܡܚܹܗ ܕܐܲܒ݂ܵܐ ܡܬ݂ܘܿܡܵܝܵܐ: ܝܼܠܝܼܕ݂ܵܐ ܕܠܵܐ ܫܘܼܪܵܝܵܐ: ܠܥܸܠ ܡ̣ܢ ܙܲܒ݂ܢܹ̈ܐ ܘܲܗܘܲܝ̈ܵܐ: ܕܠܲܝܬ݁ ܠܲܢ ܣܲܒ݂ܪܵܐ ܘܣܘܼܟܵܝܵܐ: ܐܸܠܵܐ ܐܸܢ ܐܲܢ݇ܬ݁ ܒܵܪܘܿܝܵܐ. ܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܘܲܓ݂ܒܲܝ̈ܵܐ: ܕܲܫ̣ܦܲܪܘ ܠܵܟ݂ ܡ̣ܢ ܫܘܼܪܵܝܵܐ: ܥܒܸܕ݂ ܠܲܚܛܵܗܲܝ̈ܢ ܚܘܼܣܵܝܵܐ: ܘܲܠܣܲܟ݂ܠ̱̈ܘܵܬܲܢ ܫܪܵܝܵܐ: ܘܲܠܥܵܩ̈ܵܬܲܢ ܦܘܼܨܵܝܵܐ: ܘܲܠܫܸܐ̈ܠܵܬܲܢ ܦܘܼܢܵܝܵܐ: ܘܡܲܢܲܥܲܝܢ ܠܢܘܼܗܪܵܟ݂ ܓܲܐܝܵܐ: ܘܢܲܛܲܪܲܝܢ ܒܢܝܼܫܵܟ݂ ܚܲܝܵܐ: ܡ̣ܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܟܣܲܝ̈ܵܐ ܘܲܓ݂ܠܲܝ̈ܵܐ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ). ܗܵܫܵܐ ♰ ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܪܵܫܹ̇ܡ ܨܠܝܼܒ݂ܵܐ ܥܲܠ ܥܲܡܵܐ: ܟܲܕ݂ ܦܵܫܹ̇ܛ ܐܝܼܕܹܗ ܠܥܸܠ ܠܐܲܦܲܝ̈ ܡܲܥ̱ܪܒ݂ܵܐ: ܘܡܲܝܬܸ݁ܐ ܠܵܗ̇ ܠܡܲܕ݂ܢܚܵܐ: ܘܟܸܢ ܦܵܫܹ̇ܛ ܠܵܗ̇ ܠܓܲܒܵܐ ܕܬܲܝ̱ܡܢܵܝܵܐ: ܘܡܵܘܒܸ݁ܠ ܠܵܗ̇ ܠܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ. ܐܲܝܟܲܢܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܸܣܬ݁ܡܸܟ݂ ܥܲܠ ܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܓܲܒܵܐ ܣܸܡܵܠܵܝܵܐ ܕܟܸܢܫܵܐ.
ܘܥܵܢܹ̇ܝܢ: ܡܫܝܼܚܵܐ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬ݂ܵܟ݂: ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܟܵܗܢܘܼܬ݂ܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
ܐ݇ܚܪܹܢܵܐ: ܬܸܬ݂ܩܲܒܲܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܚܲܠܵܫܹ̈ܐ: ܩܕ݂ܵܡ ܒܹܝܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܬܸܗܘܸܐ ܟܢܝܼܫܘܼܬܲܢ ܗܵܕܹܐ ܠܲܢܝܵܚ ܨܸܒ݂ܝܵܢ ܡܵܪܘܼܬ݂ܵܟ݂: ܠܡܸܣܲܒ݂ ܡܸܢܵܟܼ ܒܛܲܝܒ݁ܘܼ ܚܘܼܠܡܵܢܵܐ ܕܦܲܓ݂ܪܵܐ: ܘܢܘܼܛܵܪܵܐ ܕܢܲܦ̮ܫܵܐ: ܘܫܘܼܪܬܵܚܵܐ ܕܡܵܙܘܿܢܵܐ: (ܘܚܘܼܣܵܝܵܐ ܕܚܵܘܒܹ̈ܐ ܘܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ): ܘܫܲܝܢܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ: ܘܲܫܠܵܡܵܐ ܢܲܓ̇ܝܼܪܵܐ: ܘܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ ܕܠܵܐ ܥܵܒ݂ܪܵܐ ܘܠܵܐ ܡܸܫܬܲܪܝܵܐ ܡ̣ܢ ܒܲܝܢܵܬܲܢ: ܒܟ݂ܠܹܗ ܙܲܒ݂ܢܵܐ ܕܩܵܐܹ̇ܡ ܒܹܗ ܥܵܠܡܵܐ ܗܵܢܵܐ: )ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܒܪܝܼܟ݂ ܐܲܠܵܗܵܐ ܠܥܵܠܡܝܼܢ: ܘܲܡܫܲܒܲܚ ܫܡܹܗ ܩܲܕ̇ܝܼܫܵܐ ܠܕ݂ܵܪܕܵܖ̈ܝܼܢ: ܕܡ̣ܢܗ ܒܵܥܹܝ̇ܢܲܢ: ܘܲܠܝܲܡܵܐ ܫܦܝܼܥܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ: ܕܗ̤ܘ ܢܲܫܘܸܐ ܠܲܢ ܠܫܘܼܒ݂ܚܵܐ ܦܲܐܝܵܐ ܕܡܲܠܟ̇ܘܼܬܹܗ: ܘܲܠܒ݂ܘܼܣܵܡܵܐ ܕܥܲܡ ܡܲܠܲܐܟ݂ܵܘ̈ܗܝ ܩܲܕ̇ܝܼܫܹ̈ܐ: ܘܲܠܓܲܠܝܘܼܬ݂ ܐܲܦܹ̈ܐ ܕܲܩܕ݂ܵܡܵܘܗܝ: ܘܲܠܩܵܘܡܵܐ ܕܡ̣ܢ ܝܲܡܝܼܢܹܗ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ: ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܖ̈ܲܚܡܵܘܗܝ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟ̇ܠ ܕܐܸܬ݂ܟܲܢܲܫܢܲܢ ܒܒܲܝܬܹ݁ܗ: ܘܨܲܠܝܼܢܲܢ ܩܕ݂ܵܡ ܪܲܒ݁ܘܼܬܹܗ: ܥܲܠ ܣܲܒ݂ܪܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܗ̤ܘ ܢܸܫܡܲܥ ܨܠܘܿܬܲܢ ܒܲܚܢܵܢܹܗ: ܘܲܢܩܲܒܸ݁ܠ ܒܵܥܘܼܬܲܢ ܒܚܵܘܣܵܢܹܗ: ܘܲܢܫܝܼܓ݂ ܘܢܸܡܪܘܿܩ ܨܵܐܬ݂ܵܐ ܕܚܵܘܒܲܝ̈ܢ ܘܕܲܚܛܵܗܲܝ̈ܢ ܒܙܘܿܦܵܐ ܫܦܝܼܥܵܐ ܕܚܲܝܘܼܣܬ݂ܵܢܘܼܬܹܗ: ܘܲܢܢܝܼܚ ܢܲܦ̮̈ܫܵܬ݂ܵܐ ܕܥܲܢܝܼܕܲܝ̈ܢ ܒܐܵܘܵܢܹ̈ܐ ܫܒ݂ܝܼܚܹ̈ܐ ܕܡܲܠܟ̇ܘܼܬܹܗ: ܘܢܸܪܘܿܣ ܥܲܠ ܟܠܲܢ ܛܲܠܵܐ ܕܒܲܣܝܼܡܘܼܬܹܗ: ܘܲܥܠܲܝܢ ܘܥܲܠ ܟܠ ܒܸܖ̈ܝܵܢ ܬܲܓܸ̇ܢ ܝܲܡܝܼܢܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬܹܗ: ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܖ̈ܲܚܡܵܘܗܝ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟܠ ܕܝܲܗ݇ܒ݂ ܬܸܫܒ݁ܚܵܬܹ̈ܗ ܒܦܘܼܡܲܢ: ܘܲܙܡܝܼܖ̈ܵܬܹܗ ܒܠܸܫܵܢܲܢ: ܘܩܝܼܢܵܬܹ̈ܗ ܒܓܲܓ̇ܖ̈ܵܬܲܢ: ܘܬ݂ܵܘܕ̇ܝܼܬܹܗ ܒܣܸܦ̈ܘܵܬܲܢ: ܘܗܲܝܡܵܢܘܼܬܹܗ ܒܠܸܒܵܘ̈ܵܬܲܢ: ܗ̤ܘ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬܲܢ: ܘܲܢܩܲܒܸ݁ܠ ܒܵܥ̈ܘܵܬܲܢ: ܘܢܸܬ݂ܪܲܥܸܐ ܒܬܲܟ݂ܫ̈ܦܵܬܲܢ: (ܘܲܢܚܲܣܸܐ ܚܵܘ̈ܒܵܬܲܢ:)ܘܲܢܦܲܢܸܐ ܠܲܢ ܫܸܐ̈ܠܵܬܲܢ ܫܲܦܝܼܖ̈ܵܬ݂ܵܐ ܠܵܐ ܥܕ݂ܝܼ̈ܠܵܬ݂ܵܐ. ܘܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܢܲܫܦܲܥ ܖ̈ܲܚܡܵܘܗܝ ܘܲܚܢܵܢܹܗ: ܥܠܲܝܢ ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡ̣ܢ ܣܵܟܹܗ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܢܸܫܬܲܒܲܚ ܫܡܹܗ ܕܐܲܠܵܗܵܐ ܡܵܪܹܟܠ ܒܲܝܢܵܬܲܢ: ܡܛܲܟ̇ܣܵܢܗܘܿܢ ܕܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ. ܘܲܥܠܲܝܢ ܡܚܝܼ̈ܠܹܐ ܘܚܲܛܵܝܹ̈ܐ: ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡ̣ܢ ܣܵܟܹܗ. ܘܥܲܠ ܥܹܕ݁ܬܵܐ ܩܲܕ̇ܝܼܫܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇: ܘܥܲܠ ܐܲܒ݂ܵܗܲܝ̈ܢ ܘܖ̈ܲܒܵܢܲܝܢ ܘܡܲܠܦܵܢܲܝ̈ܢ ܘܥܲܢܝܼܕܲܝ̈ܢ: ܘܥܲܢܝܼܕܹ̈ܐ ܕܲܦܪܲܫ̣ܘ ܘܫܲܢ̣ܝܼܘ ܡ̣ܢ ܒܲܝܢܵܬܲܢ: ܘܥܲܠ ܟܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܬܲܓܸ̇ܢ ܝܲܡܝܼܢܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬܹܗ ܘܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܠܐܲܠܵܗܵܐ ܫܘܼܒ݂ܚܵܐ: ܘܲܠܡܲܠܲܐܟܹ̈ܐ ܐܝܼܩܵܪܵܐ: ܘܲܠܣܵܛܵܢܵܐ ܒܸܗܬ݁ܬ݂ܵܐ: ܘܠܲܨܠܝܼܒ݂ܵܐ ܣܸܓ݂ܕ̇ܬ݂ܵܐ: ܘܲܠܥܹܕ݁ܬܵܐ ܪܘܼܡܪܵܡܵܐ: ܘܲܠܥܲܢܝܼܕܹ̈ܐ ܢܘܼܚܵܡܵܐ: ܘܲܠܬܲܝܵܒܹ̈ܐ ܩܘܼܒܵܠܵܐ: ܘܠܲܐܣܝܼܖܹ̈ܐ ܫܪܵܝܵܐ: ܘܲܠܡܲܖ̈ܥܹܐ ܘܲܟ݂ܖ̈ܝܼܗܹܐܚܘܼܠܡܵܢܵܐ ܘܐܵܣܝܘܼܬ݂ܵܐ: ܘܠܲܐܖ̈ܒܲܥܦܸܢܝܵܬܹܗ ܕܥܵܠܡܵܐ ܫܲܝܢܵܐ ܪܲܒܵܐ ܘܲܫܠܵܡܵܐ. ܘܐܵܦ ܥܠܲܝܢ ܕܐܝܼܬܲܝܢ ܡܚܝܼ̈ܠܹܐ ܘܚܲܛܵܝܹ̈ܐ: ܚܢܵܢܹܗ ܘܖ̈ܲܚܡܵܘܗܝ ܕܐܲܠܵܗܲܢ ܣܓ݂ܝܼܕ݂ܵܐ ܢܹܐܬ݂ܘܿܢ ܘܢܲܓ̇ܢܘܼܢ ܘܢܸܫܬܲܦܥܘܼܢ ܘܲܢܩܵܘܘܿܢ ܘܢܲܡ̱ܠܟ݂ܘܼܢ ܒܐܲܡܝܼܢܘܼܬ݂ܵܐ: )ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: )ܒܝܲܡܝܼܢ ܡܵܪܘܼܬ݂ܵܟ݂ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ): ܒܲܪܸܟ݂ܡܵܪܝ ܠܟ݂ܠܲܢ: ܘܢܲܛܲܪ ܠܟ݂ܠܲܢ: ܘܥܲܕܲܪ ܠܟ݂ܠܲܢ: ܘܣܲܝܲܥ ܘܣܲܬܲܪ ܠܟ݂ܠܲܢ: ܘܢܲܚܸܡ ܥܲܢܝܼܕܹ̈ܐ ܕܟ݂ܠܲܢ: ܘܐܲܓܸ̇ܢ ܝܲܡܝܼܢܵܟ݂ ܥܲܠ ܟܠܲܢ: ܘܖ̈ܲܚܡܲܝܟ ܘܲܚܢܵܢܵܟ݂ ܢܸܫܬܲܦܥܘܼܢ ܥܲܠ ܟܠܲܢ:ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ̇ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ̇ܬ݂ܵܐ: ܘܩܘܼܒܵܠܛܲܝܒ݁ܘܼܬ݂ܵܐ ܐܲܡܝܼܢܬܵܐ ܬܸܣܲܩ ܠܵܟ݂ ܡ̣ܢ ܦܘܼܡ ܟܠܲܢ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟ̇ܠ ܢܒܲܪܸܟ݂ ܠܟܸܢܫܲܢ ܒܛܲܝܒ݁ܘܼܬܹܗ: ܘܲܢܢܲܛܲܪ ܠܲܩܝܵܡܲܢ ܒܣܘܿܓ݂ܐܵܐ ܫܦܝܼܥܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܘܲܢܦܲܢܸܐ ܫܸܐ̈ܠܵܬܲܢ ܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܹܗ: ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡ̣ܢ ܣܵܟܹܗ: ܘܥܲܠ ܥܹܕ݁ܬܵܐ ܩܲܕ̇ܝܼܫܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇: ܘܥܲܠ ܐܲܬ݂ܪܵܐ ܗܵܢܵܐ ܘܥܵܡܘܿܖ̈ܵܘܗܝ: ܘܥܲܠ ܥܘܼܡܪܵܐ ܗܵܢܵܐ ܘܝܲܬܵܒ݂ܵܘ̈ܗܝ: ܘܥܲܠ ܟܠܲܢ ܒܟ݂ܠܲܢ ܟܢܝܼܫܵܐܝܼܬ݂: ܚܢܵܢܹܗ ܘܖ̈ܲܚܡܵܘܗܝ ܕܐܲܠܵܗܲܢ ܛܵܒ݂ܵܐ: ܢܹܐܬ݂ܘܿܢ ܘܢܸܫܬܲܦܥܘܼܢ ܒܐܲܡܝܼܢܘܼ (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܡܲܩܦܝܼܢ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܲܬ݂ܠܵܬ݂ܡܐܵܐ ܘܲܬ݂ܡܲܢܬܲܥܣܲܪ ܐܲܒ݂ܵܗ̈ܵܬ݂ܵܐ ܐܲܦܸܣܩܘܿܦܹ̈ܐ: ܕܐܸܬ݂ܟܲܢܲܫܘ ܒܢܝܼܩܝܼܵܐ ܡܕ݂ܝܼܢ݇ܬܵܐ ܕܒܹܝܬ݂ܘܿܢܝܼܵܐ: ܒܙܲܒ݂ܢܵܐ ܕܩܘܿܣܛܲܢܛܝܼܢܘܿܣ ܡܲܠܟܵܐ ܙܲܟܵܝܵܐ. ܘܥܸܠܬ݂ܵܐ ܕܹܝܢ ܕܲܟ݂ܢܘܼܫܝܼܗܘܿܢ ܡܸܛܠ ܪܘܼܫܥܹܗ ܕܐܲܪܝܼܘܿܣ܀ ܘܐܲܝܟ݂ ܥܝܵܕ݂ܵܐ ܕܹܝܢ ܕܪܵܕܹܐ ܒܲܝܢܵܬ݂ ܒܢܹܝ̈ܗ̇ ܕܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܲܫܠܝܼܚܵܝܬܵܐ ܕܝܼܠܲܢ ܕܡܲܕ݂ܢ̱ܚܵܐ: ܘܲܟ݂ܒܲܪ ܕܡ̣ܢ ܪܵܘܡܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܪܕ݂ܵܐ: ܘܡ̣ܢ ܡܲܫܠ̱ܡܵܢܘܼܬ݂ܵܐ ܘܝܘܼܒܵܠܵܐ ܕܥܲܬ݁ܝܼܩܹ̈ܐ ܡܓ݂ܵܘܝܲܝ̈ ܒܬܹܐܘܿܖ̈ܝܼܵܣ ܕܲܫܪܵܪܵܐ ܐܲܠܵܗܵܝܵܐ ܡܬ݂ܝܼܚ ܗ݇ܘ̣ܵ ܥܕܲܡܵܐ ܠܝܵܘ̈ܡܵܬܲܢ. ܘܗܵܟܲܢܵܐ ܡܫܲܪܹܝܢ ܚ݇ܢܲܢ ܠܡܸܬ݂ܢܵܐ ܩܵܢܘܿܢܵܐ ܕܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܢܝܼܩܝܼܵܐ.
ܒܲܫܪܵܪܵܐ ܓܹܝܪ ܕܠܵܐ ܦܘܼܠܵܓ݂ܵܐ: ܡܗܲܝܡܢܝܼܢܲܢ ܒܚܲܕ݂ ܐܲܠܵܗܵܐ: ܐܲܒ݂ܵܐ ܐܲܚܝܼܕ݂ ܟܠ. ܥܵܒ݂ܘܿܕ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܐܲܝܠܹܝܢ ܕܡܸܬ݂ܚܲܙ̈ܝܵܢ ܘܲܕ݂ܠܵܐ ܡܸܬ݂ܚܲܙ̈ܝܵܢ. ܘܲܒ݂ܚܲܕ݂ ܡܵܪܝܵܐ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ. ܒܪܹܗ ܕܐܲܠܵܗܵܐ ܝܼܚܝܼܕ݂ܵܝܵܐ: ܒܘܼܟ݂ܪܵܐ ܕܟ݂ܠܗܹܝܢ ܒܸܖ̈ܝܵܬ݂ܵܐ. ܗܵܘ̇ ܕܡ̣ܢ ܐܲܒ݂ܘܼܗܝ ܐܸܬ݂ܝܼܠܸܕ݂ ܩܕ݂ܵܡ ܟܠܗܘܿܢ ܥܵܠܡܹ̈ܐ: ܘܠܵܐ ܐܸܬ݂ܥ̱ܒܸܕ݂. ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܕܡ̣ܢ ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܒܲܪ ܟܝܵܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܕܒ݂ܐܝܼ̈ܕ݂ܵܘܗܝ ܐܸܬ݁ܬܲܩܲܢܘ ܥܵܠܡܹ̈ܐ: ܘܐܸܬ݂ܒ݁ܪܝܼ ܟܠ ܡܸܕ݁ܡ. ܗܵܘ̇ ܕܡܸܛܠܵܬܲܢ ܒܢܲܝ̈ܢܵܫܵܐ: ܘܡܸܛܠ ܦܘܼܪܩܵܢܲܢ: ܢ̣ܚܸܬ݂ ܡ̣ܢ ܫܡܲܝܵܐ: ܘܐܸܬ݂ܓܲܫܲܡ ܡ̣ܢ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܘܲܗܘ̤ܵܐ ܒܲܪܢܵܫܵܐ: ܘܐܸܬ݂ܒ݁ܛܸܢ ܘܐܸܬ݂ܝܼܠܸܕ݂ ܡ̣ܢ ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܘܚܲܫ̣ ܘܐܸܙܕ݁ܩܸܦ: ܒܝܵܘܡܲܝ̈ ܦܲܢܛܝܼܘܿܣ ܦܝܼܠܵܛܘܿܣ. ܘܐܸܬ݂ܩ̄ܒܲܪ: ܘܩ̣ܵܡ ܠܲܬ݂ܠܵܬ݂ܵܐ ܝܵܘܡܝܼ̈ܢ ܐܲܝܟ݂ ܕܲܟ݂ܬ݂ܝܼܒ݂. ܘܲܣ̣ܠܸܩ ܠܲܫܡܲܝܵܐ: ܘܝܼܬܸܒ݂ ܡ̣ܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܘܬ݂ܘܼܒ݂ ܥܬ݂ܝܼܕ݂ ܠܡܹܐܬ݂ܵܐ ܠܲܡܕ݂ܵܢ ܠܡܝܼܬܹ̈ܐ ܘܲܠܚܲܝܹ̈ܐ. ܘܲܒ݂ܚܲܕ݂ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܪܘܼܚܵܐ ܕܲܫܪܵܪܵܐ: ܗܵܘ̇ ܕܡ̣ܢ ܐܲܒ݂ܵܐ ܢܵܦܹ̇ܩ: ܪܘܼܚܵܐ ܡܲܚܝܵܢܵܐ. ܘܒܲܚܕ݂ܵܐ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܘܲܫܠܝܼܚܵܝܬܵܐ ܩܵܬ݂ܘܿܠܝܼܩܝܼ. ܘܡܵܘܕܹ݁ܝܢܲܢ ܒܲܚܕ݂ܵܐ ܡܲܥܡܘܿܕ݂ܝܼܬ݂ܵܐ: ܠܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ. ܘܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܲܒ݂ܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ: ܐܵܡܹܝܢ.
