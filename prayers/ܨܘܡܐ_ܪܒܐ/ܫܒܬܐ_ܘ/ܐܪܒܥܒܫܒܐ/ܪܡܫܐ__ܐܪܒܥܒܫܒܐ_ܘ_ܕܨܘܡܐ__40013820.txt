ܕܡܲܓܵܗܲܝ ܐܲܪܒܲܥܒ݁ܫܲܒܵܐ ܕܸܫܬ݂ܵܐ ܕܨܵܘܡܵܐ.
      ܕܲܐَܚܪ̈ܵܝܹܐ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܢܵܘܕܸ݁ܐ ܡܵܪܝ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܡܵܪܘܼܬ݂ܵܟ݂ ܘܫܘܼܒ݂ܚܵܐ ܐܲܡܝܼܢܵܐ ܕܠܵܐ ܫܲܠܘܵܐ ܢܲܣܸܩ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ: ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ. ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܗܘܼܠܵܠܵܐ. ܦܲܨܵܢܝ ܡܢ ܒܥܸܠܕܒ݂ܵܒܲܝ̈. ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܢܛ‐ܣܙ.
ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ. ܬܢܝܼ. ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ.
 ܘܡܲܩܦܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ. ܡܫܲܒܲܚܬ݁ ܡܵܪܲܢ ܘܠܵܟ݂ ܝܵܐܹܐ ܠܡܲܣܵܩܘܼ ܫܘܼܒ݂ܚܵܐ ܟܠܝܘܿܡ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ܀ ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ ܦܘܼܡܲܢ ܘܝܲܗ݇ܒ݂ܠܲܢ ܕܲܢܗܲܠܸܠ ܒܬܸܫܒ݁ܚܵܬܹ̈ܗ. ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ ܦܘܼܡܲܢ ܘܝܲܗ݇ܒ݂ܠܲܢ ܕܲܢܙܲܡܲܪ ܒܬܸܫܒ݁ܚܵܬܹ̈ܗ. ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ ܦܘܼܡܲܢ ܘܝܲܗ݇ܒ݂ܠܲܢ ܕܲܢܫܲܒܲܚ ܒܬܸܫܒ݁ܚܵܬܹ̈ܗ܀ ܠܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܫܘܼܒ݂ܚܵܐ ܢܲܣܸܩ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܬܢܝܼ ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܲܢܩܲܠܣܵܟ݂ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܲܢܫܲܒܚܵܟ݂ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܕܲܠܘܵܬ݂ ܡܵܝܘܿܬܹ̈ܐ ܡܵܪܵܐ ܡܪܲܚܡܵܢܵܐ. ܚܘܼܢܲܝܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ܀ ܕܠܵܐ ܙܵܟܹ݁ܐ ܩܕ݂ܵܡܲܝܟ ܟܠ ܕܚܲܝ ܒܲܥܒ݂ܝܼܕܹ̈ܐ. ܐܲܢ݇ܬ݁ ܡܵܪܝ ܐܲܦܢܝܼܬܵܢ ܡܢ ܟܠܵܗ̇ ܛܘܼܥܝܲܝ. ܕܐܲܢ݇ܬ݁ܘܼ ܐܲܠܵܗܵܐ. ܘܠܵܟ݂ ܝܵܐܹܐ ܫܘܼܒ݂ܚܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܘܥܲܠ ܐܲܦܲܝ̈ ܟܠܗܘܿܢ ܥܘܼܕ݂ܪ̈ܵܢܲܝܟ ܘܛܲܝ̈ܒ݁ܘܵܬ݂ܵܟ݂ ܕܲܠܘܵܬܲܢ ܕܠܵܐ ܡܸܬ݂ܦܲܪܥܝܼܢ: ܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ ܒܥܹܕ݁ܬܵܟ݂ ܡܟܲܠܲܠܬܵܐ ܡܲܠܝܲܬ݂ ܟܠ ܥܘܼܕ݂ܪ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܵܐ ܘܒ݂ܵܪܘܿܝܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܬܲܪ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
(ܓܝܼܘܿܪܵܐ ܕܲܒ݂ܥܹܕ݇ܬܵܐ.( ܚܕ݂ܝܼܬ݂ ܟܲܕ݂ ܐܵܡ̇ܪܝܼܢ ܗ݇ܘܵܘ ܠܝܼ: ܠܒܲܝܬܹ݁ܗ ܕܡܵܪܝܵܐ ܐܵܙ̇ܠ݇ܝܼܢܲܢ(ܘܲܠܒܲܪ ܡܢ ܥܹܕ݁ܬܵܐ ܐܵܡ̇ܪܝܼܢ) ܒܟ݂ܠܕܘܼܟ ܐܝܼܬܲܝܟ ܐܲܠܵܗܵܐܩܲܒܸ݁ܠ ܬܲܟ݂ܫܲܦ̮ܬܲܢ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ:ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ.ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܟ.ܐܲܢ݇ܬ݁ܘܼ ܡܵܪܝ ܫܲܪܝܼܪܵܐܝܼܬ݂ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܘܲܡܢܲܛܪܵܢܵܐ ܐܲܡܝܼܢܵܐ ܕܚܲܝܲܝ̈ܢ: ܘܠܵܟ݂ ܡܵܪܝ ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.

(ܘܪܵܡܹܐ ܡܫܲܡܫܵܢܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ ܐܘܼܪܵܪܵܐ: ܘܐܵܣܲܪ ܙܘܼܢܵܪܵܐ: ܘܩܵܐܹ̇ܡ ܩܕ݂ܵܡ ܫܟ݂ܝܼܢܬܵܐ ܡܛܲܟ݁ܣܵܐܝܼܬ݂ (ܘܐܵܡܲܪ :
ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ: ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܫܵܐ ܘܒܲܚܦܝܼܛܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ ܘܐܲܠܵܗܵܐ ܕܟ݂ܠܹܗ ܒܘܼܝܵܐܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܦܵܪܘܿܩܲܢ ܘܝܲܨܘܼܦܬ݂ܵܢܲܢ ܘܲܡܦܲܪܢܣܵܢܵܐ ܕܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ ܘܩܘܼܝܵܡܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ ܘܲܕ݂ܟ݂ܠܗܹܝܢ ܥܹ̈ܕ݂ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܐܲܬ݂ܪܲܢ ܘܥܲܠ ܟܠ ܐܲܬ݂ܪ̈ܵܘܵܢ: ܘܥܲܠ ܐܲܝܠܹܝܢ ܕܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܥܵܡ̇ܪܝܼܢ ܒܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܘܼܙܵܓ݂ܵܘ̈ܗܝ ܕܐܵܐܲܪ ܘܟܲܗܝܼܢܘܼܬ݂ܵܐ ܕܫܲܢ݇ܬܵܐ ܘܥܲܠ̈ܠܵܬ݂ܵܐ ܕܦܹܐܪܹ̈ܐ ܘܥܲܠ ܬܘܼܩܵܢܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܲܠܟܹ̈ܐ ܐܲܚܝܼܕܲܝ̈ ܫܘܼܠܛܵܢܹܗ ܕܥܵܠܡܵܐ ܗܵܢܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܩܲܫܝܼܫܹ̈ܐ ܘܲܡܫܲܡܫܵܢܹ̈ܐ ܐܲܪ̈ܬܵܕ݁ܘܿܟ݁ܣܹܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ܗ̇ܘ ܕܥܲܬ݁ܝܼܪ ܒܪ̈ܲܚܡܵܘܗܝ ܘܲܫܦܝܼܥ ܒܲܚܢܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܡܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ ܐܝܼܬ݂ܵܘܗܝ ܘܲܠܕ݂ܵܪܕܵܪ̈ܝܼܢ ܩܲܝܵܡ ܫܘܼܠܛܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܟ݂ܝܵܢܹܗ ܐܝܼܬ݂ܵܘܗܝ ܛܵܒ݂ܵܐ ܘܝܵܗܘܿܒ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹܐ ܒܡܵܘܬܹ݁ܗ ܕܚܲܛܵܝܵܐ ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܢ ܥܵܘܠܹܗ ܘܢܸܚܸܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܲܒ݂ܝܲܠܕܹ݁ܗ ܐܵܘ ܕܲܒ݂ܕܸܢܚܹܗ ܐܵܘ ܕܲܒ݂ܨܵܘܡܹܗ ܐܵܘ ܕܲܒ݂ܡܲܥܲܠܬܹ݁ܗ ܐܵܘ ܕܒܲܩܝܵܡܬܹ݁ܗ ܐܵܘ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܐܵܘ ܕܲܒ݂ܡܲܚܲܬ݁ܬܹ݁ܗ ܐܵܘ ܕܒܲܨܠܝܼܒܹܗ: ܩܲܕ݁ܝܼܫܵܐ (ܐܵܘ ܩܲܕܝܼܫܬܵܐ) ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܐܲܦܨܲܚ ܠܲܫܡܲܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܢܟ݂܀ ܗ̇ܘ ܕܠܹܗ ܗܘ ܟܝܵܢܵܐܝܼܬ݂ ܐܝܼܬ݂ ܠܵܐ ܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܘܝܲܬ݁ܝܼܪܵܐܝܼܬ݂ ܕܐܲܝܠܹܝܢ ܕܲܡܗܲܝܡܢܝܼܢ ܒܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܣܓܵܐ ܥܲܡܲܢ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀
 ܘܡܵܐ ܕܫܸܠܡܲܬ݀ ܡܲܩܸܦ: ܢܲܦ̮̈ܫܵܬܲܢ ܘܕܲܚܕ݂ܵܕܹ̈ܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܲܓ݂ܥܸܠ.
ܘܥܵܢܹܝ̇ܢ: ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ.
	ܘܲܡܨܲܠܸܐ ܟܵܗܢܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ. ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܡܲܓ݂ܥܠܝܼܢܲܢ ܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ: ܘܡܢܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܫܵܐ̇ܠܝܼܢܲܢ ܫܘܼܒ݂ܩܵܢܵܐ ܕܚܵܘܒܹ̈ܐ ܘܕܲܚܛܵܗܹ̈ܐ: ܗܲܒ݂ܠܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܪ̈ܲܚܡܲܝܟ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ ܟܵܪܘܿܙܘܼܬ݂ܵܐ: ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ.
ܘܐܵܡ̇ܪܝܼܢ ܒܩܵܠܵܐ ܦܫܝܼܛܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܫܲܒܲܚ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܥܵܠܲܡ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܒ݂ܵܬܲܪ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ ܒܓ݂ܘܼܕܵܐ ܕܒ݂ܵܬܲܪ. ܩܲܕ݁ܝܼܫܵܐ ܘܲܫܒ݂ܝܼܚܵܐ ܘܚܲܝܠܬ݂ܵܢܵܐ ܘܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܕܲܒ݂ܩܲܕ݁ܝܼܫܹ̈ܐ ܥܵܡܲܪ ܘܡܸܬ݁ܬ݁ܢܝܼܚ ܨܸܒ݂ܝܵܢܹܗ: ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ ܐܸܬ݂ܦܲܢ ܡܵܪܝ: ܘܚܘܼܣ ܘܪܲܚܸܡ ܥܠܲܝܢ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ. ܒܲܪܸܟ݂ ܡܵܪܝ: ܐܲܪܟܸ݁ܢܘ ܪܹ̈ܫܲܝܟ݁ܘܿܢ ܠܲܣܝܵܡܝܼܕ݂ܵܐ ܘܩܲܒܸ݁ܠܘ ܒܘܼܪܟ݁ܬ݂ܵܐ.
ܘܐܵܡܲܪ ܪܲܒܵܐ. ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܬܸܫܡܸܫܬܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
	ܘܡܲܩܸܦ ܨܠܘܿܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ. ܘܟܲܕ݂ ܫܵܠ̈ܡܵܢ ܢܲܦ̮̈ܫܵܬܲܢ ܠܲܚܕ݂ܵܐ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܓܡܝܼܪܬܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܟܠܲܢ ܒܲܚܕ݂ܵܐ ܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ: ܢܸܫܬ݁ܘܸܐ ܕܢܲܣܸܩ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ... ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
      ܥܘܿܢܝܼܬ݂ܵܐ ܕܪܲܡܫܵܐ. ܗܝ ܗ݇ܝܼ ܪܹܫܩܵܠܵܐ. ܐܹܪܲܡܪܡܵܟ݂ ܡܵܪܝ ܡܲܠܟܵܐ: ܕܝܼܠܵܟ݂ ܐܸܢܘܿܢ ܫܡܲܝܵܐ ܘܕ݂ܝܼܠܵܟ݂ ܗ݇ܝܼ ܐܲܪܥܵܐ. ܒܲܒ݂ܛܝܼܠܘܼܬܹܗ ܕܨܸܒ݂ܝܵܢܵܟ݂ ܝܵܨܹܦ ܐܲܢ݇ܬ݁ ܕܲܥܒ݂ܵܕܲܝ̈ܟ ܡܵܪܵܐ ܕܟ݂ܠܵܐ. ܬܪܹܝܢ ܓܹܝܪ ܢܲܗܝܼܪܹ̈ܐ ܒܪܲܝܬ݁ ܠܬܸܫܡܸܫܬܲܢ. ܫܸܡܫܵܐ ܓܹܝܪ ܠܒ݂ܘܼܣܵܡ ܒܸܪ̈ܝܵܬ݂ܵܐ ܒܪܲܝܬܵܝܗܝ. ܘܲܠܣܲܗܪܵܐ ܥܲܡ ܟܵܘܟ݁ܒܹ̈ܐ ܠܡܲܢܗܵܪܘܼ ܒܠܸܠܝܵܐ ܬܲܩܸܢܬ݁. ܘܟܲܕ݂ ܠܲܚܕ݂ܵܕܹ̈ܐ ܐܝܼܬܲܝܗܘܿܢ ܡܝܲܒ݁ܠܵܢܹ̈ܐ. ܠܦܘܼܩܕܵܢܵܐ ܕܡܵܪܘܼܬ݂ܵܟ݂ ܒܕܸܚܠܬ݂ܵܐ ܡܸܫܬܲܥܒ݁ܕܝܼܢ. ܐܲܢ݇ܬ݁ܘܼ ܕܟ݂ܠܗܹܝܢ ܠܥܘܼܕ݂ܪܵܢܵܐ ܕܐ݇ܢܵܫ̈ܵܐ. ܘܲܠܦܘܼܪܩܵܢܵܐ ܕܟ݂ܠܢܵܫ ܝܵܨܹܦ ܐܲܢ݇ܬ݁. ܪܲܒ݁ܘܼ ܓܲܙܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂: ܬܢܝܼ. ܘܫܲܒܲܚ. ܒܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܟ݂ܠܗܘܿܢ ܐܲܪܒܲܥܒ݁ܫܲܒܹ̈ܐ ܕܟ݂ܠܵܗ̇ ܫܲܢ݇ܬܵܐ. ܙܲܝܸܢ ܠܲܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܒܙܲܝܢܵܐ ܚܲܣܝܼܢܵܐ ܘܠܵܐ ܡܸܙܕܲܟ݂ܝܵܢܵܐ: ܒܲܨܠܵܘ̈ܵܬ݂ܵܗ̇ ܕܐܸܡܵܟ݂ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܲܥܒܸܕ݂ ܠܲܢ ܥܲܡܵܗ̇ ܡܢܵܬ݂ܵܐ ܘܫܵܘܬܵܦܘܼܬ݂ܵܐ ܒܲܓ݂ܢܘܿܢܵܟ݂ ܫܡܲܝܵܢܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ... ܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
	ܫܘܼܪܵܝܵܐ. ܘܢܸܣܒܲܬ݀ ܡܲܪܝܲܡ ܢܒ݂ܝܼܬ݂ܵܐ ܚܵܬܹܗ ܕܐܲܗܪܘܿܢ ܦܠܲܓܵܐ ܒܐܝܼܕ݂ܵܗ̇* ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܘܲܢܦܲܩܝ̈ ܟܠܗܹܝܢ ܢܸܫܹ̈ܐ ܒܵܬ݂ܪܵܗ̇ ܒܲܦ̈ܠܲܓܹ݁ܐ ܘܒܲܪ̈ܒ݂ܝܼܥܹܐ* ܘܲܡܥܲܢܝܵܐ ܗ݇ܘܵܬ݂ ܠܗܹܝܢ ܡܲܪܝܲܡ* ܫܲܒܲܚܹܝ̈ܢ ܠܡܵܪܝܵܐ ܓܲܐܝܵܐ* ܫܲܒܲܚ* ܥܵܠܲܡ*
	ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܬܲܪ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
      ܕܒܹܝܡ. ܪܵܥܝܵܐ ܛܵܒ݂ܵܐ ܕܟܲܢܸܫ ܥܵܢܹ̈ܗ. ܒܦܲܓ݂ܪܵܟ݂ ܘܲܕ݂ܡܵܟ݂ ܚܲܣܵܐ ܚܵܘܒܲܝ̈ܢ. ܒܵܬܹ̈ܐ. ܠܲܚܡܵܐ ܕܚܲܝܹ̈ܐ ܐܝܼܬ݂ܵܘܗܝ ܦܲܓ݂ܪܵܟ݂. ܕܟ݂ܠ ܕܐܵܟܹ݁ܠ ܠܹܗ ܙܵܟܹ݁ܐ ܠܡܵܘܬܵܐ.
      ܣܘܼܒܵܥܵܐ. ܛܘܼܒ݂ܵܘܗܝ ܠܡܢ ܕܚܵܐܲܪ. ܐ: ܗܘܼܠܵܠܵܐ. ܥܲܡ ܨܠܘܿܬܹܗ. ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܡܐ‐ܡܛ.
	ܘܐܵܡܪܝܼܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	 ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܕܣܘܼܒܵܥܵܐ ܒܓ݂ܘܼܕܵܐ ܕܒ݂ܵܬܲܪ. ܐܲܫܘܵܐ ܠܲܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܠܪܲܡܫܵܐ ܕܫܲܝܢܵܐ: ܘܲܠܠܸܠܝܵܐ ܕܲܢܝܵܚܵܐ: ܘܲܠܨܲܦܪܵܐ ܕܲܡܣܲܒ݁ܪ̈ܵܢ ܒܹܗ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܘܠܐܝܼܡܵܡܵܐ ܕܲܥܒ݂ܵܕܹ̈ܐ ܛܵܒܹ̈ܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܕܲܒ݂ܗܘܿܢ ܢܪܲܥܸܐ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܥܘܿܢ̈ܝܵܬ݂ܵܐ. ܒܲܫܠܝܼܚܹ̈ܐ ܟܹܐܦܵܐ. ܕܲܒ݂ܪܹܫ ܟܬ݂ܵܒܹ̈ܐ ܟܬ݂ܝܼܒ݂ ܥܠܲܝ: ܐܲܝܢ̇ܵܘ ܓܲܒ݂ܪܵܐ ܕܨܵܒܹ݁ܐ ܚܲܝܹ̈ܐ. ܗܵܐ ܒܲܟ݂ܬ݂ܵܒܹ̈ܐ ܩܲܕ݁ܝܼܫܹ̈ܐ. ܕܪܝܼܫܵܐ ܐܘܼܪܚܵܐ ܠܡܲܠܟ݁ܘܼܬ݂ܵܐ. ܘܕ݂ܝܼܠܲܢ ܦܵܗܹܐ ܪܸܥܝܵܢܲܢ. ܒܲܣܪ̈ܝܼܩܵܬ݂ܵܐ ܕܠܵܐ ܝܘܼܬ݂ܪܵܢ. ܠܵܐ ܓܹܝܪ ܡܲܗܢܸܐ ܠܲܢ ܥܘܼܬ݂ܪܹܗ. ܕܥܵܠܡܵܐ ܗܵܢܵܐ ܥܵܒ݂ܘܿܪܵܐ. ܐܸܠܵܐ ܨܵܘܡܵܐ ܘܲܨܠܘܿܬ݂ܵܐ. ܕܲܒ݂ܗܘܿܢ ܐܸܬ݂ܦܲܪܩܲܬ݀ ܢܝܼܢܘܹܐ: ܬܢܝܼ. ܐܲܩܸܦ. ܩܵܠܵܐ ܕܓ݂ܵܘܵܐ. ܡܸܛܠ ܕܗܘܝܘܼ ܡܵܪܹܟ݂ܝ ܣܓ݂ܘܿܕ݂ܝ ܠܹܗ. ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ. ܐܲܦܝܼܣܝ ܘܲܒ݂ܥܵܝ ܠܲܡܫܝܼܚܵܐ. ܕܢܸܥܒܸ݁ܕ݂ ܪ̈ܲܚܡܹܐ ܥܲܠ ܥܵܠܡܵܐ. ܕܐܲܚܝܼܕ݂ ܓܵܘܣܵܐ ܒܲܨܠܘܿܬܹܟ݂ܝ. ܘܥܹܕ݁ܬܵܐ ܬܸܚܕܸ݁ܐ ܒܥܲܕ݂ܥܹܕܹܟ݂ܝ. ܘܝܲܠܕܹܝ̈ܗ ܒܓ݂ܵܘܵܗ ܢܸܬ݂ܢܲܛܪܘܼܢ. ܡܢ ܢܸܟ݂ܝܵܢܹ̈ܐ ܘܣܲܩܘܼܒ݂ܠܹ̈ܐ. ܕܐܵܟܹ݁ܠܩܲܪܨܵܐ ܡܵܪܘܿܕ݂ܵܐ. ܒܲܕܲܪ ܘܝܲܗ݇ܒ݂ ܠܒ݂ܵܝܫܹ̈ܐ. ܦܬܲܚ ܓܲܙܵܐ ܕܝܲܗ݇ܒ݂ ܠܵܟ݂ ܡܵܪܵܟ݂. ܐܘܿ ܝܘܿܚܲܢܲܢ ܟܵܪܘܿܙܵܐ. ܘܗܲܒ݂ ܥܘܼܕ݂ܪܵܢܵܐ ܠܲܣܢܝܼܩܹ̈ܐ. ܕܒ݂ܵܟ݂ ܡܸܬ݂ܓܵܘܣܝܼܢ ܘܩ̇ܵܪܹܝܢ ܫܡܵܟ݂. ܠܕܲܟ݂ܪܝܼܗ ܗܲܒ݂ܠܹܗ ܚܘܼܠܡܵܢܵܐ. ܠܕܲܡܥܵܩ ܗܘܝܼ ܠܹܗ ܠܘܼܒܵܒ݂ܵܐ. ܘܐܲܝܢܵܐ ܕܒ݂ܝܼܫܵܐ ܫܵܚܹܩ ܠܹܗ. ܨܠܘܿܬ݂ܵܟ݂ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܹܗ. ܒܟ݂ܠܵܗ ܐܲܪܥܵܐ ܢܸܦܩܲܬ݀ ܣܒܲܪܬ݂ܗܘܿܢ. ܫܠܝܼܚܹ̈ܐ ܓܒܲܝ̈ܵܐ ܘܩܲܕ݁ܝܼܫܹ̈ܐ. ܟܵܪ̈ܘܿܙܹܐ ܕܗܲܝܡܵܢܘܼܬ݂ܵܐ. ܦܲܛܪܘܿܣ ܘܦܵܘܠܘܿܣ ܕܒܲܫܪܵܪܗܘܿܢ. ܐܲܢܗܲܪܘ ܠܥܵܠܡܵܐ ܡܢ ܣܵܟܹܗ. ܘܲܥܩܲܪܘ ܛܘܼܥܝܲܝ ܕܲܦܬܲܟ݂ܪܹ̈ܐ. ܘܪܲܡܪܸܡܘ ܠܥܹܕ݁ܬܵܐ ܒܝܘܼܠܦܵܢܗܘܿܢ. ܒܲܨܠܵܘ̈ܵܬ݂ܗܘܿܢ ܢܸܗܘܘܿܢ ܥܠܲܝܢ. ܪ̈ܲܚܡܹܐ ܠܥܹܕ݁ܬܵܐ ܕܲܩܝܵܡܬܵܐ. ܡܢ ܣܵܘܦܹܝ̈ܗ ܠܣܵܘܦܹܝ̈ܗ ܕܐܲܪܥܵܐ. ܫܠܝܼܚܹ̈ܐ ܕܐܲܟ݂ܪܸܙܘ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܩܢܘܿܡܹ̈ܐ ܬܠܵܬ݂ܵܐ ܕܐܝܼܬ݂ܘܼܬ݂ܵܐ. ܒܲܨܠܵܘܵܬ݂̈ܟ݂ܘܿܢ ܢܸܬ݂ܓܲܢ݇ܒܲܪ. ܘܲܢܕܘܼܫܝܼܘܗܝ ܠܒ݂ܝܼܫܵܐ ܣܵܢܐܲܢ. ܘܢܲܡܠܸܟ݂ ܫܲܝܢܵܐ ܘܲܫܠܵܡܵܐ. ܒܐܲܪ̈ܒܲܥܦܸܢܝܵܬܹܗ ܕܥܵܠܡܵܐ. ܘܥܹܕ݁ܬܵܐ ܬܸܙܡܲܪ ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܩܘܼܕ݂ܫܵܐ ܕܲܒܲܚܘ ܠܹܗ ܕܸܒ݂ܚܹ̈ܐ ܕܫܘܼܒ݂ܚܵܐ. ܦܲܓ݂ܪܹܗ ܘܢܲܦ̮ܫܹܗ ܙܲܗܝܵܐܝܼܬ݂. ܩܲܪܸܒ݂ ܓܲܒ݂ܝܵܐ ܐܸܣܛܲܦܵܢܘܿܣ. ܕܲܚܙܵܐ ܠܡܵܪܹܗ ܕܲܦܟ݂ܝܼܪ ܗ݇ܘܵܐ. ܩܕ݂ܵܡ ܕܲܝ̇ܵܢܵܐ ܕܲܢܢܲܓ݁ܕ݂ܝܼܘܗܝ. ܘܩܲܪܸܒ݂ ܐܵܦ ܗܘ ܕܸܒ݂ܚܬ݂ܵܐ. ܦܲܓ݂ܪܹܗ ܠܡܵܘܬ݂ܵܐ ܕܲܪܓ݂ܘܼܡܝܵܐ. ܘܲܢܣܲܒ݂ ܟܠܝܼܠܵܐ ܕܙܵܟ݂ܘܼܬ݂ܵܐ. ܘܲܗܘܵܐ ܒܘܼܟ݂ܪܵܐ ܕܣܵܗܕܘܼܬ݂ܵܐ. ܘܲܢܫܲܒ݁ܚܘܼܢ ܠܲܫܡܵܟ݂ ܠܥܵܠܡܝܼܢ. ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܦܵܪܘܿܩܲܢ. ܕܲܒ݂ܝܲܕ݂ ܚܲܝܠܵܟ݂ ܐܸܬ݂ܢܲܨܲܚܘ. ܓܲܒ݂ܪܹ̈ܐ ܬܠܵܬ݂ܵܐ ܕܬܸܕ݂ܡܘܼܪܬܵܐ. ܢܵܘܣܹ̈ܐ ܕܟܲܝ̈ܵܐ ܕܪܘܼܚ ܩܘܼܕ݂ܫܵܐ. ܥܡܝܼܠ ܒܲܕ݂ܪܵܫܵܐ ܕܝܼܘܿܕܘܿܪܘܿܣ. ܘܬܹܐܵܕܘܿܪܘܿܣ ܢܲܨܝܼܚܵܐ. ܘܦܵܥܠܵܐ ܙܪܝܼܙܵܐ ܢܸܣܛܘܿܪܝܼܣ. ܨܠܘܿܬ݂ܗܘܿܢ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܲܢ. ܢܸܬ݂ܩܲܝܲܡ ܠܥܵܠܲܡ ܩܕ݂ܵܡ ܐܲܠܵܗܵܐ. ܕܘܼܟ݂ܪܵܢ ܟܵܗܢܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ. ܕܐܲܢܗܲܪܘ ܠܥܵܠܡܵܐ ܒܝܘܼܠܦܵܢܗܘܿܢ. ܢܸܗܘܸܐ ܒܥܹܕ݁ܬܵܐ ܩܕ݂ܵܡ ܡܵܪܲܢ. ܕܪܲܚܡܘܼܗܝ ܘܲܢܛܲܪܘ ܦܘܼܩܕܵܢܵܘܗܝ. ܒܲܨܠܵܘ̈ܵܬ݂ܗܘܿܢ ܬܸܬ݂ܢܛܲܪ. ܥܹܕ݁ܬܵܐ ܘܝܲܠܕܹܝ̈ܗ ܡܢ ܒܝܼܫܵܐ. ܘܟܸܢܫܵܐ ܕܙܲܝܲܚ ܕܘܼܟ݂ܪܵܢܗܘܿܢ. ܒܝܲܕ݂ ܒܵܥܘܼܬ݂ܗܘܿܢ ܢܸܣܬܲܬܲܪ. ܢܸܬ݁ܕܟܲܪ ܠܵܟ݂ ܡܵܪܝܵܐ ܟܠܗܘܿܢ ܩܘܼܪ̈ܒܵܢܲܝܟ. ܐܲܒ݂ܘܼܢ ܕܲܢܨܲܚܬ݁ ܒܐܵܓ݂ܘܿܢܵܐ. ܗܵܐ ܒܲܫܡܲܝܵܐ ܦܘܼܪܥܵܢܵܟ݂. ܡܫܝܼܚܵܐ ܕܨܲܒܸ݁ܬ݁ܬ݁ ܠܹܗ ܩܢܘܿܡܵܟ݂. ܒܥܹܕܬܹ݁ܗ ܐܵܘܪܸܒ݂ ܕܘܼܟ݂ܪܵܢܵܟ݂. ܦܝܼܪܡܵܐ ܕܲܟ݂ܝܵܐ ܚܘܼܒܵܟ݂ ܗܘܵܐ. ܘܪܲܥܝܼܬ݁ ܠܡܵܪܵܟ݂ ܒܝܲܕ݂ ܥܲܡܠܲܝ̈ܟ. ܐܲܦܝܼܣ ܥܲܡܲܢ ܕܡܵܐ ܕܡܸܬ݂ܓ݁ܠܸܐ. ܒܫܘܼܒ݂ܚܹܗ ܪܲܒܵܐ ܢܪܲܚܸܡܥܠܲܝܢ. ܚܲܠܸܠܲܝܢܝ ܒܹܗ ܘܡܢ ܬܲܠܓܵܐ ܐܸܚܘܲܪ. ܥܲܢܝܼܕܹ̈ܐ ܕܠܲܒ݂ܫܘܼܟ݂ ܡܵܪܲܢ. ܡܢ ܡܲܝ̈ܵܐ ܕܡܲܥܡܘܿܕܝܼܬ݂ܵܐ. ܒܵܟ݂ ܢܸܬ݂ܚܵܘܪܘܼܢ ܦܲܓ݂ܪ̈ܲܝܗܘܿܢ. ܡܢ ܟܘܼ̈ܬ݂ܡܵܬ݂ܵܐ ܕܲܚܛܝܼܬ݂ܵܐ. ܥܲܢܝܼܕܹ̈ܐ ܕܐܸܟܲܠܘ ܦܲܓ݂ܪܵܟ݂. ܘܒܲܕ݂ܡܵܟ݂ ܚܲܝܵܐ ܐܸܬ݂ܒܲܣܲܡܘ ܥܒܸܕ݂ ܡܵܪܲܢ ܕܘܼܟ݂ܪ̈ܵܢܲܝܗܘܿܢ. ܒܐܲܬ݂ܪܵܐ ܕܟܹܐܢܹ̈ܐ ܡܕܲܝܪܝܼܢ ܒܹܗ. ܘܐܸܡܲܪ ܕܬ݂ܘܼܒ݂ܘ ܒܢܲܝ̈ܢܵܫܵܐ. ܒܟ݂ܪܘܼܙܘܼܬܹܗ ܕܲܢܒ݂ܝܼܵܐ. ܒܨܵܘܡܵܐ ܐܸܬ݂ܦܲܪܩܲܬ݀ ܢܝܼܢܘܹܐ. ܘܲܒ݂ܒ݂ܵܥܘܼܬ݂ܵܐ ܘܒܲܨܠܘܿܬ݂ܵܐ. ܡܲܠܲܐܟ݂ ܡܵܘܬܵܐ ܡܸܢܵܗ ܟܠܵܐ. ܒܚܲܫܵܐ ܘܕܸܡܥܹ̈ܐ ܕܲܬ݂ܝܵܒ݂ܘܼܬ݂ܵܐ. ܢܹܐܚܘܿܕ݂ ܓܵܘܣܵܐ ܒܓ݂ܵܘ ܗܲܝܟ݁ܠܵܐ. ܘܢܸܩܪܸܐ ܠܪ̈ܲܚܡܹܐ ܠܥܘܼܕ݂ܪܵܢܲܢ. ܥܲܕ݂ܠܵܐ ܢܸܚܫܲܟ݂ ܢܲܗܝܼܪܵܐ. ܗܲܒ݂ܘ ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ. ܐܘܿܫܲܥܢܵܐ ܠܲܒ݂ܪܹܗ ܕܕ݂ܵܘܝܼܕ݂. ܝܲܒܸ݁ܒ݂ܘ ܫܲܒ݂ܪܹ̈ܐ ܟܲܕ݂ ܐܵܡܪܝܼܢ. ܕܲܩܪܵܐ ܠܡܝܼܬ݂ܵܐ ܡܢ ܩܲܒ݂ܪܵܐ. ܘܐܲܩܝܼܡܹܗ ܒܚܲܝܠܵܐ ܕܡܸܠܬܹܗ. ܘܲܫܩܲܠܘ ܣܵܘܟܹ̈ܐ ܕܐܝܼܠܵܢܹ̈ܐ. ܘܲܡܙܲܝܚܝܼܢ ܗ݇ܘܵܘ ܠܐܝܼܩܵܪܹܗ. ܒܪܝܼܟ݂ ܕܐܸܬ݀ܵܐ ܒܲܫܡܹܗ ܕܡܵܪܝܵܐ. ܕܢܸܦܪܘܿܩ ܠܓܸܢܣܲܢ ܡܢ ܛܘܼܥܝܲܝ. ܐܵܟܹ݁ܠ ܠܲܚܡܝ ܕܲܬ݂ܟ݂ܝܼܠ ܐ݇ܢܵܐ ܥܠܵܘܗܝ ܣܲܓ݁ܝܼ ܐܸܬ݂ܢܟܸܠ ܥܠܲܝ. ܩܨܵܐ ܗ݇ܘܵܐ ܠܲܚܡܵܐ ܦܵܪܘܿܩܲܢ. ܘܝܲܗ݇ܒ݂ ܗ݇ܘܵܐ ܠܹܗ ܠܬܲܠܡܝܼܕ݂ ܙܹܐܦܵܐ. ܘܐܸܬ݂ܥܲܠܲܠ ܒܹܗ ܣܵܛܵܢܵܐ. ܘܲܗܘܵܐ ܡܵܐܢܵܐ ܕܠܵܐ ܚܲܫܚܘܼ. ܘܲܕ݂ܠܵܐ ܫܲܚܠܸܦ ܪܸܥܝܵܢܹܗ. ܘܐܲܪܚܸܩ ܢܸܟ݂ܠܵܐ ܡܢ ܓܵܘܹܗ. ܗܘܵܐ ܢܘܼܟ݂ܪܵܝܵܐ ܠܲܓ݂ܒ݂ܝܼܬܹܗ. ܘܠܵܐ ܐܸܬ݂ܥܲܕܲܪ ܡܢ ܒܘܼܪܟ݁ܬܹܗ. ܚܵܘܝܼ ܡܵܪܝܵܐ ܦܘܼܪܩܵܢܹܗ. ܨܵܪ ܗ݇ܘܵܐ ܡܵܪܲܢ ܒܲܩܢܘܿܡܹܗ. ܘܚܵܘܝܼ ܚܲܫܵܐ ܕܲܟ݂ܡܵܐ ܩܫܸܐ. ܘܲܫܗܲܪ ܘܲܥܡܲܠ ܕܐܸܢ ܡܸܫܟ݁ܚܵܐ. ܬܸܥܒܲܪ ܡܸܢܹܗ ܗ̇ܝ ܫܵܥܬ݂ܵܐ. ܐܸܬ݁ܬ݁ܥܝܼܪܘ ܘܨܲܠܵܘ ܙܲܗܲܪ ܠܲܢ. ܕܲܠܢܸܣܝܘܿܢܵܐ ܠܵܐ ܬܸܥܠܘܼܢ. ܕܲܩܫܸܐ ܚܲܫܵܐ ܡܵܪܲܢ ܣܗܸܕ݂. ܕܕܘܼܥܬܹܗ ܠܲܕ݂ܡܵܐ ܐܸܫܬܲܚܠܦܲܬ݀. ܗܝ ܗܘܵܬ݂ ܠܪܹܫܵܐ ܕܒܸܢܝܵܢܵܐ. ܩܝܵܡܬܵܟ݂ ܡܵܪܲܢ ܚܲܝܹ̈ܐ ܗܘܵܬ݂. ܠܓܸܢܣܵܐ ܕܐ݇ܢܵܫܵܐ ܕܐܲܒ݁ܝܼܕ݂ ܗ݇ܘܵܐ. ܕܩܵܡܘ ܘܐܸܬ݂ܢܲܚܲܡܘ ܘܚܲܝܹ̈ܐ ܩܢܵܘ. ܒܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ. ܘܲܫܠܲܚܘ ܘܲܫܕ݂ܵܘ ܚܲܢܦܘܼܬ݂ܵܐ. ܘܛܘܼܥܝܲܝ ܟܠܵܗ̇ ܕܲܦܬܲܟ݂ܪܹ̈ܐ. ܘܲܒ݂ܪܲܟ݂ܘ ܘܲܣܓܸܕ݂ܘ ܠܚܲܕ݂ ܐܝܼܬ݂ܝܵܐ. ܕܲܒܲܒ݂ܪܹܗ ܚܲܝܵܐ ܦܪܲܩ ܥܵܠܡܹ̈ܐ. ܘܠܵܐ ܚܣܲܟ݂ܘ ܡܢ ܡܵܘܬܵܐ ܢܲܦ̮ܫܵܬ݂̈ܗܘܿܢ. ܩܛܝܼܠܹ̈ܐ ܕܲܪܚܸܡܘ ܒܵܪܘܿܝܗܘܿܢ. ܘܩܲܒܸ݁ܠܘ ܡܵܘܬܵܐ ܒܨܸܒ݂ܝܵܢܗܘܿܢ. ܘܲܗܘܵܘ ܕܸܒ݂ܚܹ̈ܐ ܡܪ̈ܲܥܝܵܢܹܐ. ܠܡܲܠܟܵܐ ܡܫܝܼܚܵܐ ܡܟܲܠܠܵܢܗܘܿܢ. ܩܲܪܸܒ݂ܘ ܥܲܡܲܢ ܒܵܥܘܼܬ݂ܵܐ. ܒܝܵܘܡܵܐ ܪܲܒܵܐ ܕܒ݂ܘܼܚܪܵܢܵܐ. ܕܢܸܫܬܵܘܙܲܒ݂ ܡܢ ܬܲܫܢܝܼܩܵܐ. ܘܢܹܐܪܲܬ݂ ܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ. ܐܲܪܡܵܐ ܣܲܝܦܵܐ ܒܚܲܨܲܝ̈ܟ ܓܲܢ݇ܒܵܪܵܐ. ܓܲܢ݇ܒܵܪ ܚܲܝܠܵܐ ܡܵܪܝ ܓܝܼܘܲܪܓܝܼܣ. ܕܫܵܛ ܗ݇ܘܵܐ ܠܡܵܘܬܵܐ ܘܲܠܣܲܝܦܵܐ. ܘܲܠܦܘܼܣܵܩܹ̈ܐ ܡܲܪܝܼܪܹ̈ܐ. ܘܲܠܢܸܓ݂ܕܹ̈ܐ ܕܟ݂ܠ ܐܸܣܟܹ݁ܝܡܝܼܢ. ܘܲܣܥܲܪ ܚܲܝܠܹ̈ܐ ܘܬܸܕ݂ܡܪ̈ܵܬ݂ܵܐ. ܘܐܲܦ̮ܢܝܼ ܠܟ݂ܠܢܵܫ ܠܲܫܪܵܪܵܐ. ܒܪܝܼܟ݂ܘܼ ܕܢܲܨܚܹܗ ܠܐܲܬ݂ܠܹܝܛܵܐ. ܕܲܒ݂ܚܲܝܠܹܗ ܢܸܙܟܸ݁ܐ ܠܛܘܼܥܝܲܝ. ܠܡܸܠܬܹܗ ܕܐܲܠܵܗܵܐ ܐܹܫܲܒܲܚ. ܡܸܠܬܹܗ ܕܐܲܒ݂ܵܐ ܒܚܘܼܒܹܗ ܨܒ݂ܵܐ. ܕܢܸܦܪܘܿܩ ܠܓܸܢܣܲܢ ܕܐܲܒܝܼܕ݂ ܗ݇ܘܵܐ. ܕܡܘܼܬ݂ܵܐ ܕܥܲܒ݂ܕܵܐ ܡܸܢܲܢ ܫܩܲܠ. ܘܡܢ ܝܲܡܝܼܢܵܐ ܣܠܸܩ ܐܵܘܬ݁ܒܹܗ. ܫܲܒܲܚܘ ܡܸܟܹ݁ܝܠ ܡܵܝܘܿܬܹ̈ܐ. ܕܗܵܐ ܒܲܪ ܓܸܢܣܲܢ ܒܲܫܡܲܝܵܐ. ܟܲܕ݂ ܡܸܙܕܲܝܲܚ ܡܢ ܥܸܠܵܝܹ̈ܐ. ܟܪ̈ܘܿܒܹܐ ܣܪ̈ܵܦܹܐ ܘܡܲܠܲܐܟܹ̈ܐ. ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܬܸܫܒ݁ܘܿܚܬܵܐ ܚܕܲܬܵܐ. ܚܕܲܬܵܐ ܗ݇ܝܼ ܬܸܕܡܘܼܪܬܵܐ ܕܲܣܥܲܪܘ. ܫܠܝܼܚܹ̈ܐ ܒܗܲܝܟ݁ܠܵܐ ܕܐܘܿܪܸܫܠܸܡ. ܕܠܲܚܓ݂ܝܼܪܵܐ ܕܡܢ ܟܪܸܣ ܐܸܡܹܗ. ܒܲܫܡܹܗ ܕܝܼܫܘܿܥ ܗܠܲܟ݂ܬ݂ܵܐ ܝܲܗ݇ܒ݂ܘ. ܘܲܬ݂ܗܲܪܘ ܟܸܢܫܹ̈ܐ ܕܙܵܩܘܿܦܹ̈ܐ. ܕܲܚܙܵܘ ܪܲܒ݁ܘܼܬ݂ ܬܸܕ݂ܡܘܼܪܬܵܐ. ܕܒܲܫܡܹܗ ܕܝܼܫܘܿܥ ܕܲܙܩܲܦ̮ܘ ܗ݇ܘܵܘ. ܚܓ݂ܝܼܪܵܐ ܗܠܲܟ݂ܬ݂ܵܐ ܩܲܒܸ݁ܠ ܗ݇ܘܵܐ. ܚܵܘܝܼ ܡܵܪܝܵܐ ܦܘܼܪܩܵܢܹܗ. ܨܠܝܼܒ݂ܵܐ ܕܢܘܼܗܪܵܐ ܕܐܸܬ݂ܚܵܘܝܼ. ܠܩܘܼܣܛܲܢܛܝܼܢܘܿܣ ܒܲܫܡܲܝܵܐ. ܐܲܝܟ݂ ܪܲܒ݁ ܚܲܝܠܵܐ ܐܵܙܹܠ ܗ݇ܘܵܐ. ܒܪܹܫ ܡܲܫܪܝܼܬܹܗ ܠܲܩܪܵܒ݂ܵܐ. ܘܙܵܥ ܘܐܸܬ݂ܒܲܠܗܝܼ ܓܘܼܕܲܝ̈ܗܘܿܢ. ܕܚܲܢܦܹ̈ܐ ܣܵܓ݂ܕܲܝ̈ ܠܒܸܪ̈ܝܵܬ݂ܵܐ. ܘܐܲܦܝܼܘ ܛܘܼܥܝܲܝ ܕܚܲܢܦܘܼܬ݂ܵܐ. ܘܲܣܓܸܕ݂ܘ ܠܲܨܠܝܼܒ݂ܵܐ ܘܝܲܩܪܘܼܗܝ. ܒܲܪ݉ܬ݂ ܡܲܠܟܵܐ ܒܫܘܼܒ݂ܚܵܐ ܩܵܡܲܬ݂. ܥܹܕ݁ܬܵܐ ܡܟ݂ܝܼܪܬܹ݁ܗ ܕܲܡܫܝܼܚܵܐ. ܕܦܲܪܩܵܗ̇ ܒܲܕܡܹܗ ܡܢ ܛܘܼܥܝܲܝ: ܘܐܸܫܬܵܘܕ݁ܝܼ ܠܵܗ̇ ܒܲܩܝܵܡܬܹ݁ܗ. ܚܲܝܹ̈ܐ ܘܛܘܼܒܹ̈ܐ ܕܠܵܐ ܡܸܫܬ݁ܪܝܼܢ. ܨܲܒܸ݁ܬ݂ܝ ܢܲܦ̮ܫܹܟ݂ܝ ܒܙܲܗܝܘܼܬ݂ܵܐ. ܘܲܦܪܘܿܥܝ ܩܘܼܒܵܠܛܲܝܒ݁ܘܼܬ݂ܵܐ. ܬܵܘܕ݁ܝܼܬ݂ܵܐ ܕܠܵܐ ܦܘܼܠܵܓ݂ܵܐ. ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܫܲܪܝܼܪܬܵܐ.
ܕܲܨܠܘܿܬ݂ܵܐ. ܡܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂. ܩܵܠܵܐ: ܝܗ. ܒܪܘܼܟܵܒ݂ ܡܵܪܵܐ ܕܟ݂ܠܵܐ.
	ܡܢ ܒܲܣܝܼܡܘܼܬ݂ܵܐ ܕܙܵܟ݂ܘܼܬ݂ܵܐ ܕܝܲܡܝܼܢܵܟ݂. ܡܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܡܵܪܵܐ ܕܟ݂ܠ. ܫܹܐ̈ܠܵܬ݂ ܥܲܒ݂ܕܲܝ̈ܟ ܢܸܬ݂ܦܲܢ̈ܝܵܢ. ܕܐܲܢ݇ܬ݁ܘܼ ܐܝܼܬܲܝܟ݂ ܐܝܼܵܠܗܘܿܢ. ܘܲܥܠܲܝܟ݁ܘܼ ܣܝܼܡ ܬܘܼܟ݂ܠܵܢܗܘܿܢ. ܢܲܛܲܪ ܐܸܢܘܿܢ ܒܲܚܢܵܢܵܟ݂. ܘܣܲܝܲܥ ܐܸܢܘܿܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܘܗܲܒ݂ ܠܗܘܿܢ ܐܝܼܕ݂ܵܐ ܕܥܘܼܕ݂ܪ̈ܵܢܹܐ. ܕܠܵܟ݂ ܡܵܪܝ ܬܸܫܒ݁ܘܿܚܬܵܐ ܢܲܣܩܘܼܢ. ܐܲܝܟ݂ ܥܲܝܢܹ̈ܐ ܕܥܲܒ݂ܕܹ̈ܐ ܠܘܵܬ݂ ܡܵܪ̈ܲܝܗܘܿܢ. ܠܘܵܬ݂ܵܟ݂ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܗܵܐ ܬܠܸܐ ܚܝܵܪܵܐ ܕܲܡܥܵܩܹ̈ܐ. ܐܲܣܵܐ ܟܹܐܒܹ̈ܐ ܘܟ݂ܘܼܪ̈ܗܵܢܹܐ. ܕܐܲܝܠܹܝܢ ܕܲܪܡܹܝܢ ܒܢܸܣܝܘܿܢܹ̈ܐ. ܐܲܝܟ݂ ܕܐܸܫܬܵܘܕܝܼܬ݁ ܒܲܣܒܲܪܬ݂ܵܟ݂. ܡܲܠܝܲܬ݂ ܚܲܝܹ̈ܐ ܘܒ݂ܘܼܣܵܡܹ̈ܐ. ܕܐܲܝܢܵܐ ܕܐܲܡܝܼܢ ܘܢܵܩܹܫ. ܒܲܬ݂ܪܲܥ ܡܵܪܘܼܬ݂ܝ ܡܸܬ݂ܦܬܲܚ ܠܹܗ. ܡܪܲܚܡܵܢܲܬ݁ ܡܵܪܝܵܐ ܘܙܲܕܝܼܩ. ܚܲܢܵܢܵܐ ܐܲܢ݇ܬ݁ ܡܢ ܥܵܠܲܡ. ܘܲܡܪܲܚܡܵܢܲܬ݁ ܠܕ݂ܵܪܕܵܪ̈ܝܼܢ. ܡܵܢܵܘ ܥܵܘܠܵܗ̇ ܕܲܒ݂ܪܝܼܬ݂ܵܐ. ܠܪ̈ܲܚܡܹܐ ܫܦܝܼܥܹ̈ܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂.   ܪܘܿܣ ܠܹܗ ܒܐܲܦܵܘ̈ܗܝ ܕܲܟ݂ܝܵܢܲܢ. ܛܲܠܵܐ ܕܪ̈ܲܚܡܹܐ ܘܚܵܘܣܵܢܵܐ. ܘܫܵܘܙܸܒܲܝܢ ܡܢ ܝܲܕ݂ ܒܝܼܫܵܐ. ܘܡܢ ܙܝܼܙܵܢܹ̈ܐ ܒܢܲܝ̈ ܛܘܼܥܝܲܝ. ܐܲܠܵܗܵܐ ܡܲܠܸܠ ܒܩܘܼܕ݂ܫܹܗ. ܐܲܢ݇ܬ݁ܘܼ ܕܡܲܠܸܠܬ݁ ܒܲܢܒ݂ܝܼܹ̈ܐ. ܡܵܪܹܐ ܢܒ݂ܝܼܹ̈ܐ ܘܲܫܠܝܼܚܹ̈ܐ. ܕܠܵܐ ܨܵܒܹ݁ܐ ܐ݇ܢܵܐ ܒܡܵܘܬܲܝܟ̈ܘܿܢ. ܚܲܛܵܝܹ̈ܐ ܦܢܵܘ ܡܢ ܥܵܘܠܟ݂ܘܿܢ. ܦܲܢܵܐ ܘܗܲܒ݂ܠܲܢ ܫܸܐ̈ܠܵܬܲܢ. ܚܘܼܠܡܵܢ ܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ. ܙܲܒ݂ܢܹ̈ܐ ܛܥܝܼܢܲܝ̈ ܚܲܕ݂̈ܘܵܬ݂ܵܐ. ܡܦܲܠܓܲܝ̈ ܕܵܫܢܹ̈ܐ ܘܡܵܘܗ̈ܒ݂ܵܬ݂ܵܐ. ܐܲܠܵܗܵܐ ܒܲܫܡܵܟ݂ ܦܪܘܿܩܲܝܢܝ. ܡܲܠܟܵܐ ܕܡܲܠܟܹ̈ܐ ܐܝܼܵܠܲܢ. ܐܲܢ݇ܬ݁ܘܼ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܚܘܼܣ ܥܲܠ ܥܲܒ݂ܕܲܝ̈ܟ ܕܩܵܪܹܝܢ ܠܵܟ݂. ܒܗܵܠܹܝܢ ܙܲܒ݂ܢܹ̈ܐ ܕܐܘܼܠܨܵܢܵܐ. ܕܗܵܐ ܚܕ݂ܝܼܪܝܼܢ ܠܲܢ ܕܵܘܘܿܢܹ̈ܐ. ܘܣܘܼܪ̈ܵܕܹܐ ܡܢ ܟܠ ܓܲܒܝܼ̈ܢ. ܒܲܥܓܲܠ ܢܩܲܕܡܘܼܢܵܢ ܪ̈ܲܚܡܲܝܟ. ܘܐܲܢܗܲܪ ܐܲܦܲܝ̈ܟ ܘܲܦܪܘܿܩܲܝܢ. ܒܲܫܡܲܝܵܐ ܘܐܲܪܥܵܐ. ܡܵܪܲܢ ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܘܢܸܗܘܸܐ ܒܐܲܪܥܵܐ ܨܸܒ݂ܝܵܢܵܟ݂. ܐܲܟ݂ܡܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܒܲܫܡܲܝܵܐ. ܨܹܝܕ݂ ܪ̈ܘܼܚܵܢܹܐ ܕܠܵܟ݂ ܡܫܲܡܫܝܼܢ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ. ܘܠܵܐ ܬܲܥܸܠܲܝܢ ܠܢܸܣܝܘܿܢܵܐ. ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ. ܕܕ݂ܝܼܠܵܟ݂ ܐܝܼܬܹܝܗ̇ ܡܲܠܟ݁ܘܼܬ݂ܵܐ. ܪܲܒ݁ܘܼ ܓܹܝܪ ܚܲܝܠܵܐ ܕܲܨܠܘܿܬ݂ܵܐ ܐܲܝܕܵܐ ܕܙܲܕܝܼܩܵܐ ܡܨܲܠܸܐ ܠܵܗ̇. ܪܲܒ݁ܘܼ ܚܲܝܠܵܗ̇ ܕܲܨܠܘܿܬ݂ܵܐ. ܕܚܘܼܒܵܐ ܘܫܲܝܢܵܐ ܢܵܩܹܦ ܠܵܗ̇. ܘܥܵܛܝܵܐ ܣܘܿܓ݂ܐܵܐ ܕܲܚܛܵܗܹ̈ܐ. ܡܩܲܪܒ݂ܵܐ ܠܝܼܕܲܥܬܹܗ ܕܡܵܪܝܵܐ. ܘܝܵܗ݇ܒܵܐ ܐܲܓ݂ܪܵܐ ܘܦܘܼܪܥܵܢܵܐ. ܘܓܲܠܝܘܼܬ݂ ܐܲܦܹ̈ܐ ܠܲܡܗܲܝܡ̈ܢܹܐ. ܒܝܵܘܡܵܐ ܪܲܒܵܐ ܕܒ݂ܘܼܚܪܵܢܵܐ. ܕܦܵܪܫܝܼܢ ܛܵܒܹ̈ܐ ܡܢ ܒܝܼܫܹ̈ܐ. ܗܝ ܗܘܵܬ݂ ܠܪܹܫܹܗ ܕܒܸܢܝܵܢܵܐ. ܨܠܘܿܬ݂ܵܐ ܩܠܝܼܕ݂ܵܐ ܗ݇ܝ ܕܲܒ݂ܪܝܼܬ݂ܵܐ. ܕܗܝ ܦܵܬ݂ܚܵܐ ܬܪܲܥ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܘܝܵܗ݇ܒܵܐ ܐܲܓ݂ܪܵܐ ܘܦܘܼܪܥܵܢܵܐ. ܡܢ ܗܵܘ̇ ܓܲܙܵܐ ܡܲܥܬܲܪ ܠܟ݂ܠ. ܒܵܗ̇ ܡܵܪܝ ܬܸܦ̮ܬܲܚ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܬܪܲܥ ܡܵܘܗܲܒ݂ܬ݂ܵܟ݂ ܠܲܣܢܝܼܩܹ̈ܐ. ܘܐܲܫܦܲܥ ܪ̈ܲܚܡܲܝܟ ܥܲܠ ܥܲܒ݂ܕܲܝ̈ܟ. ܕܒ݂ܵܟ݂ ܡܸܬ݂ܓܵܘܣܝܼܢ ܘܩܵܪܹܝܢ ܫܡܵܟ݂. ܡܸܛܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܐܲܠܵܗܵܐ. ܠܦܘܼܠܚܵܢ ܟܲܪܡܵܐ ܪܘܼܚܵܢܵܐ. ܩܪܲܝܬܵܢ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܗܲܒ݂ܠܲܢ ܕܢܸܓ݂ܡܘܿܪ ܨܸܒ݂ܝܵܢܵܟ݂. ܒܫܘܼܡܠܵܝ ܦܹܐܪܹ̈ܐ ܕܦܘܼܩܕܵܢܲܝ̈ܟ. ܕܐܲܢ݇ܬ݁ܘܼ ܓܦܸܬܵܐ ܕܲܫܪܵܪܵܐ. ܘܐܲܒ݂ܘܼܟ݂ ܐܝܼܬ݂ܵܘܗܝ ܦܲܠܵܚܵܐ. ܘܲܚܢܲܢ ܟܠܲܢ ܣܵܘ̈ܟܵܬ݂ܵܐ. ܩܲܒܸ݁ܠ ܦܹܐܪܹ̈ܐ ܕܣܸܦ̈ܘܵܬܲܢ. ܐܲܢܗܲܪ ܐܲܦܲܝܟ ܘܢܸܬ݂ܦܪܸܩ. ܐܲܢܗܲܪ ܐܲܒ݂ܵܐ ܒܢܲܦ̮̈ܫܵܬܲܢ. ܢܘܼܗܪܵܐ ܕܨܸܡܚܹܗ ܕܝܼܚܝܼܕ݂ܵܟ݂. ܘܢܸܙܕܲܗܘܿܢ ܡܲܕܥܲܝ̈ܢ ܟܠܫܵܥ. ܡܢ ܙܲܠܝܼܩܹ̈ܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܘܬܸ݁ܕ݂ܠܲܩ ܒܲܢ ܢܘܼܪ ܪܸܚܡܬ݂ܵܟ݂. ܘܒ݂ܵܗ̇ ܢܸܫܬܲܠܗܲܒ݂ ܒܟ݂ܠܥܸܕܵܢ. ܘܲܕ݂ܠܵܐ ܫܲܠܘܵܐ ܢܲܣܸܩ ܠܵܟ݂. ܫܘܼܒ݂ܚܵܐ ܐܝܼܬ݂ܝܵܐ ܡܵܪܵܐ ܕܟ݂ܠ. ܫܵܘܙܸܒܲܝܢ ܘܦܲܨܵܢ ܡܢ ܕܵܪܵܐ ܗܵܢܵܐ ܠܥܵܠܲܡ. ܡܵܪܲܢ ܫܵܘܙܸܒ݂ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܡܢ ܒܝܼܫܵܐ ܘܡܢ ܒܥܸܠܕܒ݂ܵܒܹ̈ܐ. ܘܡܢ ܫܹܐܕܹ̈ܐ ܘܕܲܝܘܹ̈ܐ ܛܪ̈ܘܼܢܹܐ. ܘܡܢ ܡܲܚܫ̈ܒ݂ܵܬ݂ܵܐ ܒܝܼ̈ܫܵܬ݂ܵܐ. ܘܡܢ ܗܘܼܦܵܟ݂ܵܐ ܕܒܲܚܛܵܗܹ̈ܐ. ܘܡܢ ܚܲܪܒܵܐ ܘܫܸܒ݂ܝܵܐ ܘܒܸܙܬ݂ܵܐ. ܘܡܢ ܟܹܐܒܹ̈ܐ ܘܡܢ ܟܘܼܪ̈ܗܵܢܹܐ. ܘܡܢ ܒܘܼܚܪܵܢ ܕܝܼܢܵܟ݂ ܟܹܐܢܵܐ. ܪܲܚܸܡܥܠܲܝܢ ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ. ܡܸܛܠ ܒܵܥܘܼܬ݂ܵܐ ܕܟܹܐܢܹ̈ܐ. ܘܡܸܛܠ ܨܠܘܿܬ݂ܵܐ ܕܲܢܒ݂ܝܼܹ̈ܐ. ܘܡܸܛܠ ܦܝܵܣܵܐ ܕܲܫܠܝܼܚܹ̈ܐ. ܘܡܸܛܠ ܬܲܟ݂ܫܲܦ̮ܬܵܐ ܕܣܵܗܕܹ̈ܐ. ܡܵܪܝܵܐ ܘܡܸܛܠ ܬܲܚܢܲܢܬܵܐ. ܕܟ݂ܠܵܗ̇ ܥܹܕܬܵܐ ܩܲܕܝܼܫܬܵܐ. ܗܵܪܟܵܐ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܬܲܡܵܢ ܠܵܐ ܬܲܥܠܲܢ ܠܕ݂ܝܼܢܵܐ. ܨܠܘܿܬ݂ܝ ܐܲܝܟ݂ ܥܸܛܪܵܐ ܩܕ݂ܵܡܲܝܟ. ܐܲܝܟ݂ ܦܝܼܪܡܵܐ ܕܩܲܪܸܒ݂ ܐܲܗܪܘܿܢ. ܪܹܝܚܹܗ ܕܟܸܢܫܲܢ ܢܸܒ݂ܣܲܡ ܠܵܟ݂. ܘܐܲܝܟ݂ ܒܵܥܘܼܬ݂ܵܐ ܕܢܝܼܢܘܵܝܹ̈ܐ. ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܩܲܒܸ݁ܠ ܡܵܪܝ. ܘܐܲܝܟ ܕܲܥܢܲܝܬܵܝܗܝ ܠܕ݂ܵܢܝܼܐܹܝܠ. ܡܢ ܓܘܼܒܵܐ ܕܐܲܪ̈ܝܵܘܵܬ݂ܵܐ. ܥܢܝܼ ܡܵܪܝ ܘܥܲܕܲܪ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܒܗܵܠܹܝܢ ܙܲܒ݂ܢܹ̈ܐ ܕܐܘܼܠܨܵܢܵܐ. ܐܸܬ݁ܛܥܝܼܬ݂ ܐܲܝܟ݂ ܢܸܩܝܵܐ ܐܲܒ݁ܝܼܕܬܵܐ ܒܥܝܼܘܗܝ ܠܥܲܒ݂ܕܵܟ݂. ܡܢ ܡܸܢܝܵܢ ܡܐܵܐ ܕܲܓ݂ܙܵܪܵܟ݂. ܪܵܥܝܵܐ ܛܵܒ݂ܵܐ ܘܫܲܪܝܼܪܵܐ. ܐܸܢܵܐ ܗ݇ܘ ܕܲܛܥܹܝܬ݂ ܒܲܠܚܘܿܕܲܝ. ܘܲܒ݂ܣܘܼܪ̈ܚܵܢܲܝ ܡܲܪܡܲܪܬܵܟ݂. ܠܵܐ ܡܵܪܝ ܬܲܗܡܸܐ ܡܢ ܒܵܥܘܼܬ݂ܝ. ܕܐܲܢ݇ܬ݁ܘܼ ܒܲܠܚܘܿܕ݂ ܚܵܐܹܢ ܠܟ݂ܠ. ܐܸܠܵܐ ܒܥܝܼܵܝܢܝ ܒܲܚܢܵܢܵܟ݂. ܘܲܫܒ݂ܘܿܩ ܚܵܘܒܲܝ̈ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܡܵܪܝܵܐ ܢܒܲܪܸܟ݂ ܠܥܲܡܹܗ ܒܲܫܠܵܡܵܐ. ܠܟܸܢܫܲܢ ܒܲܪܸܟ݂ ܦܵܪܘܿܩܲܢ. ܘܐܲܫܪܵܐ ܒܓ݂ܵܘܹܗ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܘܗܲܒ݂ܠܲܢ ܚܲܝܠܵܐ ܕܲܢܫܲܒ݁ܚܵܟ݂. ܟܸܢܫܹ̈ܐ ܕܲܠܥܸܠ ܥܲܡ ܕܲܠܬܲܚܬ݁. ܘܢܲܚܸܒ݂ ܟܠܲܢ ܠܲܚ̈ܕ݂ܵܕܹܐ. ܘܢܸܩܢܸܐ ܚܘܼܒܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ. ܐܲܟ݂ܡܵܐ ܕܲܩܢܹܝܢ ܪ̈ܘܼܚܵܢܹܐ. ܚܲܕ݂ ܨܸܒ݂ܝܵܢܵܐ ܕܠܵܟ݂ ܢܸܫܦܪܘܼܢ. ܝܵܬܹܒ݂ ܥܲܠ ܟܪ̈ܘܿܒܹܐ ܐܸܬ݂ܓܲܠܝ. ܝܵܬܹܒ݂ ܡܢ ܝܲܡܝܼܢ ܐܲܒ݂ܵܐ. ܒܝܲܡܝܼܢܵܟ݂ ܒܲܪܸܟ݂ ܟܸܢܫܲܢ. ܘܒܲܕ݂ܪܵܥܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂. ܢܲܛܲܪ ܠܲܙܒ݂ܝܼܢܲܝ̈ ܒܲܕ݂ܡܵܟ݂. ܘܢܘܼܗܪܵܟ݂ ܢܸܪܕܘܿܦ ܡܢ ܥܹܕܬܵܟ݂. ܚܸܫܘܿܟ݂ܵܐ ܕܠܵܐ ܝܼܕܲܥܬ݂ܵܐ. ܘܥܲܡ ܟܹܐܢܹ̈ܐ ܕܲܫܦܲܪܘ ܩܘܼܕ݂ܡܲܝܟ. ܐܲܫܘܵܢ ܕܢܹܐܪܲܬ݂ ܡܲܠܟ݁ܘܼܬ݂ܵܐ. ܢܲܛܲܪܲܝܢܝ ܐܲܝܟ݂ ܒܵܒ݂ܬ݂ܵܐ ܕܥܲܝܢܵܐ. ܠܟܸܢܫܵܐ ܪܲܒܵܐ ܕܡܲܪܥܝܼܬ݂ܵܟ݂. ܢܲܛܲܪ ܡܵܪܲܢ ܒܲܚܢܵܢܵܟ݂. ܕܠܵܐ ܢܸܫܬܲܒ݂ܗܲܪ ܘܢܸܣܥܸܐ ܥܠܲܝܢ. ܐܵܟܹܠܩܲܪܨܵܐ ܐܲܝܟ݂ ܕܡܢ ܩܕ݂ܝܼܡ. ܫܲܕܲܪ ܚܲܝܠܵܟ݂ ܡܢ ܪܵܘܡܵܐ. ܘܢܸܗܘܸܐ ܫܘܼܪܵܐ ܠܡܲܫܪܝܼܬܲܢ. ܘܲܢܩܘܼܡ ܒܐܲܦܲܝ̈ܢ ܘܠܵܐ ܢܸܬ݂ܢܟܸܐ. ܘܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬ݂ܵܟ ܬܸܬ݂ܢܛܲܪ. ܡܢ ܦܘܼܡܵܐ ܕܲܥܠܲܝܡܹ̈ܐ ܘܕܲܛܠܵܝܹ̈ܐ. ܫܘܼܒ݂ܚܵܐ ܠܪ̈ܲܚܡܲܝܟ ܡܵܪܵܐ ܕܟ݂ܠ. ܫܘܼܒ݂ܚܵܐ ܠܪܸܡܙܵܟ݂ ܐܲܚܝܼܕ݂ ܟܠ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܡܢ ܥܸܠܵܝܹ̈ܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܡܢ ܬܲܚܬܵܝܹ̈ܐ. ܣܥܘܿܪ ܠܲܟ݂ܪ̈ܝܼܗܹܐ ܒܲܚܢܵܢܵܟ݂. ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬ݂ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܢܸܪܘܲܙ ܠܸܒ݁ܗܘܿܢ ܒܦܘܼܪܩܵܢܵܟ݂. ܘܢܹܐܡܪܘܼܢ ܟܠܫܵܥ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܥܲܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܥܲܠ ܩܘܼܫܬܵܟ݂. ܫܘܼܒ݂ܚܵܐ ܠܪ̈ܲܚܡܲܝܟ ܕܲܗܘܵܘ ܠܲܢ. ܐܘܼܪܚܵܐ ܠܲܬ݂ܪܲܥ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܡܫܝܼܚܵܐ ܡܸܫܚܵܐ ܕܕ݂ܘܼܗܵܢܲܢ. ܥܨܵܒ݂ܵܐ ܕܡܲܓܵܢ ܐܸܫܬܲܕܲܪ. ܦܘܿܩ ܒܲܒ݂ܥܵܬܲܢ ܪܲܒ݁ܘܼܠܝ. ܐܲܝܟ݂ ܫܘܼܘܕܵܝܵܟ݂ ܫܲܪܝܼܪܵܐ. ܟܪ̈ܝܼܗܹܐ ܣܢܝܼܩܝܼܢ ܥܲܠ ܗܵܕܹܐ. ܠܵܘ ܫܲܪܝܼܪܹ̈ܐ ܒܢܲܝ̈ ܐ݇ܪ̈ܵܙܲܝܟ. ܦܬܲܚܠܝܼ ܬܲܪ̈ܥܹܐ ܕܙܲܕܝܼܩܘܼܬ݂ܵܐ. ܚܲܢܵܢܵܐ ܘܲܡܠܸܐ ܪ̈ܲܚܡܹܐ. ܦܬܲܚ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܠܲܣܢܝܼܩܹ̈ܐ. ܘܦܲܢܵܐ ܫܹܐ̈ܠܵܬ݂ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܩܵܪܹܝܢ ܠܲܫܡܵܟ݂ ܩܲܕܝܼܫܵܐ. ܘܡܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܢܸܣܬܲܝܥܘܼܢ. ܠܫܘܼܡܠܵܝ ܟܠܗܘܿܢ ܦܘܼܩܕܵܢܲܝ̈ܟ. ܕܐܲܢ݇ܬ݁ܘܼ ܝܲܡܵܐ ܕܲܚܢܵܢܵܐ. ܘܫܵܒܹ݁ܩ ܚܵܘܒܹ̈ܐ ܕܡܵܝܘܿܬܹ̈ܐ. ܐܲܣܵܢܝ ܡܵܪܝܵܐ ܡܸܛܠ ܕܙܵܥܘ ܓܲܪ̈ܡܲܝ. ܡܫܝܼܚܵܐ ܐܵܣܝܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ. ܥܨܘܿܒ݂ ܫܘܼ̈ܡܵܬܲܢ ܒܲܚܢܵܢܵܟ݂. ܘܐܲܣܵܐ ܟܹܐܒܲܝ̈ܢ ܘܟ݂ܘܼܪ̈ܗܵܢܲܢ. ܒܣܲܡܵܐ ܪܲܒܵܐ ܕܲܚܢܵܢܵܟ݂. ܘܚܲܝܸܠ ܒܪ̈ܲܚܡܲܝܟ ܡܚܝܼܠܘܼܬܲܢ. ܘܕܲܟܵܐ ܒܙܘܿܦܵܟ݂ ܟܘܼܬ݂̈ܡܵܬܲܢ. ܘܡܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܥܲܬ݁ܝܼܪܵܐ. ܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ. ܐܲܣܵܐ ܬܒ݂ܵܪܹܗ ܕܠܹܒ݁ܝ ܘܲܥܨܘܿܒ݂. ܛܵܒ݂ܵܐ ܕܐܸܬ݁ܬܲܚܬܝܼ ܒܚܘܼܒܹ݁ܗ. ܘܲܠܒܸܫ ܦܲܓ݂ܪܵܐ ܕܡܵܝܘܿܬܹ̈ܐ. ܘܐܲܣܝܼ ܟܹܐܒ݂ܵܘ̈ܗܝ ܕܲܟ݂ܝܵܢܲܢ. ܘܪܲܡܪܸܡ ܫܸܦܠܵܐ ܕܡܘܼܟܵܟܹܗ. ܐܲܣܵܐ ܡܘܼܡܲܝ̈ܢ ܒܲܚܢܵܢܵܟ݂. ܘܲܡܪܘܿܩ ܨܵܐܬ݂ܵܐ ܕܲܚܛܵܗܲܝ̈ܢ. ܘܢܲܛܲܪ ܚܲܝܲܝ̈ܢ ܡܸܢ ܒܝܼܫܵܐ. ܘܐܲܫܘܵܢ ܠܲܓ݂ܢܘܿܢ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܡܵܪܝܵܐ ܪ̈ܲܚܡܲܝܟ ܠܥܵܠܲܡ. ܪ̈ܲܚܡܲܝܟ ܢܚܘܼܢܘܼܢ ܚܵܘ̈ܒܵܬܲܢ. ܚܢܵܢܵܟ݂ ܐܵܣܝܵܐ ܢܸܗܘܸܐ ܠܲܢ. ܘܥܘܼܕ݂ܪܵܢܵܟ݂ ܢܸܬ݂ܠܵܘܸܐ ܠܲܢ. ܘܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܗܝ ܬܩܘܼܡ ܒܐܲܦܲܝ̈ܢ. ܐܵܦ ܝܲܡܝܼܢܵܟ݂ ܐܵܘܫܸܛ ܠܲܢ. ܘܲܙܓ݂ܘܿܪ ܡܸܢܲܢ ܠܲܢܟ݂ܝܼܠܵܐ. ܕܨܵܒܹ݁ܐ ܕܢܸܫܒܹ݁ܝܢ ܡܢ ܛܝܵܪܵܟ݂. ܒܝܲܕ݂ ܦܘܼܩܕܵܢܵܟ݂ ܢܸܫܬܲܬܲܩ. ܗܵܘ ܕܐܝܼܬ݂ܵܘܗܝ ܡܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ. ܐܝܼܬ݂ܝܵܐ ܕܡܢ ܡܬ݂ܘܿܡ ܠܵܐ ܡܲܗܡܸܐ. ܡܢ ܚܲܛܵܝܹ̈ܐ ܕܩܵܪܹܝܢ ܠܹܗ. ܐܲܫܪܵܐ ܫܲܝܢܵܟ݂ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܲܫܠܵܡܵܟ݂ ܒܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ. ܘܟ݂ܵܗܢܹ̈ܐ ܘܡܲܠܟܹ̈ܐ ܢܸܬ݂ܐܲܣܪܘܼܢ. ܒܚܘܼܒܵܐ ܫܲܠܡܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ. ܕܥܘܼܡܪܵܐ ܫܲܠܝܵܐ ܘܲܡܫܲܝܢܵܐ. ܢܸܥܡܲܪ ܒܕܲܟ݂ܝܘܼܬ݂ ܕܘܼܒܵܪ̈ܲܝܢ. ܠܵܐ ܬܲܗܦܸܟ݂ ܐܲܦܲܝ̈ܟ ܡܸܢܲܢ. ܒܚܲܫܵܐ ܘܕܸܡܥܹ̈ܐ ܡܦܝܼܣܝܼܢܲܢ. ܡܵܪܵܐ ܛܵܒ݂ܵܐ ܘܒܲܣܝܼܡܵܐ. ܦܬܲܚ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܠܲܣܢܝܼܩܹ̈ܐ. ܕܢܵܩ̇ܫܝܼܢ ܒܲܬ݂ܪܲܥ ܡܵܪܘܼܬ݂ܵܟ݂. ܘܫܲܝܸܢ ܥܵܠܡܵܐ ܒܝܲܕ݂ ܪܸܡܙܵܟ݂. ܘܒܲܛܸܠ ܡܸܢܹܗ ܟܠ ܩܹܐܪ̈ܣܝܼܢ. ܘܲܒ݂ܦܘܼܩܕܵܢܵܟ݂ ܢܸܬ݂ܒܲܛܠܘܼܢ. ܩܪ̈ܵܒܹܐ ܕܫܵܓ݂ܫܝܼܢ ܠܲܒ݂ܪܝܼܬ݂ܵܐ. ܕܐܘܼܪܚܵܐ܀ ܥܲܕܲܪܲܝܢ ܐܲܠܵܗܵܐ ܦܵܪܘܿܩܲܢ. ܡܵܪܝܵܐ ܕܲܗܘܵܐ ܥܲܡ ܝܲܥܩܘܿܒ݂. ܘܫܵܘܙܸܒ݂ ܚܲܝܵܘ̈ܗܝ ܡܸܢ ܥܹܣܘܿ. ܗܘܝܼ ܥܲܡ ܥܲܒ݂ܕܲܝ̈ܟ ܚܲܢܵܢܵܐ. ܒܐܘܼܪܚܵܐ ܕܣܝܼܡܵܐ ܩܕ݂ܵܡܲܝܗܘܿܢ. ܘܢܲܛܲܪ ܐܸܢܘܿܢ ܕܠܵܐ ܢܸܟ݂ܝܵܢ. ܡܸܢ ܒܝܼܫܘܼܬ݂ܵܐ ܕܲܢܟ݂ܝܼ̈ܠܹܐ. ܘܐܲܗܦܸܟ ܐܸܢܘܿܢ ܒܲܫܠܵܡܵܐ. ܘܚܲܝܠܵܟ݂ ܠܸܘܝܵܐ ܢܸܗܘܸܐ ܠܗܘܿܢ. ܐܸܢܵܐ ܕܹܝܢ ܥܠܲܝܟ ܡܵܪܝܵܐ ܐܸܬ݁ܬܲܟ݂ܠܹܬ݂. ܥܲܠ ܬܘܼܟ݂ܠܵܢܵܟ݂ ܦܵܪܘܿܩܲܢ. ܪܵܕܹܝܢ ܒܐܘܼܪܚܵܐ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܐܲܠܘܵܐ ܐܸܢܘܿܢ ܥܘܼܕ݂ܪܵܢܵܟ݂. ܐܲܝܟ݂ ܝܵܘܣܸܦ ܒܐܲܪܥܵܐ ܕܡܸܨܪܹܝܢ. ܘܫܵܘܙܸܒ݂ ܐܸܢܘܿܢ ܡܸܢ ܒܝܼܫܵܐ. ܐܲܝܟ݂ ܕܠܹܐܪܲܡܝܵܐ ܒܓ݂ܵܘ ܝܼܗܘܼܕ݂. ܘܐܲܗܦܸܟ݂ ܐܸܢܘܿܢ ܠܥܘܼܡܪ̈ܲܝܗܘܿܢ. ܐܲܝܟ݂ ܝܲܥܩܘܿܒ݂ ܡܸܢ ܒܹܝܬ݂ ܠܵܒ݂ܵܢ. ܕܡܸܛܪܵܐ܀ ܫܡܲܥ ܐܲܠܵܗܵܐ ܩܵܠܝ ܟܲܕ݂ ܐܸܬ݂ܟܲܫܲܦ ܠܵܟ݂. ܫܡܲܥ ܡܵܪܝ ܒܩܵܠܵܐ ܕܐܲܟܵܪܹ̈ܐ. ܕܚܵܙܹܝܢ ܙܲܪ̈ܥܹܐ ܕܲܟ݂ܡܝܼܪܝܼܢ. ܫܲܕܲܪ ܛܲܠܵܟ݂ ܠܒ݂ܘܼܝܵܐܗܘܿܢ. ܘܟܲܦܲܪ ܕܸܡܥܹ̈ܐ ܕܥܲܝܢܲܝ̈ܗܘܿܢ. ܘܐܲܝܟ݂ ܕܐܸܬ݂ܡܲܟܲܟ݂ܘ ܒܡܲܪܕܘܼܬ݂ܵܐ. ܢܸܬ݂ܠܲܒ݁ܒ݂ܘܼܢ ܗܵܫܵܐ ܒܪ̈ܲܚܡܹܐ. ܕܢܹܐܟ݂ܠܘܼܢ ܟܠ ܬܸܓ݂ܡܝܼ̈ܢ ܘܢܸܣܒ݁ܥܘܼܢ. ܘܲܢܩܲܪܒ݂ܘܼܢ ܠܵܟ݂ ܩܘܼܪ̈ܒܵܢܹܐ. ܫܡܲܥ ܐܲܠܵܗܵܐ ܘܪܲܚܸܡ ܥܠܲܝ. ܫܡܲܝܵܐ ܘܐܲܪܥܵܐ ܗܵܐ ܒܵܟܹܝܢ. ܘܕܸܡܥܹ̈ܐ ܕܚܲܫܵܐ ܗܵܐ ܡܲܪܕܹܝܢ. ܕܚܵܙܹܝܢ ܥܲܒ݂ܕܲܝ̈ܟ ܕܐܲܠܝܼܨܝܼܢ. ܘܠܲܝܬ݁ ܐܝܼܵܠܵܐ ܕܲܡܫܵܘܙܸܒ݂. ܐܸܬ݁ܬ݁ܥܝܼܪ ܡܵܪܝܵܐ ܐܲܝܟ݂ ܕܲܡܟܵܐ. ܘܐܲܝܟ݂ ܕܲܒ݂ܙܲܒ݂ܢܹ̈ܐ ܩܲܕ݂ܡܵܝܹ̈ܐ. ܘܲܦܪܘܿܩ ܘܫܵܘܙܸܒ݂ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܡܸܢ ܐܝܼܕܲܝ̈ܗܘܿܢ ܕܪ̈ܵܕ݂ܘܿܦܹܐ. ܥܘܿܢ̈ܝܵܬ݂ܵܐ ܕܓ݂ܵܘܵܐ܀ ܢܲܛܲܪ ܠܢܲܦ̮ܫܝ ܡܸܛܠ ܕܛܵܒ݂ ܐܲܢ݇ܬ݁. ܒܲܨܠܘܿܬ݂ ܡܲܪܝܲܡ ܕܝܼܠܸܕ݂ܬ݂ܵܟ݂. ܘܝܘܿܚܲܢܵܢ ܕܫܲܡܸܫ ܥܡܵܕ݂ܵܟ݂. ܘܦܲܛܪܘܿܣ ܘܦܵܘܠܘܿܣ ܥܲܡ ܡܲܬܲܝ. ܘܡܲܪܩܘܿܣ ܘܠܘܼܩܵܐ ܘܝܘܿܚܲܢܵܢ. ܘܣܵܗܕܵܐ ܙܲܗܝܵܐ ܐܸܣܛܲܦܵܢܘܿܣ. ܘܡܵܪܝ ܢܸܣܛܘܿܪܝܼܣ ܘܒܹܝܬ݂ ܢܲܪܣܲܝ. ܘܡܵܪܝ ܐܲܒ݂ܵܐ ܥܲܡ ܥܲܢܝܼܕܹ̈ܐ. ܢܲܛܲܪ ܠܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܘܼܟ݂ܪܵܢܵܐ ܕܛܲܝܒ̈ܘܵܬ݂ܵܟ݂ ܣܲܓܝܼ̈ܐܵܬ݂ܵܐ. ܒܨܵܘܡܵܟ݂ ܘܚܲܫܵܟ݂ ܘܲܩܝܵܡܬܵܟ݂. ܘܣܵܗܕܹ̈ܐ ܕܐܵܘܕܝܼܘ ܒܡܵܪܘܼܬ݂ܵܟ݂. ܘܡܵܪܝ ܓܝܼܘܲܪܓ݁ܝܼܣ ܢܲܨܝܼܚܵܐ. ܘܣܘܼܠܵܩܵܟ݂ ܘܡܲܚܬܹܗ ܕܪܘܼܚܵܟ݂. ܘܲܫ̈ܟ݂ܵܚܵܬ݂ܵܐ ܕܲܨܠܝܼܒ݂ܵܟ݂. ܘܩܘܼܕܵܫܵܗ̇ ܕܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ. ܢܲܛܲܪ ܠܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܡܢ ܢܸܟ݂ܝܵܢܵܘ̈ܗܝ ܕܡܵܪܘܿܕ݂ܵܐ. ܕܡܲܪܝܲܡ܀ ܡܸܛܠ ܕܡܢ ܠܘܵܬܹܗ ܐܸܢܘܿܢ ܪ̈ܲܚܡܹܐ. ܘܣܲܓ݁ܝܼ ܠܘܵܬܹܗ ܦܘܼܪܩܵܢܵܐ. ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܩܲܕܝܼܫܬܵܐ. ܐܲܦܝܼܣܝ ܘܲܒ݂ܥܵܝ ܠܲܡܫܝܼܚܵܐ. ܕܢܸܥܒܸ݁ܕ݂ ܪ̈ܲܚܡܹܐ ܥܲܠ ܥܵܠܡܵܐ. ܕܐܲܚܝܼܕ݂ ܓܵܘܣܵܐ ܒܲܨܠܘܿܬܹܟ݂ܝ. ܘܥܹܕܬܵܐ ܬܸܚܕܸܐ ܒܥܲܕ݂ܥܹܕܹܟ݂ܝ. ܘܝܲܠܕܹܝ̈ܗ̇ ܒܓ݂ܵܘܵܗ̇ ܢܸܬ݂ܢܲܛܪܘܼܢ. ܡܢ ܢܸܟ݂ܝܵܢܹ̈ܐ ܘܣܲܩܘܼܒ݂̈ܠܹܐ. ܕܐܵܟܹܠܩܲܪܨܵܐ ܡܵܪܘܿܕ݂ܵܐ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ ܐܲܢ݇ܬ݁ ܥܢܲܝܬ݁ ܐܸܢܘܿܢ. ܐܲܠܵܗܵܐ ܬܵܒ݂ܘܿܥܵܐ ܗܘܲܝܬ݁ ܠܗܘܿܢ. ܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܘܙܲܕܝܼܩܹ̈ܐ. ܢܒ݂ܝܼܹ̈ܐ ܫܠܝܼܚܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ. ܘܣܵܗܕܹ̈ܐ ܘܟ݂ܵܗܢܹ̈ܐ ܘܥܵܢܘܵܝܹ̈ܐ. ܕܪܲܥܝܘܼܟ݂ ܘܲܫܦܲܪܘ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܒܲܛܸܠ ܩܹܐܪ̈ܣܹܐ ܘܲܩܪ̈ܵܒܹܐ. ܘܫܲܝܸܢ ܟܵܗܢܹ̈ܐ ܥܲܡ ܡܲܠܟܹ̈ܐ. ܘܐܲܦܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܠܪ̈ܲܚܝܼܩܹܐ. ܘܢܲܛܲܪ ܒܪܸܡܙܵܟ݂ ܠܩܲܪ̈ܝܼܒܹܐ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܚܵܘܝܼ ܡܵܪܝܵܐ ܦܘܼܪܩܵܢܹܗ. ܨܠܝܼܒ݂ܵܐ ܕܝܼܫܘܿܥ ܥܠܵܘܗܝ ܐܸܨܛܠܸܒ݂. ܡܵܘܬܵܐ ܘܚܲܝܹ̈ܐ ܡܸܢܹܗ ܢܒܲܥܘ. ܡܵܘܬܵܐ ܠܥܲܡܵܐ ܕܠܵܐ ܗܲܝܡܸܢ. ܘܚܲܝܹ̈ܐ ܠܥܲܡܡܹ̈ܐ ܕܗܲܝܡܸܢܘ ܒܹܗ. ܟܹܐܢ ܕܲܝܵܢܵܐ ܘܕ݂ܝܼܢܹܗ ܬܪܝܼܨ. ܕܥܲܡܵܐ ܕܨܲܠܒܹܗ ܕܲܝܪܹܗ ܚܪܲܒ݂. ܘܥܲܡܡܹ̈ܐ ܕܗܲܝܡܸܢܘ ܒܡܵܪܘܼܬܹܗ. ܫܘܼܒ݂ܚܵܐ ܒܥܹܕܬܹܗ ܗܵܐ ܙܵܡܪܝܼܢ. ܕܐܲܒ݂ܘܼܢ܀ ܢܸܥܒܸ݁ܕ݂ ܡܵܪܝܵܐ ܟܠܹܗ ܨܸܒ݂ܝܵܢܵܟ݂. ܐܲܒ݂ܘܼܢ ܕܲܢܨܲܚܬ݁ ܒܐܵܓ݂ܘܿܢܵܐ. ܗܵܐ ܒܲܫܡܲܝܵܐ ܦܘܼܪܥܵܢܵܟ݂. ܡܫܝܼܚܵܐ ܕܨܲܒܸ݁ܬ݁ܬ݁ ܠܹܗ ܩܢܘܿܡܵܟ݂. ܒܥܹܕܬܹܗ ܐܵܘܪܸܒ݂ ܕܘܼܟ݂ܪܵܢܵܟ݂. ܦܝܼܪܡܵܐ ܕܲܟ݂ܝܵܐ ܚܘܼܒܵܟ݂ ܗܘܵܐ. ܘܪܲܥܝܼܬ݁ ܠܡܵܪܵܟ݂ ܒܝܲܕ݂ ܥܲܡܠܲܝ̈ܟ. ܐܲܦܝܼܣ ܥܲܡܲܢ ܕܡܵܐ ܕܡܸܬ݂ܓܠܸܐ. ܒܫܘܼܒ݂ܚܹܗ ܪܲܒܵܐ ܢܪܲܚܸܡܥܠܲܝܢ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܠܵܐ ܬܸܫܕܹܝܢܝ ܡܢ ܩܕ݂ܵܡܲܝܟ. ܒܝܵܘܡܵܐ ܪܲܒܵܐ ܕܓܸܠܝܵܢܵܟ݂. ܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ. ܡܵܐ ܕܙܵܐܲܚ ܐܲܢ݇ܬ݁ ܠܡܹܐܬ݂ܵܐ. ܠܲܡܕ݂ܵܢ ܟܠܵܐ ܒܟܹܐܢܘܼܬ݂ܵܐ. ܚܘܼܣ ܥܲܠ ܐܲܝܠܹܝܢ ܕܐܸܫܬܵܘܬܲܦܘ. ܒܦܲܓ݂ܪܵܟ݂ ܘܲܕ݂ܡܵܟ݂ ܡܚܲܣܝܵܢܵܐ. ܘܐܲܫܘܵܢ ܐܸܢܘܿܢ ܠܲܓ݂ܢܘܿܢܵܟ݂. ܕܥܲܡ ܩܲܕܝܼܫܲܝ̈ܟ ܢܸܬ݂ܒܲܣܡܘܼܢ. ܢܸܪܘܙܘܼܢ ܘܲܢܫܲܒ݁ܚܘܼܢ. ܥܲܢܝܼ̈ܕܹܐ ܕܠܲܒ݂ܫܘܼܟ݂ ܡܵܪܲܢ. ܡܢ ܡܲܝ̈ܵܐ ܕܡܲܥܡܘܿܕ݂ܝܼܬ݂ܵܐ. ܒܵܟ݂ ܢܸܬ݂ܚܵܘܪܘܼܢ ܦܲܓ݂ܪ̈ܲܝܗܘܿܢ. ܡܢ ܟܘܼܬ݂̈ܡܵܬ݂ܵܐ ܕܲܚܛܝܼܬ݂ܵܐ. ܥܲܢܝܼ̈ܕܹܐ ܕܐܸܟܲܠܘ ܦܲܓ݂ܪܵܟ݂. ܘܒܲܕ݂ܡܵܟ݂ ܚܲܝܵܐ ܐܸܬ݂ܒܲܣܲܡܘ. ܥܒܸܕ݂ ܡܵܪܲܢ ܕܘܼܟ݂ܪ̈ܵܢܲܝܗܘܿܢ. ܒܐܲܬ݂ܪܵܐ ܕܟܹܐܢܹ̈ܐ ܡܕܲܝܪܝܼܢ ܒܹܗ܀
	ܫܲܒܲܚ. ܪܵܥܝܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮ܫ̈ܵܬܲܢ. ܥܵܕ݂ܘܿܪܵܐ ܘܦܵܪܘܿܩܵܐ ܕܟ݂ܠ. ܕܲܒܲܪ ܠܲܢ ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܵܟ݂. ܬܚܹܝܬ݂ ܛܸܠܵܠܵܐ ܕܝܲܡܝܼܢܵܟ݂. ܘܩܲܒܸ݁ܠ ܦܹܐܪ̈ܐ ܡܠܝܼ̈ܠܹܐ ܕܣܸܦ̈ܘܵܬܲܢ. ܕܲܒ݂ܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܩܲܪܒ݂ܝܼܢܲܢ ܠܵܟ݂. ܘܗܲܒ݂ܠܲܢ ܪ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ. ܡܼܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܥܲܬ݁ܝܼܪܵܐ. ܥܵܠܲܡ. ܒܨܵܘܡܵܐ ܘܚܲܫܵܐ ܘܲܨܠܘܿܬ݂ܵܐ. ܢܪܲܥܹܝܘܗܝ ܟܠܲܢ ܠܲܡܫܝܼܚܵܐ. ܕܗܘܝܘܼ ܫܵܒܹ݁ܩ ܚܵܘ̈ܒܵܬܲܢ. ܘܥܵܛܹܐ ܣܘܿܓ݂ܵܐܐ ܕܲܚܛܵܗܲܝ̈ܢ. ܘܢܸܩܪܘܿܒ݂ ܘܢܸܣܲܒ݂ ܕܲܟ݂ܝܵܐܝܼܬ݂. ܐ݇ܪ̈ܵܙܹܐ ܫܒ݂ܝܼܚܹ̈ܐ ܘܩܲܕ݁ܝܼܫܹ̈ܐ. ܘܢܸܩܥܸܐ ܘܢܹܐܡܲܪ ܫܲܘܝܵܐܝܼܬ݂. ܒܪܝܼܟ݂ܘܼ ܐܝܼܩܵܪܹܗ ܕܡܵܪܝܵܐ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܩܵܢܘܿܢܵܐ. ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ. ܙܲܝܸܢ ܠܲܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܒܙܲܝܢܵܐ ܚܲܣܝܼܢܵܐ ܘܠܵܐ ܡܸܙܕܲܟ݂ܝܵܢܵܐ: ܒܲܨܠܵܘ̈ܵܬ݂ܵܗ̇ ܕܐܸܡܵܟ݂ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܲܥܒܸܕ݂ ܠܲܢ ܥܲܡܵܗ̇ ܡܢܵܬ݂ܵܐ ܘܫܵܘܬܵܦܘܼܬ݂ܵܐ ܒܲܓ݂ܢܘܿܢܵܟ݂ ܫܡܲܝܵܢܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ... ܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
	ܩܵܢܘܿܢܵܐ. ܡܢ ܥܘܼܡܩܵܐ ܩܪܹܝܬ݂ܵܟ݂ ܡܵܪܝܵܐ ܘܲܫܡܲܥܬ݁ ܒܩܵܠܝ* ܢܸܗܘ̈ܝܵܢ ܐܸܕ݂ܢܲܝ̈ܟ ܨܵܝ̈ܬܵܢ ܠܩܵܠܵܐ ܕܟ݂ܘܼܫܵܦܝ* ܛܵܒ݂ܵܐ ܘܟܹܐܢܵܐ ܚܲܢܵܢܵܐ: ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܪܲܚܸܡܥܠܲܝ. ܐܸܢ ܚܛܵܗܹ̈ܐ ܢܵܛܲܪ ܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܡܲܢܘܼ ܡܸܫܟܲܚ ܠܲܡܩܵܡ* ܡܸܛܠ ܕܡܢ ܠܘܵܬ݂ܵܟ݂ ܗ݇ܘ ܫܘܼܒ݂ܩܵܢܵܐ* ܣܲܒ݁ܪܹܬ݂ ܒܡܵܪܝܵܐ ܘܣܲܟ݁ܝܲܬ݀ ܢܲܦ̮ܫܝ ܠܡܸܠܬܹܗ* ܣܲܟ݁ܝܼܬ݀ ܠܡܵܪܝܵܐ ܡܢ ܡܲܛܲܪܬܵܐ ܕܨܲܦܪܵܐ ܘܲܥܕܲܡܵܐ ܠܡܲܛܲܪܬܵܐ ܕܨܲܦܪܵܐ* ܢܣܲܟܸ݁ܐ ܝܼܣܪܵܝܹܠ ܠܡܵܪܝܵܐ* ܡܸܛܠ ܕܡܢ ܠܘܵܬܹܗ ܐܸܢܘܿܢ ܪ̈ܲܚܡܹܐ* ܘܣܲܓ݁ܝܼ ܠܘܵܬܹܗ ܦܘܼܪܩܵܢܵܐ* ܗܘ ܢܸܦܪܩܝܼܘܗܝ ܠܝܼܣܪܵܝܹܠ ܡܸܢ ܟܠܹܗ ܥܵܘܠܹܗ*

ܛܵܒ݂ܵܐ ܘܟܹܐܢܵܐ ܘܚܲܢܵܢܵܐ. ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܪܲܚܸܡܥܠܲܝ܀

	ܫܲܒܲܚ. ܥܵܠܲܡ.

ܒܲܨܠܘܿܬ݂ ܡܲܪܝܲܡ ܕܝܼܠܸܕ݂ܬ݂ܵܟ݂. ܢܲܛܲܪ ܡܵܪܲܢ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ.

	ܡܢ ܥܘܼܡܩܵܐ ܩܪܹܝܬ݂ܵܟ݂ ܡܵܪܝܵܐ ܘܲܫܡܲܥܬ݁ ܒܩܵܠܝ* ܢܸܗܘ̈ܝܵܢ ܐܸܕ݂ܢܲܝ̈ܟ ܨܵܝ̈ܬܵܢ ܠܩܵܠܵܐ ܕܟ݂ܘܼܫܵܦܝ*

 ܒܲܨܠܘܿܬ݂ ܡܲܪܝܲܡ ܕܝܼܠܸܕ݂ܬ݂ܵܟ݂. ܢܲܛܲܪ ܡܵܪܲܢ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ.

	ܬܸܫܒ݁ܘܿܚܬܵܐ. ܚܸܕ݂ܝܲܬ݀ ܡܲܪܝܲܡ ܕܝܼܠܸܕ݂ܬܹܗ. ܘܝܘܿܚܲܢܵܢ ܟܲܕ݂ ܡܲܥܡܸܕ݂ ܠܹܗ. ܚܵܕܹ݁ܝܢ ܒܹܗ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ. ܒܟ݂ܵܘܟ݁ܒ݂ܵܐ ܕܲܕ݂ܢܲܚ ܡܢ ܝܲܥܩܘܿܒ݂܀ ܫܠܝܼܚܹ̈ܐ ܕܲܗܘܵܘ ܡܸܨܥܵܝܹ̈ܐ. ܒܹܝܬ݂ ܐܲܠܵܗܵܐ ܠܲܒ݂ܪܝܼܬ݂ܵܐ. ܥܲܕܲܪܘ ܠܲܢ ܒܲܨܠܵܘ̈ܵܬ݂ܟ݂ܘܿܢ. ܕܢܸܫܬ݁ܘܸܐ ܠܪ̈ܲܚܡܹܐ ܘܠܲܚܢܵܢܵܐ.
	ܟܵܪܘܿܙܘܼܬ݂ܵܐ. ܐܲܚܝܼܕ݂ ܟܠ. ܘܡܸܬ݂ܐܲܡܪܵܐ ܒܟ݂ܠܗܘܿܢ ܣܘܼܒܵܥܹ̈ܐ ܕܨܵܘܡܵܐ.
 ܢܩܘܼܡ ܫܲܦܝܼܪ...܀ ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܲܚܝܼܕ݂ ܟܠ ܐܲܠܵܗܵܐ ܕܐܲܒ݂̈ܵܗܵܬܲܢ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܩܲܕ݁ܝܼܫܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܲܒ݂ܩܲܕ݁ܝܼܫܹ̈ܐ ܥܵܡܲܪ ܘܡܸܬ݁ܬ݁ܢܝܼܚ ܨܸܒ݂ܝܵܢܹܗ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܡܲܠܟܵܐ ܕܡܲܠܟܹ̈ܐ ܘܡܵܪܵܐ ܕܡܵܪ̈ܵܘܵܬ݂ܵܐ ܕܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܐ݇ܢܵܫ ܡܢ ܒܢܲܝ̈ܢܵܫܵܐ ܠܵܐ ܚܙܵܝܗܝ ܘܐܵܦܠܵܐ ܡܨܸܐ ܠܡܸܚܙܝܹܗ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܨܵܒܹ݁ܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܢܸܚܘܿܢ. ܘܢܸܬ݂ܦܢܘܿܢ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܢܟ݂...܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒ݂ܐܲܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂ ܐܲܫܪܵܐ ܒܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ: ܘܐܸܬ݂ܪܲܚܸܡܥܠܲܝܢ.
 ܘܟܸܢ ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ. ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܫܲܒܲܚ܀ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܥܵܠܲܡ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
 ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܦܵܬܲܚ ܬܲܪܥܹܗ ܠܲܕ݂ܢܵܩܫܝܼܢ ܒܹܗ: ܡܦܲܢܸܐ ܫܹܐ̈ܠܵܬ݂ܵܐ ܠܲܕ݂ܫܵܐܠܝܼܢ ܠܹܗ. ܦܬܲܚ ܠܵܗ̇ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܬܲܪܥܵܐ ܕܪ̈ܲܚܡܹܐ ܠܲܨܠܘܿܬܲܢ. ܘܩܲܒܸ݁ܠܹܝܗ̇ ܠܒ݂ܵܥܘܼܬܲܢ. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ ܡܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܥܲܬ݁ܝܼܪܵܐ ܘܲܫܦܝܼܥܵܐ: ܛܵܒ݂ܵܐ ܕܠܵܐ ܟܵܠܹܐ ܪ̈ܲܚܡܵܘܗܝ ܘܡܵܘܗܒ݂ܵܬܹ̈ܗ: ܡܢ ܣܢܝܼܩܹ̈ܐ ܘܐܲܠܝܼܨܹ̈ܐ ܥܲܒ݂ܕܵܘ̈ܗܝ ܕܩ̇ܵܪܹܝܢ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܹܗ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܐ݇ܚܪܹܬ݂ܵܐ. ܗܘܝܼ ܡܵܪܝ ܥܲܡܲܢ ܒܪ̈ܲܚܡܲܝܟ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܘܕܲܒܲܪ ܠܲܢ ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܘܼܪܟ݁ܬ݂ܵܐ ܡܵܪܝ ܕܪ̈ܲܚܡܹܐ: ܡܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܕܪ̈ܲܚܡܹܐ: ܒܝܲܕ݂ ܡܲܠܲܐܟ݂ܵܐ ܕܪ̈ܲܚܡܹܐ: ܡܢ ܠܘܵܬ݂ ܐܲܠܵܗܵܐ ܣܲܓ݁ܝܼ ܪ̈ܲܚܡܹܐ: ܬܸܫܬܲܕܲܪ ܘܬܸܫܬܲܦܲܥ ܥܲܠ ܟܸܢܫܹ̈ܐ ܕܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ: ܕܐܲܡܝܼܢܝܼܢ ܘܩܵܪܹܝ̇ܢ ܠܲܫܡܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ: ܘܚܵܝ̇ܪܝܼܢ ܘܲܡܣܲܟܹ݁ܝܢ ܠܪ̈ܲܚܡܹܐ: ܘܲܠܦܘܼܪܩܵܢܵܐ ܕܡܢ ܠܘܵܬ݂ܵܟ݂: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܡܲܩܦܝܼܢ. ܨܠܘܿܬ݂ܵܗ̇ ܡܵܪܝ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܗ̇ ܕܐܸܡܵܐ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܬܲܟ݂ܫܲܦ̮ܬܵܗ̇ ܘܬܲܚܢܲܢܬܵܗ̇ ܕܡܲܠܝܲܬ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܐ: ܡܵܪܬ݁ܝ ܡܲܪܝܲܡ ܛܘܼܒ݂ܵܢܝܼܬ݂ܵܐ: ܘܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ ܙܲܟܵܝܵܐ: ܘܥܘܼܕ݂ܪܵܢܵܐ ܐܲܠܵܗܵܝܵܐ: ܘܒ݂ܵܥܘܼܬܹܗ ܕܡܵܪܝ ܝܘܿܚܲܢܵܢ ܡܲܥܡܕ݂ܵܢܵܐ: ܘܕܲܫܠܝܼܚܹ̈ܐ ܛܘܼܒ݂ܵܢܹ̈ܐ: ܘܕܲܐܒ݂ܵܗ̈ܵܬ݂ܵܐ ܢܲܨܝܼܚܹ̈ܐ ܡܣܲܒ݁ܪ̈ܵܢܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܘܙܵܪ̈ܥܲܝ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܕܲܐܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܡܵܪܝ ܐܲܒ݂ܵܐ ܩܵܬ݂ܘܿܠܝܼܩܵܐ: ܘܡܵܪܝ ܐܸܣܛܲܦܵܢܘܿܣ ܒܘܼܟ݂ܪܵܐ ܕܣܵܗܕܹ̈ܐ: ܘܲܕ݂ܓܲܢ݇ܒܵܪ ܚܲܝܠܵܐ ܡܵܪܝ ܓܝܼܘܲܪܓܝܼܣ ܣܵܗܕܵܐ ܢܲܨܝܼܚܵܐ: ܘܲܕ݂ܡܵܪܝ ܦܠܵܢ: ܘܲܕ݂ܡܵܪܝ ܐܵܘܓܹ݁ܝܢ ܛܘܼܒ݂ܵܢܵܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܣܝܼܥܬܹ݁ܗ ܪܘܼܚܵܢܵܝܬܵܐ. ܘܕܲܫܡܘܿܢܝܼ ܘܕܲܒ݂ܢܹܝ̈ܗ̇: ܘܲܕ݂ܟ݂ܠܗܘܿܢ ܣܵܗܕܵܘ̈ܗܝ ܘܩܲܕ݁ܝܼ̈ܫܵܘܗܝ ܕܡܵܪܲܢ: ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܫܘܼܪܵܐ ܪܵܡܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܚܲܣܝܼܢܵܐ: ܕܬܸܦܪܘܿܩ ܘܲܬ݂ܦܲܨܸܐ ܘܲܬ݂ܫܵܘܙܸܒ݂ ܘܲܬ݂ܢܲܛܲܪ ܠܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ ܡܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹܗ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܘܟܸܢ ܚܵܬ݂ܡ̇ܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܪ̈ܵܝܹܐ. ܘܢܵܣܹ̇ܒ݂ ܟܵܗܢܵܐ ܨܠܝܼܒ݂ܵܐ ܒܐܝܼܕܹܗ ܕܝܲܡܝܼܢܵܐ: ܘܩܵܐܹ̇ܡ ܒܓܲܒܵܐ ܬܲܝܡܢܵܝܵܐ: ܘܡܲܗܦܸܟ݂ ܐܲܦܵܘ̈ܗܝ ܠܥܲܡܵܐ ܘܐܵܡܲܪ: ܒܲܪܸܟ݂ ܡܵܪܝ ܒܦܘܼܩܕܵܢܟ݂ܘܿܢ. ܘܥܵܢܹ̇ܝܢ: ܒܦܘܼܩܕܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܡܵܪܲܢ: ܘܫܘܼܒ݂ܚܵܐ ܠܲܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ. ܘܡܲܪܟ݁ܢܝܼܢ ܥܲܡܵܐ ܪܹ̈ܫܲܝܗܘܿܢ.
ܘܡܲܩܸܦ ܚܘܼܬܵܡܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ: ܨܸܡܚܹܗ ܕܐܲܒ݂ܵܐ ܡܬ݂ܘܿܡܵܝܵܐ: ܝܼܠܝܼܕ݂ܵܐ ܕܠܵܐ ܫܘܼܪܵܝܵܐ: ܠܥܸܠ ܡܢ ܙܲܒ݂ܢܹ̈ܐ ܘܲܗܘܲܝ̈ܵܐ: ܕܠܲܝܬ݁ ܠܲܢ ܣܲܒ݂ܪܵܐ ܘܣܘܼܟܵܝܵܐ: ܐܸܠܵܐ ܐܸܢ ܐܲܢ݇ܬ݁ ܒܵܪܘܿܝܵܐ. ܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܘܲܓ݂ܒܲܝ̈ܵܐ: ܕܲܫܦܲܪܘ ܠܵܟ݂ ܡܢ ܫܘܼܪܵܝܵܐ: ܥܒܸܕ݂ ܠܲܚܛܵܗܲܝ̈ܢ ܚܘܼܣܵܝܵܐ: ܘܲܠܣܲܟ݂ܠ̈ܘܵܬܲܢ ܫܪܵܝܵܐ: ܘܲܠܥܵܩ̈ܵܬܲܢ ܦܘܼܨܵܝܵܐ: ܘܲܠܫܸܐ̈ܠܵܬܲܢ ܦܘܼܢܵܝܵܐ: ܘܡܲܢܲܥܲܝܢ ܠܢܘܼܗܪܵܟ݂ ܓܲܐܝܵܐ: ܘܢܲܛܲܪܲܝܢ ܒܢܝܼܫܵܟ݂ ܚܲܝܵܐ: ܡܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܟܣܲܝ̈ܵܐ ܘܲܓ݂ܠܲܝ̈ܵܐ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ). ܗܵܫܵܐ ♰ ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܪܵܫܹ̇ܡ ܨܠܝܼܒ݂ܵܐ ܥܲܠ ܥܲܡܵܐ: ܟܲܕ݂ ܦܵܫܹ̇ܛ ܐܝܼܕܹܗ ܠܥܸܠ ܠܐܲܦܲܝ̈ ܡܲܥܪܒ݂ܵܐ: ܘܡܲܝܬܸ݁ܐ ܠܵܗ̇ ܠܡܲܕ݂ܢܚܵܐ: ܘܟܸܢ ܦܵܫܹ̇ܛ ܠܵܗ̇ ܠܓܲܒܵܐ ܕܬܲܝܡܢܵܝܵܐ: ܘܡܵܘܒܸ݁ܠ ܠܵܗ̇ ܠܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ. ܐܲܝܟܲܢܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܸܣܬ݁ܡܸܟ݂ ܥܲܠ ܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܓܲܒܵܐ ܣܸܡܵܠܵܝܵܐ ܕܟܸܢܫܵܐ.
ܘܥܵܢܹ̇ܝܢ: ܡܫܝܼܚܵܐ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬ݂ܵܟ݂: ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܟܵܗܢܘܼܬ݂ܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
ܐ݇ܚܪܹܢܵܐ: ܬܸܬ݂ܩܲܒܲܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܚܲܠܵܫܹ̈ܐ: ܩܕ݂ܵܡ ܒܹܝܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܬܸܗܘܸܐ ܟܢܝܼܫܘܼܬܲܢ ܗܵܕܹܐ ܠܲܢܝܵܚ ܨܸܒ݂ܝܵܢ ܡܵܪܘܼܬ݂ܵܟ݂: ܠܡܸܣܲܒ݂ ܡܸܢܵܟ݂ ܒܛܲܝܒ݁ܘܼ ܚܘܼܠܡܵܢܵܐ ܕܦܲܓ݂ܪܵܐ: ܘܢܘܼܛܵܪܵܐ ܕܢܲܦ̮ܫܵܐ: ܘܫܘܼܪܬܵܚܵܐ ܕܡܵܙܘܿܢܵܐ:(ܘܚܘܼܣܵܝܵܐ ܕܚܵܘܒܹ̈ܐ ܘܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ):ܘܫܲܝܢܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ: ܘܲܫܠܵܡܵܐ ܢܲܓ݁ܝܼܪܵܐ: ܘܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ ܕܠܵܐ ܥܵܒ݂ܪܵܐ ܘܠܵܐ ܡܸܫܬܲܪܝܵܐ ܡܢ ܒܲܝܢܵܬܲܢ: ܒܟ݂ܠܹܗ ܙܲܒ݂ܢܵܐ ܕܩܵܐܹ̇ܡ ܒܹܗ ܥܵܠܡܵܐ ܗܵܢܵܐ: )ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܒܪܝܼܟ݂ ܐܲܠܵܗܵܐ ܠܥܵܠܡܝܼܢ: ܘܲܡܫܲܒܲܚ ܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ ܠܕ݂ܵܪܕܵܪ̈ܝܼܢ: ܕܡܢܗ ܒܵܥܹܝ̇ܢܲܢ: ܘܲܠܝܲܡܵܐ ܫܦܝܼܥܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ: ܕܗܘ ܢܲܫܘܸܐ ܠܲܢ ܠܫܘܼܒ݂ܚܵܐ ܦܲܐܝܵܐ ܕܡܲܠܟ݁ܘܼܬܹܗ: ܘܲܠܒ݂ܘܼܣܵܡܵܐ ܕܥܲܡ ܡܲܠܲܐܟ݂ܵܘ̈ܗܝ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܘܲܠܓܲܠܝܘܼܬ݂ ܐܲܦܹ̈ܐ ܕܲܩܕ݂ܵܡܵܘܗܝ: ܘܲܠܩܵܘܡܵܐ ܕܡܢ ܝܲܡܝܼܢܹܗ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ: ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܪ̈ܲܚܡܵܘܗܝ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟ݁ܠ ܕܐܸܬ݂ܟܲܢܲܫܢܲܢ ܒܒܲܝܬܹ݁ܗ: ܘܨܲܠܝܼܢܲܢ ܩܕ݂ܵܡ ܪܲܒ݁ܘܼܬܹܗ: ܥܲܠ ܣܲܒ݂ܪܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܗܘ ܢܸܫܡܲܥ ܨܠܘܿܬܲܢ ܒܲܚܢܵܢܹܗ: ܘܲܢܩܲܒܸ݁ܠ ܒܵܥܘܼܬܲܢ ܒܚܵܘܣܵܢܹܗ: ܘܲܢܫܝܼܓ݂ ܘܢܸܡܪܘܿܩ ܨܵܐܬ݂ܵܐ ܕܚܵܘܒܲܝ̈ܢ ܘܕܲܚܛܵܗܲܝ̈ܢ ܒܙܘܿܦܵܐ ܫܦܝܼܥܵܐ ܕܚܲܝܘܼܣܬ݂ܵܢܘܼܬܹܗ: ܘܲܢܢܝܼܚ ܢܲܦ̮̈ܫܵܬ݂ܵܐ ܕܥܲܢܝܼܕܲܝ̈ܢ ܒܐܵܘܵܢܹ̈ܐ ܫܒ݂ܝܼܚܹ̈ܐ ܕܡܲܠܟ݁ܘܼܬܹܗ: ܘܢܸܪܘܿܣ ܥܲܠ ܟܠܲܢ ܛܲܠܵܐ ܕܒܲܣܝܼܡܘܼܬܹܗ: ܘܲܥܠܲܝܢ ܘܥܲܠ ܟܠ ܒܸܪ̈ܝܵܢ ܬܲܓܸ݁ܢ ܝܲܡܝܼܢܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬܹܗ: ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܪ̈ܲܚܡܵܘܗܝ:(ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.

ܐ݇ܚܪܹܢܵܐ

: ܐܲܠܵܗܵܐ ܡܵܪܹܟܠ ܕܝܲܗ݇ܒ݂ ܬܸܫܒ݁ܚܵܬܹ̈ܗ ܒܦܘܼܡܲܢ: ܘܲܙܡܝܼܪ̈ܵܬܹܗ ܒܠܸܫܵܢܲܢ: ܘܩܝܼܢܵܬܹ̈ܗ ܒܓܲܓ݁ܪ̈ܵܬܲܢ: ܘܬ݂ܵܘܕ݁ܝܼܬܹܗ ܒܣܸܦ̈ܘܵܬܲܢ: ܘܗܲܝܡܵܢܘܼܬܹܗ ܒܠܸܒܵܘ̈ܵܬܲܢ: ܗܘ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬܲܢ: ܘܲܢܩܲܒܸ݁ܠ ܒܵܥ̈ܘܵܬܲܢ: ܘܢܸܬ݂ܪܲܥܸܐ ܒܬܲܟ݂ܫ̈ܦܵܬܲܢ: (ܘܲܢܚܲܣܸܐ ܚܵܘ̈ܒܵܬܲܢ:)ܘܲܢܦܲܢܸܐ ܠܲܢ ܫܸܐ̈ܠܵܬܲܢ ܫܲܦܝܼܪ̈ܵܬ݂ܵܐ ܠܵܐ ܥܕ݂ܝܼ̈ܠܵܬ݂ܵܐ. ܘܡܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܢܲܫܦܲܥ ܪ̈ܲܚܡܵܘܗܝ ܘܲܚܢܵܢܹܗ: ܥܠܲܝܢ ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡܢ ܣܵܟܹܗ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܢܸܫܬܲܒܲܚ ܫܡܹܗ ܕܐܲܠܵܗܵܐ ܡܵܪܹܟܠ ܒܲܝܢܵܬܲܢ: ܡܛܲܟ݁ܣܵܢܗܘܿܢ ܕܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ. ܘܲܥܠܲܝܢ ܡܚܝܼ̈ܠܹܐ ܘܚܲܛܵܝܹ̈ܐ: ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡܢ ܣܵܟܹܗ. ܘܥܲܠ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇: ܘܥܲܠ ܐܲܒ݂ܵܗܲܝ̈ܢ ܘܪ̈ܲܒܵܢܲܝܢ ܘܡܲܠܦܵܢܲܝ̈ܢ ܘܥܲܢܝܼܕܲܝ̈ܢ: ܘܥܲܢܝܼܕܹ̈ܐ ܕܲܦܪܲܫܘ ܘܫܲܢܝܼܘ ܡܢ ܒܲܝܢܵܬܲܢ: ܘܥܲܠ ܟܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܬܲܓܸ݁ܢ ܝܲܡܝܼܢܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬܹܗ ܘܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܠܐܲܠܵܗܵܐ ܫܘܼܒ݂ܚܵܐ: ܘܲܠܡܲܠܲܐܟܹ̈ܐ ܐܝܼܩܵܪܵܐ: ܘܲܠܣܵܛܵܢܵܐ ܒܸܗܬ݁ܬ݂ܵܐ: ܘܠܲܨܠܝܼܒ݂ܵܐ ܣܸܓ݂ܕ݁ܬ݂ܵܐ: ܘܲܠܥܹܕ݁ܬܵܐ ܪܘܼܡܪܵܡܵܐ: ܘܲܠܥܲܢܝܼܕܹ̈ܐ ܢܘܼܚܵܡܵܐ: ܘܲܠܬܲܝܵܒܹ̈ܐ ܩܘܼܒܵܠܵܐ: ܘܠܲܐܣܝܼܪܹ̈ܐ ܫܪܵܝܵܐ: ܘܲܠܡܲܪ̈ܥܹܐ ܘܲܟ݂ܪ̈ܝܼܗܹܐܚܘܼܠܡܵܢܵܐ ܘܐܵܣܝܘܼܬ݂ܵܐ: ܘܠܲܐܪ̈ܒܲܥܦܸܢܝܵܬܹܗ ܕܥܵܠܡܵܐ ܫܲܝܢܵܐ ܪܲܒܵܐ ܘܲܫܠܵܡܵܐ. ܘܐܵܦ ܥܠܲܝܢ ܕܐܝܼܬܲܝܢ ܡܚܝܼ̈ܠܹܐ ܘܚܲܛܵܝܹ̈ܐ: ܚܢܵܢܹܗ ܘܪ̈ܲܚܡܵܘܗܝ ܕܐܲܠܵܗܲܢ ܣܓ݂ܝܼܕ݂ܵܐ ܢܹܐܬ݂ܘܿܢ ܘܢܲܓ݁ܢܘܼܢ ܘܢܸܫܬܲܦܥܘܼܢ ܘܲܢܩܵܘܘܿܢ ܘܢܲܡܠܟ݂ܘܼܢ ܒܐܲܡܝܼܢܘܼܬ݂ܵܐ: )ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: )ܒܝܲܡܝܼܢ ܡܵܪܘܼܬ݂ܵܟ݂ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ): ܒܲܪܸܟ݂ܡܵܪܝ ܠܟ݂ܠܲܢ: ܘܢܲܛܲܪ ܠܟ݂ܠܲܢ: ܘܥܲܕܲܪ ܠܟ݂ܠܲܢ: ܘܣܲܝܲܥ ܘܣܲܬܲܪ ܠܟ݂ܠܲܢ: ܘܢܲܚܸܡ ܥܲܢܝܼܕܹ̈ܐ ܕܟ݂ܠܲܢ: ܘܐܲܓܸ݁ܢ ܝܲܡܝܼܢܵܟ݂ ܥܲܠ ܟܠܲܢ: ܘܪ̈ܲܚܡܲܝܟ ܘܲܚܢܵܢܵܟ݂ ܢܸܫܬܲܦܥܘܼܢ ܥܲܠ ܟܠܲܢ:ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ: ܘܩܘܼܒܵܠܛܲܝܒ݁ܘܼܬ݂ܵܐ ܐܲܡܝܼܢܬܵܐ ܬܸܣܲܩ ܠܵܟ݂ ܡܢ ܦܘܼܡ ܟܠܲܢ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟ݁ܠ ܢܒܲܪܸܟ݂ ܠܟܸܢܫܲܢ ܒܛܲܝܒ݁ܘܼܬܹܗ: ܘܲܢܢܲܛܲܪ ܠܲܩܝܵܡܲܢ ܒܣܘܿܓ݂ܐܵܐ ܫܦܝܼܥܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܘܲܢܦܲܢܸܐ ܫܸܐ̈ܠܵܬܲܢ ܡܢ ܒܹܝܬ݂ ܓܲܙܹܗ: ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡܢ ܣܵܟܹܗ: ܘܥܲܠ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇: ܘܥܲܠ ܐܲܬ݂ܪܵܐ ܗܵܢܵܐ ܘܥܵܡܘܿܪ̈ܵܘܗܝ: ܘܥܲܠ ܥܘܼܡܪܵܐ ܗܵܢܵܐ ܘܝܲܬܵܒ݂ܵܘ̈ܗܝ: ܘܥܲܠ ܟܠܲܢ ܒܟ݂ܠܲܢ ܟܢܝܼܫܵܐܝܼܬ݂: ܚܢܵܢܹܗ ܘܪ̈ܲܚܡܵܘܗܝ ܕܐܲܠܵܗܲܢ ܛܵܒ݂ܵܐ: ܢܹܐܬ݂ܘܿܢ ܘܢܸܫܬܲܦܥܘܼܢ ܒܐܲܡܝܼܢܘܼ (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܡܲܩܦܝܼܢ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܲܬ݂ܠܵܬ݂ܡܐܵܐ ܘܲܬ݂ܡܲܢܬܲܥܣܲܪ ܐܲܒ݂ܵܗ̈ܵܬ݂ܵܐ ܐܲܦܸܣܩܘܿܦܹ̈ܐ: ܕܐܸܬ݂ܟܲܢܲܫܘ ܒܢܝܼܩܝܼܵܐ ܡܕ݂ܝܼܢ݇ܬܵܐ ܕܒܹܝܬ݂ܘܿܢܝܼܵܐ: ܒܙܲܒ݂ܢܵܐ ܕܩܘܿܣܛܲܢܛܝܼܢܘܿܣ ܡܲܠܟܵܐ ܙܲܟܵܝܵܐ. ܘܥܸܠܬ݂ܵܐ ܕܹܝܢ ܕܲܟ݂ܢܘܼܫܝܼܗܘܿܢ ܡܸܛܠ ܪܘܼܫܥܹܗ ܕܐܲܪܝܼܘܿܣ܀ ܘܐܲܝܟ݂ ܥܝܵܕ݂ܵܐ ܕܹܝܢ ܕܪܵܕܹܐ ܒܲܝܢܵܬ݂ ܒܢܹܝ̈ܗ̇ ܕܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܲܫܠܝܼܚܵܝܬܵܐ ܕܝܼܠܲܢ ܕܡܲܕ݂ܢܚܵܐ: ܘܲܟ݂ܒܲܪ ܕܡܢ ܪܵܘܡܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܪܕ݂ܵܐ: ܘܡܢ ܡܲܫܠܡܵܢܘܼܬ݂ܵܐ ܘܝܘܼܒܵܠܵܐ ܕܥܲܬ݁ܝܼܩܹ̈ܐ ܡܓ݂ܵܘܝܲܝ̈ ܒܬܹܐܘܿܪ̈ܝܼܵܣ ܕܲܫܪܵܪܵܐ ܐܲܠܵܗܵܝܵܐ ܡܬ݂ܝܼܚ ܗ݇ܘܵ ܥܕܲܡܵܐ ܠܝܵܘ̈ܡܵܬܲܢ. ܘܗܵܟܲܢܵܐ ܡܫܲܪܹܝܢ ܚ݇ܢܲܢ ܠܡܸܬ݂ܢܵܐ ܩܵܢܘܿܢܵܐ ܕܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܢܝܼܩܝܼܵܐ.

ܒܲܫܪܵܪܵܐ ܓܹܝܪ ܕܠܵܐ ܦܘܼܠܵܓ݂ܵܐ:

ܡܗܲܝܡܢܝܼܢܲܢ ܒܚܲܕ݂ ܐܲܠܵܗܵܐ: ܐܲܒ݂ܵܐ ܐܲܚܝܼܕ݂ ܟܠ. ܥܵܒ݂ܘܿܕ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܐܲܝܠܹܝܢ ܕܡܸܬ݂ܚܲܙ̈ܝܵܢ ܘܲܕ݂ܠܵܐ ܡܸܬ݂ܚܲܙ̈ܝܵܢ. ܘܲܒ݂ܚܲܕ݂ ܡܵܪܝܵܐ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ. ܒܪܹܗ ܕܐܲܠܵܗܵܐ ܝܼܚܝܼܕ݂ܵܝܵܐ: ܒܘܼܟ݂ܪܵܐ ܕܟ݂ܠܗܹܝܢ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܗܵܘ̇ ܕܡܢ ܐܲܒ݂ܘܼܗܝ ܐܸܬ݂ܝܼܠܸܕ݂ ܩܕ݂ܵܡ ܟܠܗܘܿܢ ܥܵܠܡܹ̈ܐ: ܘܠܵܐ ܐܸܬ݂ܥܒܸܕ݂. ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܕܡܢ ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܒܲܪ ܟܝܵܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܕܒ݂ܐܝܼ̈ܕ݂ܵܘܗܝ ܐܸܬ݁ܬܲܩܲܢܘ ܥܵܠܡܹ̈ܐ: ܘܐܸܬ݂ܒ݁ܪܝܼ ܟܠ ܡܸܕ݁ܡ. ܗܵܘ̇ ܕܡܸܛܠܵܬܲܢ ܒܢܲܝ̈ܢܵܫܵܐ: ܘܡܸܛܠ ܦܘܼܪܩܵܢܲܢ: ܢܚܸܬ݂ ܡܢ ܫܡܲܝܵܐ: ܘܐܸܬ݂ܓܲܫܲܡ ܡܢ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܘܲܗܘܵܐ ܒܲܪܢܵܫܵܐ: ܘܐܸܬ݂ܒ݁ܛܸܢ ܘܐܸܬ݂ܝܼܠܸܕ݂ ܡܢ ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܘܚܲܫ ܘܐܸܙܕ݁ܩܸܦ: ܒܝܵܘܡܲܝ̈ ܦܲܢܛܝܼܘܿܣ ܦܝܼܠܵܛܘܿܣ. ܘܐܸܬ݂ܩ̄ܒܲܪ: ܘܩܵܡ ܠܲܬ݂ܠܵܬ݂ܵܐ ܝܵܘܡܝܼ̈ܢ ܐܲܝܟ݂ ܕܲܟ݂ܬ݂ܝܼܒ݂. ܘܲܣܠܸܩ ܠܲܫܡܲܝܵܐ: ܘܝܼܬܸܒ݂ ܡܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܘܬ݂ܘܼܒ݂ ܥܬ݂ܝܼܕ݂ ܠܡܹܐܬ݂ܵܐ ܠܲܡܕ݂ܵܢ ܠܡܝܼܬܹ̈ܐ ܘܲܠܚܲܝܹ̈ܐ. ܘܲܒ݂ܚܲܕ݂ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܪܘܼܚܵܐ ܕܲܫܪܵܪܵܐ: ܗܵܘ̇ ܕܡܢ ܐܲܒ݂ܵܐ ܢܵܦܹ݁ܩ: ܪܘܼܚܵܐ ܡܲܚܝܵܢܵܐ. ܘܒܲܚܕ݂ܵܐ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܘܲܫܠܝܼܚܵܝܬܵܐ ܩܵܬ݂ܘܿܠܝܼܩܝܼ. ܘܡܵܘܕܹ݁ܝܢܲܢ ܒܲܚܕ݂ܵܐ ܡܲܥܡܘܿܕ݂ܝܼܬ݂ܵܐ: ܠܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ. ܘܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܲܒ݂ܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ: ܐܵܡܹܝܢ.
