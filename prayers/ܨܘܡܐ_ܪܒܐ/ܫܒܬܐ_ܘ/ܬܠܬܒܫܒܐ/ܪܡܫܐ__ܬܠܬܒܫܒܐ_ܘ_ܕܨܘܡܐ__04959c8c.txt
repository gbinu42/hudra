ܕܡܲܓܵܗܲܝ ܬܠܵܬ݂ܒ݁ܫܲܒܵܐ ܕܸܫܬ݂ܵܐ ܕܨܵܘܡܵܐ.
ܕܲܐَܚܪ̈ܵܝܹܐ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܢܵܘܕܸ݁ܐ ܡܵܪܝ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܡܵܪܘܼܬ݂ܵܟ݂ ܘܫܘܼܒ݂ܚܵܐ ܐܲܡܝܼܢܵܐ ܕܠܵܐ ܫܲܠܘܵܐ ܢܲܣܸܩ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
      ܗܘܼܠܵܠܵܐ. ܐܲܠܵܗܝ ܐܲܠܵܗܝ. ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܟܒ‐ܠ.
      ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ. ܬܢܝܼ. ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ.
 ܘܡܲܩܦܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ. ܡܫܲܒܲܚܬ݁ ܡܵܪܲܢ ܘܠܵܟ݂ ܝܵܐܹܐ ܠܡܲܣܵܩܘܼ ܫܘܼܒ݂ܚܵܐ ܟܠܝܘܿܡ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ܀ ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ ܦܘܼܡܲܢ ܘܝܲܗ݇ܒ݂ܠܲܢ ܕܲܢܗܲܠܸܠ ܒܬܸܫܒ݁ܚܵܬܹ̈ܗ. ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ ܦܘܼܡܲܢ ܘܝܲܗ݇ܒ݂ܠܲܢ ܕܲܢܙܲܡܲܪ ܒܬܸܫܒ݁ܚܵܬܹ̈ܗ. ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ ܦܘܼܡܲܢ ܘܝܲܗ݇ܒ݂ܠܲܢ ܕܲܢܫܲܒܲܚ ܒܬܸܫܒ݁ܚܵܬܹ̈ܗ܀ ܠܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܫܘܼܒ݂ܚܵܐ ܢܲܣܸܩ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܬܢܝܼ ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܲܢܩܲܠܣܵܟ݂ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܲܢܫܲܒܚܵܟ݂ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܕܲܠܘܵܬ݂ ܡܵܝܘܿܬܹ̈ܐ ܡܵܪܵܐ ܡܪܲܚܡܵܢܵܐ. ܚܘܼܢܲܝܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ܀ ܕܠܵܐ ܙܵܟܹ݁ܐ ܩܕ݂ܵܡܲܝܟ ܟܠ ܕܚܲܝ ܒܲܥܒ݂ܝܼܕܹ̈ܐ. ܐܲܢ݇ܬ݁ ܡܵܪܝ ܐܲܦܢܝܼܬܵܢ ܡܢ ܟܠܵܗ̇ ܛܘܼܥܝܲܝ. ܕܐܲܢ݇ܬ݁ܘܼ ܐܲܠܵܗܵܐ. ܘܠܵܟ݂ ܝܵܐܹܐ ܫܘܼܒ݂ܚܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ: ܘܥܲܠ ܐܲܦܲܝ̈ ܟܠܗܘܿܢ ܥܘܼܕ݂ܪ̈ܵܢܲܝܟ ܘܛܲܝ̈ܒ݁ܘܵܬ݂ܵܟ݂ ܕܲܠܘܵܬܲܢ ܕܠܵܐ ܡܸܬ݂ܦܲܪܥܝܼܢ: ܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ ܒܥܹܕ݁ܬܵܟ݂ ܡܟܲܠܲܠܬܵܐ ܡܲܠܝܲܬ݂ ܟܠ ܥܘܼܕ݂ܪ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܵܐ ܘܒ݂ܵܪܘܿܝܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܬܲܪ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
(ܓܝܼܘܿܪܵܐ ܕܲܒ݂ܥܹܕ݇ܬܵܐ.( ܚܕ݂ܝܼܬ݂ ܟܲܕ݂ ܐܵܡ̇ܪܝܼܢ ܗ݇ܘܵܘ ܠܝܼ: ܠܒܲܝܬܹ݁ܗ ܕܡܵܪܝܵܐ ܐܵܙ̇ܠ݇ܝܼܢܲܢ(ܘܲܠܒܲܪ ܡܢ ܥܹܕ݁ܬܵܐ ܐܵܡ̇ܪܝܼܢ) ܒܟ݂ܠܕܘܼܟ ܐܝܼܬܲܝܟ ܐܲܠܵܗܵܐܩܲܒܸ݁ܠ ܬܲܟ݂ܫܲܦ̮ܬܲܢ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ:ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ.ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܟ. ܐܲܢ݇ܬ݁ܘܼ ܡܵܪܝ ܫܲܪܝܼܪܵܐܝܼܬ݂ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܘܲܡܢܲܛܪܵܢܵܐ ܐܲܡܝܼܢܵܐ ܕܚܲܝܲܝ̈ܢ: ܘܠܵܟ݂ ܡܵܪܝ ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	ܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܡ. (ܡܙ:ܣܙ: 1-11). ܐܲܠܵܗܲܢ ܢܪܲܚܸܡ ܥܠܲܝܢ. ܗܲܠܸܠܘܼܝܲܗ ܗܲܠܸܠܘܼܝܲܗ ܐܹܗܲܠܸܠܘܼܝܲܗ. ܢܒܲܪܟܲܢ ܘܢܲܢܗܲܪ ܐܲܦܵܘ̈ܗܝ ܒܲܢ.

ܨܠܘܿܬ݂ܵܐ ܕܡܵܪܝܵܐ ܩܪܹܝ̇ܬ݂ܵܟ݂ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܪ̈ܵܝܹܐ: ܠܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ: ܘܠܲܒ݂ܛܝܼܠܘܼܬܹܗ ܕܨܸܒ݂ܝܵܢܵܟ݂ ܕܲܥܠܲܝܢ: ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܘܐܵܡܪܝܼܢ ܡܵܪܝܵܐ ܩܪܹܝܬ݂ܵܟ݂ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܪ̈ܵܝܹܐ ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܕܡܪܝܐ ܕܩܪܝܬܟ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܒ݂ܵܬܲܪ: ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܪ̈ܵܝܹܐ. ܫܡܲܥ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܒܲܚܢܵܢܵܟ݂: ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܒܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂: ܘܚܘܼܢ ܠܚܲܛܵܝܘܼܬܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܪ̈ܲܚܡܲܝܟ: ܐܵܣܝܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܫܘܼܪܵܝܵܐ ܕܒ݂ܵܬܲܪ. (ܡܙ:ܥ: 1-7). ܐܲܠܵܗܵܐ ܦܲܨܵܢܝ ܡܵܪܝܵܐ ܠܥܘܼܕ݂ܪܵܢܝ ܟܲܬܲܪ. ܗܲܠܸܠܘܼܝܲܗ ܗܲܠܸܠܘܼܝܲܗ ܐܹܗܲܠܸܠܘܼܝܲܗ. ܢܸܒ݂ܗܬ݂ܘܼܢ ܘܢܸܚܦܪܘܼܢ ܐܲܝܠܹܝܢ ܕܒ݂ܵܥܹܝ̇ܢ ܠܢܲܦ̮ܫܝ.

 ܘܢܹܐܡܪܘܼܢ ܒܟ݂ܠܙܒܲܢ ܕܪܲܒ݁ܘܼ ܐܲܠܵܗܵܐ

܀

 ܫܲܒܲܚ܀ ܥܵܠܲܡ܀
ܘܐܵܡܪܝܼܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ
(ܘܪܵܡܹܐ ܡܫܲܡܫܵܢܵܐ ܐܘܼܪܵܪܵܐ: ܘܐܵܣܲܪ ܙܘܼܢܵܪܵܐ: ܘܩܵܐܹ̇ܡ ܩܕ݂ܵܡ ܫܟ݂ܝܼܢܬܵܐ ܡܛܲܟ݁ܣܵܐܝܼܬ݂ (ܘܐܵܡܲܪ :
ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ: ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܫܵܐ ܘܒܲܚܦܝܼܛܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ ܘܐܲܠܵܗܵܐ ܕܟ݂ܠܹܗ ܒܘܼܝܵܐܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ

܀

 ܦܵܪܘܿܩܲܢ ܘܝܲܨܘܼܦܬ݂ܵܢܲܢ ܘܲܡܦܲܪܢܣܵܢܵܐ ܕܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ ܘܩܘܼܝܵܡܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ ܘܲܕ݂ܟ݂ܠܗܹܝܢ ܥܹ̈ܕ݂ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܐܲܬ݂ܪܲܢ ܘܥܲܠ ܟܠ ܐܲܬ݂ܪ̈ܵܘܵܢ: ܘܥܲܠ ܐܲܝܠܹܝܢ ܕܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܥܵܡ̇ܪܝܼܢ ܒܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܘܼܙܵܓ݂ܵܘ̈ܗܝ ܕܐܵܐܲܪ ܘܟܲܗܝܼܢܘܼܬ݂ܵܐ ܕܫܲܢ݇ܬܵܐ ܘܥܲܠ̈ܠܵܬ݂ܵܐ ܕܦܹܐܪܹ̈ܐ ܘܥܲܠ ܬܘܼܩܵܢܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܲܠܟܹ̈ܐ ܐܲܚܝܼܕܲܝ̈ ܫܘܼܠܛܵܢܹܗ ܕܥܵܠܡܵܐ ܗܵܢܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܩܲܫܝܼܫܹ̈ܐ ܘܲܡܫܲܡܫܵܢܹ̈ܐ ܐܲܪ̈ܬܵܕ݁ܘܿܟ݁ܣܹܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ܗ̇ܘ ܕܥܲܬ݁ܝܼܪ ܒܪ̈ܲܚܡܵܘܗܝ ܘܲܫܦܝܼܥ ܒܲܚܢܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܡܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ ܐܝܼܬ݂ܵܘܗܝ ܘܲܠܕ݂ܵܪܕܵܪ̈ܝܼܢ ܩܲܝܵܡ ܫܘܼܠܛܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܟ݂ܝܵܢܹܗ ܐܝܼܬ݂ܵܘܗܝ ܛܵܒ݂ܵܐ ܘܝܵܗܘܿܒ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹܐ ܒܡܵܘܬܹ݁ܗ ܕܚܲܛܵܝܵܐ ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܢ ܥܵܘܠܹܗ ܘܢܸܚܸܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܲܒ݂ܝܲܠܕܹ݁ܗ ܐܵܘ ܕܲܒ݂ܕܸܢܚܹܗ ܐܵܘ ܕܲܒ݂ܨܵܘܡܹܗ ܐܵܘ ܕܲܒ݂ܡܲܥܲܠܬܹ݁ܗ ܐܵܘ ܕܒܲܩܝܵܡܬܹ݁ܗ ܐܵܘ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܐܵܘ ܕܲܒ݂ܡܲܚܲܬ݁ܬܹ݁ܗ ܐܵܘ ܕܒܲܨܠܝܼܒܹܗ: ܩܲܕ݁ܝܼܫܵܐ (ܐܵܘ ܩܲܕܝܼܫܬܵܐ) ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܐܲܦܨܲܚ ܠܲܫܡܲܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܢܟ݂܀ ܗ̇ܘ ܕܠܹܗ ܗܘ ܟܝܵܢܵܐܝܼܬ݂ ܐܝܼܬ݂ ܠܵܐ ܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܘܝܲܬ݁ܝܼܪܵܐܝܼܬ݂ ܕܐܲܝܠܹܝܢ ܕܲܡܗܲܝܡܢܝܼܢ ܒܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܣܓܵܐ ܥܲܡܲܢ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀
 ܘܡܵܐ ܕܫܸܠܡܲܬ݀ ܡܲܩܸܦ: ܢܲܦ̮̈ܫܵܬܲܢ ܘܕܲܚܕ݂ܵܕܹ̈ܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܲܓ݂ܥܸܠ.
ܘܥܵܢܹܝ̇ܢ: ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ.
	ܘܲܡܨܲܠܸܐ ܟܵܗܢܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ. ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܡܲܓ݂ܥܠܝܼܢܲܢ ܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ: ܘܡܢܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܫܵܐ̇ܠܝܼܢܲܢ ܫܘܼܒ݂ܩܵܢܵܐ ܕܚܵܘܒܹ̈ܐ ܘܕܲܚܛܵܗܹ̈ܐ: ܗܲܒ݂ܠܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܪ̈ܲܚܡܲܝܟ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ ܟܵܪܘܿܙܘܼܬ݂ܵܐ: ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ.
ܘܐܵܡ̇ܪܝܼܢ ܒܩܵܠܵܐ ܦܫܝܼܛܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܫܲܒܲܚ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܥܵܠܲܡ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܒ݂ܵܬܲܪ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܪ̈ܵܝܹܐ. ܩܲܕ݁ܝܼܫܵܐ ܘܲܫܒ݂ܝܼܚܵܐ ܘܚܲܝܠܬ݂ܵܢܵܐ ܘܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܕܲܒ݂ܩܲܕ݁ܝܼܫܹ̈ܐ ܥܵܡܲܪ ܘܡܸܬ݁ܬ݁ܢܝܼܚ ܨܸܒ݂ܝܵܢܹܗ: ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ ܐܸܬ݂ܦܲܢ ܡܵܪܝ: ܘܚܘܼܣ ܘܪܲܚܸܡ ܥܠܲܝܢ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ. ܒܲܪܸܟ݂ ܡܵܪܝ: ܐܲܪܟܸ݁ܢܘ ܪܹ̈ܫܲܝܟ݁ܘܿܢ ܠܲܣܝܵܡܝܼܕ݂ܵܐ ܘܩܲܒܸ݁ܠܘ ܒܘܼܪܟ݁ܬ݂ܵܐ.
ܘܐܵܡܲܪ ܪܲܒܵܐ. ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܬܸܫܡܸܫܬܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.

      ܥܘܿܢܝܼܬ݂ܵܐ ܕܪܲܡܫܵܐ. ܒܩܵܠ ܒܲܒ݂ܛܝܼܠܘܼܬܹܗ. ܐܲܠܵܗܵܐ ܛܥܲܝܬܵܢ ܘܲܕ݂ܚܲܩܬܵܢ ܘܲܪܓܸ݁ܙܬ݁ ܥܠܲܝܢ: ܡܸܛܠ ܕܚܲܕ݂ܪ̈ܵܢܝ ܒܝܼܫܵܬ݂ܵܐ ܕܠܵܐ ܡܸܢܝܵܢ. ܡܸܛܠ ܒܝܼܫܘܼܬ݂ ܥܒ݂ܵܕܲܝ̈ܢ ܣܓ݂ܝܼ ܘܲܥܫܸܢ ܡܲܚ̈ܘܵܬܲܢ ܕܠܵܐ ܡܸܢܝܵܢܵܐ. ܘܠܸܒܲܢ ܠܵܐ ܙܵܥ ܡܢ ܫܲܒ݂ܛܵܐ ܕܪܘܼܓ݂ܙܵܐ. ܘܵܝܠܲܢ ܕܹܝܢ ܕܕܲܠܡܵܐ ܒܲܢ ܐܸܫܬܲܠܡܲܬ݀. ܚܲܪܬܹܗ ܕܥܵܠܡܵܐ ܘܫܘܼܠܵܡܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ. ܕܗܵܐ ܡܢ ܟܲܕ݁ܘܼ ܛܠܸܩ ܚܲܝܲܝ̈ܢ. ܘܲܓ݂ܡܲܪ ܘܐܵܘܦ̮ܝܼ ܡܵܙܘܿܢܲܢ. ܘܲܒ݂ܛܸܠ ܡܸܢܲܢ ܣܲܒ݂ܪܲܢ. ܒܪܝܼܟ݂ ܕܲܝ̇ܵܢܵܐ ܕܟܹܐܢܵܐ ܗ݇ܝܼ ܡܲܪܕܘܼܬܹܗ. ܘܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܹܗ ܡܪܲܚܸܡ ܥܲܠ ܟܠܵܐ. ܡܵܪܵܐ ܕܨܵܒܹ݁ܐ ܒܬܲܝܵܒܹ̈ܐ ܚܘܼܣ ܥܠܲܝܢ: ܬܢܝܼ. ܘܫܲܒܲܚ. ܒܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܩܵܢܘܿܢܵܐ. ܚܘܼܣ ܥܠܲܝܢ ܐܲܢ݇ܬ݁ ܚܲܢܵܢܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܸܬ݂ܦܲܢ ܥܠܲܝܢ ܡܠܸܐ ܪ̈ܲܚܡܹܐ: ܘܠܵܐ ܡܵܪܝ ܬܲܗܡܸܐ ܚܝܵܪܵܟ݂ ܘܲܒ݂ܛܝܼܠܘܼܬ݂ܵܟ݂ ܡܢܢ: ܕܲܥܠܲܝܟ݁ܘܼ ܣܲܒ݂ܪܲܢ ܘܬ݂ܘܼܟ݂ܠܵܢܲܢ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ: ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	      ܐܵܬ݂ܘ̈ܵܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܒ݂ܵܬܲܪ (ܡܙ: ܩܝܚ: ܣ ، ܥ). ܣܢܹܝܬ݂ ܠܥܵܘ̈ܵܠܹܐ ܘܢܵܡܘܿܣܵܟ݂ ܪܸܚܡܹܬ݂* ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܣܸܬܵܪܝ ܘܒܹܝܬ݂ ܓܵܘܣܝ ܐܲܢ݇ܬ݁ܘܼ ܘܲܠܡܸܠܬ݂ܵܟ݂ ܣܲܒ݁ܪܹܬ݂* ܥܒܲܪܘ ܡܹܢܝ ܥܵܘ̈ܵܠܹܐ ܕܐܸܛܲܪ ܦܘܼܩܕܵܢܵܘ̈ܗܝ ܕܐܲܠܵܗܝ* ܐܲܫܲܪܲܝܢܝ ܒܡܸܠܬ݂ܵܟ݂ ܘܐܸܚܸܐ ܘܠܵܐ ܬܲܒ݂ܗܸܬܲܝܢܝ ܡܢ ܣܲܒ݂ܪܝ* ܣܲܝܲܥܲܝܢܝ ܘܐܸܬ݂ܦܪܸܩ ܘܐܸܬ݂ܝܲܠܲܦ ܒܦܘܼܩܕܵܢܲܝ̈ܟ* ܒܟ݂ܠܙܒܲܢ ܐܲܣܠܝܼܬ݂ ܠܟ݂ܠ ܕܛܵܥܹܝܢ ܡܸܢܵܟ݂ ܡܸܛܠ ܕܥܵܘܠܵܐ ܗ݇ܘ ܪܸܢܝܼܗܘܿܢ* ܣܲܡܸܟܲܝܢܝ ܘܐܸܬ݂ܦܪܸܩ ܘܐܸܪܢܸܐ ܒܟ݂ܠܙܒܲܢ ܒܦܘܼܩܕܵܢܲܝ̈ܟ* ܩܦܘܿܕ݂ ܒܸܣܪܝ ܡܢ ܕܸܚܠܬ݂ܵܟ݂ ܘܡܢ ܕܝܼܢܲܝ̈ܟ ܕܸܚܠܹܬ݂* ܥܵܒܹ݁ܕ݂ ܕܝܼܢܵܐ ܘܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ ܠܵܐ ܬܸܫܒ݁ܘܿܩܲܝܢܝ ܒܐܝܼܕ݂ܵܐ ܕܛܵܠܘܿܡܲܝ̈* ܒܲܣܸܡܲܝܗܝ ܠܥܲܒ݂ܕܵܟ݂ ܒܛܵܒ݂̈ܵܬ݂ܵܐ ܘܠܵܐ ܢܸܥܫܩܘܼܢܵܢܝ ܫܲܒ݂ܗܪ̈ܵܢܹܐ* ܥܲܝܢܲܝ̈ ܣܲܟ݁ܝܼ ܠܦܘܼܪܩܵܢܵܟ݂ ܘܲܠܡܹܐܡܪܵܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܟ݂* ܥܒܸܕ݂ ܠܥܲܒ݂ܕܵܟ݂ ܐܲܝܟ݂ ܪ̈ܲܚܡܲܝܟ ܘܐܲܠܸܦܲܝܢܝ ܢܵܡܘܿܣܵܟ݂* ܥܲܒ݂ܕܵܟ݂ ܐ݇ܢܵܐ ܣܲܟܸ݁ܠܲܝܢܝ ܘܐܸܕܲܥ ܣܵܗܕ݁ܘܼܬ݂ܵܟ݂* ܙܲܒ݂ܢܵܐ ܗ݇ܘ ܠܡܸܦܠܚܹܗ ܠܡܵܪܝܵܐ ܘܗܵܐ ܒܲܛܸܠܘ ܢܵܡܘܿܣܵܟ݂* ܡܸܛܠ ܗܵܢܵܐ ܪܸܚܡܹܬ݂ ܦܘܼܩܕܵܢܲܝ̈ܟ ܛܵܒ݂ ܡܸܢ ܕܲܗܒ݂ܵܐ ܘܛܵܒ݂ ܡܢ ܟܹܐܦܹ̈ܐ ܛܵܒ݂̈ܵܬ݂ܵܐ* ܟܠܗܘܿܢ ܦܘܼܩܕܵܢܲܝ̈ܟ ܪܸܚܡܹܬ݂ ܘܟ݂ܠܵܗ̇ ܐܘܼܪܚܵܐ ܕܥܵܘ̈ܵܠܹܐ ܣܢܹܝܬ݂* ܫܲܒܲܚ* ܥܵܠܲܡ*
	ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܬܲܪ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
      ܕܒܹܝܡ. ܒܪܝܼܟ݂ ܕܐܲܫܠܸܡ ܠܵܗ ܠܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ. ܐ݇ܪܵܙܵܐ ܕܡܵܘܬܹ݁ܗ ܘܲܕ݂ܢܘܼܚܵܡܹܗ. ܒܵܬܹ̈ܐ. ܟܵܗܢܵܐ ܘܟ݂ܘܼܡܪܵܐ ܕܟܲܗܸܢ ܒܲܕ݂ܡܹܗ. ܒܵܟ݂ ܢܸܬ݂ܐܲܣܘܿܢ ܡܘܼܡܹ̈ܐ ܕܢܲܦ̮ܫܲܢ.
      ܣܘܼܒܵܥܵܐ. ܛܘܼܒ݂ܵܘܗܝ ܠܓܲܒ݂ܪܵܐ ܕܒ݂ܐܘܼܪܚܵܐ. ܐ: ܗܘܼܠܵܠܵܐ ܥܲܡ ܨܠܘܿܬܹܗ. ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܐ‐ܝ
ܘܐܵܡܪܝܼܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	 ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܕܣܘܼܒܵܥܵܐ ܒܓ݂ܘܼܕܵܐ ܕܒ݂ܵܬܲܪ. ܐܲܫܘܵܐ ܠܲܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܠܪܲܡܫܵܐ ܕܫܲܝܢܵܐ: ܘܲܠܠܸܠܝܵܐ ܕܲܢܝܵܚܵܐ: ܘܲܠܨܲܦܪܵܐ ܕܲܡܣܲܒ݁ܪ̈ܵܢ ܒܹܗ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܘܠܐܝܼܡܵܡܵܐ ܕܲܥܒ݂ܵܕܹ̈ܐ ܛܵܒܹ̈ܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܕܲܒ݂ܗܘܿܢ ܢܪܲܥܸܐ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
      ܥܘܿܢ̈ܝܵܬ݂ܵܐ. ܒܛܘܼܒܹܝ̈ܗ ܠܪܘܼܚܵܟ݂. ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܒܩܘܼܕ݂ܫܹܗ: ܘܲܚܢܲܢ ܩܵܡܢ ܘܐܸܬ݁ܛܲܝܲܒ݂ܢ. ܡܵܐ ܕܲܫܡܲܥܢܲܢ ܩܵܠܹ̈ܐ ܕܚܝܼ̈ܠܹܐ. ܕܐ݇ܪ̈ܵܙܵܘܗܝ ܩܲܕ݁ܝܼܫܹ̈ܐ ܕܲܡܫܝܼܚܵܐ. ܒܗܵܘ ܥܸܕܵܢܵܐ ܕܡܲܣܸܩ ܟܵܗܢܵܐ ܩܘܼܪܒܵܢܵܐ ܠܐܲܠܵܗܵܐ. ܢܸܒ݂ܪܘܼܟ݂ ܟܠܲܢ ܒܕܸܚܠܬ݂ܵܐ ܩܕ݂ܵܡ ܡܵܪܝܵܐ. ܘܢܹܐܡܲܪ ܐܸܬ݂ܪܲܥܸܐ ܠܥܲܒ݂ܕܲܝ̈ܟ: ܬܢܝܼ. ܗܘܝܘܼ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ. ܡܫܝܼܚܵܐ ܕܲܙܟ݂ܵܐ ܒܝܲܕ݂ ܨܵܘܡܹܗ. ܠܕܲܪܹ̈ܐ ܬܠܵܬ݂ܵܐ ܘܐܸܬ݂ܢܲܨܲܚ. ܪܸܓ݁ܬ݂ܵܐ ܘܫܘܼܒ݂ܚܵܐ ܘܝܲܥܢܘܼܬ݂ܵܐ. ܕܠܵܐܕ݂ܵܡ ܐܲܦܩܲܬ݀ ܡܢ ܓܲܢܬ݂ܵܐ. ܗܲܒ݂ܠܲܢ ܚܲܝܠܵܐ ܕܲܒ݂ܨܵܘܡܵܐ ܢܸܫܦܲܪ ܠܲܫܡܵܟ݂. ܒܕ݂ܘܼܒܵܪܹ̈ܐ ܕܲܡܪܲܥܹܝܢ ܠܵܟ݂. ܬܵܘ ܒܢܲܝ̈ܵܐ ܫܘܼܡܥܘܼܢܝ. ܒܨܵܘܡܵܐ ܢܸܫܒ݁ܘܿܩ ܠܚܲܝ̇ܵܒܲܝ̈ܢ. ܕܢܸܫܒ݁ܘܿܩ ܠܲܢ ܡܵܪܝܵܐ ܚܵܘܒܲܝ̈ܢ. ܘܢܸܬܸ݁ܠ ܠܲܚܡܲܢ ܠܡܸܣܟܹ݁ܢܹ̈ܐ. ܐܲܝܟ݂ ܕܐܲܠܸܦ ܠܲܢ ܐܹܫܲܥܝܵܐ. ܘܲܒ݂ܙܸܕ݂̈ܩܵܬ݂ܵܐ ܘܒܲܨܠܵܘ̈ܵܬ݂ܵܐ ܬܟ݂ܝܼ̈ܒ݂ܵܬ݂ܵܐ. ܢܸܗܘܸܐ ܬܲܠܡܝܼܕܹ̈ܐ ܠܡܵܪܲܢ. ܕܲܨܠܘܿܬ݂ܵܐ. ܒܟ݂ܘܼܪܗܵܢܵܐ ܕܥܲܣܩܵܐ. ܫܘܼܚܠܵܦܵܐ: ܗ. ܩܵܠܵܐ: ܟܟ. ܒܪܘܼܟܵܒ݂ ܡܵܐ ܕܲܫܡܲܥܢܲܢ.
ܐܲܣܵܢܝ ܡܵܪܝܵܐ ܡܸܛܠ ܕܙܵܥܘ ܓܲܪ̈ܡܲܝ. ܒܟ݂ܘܼܪܗܵܢܵܐ ܥܲܣܩܵܐ ܕܦܲܓ݂ܪܵܐ. ܡܸܬ݂ܢܵܘܲܠ ܐ݇ܢܵܐ ܡܵܪܝܵܐ. ܐܲܣܵܐ ܠܝܼ ܟܹܐܒ݂ܝ ܒܲܚܢܵܢܵܟ݂. ܐܲܝܟ݂ ܕܲܠܟܹܐܢܵܐ ܚܹܙܲܩܝܵܐ. ܕܒܲܠܬ݂ܵܐ ܕܬܹܐܢܹ̈ܐ ܣܵܡ ܠܹܗ ܢܒ݂ܝܼܵܐ ܘܐܸܬ݂ܐܲܣܝܼ. ܘܠܝܼ ܡܵܪܝ ܣܝܼܡ ܣܲܡܵܐ ܕܪ̈ܲܚܡܲܝܟ. ܐܲܣܵܐ ܬܒ݂ܵܪܹܗ ܕܠܹܒ݁ܝ ܘܲܥܨܘܿܒ݂. ܒܟ݂ܘܼܪܗܵܢܵܐ ܥܲܣܩܵܐ ܪܡܸܐ ܐ݇ܢܵܐ. ܘܩܲܫܝܵܐܝܼܬ݂ ܡܸܬ݂ܢܵܘܲܠ ܐ݇ܢܵܐ. ܐܲܚܠܸܡܲܝܢܝ ܐܲܝܟ݂ ܚܹܙܲܩܝܵܐ. ܕܐܸܦܪܘܿܥ ܫܘܼܒ݂ܚܵܐ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܡܵܪܵܐ ܛܵܒ݂ܵܐ ܣܝܼܡ ܠܵܗ̇ ܥܨܵܒ݂ܵܐ ܠܲܟ݂ܪܝܼܗܘܼܬ݂ܝ. ܒܪ̈ܲܚܡܵܘܗܝ ܕܲܒ݂ܪܵܟ݂ ܚܲܒ݁ܝܼܒ݂ܵܐ. ܐܲܣܵܢܝ ܡܵܪܝܵܐ ܘܐܸܬ݂ܐܲܣܸܐ. ܒܟ݂ܘܼܪܗܵܢܵܐ ܥܲܣܩܵܐ ܪܡܸܐ ܐ݇ܢܵܐ. ܘܲܠܐܸܐ ܐ݇ܢܵܐ ܘܡܸܬ݁ܛܲܪܲܦ ܐ݇ܢܵܐ. ܘܠܲܝܬ݁ ܐܵܣܝܵܐ ܕܲܡܨܸܐ ܕܢܲܐܣܹܝܢܝ. ܐܸܠܵܐ ܐܸܢ ܐܲܢ݇ܬ݁ ܡܵܪܹܟ݁ܠ. ܐܲܣܵܐ ܟܹܐܒܲܝ̈ ܐܲܝܟ݂ ܕܐܲܣܝܼܬ݁ ܠܲܡܡܲܚܲܝܬܵܐ. ܕܩܸܪܒܲܬ݀ ܨܹܐܕܲܝܟ ܘܐܸܬ݂ܐܲܣܝܲܬ݀. ܥܢܝܼܢܝ ܡܵܪܝܵܐ ܡܸܛܠ ܕܛܵܒ݂ܵܐ ܗ݇ܝ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܗܵܐ ܟܪܝܼܗܝܼܢܲܢ ܘܲܪܡܹܝܢܲܢ. ܒܲܬ݂ܪܲܥ ܪ̈ܲܚܡܲܝܟ ܡܵܪܹܟ݁ܠ. ܡܫܝܼܚܵܐ ܐܵܣܝܵܐ ܕܐܸܫܬܲܕܲܪ. ܡܢ ܐܲܒ݂ܵܐ ܕܢܲܐܣܸܐ ܟܹܐܒܲܝ̈ܢ. ܚܘܼܪ ܡܵܪܝ ܘܲܚܙܝܼ ܕܲܣܪ̈ܝܼ ܘܐܸܬ݂ܡܣܝܼ̈ ܚܒ݂ܵܪ̈ܵܬܲܢ. ܘܐܲܚܠܸܡܲܝܢܝ ܒܣܲܡܵܐ ܕܪ̈ܲܚܡܲܝܟ. ܘܐܸܡܲܪ ܦܸܠ̈ܐܵܬ݂ܵܐ ܕܡܢ ܩܕ݂ܝܼܡ. ܒܐܲܪܡܲܠܬܵܐ ܡܬ݂ܝܼܠܝܼܢܲܢ ܠܵܟ݂. ܘܒܲܒ݂ܪܵܐ ܕܦܲܪܲܚ ܩܸܢܝܵܢܵܘ̈ܗܝ. ܫܲܠܸܡ ܨܹܐܕܲܝܢ ܒܲܥܒ݂ܵܕ݂ܵܐ. ܦܸܠܐܬ݂ܵܐ ܕܪܲܟܸ݁ܒ݂ ܚܘܼܒܵܟ݂. ܘܢܸܫܡܲܥ ܡܸܢܵܟ݂ ܕܗܵܢ ܒܹܪܝ ܡܝܼܬ݂ܵܐ ܗ݇ܘܵܐ ܘܲܚܝܵܐ. ܘܢܸܚܕܘܿܢ ܥܝܼܪܹ̈ܐ ܒܦܘܼܢܵܝܲܢ. ܐܸܬ݂ܦܲܢ ܥܠܲܝ ܘܪܲܚܸܡܥܠܲܝ. ܡܚܝܼܠܘܼ ܟܝܵܢܲܢ ܚܲܠܵܫܵܐ. ܘܲܢܩܸܐ ܠܟܹܐܒܹ̈ܐ ܘܟ݂ܘܼܪ̈ܗܵܢܹܐ. ܘܐܸܢ ܗܘ ܕܣܲܡܵܐ ܕܲܚܢܵܢܵܟ݂. ܠܵܐ ܡܲܐܣܸܐ ܠܲܚܒ݂ܵܪ̈ܵܬܲܢ. ܡܢ ܫܹܠܝ ܛܵܠܩܝܼܢ. ܘܡܵܘܦܹܝܢ ܚܲܝܲܝ̈ܢ ܡܢ ܐܲܪܥܵܐ. ܐܲܢ݇ܬ݁ ܡܵܪܝ ܗܘܝܼ ܐܵܣܝܵܐ ܠܟܹܐܒܲܝ̈ܢ. ܩܘܼܪܒܵܢܵܐ ܕܐܝܼܕܲܝ̈ ܐܲܝܟ݂ ܩܘܼܪܒܵܢܹܗ ܕܪܲܡܫܵܐ. ܐܲܝܟ݂ ܦܝܼܪܡܵܐ ܕܩܲܪܸܒ݂ ܐܲܗܪܘܿܢ. ܪܹܝܚܹܗ ܕܟܸܢܫܲܢ ܢܸܒ݂ܣܲܡ ܠܵܟ݂. ܘܐܲܝܟ݂ ܒܵܥܘܼܬ݂ܵܐ ܕܢܝܼܢܘܵܝܹ̈ܐ. ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܩܲܒܸ݁ܠ ܡܵܪܝ. ܘܐܲܝܟ݂ ܕܲܥܢܲܝܬܵܝܗܝ ܠܕ݂ܵܢܝܼܐܹܝܠ ܡܢ ܓܵܘ ܓܘܼܒܵܐ. ܥܢܝܼܢ ܡܵܪܝ ܘܥܲܕܲܪܲܝܢ ܒܪ̈ܲܚܡܲܝܟ. ܣܲܒ݂ܪܵܐ ܕܟ݂ܠܗܘܿܢ ܣܵܘܦܹܝ̈ܗ̇ ܕܐܲܪܥܵܐ. ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ. ܕܐܸܬ݂ܵܐ ܡܸܛܠ ܦܘܼܪܩܵܢܲܢ. ܓܡܘܿܪ ܒܲܢ ܢܝܼܫܵܐ ܕܡܹܐܬ݂ܝܼܬ݂ܵܟ݂. ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܹܗ ܕܫܵܠܘܿܚܵܟ݂ ܘܒ݂ܵܟ݂ ܢܸܬ݂ܚܵܘܲܪ ܡܢ ܟܘܼܬ݂̈ܡܵܬ݂ܵܐ ܕܲܚܛܝܼܬ݂ܵܐ. ܕܣܵܡ ܒܲܢ ܒܝܼܫܵܐ ܒܥܸܠܕܲܪܲܢ. ܡܪܲܚܡܵܢܲܬ݁ ܡܵܪܝܵܐ ܘܙܲܕܝܼܩ. ܡܫܝܼܚܵܐ ܐܵܣܝܵܐ ܫܲܪܝܼܪܵܐ. ܕܠܲܚܡܵܬܹܗ ܕܫܸܡܥܘܿܢ ܐܲܣܝܼ. ܡܢ ܟܢܲܥܢܵܝܬܵܐ ܪܕܲܦ ܫܹܐܕ݂ܵܐ. ܘܠܲܡܡܲܚܲܝܬܵܐ ܝܲܗ݇ܒ݂ ܚܲܝܹ̈ܐ. ܒܚܲܝܠܵܟ݂ ܪܲܒܵܐ ܟܹܐܒܹ̈ܐ ܕܦܲܓ݂ܪ̈ܲܝܢ ܢܸܬ݂ܐܲܣܘܿܢ. ܕܢܵܘܕܸܐ ܠܲܫܡܵܟ݂ ܩܲܕܝܼܫܵܐ. ܕܐܸܥܘܿܠ ܒܗܘܿܢ ܘܐܵܘܕܸܐ ܠܡܵܪܝܵܐ. ܒܲܬ݂ܪܲܥ ܪ̈ܲܚܡܲܝܟ ܡܵܪܹܟ݁ܠ. ܢܵܩ̇ܫܝܼܢ ܥܲܒ݂ܕܲܝ̈ܟ ܕܐܲܠܝܼܨܝܼܢ. ܘܡܢ ܓܲܙܵܐ ܕܒܲܣܝܼܡܘܼܬ݂ܵܟ݂. ܫܵܐ̇ܠܝܼܢ ܪ̈ܲܚܡܹܐ ܘܦܘܼܪܩܵܢܵܐ. ܫܡܲܥ ܡܵܪܝ ܒܩܵܠܲܝ̈ܢ. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ. ܛܵܒ݂ܵܐ ܕܠܵܐ ܟܵܠܹܐ ܪ̈ܲܚܡܵܘܗܝ. ܚܵܘܝܼܬ݁ ܒܥܲܡܡܹ̈ܐ ܥܘܼܫܢܵܟ݂. ܠܢܲܥܡܵܢ ܓܲܪܒܵܐ ܕܲܟ݁ܝܼ ܗ݇ܘܵܐ. ܐܹܠܝܼܫܲܥ ܗܵܘ̇ ܒܲܪ ܫܵܦܵܛ. ܘܠܲܚܒ݂ܵܪ̈ܵܬܹܗ ܕܲܟ݂ܝܵܢܲܢ. ܝܼܫܘܿܥ ܒܡܵܘܬܵܟ݂ ܐܲܢ݇ܬ݁ ܐܲܣܝܼܬ݁. ܘܠܲܡܚܝܼܠܘܼܬܲܢ ܐܵܘܫܸܛ ܐܝܼܕ݂ܵܐ ܕܥܘܼܕ݂ܪ̈ܵܢܹܐ. ܡܵܪܲܢ ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ. ܐܸܬ݁ܕܲܟ݂ܪ ܡܵܪܝܵܐ ܪ̈ܲܚܡܲܝܟ ܕܡܢ ܥܵܠܲܡ ܘܛܲܝܒ̈ܘܵܬ݂ܵܟ݂. ܡܢ ܦܘܼܡܵܟ݂ ܡܵܪܝ ܩܲܕܝܼܫܵܐ. ܫܸܡܥܹ̇ܬ݂ ܕܐܸܡܲܪܬ݁ ܒܲܣܒܲܪܬ݂ܵܟ݂. ܕܩܵܪܹܐ ܠܝܼ ܥܵܢܹ̇ܐ ܐ݇ܢܵܐ ܠܹܗ. ܘܲܕ݂ܢܵܩܹ̇ܫ ܦܵܬܲܚ ܐ݇ܢܵܐ ܠܹܗ. ܦܬܲܚ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܡܫܝܼܚܵܐ ܡܲܠܟܲܢ ܘܡܲܚܝܵܢܲܢ. ܘܦܲܪܢܸܣ ܥܘܼܬ݂ܪܵܟ݂ ܠܲܣܢܝܼܩܹ̈ܐ. ܐܲܣܵܐ ܬܒ݂ܵܪܹܗ ܕܠܹܒ݁ܝ ܘܲܥܨܘܿܒ݂. ܡܫܝܼܚܵܐ ܐܵܣܝܵܐ ܫܲܪܝܼܪܵܐ. ܐܲܣܵܐ ܟܹܐܒܲܝ̈ܢ ܒܲܚܢܵܢܵܟ݂. ܒܡܸܠܬ݂ܵܟ݂ ܚܲܝܸܨܬ݁ ܡܫܲܪ̈ܲܝܵܐ. ܘܲܪܕܲܦ̮ܬ݁ ܟܹܐܒܹ̈ܐ ܒܦܘܼܩܕܵܢܵܟ݂. ܘܠܲܢ ܚܲܛܵܝܹ̈ܐ ܕܲܒ݂ܟܹܐܒ݂ ܦܲܓ݂ܪܵܐ ܡܢܵܘܠܝܼܢܲܢ. ܐܲܚܠܸܡܲܝܢ ܕܢܵܘܕܸܐ ܠܲܫܡܵܟ݂. ܡܵܪܝܵܐ ܐܲܢ݇ܬ݁ ܝܵܕܲܥ ܐܲܢ݇ܬ݁. ܟܘܼܪ̈ܗܵܢܲܝܢ ܠܵܐ ܟܣܹܝܢ ܡܸܢܵܟ݂. ܡܫܝܼܚܵܐ ܐܵܣܝܵܐ ܫܲܪܝܼܪܵܐ. ܘܲܚܢܵܢܵܟ݂ ܡܲܐܣܸܐ ܡܲܓܵܢ. ܠܲܕ݂ܫܵܐ̇ܠܝܼܢ ܒܗܲܝܡܵܢܘܼܬ݂ܵܐ. ܡܲܒ݁ܘܼܥ ܚܲܝܹ̈ܐ ܕܢܵܒ݂ܥܝܼܢ ܡܸܢܹܗ ܥܘܼܕ݂ܪ̈ܵܢܹܐ. ܥܢܝܼ ܠܣܵܓ݂ܘܕܲܝ̈ܟ ܕܩ̇ܵܪܹܝܢ ܠܵܟ݂. ܪܥܝܼ ܐܸܢܘܿܢ ܘܕܲܒܲܪ ܐܸܢܘܿܢ ܥܕܲܡܵܐ ܠܥܵܠܲܡ. ܪܵܥܝܵܐ ܪܲܒܵܐ ܕܡܲܪܥܝܼܬ݂ܵܐ ܕܟܲܢܸܫ ܥܸܪ̈ܒܹܐ ܐܲܒ݁ܝܼ̈ܕܹܐ. ܐܲܦܢܵܢ ܨܹܐܕܲܝܟ ܒܲܚܢܵܢܵܟ݂. ܡܢ ܦܸܗܝܵܐ ܕܩܵܛܹܠ ܐ݇ܢܵܫܵܐ. ܘܢܸܫܬ݁ܟܲܚ ܙܘܼܙܹܗ ܕܲܩܢܘܿܡܲܢ ܕܲܛܒ݂ܝܼܥ ܒܲܫܡܵܟ݂. ܒܲܫܪܵܓ݂ܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬ݂ܵܟ݂. ܡܲܐܣܸܐ ܠܲܬ݂ܒ݂ܝܼܪ̈ܲܝ ܠܸܒܵܐ ܘܥܵܨܹܒ݂ ܟܹܐܒܲܝ̈ܗܘܿܢ. ܡܫܝܼܚܵܐ ܕܐܲܣܝܼ ܠܲܡܫܲܪܝܵܐ ܘܲܠܒܲܪܬ݂ܵܗ̇ ܕܲܟ݂ܢܲܥܢܵܝܬܵܐ. ܘܠܲܡܡܲܚܲܝܬܵܐ ܕܲܥܒܲܪ ܗ݇ܘܵܐ. ܟܘܼܪܗܵܢܵܗ̇ ܠܐܵܣܵܘ̈ܵܬ݂ܵܐ. ܐܲܣܵܐ ܟܹܐܒܲܝ̈ܢ ܘܟ݂ܘܼܪ̈ܗܵܢܲܝܢ ܕܲܡܫܲܢܩܝܼܢ ܠܲܢ. ܐܵܣܝܵܐ ܡܲܐܣܸܐ ܟܠ ܟܹܐܒ݂ܝܼ̈ܢ. ܐܲܢܗܲܪ ܥܲܝܢܲܝ̈ ܕܠܵܐ ܐܸܕ݂ܡܲܟ݂ ܠܡܵܘܬܵܐ. ܡܫܝܼܚܵܐ ܕܐܲܢܗܲܪ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܒܢܘܼܗܪܵܐ ܫܒ݂ܝܼܚܵܐ ܕܓܸܠܝܵܢܹܗ. ܐܲܢܗܲܪ ܙܵܘܥܲܝ̈ܢ ܒܲܫܪܵܪܵܟ݂. ܕܢܸܣܬܲܟ݁ܠܘܼܢ ܐ݇ܪܵܙ ܡܹܐܬ݂ܝܼܬ݂ܵܟ݂. ܘܢܸܗܘܘܿܢ ܪܸ̈ܓ݂ܫܲܝܢ ܡܲܣܩܝܼܢ ܟܠܫܵܥ ܡܲܕܲܐܬ݂ܵܐ. ܬܵܘܕܝܼܬ݂ܵܐ ܕܦܲܐܝܵܐ ܠܲܫܡܵܟ݂. ܡܸܛܠ ܕܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܛܵܒ݂ ܐܲܢ݇ܬ݁. ܛܵܒ݂ܵܐ ܩܲܒܸ݁ܠ ܒܵܥܘܼܬܲܢ. ܘܲܒ݂ܪܹܗ ܕܛܵܒ݂ܵܐ ܬܲܟ݂ܫܲܦܬܲܢ. ܕܐܲܢ݇ܬ݁ܘܼ ܣܲܒ݂ܪܵܐ ܕܬܲܝܵܒܹ̈ܐ. ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܕܟ݂ܠ ܕܐܲܠܝܼܨܝܼܢ. ܦܬܲܚ ܠܵܗ̇ ܡܵܪܲܢ. ܬܲܪܥܵܐ ܕܪ̈ܲܚܡܹܐ ܠܲܨܠܘܿܬܲܢ. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ. ܕܡܲܪܝܲܡ܀ ܐܸܒ݂ܥܸܐ ܒܛܵܒ݂ܵܬܹ̈ܟ݂ܝ. ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܩܲܕܝܼܫܬܵܐ. ܐܲܦܝܼܣܝ ܘܲܒ݂ܥܵܝ ܠܲܡܫܝܼܚܵܐ. ܕܢܸܥܒܸ݁ܕ݂ ܪ̈ܲܚܡܹܐ ܥܲܠ ܥܵܠܡܵܐ. ܕܐܲܚܝܼܕ݂ ܓܵܘܣܵܐ ܒܲܨܠܘܿܬܹܟ݂ܝ. ܘܥܹܕܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇ ܬܸܬ݂ܢܛܲܪ ܡܢ ܢܸܟ݂ܝܵܢܵܘ̈ܗܝ. ܕܐܵܟܹܠܩܲܪܨܵܐ ܡܵܪܘܿܕ݂ܵܐ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܛܲܪ ܪ̈ܲܚܡܲܝܟ ܠܩܲܪ̈ܝܼܒܲܝܟ. ܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܡܪ̈ܲܥܝܵܢܲܝܟ. ܘܙܲܕܝܼܩܹ̈ܐ ܕܲܫܦܲܪܘ ܩܘܼܕ݂ܡܲܝܟ. ܢܒ݂ܝܼܹ̈ܐ ܫܠܝܼܚܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ. ܘܣܵܗܕܹ̈ܐ ܘܟ݂ܵܗܢܹ̈ܐ ܘܥܵܢܘܵܝܹ̈ܐ. ܢܲܛܲܪ ܡܵܪܲܢ ܠܟܸܢܫܵܐ ܟܠܹܗ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܡܢ ܢܸܟ݂ܝܵܢܵܘ̈ܗܝ ܕܐܵܟܹܠܩܲܪܨܵܐ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܡܸܛܠ ܕܡܲܒ݁ܘܼܥܵܐ ܕܚܲܝܹ̈ܐ ܥܲܡܵܟ݂ ܗ݇ܘ. ܨܠܝܼܒ݂ܵܐ ܢܸܒ݂ܥܵܐ ܕܥܘܼܕ݂ܪ̈ܵܢܹܐ. ܢܲܛܲܪ ܠܥܹܕܬܵܐ ܘܲܠܝܲܠܕܹܝ̈ܗ̇. ܡܢ ܒܝܼܫܵܐ ܘܚܲܝ̈ܠܵܘܵܬܹܗ. ܕܡܲܩܪܸܒ݂ ܥܲܡܵܗ̇ ܒܟ݂ܠܥܸܕܵܢ. ܘܲܦܩܘܿܕ݂ ܫܲܝܢܵܐ ܒܥܵܠܡܵܐ ܟܠܹܗ ܘܥܵܡܘܿܪ̈ܵܘܗܝ. ܘܲܬ݂ܚܹܝܬ݂ ܟܹܢܦܲܝ̈ܟ ܢܸܣܬܲܬܲܪ. ܕܐܲܒ݂ܘܼܢ܀ ܛܘܼܒܲܝܟ ܘܛܵܒ݂ܬ݂ܵܐ ܠܢܲܦ̮ܫܵܟ݂. ܛܘܼܒܲܝܟ ܐܲܒ݂ܘܼܢ ܩܲܕܝܼܫܵܐ. ܒܹܝܬ݂ ܓܵܘܣܵܐ ܕܟ݂ܠ ܕܐܲܠܝܼܨܝܼܢ. ܘܡܲܒ݁ܘܼܥ ܫܸܩܝܵܐ ܪܘܼܚܵܢܵܐ. ܕܲܣܢܝܼܩܹ̈ܐ ܒܹܗ ܡܸܬ݂ܦܲܪܩܝܼܢ. ܨܠܘܿܬ݂ܵܟ݂ ܬܸܗܘܸܐ ܫܘܼܪܵܐ ܪܵܡܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ. ܠܟܸܢܫܵܐ ܕܙܲܝܲܚ ܕܘܼܟ݂ܪܵܢܵܟ݂. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܠܵܐ ܬܸܫܕܹܝܢܝ ܡܢ ܩܕ݂ܵܡܲܝܟ. ܒܗܵܘ̇ ܒܹܝܬ݂ ܕܝܼܢܵܐ ܕܟܹܐܢܘܼܬ݂ܵܟ݂. ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܘܦܵܪܘܿܩܲܢ. ܚܲܣܵܐ ܘܲܫܒ݂ܘܿܩ ܚܵܘ̈ܒܵܬܲܢ. ܘܲܥܛܝܼ ܟܠܗܘܿܢ ܣܘܼܪ̈ܚܵܢܲܝܢ. ܘܐܲܫܘܵܢ ܒܛܲܝܒ݁ܘܼ ܕܢܸܥܘܿܠ ܥܲܡܵܟ݂ ܠܲܓ݂ܢܘܿܢܵܐ. ܘܥܲܡ ܩܲܕܝܼܫܲܝ̈ܟ ܢܸܬ݂ܒܲܣܲܡ. ܟܠܗܘܿܢ ܠܵܟ݂ܘܼ ܡܣܲܟܹ݁ܝܢ. ܢܵܦܩܝܼܢ ܠܐܘܼܪܥܵܟ݂ ܡܵܐ ܕܐܵܬܹ݁ܝܬ݁. ܕܬܸܥܒܸ݁ܕ݂ ܩܝܵܡܬܵܐ ܠܥܲܢܝܼ̈ܕܹܐ. ܘܐܵܪܥܝܼܢ ܠܵܟ݂ ܒܲܚܕ݂ܵܐ ܫܲܘܝܘܼ. ܡܝܼ̈ܬܹܐ ܕܲܒ݂ܠܝܼܘ ܘܐܸܬ݂ܚܲܒܲܠܘ. ܫܵܠ̇ܚܝܼܢ ܘܫܵܕܹܝܢ ܠܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܡܵܝܘܿܬܹ̈ܐ. ܘܦܵܪܚܝܼܢ ܠܐܘܼܪܥܵܟ݂ ܒܲܥܢܵܢܹ̈ܐ܀
ܫܲܒܲܚ. ܡܫܝܼܚܵܐ ܕܨܵܡ ܕܢܸܦܪܘܿܥ ܚܵܘܒ݁ܬ݂ܵܐ. ܕܠܵܐ ܐ݇ܢܵܫ ܡܨܸܐ ܗ݇ܘܵܐ ܕܢܸܦܪܥܝܼܗ. ܚܘܼܪ ܒܵܗ ܡܵܪܝ ܒܚܲܝܵܒ݂ܘܼܬܲܢ. ܕܚܵܘܒܹ̈ܐ ܚܲܢܩܘܼܢ ܡܢ ܟܠ ܕܘܼܟ. ܘܡܸܠܟܲܝ̈ܟ ܛܵܒܹ̈ܐ ܘܦܘܼܩܕܵܢܲܝ̈ܟ ܐܲܠܵܗܵܝܹ̈ܐ. ܢܲܦܩܘܼܢ ܕܝܼܢܲܢ ܒܙܵܟ݂ܘܼܬ݂ܵܐ.
ܥܵܠܲܡ. ܓܢܘܿܢܵܐ ܘܛܘܼܒ݂ܵܐ ܕܠܵܐ ܥܵܒܲܪ. ܡܠܲܟ݂ ܦܵܪܘܿܩܲܢ ܠܩܲܕ݁ܝܼܫܵܘ̈ܗܝ. ܕܐܸܬ݁ܛܲܪܲܦܘ ܘܲܠܐܝܼܘ ܗܵܪܟܵܐ. ܒܫܲܗܪܵܐ ܘܨܵܘܡܵܐ ܘܲܨܠܘܿܬ݂ܵܐ. ܘܠܲܢ ܡܵܪܝ ܒܛܲܝܒ݁ܘܼ ܒܡܲܥܪ̈ܵܒܲܝ ܫܸܡܫܹܗ ܕܥܵܠܡܵܐ. ܐܲܫܘܵܢ ܕܥܲܡܗܘܿܢ ܢܵܘܕܸ݁ܐ ܠܵܟ݂.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܩܵܢܘܿܢܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ. ܚܘܼܣ ܥܠܲܝܢ ܐܲܢ݇ܬ݁ ܚܲܢܵܢܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܸܬ݂ܦܲܢ ܥܠܲܝܢ ܡܠܸܐ ܪ̈ܲܚܡܹܐ: ܘܠܵܐ ܡܵܪܝ ܬܲܗܡܸܐ ܚܝܵܪܵܟ݂ ܘܲܒ݂ܛܝܼܠܘܼܬ݂ܵܟ݂ ܡܢܢ: ܕܲܥܠܲܝܟ݁ܘܼ ܣܲܒ݂ܪܲܢ ܘܬ݂ܘܼܟ݂ܠܵܢܲܢ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ: ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
            ܩܵܢܘܿܢܵܐ. ܣܓ݂ܝܼܘ ܐܵܠܘܿܨܲܝ̈ ܡܢ ܛܲܠܝܘܼܬ݂ܝ* ܢܹܐܡܲܪ ܝܼܣܪܵܝܹܠ ܕܲܣܓ݂ܝܼܘ ܐܵܠܘܿܨܲܝ̈ ܡܢ ܛܲܠܝܘܼܬ݂ܝ* ܫܡܲܥ ܡܵܪܲܢ ܩܵܠ ܬܲܟ݂ܫܲܦ̮ܬܲܢ: ܘܢܲܛܲܪ ܠܲܢ ܡܢ ܒܝܼܫܵܐ. ܘܠܵܐ ܐܸܬ݂ܡܨܝܼܘ ܒܚܲܝܠܝ* ܥܲܠ ܚܲܨܝ ܢܲܓܸ݁ܕ݂ܘ ܢܸܓ݂ܕܹ̈ܐ ܘܐܲܓܲܪܘ ܡܘܼܟܵܟ݂ܗܘܿܢ* ܡܵܪܝܵܐ ܙܲܕ݁ܝܼܩ ܘܲܡܦܲܫܲܚ ܣܵܘܟܲܝ̈ܗܘܿܢ ܕܥܵܘܵܠܹ̈ܐ* ܢܸܗܦܟ݂ܘܼܢ ܠܒܸܣܬܲܪܗܘܿܢ ܟܠܗܘܿܢ ܣܵܢܐܹܝ̈ܗ ܕܨܸܗܝܘܿܢ* ܘܢܸܗܘܘܿܢ ܐܲܝܟ݂ ܥܡܝܼܪܵܐ ܕܐܸܓܵܪܵܐ* ܕܡܵܐ ܕܢܸܫܒܲܬ݀ ܒܹܗ ܪܘܼܚܵܐ ܫ̇ܵܠܹܦ ܘܝܵܒܹ݁ܫ* ܘܠܵܐ ܡܵܠܹܐ ܡܸܢܹܗ ܟܲܦܹܗ ܚܵܨܘܿܕ݂ܵܐ* ܘܐܵܦܠܵܐ ܩܵܦܠܹܗ ܡܩܲܒ݁ܪܵܢܵܐ* ܘܠܵܐ ܐܸܡܲܪܘ ܐܲܝܠܹܝܢ ܕܥܵܒ݂ܪܝܼܢ* ܕܒ݂ܘܼܪܟ݁ܬܹܗ ܕܡܵܪܝܵܐ ܥܠܲܝܟ݁ܘܿܢ* ܒܲܪܸܟ݂ܢܵܟ݂ܘܿܢ ܒܲܫܡܹܗ ܕܡܵܪܝܵܐ*
 ܫܲܒܲܚ. ܥܵܠܲܡ.

ܫܡܲܥ ܡܵܪܲܢ ܩܵܠ ܬܲܟ݂ܫܲܦ̮ܬܲܢ. ܘܢܲܛܲܪ ܠܲܢ ܡܢ ܒܝܼܫܵܐ.

ܣܓ݂ܝܼܘ ܐܵܠܘܿܨܲܝ̈ ܡܢ ܛܲܠܝܘܼܬ݂ܝ* ܢܹܐܡܲܪ ܝܼܣܪܵܝܹܠ ܕܲܣܓ݂ܝܼܘ ܐܵܠܘܿܨܲܝ̈ ܡܢ ܛܲܠܝܘܼܬ݂ܝ* ܫܡܲܥ ܡܵܪܲܢ ܩܵܠ ܬܲܟ݂ܫܲܦ̮ܬܲܢ: ܘܢܲܛܲܪ ܠܲܢ ܡܢ ܒܝܼܫܵܐ.
ܬܸܫܒ݁ܘܿܚܬܵܐ. ܢܸܗܘܸܐ ܨܠܝܼܒ݂ܵܟ݂ ܢܵܛܘܿܪܵܐ. ܠܟܲܪܟ݂ܵܐ ܕܥܵܢ̈ܵܟ݂ ܫ̇ܵܪܝܵܐ ܒܹܗ. ܘܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܬܸܗܘܸܐ ܣܲܟ݁ܪܵܐ. ܘܬܲܛܸܠ ܥܲܠ ܟܠܹܗ ܟܸܢܫܲܢ. ܬܢܝܼ: ܓ: ܙܲܒ݂ܢ̈ܝܼܢ.
      ܟܵܪܘܿܙܘܼܬ݂ܵܐ. ܐܲܚܝܼܕ݂ ܟܠ. ܘܡܸܬ݂ܐܲܡܪܵܐ ܒܟ݂ܠܗܘܿܢ ܣܘܼܒܵܥܹ̈ܐ ܕܨܵܘܡܵܐ.
 ܢܩܘܼܡ ܫܲܦܝܼܪ...܀ ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܲܚܝܼܕ݂ ܟܠ ܐܲܠܵܗܵܐ ܕܐܲܒ݂̈ܵܗܵܬܲܢ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܩܲܕ݁ܝܼܫܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܲܒ݂ܩܲܕ݁ܝܼܫܹ̈ܐ ܥܵܡܲܪ ܘܡܸܬ݁ܬ݁ܢܝܼܚ ܨܸܒ݂ܝܵܢܹܗ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܡܲܠܟܵܐ ܕܡܲܠܟܹ̈ܐ ܘܡܵܪܵܐ ܕܡܵܪ̈ܵܘܵܬ݂ܵܐ ܕܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܐ݇ܢܵܫ ܡܢ ܒܢܲܝ̈ܢܵܫܵܐ ܠܵܐ ܚܙܵܝܗܝ ܘܐܵܦܠܵܐ ܡܨܸܐ ܠܡܸܚܙܝܹܗ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܨܵܒܹ݁ܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܢܸܚܘܿܢ. ܘܢܸܬ݂ܦܢܘܿܢ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܢܟ݂...܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒ݂ܐܲܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂ ܐܲܫܪܵܐ ܒܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ: ܘܐܸܬ݂ܪܲܚܸܡܥܠܲܝܢ.
 ܘܟܸܢ ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ. ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܫܲܒܲܚ܀ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܥܵܠܲܡ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
 ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܦܵܬܲܚ ܬܲܪܥܹܗ ܠܲܕ݂ܢܵܩܫܝܼܢ ܒܹܗ: ܡܦܲܢܸܐ ܫܹܐ̈ܠܵܬ݂ܵܐ ܠܲܕ݂ܫܵܐܠܝܼܢ ܠܹܗ. ܦܬܲܚ ܠܵܗ̇ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܬܲܪܥܵܐ ܕܪ̈ܲܚܡܹܐ ܠܲܨܠܘܿܬܲܢ. ܘܩܲܒܸ݁ܠܹܝܗ̇ ܠܒ݂ܵܥܘܼܬܲܢ. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ ܡܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܥܲܬ݁ܝܼܪܵܐ ܘܲܫܦܝܼܥܵܐ: ܛܵܒ݂ܵܐ ܕܠܵܐ ܟܵܠܹܐ ܪ̈ܲܚܡܵܘܗܝ ܘܡܵܘܗܒ݂ܵܬܹ̈ܗ: ܡܢ ܣܢܝܼܩܹ̈ܐ ܘܐܲܠܝܼܨܹ̈ܐ ܥܲܒ݂ܕܵܘ̈ܗܝ ܕܩ̇ܵܪܹܝܢ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܹܗ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܐ݇ܚܪܹܬ݂ܵܐ. ܗܘܝܼ ܡܵܪܝ ܥܲܡܲܢ ܒܪ̈ܲܚܡܲܝܟ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܘܕܲܒܲܪ ܠܲܢ ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܘܼܪܟ݁ܬ݂ܵܐ ܡܵܪܝ ܕܪ̈ܲܚܡܹܐ: ܡܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܕܪ̈ܲܚܡܹܐ: ܒܝܲܕ݂ ܡܲܠܲܐܟ݂ܵܐ ܕܪ̈ܲܚܡܹܐ: ܡܢ ܠܘܵܬ݂ ܐܲܠܵܗܵܐ ܣܲܓ݁ܝܼ ܪ̈ܲܚܡܹܐ: ܬܸܫܬܲܕܲܪ ܘܬܸܫܬܲܦܲܥ ܥܲܠ ܟܸܢܫܹ̈ܐ ܕܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ: ܕܐܲܡܝܼܢܝܼܢ ܘܩܵܪܹܝ̇ܢ ܠܲܫܡܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ: ܘܚܵܝ̇ܪܝܼܢ ܘܲܡܣܲܟܹ݁ܝܢ ܠܪ̈ܲܚܡܹܐ: ܘܲܠܦܘܼܪܩܵܢܵܐ ܕܡܢ ܠܘܵܬ݂ܵܟ݂: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܡܲܩܦܝܼܢ. ܨܠܘܿܬ݂ܵܗ̇ ܡܵܪܝ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܗ̇ ܕܐܸܡܵܐ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܬܲܟ݂ܫܲܦ̮ܬܵܗ̇ ܘܬܲܚܢܲܢܬܵܗ̇ ܕܡܲܠܝܲܬ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܐ: ܡܵܪܬ݁ܝ ܡܲܪܝܲܡ ܛܘܼܒ݂ܵܢܝܼܬ݂ܵܐ: ܘܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ ܙܲܟܵܝܵܐ: ܘܥܘܼܕ݂ܪܵܢܵܐ ܐܲܠܵܗܵܝܵܐ: ܘܒ݂ܵܥܘܼܬܹܗ ܕܡܵܪܝ ܝܘܿܚܲܢܵܢ ܡܲܥܡܕ݂ܵܢܵܐ: ܘܕܲܫܠܝܼܚܹ̈ܐ ܛܘܼܒ݂ܵܢܹ̈ܐ: ܘܕܲܐܒ݂ܵܗ̈ܵܬ݂ܵܐ ܢܲܨܝܼܚܹ̈ܐ ܡܣܲܒ݁ܪ̈ܵܢܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܘܙܵܪ̈ܥܲܝ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܕܲܐܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܡܵܪܝ ܐܲܒ݂ܵܐ ܩܵܬ݂ܘܿܠܝܼܩܵܐ: ܘܡܵܪܝ ܐܸܣܛܲܦܵܢܘܿܣ ܒܘܼܟ݂ܪܵܐ ܕܣܵܗܕܹ̈ܐ: ܘܲܕ݂ܓܲܢ݇ܒܵܪ ܚܲܝܠܵܐ ܡܵܪܝ ܓܝܼܘܲܪܓܝܼܣ ܣܵܗܕܵܐ ܢܲܨܝܼܚܵܐ: ܘܲܕ݂ܡܵܪܝ ܦܠܵܢ: ܘܲܕ݂ܡܵܪܝ ܐܵܘܓܹ݁ܝܢ ܛܘܼܒ݂ܵܢܵܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܣܝܼܥܬܹ݁ܗ ܪܘܼܚܵܢܵܝܬܵܐ. ܘܕܲܫܡܘܿܢܝܼ ܘܕܲܒ݂ܢܹܝ̈ܗ̇: ܘܲܕ݂ܟ݂ܠܗܘܿܢ ܣܵܗܕܵܘ̈ܗܝ ܘܩܲܕ݁ܝܼ̈ܫܵܘܗܝ ܕܡܵܪܲܢ: ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܫܘܼܪܵܐ ܪܵܡܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܚܲܣܝܼܢܵܐ: ܕܬܸܦܪܘܿܩ ܘܲܬ݂ܦܲܨܸܐ ܘܲܬ݂ܫܵܘܙܸܒ݂ ܘܲܬ݂ܢܲܛܲܪ ܠܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ ܡܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹܗ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܘܟܸܢ ܚܵܬ݂ܡ̇ܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܪ̈ܵܝܹܐ. ܘܢܵܣܹ̇ܒ݂ ܟܵܗܢܵܐ ܨܠܝܼܒ݂ܵܐ ܒܐܝܼܕܹܗ ܕܝܲܡܝܼܢܵܐ: ܘܩܵܐܹ̇ܡ ܒܓܲܒܵܐ ܬܲܝܡܢܵܝܵܐ: ܘܡܲܗܦܸܟ݂ ܐܲܦܵܘ̈ܗܝ ܠܥܲܡܵܐ ܘܐܵܡܲܪ: ܒܲܪܸܟ݂ ܡܵܪܝ ܒܦܘܼܩܕܵܢܟ݂ܘܿܢ. ܘܥܵܢܹ̇ܝܢ: ܒܦܘܼܩܕܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܡܵܪܲܢ: ܘܫܘܼܒ݂ܚܵܐ ܠܲܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ. ܘܡܲܪܟ݁ܢܝܼܢ ܥܲܡܵܐ ܪܹ̈ܫܲܝܗܘܿܢ.
ܘܡܲܩܸܦ ܚܘܼܬܵܡܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ: ܨܸܡܚܹܗ ܕܐܲܒ݂ܵܐ ܡܬ݂ܘܿܡܵܝܵܐ: ܝܼܠܝܼܕ݂ܵܐ ܕܠܵܐ ܫܘܼܪܵܝܵܐ: ܠܥܸܠ ܡܢ ܙܲܒ݂ܢܹ̈ܐ ܘܲܗܘܲܝ̈ܵܐ: ܕܠܲܝܬ݁ ܠܲܢ ܣܲܒ݂ܪܵܐ ܘܣܘܼܟܵܝܵܐ: ܐܸܠܵܐ ܐܸܢ ܐܲܢ݇ܬ݁ ܒܵܪܘܿܝܵܐ. ܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܘܲܓ݂ܒܲܝ̈ܵܐ: ܕܲܫܦܲܪܘ ܠܵܟ݂ ܡܢ ܫܘܼܪܵܝܵܐ: ܥܒܸܕ݂ ܠܲܚܛܵܗܲܝ̈ܢ ܚܘܼܣܵܝܵܐ: ܘܲܠܣܲܟ݂ܠ̈ܘܵܬܲܢ ܫܪܵܝܵܐ: ܘܲܠܥܵܩ̈ܵܬܲܢ ܦܘܼܨܵܝܵܐ: ܘܲܠܫܸܐ̈ܠܵܬܲܢ ܦܘܼܢܵܝܵܐ: ܘܡܲܢܲܥܲܝܢ ܠܢܘܼܗܪܵܟ݂ ܓܲܐܝܵܐ: ܘܢܲܛܲܪܲܝܢ ܒܢܝܼܫܵܟ݂ ܚܲܝܵܐ: ܡܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܟܣܲܝ̈ܵܐ ܘܲܓ݂ܠܲܝ̈ܵܐ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ). ܗܵܫܵܐ ♰ ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܪܵܫܹ̇ܡ ܨܠܝܼܒ݂ܵܐ ܥܲܠ ܥܲܡܵܐ: ܟܲܕ݂ ܦܵܫܹ̇ܛ ܐܝܼܕܹܗ ܠܥܸܠ ܠܐܲܦܲܝ̈ ܡܲܥܪܒ݂ܵܐ: ܘܡܲܝܬܸ݁ܐ ܠܵܗ̇ ܠܡܲܕ݂ܢܚܵܐ: ܘܟܸܢ ܦܵܫܹ̇ܛ ܠܵܗ̇ ܠܓܲܒܵܐ ܕܬܲܝܡܢܵܝܵܐ: ܘܡܵܘܒܸ݁ܠ ܠܵܗ̇ ܠܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ. ܐܲܝܟܲܢܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܸܣܬ݁ܡܸܟ݂ ܥܲܠ ܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܓܲܒܵܐ ܣܸܡܵܠܵܝܵܐ ܕܟܸܢܫܵܐ.
ܘܥܵܢܹ̇ܝܢ: ܡܫܝܼܚܵܐ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬ݂ܵܟ݂: ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܟܵܗܢܘܼܬ݂ܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
ܐ݇ܚܪܹܢܵܐ: ܬܸܬ݂ܩܲܒܲܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܚܲܠܵܫܹ̈ܐ: ܩܕ݂ܵܡ ܒܹܝܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܬܸܗܘܸܐ ܟܢܝܼܫܘܼܬܲܢ ܗܵܕܹܐ ܠܲܢܝܵܚ ܨܸܒ݂ܝܵܢ ܡܵܪܘܼܬ݂ܵܟ݂: ܠܡܸܣܲܒ݂ ܡܸܢܵܟ݂ ܒܛܲܝܒ݁ܘܼ ܚܘܼܠܡܵܢܵܐ ܕܦܲܓ݂ܪܵܐ: ܘܢܘܼܛܵܪܵܐ ܕܢܲܦ̮ܫܵܐ: ܘܫܘܼܪܬܵܚܵܐ ܕܡܵܙܘܿܢܵܐ:(ܘܚܘܼܣܵܝܵܐ ܕܚܵܘܒܹ̈ܐ ܘܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ):ܘܫܲܝܢܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ: ܘܲܫܠܵܡܵܐ ܢܲܓ݁ܝܼܪܵܐ: ܘܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ ܕܠܵܐ ܥܵܒ݂ܪܵܐ ܘܠܵܐ ܡܸܫܬܲܪܝܵܐ ܡܢ ܒܲܝܢܵܬܲܢ: ܒܟ݂ܠܹܗ ܙܲܒ݂ܢܵܐ ܕܩܵܐܹ̇ܡ ܒܹܗ ܥܵܠܡܵܐ ܗܵܢܵܐ: )ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܒܪܝܼܟ݂ ܐܲܠܵܗܵܐ ܠܥܵܠܡܝܼܢ: ܘܲܡܫܲܒܲܚ ܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ ܠܕ݂ܵܪܕܵܪ̈ܝܼܢ: ܕܡܢܗ ܒܵܥܹܝ̇ܢܲܢ: ܘܲܠܝܲܡܵܐ ܫܦܝܼܥܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ: ܕܗܘ ܢܲܫܘܸܐ ܠܲܢ ܠܫܘܼܒ݂ܚܵܐ ܦܲܐܝܵܐ ܕܡܲܠܟ݁ܘܼܬܹܗ: ܘܲܠܒ݂ܘܼܣܵܡܵܐ ܕܥܲܡ ܡܲܠܲܐܟ݂ܵܘ̈ܗܝ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܘܲܠܓܲܠܝܘܼܬ݂ ܐܲܦܹ̈ܐ ܕܲܩܕ݂ܵܡܵܘܗܝ: ܘܲܠܩܵܘܡܵܐ ܕܡܢ ܝܲܡܝܼܢܹܗ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ: ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܪ̈ܲܚܡܵܘܗܝ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟ݁ܠ ܕܐܸܬ݂ܟܲܢܲܫܢܲܢ ܒܒܲܝܬܹ݁ܗ: ܘܨܲܠܝܼܢܲܢ ܩܕ݂ܵܡ ܪܲܒ݁ܘܼܬܹܗ: ܥܲܠ ܣܲܒ݂ܪܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܗܘ ܢܸܫܡܲܥ ܨܠܘܿܬܲܢ ܒܲܚܢܵܢܹܗ: ܘܲܢܩܲܒܸ݁ܠ ܒܵܥܘܼܬܲܢ ܒܚܵܘܣܵܢܹܗ: ܘܲܢܫܝܼܓ݂ ܘܢܸܡܪܘܿܩ ܨܵܐܬ݂ܵܐ ܕܚܵܘܒܲܝ̈ܢ ܘܕܲܚܛܵܗܲܝ̈ܢ ܒܙܘܿܦܵܐ ܫܦܝܼܥܵܐ ܕܚܲܝܘܼܣܬ݂ܵܢܘܼܬܹܗ: ܘܲܢܢܝܼܚ ܢܲܦ̮̈ܫܵܬ݂ܵܐ ܕܥܲܢܝܼܕܲܝ̈ܢ ܒܐܵܘܵܢܹ̈ܐ ܫܒ݂ܝܼܚܹ̈ܐ ܕܡܲܠܟ݁ܘܼܬܹܗ: ܘܢܸܪܘܿܣ ܥܲܠ ܟܠܲܢ ܛܲܠܵܐ ܕܒܲܣܝܼܡܘܼܬܹܗ: ܘܲܥܠܲܝܢ ܘܥܲܠ ܟܠ ܒܸܪ̈ܝܵܢ ܬܲܓܸ݁ܢ ܝܲܡܝܼܢܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬܹܗ: ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܪ̈ܲܚܡܵܘܗܝ:(ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.

ܐ݇ܚܪܹܢܵܐ

: ܐܲܠܵܗܵܐ ܡܵܪܹܟܠ ܕܝܲܗ݇ܒ݂ ܬܸܫܒ݁ܚܵܬܹ̈ܗ ܒܦܘܼܡܲܢ: ܘܲܙܡܝܼܪ̈ܵܬܹܗ ܒܠܸܫܵܢܲܢ: ܘܩܝܼܢܵܬܹ̈ܗ ܒܓܲܓ݁ܪ̈ܵܬܲܢ: ܘܬ݂ܵܘܕ݁ܝܼܬܹܗ ܒܣܸܦ̈ܘܵܬܲܢ: ܘܗܲܝܡܵܢܘܼܬܹܗ ܒܠܸܒܵܘ̈ܵܬܲܢ: ܗܘ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬܲܢ: ܘܲܢܩܲܒܸ݁ܠ ܒܵܥ̈ܘܵܬܲܢ: ܘܢܸܬ݂ܪܲܥܸܐ ܒܬܲܟ݂ܫ̈ܦܵܬܲܢ: (ܘܲܢܚܲܣܸܐ ܚܵܘ̈ܒܵܬܲܢ:)ܘܲܢܦܲܢܸܐ ܠܲܢ ܫܸܐ̈ܠܵܬܲܢ ܫܲܦܝܼܪ̈ܵܬ݂ܵܐ ܠܵܐ ܥܕ݂ܝܼ̈ܠܵܬ݂ܵܐ. ܘܡܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܢܲܫܦܲܥ ܪ̈ܲܚܡܵܘܗܝ ܘܲܚܢܵܢܹܗ: ܥܠܲܝܢ ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡܢ ܣܵܟܹܗ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܢܸܫܬܲܒܲܚ ܫܡܹܗ ܕܐܲܠܵܗܵܐ ܡܵܪܹܟܠ ܒܲܝܢܵܬܲܢ: ܡܛܲܟ݁ܣܵܢܗܘܿܢ ܕܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ. ܘܲܥܠܲܝܢ ܡܚܝܼ̈ܠܹܐ ܘܚܲܛܵܝܹ̈ܐ: ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡܢ ܣܵܟܹܗ. ܘܥܲܠ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇: ܘܥܲܠ ܐܲܒ݂ܵܗܲܝ̈ܢ ܘܪ̈ܲܒܵܢܲܝܢ ܘܡܲܠܦܵܢܲܝ̈ܢ ܘܥܲܢܝܼܕܲܝ̈ܢ: ܘܥܲܢܝܼܕܹ̈ܐ ܕܲܦܪܲܫܘ ܘܫܲܢܝܼܘ ܡܢ ܒܲܝܢܵܬܲܢ: ܘܥܲܠ ܟܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܬܲܓܸ݁ܢ ܝܲܡܝܼܢܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬܹܗ ܘܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܠܐܲܠܵܗܵܐ ܫܘܼܒ݂ܚܵܐ: ܘܲܠܡܲܠܲܐܟܹ̈ܐ ܐܝܼܩܵܪܵܐ: ܘܲܠܣܵܛܵܢܵܐ ܒܸܗܬ݁ܬ݂ܵܐ: ܘܠܲܨܠܝܼܒ݂ܵܐ ܣܸܓ݂ܕ݁ܬ݂ܵܐ: ܘܲܠܥܹܕ݁ܬܵܐ ܪܘܼܡܪܵܡܵܐ: ܘܲܠܥܲܢܝܼܕܹ̈ܐ ܢܘܼܚܵܡܵܐ: ܘܲܠܬܲܝܵܒܹ̈ܐ ܩܘܼܒܵܠܵܐ: ܘܠܲܐܣܝܼܪܹ̈ܐ ܫܪܵܝܵܐ: ܘܲܠܡܲܪ̈ܥܹܐ ܘܲܟ݂ܪ̈ܝܼܗܹܐܚܘܼܠܡܵܢܵܐ ܘܐܵܣܝܘܼܬ݂ܵܐ: ܘܠܲܐܪ̈ܒܲܥܦܸܢܝܵܬܹܗ ܕܥܵܠܡܵܐ ܫܲܝܢܵܐ ܪܲܒܵܐ ܘܲܫܠܵܡܵܐ. ܘܐܵܦ ܥܠܲܝܢ ܕܐܝܼܬܲܝܢ ܡܚܝܼ̈ܠܹܐ ܘܚܲܛܵܝܹ̈ܐ: ܚܢܵܢܹܗ ܘܪ̈ܲܚܡܵܘܗܝ ܕܐܲܠܵܗܲܢ ܣܓ݂ܝܼܕ݂ܵܐ ܢܹܐܬ݂ܘܿܢ ܘܢܲܓ݁ܢܘܼܢ ܘܢܸܫܬܲܦܥܘܼܢ ܘܲܢܩܵܘܘܿܢ ܘܢܲܡܠܟ݂ܘܼܢ ܒܐܲܡܝܼܢܘܼܬ݂ܵܐ: )ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: )ܒܝܲܡܝܼܢ ܡܵܪܘܼܬ݂ܵܟ݂ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ): ܒܲܪܸܟ݂ܡܵܪܝ ܠܟ݂ܠܲܢ: ܘܢܲܛܲܪ ܠܟ݂ܠܲܢ: ܘܥܲܕܲܪ ܠܟ݂ܠܲܢ: ܘܣܲܝܲܥ ܘܣܲܬܲܪ ܠܟ݂ܠܲܢ: ܘܢܲܚܸܡ ܥܲܢܝܼܕܹ̈ܐ ܕܟ݂ܠܲܢ: ܘܐܲܓܸ݁ܢ ܝܲܡܝܼܢܵܟ݂ ܥܲܠ ܟܠܲܢ: ܘܪ̈ܲܚܡܲܝܟ ܘܲܚܢܵܢܵܟ݂ ܢܸܫܬܲܦܥܘܼܢ ܥܲܠ ܟܠܲܢ:ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ: ܘܩܘܼܒܵܠܛܲܝܒ݁ܘܼܬ݂ܵܐ ܐܲܡܝܼܢܬܵܐ ܬܸܣܲܩ ܠܵܟ݂ ܡܢ ܦܘܼܡ ܟܠܲܢ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟ݁ܠ ܢܒܲܪܸܟ݂ ܠܟܸܢܫܲܢ ܒܛܲܝܒ݁ܘܼܬܹܗ: ܘܲܢܢܲܛܲܪ ܠܲܩܝܵܡܲܢ ܒܣܘܿܓ݂ܐܵܐ ܫܦܝܼܥܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܘܲܢܦܲܢܸܐ ܫܸܐ̈ܠܵܬܲܢ ܡܢ ܒܹܝܬ݂ ܓܲܙܹܗ: ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡܢ ܣܵܟܹܗ: ܘܥܲܠ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇: ܘܥܲܠ ܐܲܬ݂ܪܵܐ ܗܵܢܵܐ ܘܥܵܡܘܿܪ̈ܵܘܗܝ: ܘܥܲܠ ܥܘܼܡܪܵܐ ܗܵܢܵܐ ܘܝܲܬܵܒ݂ܵܘ̈ܗܝ: ܘܥܲܠ ܟܠܲܢ ܒܟ݂ܠܲܢ ܟܢܝܼܫܵܐܝܼܬ݂: ܚܢܵܢܹܗ ܘܪ̈ܲܚܡܵܘܗܝ ܕܐܲܠܵܗܲܢ ܛܵܒ݂ܵܐ: ܢܹܐܬ݂ܘܿܢ ܘܢܸܫܬܲܦܥܘܼܢ ܒܐܲܡܝܼܢܘܼ (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܡܲܩܦܝܼܢ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܲܬ݂ܠܵܬ݂ܡܐܵܐ ܘܲܬ݂ܡܲܢܬܲܥܣܲܪ ܐܲܒ݂ܵܗ̈ܵܬ݂ܵܐ ܐܲܦܸܣܩܘܿܦܹ̈ܐ: ܕܐܸܬ݂ܟܲܢܲܫܘ ܒܢܝܼܩܝܼܵܐ ܡܕ݂ܝܼܢ݇ܬܵܐ ܕܒܹܝܬ݂ܘܿܢܝܼܵܐ: ܒܙܲܒ݂ܢܵܐ ܕܩܘܿܣܛܲܢܛܝܼܢܘܿܣ ܡܲܠܟܵܐ ܙܲܟܵܝܵܐ. ܘܥܸܠܬ݂ܵܐ ܕܹܝܢ ܕܲܟ݂ܢܘܼܫܝܼܗܘܿܢ ܡܸܛܠ ܪܘܼܫܥܹܗ ܕܐܲܪܝܼܘܿܣ܀ ܘܐܲܝܟ݂ ܥܝܵܕ݂ܵܐ ܕܹܝܢ ܕܪܵܕܹܐ ܒܲܝܢܵܬ݂ ܒܢܹܝ̈ܗ̇ ܕܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܲܫܠܝܼܚܵܝܬܵܐ ܕܝܼܠܲܢ ܕܡܲܕ݂ܢܚܵܐ: ܘܲܟ݂ܒܲܪ ܕܡܢ ܪܵܘܡܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܪܕ݂ܵܐ: ܘܡܢ ܡܲܫܠܡܵܢܘܼܬ݂ܵܐ ܘܝܘܼܒܵܠܵܐ ܕܥܲܬ݁ܝܼܩܹ̈ܐ ܡܓ݂ܵܘܝܲܝ̈ ܒܬܹܐܘܿܪ̈ܝܼܵܣ ܕܲܫܪܵܪܵܐ ܐܲܠܵܗܵܝܵܐ ܡܬ݂ܝܼܚ ܗ݇ܘܵ ܥܕܲܡܵܐ ܠܝܵܘ̈ܡܵܬܲܢ. ܘܗܵܟܲܢܵܐ ܡܫܲܪܹܝܢ ܚ݇ܢܲܢ ܠܡܸܬ݂ܢܵܐ ܩܵܢܘܿܢܵܐ ܕܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܢܝܼܩܝܼܵܐ.

ܒܲܫܪܵܪܵܐ ܓܹܝܪ ܕܠܵܐ ܦܘܼܠܵܓ݂ܵܐ:

ܡܗܲܝܡܢܝܼܢܲܢ ܒܚܲܕ݂ ܐܲܠܵܗܵܐ: ܐܲܒ݂ܵܐ ܐܲܚܝܼܕ݂ ܟܠ. ܥܵܒ݂ܘܿܕ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܐܲܝܠܹܝܢ ܕܡܸܬ݂ܚܲܙ̈ܝܵܢ ܘܲܕ݂ܠܵܐ ܡܸܬ݂ܚܲܙ̈ܝܵܢ. ܘܲܒ݂ܚܲܕ݂ ܡܵܪܝܵܐ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ. ܒܪܹܗ ܕܐܲܠܵܗܵܐ ܝܼܚܝܼܕ݂ܵܝܵܐ: ܒܘܼܟ݂ܪܵܐ ܕܟ݂ܠܗܹܝܢ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܗܵܘ̇ ܕܡܢ ܐܲܒ݂ܘܼܗܝ ܐܸܬ݂ܝܼܠܸܕ݂ ܩܕ݂ܵܡ ܟܠܗܘܿܢ ܥܵܠܡܹ̈ܐ: ܘܠܵܐ ܐܸܬ݂ܥܒܸܕ݂. ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܕܡܢ ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܒܲܪ ܟܝܵܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܕܒ݂ܐܝܼ̈ܕ݂ܵܘܗܝ ܐܸܬ݁ܬܲܩܲܢܘ ܥܵܠܡܹ̈ܐ: ܘܐܸܬ݂ܒ݁ܪܝܼ ܟܠ ܡܸܕ݁ܡ. ܗܵܘ̇ ܕܡܸܛܠܵܬܲܢ ܒܢܲܝ̈ܢܵܫܵܐ: ܘܡܸܛܠ ܦܘܼܪܩܵܢܲܢ: ܢܚܸܬ݂ ܡܢ ܫܡܲܝܵܐ: ܘܐܸܬ݂ܓܲܫܲܡ ܡܢ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܘܲܗܘܵܐ ܒܲܪܢܵܫܵܐ: ܘܐܸܬ݂ܒ݁ܛܸܢ ܘܐܸܬ݂ܝܼܠܸܕ݂ ܡܢ ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܘܚܲܫ ܘܐܸܙܕ݁ܩܸܦ: ܒܝܵܘܡܲܝ̈ ܦܲܢܛܝܼܘܿܣ ܦܝܼܠܵܛܘܿܣ. ܘܐܸܬ݂ܩ̄ܒܲܪ: ܘܩܵܡ ܠܲܬ݂ܠܵܬ݂ܵܐ ܝܵܘܡܝܼ̈ܢ ܐܲܝܟ݂ ܕܲܟ݂ܬ݂ܝܼܒ݂. ܘܲܣܠܸܩ ܠܲܫܡܲܝܵܐ: ܘܝܼܬܸܒ݂ ܡܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܘܬ݂ܘܼܒ݂ ܥܬ݂ܝܼܕ݂ ܠܡܹܐܬ݂ܵܐ ܠܲܡܕ݂ܵܢ ܠܡܝܼܬܹ̈ܐ ܘܲܠܚܲܝܹ̈ܐ. ܘܲܒ݂ܚܲܕ݂ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܪܘܼܚܵܐ ܕܲܫܪܵܪܵܐ: ܗܵܘ̇ ܕܡܢ ܐܲܒ݂ܵܐ ܢܵܦܹ݁ܩ: ܪܘܼܚܵܐ ܡܲܚܝܵܢܵܐ. ܘܒܲܚܕ݂ܵܐ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܘܲܫܠܝܼܚܵܝܬܵܐ ܩܵܬ݂ܘܿܠܝܼܩܝܼ. ܘܡܵܘܕܹ݁ܝܢܲܢ ܒܲܚܕ݂ܵܐ ܡܲܥܡܘܿܕ݂ܝܼܬ݂ܵܐ: ܠܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ. ܘܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܲܒ݂ܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ: ܐܵܡܹܝܢ.
