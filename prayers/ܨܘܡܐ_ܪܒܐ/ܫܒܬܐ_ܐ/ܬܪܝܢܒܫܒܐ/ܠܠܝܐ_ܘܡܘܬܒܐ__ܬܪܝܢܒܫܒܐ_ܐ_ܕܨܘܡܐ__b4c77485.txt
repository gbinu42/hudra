ܒܠܠܝܐ ܕܹܝܢ ܡܢ ܩܕ݂ܵܡ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝ̈ܵܐ. ܐܵܡܪܝܼܢ ܩܢܘܢܐ ܗܵܢܵܐ ܒܟ݂ܠܗܹܝܢ ܫܲܒܹ̈ܐ ܕܐ݇ܪ̈ܵܙܹܐ: ܘܲܥܪ̈ܘܼܒ݂ܵܬ݂ܵܐ.
	 ܐܸܢܵܐ ܫܟ݂ܒܹ݁ـܬ݂ ܘܕܸܡܟܹ݁ܬ݂ ܘܐܸܬ݁ܬ݁ܥܝܼـܪܹܬ݂* ܡܸܛـܠ ܕܡܵܪܝܵܐ ܣܲܡܟܲܢܝ. ܝܵܐܹܐ ܘܙܵܕܹ݁ܩ ܕܲܢܩܲܕܸ݁ܡ. ܘܲܢܫܲܒܲܚ ܠܵܟ݂ ܦܵܪܘܿܩܲܢ. ܘܲܢܝܲܒܸ݁ܒ݂  ܥܲܡ ܡܲܠܲܐܟܹ̈ܐ. ܒܩܵܠܹ̈ܐ ܕܲܡܠܹܝܢ ܬܵܘܕ݁ܝܼܬ݂ܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܐܸܢܵܐ ܫܸܟ݂ܒܹ݁ܬ݂ ܘܕܸܡܟܹ݁ܬ݂ ܘܐܸܬ݁ܬ݁ܥܝܼܪܹܝܬ݂* ܡܸܛܠ ܕܡܵܪܝܐ ܣܲܡܟܲܢܝ܀ ܠܵܐ ܠܵܐ ܐܸܕ݂ܚܲܠ ܡܢ ܪܸ̈ܒ݁ܘܵܬܹܗ ܕܥܲܡܵܐ* ܕܚܲܕ݂ܪܘܼܢܝ ܘܣܵܡܘ ܥܠܲܝ܀ ܩܘܼܡ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܘܲܦܪܘܿܩܲܝܢܝ* ܡܸܛܠ ܕܐܲܢ݇ܬ݁ ܡܚܲܝܬ݁ܠܟ݂ܠܗܘܿܢ ܒܥܸܠܕܒ݂ܵܒܲܝ̈ ܥܲܠ ܦܲܟܲܝ̈ܗܘܿܢ܀ ܘܫܸܢܹ̈ܐ ܕܪ̈ܲܫܝܼܥܹܐ ܬܲܒܲܪܬ݁* ܕܡܵܪܝܵܐ ܗ݇ܘ ܦܘܼܪܩܵܢܵܐ* ܘܥܲܠ ܥܲܡܵܟ݂ ܒܘܼܪܟ݁ܬ݂ܵܟ݂ ܠܥܵܠܲܡ܀ ܝܵܐܹܐ ܘܙܵܕܹܩ ܕܲܢܩܲܕܸܡ. ܘܡܲܩܦܝܼܢ. ܒܲܪܟ݂ܘ ܠܡܵܪܝܵܐ ܟܠܟ݂ܘܿܢ ܥܲܒ݂ܕܵܘ̈ܗܝ ܕܡܵܪܝܵܐ. ܐܲܝܠܹܝܢ ܕܩܵܝܡܝܼܢ ܒܒܲܝܬܹ݁ܗ ܕܡܵܪܝܵܐ ܒܠܲܝ̈ܠܵܘܵܬ݂ܵܐ. ܫܘܼܒ݂ܚܵܐ ܙܵܡܪܝܼܢ ܩܲܕ݁ܝܼܫܹ̈ܐ ܒܠܲܝ̈ܠܵܘܵܬ݂ܵܐ ܥܲܡ ܥܝܼܪܹ̈ܐ: ܘܪܸ̈ܒܘܵܬ݂ܵܐ ܕܟܸܢܫܲܝ̈ ܪܵܘܡܵܐ. ܬܵܢܹܝܢ ܫܘܼܒ݂ܚܵܐ ܕܪܲܒܘܼܬܹܗ ܕܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ. ܒܲܪܸܟ݂ܘ ܠܡܵܪܝܵܐ ܟܠܟ݂ܘܿܢ ܥܲܒ݂ܕܵܘ̈ܝ ܕܡܵܪܝܵܐ* ܐܲܝܠܹܝܢ ܕܩܵܝܡܝܼܢ ܒܒܲܝܬܹ݁ܗ ܕܡܵܪܝܵܐ ܒܠܲܝ̈ܠܵܘܵܬ݂ܵܐ܀ ܐܲܪܝܼܡܘ ܐܝܼܕܲܝ̈ܟ݁ܘܿܢ ܠܩܘܼܕ݂ܫܵܐ ܘܒܲܪܸܟ݂ܘ ܠܡܵܪܝܵܐ* ܢܒܲܪܟ݂ܟ݂ ܡܵܪܝܵܐ ܡܢ ܨܸܗܝܘܿܢ* ܗ̇ܘ ܕܲܥܒܲܕ݂ ܫܡܲܝܵܐ ܘܐܲܪܥܵܐ. ܫܘܼܒ݂ܚܵܐ ܙܵܡܪܝܼܢ ܩܲܕ݁ܝܼܫܹ̈ܐ. ܫܲܒܲܚ. ܠܗ̇ܘ ܕܲܟ݂ܪ̈ܘܿܒܹܐ ܟܪܝܼܟ݂ܝܼܢ ܠܹܗ. ܣܪ̈ܵܦܹܐ ܡܩܲܕ݁ܫܝܼܢ ܠܐܝܼܩܵܪܹܗ. ܡܲܠܲܐܟܹ̈ܐ ܘܲܒ݂ܢܲܝ̈ܢܵܫܵܐ. ܬܵܢܹܝܢ ܫܘܼܒ݂ܚܵܐ ܕܪܲܒ݁ܘܼܬܹܗ. ܕܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ. ܕܐܸܬ݂ܵܐ ܘܦܲܪܩܲܢ ܒܲܨܠܝܼܒܹܗ. ܘܐܸܫܬܵܘܕ݁ܝܼ ܠܲܢ ܡܲܠܟ݁ܘܼܬ݂ܵܐ: ܕܠܵܐ ܥܵܒ݂ܪܵܐ ܘܠܵܐ ܡܸܫܬܲܪܝܵܐ: ܘܒ݂ܘܼܣܵܡܵܐ ܕܠܵܐ ܫܘܼܠܵܡܵܐ: ܠܹܗ ܫܘܼܒ݂ܚܵܐ ܘܲܥܠܲܝܢ ܪ̈ܲܚܡܵܘܗܝ. ܘܗܵܦ݁ܟ݁ܝܼܢ ܘܐܵܡܪܝܼܢ. ܐܸܢܵܐ ܫܸܟ݂ܒܹ݁ܬ݂ ܘܕܸܡܟܹ݁ܬ݂ ܘܐܸܬ݁ܬ݁ܥܝܼܪܹܝܬ݂* ܡܸܛܠ ܕܡܵܪܝܵܐ ܣܲܡܟܲܢܝ. ܫܵܘܹܐ ܠܫܘܼܒ݂ܚܵܐ ܘܠܐܝܼܩܵܪܵܐ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܩܘܼܕ݂ܫܵܐ. ܡܢ ܥܸܠܵܝܹ̈ܐ ܘܬܲܚܬܵܝܹ̈ܐ. ܡܢ ܪ̈ܘܼܚܵܢܹܐ ܘܦܲܓ݂ܪ̈ܵܢܹܐ. ܝܲܬ݁ܝܼܪ ܕܹܝܢ ܡܢ ܒܢܲܝ̈ ܥܹܕ݁ܬܵܐ. ܕܐܲܫܘܝܼ ܐܸܢܘܿܢ ܠܬܸܫܒ݁ܘܿܚܬܹܗ. ܕܲܢܫܲܒ݁ܚܘܼܢܵܝܗܝ ܬܠܝܼܬ݂ܵܐܝܼܬ݂. ܙܡܝܼܪܬܵܐ ܬܠܝܼܬ݂ܵܝܬܵܐ ܕܩܘܼܕܫܹܗ. ܕܠܹܗ ܫܘܼܒ݂ܚܵܐ ܡܢ ܟܠ ܘܲܒ݂ܟ݂ܠ܀ ܘܐܸܢܵܐ ܕܒܲܚܛܵܗܲܝ̈ ܫܸܟ݂ܒܹ݁ܬ݂. ܘܲܒ݂ܫܸܢܬܵܐ ܕܥܵܘܠܵܐ ܕܸܡܟܹ݁ܬ݂: ܘܗܵܫܵܐ ܒܪܸܡܙܵܟ݂ ܐܸܬ݁ܬ݁ܥܝܼܪܹܬ݂. ܘܥܲܡ ܡܫܲܒ݁ܚܵܢܲܝ̈ܟ ܐܸܫܬܵܘܬ݁ܦܹܬ݂: ܫܒ݂ܘܿܩ ܠܝܼ ܟܠ ܕܲܚܛܹܝܬ݂ ܘܐܲܣܟ݁ܠܹܬ݂: ܘܐܲܫܘܵܢܝ ܠܫܘܼܒ݂ܚܵܟ݂ ܕܠܹܗ ܪܸܚܡܹܬ݂. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܡܥܝܼܪܵܢܲܢ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܡܩܝܼܡܵܢܲܢ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܡܢܲܚܡܵܢܲܢ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܡܚܲܕܬ݂ܵܢܲܢ.
 ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ. ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
 ܘܐܵܡܵܪ ܟܵܗܢܵܐ ܗܵܘ ܕܐܸܡܲܪ ܩܵܢܘܿܢܵܐ ܗܵܢܵܐ. ܩܕ݂ܵܡ ܬܲܪܥܵܐ ܕܡܲܕ݂ܒ݁ܚܵܐ. ܟܲܕ݂ ܓܠܸܐ ܪܹܫܹܗ. ܗܵܕܹܐ ܨܠܘܿܬ݂ܵܐ. ܠܵܟ݂ ܡܥܝܼܪܵܢܵܐ ܕܕܲܡܟܹ̈ܐ. ܬܢܝܼ. ܘܲܡܩܝܼܡܵܢܵܐ ܕܢܲܦܠܹ̈ܐ: ܘܡܲܪܘܚܵܢܵܐ ܕܐܲܠܝܼܨܹ̈ܐ: ܘܲܡܚܲܣܝܵܢܵܐ ܕܚܲܛܵܝܹ̈ܐ: ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܪܲܒܵܐ ܕܬܲܝܵܒܹ̈ܐ: ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ ܘܫܵܐܠܝܼܢܲܢ. ܐܲܥܝܼܪ ܡܵܪܝ ܕܲܡܟ݁ܘܼܬܲܢ ܒܲܚܢܵܢܵܟ݂: ܘܢܲܦܸܨ ܝܘܼܩܪܵܐ ܕܡܲܐܝܼܢܘܼܬܲܢ ܒܚܵܘܣܵܢܵܟ݂. ܘܗܲܒ݂ܠܲܢ ܘܐܲܫܘܵܐ ܠܲܢ ܕܲܢܩܘܼܡ ܘܲܢܫܲܡܸܫ ܩܕ݂ܵܡܲܝܟ. ܕܲܟ݂ܝܵܐܝܼܬ݂ ܘܙܲܗܝܵܐܝܼܬ݂ ܘܥܝܼܪܵܐܝܼܬ݂ ܘܲܚܦܝܼܛܵܐܝܼܬ݂. ܥܝܼܪܵܐ ܕܲܡܫܲܡܫܝܼܢ ܠܹܗ ܥܝܼܪܹ̈ܐ ܒܗܘܼܠܵܠܲܝ̈ܗܘܿܢ. ܘܲܣܪ̈ܵܦܹܐ ܒܩܘܼܕ݂ܵܫܲܝ̈ܗܘܿܢ: ܘܬܲܚܬܵܝܹ̈ܐ ܒܲܙܡܝܼܪ̈ܵܬ݂ܗܘܿܢ: ܘܥܲܡܡܹ̈ܐ ܟܠܗܘܿܢ ܒܣܸܓ݂ܕܵܬ݂ܗ̈ܘܿܢ. ܡܵܪܵܐ ܕܟ݂ܠ.
	ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܘܐܵܡܲܪ ܟܵܗܢܵܐ ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟ݂: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟ݂: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬ݂ܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬܵܐ ܘܣܸܓ݂ܕܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.
	ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.
	ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ.
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܚܲܝܸܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܠܲܡܚܝܼܠܘܼܬܲܢ ܒܲܚܢܵܢܵܟ݂: ܘܠܲܒܸ݁ܒ݂ ܘܥܲܕܲܪ ܠܲܙܥܘܿܪܘܼܬ݂ ܢܲܦ̮ܫܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܥܝܼܪ ܕܲܡܟ݁ܘܼܬ݂ ܪܸܥܝܵܢܲܢ: ܘܐܲܩܸܠ ܝܘܼܩܪܵܐ ܕܗܲܕܵܡܲܝ̈ܢ: ܘܐܲܫܝܼܓ݂ ܘܲܡܪܘܿܩ ܨܵܐܬ݂ܵܐ ܕܚܵܘܒܲܝ̈ܢ ܘܕܲܚܛܵܗܲܝ̈ܢ: ܘܐܲܢܗܲܪ ܠܚܸܫܘܿܟ݂ܘܼܬܹܗ ܕܡܲܕ݁ܥܲܢ: ܘܐܵܘܫܸܛ ܘܗܲܒ݂ ܒܲܢ ܚܲܝܠܵܐ ܘܐܝܼܕ݂ܵܐ ܕܥܘܼܕ݂ܪ̈ܵܢܹܐ: ܕܒ݂ܵܗ̇ ܢܩܘܼܡ ܘܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.
	ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.
	ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ.
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܚܲܝܠܵܐ ܡܵܪܝ ܟܲܣܝܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܥܘܼܕ݂ܪܵܢܵܐ ܬܡܝܼܗܵܐ ܕܡܵܪܘܼܬ݂ܵܟ݂: ܘܣܘܼܝܵܥܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂: ܗܘ ܢܚܲܝܸܠܠܲܡܚܝܼܠܘܼܬܹܗ ܕܲܟ݂ܝܵܢܲܢ ܚܲܠܵܫܵܐ: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬܵܐ ܘܣܸܓ݂ܕܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.
	ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.
	ܗܲܠܸܠܘܼܝܲܗ ܗܲܠܸܠܘܼܝܵܗ: ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ.
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.

	ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.
	ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.
	ܗܲܠܸܠܘܼܝܲܗ ܗܲܠܸܠܘܼܝܵܗ: ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ.
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܡܵܘܬ݁ܒ݂ܵܐ.
      ܨܠܘܿܬ݂ܵܐ. ܬܸܒ݂ܣܲܡ ܠܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܚܲܠܵܫܘܼܬܲܢ: ܘܬܸܥܘܿܠ ܩܕ݂ܵܡܲܝܟ ܒܵܥܘܼܬ݂ܵܐ ܕܲܡܚܝܼܠܘܼܬܲܢ: ܘܪ̈ܲܚܡܲܝܟ ܢܸܗܘܘܿܢ ܣܢܹܐܓ݂ܪܹ̈ܐ ܛܵܒܹ̈ܐ ܠܚܲܝܵܒ݂ܘܼܬܲܢ: ܘܡܢ ܒܹܝܬ݂ ܓܲܙܹܗ ܪܲܒܵܐ ܕܲܚܢܵܢܵܟ݂ ܢܸܬ݂ܦܲܢ̈ܝܵܢ ܫܹܐ̈ܠܵܬܵܐ ܠܲܣܢܝܼܩܘܼܬܲܢ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
 ܡܘܬ݁ܒ݂ܐ. ܒܐܲܠܵܗܵܐ ܡܸܠܬ݂ܵܐ. ܡܲܢܘܼ ܐܲܟ݂ܘܵܬ݂ܵܟ݂ ܡܵܪܝܵܐ: ܡܲܢܘܼ ܢܸܫܬܲܥܸܐ ܬܸܕ݂ܡܪ̈ܵܬܹܗ ܕܡܵܪܝܵܐ. ܡܲܢܘܼ ܣܵܦܹ݁ܩ ܥܵܒ݂ܘܿܕܲܢ ܕܢܵܘܕܸ݁ܐ  ܚܠܵܦܲܝܢ ܠܲܡܪܲܚܡܵܢܘܼܬ݂ܟ݂. ܕܲܒ݂ܫܘܼܪܵܝܵܐ ܒܨܲܠܡܵܟ݂ ܝܲܩܝܼܪܵܐ ܐܲܬ݂ܩܸܢܬܵܢ. ܘܲܒ݂ܚܲܪܬܵܐ ܕܙܲܒ݂ܢܹ̈ܐ ܠܒܸܫܬ݂ܵܢ ܘܐܲܦ̮ܢܝܼܬܵܢ ܠܝܼܕܲܥܬ݂ܵܟ݂. ܡܵܘܪܸܒ݂ ܓܸܢܣܲܢ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂: ܬܢܝܼ. ܢܵܘܕܸܐ ܠܵܟ݂ ܠܥܵܠܲܡ. ܠܵܟ݂ ܡܵܘܕܹ݁ܝܢܲܢ ܥܵܒ݂ܘܿܕܲܢ. ܕܐܲܢ݇ܬ݁ܘܼ ܡܪܲܚܡܲܬ݁ ܥܲܠ ܟܠܵܗ̇ ܒܪܝܼܬ݂ܵܐ. ܠܛܵܒܹ̈ܐ ܓܹܝܪ ܘܲܠܒ݂ܝܼܫܹ̈ܐ ܛܥܝܼܢܲܬ݁ ܒܲܚܢܵܢܵܟ݂. ܘܲܫܦܝܼܥܵܐܝܼܬ݂ ܝܵܗܹܒ݂ ܐܲܢ݇ܬ݁ ܥܘܼܕ݂ܪ̈ܵܢܹܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ. ܣܲܓܝܼ ܪ̈ܲܚܡܹܐ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܬܵܘ ܢܫܲܒܲܚ ܠܡܵܪܝܵܐ. ܬܵܘ ܢܵܘܕܸ݁ܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܕܲܒ݂ܓܸܠܝܵܢܹܗ ܚܲܪܲܪ ܠܲܢ ܡܢ ܛܘܼܥܝܲܝ. ܘܪܲܥܝܼ ܠܲܢ ܥܲܡ ܐܲܒ݂ܘܼܗܝ ܒܩܘܼܪܒܵܢܵܐ ܕܦܲܓ݂ܪܹܗ. ܘܫܲܝܸܢ ܥܲܡܲܢ ܥܸܠܵܝܹ̈ܐ ܕܪܲܓ݁ܝܼܙܝܼܢ ܗ݇ܘܵܘ ܡܸܛܠ ܥܵܘܠܲܢ. ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ ܥܲܡ ܫ̇ܵܠܘܿܚܹܗ. ܕܲܨܠܘܿܬ݂ܵܐ. ܦܬܲܚܠܲܢ ܡܵܪܲܢ. ܩܵܠܵܐ: ܐ.
  ܒܡܲܢܘܼ ܣܵܦܹ݁ܩ. ܦܬܲܚܠܝܼ ܬܲܪ̈ܥܹܐ ܕܙܲܕܝܼܩܘܼܬ݂ܵܐ. ܦܬܲܚܠܲܢ ܡܵܪܲܢ: ܒܹܝܬ݂ ܓܲܙܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂. ܕܢܸܣܲܒ݂ ܪ̈ܲܚܡܹܐ ܘܦܘܼܪܩܵܢܵܐ: ܠܓܸܢܣܲܢ ܡܸܣܟܹ݁ܢܵܐ. ܘܐܲܣܵܐ ܟܹܐܒܹ̈ܐ ܕܲܚܛܵܗܲܝ̈ܢ: ܒܣܲܡܵܐ ܪܲܒܵܐ ܕܲܚܢܵܢܵܟ݂. ܕܲܡܚܝܼܠ ܓܸܢܣܲܢ: ܒܪ̈ܲܚܡܲܝܟ ܢܸܬ݂ܚܢܸܢ. ܕܐܸܥܘܿܠ ܒܗܘܿܢ ܘܐܵܘܕܹܐ ܠܡܵܪܝܵܐ. ܒܲܬ݂ܪܲܥ ܪ̈ܲܚܡܲܝܟ: ܢܵܩܫܝܼܢܲܢ ܡܵܪܵܐ ܛܵܒ݂ܵܐ ܘܒܲܣܝܼܡܵܐ. ܦܬܲܚܠܲܢ ܘܦܲܢܵܐ ܫܹ̈ܐܠܵܬܲܢ: ܐܲܝܟ ܕܐܸܫܬܵܘܕܝܼܬ݁ ܒܲܣܒܲܪܬܵܟ݂. ܕܟ݂ܠ ܐܲܝܢܵܐ ܕܫܵܐܹܠ ܢܵܣܹܒ݂: ܘܐܲܝܢܵܐ ܕܢܵܩܹܫ ܡܸܬ݂ܦܬܲܚ ܠܹܗ. ܣܲܓ݁ܝܼ ܪ̈ܲܚܡܹܐ: ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ. ܠܸܒܵܐ ܕܲܟ݂ܝܵܐ ܒܪܝܼ ܒܝܼ ܐܲܠܵܗܵܐ. ܗܲܒ݂ܠܲܢ ܡܵܪܲܢ: ܒܲܚܢܵܢܵܟ݂ ܕܢܸܬ݂ܦܢܸܐ ܟܠܲܢ ܠܝܼܕܲܥܬ݂ܵܟ݂. ܟܲܕ݂ ܫܵܐܠܝܼܢܲܢ ܪ̈ܲܚܡܲܝܟ: ܫܘܼܒ݂ܩܵܢܵܐ ܕܚܵܘ̈ܒܹ݁ܐ. ܘܲܒ݂ܝܲܕ݂ ܙܘܿܦܵܟ݂ ܢܸܬ݂ܚܵܘܲܪ: ܡܢ ܟܘܼ̈ܬ݂ܡܵܬ݂ܵܐ ܕܲܚܛܝܼܬ݂ܵܐ. ܪܵܚܹܡ ܐ݇ܢܵܫܵܐ: ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ. ܒܪܲܡܫܵܐ ܘܲܒ݂ܨܲܦܪܵܐ ܘܲܒ݂ܛܲܗܪܵܐ. ܐܲܫܘܵܢ ܡܵܪܲܢ: ܒܲܚܢܵܢܵܟ݂ ܕܢܸܦܠܘܿܚ ܒܟܲܪܡܵܐ ܕܲܣܒܲܪܬ݂ܵܟ݂. ܘܢܸܣܲܒ݂ ܟܲܟ݁ܪܹ̈ܐ ܕܝܘܼܠܦܵܢܵܐ: ܡܢ ܓܲܙܵܐ ܕܣܸܦܪ̈ܲܝܟ. ܘܢܲܪܡܸܐ ܡܸܫܚܵܐ ܒܠܲܡܦܹ̈ܝܕܲܝܢ: ܕܲܠܣܲܟ݂̈ܠܵܬ݂ܵܐ ܠܵܐ ܢܸܕ݂ܡܸܐ. ܚܲܬ݂ܢܵܐ ܕܪܵܘܡܵܐ: ܐܲܫܘܵܢ ܠܲܓ݂ܢܘܿܢܵܟ݂. ܩܪܹܝܬ݂ܵܟ݂ ܡܵܪܝܵܐ ܒܟ݂ܠܝܘܿܡ. ܠܵܟ݂ ܩܵܪܹܝܢܲܢ: ܐܲܠܵܗܲܢ ܛܵܒ݂ܵܐ ܪܲܚܸܡ ܥܲܠ ܚܲܝܲܝ̈ܢ. ܘܐܲܫܦܲܥ ܪ̈ܲܚܡܹܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܥܲܠ ܥܲܒ݂ܕܲܝ̈ܟ ܡܵܪܝܵܐ. ܘܐܲܣܵܐ ܟܹܐܒܹ̈ܐ ܘܟ݂ܘܼܪ̈ܗܵܢܹܐ: ܕܐܲܝܠܹܝܢ ܕܲܪܡܹܝܢ ܒܢܸܣܝܘܿܢܹ̈ܐ. ܡܵܪܵܐ ܛܵܒ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ. ܡܸܛܠ ܕܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܛܵܒ݂ ܐܲܢ݇ܬ݁. ܛܵܒ݂ܵܐ ܘܟܹܐܢܵܐ: ܘܚܲܢܵܢܵܐ ܥܠܲܝܟ݁ܘܼ ܐܝܼܬ݂ܵܘܗܝ ܬܘܼܟ݂ܠܵܢܲܢ. ܕܐܲܢ݇ܬ݁ܘܼ ܒܲܠܚܘܿܕ݂ ܝܵܕܲܥ ܐܲܢ݇ܬ݁: ܟܠܗܘܿܢ ܚܲܫܲܝ̈ܢ ܘܟ݂ܘܼܪ̈ܗܵܢܲܝܢ. ܬܵܐ ܠܐܝܼܵܠܲܢ ܘܲܦܪܘܿܩܲܝܢ: ܘܫܵܘܙܸܒ݂ ܚܲܝܲܝ̈ܢ ܡܢ ܚܒ݂ܵܠܵܐ. ܕܐܲܢ݇ܬ݁ܘܼ ܒܲܠܚܘܿܕ݂: ܣܲܓ݁ܝܼ ܛܲܝܒ݁ܘܼܬ݂ܵܐ. ܬܵܘܫܸܛ ܐܝܼܕ݂ܵܟ݂ ܘܬܸܦ̄ܪܩܲܢܝ. ܒܝܲܡܹܗ ܕܥܵܠܡܵܐ: ܥܵܒ݂ܘܿܪܵܐ ܡܸܬ݁ܛܲܪܲܦ ܐ݇ܢܵܐ ܡܵܪܹܟܠ. ܐܵܘܫܸܛ ܐܝܼܕ݂ܵܐ ܕܥܘܼܕܪ̈ܵܢܹܐ: ܘܕܲܠܵܢܝ ܡܸܢܹܗ ܒܲܚܢܵܢܵܟ݂. ܐܲܝܟ݂ ܕܲܠܦܲܛܪܘܿܣ ܕܲܩܪܲܝܬܵܝܗܝ: ܘܡܢ ܓܵܘ ܝܲܡܵܐ ܕܲܠܝܼܬܵܝܗܝ. ܕܥܲܡܹܗ ܐܸܙܡܲܪ: ܫܘܼܒ݂ܚܵܐ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܐܲܤܵܢܝ ܡܵܪܝܵܐ ܡܸܛܠ ܕܙܵܥܘ ܓܲܪ̈ܡܲܝ. ܒܟ݂ܘܼܪܗܵܢܵܐ ܥܲܣܩܵܐ: ܡܸܬ݂ܢܵܘܲܠ ܐ݇ܢܵܐ ܡܵܪܵܐ ܛܵܒ݂ܵܐ ܘܒܲܣܝܼܡܵܐ. ܫܲܕܲܪ ܚܲܝܠܵܟ݂ ܡܢ ܪܵܘܡܵܐ: ܘܢܲܐܣܸܐ ܟܹܐܒܲܝ̈ ܘܟ݂ܘܼܪ̈ܗܵܢܲܝ. ܕܐܲܢ݇ܬ݁ܘܼ ܐܵܣܝܵܐ ܫܲܪܝܼܪܵܐ: ܕܡܲܐܣܸܐ ܟܠܫܵܥ ܡܡܲܚܲܝ̈ܵܐ. ܐܲܣܵܢܝ ܒܪ̈ܲܚܡܲܝܟ: ܘܲܒ݂ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܬܸܥܘܿܠ ܩܕ݂ܵܡܲܝܟ ܨܠܘܿܬ݂ܝ. ܨܠܘܿܬܲܢ ܩܲܒܸܠ: ܦܵܪܘܿܩܲܢ ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹ̈ܐܠܵܬܲܢ. ܘܲܨܠܝܼܒ݂ܵܟ݂ ܢܸܗܘܸܐ ܫܘܼܪܵܐ ܠܟܸܢܫܲܢ ܡܢ ܒܝܼܫܵܐ ܘܚܲܝܠܹܗ. ܘܢܸܗܘܸܐ ܠܵܟ݂ ܥܲܒ݂̈ܕܹܐ ܛܵܒܹ̈ܐ: ܘܢܲܣܸܩ ܫܘܼܒ݂ܚܵܐ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܫܵܘܹܐ ܠܫܘܼܒ݂ܚܵܐ: ܡܢ ܟܠ ܦܘܼ̈ܡܝܼܢ. ܨܘܼܬ݂ ܡܸܠܲܝ̈ ܘܩܲܒܸ݁ܠ. ܡܵܪܲܢ ܩܲܒܸ݁ܠ: ܒܵܥܘܼܬܲܢ ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹ̈ܐܠܵܬܲܢ. ܘܗܲܒ݂ܠܲܢ ܙܲܒ݂ܢܹ̈ܐ ܕܲܢܝܵܚܵܐ: ܡܢ ܩܕ݂ܵܡ ܦܲܪܨܘܿܦ ܡܵܪܘܼܬ݂ܵܟ݂. ܕܢܵܘܕܸܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂: ܝܵܗܘܿܒ݂ܵܐ ܕܛܵܒ݂̈ܵܬܲܢ. ܡܵܪܵܐ ܛܵܒ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ. ܡܲܠܟܵܐ ܕܪܲܒ ܥܲܠ ܟܠܗܘܿܢ ܐܲܠܵܗܹ̈ܐ. ܡܲܠܟܵܐ ܡܫܝܼܚܵܐ: ܕܒܲܨܠܝܼܒ݂ܵܟ݂ ܐܸܚܕܹܬ݂ ܓܵܘܣܵܐ ܠܵܐ ܬܲܪܦܹܝܢܝ. ܒܐܝܼ̈ܕܲܝ ܫܹܐܕܹ̈ܐ ܡܵܪ̈ܘܿܕܹܐ: ܕܟ݂ܠܫܵܥ ܕܲܠܩܘܼܒ݂ܠܵܐ ܩܵܝܡܝܼܢ. ܐܸܠܵܐ ܫܲܪܲܪ ܚܘܼ̈ܫܵܒܲܝ: ܠܲܡܥܲܙܵܝܘܼ ܩܕ݂ܵܡ ܚܲܫܹ̈ܐ. ܘܐܲܫܘܵܢܝ ܕܐܸܣܲܒ݂: ܟܠܝܼܠ ܙܵܟ݂ܘܼܬ݂ܵܐ. ܡܢ ܦܘܼܡܵܐ ܕܲܥܠܲܝܡܹ̈ܐ ܘܕܲܛ̈ܠܵܝܹܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ: ܘܬ݂ܵܘܕܝܼܬ݂ܵܐ ܠܲܒ݂ܪܵܐ ܘܣܸܓ݂ܕܬܵܐ ܠܪܘܼܚܩܘܼܕ݂ܫܵܐ. ܟܝܵܢܵܐ ܣܓ݂ܝܼܕ݂ܵܐ ܘܲܫܒ݂ܝܼܚܵܐ: ܕܲܒ݂ܛܲܝܒ݁ܘܼܬܹܗ ܡܕܲܒܲܪ ܠܟ݂ܠ. ܘܡܲܕ݂ܢܲܚ ܫܸܡܫܵܐ ܕܛܵܒ݂ܘܼܬܹܗ: ܥܲܠ ܟܠܗܘܿܢ ܥܒ݂ܝܼ̈ܕܹܐ. ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ: ܘܲܥܠܲܝܢ ܪ̈ܲܚܡܵܘܗܝ. ܩܲܕܝܼܫܘܼ ܘܲܕ݂ܚܝܼܠ ܫܡܹܗ. ܩܲܕܝܼܫ ܡܢ ܟܠ: ܕܲܡܣܲܝܸܟ݂ ܥܲܒ݂ܕܵܘ̈ܗܝܔ ܘܗܘ ܠܵܐ ܡܸܣܬܲܝܲܟ݂. ܒܲܪܸܟ݂ ܒܵܬܹ̈ܐ ܕܲܡܗܲܝܡܢܹ̈ܐ: ܕܐܵܘܕܝܼܘ ܒܲܨܠܝܼܒ݂ܵܟ݂. ܘܐܲܣܓܵܐ ܠܘܵܬ݂ܗܘܿܢ ܥܘܼܕ݂ܪ̈ܵܢܲܝܟ: ܐܲܝܟ݂ ܕܲܒ݂ܒܲܝܬܹ݁ܗ ܕܐܲܒ݂ܪܵܗܵܡ. ܘܢܲܣܩܘܼܢ ܫܘܼܒ݂ܚܵܐ: ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂. ܬܵܘ ܢܸܒ݂ܪܘܿܟ݂ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܹܗ. ܠܵܟ݂ ܣܵܓ݂ܕܝܼܢܲܢ: ܒܲܪ ܛܵܒ݂ܵܐ ܕܡܢ ܡܬ݂ܘܿܡ ܠܵܐ ܡܲܗܡܸܐ ܡܢ ܥܲܒ݂ܕܵܘ̈ܗܝ: ܒܲܛܸܠ ܩܪ̈ܵܒܹܐ ܘܬܲܟ݂ܬܘܼܫܹ̈ܐ: ܘܩܲܝܸܡ ܥܹܕܬܵܟ݂ ܒܲܨܠܝܼܒ݂ܵܟ݂. ܘܢܲܛܲܪ ܝܲܠܕܹܝ̈ܗ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܕܒܲܕ݂ܡܵܟ݂ ܚܲܝܵܐ ܐܸܙܕܲܒ݂ܢܲܬ݀. ܕܬܲܣܸܩ ܫܘܼܒ݂ܚܵܐ: ܠܲܫܡܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ. ܦܬܲܚܠܝܼ ܬܲܪ̈ܥܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ. ܦܬܲܚ ܦܵܪܘܿܩܲܢ: ܒܹܝܬ݂ ܓܲܙܵܐ ܘܐܲܥܬܲܪ ܠܐܲܝܠܹܝܢ ܕܲܣܢܝܼܩܝܼܢ. ܘܦܲܢܵܐ ܫܹ̈ܐܠܵܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ: ܕܲܥܠܲܝܟ ܐܝܼܬ݂ܵܘܗܝ ܬܘܼܟ݂ܠܵܢܗܘܿܢ. ܘܚܲܣܵܐ ܘܲܥܛܝܼ ܚܛܵܗܲܝ̈ܗܘܿܢ: ܘܐܲܚܠܸܡ ܒܪ̈ܲܚܡܲܝܟ ܟܹܐܒܲܝ̈ܗܘܿܢ. ܕܢܲܣܩܘܼܢ ܫܘܼܒ݂ܚܵܐ: ܠܲܫܡܵܟ݂ ܒܟ݂ܠܥܸܕܵܢ. ܡܢ ܒܲܣܝܼܡܘܼܬ݂ܵܐ ܕܙܵܟ݂ܘܼܬ݂ܵܐ ܕܝܲܡܝܼܢܵܟ݂. ܡܢ ܒܹܝܬ݂ ܓܲܙܵܐ: ܫܵܐܠܝܼܢܲܢ ܪ̈ܲܚܡܹܐ ܘܫܘܼܒ݂ܩܵܢ ܚܵܘ̈ܒܵܬ݂ܵܐ. ܦܬܲܚܠܲܢ ܘܦܲܢܵܐ ܫܹ̈ܐܠܵܬܲܢ: ܐܲܝܟ݂ ܕܲܡܥܵܕܵܐ ܗ݇ܝܼ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܘܚܘܼܢ ܠܐܲܠܝܼܨܹ̈ܐ ܘܠܲܡܛܲܪ̈ܦܹܐ: ܐܵܦ ܠܲܡܥܵܩܹ̈ܐ ܕܩܵܪܹܝܢ ܠܵܟ݂. ܕܐܲܢ݇ܬ݁ܘܼ ܒܲܠܚܘܿܕ݂: ܚܵܐܹܢ ܠܚܲܛܵܝܹ̈ܐ. ܗܵܘ ܕܐܝܼܬ݂ܵܘܗܝ ܡܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ. ܐܲܢ݇ܬ݁ܘܼ ܡܵܪܲܢ: ܒܹܝܬ݂ ܓܵܘܣܵܐ ܪܲܒܵܐ ܘܣܲܒ݂ܪܵܐ ܫܲܪܝܼܪܵܐ. ܘܲܦܬ݂ܝܼܚ ܬܲܪܥܹܗ ܕܲܚܢܵܢܵܟ݂: ܠܟ݂ܠܗܘܿܢ ܕܐܲܝܠܹܝܢ ܕܩܵܪܹܝܢ ܠܵܟ݂. ܘܡܲܓܵܢ ܝܵܗ݇ܒܲܬ݁ ܥܘܼܕ݂ܪ̈ܵܢܹܐ: ܘܐܵܣܝܘܼܬ݂ܵܐ ܠܟܹܐܒܲܝ̈ܢ. ܡܵܘܪܸܒ݂ ܓܸܢܣܲܢ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂. ܐܸܬ݁ܕܲܟ݂ܪ ܥܹܕܬܵܟ݂ ܕܲܩܢܲܝܬ݁ ܡܢ ܩܕ݂ܝܼܡ. ܚܘܼܣ ܥܲܠ ܥܹܕܬܵܟ݂: ܦܵܪܘܿܩܲܢ ܕܡܸܬ݂ܟܲܫܦܵܐ ܠܵܟ݂ ܒܟ݂ܠܥܸܕܵܢ. ܘܗܲܒ݂ܠܵܗ ܚܲܝܠܵܐ ܕܠܵܐ ܡܸܙܕܟܸܐ: ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܕܠܵܐ ܡܸܫܬܲܬ݁ܩܵܐ. ܘܫܲܥܒܸ݁ܕ݂ ܩܘܼܕ݂ܡܹܝܗ ܒܥܸܠܕܒ݂ܵܒܹܝ̈ܗ: ܒܚܲܝܠܵܟ݂ ܪܲܒܵܐ ܙܵܟܹ݁ܐ ܠܟ݂ܠ. ܕܢܸܕܥܘܼܢ ܥܲܡܡܹ̈ܐ: ܕܐܲܢ݇ܬ݁ܘܼ ܐܲܠܵܗܵܗ. ܕܐܘܼܪܚܵܐ܀ ܟܠܗܘܿܢ ܒܚܸܟ݂ܡܬ݂ܵܐ ܒܪܲܝܬ݁ ܐܸܢܘܿܢ. ܝܲܡܵܐ ܘܝܲܒ݂ܫܵܐ: ܒܦܘܼܩܕܵܢܵܟ݂ ܡܵܪܲܢ ܡܸܬ݂ܩܲܝܡܝܼܢ. ܘܐܲܝܠܹܝܢ ܕܒ݂ܐܘܼܪ̈ܚܵܬ݂ܵܐ ܪܵܕܹܝܢ: ܘܲܡܣܲܒ݁ܪܝܼܢܒܵܟ݂ ܠܵܐ ܒܵܗܬܝܼܢ. ܘܲܚܢܲܢ ܗܵܫܵܐ ܕܥܲܠ ܣܲܒ݂ܪܵܟ݂: ܒܐܘܼܪܚܵܐ ܗܵܕܹܐ ܚܵܙܩܝܼܢܲܢ. ܚܵܘܵܢ ܐܘܼܪܚܵܟ݂: ܘܗܲܕܵܢ ܒܲܫܒ݂ܝܼܠܲܝ̈ܟ݂. ܕܲܒܲܪܲܝܢܝ ܒܲܫܒ݂ܝܼܠܲܝ̈ܟ ܬܪ̈ܝܼܨܹܐ. ܠܡܹܐܢܵܐ ܢܝܼܚܵܐ: ܘܲܡܫܲܝܢܵܐ ܛܲܝܸܒ݂ ܠܝܼ ܡܵܪܲܢ. ܒܐܘܼܪܚܵܐ ܗܵܕܹܐ ܕܫܲܪܝܼܬ݁ ܒܵܗ: ܕܐܸܚܙܘܿܩ ܘܐܸܦܢܸܐ ܒܲܫܠܵܡܵܐ. ܘܢܲܨܠܲܚ ܣܘܼܥܪܵܢܝ ܟܹܐܢܵܐܝܼܬ݂: ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܵܐ ܕܡܵܪܘܼܬ݂ܵܟ݂. ܡܫܝܼܚܵܐ ܐܘܼܪܚܵܐ: ܗܲܕܵܢܝ ܒܲܫܒ݂ܝܼ̈ܠܲܝܟ. ܕܡܸܛܪܵܐ܀ ( 1 ) ܗܵܢܵܐ ܝܲܡܵܐ ܪܲܒܵܐ ܘܲܪܘܲܚܬ݂ܵܐ ܕܐܝܼ̈ܕܲܝܵܐ. ܝܲܡܵܐ ܕܪ̈ܲܚܡܹܐ: ܠܵܐ ܬܸܡܚܹܝܗ ܠܐܲܪܥܵܐ ܒܫܲܒ݂ܛܵܐ ܕܡܲܪܕܘܼܬ݂ܵܐ. ܘܐܸܢܗܘ ܕܒ݂ܝܼܫܝܼܢ ܥܵܡܘܿܪܹ̈ܝܗ: ܕܝܼܠܵܟ݂ ܫܦܝܼܥܵܐ ܗ݇ܝܔܼ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܣܥܘܿܪܹܝܗ ܡܵܪܝ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁: ܒܡܸܛܪܵܐ ܢܝܼܚܵܐ ܕܨܸܒ݂ܝܵܢܵܐ. ܘܢܸܚܘܿܢ ܝܲܠܕܹܝ̈ܗ: ܘܲܢܫܲܒ݁ܚܘܼܢܵܟ݂. ܥܵܒܹܕ݂ ܐܲܢ݇ܬ݁ ܬܸܕ݂ܡܪ̈ܵܬ݂ܵܐ ܐܲܢ݇ܬ݁ ܐܲܠܵܗܵܐ ܒܲܠܚܘܿܕܲܝܟ. ܐܲܢ݇ܬ݁ܘܼ ܕܲܟ݂ܠܲܝܬ݁: ܠܲܫܡܲܝܵܐ ܘܠܵܐ ܢܚܸܬ݂ ܡܸܛܪܵܐ ܒܙܲܒ݂ܢܹܗ ܕܐܹܠܝܼܵܐ. ܘܲܒ݂ܝܲܕ݂ ܨܠܘܿܬܹܗ ܕܙܲܕܝܼܩܵܐ: ܓܵܠܘ ܗ݇ܘܵܘ ܪ̈ܲܚܡܲܝܟ ܥܠܲܝܗܘܿܢ. ܐܵܦܠܲܢ ܡܵܪܝ ܕܠܵܟ݂ ܩܵܪܹܝܢܲܢ: ܗܲܒ݂ܠܲܢ ܡܸܛܪܵܐ ܕܨܸܒ݂ܝܵܢܵܐ. ܘܒܲܪܸܟ݂ ܟܠܝܼܠܵܐ: ܕܫܲܢ݇ܬܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܕܡܲܪܝܲܡ܀ ܬܹܒܹܝܠ ܘܟ݂ܠܗܘܿܢ ܥܵܡܘܿܪܹ̈ܝܗ. ܪܵܘܡܵܐ ܘܥܘܼܡܩܵܐ: ܘܟ݂ܠ ܕܲܒ݂ܗܘܿܢ ܙܵܡܪܝܼܢ ܫܘܼܒ݂ܚܵܐ ܒܕ݂ܘܼܟ݂ܪܵܢܵܗ. ܕܡܲܪܝܲܡ ܐܸܡܵܐ ܡܒܲܪܲܟ݂ܬ݂ܵܐ: ܡܥܝܼܢܵܐ ܡܲܪܕܸ݁ܐ ܥܘܼܕ݂ܪ̈ܵܢܹܐ. ܠܐܲܝܠܹܝܢ ܕܡܸܬ݂ܓܵܘܣܝܼܢ ܒܲܫܡܵܗ: ܘܐܵܚܕܝܼܢ ܓܵܘܣܵܐ ܒܲܨܠܘܿܬ݂ܵܗ. ܘܩܵܥܹܝܢ ܐܲܟܲܚ݇ܕ݂: ܒܪܝܼܟ݂ ܕܘܼܟ݂ܪܵܢܹܟ݂ܝ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܐܹܪܲܡܪܡܵܟ݂ ܡܵܪܝ ܡܲܠܟܵܐ. ܡܫܝܼܚܵܐ ܕܐܵܘܪܸܒ݂: ܠܩܲܕ݁ܝܼܫܵܘ̈ܗܝ ܘܲܢܨܲܚ ܐܸܢܘܿܢ ܒܟ݂ܠ ܕܵܪ̈ܝܼܢ. ܗܲܒ݂ܠܲܢ ܚܲܝܠܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܕܲܢܡܲܪܸܐ ܒܕܘܼܒܵܪ̈ܲܝܗܘܿܢ. ܘܢܸܗܘܸܐ ܥܲܡܗܘܿܢ ܝܵܪ̈ܘܿܬܹܐ: ܒܥܘܼܡܪܵܐ ܦܲܐܝܵܐ ܕܲܫܡܲܝܵܐ. ܡܵܘܪܸܒ݂ ܥܲܒ݂ܕܵܘ̈ܗܝ: ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܡܸܟܹ݁ܝܠ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ. ܨܠܝܼܒ݂ܵܟ݂ ܡܵܪܲܢ: ܢܸܗܘܸܐ ܢܵܛܘܿܪܵܐ ܠܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕ݂ܵܘ̈ܗܝ. ܘܲܬܚܹܝܬ݂ ܟܸܢܦܵܘ̈ܗܝ ܢܸܣܬܲܬܲܪ: ܡܢ ܨܸܢ̈ܥܵܬܹ̈ܗ ܕܲܢܟ݂ܝܠܵܐ. ܘܡܵܐ ܡܸܙܕܲܝܲܚ ܒܬܸܫܒ݁ܘܿܚܬܵܐ: ܒܝܵܘܡܵܐ ܐ݇ܚܪܵܝܵܐ ܕܓܸܠܝܵܢܵܟ݂. ܢܸܚܙܹܝܘܗܝ ܬܲܡܵܢ: ܘܢܵܘܕܸ݁ܐ ܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܕܐܲܒ݂ܘܼܢ܀ ܐܹܥܲܫܢܝܼܘܗܝ ܘܐܹܝܲܩܪܝܼܘܗܝ. ܠܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ: ܘܩܲܕܝܼܫܵܐ ܙܵܕܹܩ ܠܟ݂ܠܲܢ ܕܲܢܝܲܩܲܪ. ܕܲܗܘܵܐ ܡܵܘܬܹ݁ܗ ܠܥܘܼܕ݂ܪܵܢܵܐ: ܒܥܹܕܬܵܐ ܡܟ݂ܝܼܪܬܹܗ ܕܲܡܫܝܼܚܵܐ. ܘܗܵܐ ܐܝܼܬ݂ܵܘܗܝ ܦܲܓ݂ܪܹܗ ܙܲܗܝܵܐ: ܢܸܒ݂ܥܵܐ ܡܲܪܕܸܐ ܥܘܼܕ݂ܪ̈ܵܢܹܐ: ܠܟ݂ܠ ܕܐܲܠܝܼܨܝܼܢ: ܘܒܹܗ ܡܬ݂ܓܵܘܣܝܼܢ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܡܸܛܠ ܐܝܼܩܵܪܵܐ ܕܲܫܡܵܟ݂. ܐܲܢܝܼܚ ܡܵܪܲܢ: ܒܲܓ݂ܢܘܿܢܵܟ݂ ܠܥܲܢܝܼ̈ܕܹܐ ܕܠܲܒ݂ܫܘܼܟ݂. ܘܕܐܸܫܬܵܘܬܲܦܘ ܒܦܲܓ݂ܪܵܟ݂ ܘܲܕ݂ܡܵܟ݂: ܒܥܹܕܬܵܐ ܗܵܕܹܐ ܕܥܲܠ ܐܲܪܥܵܐ. ܫܵܘܬܸܦ ܐܸܢܘܿܢ ܒܡܲܠܟܘܼܬ݂ܵܟ݂: ܚܲܝܹ̈ܐ ܘܛܘܼܒܹ̈ܐ ܐܲܡܝܼܢܹ̈ܐ. ܕܐܲܢ݇ܬ݁ܘܼ ܐܝܼܬܲܝܟ: ܣܲܒ݂ܪܵܐ ܕܲܩܒ݂ܝܼܪܹ̈ܐ. ܡܗܲܝܡܲܢܘ ܡܵܪܝܵܐ ܒܡܸܠܵܘ̈ܗܝ. ܐܲܝܟ݂ ܕܐܸܫܬܵܘܕܝܼܬ݁: ܦܵܪܘܿܩܲܢ ܢܲܚܸܡ ܘܐܲܚܵܐ ܠܥܲܢ̈ܝܼܕܲܝܢ. ܘܐܲܠܒܸ݁ܫ ܫܘܼܒ݂ܚܵܐ ܦܲܓ݂ܪ̈ܲܝܗܘܿܢ: ܒܝܵܘܡܵܐ ܪܲܒܵܐ ܕܲܩܝܵܡܬܵܐ. ܘܲܢܩܘܼܡܘܼܢ ܡܢ ܝܲܡܝܼܢܵܟ݂: ܘܢܹܐܪܬ݁ܘܼܢ ܛܘܼܒܹ̈ܐ ܕܠܵܐ ܦܵܛܪܝܼܢ. ܡܵܪܵܐ ܕܡܵܘܬܲܢ: ܝܵܗܘܿܒ݂ܵܐ ܕܚܲܝܲܝ̈ܢ ܀
ܕܥܲܢܝܼ̈ܕܹܐ. ܐܵܘܕܵܘ ܠܹܗ ܘܒܲܪܸܟ݂ ܠܲܫܡܹܗ. ܐܵܘܕܵܘ ܠܹܗ ܡܵܝܘܿܬܹ̈ܐ ܠܲܒ݂ܪܵܐ ܕܦܲܪܩܲܢ. ܡܢ ܫܘܼܥܒܵܕܹܗ ܕܡܵܘܬܵܐ. ܕܐܲܚܝܼܕ݂ ܗ݇ܘܵܐ ܠܲܢ ܒܲܚܛܵܗܲܝ̈ܢ. ܟܠܹܗ ܓܹܝܪ ܕܲܢܚܸܬ݂ ܠܲܫܝܘܿܠ. ܕܢܲܚܸܐ ܡܝܼܬܹ̈ܐ ܡܢ ܩܲܒ݂ܪܹ̈ܐ. ܡܲܢܘܼ ܡܨܸܐ ܦܵܪܲܥ ܛܲܝܒ݁ܘܼܬ݂ܵܐ. ܕܲܗܘܵܬ݀ ܒܹܗ ܒܓܸܢܣܵܐ ܕܡܵܝܘܿܬܹ̈ܐ. ܦܘܼܡܝ ܢܡܲܠܸܠ ܚܸܟ݂ܡܬ݂ܵܐ. ܐܸܬ݂ܓܵܘܲܣܘ ܚܲܛܵܝܹ̈ܐ ܒܲܬ݂ܝܵܒ݂ܘܼܬ݂ܵܐ. ܕܩܲܠܝܼܠ ܗ݇ܘܼ ܓܹܝܪ ܙܲܒ݂ܢܵܐ. ܡܗܲܒܸ݁ܒ݂ ܥܵܠܡܵܐ ܘܥܵܒܲܪ ܠܹܗ. ܛܘܼܒ݂ܵܐ ܠܬܲܝܵܒܹ̈ܐ ܘܕ݂ܝܼܢܵܐ ܠܥܵܘܵܠܹ̈ܐ ܕܠܵܐ ܐܸܙܕܲܕܲܩܘ. ܕܐܸܢ ܡܵܪܝ ܟܹܐܢܵܐܝܼܬ݂ ܕܵܐܸܢ ܐܲܢ݇ܬ݁ ܡܲܢܘܼ ܡܨܸܐ ܙܵܟܹ݁ܐ ܒܹܝܬ݂ ܕܝܼܢܵܐ.
ܫܲܒܲܚ. ܡܢ ܢܸܬܸ݁ܠ ܠܝܼ ܡܲܒ݁ܘܼܥܵܐ ܕܣܘܿܓ݂ܵܐܐ ܕܕܸܡܥܹ̈ܐ. ܘܚܘܼܫܵܒܹ̈ܐ ܕܪܸܚܡܲܬ݂ ܢܲܟ݂ܦܘܼܬ݂ܵܐ. ܕܐܸܥܛܸܐ ܒܲܬ݂ܝܵܒ݂ܘܼܬ݂ܵܐ ܚܵܘܒܲܝ̈ ܘܲܚܛܵܗܲܝ̈ ܕܲܣܥܝܼܪܝܼܢ ܠܝܼ ܒܟ݂ܠ ܥܸܕܵܢ. ܣܢܝܼܩ ܥܲܠ ܪ̈ܲܚܡܲܝܟ ܐܝܼܬܲܝ ܛܵܒ݂ܵܐ ܘܒܲܣܝܼܡܵܐ. ܡܸܛܠ ܕܡܲܚܒܲܬ݁ ܠܐܲܝܠܹܝܢ ܕܬ݂ܵܝܒ݁ܝܼܢ. ܘܡܸܛܠ ܗܵܢܵܐ ܠܵܐ ܦܵܣܹܩ ܐ݇ܢܵܐ ܣܲܒ݂ܪܵܐ ܕܚܲܝܲܝ̈ ܡܵܪܝܵܐ.
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܒܵܚܵܐ. ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܢ ܥܸܠܵܝܹ̈ܐ: ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܢ ܬܲܚܬܵܝܹ̈ܐ: ܘܣܸܓ݂ܕܬ݂ܵܐ ܘܗܸܕ݂ܪܵܐ ܘܪܘܼܡܪܵܡܵܐ: ܡܢ ܟܠ ܕܲܒ݂ܪܲܝܬ݁: ܘܐܲܬ݂ܩܸܢܬ݁ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ. ܡܵܪܵܐ ܕܟ݂ܠ.
 ܫܘܼܒܵܚܵܐ. ܦܪܘܿܩ ܡܵܪܝܵܐ ܡܸܛܠ ܕܲܓ݂ܡܲܪ ܛܵܒ݂ܵܐ* ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܬܢܝܼ ܥܲܡ ܟܠ ܦܸܬ݂ܓ݂ܵܡܵܐ. ܘܒܸܛܠܲܬ݀ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܡܢ ܐܲܪܥܵܐ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܒܢܲܝ̈ܢܵܫܵܐ ܣܪܝܼܩܘܼܬ݂ܵܐ ܡܡܲܠܠܝܼܢ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܘܓܲܒ݂ܪܵܐ ܥܲܡ ܚܲܒ݂ܪܹܗ ܒܣܸܦ̈ܘܵܬ݂ܵܐ ܡܦܲܠ̈ܓ݂ܵܬ݂ܵܐ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܒܠܸܒܵܐ ܘܲܒ݂ܠܸܒܵܐ ܡܡܲܠܠܝܼܢ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܢܵܘܒܸܕ ܡܵܪܝܵܐ ܟܠܗܹܝܢ ܣܸܦ̈ܘܵܬ݂ܵܐ ܡܦܲܠ̈ܓ݂ܵܬ݂ܵܐ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܘܠܸܫܵܢܹ̈ܐ ܕܲܡܡܲܠܠܝܼܢ ܪܵܘܪ̈ܒ݂ܵܬ݂ܵܐ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܕܐܸܡܲܪܘ ܠܸܫܵܢܲܢ ܢܵܘܪܸܒ݂ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܘܣܸܦ̈ܘܵܬ݂ܵܐ ܕܝܼܠܲܢ ܐܸܢܘܿܢ ܡܲܢܘܼ ܡܵܪܲܢ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܡܸܛܠ ܒܸܙܬ݂ܵܐ ܕܡܸܣܟܹܢܹ̈ܐ ܘܐܸܢܲܩܬ݂ܵܐ ܕܒ݂ܵܝܫܹ̈ܐ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܡܸܟܹ݁ܝܠ ܐܹܩܘܼܡ ܐܵܡܲܪ ܡܵܪܝܵܐ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܘܐܸܥܒܸ݁ܕ݂ ܦܘܼܪܩܵܢܵܐ ܓܲܠܝܵܐܝܼܬ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܡܹܐܡܪܹܗ ܕܡܵܪܝܵܐ ܡܹܐܡܪܵܐ ܕܲܟ݂ܝܵܐ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܣܹܐܡܵܐ ܓܲܒ݂ܝܵܐ ܕܲܒ݂ܚܝܼܪ ܒܐܲܪܥܵܐ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܘܲܡܕܲܟܲܝ ܚܲܕ݂ ܒܫܲܒ݂ܥܵܐ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܘܐܲܢ݇ܬ ܡܵܪܝܵܐ ܬܸܛܲܪ ܐܸܢܘܿܢ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܫܵܘܙܸܒܲܝܢܝ ܘܦܲܨܵܢܝ ܡܢ ܕܵܪܵܐ ܗܵܢܵܐ ܠܥܵܠܲܡ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܕܲܟ݂ܪܝܼܟ݂ܝܼܢ ܪ̈ܲܫܝܼܥܹܐ ܘܲܡܗܲܠܟ݂ܝܼܢ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܐܲܝܟ݂ ܪܵܘܡܵܐ ܙܲܠܝܼܠܵܐ ܕܲܒ݂ܢܲܝ̈ ܐܲܕ݂ܘܿܡ ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ*
ܫܲܒܲܚ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ* ܥܵܠܲܡ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ*
ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܬܢܝܼ ܥܲܡ ܟܠ ܦܸܬ݂ܓ݂ܵܡܵܐ. ܒܩܵܠܵܐ ܚܢܝܼܓ݂ܵܐ: ܘܠܵܘ ܕܲܥܪ̈ܘܼܒ݂ܵܬܵܐ: ܒܟ݂ܠܗܹܝܢ ܫܲܒܹ̈ܐ ܕܐ݇ܪ̈ܵܙܹܐ.
 ܬܸܫܒ݁ܘܿܚܬܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ܀ ܒܐܝܼܡܵܡܵܐ ܘܲܒ݂ܠܸܠܝܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂܀ ܒܟ݂ܠܙܒܲܢ ܘܲܒ݂ܟ݂ܠܥܸܕܵܢ. ܒܪ̈ܲܚܡܲܝܟ ܚܘܼܢܲܝܢ ܐܲܠܵܗܵܐ܀ ܫܵܐܠܹܝܢܲܢ ܡܸܢܵܟ݂ ܪ̈ܲܚܡܹܐ: ܘܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ܀ ܒܲܛܸܠܲܝܗܝ ܠܒ݂ܝܼܫܵܐ ܡܸܢܲܢ: ܕܟ݂ܠܫܵܥ ܦܲܚܹ̈ܐ ܨܵܠܹܐ ܠܲܢ܀ ܢܲܛܲܪ ܠܲܢ ܒܢܝܼܫܵܟ݂ ܚܲܝܵܐ. ܕܢܸܚܙܸܐ ܒܝܼܫܵܐ ܘܲܢܫܲܢܸܐ܀ ܘܝܲܡܝܼܢܵܟ݂ ܕܡܲܠܝܵܐ ܪ̈ܲܚܡܹܐ. ܡܵܪܝܵܐ ܐܲܓܸ݁ܢ ܥܠܲܝܢ܀ ܡܵܪܝܵܐ ܪ̈ܲܚܡܵܝܟ ܠܥܵܠܲܡ. ܘܲܥܒ݂ܵܕ݂ ܐܝܼܕܲܝ̈ܟ ܠܵܐ ܬܲܪܦܸܐ܀ ܨܠܝܼܒ݂ܵܟ݂ ܫܘܼܪܵܐ ܢܸܗܘܸܐ ܠܲܢ. ܡܫܝܼܚܵܐ ܕܦܲܪܩܲܢ ܒܲܨܠܝܼܒܹܗ܀ ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܵܪܲܢ ܝܼܫܘܿܥ. ܘܬ݂ܵܘܕܝܼܬܵܐ ܠܐܲܒ݂ܵܐ ܕܫܲܠܚܵܟ݂܀ ܘܲܠܪܘܼܚ ܩܘܼܕ݂ܫܵܐ ܗܘܼܠܵܠܵܐ: ܡܢ ܥܵܠܲܡ ܥܕܲܡܵܐ ܠܥܵܠܲܡ. ܘܡܸܬ݂ܐܲܡܪܵܐ ܒܠܲܝܠܵܘ̈ܵܬ݂ܵܐ ܕܫܲܒܹ̈ܐ ܕܐ݇ܪ̈ܵܙܹܐ: ܘܒܲܥܪ̈ܘܼܒ݂ܵܬ݂ܵܐ.
ܟܵܪܘܿܙܘܼܬ݂ܵܐ. ܢܩܘܼܡ ܫܲܦܝܼܪ...܀ ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ݁ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡܠܲܟ݂ ܛܵܒ݂̈ܵܬ݂ܵܐ ܠܪ̈ܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕܬܹܗ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹ݁ܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܵܢ: ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܗܘܝܘܼ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܪ̈ܝܵܬ݂ܵܐ: ܘܥܲܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܢܟ݂...܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒ݂ܐܲܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܢ.
	ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ. ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ. ܣܝܼܡܘ ܒܘܼܪܟܵܐ. ܩܘܼܡܘ ܒܚܲܝܠܵܐ ܕܐܲܠܵܗܵܐ.
	ܘܲܡܨܲܠܸܐ ܪܲܒܵܐ. ܡܸܢܵܟ݂ ܡܠܸܐ ܪ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ: ܘܡܢ ܥܘܼܬ݂ܪܵܐ ܪܲܒܵܐ ܕܒܲܣܝܼܡܘܼܬܹܗ ܕܚܘܼܒܵܟ݂: ܘܓܲܙܵܐ ܫܦܝܼܥܵܐ ܕܲܚܢܵܢܵܟ݂: ܫܵܐܠܝܼܢܲܢ ܥܘܼܕ݂ܪܵܢܵܐ ܘܚܘܼܝܵܠܵܐ ܘܫܘܼܘܙܵܒ݂ܵܐ ܘܢܘܼܛܵܪܵܐ ܘܐܵܣܝܘܼܬ݂ܵܐ ܠܟܹܐܒܹ̈ܐ ܕܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮ܫ̈ܵܬܲܢ: ܗܲܒ݂ܠܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܪ̈ܲܚܡܲܝܟ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.܀

ܘܥܵܢܹܝ̇ܢ:

 ܐܵܡܹܝܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
	ܚܵܬ݂ܵܗ̇. ܒܪܝܼܟ݂ܝܼܢ ܘܲܣܓ݂ܝܼܕ݂ܝܼܢ: ܘܪܵܡܝܼܢ ܘܲܡܥܲܠܝܼܢ: ܘܠܵܐ ܡܸܬ݁ܕܲܪܟ݁ܝܼܢ ܪ̈ܲܚܡܹܐ ܡܬ݂ܘܿܡܵܝܹ̈ܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܕܚܵܢܝܼܢ ܡܲܓܵܢ ܠܚܲܛܵܝܹ̈ܐ: ܣܲܒ݂ܪܲܢ ܛܵܒ݂ܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܲܢ ܡܠܸܐ ܪ̈ܲܚܡܹܐ ܘܫܵܒܹ݁ܩ ܚܵܘܒܹ̈ܐ ܘܲܚܛܵܗܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܘܐܵܡܪܝܼܢ ܚܠܵܦ ܡܲܕ݂ܪܵܫܵܐ ܒܵܬܹ̈ܐ. ܘܡܵܐ ܕܡܸܬ݂ܐܲܡܪܝܼܢ ܝܵܬܹ݁ܒ݂ ܟܠܹܗ ܥܲܡܵܐ. ܘܐܵܡܲܪ ܐܵܡܘܿܪܵܐ. ܒܪܝܼܟ݂ ܕܲܒ݂ܨܵܘܡܹܗ ܐܲܠܸܦ̮ ܠܲܢ. ܕܢܸܙܟܹ݁ܝܘܗܝ ܠܒ݂ܝܼܫܵܐ ܒܐܵܓ݂ܘܿܢܵܐ. ܒܪܝܼܟ݂ ܕܲܒ݂ܨܵܘܡܹܗ ܐܲܠܸܦ̮ ܠܲܢ. ܘܥܵܢܹܝܢ. ܕܢܸܙܟܹ݁ܝܘܗܝ ܠܒ݂ܝܼܫܵܐ ܒܐܵܓ݂ܘܿܢܵܐ. ܒܪܝܼܟ݂ ܕܲܒ݂ܨܵܘܡܹܗ ܐܲܠܸܦ̮ ܠܲܢ. ܕܢܸܙܟܹ݁ܝܘܗܝ ܠܒ݂ܝܼܫܵܐ ܒܐܵܓ݂ܘܿܢܵܐ. ܦܸܬ݂ܓ݂ܵܡܹ̈ܐ. ܕܗܵܐ ܐܲܢ݇ܬ݁ ܐܲܠܵܗܵܐ ܛܥܲܝܬܵܢ ܘܠܵܐ ܢܵܦܹ݁ܩ ܐܲܢ݇ܬ݁ ܒܚܲܝܠܲܢ* ܗܲܒ݂ܠܲܢ ܥܘܼܕ݂ܪܵܢܵܐ ܥܲܠ ܒܥܸܠܕܒ݂ܵܒܲܝ̈ܢ. ܕܢܸܙܟܹ݁ܝܘܗܝ ܠܒ݂ܝܼܫܵܐ. ܨܠܝܼ ܡܵܪܝܵܐ ܐܸܕ݂ܢܵܟ݂ ܘܲܥܢܝܼܢܝ* ܡܸܛܠ ܕܡܸܣܟܹ݁ܢܵܐ ܐ݇ܢܵܐ ܘܒ݂ܵܝܫܵܐ* ܢܲܛܲܪ ܠܢܲܦ̮ܫܝ ܡܸܛܠ ܕܛܵܒ݂ ܐܲܢ݇ܬ݁* ܦܪܘܿܩ ܠܥܲܒ݂ܕܵܟ݂ ܐܲܢ݇ܬ݁ ܐܲܠܵܗܵܐ ܕܲܡܣܲܒܲܪ ܒܵܟ݂. ܕܢܸܙܟܹ݁ܝܘܗܝ ܠܒ݂ܝܼܫܵܐ. ܘܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܘܲܡܪܲܚܦܵܢܵܐ* ܢܲܓ݁ܝܼܪ ܪܘܼܚܵܐ ܘܣܲܓ݁ܝܼ ܛܲܝܒ݁ܘܼܬ݂ܵܐ ܘܩܘܼܫܬܵܐ* ܐܸܬ݂ܦܲܢ ܥܠܲܝ ܘܪܲܚܸܡܥܠܲܝܢ* ܗܲܒ݂ ܚܲܝܠܵܐ ܠܥܲܒ݂ܕܵܟ݂ ܘܲܦܪܘܿܩ ܠܒܲܪ ܐܲܡܬ݂ܵܟ݂. ܕܢܸܙܟܹ݁ܝܘܗܝ ܠܒ݂ܝܼܫܵܐ. ܘܫܲܪ܊
	ܣܘܼܟܵܝܹ̈ܐ. ܛܘܼܒ݂ܵܘܗܝ ܠܡܢ ܕܚܵܐܲܪ ܒܡܸܣܟܹ݁ܢܵܐ. ܩܲܕ݂ܡܵܝܹ̈ܐ. ܚܙܝܼ. ܡܙܡܘܪ̈ܐ ܡܐ‐ܡܛ
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܩܠ̈ܐ ܕܫܗܪܐ. ܗܘܼܠܵܠܵܐ. ܐܲܠܵܗ ܐܲܠܵܗܝܼ̈ܢ. ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܢܢ‐ܢܚ. ܩܲܕ݂ܡܵܝܹ̈ܐ ܕܨܵܘܡܵܐ.
	ܦܘܼܚܵܡܵܐ ܕܩܵܠܹ̈ܐ ܕܫܲܗܪܵܐ ܕܩܲܕ݂ܡܵܝܹ̈ܐ.
	ܩܲܕ݂ܡܵܝܵܐ. ܐܲܠܵܗ ܐܲܠܵܗܝܼ̈ܢ. ܐܲܩܲܦܬ݂ܵܐ. ܠܚܲܛܵܝܵܐ ܐܵܡܲܪ ܠܹܗ ܐܲܠܵܗܵܐ.
 ܬܪܲܝܵܢܵܐ. ܐܵܡܲܪ ܥܵܘ̇ܵܠܵܐ ܒܠܸܒܹ݁ܗ. ܒܨܘܼܬ݂ ܫܡܲܝ̈ܵܐ. ܐܲܝܟ݂ ܩܲܕ݂ܡܵܝܹ̈ܐ ܕܲܠܒܲܪ ܡܢ ܨܵܘܡܵܐ. ܐܲܩܲܦܬ݂ܵܐ. ܐܲܠܵܗܵܐ ܒܲܫܡܵܟ݂ ܦܪܘܿܩܲܝܢܝ. ܐܲܝܟ݂ ܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ.
	ܬܠܝܼܬ݂ܵܝܵܐ. ܪܲܚܸܡܥܠܲܝ ܐܲܠܵܗܵܐ ܡܸܛܠ ܕܕ݂ܵܫܲܢܝ. ܐܲܩܲܦܬ݂ܵܐ. ܪܲܚܸܡܥܠܲܝ ܐܲܠܵܗܵܐ ܡܸܛܠ ܕܒ݂ܵܟ݂ܘܼ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ ܕܚܲܕ݂ܒܫܲܒܹ̈ܐ. ܡܢ ܦܘܼܡܹ̈ܐ ܡܵܪܝ ܡܠܝܼܠܹ̈ܐ ܕܲܒ݂ܪܲܝܬ: ܘܡܢ ܠܸܫܵܢܹ̈ܐ ܡܫܲܒܚܵܢܹ̈ܐ ܕܪܲܟܸܒ݂ܬ: ܘܡܢ ܟܸܢܫܹ̈ܐ ܟܠܗܘܿܢ ܕܥܸܠܵܝܹ̈ܐ ܘܲܕܬܲܚܬܵܝܹ̈ܐ: ܫܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܲܕ݂ܡܵܪܘܼܬ݂ܵܟ݂ ܢܸܣܬܓܸܕ݂ ܘܢܸܫܬܲܒܲܚ ܘܢܸܬ݂ܝܲܩܲܪ ܘܢܸܬ݂ܪܲܡܪܲܡ ܘܢܸܬܬܵܘܕܸܐ ܘܢܸܬ݂ܒܲܪܲܟ݂ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
	ܕܠܸܠܝܵܐ. ܒܫܘܼܪܵܝ ܒܪܵܡܘܼܬ݂ܵܟ݂ ܢܸܦܠܹܬ݂. ܒܪܝܼܟ݂ܘܼ ܕܐܵܬܹ݁ܐ ܒܲܫܡܹܗ ܕܡܵܪܝܵܐ: ܡܹܐܬ݂ܵܐ ܐܵܬܹ݁ܐ ܒܚܲܕ݂ܘܼܬ݂ܵܐ: ܐܲܝܢ̇ܵܘ ܓܲܒ݂ܪܵܐ ܕܨܵܒܹ݁ܐ ܚܲܝܹ̈ܐ: ܢܩܲܕܸܡ ܐܲܦܵܘ̈ܗܝ ܒܬܵܘܕ݁ܝܼܬ݂ܵܐ. ܕܨܘܼܪܬܵܐ. ܕܢܸܬ݂ܪܘܿܨ ܪܸ̈ܓ݂ܠܲܝܢ ܒܐܘܼܪܚܵܐ ܕܲܫܠܵܡܵܐ. ܗܵܐ ܨܵܘܡܵܐ ܦܲܐܝܵܐ ܒܲܕ݂ܡܘܼܬ݂ ܡܲܠܟܵܐ ܐܸܬ݀ܵܐ ܠܲܢ. ܟܠܢܵܫ ܢܲܦ̮ܫܹܗ ܬܸܗܘܸܐ ܡܨܲܒ݁ܬܵܐ ܠܹܗ ܐܲܝܟ݂ ܡܕ݂ܝܼܢ݇ܬܵܐ. ܘܐܲܝܟ݂ ܕܡܢ ܫܘܼܩܹ̈ܐ ܡܢ ܠܸܒܵܘ̈ܵܬ݂ܵܐ ܨܵܐܘܼܬ݂ܵܐ ܕܥܵܘܠܵܐ ܢܕܲܟܸ݁ܐ. ܘܲܠܚܘܼܫܵܒܹ̈ܐ ܐܲܝܟ݂ ܕܥܲܠ ܟܸܢܫܹ̈ܐ. ܪܹܫܵܐ ܢܩܝܼܡ ܠܗܘܿܢ ܗܵܘܢܵܐ ܕܲܡܨܲܠܲܠ. ܐܵܦ ܠܲܚܙܵܬ݂ܵܐ ܕܥܲܝܢܹ̈ܐ ܢܚܵܘܸܐ. ܒܲܕ݂ܡܘܼܬ݂ ܚܸܙܘܵܢܹ̈ܐ ܟܬ݂ܵܒܹ̈ܐ ܕܣܘܼܟܵܠܵܐ. ܘܢܸܩܥܸܐ ܟܠܲܢ ܡܵܪܝܵܐ ܒܢܲܟ݂ܦܘܼܬ݂ܵܐ ܘܲܒ݂ܩܲܕܝܼܫܘܼܬ݂ܵܐ ܪܲܚܸܡܥܡܥܠܲܝܢ: ܬܢܝܼ. ܘܫܲܒܲܚ. ܒܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܩܵܢܘܿܢܵܐ: ܒܓ݂ܘܼܕܵܐ ܥܸܠܵܝܬܵܐ. ܚܘܼܣ ܥܠܲܝܢ ܐܲܢ݇ܬ ܚܲܢܵܢܵܐ ܒܛܲܝܒܘܼܬ݂ܵܟ݂: ܘܐܸܬ݂ܦܲܢ ܥܠܲܝܢ ܡܠܸܐ ܪ̈ܲܚܡܹܐ: ܘܠܵܐ ܡܵܪܝ ܬܲܗܡܸܐ ܚܝܵܪܵܟ݂ ܘܲܒ݂ܛܝܼܠܘܼܬ݂ܵܟ݂ ܡܸܢܲܢ: ܕܲܥܠܲܝܟܘܼ ܣܲܒ݂ܪܲܢ ܘܬ݂ܘܼܟ݂ܠܵܢܲܢ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܵܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ: ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
ܩܵܢܘܿܢܵܐ. ܛܘܼܒ݂ܵܘܗܝ ܠܓܲܒ݂ܪܵܐ ܕܒ݂ܐܘܼܪܚܵܐ ܕܥܵܘ̈ܵܠܹܐ ܠܵܐ ܗܲܠܸܟ݂* ܘܲܒ݂ܪܸܥܝܵܢܵܐ ܕܚܲܛܵܝܹ̈ܐ ܠܵܐ ܩܵܡ܀ ܛܘܼܒ݂ ܠܐܲܝܢܵܐ ܕܲܛܥܲܢ ܢܝܼܪܵܟ݂ ܘܐܸܬ݂ܗܲܓ݁ܝܼ ܒܢܵܡܘܿܣܵܟ݂. ܡܵܪܝܵܐ ܒܠܸܠܝܵܐ. ܘܒ݂ܐܝܼܡܵܡܵܐ܀ ܛܘܼܒ݂ܵܘܗܝ ܠܓܲܒ݂ܪܵܐ ܕܒ݂ܐܘܼܪܚܵܐ ܕܥܵܘ̈ܵܠܹܐ ܠܵܐ ܗܲܠܸܟ݂* ܘܲܒ݂ܪܸܥܝܵܢܵܐ ܕܚܲܛܵܝܹ̈ܐ ܠܵܐ ܩܵܡ܀ ܘܥܲܠ ܡܵܘܬ݁ܒ݂ܵܐ ܕܲܡܡܲܝܩܵܢܹ̈ܐ ܠܵܐ ܝܑܼܬܸܒ݂* ܐܸܠܵܐ ܒܢܵܡܘܿܣܹܗ ܕܡܵܪܝܵܐ ܨܸܒ݂ܝܵܢܹܗ܀ ܘܲܒ݂ܢܵܡܘܿܣܹܗ ܢܸܬ݂ܗܲܓܸ݁ܐ ܐܝܼܡܵܡܵܐ ܘܠܸܠܝܵܐ* ܢܸܗܘܸܐ ܐܲܝܟ݂ ܐܝܼܠܵܢܵܐ ܕܲܢܨܝܼܒ݂ ܥܲܠ ܬܲܦܵܐ ܕܡܲܝ̈ܵܐ܀ ܕܦܹܐܪ̈ܵܘܜ ܝܵܗܹܒ݂ ܒܙܲܒ݂ܢܹܗ ܘܛܲܪ̈ܦܵܘܗܝ ܠܵܐ ܢܵܬ݂ܪܝܼܢ* ܘܟ݂ܠ ܕܥܵܒܹ݁ܕ݂ ܡܫܲܠܸܡ܀ ܠܵܐ ܗܵܟܲܢܵܐ ܪ̈ܲܫܝܼܥܹܐ* ܐܸܠܵܐ ܐܲܝܟ݂ ܥܘܼܪܵܐ ܕܕ݂ܵܪܝܵܐ ܪܘܼܚܵܐ܀ ܡܸܛܠ ܗܵܢܵܐ ܠܵܐ ܢܩܘܼܡܘܼܢ ܪ̈ܲܫܝܼܥܹܐ ܒܕ݂ܝܼܢܵܐ* ܐܵܦܠܵܐ ܚܲܛܵܝܹ̈ܐ ܒܲܟ݂ܢܘܼܫܬܵܐ ܕܙܲܕ݁ܝܼܩܹ̈ܐ܀ ܡܸܛܠ ܕܝ̇ܵܕܲܥ ܗ݇ܘ ܡܵܪܝܵܐ ܐܘܼܪܚܵܐ ܕܙܲܕܝܼܩܹ̈ܐ* ܘܐܘܼܪܚܵܐ ܕܪ̈ܲܫܝܼܥܹܐ ܬܹܐܒܲܕ݂܀ ܛܘܼܒ݂ ܠܐܲܝܢܵܐ ܕܲܛܥܲܢ ܢܝܼܪܵܟ݂ ܘܐܸܬ݂ܗܲܓ݁ܝܼ ܒܢܵܡܘܿܣܵܟ݂. ܡܵܪܝܵܐ ܒܠܸܠܝܵܐ. ܘܒ݂ܐܝܼܡܵܡܵܐ܀
	ܫܲܒܲܚ ܀ ܥܵܠܲܡ. ܛܘܼܒ݂ܵܘܗܝ ܠܐܲܝܢܵܐ ܕܐܸܬ݁ܕܒܲܪ ܒܢܵܡܘܿܣܹܗ ܕܲܡܫܝܼܚܵܐ. ܘܲܢܛܲܪ ܟܠܗܘܿܢ ܦܘܼܩܕܵܢܵܘ̈ܗܝ.
	ܛܘܼܒ݂ܵܘܗܝ ܠܓܲܒ݂ܪܵܐ ܕܒ݂ܐܘܼܪܚܵܐ ܕܥܵܘ̈ܵܠܹܐ ܠܵܐ ܗܲܠܸܟ݂* ܘܲܒ݂ܪܸܥܝܵܢܵܐ ܕܚܲܛܵܝܹ̈ܐ ܠܵܐ ܩܵܡ܀ ܛܘܼܒ݂ܵܘܗܝ ܠܐܲܝܢܵܐ ܕܐܸܬ݁ܕܒܲܪ ܒܢܵܡܘܿܣܹܗ ܕܲܡܫܝܼܚܵܐ. ܘܲܢܛܲܪ ܟܠܗܘܿܢ ܦܘܼܩܕܵܢܵܘ̈ܗܝ.
	ܬܸܫܒ݁ܘܿܚܬܵܐ. ܐܸܬ݂ܦܲܢ ܥܲܠ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܦܵܪܘܿܩܲܢ: ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬܲܢ ܘܦܲܢܵܐ ܫܹ̈ܐܠܵܬܲܢ܀ ܕܐܲܢ݇ܬ݁ܘܼ ܒܹܝܬ݂ ܓܵܘܣܵܐ ܘܣܲܒ݂ܪܵܐ ܕܲܡܚܝܼܠܹ̈ܐ: ܒܥܘܼܕ݂ܪܵܢܵܟ݂ ܢܸܙܟܸ݁ܐ ܠܨܸܢܥܵܬܹ̈ܗ ܕܒ݂ܝܼܫܵܐ܀ ܕܗܵܐ ܨܵܠܹܐ ܟܠܫܵܥ ܦܲܚܹ̈ܐ ܠܢܲܦ̮ܫ̈ܵܬܲܢ: ܘܨܵܒܹ݁ܐ ܢܵܟ݂ܘܿܠܵܐ ܕܲܢܟܲܬ݁ܡܲܢ ܒܥܵܘܠܹܗ܀ ܫܲܕܲܪ ܡܵܪܝ ܚܲܝܠܵܟ݂ ܘܲܢܦܲܣܸܩ ܢܸܫܒܵܘ̈ܗܝ: ܕܠܵܐ ܢܨܘܼܕܲܢ ܘܢܸܦܸܠ ܒܗܵܘܬ݂ܵܐ ܕܲܚܛܝܼܬܵܐ܀ ܚܲܝܠܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܢܚܲܝܸܠ ܡܚܝܼܠܘܼܬܲܢ: ܕܢܸܫܦܲܪ ܒܲܥܒ݂ܵܕܲܝ̈ܢ ܩܕ݂ܵܡܲܝܟ ܚܲܢܵܢܵܐ܀ ܘܐܲܫܘܵܐ ܕܢܸܙܡܲܪ ܠܵܟ݂ ܒܝܵܘܡܵܐ ܕܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܫܘܼܒ݂ܚܵܐ ܕܠܵܐ ܦܵܛܲܪ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ܀
ܟܵܪܘܿܙܘܼܬ݂ܵܐ. ܢܩܘܼܡ ܫܲܦܝܼܪ........܀ ܗܵܘ ܕܐܲܠܸܦ̮ ܠܲܢ ܕܲܢܨܲܠܸܐ ܕܠܵܐ ܡܲܐܝܼܢܘܼܬ݂ܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂.....܀ ܗܵܘ ܕܐܲܓܵܗ ܗ݇ܘܵܐ ܒܲܨܠܘܿܬܹܗ ܕܐܲܠܵܗܵܐ: ܡܸܛܠ ܦܘܼܪܩܵܢܹܗ ܕܓܸܢܣܲܢ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂.....܀ ܗ̇ܘ ܕܝܲܗ݇ܒ݂ܠܲܢ ܬܲܚܘܝܼܬ݂ܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܒܐܲܒ݂ܵܗܲܝ̈ܢ ܦܲܓ݂ܪ̈ܵܢܵܝܹܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂....܀ ܗ̇ܘ ܕܡܢ ܡܵܘܬܹ̈ܐ ܚܲܣܝܼܢܹ̈ܐ ܦܲܪܩܲܢ ܘܬ݂ܘܼܒ݂ ܡܣܲܒ݁ܪܝܼܢܲܢ ܕܦܵܪܹܩ ܠܲܢ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂......܀ ܗ̇ܘ ܕܦܲܪܩܲܢ ܡܢ ܫܘܼܠܛܵܢܹܗ ܕܚܸܫܘܿܟ݂ܐ: ܘܐܲܝܬ݁ܝܲܢ ܠܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܒ݂ܪܹܗ ܚܲܒ݁ܝܼܒ݂ܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂......܀ ܗ̇ܘ ܕܐܸܡܲܪ ܕܲܫܐܲܠܘ ܘܢܸܬ݂ܝܼܗܸܒ݂ ܠܟ݂ܘܿܢ: ܒܥܵܘ ܘܬܸܫܟ݁ܚܘܼܢ: ܩܘܿܫܘ ܘܡܸܬ݂ܦܬܲܚ ܠܟ݂ܘܿܢ ܓܲܙܵܐ ܕܪ̈ܲܚܡܹܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂......܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟ݁ܝܼܣ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܦܵܐ ܡܝܼܛܪܵܦܘܠܹܝܛܝܼܣ: ( ܌ ) ܘܥܲܠ ܟܠܗܘܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܢ ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒ݂ܐܲܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂ ܘܫܲܝܸܢ ܥܹܕܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
