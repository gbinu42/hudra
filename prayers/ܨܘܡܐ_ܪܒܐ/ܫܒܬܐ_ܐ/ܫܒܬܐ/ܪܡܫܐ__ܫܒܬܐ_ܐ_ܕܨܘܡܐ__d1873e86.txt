ܕܡܲܓܵܗܲܝ ܫܲܒ݁ܬ݂ܵܐ ܩܲܕ݂ܡܵܝܬܵܐ ܕܨܵܘܡܵܐ.
ܕܩܲܕ݂ܡܵܝܹ̈ܐ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܢܵܘܕܸ݁ܐ ܡܵܪܝ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܡܵܪܘܼܬ݂ܵܟ݂ ܘܫܘܼܒ݂ܚܵܐ ܐܲܡܝܼܢܵܐ ܕܠܵܐ ܫܲܠܘܵܐ ܢܲܣܸܩ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ: ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ. ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
    ܨܠܘܿܬ݂ܵܐ. ܢܓ݂ܘܼܠܘܼܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܪ̈ܲܚܡܹܐ ܡܬ݂ܘܿܡܵܝܹ̈ܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ. ܥܲܠ ܚܲܛܵܝܹ̈ܐ ܘܲܡܚܝܼ̈ܠܹܐ ܥܲܒ݂ܕܲܝ̈ܟ ܕܩܵܪܹܝܢ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܵܟ݂ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ ܡܵܪܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ . . .
ܗܘܼܠܵܠܵܐ. ܢܩܘܼܡ ܐܲܠܵܗܵܐ. ܚܙܝܼ: ܡܙܡܘܪܹ̈ܐ ܣܚ‐ܥ.
ܘܲܒ݂ܫܘܼܠܵܡܹܗ ܐܵܡܪܝܼܢ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ. ܬܢܝܼ. ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ.
     ܘܡܲܩܦܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ. ܡܫܲܒܲܚܬ݁ ܡܵܪܲܢ ܘܠܵܟ݂ ܝܵܐܹܐ ܠܡܲܣܵܩܘܼ ܫܘܼܒ݂ܚܵܐ ܟܠܝܘܿܡ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ܀ ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ ܦܘܼܡܲܢ ܘܝܲܗ݇ܒ݂ܠܲܢ ܕܲܢܗܲܠܸܠ ܒܬܸܫܒ݁ܚܵܬܹ̈ܗ. ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ ܦܘܼܡܲܢ ܘܝܲܗ݇ܒ݂ܠܲܢ ܕܲܢܙܲܡܲܪ ܒܬܸܫܒ݁ܚܵܬܹ̈ܗ. ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ ܦܘܼܡܲܢ ܘܝܲܗ݇ܒ݂ܠܲܢ ܕܲܢܫܲܒܲܚ ܒܬܸܫܒ݁ܚܵܬܹ̈ܗ܀ ܠܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܫܘܼܒ݂ܚܵܐ ܢܲܣܸܩ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܬܢܝܼ ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܲܢܩܲܠܣܵܟ݂ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܲܢܫܲܒܚܵܟ݂ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂܀ ܕܲܠܘܵܬ݂ ܡܵܝܘܿܬܹ̈ܐ ܡܵܪܵܐ ܡܪܲܚܡܵܢܵܐ. ܚܘܼܢܲܝܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ܀ ܕܠܵܐ ܙܵܟܹ݁ܐ ܩܕ݂ܵܡܲܝܟ ܟܠ ܕܚܲܝ ܒܲܥܒ݂ܝܼܕܹ̈ܐ. ܐܲܢ݇ܬ݁ ܡܵܪܝ ܐܲܦܢܝܼܬܵܢ ܡܢ ܟܠܵܗ̇ ܛܘܼܥܝܲܝ. ܕܐܲܢ݇ܬ݁ܘܼ ܐܲܠܵܗܵܐ. ܘܠܵܟ݂ ܝܵܐܹܐ ܫܘܼܒ݂ܚܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܘܲܡܨܲܠܹܝܢ. ܠܵܟ݂ ܛܵܒ݂ܵܐ ܘܒܲܣܝܼܡܵܐ: ܚܲܢܵܢܵܐ ܘܲܡܠܸܐ ܪ̈ܲܚܡܹܐ: ܡܲܠܟܵܐ ܪܲܒܵܐ ܕܬܸܫܒ݁ܘܿܚܬܵܐ: ܐܝܼܬ݂ܝܵܐ ܕܐܝܼܬ݂ܵܘܜ ܡܢ ܥܵܠܲܡ: ܡܵܘܕܹܝܢܲܢ ܘܣܵܓ݂ܕܝܼܢܲܢ ܘܲܡܫܲܒ݁ܚܝܼܢܲܢ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܡ (ܡܙ؛ ܠ؛ 1-7). ܐܹܪܲܡܪܡܵܟ݂ ܡܵܪܝܵܐ ܕܐܲܪܝܼܡܬܵܢܝ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܘܠܵܐ ܚܲܕ݁ܝܼܬ݁ ܒܝܼ ܠܲܒ݂ܥܸܠܕܒ݂ܵܒܲܝ̈* ܡܵܪܝܵܐ ܐܲܠܵܗܝ ܒܥܹ̇ܝܬ݂ ܡܸܢܵܟ݂ ܘܐܲܣܝܼܬܵܢܝ* ܘܐܲܣܸܩܬ݁ ܡܢ ܫܝܘܿܠ ܠܢܲܦ̮ܫܝ* ܘܐܲܚܝܼܬܵܢܝ ܡܢ ܢܵܚܬܲܝ̈ ܓܘܼܒܵܐ* ܙܡܲܪܘ ܠܡܵܪܝܵܐ ܓܒ݂ܵܘ̈ܗܝ* ܘܐܵܘܕܵܘ ܠܕ݂ܘܼܟ݂ܪܵܢܵܐ ܕܩܘܼܕ݂ܫܹܗ*  ܫܲܒܲܚ܀ ܥܵܠܲܡ܀
      ܨܠܘܿܬ݂ܵܐ ܕܡܵܪܝܵܐ ܩܪܹܝ̇ܬ݂ܵܟ݂ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܪ̈ܵܝܹܐ: ܠܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ: ܘܠܲܒ݂ܛܝܼܠܘܼܬܹܗ ܕܨܸܒ݂ܝܵܢܵܟ݂ ܕܲܥܠܲܝܢ: ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
    ܡܵܪܝܵܐ ܩܪܹܝܬ݂ܵܟ݂. ܒܩܵܠܵܐ ܦܪܝܼܫܵܐ ܕܨܵܘܡܵܐ. ܚܙܝܼ: ܡܙܡܘܪ̈ܐ ܕܡܪܝܐ ܕܩܪܝܬܟ
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܥܘܿܢܝܼܬ݂ܵܐ ܕܒ݂ܵܬܲܪ. ܫܡܲܥ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܒܲܚܢܵܢܵܟ݂: ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܒܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂: ܘܚܘܼܢ ܠܚܲܛܵܝܘܼܬܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܪ̈ܲܚܡܲܝܟ: ܐܵܣܝܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܫܘܼܪܵܝܵܐ ܕܒ݂ܵܬܲܪ (ܡܙ؛ ܢܕ؛ 1-8). ܐܲܠܵܗܵܐ ܒܲܫܡܵܟ݂ ܦܪܘܿܩܲܝܢܝ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܘܲܒ݂ܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂ ܕܘܼܢ ܠܝܼ* ܐܲܠܵܗܵܐ ܫܡܲܥ ܨܠܘܿܬ݂ܝ* ܘܨܘܼܬ݂ ܐܸܢܹܝܢ ܠܡܸܠܵܘ̈ܗܝ ܕܦܘܼܡܝ* ܡܸܛܠ ܕܢܘܼܟ݂ܪ̈ܵܝܹܐ ܩܵܡܘ ܥܠܲܝ* ܘܥܲܫܝܼܢܹ̈ܐ ܒܥܵܐܘܼܗ̇ ܠܢܲܦ̮ܫܝ* ܘܠܵܐ ܚܲܫܒ݁ܘܼܟ݂ ܐܲܠܵܗܵܐ* ܐܲܠܵܗܵܐ ܡܥܲܕ݁ܪܵܢܝ ܡܵܪܝܵܐ ܡܣܲܡܟ݂ܵܢܵܗ̇ ܕܢܲܦ̮ܫܝ* ܫܲܒܲܚ܀ ܥܵܠܲܡ܀
	ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
     ܘܲܡܨܲܠܹܝܢ ܒܩܲܢܟܹ݁ܐ. ܘܥܲܠ ܐܲܦܲܝ̈ ܟܠܗܘܿܢ ܥܘܼܕ݂ܪ̈ܵܢܲܝـܟ ܘܛܲܝـܒ݁ܘ̈ܵܬ݂ܵܟ݂ ܕܲܠܘܵܬܲܢ ܕܠܵܐ ܡܸܬ݂ܦܲܪܥܝܼـܢ: ܢܵܘܕܸ݁ܐ ܠܵـܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ ܒܥܹܕ݁ܬܵܟ݂ ܡܟܲܠܲܠܬܵܐ ܡܲܠܝܲܬ݂ ܟܠ ܥܘܼܕ݂ܪ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܵܐ ܘܒ݂ܵܪܘܿܝܵـܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
	ܘܢܵܦ݁ܩܝܼܢ ܒܠܵܟ݂ܘܼ ܡܵܪܵܐ ܠܒܹܝܡ ܒܲܨܠܝܼܒ݂ܵܐ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮ܫ̈ܵܬܲܢ. ܦܸܬ݂ܓ݂ܵܡܹ̈ܐ ܕܠܵܟ݂ܘܼ ܡܵܪܵܐ.
 ܐܲܫ̇ܝܼܓܹܬ݂ ܕܲܟ݂ܝܵܐܝܼܬ݂ ܐܝܼܕܲܝ̈: ܘܐܸܬ݂ܟܲܪܟܹ݁ܬ݂ ܠܡܲܕ݂ܒ݁ܚܵܟ݂ ܡܵܪܝܵܐ.ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮ܫ̈ܵܬܲܢ.

ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ.

ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.

	ܨܠܘܿܬ݂ܵܐ. ܐܲܢ݇ܬ݁ܘܼ ܡܵܪܝ ܫܲܪܝܼܪܵܐܝܼܬ݂ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܪ̈ܵܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮ܫ̈ܵܬܲܢ: ܘܲܡܢܲܛܪܵܢܵܐ ܐܲܡܝܼܢܵܐ ܕܚܲܝܲܝ̈ܢ: ܘܠܵܟ݂ ܡܵܪܝ ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
	ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ.

ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
 ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܕܣܘܼܒܵܥܵܐ. ܐܲܫܘܵܐ ܠܲܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܠܪܲܡܫܵܐ ܕܫܲܝܢܵܐ: ܘܲܠܠܸܠܝܵܐ ܕܲܢܝܵܚܵܐ: ܘܲܠܨܲܦܪܵܐ ܕܲܡܣܲܒ݁ܪ̈ܵܢ ܒܹܗ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܘܠܐܝܼܡܵܡܵܐ ܕܲܥܒ݂ܵܕܹ̈ܐ ܛܵܒܹ̈ܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܕܲܒ݂ܗܘܿܢ ܢܪܲܥܸܐ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
      ܥܘܿܢ̈ܝܵܬ݂ܵܐ. ܒܢܘܼܗܪܵܐ ܘܒܲܪ ܢܘܼܗܪܵܐ. ܙܡܲܪܘ ܠܐܲܠܵܗܵܐ ܒܫܘܼܒ݂ܚܵܐ ܙܡܲܪܘ ܠܡܲܠܟܲܢ: ܡܸܛܠ ܕܗܘܝܘܼ ܐܲܠܵܗܲܢ. ܥܹܕ݁ܬܵܐ ܙܡܲܪܝ ܫܘܼܒ݂ܚܵܐ. ܠܲܡܫܝܼܚܵܐ ܒܩܵܠ ܐܘܿܫܲܥܢܹ̈ܐ. ܘܐܵܘܕܵܝ ܘܲܣܓ݂ܘܿܕ݂ܝ ܠܹܗ. ܕܒܲܨܠܝܼܒܹܗ ܚܲܪܲܪ ܝܲܠܕܲܝ̈ܟ݁ܝ: ܬܢܝܼ. ܐܸܬ݂ܦܲܨܚܝ ܘܙܲܡܲܪܝ ܘܫܲܒܲܚܘ. ܥܹܕ݁ܬܵܐ ܙܡܲܪܝ ܫܘܼܒ݂ܚܵܐ. ܠܲܡܫܝܼܚܵܐ ܕܲܒ݂ܝܲܕ݂ ܨܠܝܼܒܹܗ. ܚܲܪܲܪ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܘܡܲܠܒܸ݁ܫ ܫܘܼܒ݂ܚܵܐ ܠܩܲܕ݁ܝܼܫܵܘ̈ܗܝ. ܫܡܲܥܝ ܒܪܲܬ݂ܝ ܘܲܚܙܵܝ ܘܲܨܠܵܝ ܐܸܕ݂ܢܹܟ݂ܝ. ܥܹܕ݁ܬܵܐ ܡܗܲܝܡܲܢܬܵܐ. ܙܡܲܪܝ ܠܹܗ ܫܘܼܒ݂ܚܵܐ ܠܲܡܫܝܼܚܵܐ. ܕܦܲܪܩܹܟ݂ܝ ܒܲܨܠܝܼܒܹܗ. ܡܢ ܡܵܘܬܵܐ ܘܡܢ ܣܵܛܵܢܵܐ. ܢܘܼܗܪܵܐ ܕܲܕ݂ܢܲܚ ܠܙܲܕ݁ܝܼܩܹ̈ܐ. ܥܹܕܬܵܐ ܡܗܲܝܡܲܢܬܵܐ. ܩܘܼܡܝ ܐܲܢܗܲܪܝ ܕܲܡܛܵܐ ܢܘܼܗܪܹܟ݂ܝ. ܘܲܩܪܸܒ݂ ܓܸܠܝܵܢܹܗ. ܕܡܲܠܟܵܐ ܕܦܲܪܩܹܟ݂ܝ ܒܲܨܠܝܼܒܹܗ. ܚܵܘܝܼ ܡܵܪܝܵܐ ܦܘܼܪܩܵܢܹܗ. ܨܠܝܼܒ݂ܵܐ ܕܐܸܬ݂ܚܵܘܝܼ ܠܩܘܼܣܛܲܢܛܝܼܢܘܿܣ ܒܲܫܡܲܝܵܐ. ܒܟܸܢܦܵܘ̈ܗܝ ܣܲܬܲܪܲܝܢ. ܡܢ ܨܸܢܥܵܬܹ̈ܗ ܕܲܢܟ݂ܝܼܠܵܐ. ܘܒܹܗܘ ܢܸܚܕܸ݁ܐ ܠܸܒܲܢ. ܒܨܵܘܡܵܟ݂ ܦܵܪܘܿܩܲܢ. ܡܠܲܟ݂ܬ݁ ܘܐܸܫܬܵܘܕ݁ܝܼܬ݁ ܠܲܟ݂ܝܵܢܲܢ. ܙܵܟ݂ܘܼܬ݂ܵܐ ܪܲܒ݁ܬ݂ܵܐ. ܠܘܼܩܒܲܠ ܒܝܼܫܵܐ ܢ̇ܵܟ݂ܘܿܠܵܐ. ܬܵܘ ܒܢܲܝ̈ܵܐ ܫܘܼܡܥܘܼܢܝ. ܒܨܵܘܡܵܐ ܚܲܒ݁ܝܼܒܲܝ̈ ܢܸܒ݂ܣܘܿܪ ܘܢܸܫܕܸ݁ܐ ܣܲܢ̈ܝܵܬܲܢ. ܘܢܸܫܦܲܪ ܠܐܲܠܵܗܵܐ. ܘܢܸܩܪܸܐ ܠܪ̈ܲܚܡܹܐ ܠܥܘܼܕ݂ܪܵܢܲܢ.
ܕܲܨܠܘܿܬ݂ܵܐ. ܫ̇ܵܡܲܥ ܘܠܵܐ ܡܲܗܡܸܐ ܫܘܼܚܠܵܦܵܐ: ܗ. ܩܵܠܵܐ: ܛ. ܒܪܘܼܟܵܒ݂ ܢܘܼܗܪܵܐ ܘܒܲܪ ܢܘܼܗܪܵܐ
ܫܡܲܥ ܐܲܠܵܗܵܐ ܘܪܲܚܸܡܥܠܲܝ. ܫܵܡܲܥ ܘܠܵܐ ܡܲܗܡܸܐ. ܥܵܢܹ̇ܐ ܘܦܵܪܹܩ ܘܲܡܫܲܘܙܸܒ݂. ܫܡܲܥ ܡܵܪܝ ܒܵܥܘܼܬܲܢ. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܸܐ̈ܠܵܬܲܢ. ܫܡܲܥ ܡܵܪܝܵܐ ܒܵܥܘܼܬ݂ܝ ܡܵܪܝܵܐ ܨܠܘܿܬ݂ܝ ܩܲܒܸܠ. ܐܲܠܵܗܵܐ ܕܲܒ݂ܪܵܢ. ܕܲܫܡܲܥ ܩܵܠܲܢ ܩܕ݂ܵܡ ܕܢܸܒ݂ܪܹܝܢ. ܫܡܲܥ ܒܚܘܼܒܵܟ݂ ܨܠܘܿܬܲܢ. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܸܐ̈ܠܵܬܲܢ. ܡܗܲܝܡܲܢܘܼ ܡܵܪܝܵܐ ܒܡܸܠܵܘ̈ܗܝ. ܡܵܪܲܢ ܐܲܢ݇ܬ݁ ܐܸܡܲܪܬ݁. ܕܟ݂ܠ ܕܢܵܩܹ̇ܫ ܒܲܬ݂ܪܲܥ ܡܵܪܘܼܬ݂ܝ. ܡܸܬ݂ܦܬܲܚ ܠܹܗ ܬܲܪܥܵܐ. ܘܡܸܬ݂ܦܲܢ̈ܝܵܢ ܠܹܗ ܫܹܐ̈ܠܵܬܹܗ. ܐܸܬ݁ܕܲܟ݂̄ܪ ܡܵܪܝܵܐ ܪ̈ܲܚܡܲܝܟ ܕܡܢ ܥܵܠܲܡ ܘܛܲܝܒ̈ܘܵܬ݂ܵܟ݂. ܡܵܪܲܢ ܐܲܝܟ݂ ܕܐܸܡܲܪܬ݁. ܢܩܲܫܢܲܢ ܬܲܪܥܵܟ݂ ܒܥܘܼܕ݂ܪܵܢܵܟ݂. ܦܬܲܚ ܠܲܢ ܬܪܲܥ ܪ̈ܲܚܡܲܝܟ. ܘܐܲܥܸܠܲܝܢ ܠܲܠܡܹܐܢ ܚܲܝܹ̈ܐ. ܥܕܲܡܵܐ ܕܲܬ݂ܪܲܚܸܡ ܥܠܲܝܢ. ܒܲܬ݂ܪܲܥ ܛܲܝܒܘܼܬ݂ܵܟ݂. ܢܵܩܫܝܼܢ ܥܲܒ݂ܕܲܝ̈ܟ ܕܐܲܠܝܼܨܝܼܢ. ܦܬܲܚ ܠܲܢ ܘܦܲܢܵܐ ܠܲܢ. ܫܹܐ̈ܠܵܬܲܢ ܪܵܚܹܡ ܐ݇ܢܵܫܵܐ. ܦܬܲܚ ܠܝܼ ܬܲܪ̈ܥܹܐ ܕܙܲܕܝܼܩܘܼܬ݂ܵܐ. ܒܬܲܪܥܵܟ݂ ܡܵܪܹܐ ܟܠ. ܢܵܩܫܝܼܢܲܢ ܒܟ݂ܠ ܥܸܕܵܢܝܼ̈ܢ. ܚܘܼܣ ܥܲܠ ܕܵܘܝܘܼܬܲܢ. ܘܐܸܬ݂ܪܲܚܲܡ ܥܲܠ ܡܚܝܼܠܘܼܬܲܢ. ܚܘܼܪ ܡܵܪܝܵܐ ܒܲܩܝܵܡܵܟ݂. ܐܲܫܦܲܥ ܒܘܼܪܟܬ݂ܵܟ݂ ܡܵܪܝ. ܥܲܠ ܟܸܢܫܹ̈ܐ ܕܲܡܫܲܒܚܵܢܲܝ̈ܟ. ܕܢܵܘܕܘܿܢ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܥܲܠ ܛܵܒ݂̈ܵܬ݂ܵܟ݂ ܕܲܠܘܵܬ݂ܗܘܿܢ. ܥܲܠ ܥܲܡܵܟ݂ ܒܘܼܪܟܬ݂ܵܟ݂ ܠܥܵܠܲܡ. ܒܘܼܪܟܬ݂ܵܟ݂ ܘܛܲܝܒܘܼܬ݂ܵܟ݂. ܘܲܚܢܵܢܵܟ݂ ܐܝܼܬ݂ܝܵܐ ܓܢܝܼܙܵܐ. ܐܲܓܸܢ ܒܐܲܡܝܼܢܘܼ. ܥܲܠ ܣܵܓ݂ܘܿܕܲܝ̈ܟ ܕܗܲܝܡܸܢܘ ܒܵܟ݂. ܥܘܼܕ݂ܪܵܢܲܢ ܒܲܫܡܹܗ ܕܡܵܪܝܵܐ. ܥܘܼܕ݂ܪܵܢ ܛܲܝܒܘܼܬ݂ܵܟ݂ ܐܲܠܘܵܐ ܡܵܪܲܢ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܲܢܚܲܡܣܢܘܼܢ ܟܠܫܵܥ. ܒܲܫܪܵܪܵܐ ܕܗܲܝܡܵܢܘܼܬ݂ܗܘܿܢ. ܥܲܕܲܪܲܝܢ ܐܲܠܵܗܵܐ ܦܵܪܘܿܩܲܢ. ܥܘܼܕ݂ܪܵܢܵܐ ܕܪ̈ܲܚܡܲܝܟ. ܘܣܘܼܝܵܥܵܐ ܕܡܢ ܛܲܝܒܘܼܬ݂ܵܟ݂. ܫܲܕܲܪ ܠܲܢ ܡܵܪܝܵܐ. ܘܚܲܝܸܠܲܝܢ ܕܢܸܫܦܲܪ ܠܲܫܡܵܟ݂. ܨܘܼܬ݂ ܡܵܪܝܵܐ ܨܠܘܿܬ݂ܝ. ܨܠܘܿܬܲܢ ܬܸܒ݂ܣܲܡ ܠܵܟ݂. ܘܒ݂ܵܥܘܼܬܲܢ ܬܸܥܘܿܠ ܩܘܼܕ݂ܡܲܝܟ. ܡܵܪܵܐ ܡܪܲܚܡܵܢܵܐ. ܠܵܐ ܒܵܗܬ݁ܝܼܢ ܕܲܡܣܲܒܪܝܼܢ ܒܵܟ݂. ܨܠܘܿܬ݂ܝ ܐܲܝܟ݂ ܥܸܛܪܵܐ ܩܕ݂ܵܡܲܝܟ. ܨܘܼܬ݂ ܡܵܪܝܵܐ ܨܠܘܿܬܲܢ. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ. ܘܐܲܫܘܵܢ ܕܲܕ݂ܠܵܐ ܡܘܼܡ. ܢܗܲܠܸܟ݂ ܩܕ݂ܵܡܲܝܟ ܒܙܲܗܝܘܼܬ݂ܵܐ. ܐܸܬ݂ܦܢܵܝ ܢܲܦ̮ܫܝ ܠܲܢܝܵܚܹܟ݂ܝ. ܐܘܿ ܢܲܦ̮ܫܝ ܕܵܘܝܼܬ݂ܵܐ. ܠܡܵܢܵܐ ܫܓ݂ܝܼܫܲܬ݁ܝ ܘܲܕ݂ܠܝܼܚܲܬ݁ܝ. ܛܲܝܸܒ݂ ܠܹܟ݂ܝ ܢܲܚ̈ܬܹܐ. ܕܠܵܐ ܒܓܹܗܲܢܵܐ ܬܸܫܬܲܢܩܝܼܢ. ܒܲܪܸܟ݂ܝ ܢܲܦ̮ܫܝ ܠܡܵܪܝܵܐ. ܢܲܦ̮ܫܝ ܡܚܝܼܠܬܵܐ ܐܸܬ݁ܬ݁ܥܝܼܪܝ. ܡܢ ܫܸܢܬ݂ܵܐ ܕܡܲܗܡܝܵܢܘܼܬ݂ܵܐ. ܘܨܲܠܵܝ ܠܐܲܠܵܗܹܟ݂ܝ. ܕܲܢܚܘܼܣ ܥܠܲܝܟ݁ܝ ܒܝܘܿܡ ܡܹܐܬ݂ܝܹܗ. ܐܲܟ݂ܚ݇ܕ݂ܵܐ ܥܲܬ݁ܝܼܪܹ̈ܐ ܘܡܸܣܟܹܢܹ̈ܐ. ܢܸܒ݂ܥܸܐ ܘܠܵܐ ܢܸܫܠܸܐ. ܘܲܢܨܲܠܸܐ ܘܠܵܐ ܬܸܡܐܲܢ ܠܲܢ. ܥܲܕ݂ ܐܝܼܬ݂ ܠܲܢ ܐܲܬ݂ܪܵܐ. ܘܚܵܢܝܼܢ ܪ̈ܲܚܡܹܐ ܠܚܲܛܵܝܹ̈ܐ. ܐܲܝܠܹܝܢ ܕܩܵܝܡܝܼܢ ܒܒܲܝܬܹܗ ܕܡܵܪܝܵܐ ܒܠܲܝ̈ܠܵܘܵܬ݂ܵܐ. ܐܸܬ݁ܬ݁ܥܝܼܪܘ ܘܩܘܼܡܘ ܨܲܠܵܘ. ܐܲܚܲܝ̈ ܥܲܕ݂ ܐܝܼܬ݂ ܠܟ݂ܘܿܢ ܐܲܬ݂ܪܵܐ. ܕܲܠܡܵܐ ܡܢ ܫܸܠܝܵܐ. ܐܵܬ݂ܝܵܐ ܫܵܥܲܬ݂ ܥܘܼܢܕܵܢܟ݂ܘܿܢ. ܐܲܣܵܢܝ ܡܵܪܝܵܐ ܡܸܛܠ ܕܙܵܥܘ ܓܲܪ̈ܡܲܝ. ܥܲܣܩܝܼܢ ܟܘܼܪ̈ܗܵܢܲܝܢ. ܘܠܲܝܬ݁ ܠܲܢ ܐܵܣܝܵܐ ܕܥܵܨܹܒ݂ ܠܲܢ. ܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܗܘܝܼ ܐܵܣܝܵܐ. ܘܣܝܼܡ ܣܲܡܵܟ݂ ܠܲܚܒ݂ܵܪ̈ܵܬܲܢ. ܡܲܐܣܸܐ ܠܲܬ݂ܒ݂ܝܼܪ̈ܲܝ ܠܸܒܵܐ ܘܥܵܨܹܒ݂ ܟܹܐܒܲܝ̈ܗܘܿܢ. ܐܵܣܝܵܐ ܫܲܪܝܼܪܵܐ. ܕܐܸܫܬܲܕܲܪ ܠܲܢ ܡܢ ܪܵܘܡܵܐ. ܐܲܣܵܐ ܟܘܼܬ݂̈ܡܵܬܲܢ. ܘܲܫܒ݂ܘܿܩ ܚܵܘܒܲܝ̈ܢ ܘܣܲܟ݂̈ܠܘܵܬܲܢ. ܡܵܪܝܵܐ ܢܸܫܪܸܐ ܒܹܗ ܠܥܵܠܲܡ. ܟܢܘܼܫܝܲܢ ܕܝܵܘܡܵܢܵܐ. ܬܠܝܼܬ݂ܵܝܘܼܬ݂ܵܐ ܬܸܫܪܸܐ ܒܹܗ. ܘܢܲܡܠܸܟ݂ ܒܲܝܢܵܬܲܢ. ܚܘܼܒܹܗ ܕܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ. ܢܸܬ݂ܒܲܪܟ݂ܘܼܢ ܒܹܗ ܟܠܗܘܿܢ ܥܲܡ̈ܡܹܐ. ܒܲܟ݂ܢܘܼܫܝܵܐ ܒܪܝܼܟ݂ܵܐ. ܕܒ݂ܘܼܟ݂ܪܹ̈ܐ ܟܬ݂ܝܼܒܲܝ̈ ܒܲܫܡܲܝܵܐ. ܐܲܫܘܵܐ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܲܢܩܲܕܫܘܼܢ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂. ܕܐܘܼܪܚܵܐ܀ ܡܸܛܠ ܕܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ ܛܵܒ݂ ܐܲܢ݇ܬ݁. ܛܵܒ݂ܵܐ ܕܐܸܬ݂ܠܵܘܝܼ ܠܒܲܪ ܥܲܡܪܲܡ ܒܝܲܡܵܐ ܘܝܲܒ݂ܫܵܐ. ܗܲܕܵܢܝ ܒܲܫܒ݂ܝܼܠܲܝ̈ܟ. ܘܫܲܪܬܲܚܲܝܢܝ ܒܟ݂ܠ ܛܵܒ݂̈ܵܬ݂ܵܐ. ܗܲܠܸܟܲܝܢܝ ܒܲܫܒ݂ܝܼܠܵܐ ܕܦܘܼܩܕܵܢܲܝ̈ܟ ܡܸܛܠ ܕܒܹܗ ܨܒܹܝ̇ܬ݂. ܚܲܪܲܪ ܡܵܪܝ ܐܘܼܪܚܲܢ. ܡܸܢ ܡܲܪ̈ܥܹܐ ܘܡܸܢ ܢܸܣܝܘܿܢܹ̈ܐ. ܘܡܲܢܲܥ ܣܘܼܥܪ̈ܵܢܲܝܢ. ܠܘܲܥܕܵܐ ܕܫܲܝܢܵܐ ܘܕܲܫܠܵܡܵܐ. ܠܹܗ ܗܘ ܝܵܐܝܵܐ ܬܸܫܒܘܿܚܬܵܐ. ܫܘܼܒ݂ܚܵܐ ܘܬ݂ܵܘܕܝܼܬ݂ܵܐ. ܢܲܣܸܩ ܠܐܲܒ݂ܵܐ ܡܬ݂ܘܿܡܵܝܵܐ. ܘܠܲܒ݂ܪܵܐ ܕܡܢ ܥܘܼܒܹܗ. ܘܲܠܪܘܼܚ ܩܘܼܕ݂ܫܵܐ ܢܵܦܘܿܩܵܐ. ܡܸܟܹܝܠ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ. ܠܥܵܠܲܡ ܘܲܠܥܵܠܡܝܼܢ. ܘܲܠܕ݂ܵܪܕܵܪ̈ܝܼܢ ܢܸܬ݁ܬܵܘܕܸܐ: ܘܢܸܫܬܲܒܲܚ ܡܢ ܟܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚ ܩܘܼܕ݂ܫܵܐ. ܕܡܲܪܝܲܡ܀ ܡܵܪܝܵܐ ܠܢܘܼܓ݂ܪܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ. ܨܠܘܿܬ݂ܵܗ̇ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ. ܡܲܪܝܲܡ ܐܸܡܹܗ ܕܲܡܫܝܼܚܵܐ. ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܲܢ. ܘܲܬ݂ܢܲܛܲܪ ܠܲܢ ܡܢ ܒܝܼܫܵܐ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܒܥܵܘ ܠܡܵܪܝܵܐ ܘܐܸܬ݂ܚܲܝܲܠܘ. ܢܒ݂ܝܼܹ̈ܐ ܘܲܫܠܝܼܚܹ̈ܐ. ܘܣܵܗܕܹ̈ܐ ܘܟ݂ܵܗܢܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ. ܨܠܘܿܬ݂ܟ݂ܘܿܢ ܬܸܗܘܸܐ ܠܲܢ. ܫܘܼܪܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܐܲܝܟ݂ ܡܕ݂ܝܼܢ݇ܬܵܐ ܕܲܟ݂ܪܝܼܟ݂ ܠܵܗ̇ ܫܘܼܪܵܐ. ܨܠܝܼܒ݂ܵܐ ܕܲܚܙܵܐ ܗ݇ܘܵܐ. ܩܘܼܣܛܲܢܛܝܼܢܘܿܣ ܒܲܫܡܲܝܵܐ. ܗܘ ܢܸܗܘܸܐ ܫܘܼܪܵܐ. ܠܥܹ̈ܕܵܬ݂ܵܐ ܘܲܠܕܲܝܪ̈ܵܬ݂ܵܐ. ܕܐܲܒ݂ܘܼܢ܀ ܒܥܝܼ ܡܢ ܡܵܪܝܵܐ ܘܨܲܠܵܐ ܩܕ݂ܵܡܵܘܗܝ. ܐܲܒ݂ܘܼܢ ܩܲܕܝܼܫܵܐ. ܡܵܪܝ ܦܠܢ ܢܲܨܝܼܚܵܐ. ܒܥܝܼ ܚܠܵܦܲܝܢ ܪ̈ܲܚܡܹܐ. ܡܢ ܡܵܪܵܟ݂ܕܲܪܚܸܡܬ݁ܚܘܼܒܹܗ. ܕܥܲܢܝܼ̈ܕܹܐ܀ܠܵܐܬܸܫܕܹܝܢܝ ܡܢ ܩܕ݂ܵܡܲܝܟ. ܒܝܵܘܡܵܐ ܕܡܸܬ݂ܓܲܠܝ̈ܵܢ. ܟܲܣ̈ܝܵܬ݂ܵܐ ܕܲܣܥܝܼܪ̈ܵܢ ܠܲܢ. ܩܕ݂ܵܡܲܝܟ ܚܲܢܵܢܵܐ. ܠܵܐ ܬܸܥܘܿܠ ܒܕ݂ܝܼܢܵܐ ܥܲܡܲܢ. ܡܵܪܝܵܐ ܒܨܲܦܪܵܐ ܬܸܫܡܲܥ ܩܵܠܝ. ܒܨܲܦܪܵܐ ܕܒܹܗ ܡܸܬ݂ܓ݁ܠܸܐ. ܢܝܼܫܵܟ݂ ܠܒ݂ܘܼܚܪܵܢ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܚܢܵܢܵܟ݂ ܢܩܘܼܡ ܒܐܲܦܲܝ̈ܢ. ܘܪ̈ܲܚܡܲܝܟ ܢܦܝܼܣܘܼܢ ܥܲܠ ܚܵܘܒܲܝ̈ܢ܀
ܫܲܒܲܚ. ܐܲܚܲܝ̈ ܙܡܲܪܘ ܫܘܼܒ݂ܚܵܐ. ܘܲܦܪܘܿܥܘ ܩܘܼܒܵܠܛܲܝܒ݁ܘܼܬ݂ܵܐ. ܠܲܡܫܝܼܚܵܐ ܕܪܲܡܪܸܡ. ܫܸܦܠܹܗ ܕܓܸܢܣܲܢ ܒܲܩܝܵܡܬܹ݁ܗ.
ܥܵܠܲܡ. ܥܲܠ ܪܹܫ ܓܵܓ݂ܘܼܠܬܵܐ. ܚܙܵܬܹܗ ܥܹܕ݁ܬܵܐ ܠܲܡܫܝܼܚܵܐ. ܘܒܸܪܟܲܬ݀ ܘܣܸܓ݂ܕܲܬ݀ ܠܹܗ. ܘܦܸܪܥܲܬ݀ ܫܘܼܒ݂ܚܵܐ ܠܫܵܠܘܿܚܹܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܩܵܢܘܿܢܵܐ. ܚܘܼܣ ܥܠܲܝܢ ܐܲܢ݇ܬ݁ ܚܲܢܵܢܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܸܬ݂ܦܲܢ ܥܠܲܝܢ ܡܠܸܐ ܪ̈ܲܚܡܹܐ: ܘܠܵܐ ܡܵܪܝ ܬܲܗܡܸܐ ܚܝܵܪܵܟ݂ ܘܲܒ݂ܛܝܼܠܘܼܬ݂ܵܟ݂ ܡܢܢ: ܕܲܥܠܲܝܟ݁ܘܼ ܣܲܒ݂ܪܲܢ ܘܬ݂ܘܼܟ݂ܠܵܢܲܢ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ: ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
      ܩܵܢܘܿܢܵܐ. ܐܲܠܵܗܲܢ ܒܹܝܬ݂ ܓܵܘܣܲܢ ܥܲܫܝܼܢܵܐ* ܘܲܡܥܲܕ݂ܪܵܢܲܢ ܒܙܲܒ݂ܢܹ̈ܐ ܕܐܘܼܠܨܵܢܵܐ* ܒܹܝܬ݂ ܓܵܘܣܲܢ ܐܲܢ݇ܬ݁ܘܼ ܡܵܪܲܢ. ܘܠܵܐ ܒܵܗܬ݁ܝܼܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܐܸܚܲܕ݂ܘ ܓܵܘܣܵܐ ܒܲܨܠܝܼܒ݂ܵܟ݂܀ ܐܸܫܬ݁ܟܲܚܬ݁ ܠܲܢ ܒܟ݂ܠܙܒܲܢ* ܡܸܛܠ ܗܵܢܵܐ ܠܵܐ ܕܵܚܠܝܼܢܲܢ܀ ܡܵܐ ܕܙܵܝܥܵܐ ܐܲܪܥܵܐ ܘܙܵܝܥܝܼܢ ܛܘܼܪܹ̈ܐ ܒܠܸܒܵܐ ܕܝܲܡܡܹ̈ܐ* ܢܸܬ݁ܕܲܠܚܘܼܢ ܡܲܝܲܝ̈ܗܘܿܢ ܘܢܸܬ݂ܗܲܦܟ݁ܘܼܢ܀ ܘܲܢܙܘܼܥܘܼܢ ܛܘܼܪܹ̈ܐ ܒܥܘܼܫܢܹܗ* ܬܲܦܹ̈ܐ ܕܢܲܗܪ̈ܵܘܵܬ݂ܵܐ ܢܚܲܕܘܿܢ ܒܲܡܕܝܼܢ݇ܬܹܗ ܕܐܲܠܵܗܲܢ܀ ܩܲܕ݁ܝܼܫܘܼ ܒܹܝܬ݂ ܡܲܫܪܝܹܗ ܕܲܡܪܲܝܡܵܐ* ܐܲܠܵܗܵܐ ܒܓ݂ܵܘܵܗ ܠܵܐ ܬܙܘܼܥ܀ ܢܥܲܕ݁ܪܝܼܗ ܐܲܠܵܗܵܐ ܒܥܸܕܵܢ ܨܲܦܪܵܐ* ܐܸܫܬ݁ܓܸܫܘ ܥܲܡܡܹ̈ܐ ܘܙܵܥܘ ܡܲܠܟ̈ܘܵܬ݂ܵܐ܀ ܐܲܪܝܼܡ ܩܵܠܹܗ ܘܙܵܥܲܬ݀ ܐܲܪܥܵܐ* ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܥܲܡܲܢ܀ ܘܲܡܥܲܕ݁ܪܵܢܲܢ ܐܲܠܵܗܹܗ ܕܝܲܥܩܘܿܒ݂* ܬܵܘ ܚܙܵܘ ܥܒ݂ܵܕ݂ܵܘ̈ܗܝ ܕܐܲܠܵܗܵܐ܀ ܕܥܵܒܹ݁ܕ݂ ܬܸܕ݂ܡܪ̈ܵܬ݂ܵܐ ܒܐܲܪܥܵܐ* ܘܲܡܒܲܛܸܠ ܩܪ̈ܵܒܹܐ ܡܢ ܣܵܘ̈ܦܹܝܗ ܕܐܲܪܥܵܐ܀ ܩܸܫ̈ܬܵܬ݂ܵܐ ܡܬܲܒܲܪ ܘܲܡܬܲܒܲܪ ܢܲܝ̈ܙܟܹܐ* ܘܡܲܪ̈ܟ݁ܒ݂ܵܬ݂ܵܐ ܡܵܘܩܸܕ݂ ܒܢܘܼܪܵܐ܀ ܐܸܬ݂ܦܢܵܘ ܘܕܲܥܘ ܕܐܸܢܵܐ ܐ݇ܢܵܐ ܐܲܠܵܗܵܐ* ܐܸܬ݁ܬ݁ܪܝܼܡܹܬ݂ ܒܥܲܡܡܹ̈ܐ ܘܐܸܬ݁ܬ݁ܪܝܼܡܹܬ݂ ܒܐܲܪܥܵܐ܀ ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܥܲܡܲܢ* ܘܲܡܥܲܕ݁ܪܵܢܲܢ ܐܲܠܵܗܹܗ ܕܝܲܥܩܘܿܒ݂܀ ܒܹܝܬ݂ ܓܵܘܣܲܢ ܐܲܢ݇ܬ݁ܘܼ ܡܵܪܲܢ. ܘܠܵܐ ܒܵܗܬ݁ܝܼܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܐܸܚܲܕ݂ܘ ܓܵܘܣܵܐ ܒܲܨܠܝܼܒ݂ܵܟ݂܀
ܫܲܒܲܚ. ܥܵܠܲܡ. ܠܵܟ݂ ܢܵܘܕܸܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܡܵܪܲܢ. ܕܲܡܠܲܝܬܵܗ ܠܲܒ݂ܪܝܼܬ݂ܵܐ. ܗܸܕ݂ܪܹܗ ܘܫܘܼܦܪܹܗ ܕܲܨܠܝܼܒ݂ܵܟ݂.
ܐܲܠܵܗܲܢ ܒܹܝܬ݂ ܓܵܘܣܲܢ ܥܲܫܝܼܢܵܐ* ܘܲܡܥܲܕ݂ܪܵܢܲܢ ܒܙܲܒ݂ܢܹ̈ܐ ܕܐܘܼܠܨܵܢܵܐ* ܠܵܟ݂ ܢܵܘܕܸܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܡܵܪܲܢ. ܕܲܡܠܲܝܬܵܗ ܠܲܒ݂ܪܝܼܬ݂ܵܐ. ܗܸܕ݂ܪܹܗ ܘܫܘܼܦܪܹܗ ܕܲܨܠܝܼܒ݂ܵܟ݂.
ܬܸܫܒ݁ܘܿܚܬܵܐ. ܒܲܨܠܝܼܒ݂ܵܟ݂ ܐܸܚܕܹܬ݂ ܓܵܘܣܵܐ ܦܵܪܘܿܩ. ܒܟܸܢܦܵܘ̈ܗܝ ܣܲܬܲܪܲܝܢܝ ܡܢ ܒܝܼܫܵܐ ܘܚܲܝܠܹܗ܀ ܘܵܝܠܲܢ ܥܲܠ ܕܲܚܛܲܝܢ ܘܐܲܪܓܸ݁ܙܢ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܬܦܝܼܣܵܟ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܕܬܸܥܒܸ݁ܕ݂ ܥܠܲܝܢ ܪ̈ܲܚܡܹܐ.
ܟܵܪܘܿܙܘܼܬ݂ܵܐ. ܐܲܚܝܼܕ݂ ܟܠ. ܘܡܸܬ݂ܐܲܡܪܵܐ ܒܟ݂ܠܗܘܿܢ ܣܘܼܒܵܥܹ̈ܐ ܕܨܵܘܡܵܐ.
     ܢܩܘܼܡ ܫܲܦܝܼܪ...܀ ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܲܚܝܼܕ݂ ܟܠ ܐܲܠܵܗܵܐ ܕܐܲܒ݂̈ܵܗܵܬܲܢ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܩܲܕܝܼܫܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܲܒ݂ܩܲܕܝܼܫܹ̈ܐ ܥܵܡܲܪ ܘܡܸܬ݁ܬ݁ܢܝܼܚ ܨܸܒ݂ܝܵܢܹܗ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܡܲܠܟܵܐ ܕܡܲܠܟܹ̈ܐ ܘܡܵܪܵܐ ܕܡܵܪ̈ܵܘܵܬ݂ܵܐ ܕܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܐ݇ܢܵܫ ܡܢ ܒܢܲܝ̈ܢܵܫܵܐ ܠܵܐ ܚܙܵܝܗܝ ܘܐܵܦܠܵܐ ܡܨܸܐ ܠܡܸܚܙܝܹܗ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܨܵܒܹܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܢܸܚܘܿܢ. ܘܢܸܬ݂ܦܢܘܿܢ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒ݂ܐܲܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂...܀ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂ ܐܲܫܪܵܐ ܒܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ: ܘܐܸܬ݂ܪܲܚܸܡܥܠܲܝܢ.
ܘܟܸܢ ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ. ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.

ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܦܵܬܲܚ ܬܲܪܥܹܗ ܠܲܕ݂ܢܵܩܫܝܼܢ ܒܹܗ: ܡܦܲܢܸܐ ܫܹܐ̈ܠܵܬ݂ܵܐ ܠܲܕ݂ܫܵܐܠܝܼܢ ܠܹܗ. ܦܬܲܚ ܠܵܗ̇ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܬܲܪܥܵܐ ܕܪ̈ܲܚܡܹܐ ܠܲܨܠܘܿܬܲܢ. ܘܩܲܒܸ݁ܠܹܝܗ̇ ܠܒ݂ܵܥܘܼܬܲܢ. ܘܦܲܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ ܡܢ ܒܹܝܬ݂ ܓܲܙܵܟ݂ ܥܲܬ݁ܝܼܪܵܐ ܘܲܫܦܝܼܥܵܐ: ܛܵܒ݂ܵܐ ܕܠܵܐ ܟܵܠܹܐ ܪ̈ܲܚܡܵܘܗܝ ܘܡܵܘܗܒ݂ܵܬܹ̈ܗ: ܡܢ ܣܢܝܼܩܹ̈ܐ ܘܐܲܠܝܼܨܹ̈ܐ ܥܲܒ݂ܕܵܘ̈ܗܝ ܕܩ̇ܵܪܹܝܢ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܹܗ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܐ݇ܚܪܹܬ݂ܵܐ. ܗܘܝܼ ܡܵܪܝ ܥܲܡܲܢ ܒܪ̈ܲܚܡܲܝܟ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܘܕܲܒܲܪ ܠܲܢ ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܘܼܪܟ݁ܬ݂ܵܐ ܡܵܪܝ ܕܪ̈ܲܚܡܹܐ: ܡܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܕܪ̈ܲܚܡܹܐ: ܒܝܲܕ݂ ܡܲܠܲܐܟ݂ܵܐ ܕܪ̈ܲܚܡܹܐ: ܡܢ ܠܘܵܬ݂ ܐܲܠܵܗܵܐ ܣܲܓ݁ܝܼ ܪ̈ܲܚܡܹܐ: ܬܸܫܬܲܕܲܪ ܘܬܸܫܬܲܦܲܥ ܥܲܠ ܟܸܢܫܹ̈ܐ ܕܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ: ܕܐܲܡܝܼܢܝܼܢ ܘܩܵܪܹܝ̇ܢ ܠܲܫܡܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ: ܘܚܵܝ̇ܪܝܼܢ ܘܲܡܣܲܟܹ݁ܝܢ ܠܪ̈ܲܚܡܹܐ: ܘܲܠܦܘܼܪܩܵܢܵܐ ܕܡܢ ܠܘܵܬ݂ܵܟ݂: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܡܲܩܦܝܼܢ. ܨܠܘܿܬ݂ܵܗ̇ ܡܵܪܝ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܗ̇ ܕܐܸܡܵܐ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܬܲܟ݂ܫܲܦ̮ܬܵܗ̇ ܘܬܲܚܢܲܢܬܵܗ̇ ܕܡܲܠܝܲܬ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܐ: ܡܵܪܬ݁ܝ ܡܲܪܝܲܡ ܛܘܼܒ݂ܵܢܝܼܬ݂ܵܐ: ܘܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ ܙܲܟܵܝܵܐ: ܘܥܘܼܕ݂ܪܵܢܵܐ ܐܲܠܵܗܵܝܵܐ: ܘܒ݂ܵܥܘܼܬܹܗ ܕܡܵܪܝ ܝܘܿܚܲܢܵܢ ܡܲܥܡܕ݂ܵܢܵܐ: ܘܕܲܫܠܝܼܚܹ̈ܐ ܛܘܼܒ݂ܵܢܹ̈ܐ: ܘܕܲܐܒ݂ܵܗ̈ܵܬ݂ܵܐ ܢܲܨܝܼܚܹ̈ܐ ܡܣܲܒ݁ܪ̈ܵܢܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܘܙܵܪ̈ܥܲܝ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܕܲܐܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܡܵܪܝ ܐܲܒ݂ܵܐ ܩܵܬ݂ܘܿܠܝܼܩܵܐ: ܘܡܵܪܝ ܐܸܣܛܲܦܵܢܘܿܣ ܒܘܼܟ݂ܪܵܐ ܕܣܵܗܕܹ̈ܐ: ܘܲܕ݂ܓܲܢ݇ܒܵܪ ܚܲܝܠܵܐ ܡܵܪܝ ܓܝܼܘܲܪܓܝܼܣ ܣܵܗܕܵܐ ܢܲܨܝܼܚܵܐ: ܘܲܕ݂ܡܵܪܝ ܦܠܵܢ: ܘܲܕ݂ܡܵܪܝ ܐܵܘܓܹ݁ܝܢ ܛܘܼܒ݂ܵܢܵܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܣܝܼܥܬܹ݁ܗ ܪܘܼܚܵܢܵܝܬܵܐ. ܘܕܲܫܡܘܿܢܝܼ ܘܕܲܒ݂ܢܹܝ̈ܗ̇: ܘܲܕ݂ܟ݂ܠܗܘܿܢ ܣܵܗܕܵܘ̈ܗܝ ܘܩܲܕ݁ܝܼ̈ܫܵܘܗܝ ܕܡܵܪܲܢ: ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܫܘܼܪܵܐ ܪܵܡܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܚܲܣܝܼܢܵܐ: ܕܬܸܦܪܘܿܩ ܘܲܬ݂ܦܲܨܸܐ ܘܲܬ݂ܫܵܘܙܸܒ݂ ܘܲܬ݂ܢܲܛܲܪ ܠܦܲܓ݂ܪ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ ܡܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹܗ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܘܟܸܢ ܚܵܬ݂ܡ̇ܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܪ̈ܵܝܹܐ. ܘܢܵܣܹ̇ܒ݂ ܟܵܗܢܵܐ ܨܠܝܼܒ݂ܵܐ ܒܐܝܼܕܹܗ ܕܝܲܡܝܼܢܵܐ: ܘܩܵܐܹ̇ܡ ܒܓܲܒܵܐ ܬܲܝܡܢܵܝܵܐ: ܘܡܲܗܦܸܟ݂ ܐܲܦܵܘ̈ܗܝ ܠܥܲܡܵܐ ܘܐܵܡܲܪ: ܒܲܪܸܟ݂ ܡܵܪܝ ܒܦܘܼܩܕܵܢܟ݂ܘܿܢ. ܘܥܵܢܹ̇ܝܢ: ܒܦܘܼܩܕܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܡܵܪܲܢ: ܘܫܘܼܒ݂ܚܵܐ ܠܲܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ. ܘܡܲܪܟ݁ܢܝܼܢ ܥܲܡܵܐ ܪܹ̈ܫܲܝܗܘܿܢ.
ܘܡܲܩܸܦ ܚܘܼܬܵܡܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ: ܨܸܡܚܹܗ ܕܐܲܒ݂ܵܐ ܡܬ݂ܘܿܡܵܝܵܐ: ܝܼܠܝܼܕ݂ܵܐ ܕܠܵܐ ܫܘܼܪܵܝܵܐ: ܠܥܸܠ ܡܢ ܙܲܒ݂ܢܹ̈ܐ ܘܲܗܘܲܝ̈ܵܐ: ܕܠܲܝܬ݁ ܠܲܢ ܣܲܒ݂ܪܵܐ ܘܣܘܼܟܵܝܵܐ: ܐܸܠܵܐ ܐܸܢ ܐܲܢ݇ܬ݁ ܒܵܪܘܿܝܵܐ. ܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܘܲܓ݂ܒܲܝ̈ܵܐ: ܕܲܫܦܲܪܘ ܠܵܟ݂ ܡܢ ܫܘܼܪܵܝܵܐ: ܥܒܸܕ݂ ܠܲܚܛܵܗܲܝ̈ܢ ܚܘܼܣܵܝܵܐ: ܘܲܠܣܲܟ݂ܠ̈ܘܵܬܲܢ ܫܪܵܝܵܐ: ܘܲܠܥܵܩ̈ܵܬܲܢ ܦܘܼܨܵܝܵܐ: ܘܲܠܫܸܐ̈ܠܵܬܲܢ ܦܘܼܢܵܝܵܐ: ܘܡܲܢܲܥܲܝܢ ܠܢܘܼܗܪܵܟ݂ ܓܲܐܝܵܐ: ܘܢܲܛܲܪܲܝܢ ܒܢܝܼܫܵܟ݂ ܚܲܝܵܐ: ܡܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܟܣܲܝ̈ܵܐ ܘܲܓ݂ܠܲܝ̈ܵܐ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ). ܗܵܫܵܐ ♰ ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܪܵܫܹ̇ܡ ܨܠܝܼܒ݂ܵܐ ܥܲܠ ܥܲܡܵܐ: ܟܲܕ݂ ܦܵܫܹ̇ܛ ܐܝܼܕܹܗ ܠܥܸܠ ܠܐܲܦܲܝ̈ ܡܲܥܪܒ݂ܵܐ: ܘܡܲܝܬܸ݁ܐ ܠܵܗ̇ ܠܡܲܕ݂ܢܚܵܐ: ܘܟܸܢ ܦܵܫܹ̇ܛ ܠܵܗ̇ ܠܓܲܒܵܐ ܕܬܲܝܡܢܵܝܵܐ: ܘܡܵܘܒܸ݁ܠ ܠܵܗ̇ ܠܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ. ܐܲܝܟܲܢܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܸܣܬ݁ܡܸܟ݂ ܥܲܠ ܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܓܲܒܵܐ ܣܸܡܵܠܵܝܵܐ ܕܟܸܢܫܵܐ.
ܘܥܵܢܹ̇ܝܢ: ܡܫܝܼܚܵܐ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬ݂ܵܟ݂: ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܟܵܗܢܘܼܬ݂ܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
ܐ݇ܚܪܹܢܵܐ: ܬܸܬ݂ܩܲܒܲܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܚܲܠܵܫܹ̈ܐ: ܩܕ݂ܵܡ ܒܹܝܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܬܸܗܘܸܐ ܟܢܝܼܫܘܼܬܲܢ ܗܵܕܹܐ ܠܲܢܝܵܚ ܨܸܒ݂ܝܵܢ ܡܵܪܘܼܬ݂ܵܟ݂: ܠܡܸܣܲܒ݂ ܡܸܢܵܟ݂ ܒܛܲܝܒ݁ܘܼ ܚܘܼܠܡܵܢܵܐ ܕܦܲܓ݂ܪܵܐ: ܘܢܘܼܛܵܪܵܐ ܕܢܲܦ̮ܫܵܐ: ܘܫܘܼܪܬܵܚܵܐ ܕܡܵܙܘܿܢܵܐ:(ܘܚܘܼܣܵܝܵܐ ܕܚܵܘܒܹ̈ܐ ܘܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ):ܘܫܲܝܢܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ: ܘܲܫܠܵܡܵܐ ܢܲܓ݁ܝܼܪܵܐ: ܘܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ ܕܠܵܐ ܥܵܒ݂ܪܵܐ ܘܠܵܐ ܡܸܫܬܲܪܝܵܐ ܡܢ ܒܲܝܢܵܬܲܢ: ܒܟ݂ܠܹܗ ܙܲܒ݂ܢܵܐ ܕܩܵܐܹ̇ܡ ܒܹܗ ܥܵܠܡܵܐ ܗܵܢܵܐ: )ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܒܪܝܼܟ݂ ܐܲܠܵܗܵܐ ܠܥܵܠܡܝܼܢ: ܘܲܡܫܲܒܲܚ ܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ ܠܕ݂ܵܪܕܵܪ̈ܝܼܢ: ܕܡܢܗ ܒܵܥܹܝ̇ܢܲܢ: ܘܲܠܝܲܡܵܐ ܫܦܝܼܥܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ: ܕܗܘ ܢܲܫܘܸܐ ܠܲܢ ܠܫܘܼܒ݂ܚܵܐ ܦܲܐܝܵܐ ܕܡܲܠܟ݁ܘܼܬܹܗ: ܘܲܠܒ݂ܘܼܣܵܡܵܐ ܕܥܲܡ ܡܲܠܲܐܟ݂ܵܘ̈ܗܝ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܘܲܠܓܲܠܝܘܼܬ݂ ܐܲܦܹ̈ܐ ܕܲܩܕ݂ܵܡܵܘܗܝ: ܘܲܠܩܵܘܡܵܐ ܕܡܢ ܝܲܡܝܼܢܹܗ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ: ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܪ̈ܲܚܡܵܘܗܝ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟ݁ܠ ܕܐܸܬ݂ܟܲܢܲܫܢܲܢ ܒܒܲܝܬܹ݁ܗ: ܘܨܲܠܝܼܢܲܢ ܩܕ݂ܵܡ ܪܲܒ݁ܘܼܬܹܗ: ܥܲܠ ܣܲܒ݂ܪܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܗܘ ܢܸܫܡܲܥ ܨܠܘܿܬܲܢ ܒܲܚܢܵܢܹܗ: ܘܲܢܩܲܒܸ݁ܠ ܒܵܥܘܼܬܲܢ ܒܚܵܘܣܵܢܹܗ: ܘܲܢܫܝܼܓ݂ ܘܢܸܡܪܘܿܩ ܨܵܐܬ݂ܵܐ ܕܚܵܘܒܲܝ̈ܢ ܘܕܲܚܛܵܗܲܝ̈ܢ ܒܙܘܿܦܵܐ ܫܦܝܼܥܵܐ ܕܚܲܝܘܼܣܬ݂ܵܢܘܼܬܹܗ: ܘܲܢܢܝܼܚ ܢܲܦ̮̈ܫܵܬ݂ܵܐ ܕܥܲܢܝܼܕܲܝ̈ܢ ܒܐܵܘܵܢܹ̈ܐ ܫܒ݂ܝܼܚܹ̈ܐ ܕܡܲܠܟ݁ܘܼܬܹܗ: ܘܢܸܪܘܿܣ ܥܲܠ ܟܠܲܢ ܛܲܠܵܐ ܕܒܲܣܝܼܡܘܼܬܹܗ: ܘܲܥܠܲܝܢ ܘܥܲܠ ܟܠ ܒܸܪ̈ܝܵܢ ܬܲܓܸ݁ܢ ܝܲܡܝܼܢܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬܹܗ: ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܪ̈ܲܚܡܵܘܗܝ:(ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.

ܐ݇ܚܪܹܢܵܐ

: ܐܲܠܵܗܵܐ ܡܵܪܹܟܠ ܕܝܲܗ݇ܒ݂ ܬܸܫܒ݁ܚܵܬܹ̈ܗ ܒܦܘܼܡܲܢ: ܘܲܙܡܝܼܪ̈ܵܬܹܗ ܒܠܸܫܵܢܲܢ: ܘܩܝܼܢܵܬܹ̈ܗ ܒܓܲܓ݁ܪ̈ܵܬܲܢ: ܘܬ݂ܵܘܕ݁ܝܼܬܹܗ ܒܣܸܦ̈ܘܵܬܲܢ: ܘܗܲܝܡܵܢܘܼܬܹܗ ܒܠܸܒܵܘ̈ܵܬܲܢ: ܗܘ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬܲܢ: ܘܲܢܩܲܒܸ݁ܠ ܒܵܥ̈ܘܵܬܲܢ: ܘܢܸܬ݂ܪܲܥܸܐ ܒܬܲܟ݂ܫ̈ܦܵܬܲܢ: (ܘܲܢܚܲܣܸܐ ܚܵܘ̈ܒܵܬܲܢ:)ܘܲܢܦܲܢܸܐ ܠܲܢ ܫܸܐ̈ܠܵܬܲܢ ܫܲܦܝܼܪ̈ܵܬ݂ܵܐ ܠܵܐ ܥܕ݂ܝܼ̈ܠܵܬ݂ܵܐ. ܘܡܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܢܲܫܦܲܥ ܪ̈ܲܚܡܵܘܗܝ ܘܲܚܢܵܢܹܗ: ܥܠܲܝܢ ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡܢ ܣܵܟܹܗ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܢܸܫܬܲܒܲܚ ܫܡܹܗ ܕܐܲܠܵܗܵܐ ܡܵܪܹܟܠ ܒܲܝܢܵܬܲܢ: ܡܛܲܟ݁ܣܵܢܗܘܿܢ ܕܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ. ܘܲܥܠܲܝܢ ܡܚܝܼ̈ܠܹܐ ܘܚܲܛܵܝܹ̈ܐ: ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡܢ ܣܵܟܹܗ. ܘܥܲܠ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇: ܘܥܲܠ ܐܲܒ݂ܵܗܲܝ̈ܢ ܘܪ̈ܲܒܵܢܲܝܢ ܘܡܲܠܦܵܢܲܝ̈ܢ ܘܥܲܢܝܼܕܲܝ̈ܢ: ܘܥܲܢܝܼܕܹ̈ܐ ܕܲܦܪܲܫܘ ܘܫܲܢܝܼܘ ܡܢ ܒܲܝܢܵܬܲܢ: ܘܥܲܠ ܟܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܬܲܓܸ݁ܢ ܝܲܡܝܼܢܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬܹܗ ܘܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܠܐܲܠܵܗܵܐ ܫܘܼܒ݂ܚܵܐ: ܘܲܠܡܲܠܲܐܟܹ̈ܐ ܐܝܼܩܵܪܵܐ: ܘܲܠܣܵܛܵܢܵܐ ܒܸܗܬ݁ܬ݂ܵܐ: ܘܠܲܨܠܝܼܒ݂ܵܐ ܣܸܓ݂ܕ݁ܬ݂ܵܐ: ܘܲܠܥܹܕ݁ܬܵܐ ܪܘܼܡܪܵܡܵܐ: ܘܲܠܥܲܢܝܼܕܹ̈ܐ ܢܘܼܚܵܡܵܐ: ܘܲܠܬܲܝܵܒܹ̈ܐ ܩܘܼܒܵܠܵܐ: ܘܠܲܐܣܝܼܪܹ̈ܐ ܫܪܵܝܵܐ: ܘܲܠܡܲܪ̈ܥܹܐ ܘܲܟ݂ܪ̈ܝܼܗܹܐܚܘܼܠܡܵܢܵܐ ܘܐܵܣܝܘܼܬ݂ܵܐ: ܘܠܲܐܪ̈ܒܲܥܦܸܢܝܵܬܹܗ ܕܥܵܠܡܵܐ ܫܲܝܢܵܐ ܪܲܒܵܐ ܘܲܫܠܵܡܵܐ. ܘܐܵܦ ܥܠܲܝܢ ܕܐܝܼܬܲܝܢ ܡܚܝܼ̈ܠܹܐ ܘܚܲܛܵܝܹ̈ܐ: ܚܢܵܢܹܗ ܘܪ̈ܲܚܡܵܘܗܝ ܕܐܲܠܵܗܲܢ ܣܓ݂ܝܼܕ݂ܵܐ ܢܹܐܬ݂ܘܿܢ ܘܢܲܓ݁ܢܘܼܢ ܘܢܸܫܬܲܦܥܘܼܢ ܘܲܢܩܵܘܘܿܢ ܘܢܲܡܠܟ݂ܘܼܢ ܒܐܲܡܝܼܢܘܼܬ݂ܵܐ: )ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: )ܒܝܲܡܝܼܢ ܡܵܪܘܼܬ݂ܵܟ݂ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ): ܒܲܪܸܟ݂ܡܵܪܝ ܠܟ݂ܠܲܢ: ܘܢܲܛܲܪ ܠܟ݂ܠܲܢ: ܘܥܲܕܲܪ ܠܟ݂ܠܲܢ: ܘܣܲܝܲܥ ܘܣܲܬܲܪ ܠܟ݂ܠܲܢ: ܘܢܲܚܸܡ ܥܲܢܝܼܕܹ̈ܐ ܕܟ݂ܠܲܢ: ܘܐܲܓܸ݁ܢ ܝܲܡܝܼܢܵܟ݂ ܥܲܠ ܟܠܲܢ: ܘܪ̈ܲܚܡܲܝܟ ܘܲܚܢܵܢܵܟ݂ ܢܸܫܬܲܦܥܘܼܢ ܥܲܠ ܟܠܲܢ:ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ: ܘܩܘܼܒܵܠܛܲܝܒ݁ܘܼܬ݂ܵܐ ܐܲܡܝܼܢܬܵܐ ܬܸܣܲܩ ܠܵܟ݂ ܡܢ ܦܘܼܡ ܟܠܲܢ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟ݁ܠ ܢܒܲܪܸܟ݂ ܠܟܸܢܫܲܢ ܒܛܲܝܒ݁ܘܼܬܹܗ: ܘܲܢܢܲܛܲܪ ܠܲܩܝܵܡܲܢ ܒܣܘܿܓ݂ܐܵܐ ܫܦܝܼܥܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܘܲܢܦܲܢܸܐ ܫܸܐ̈ܠܵܬܲܢ ܡܢ ܒܹܝܬ݂ ܓܲܙܹܗ: ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡܢ ܣܵܟܹܗ: ܘܥܲܠ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇: ܘܥܲܠ ܐܲܬ݂ܪܵܐ ܗܵܢܵܐ ܘܥܵܡܘܿܪ̈ܵܘܗܝ: ܘܥܲܠ ܥܘܼܡܪܵܐ ܗܵܢܵܐ ܘܝܲܬܵܒ݂ܵܘ̈ܗܝ: ܘܥܲܠ ܟܠܲܢ ܒܟ݂ܠܲܢ ܟܢܝܼܫܵܐܝܼܬ݂: ܚܢܵܢܹܗ ܘܪ̈ܲܚܡܵܘܗܝ ܕܐܲܠܵܗܲܢ ܛܵܒ݂ܵܐ: ܢܹܐܬ݂ܘܿܢ ܘܢܸܫܬܲܦܥܘܼܢ ܒܐܲܡܝܼܢܘܼ (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܡܲܩܦܝܼܢ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܲܬ݂ܠܵܬ݂ܡܐܵܐ ܘܲܬ݂ܡܲܢܬܲܥܣܲܪ ܐܲܒ݂ܵܗ̈ܵܬ݂ܵܐ ܐܲܦܸܣܩܘܿܦܹ̈ܐ: ܕܐܸܬ݂ܟܲܢܲܫܘ ܒܢܝܼܩܝܼܵܐ ܡܕ݂ܝܼܢ݇ܬܵܐ ܕܒܹܝܬ݂ܘܿܢܝܼܵܐ: ܒܙܲܒ݂ܢܵܐ ܕܩܘܿܣܛܲܢܛܝܼܢܘܿܣ ܡܲܠܟܵܐ ܙܲܟܵܝܵܐ. ܘܥܸܠܬ݂ܵܐ ܕܹܝܢ ܕܲܟ݂ܢܘܼܫܝܼܗܘܿܢ ܡܸܛܠ ܪܘܼܫܥܹܗ ܕܐܲܪܝܼܘܿܣ܀ ܘܐܲܝܟ݂ ܥܝܵܕ݂ܵܐ ܕܹܝܢ ܕܪܵܕܹܐ ܒܲܝܢܵܬ݂ ܒܢܹܝ̈ܗ̇ ܕܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܲܫܠܝܼܚܵܝܬܵܐ ܕܝܼܠܲܢ ܕܡܲܕ݂ܢܚܵܐ: ܘܲܟ݂ܒܲܪ ܕܡܢ ܪܵܘܡܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܪܕ݂ܵܐ: ܘܡܢ ܡܲܫܠܡܵܢܘܼܬ݂ܵܐ ܘܝܘܼܒܵܠܵܐ ܕܥܲܬ݁ܝܼܩܹ̈ܐ ܡܓ݂ܵܘܝܲܝ̈ ܒܬܹܐܘܿܪ̈ܝܼܵܣ ܕܲܫܪܵܪܵܐ ܐܲܠܵܗܵܝܵܐ ܡܬ݂ܝܼܚ ܗ݇ܘܵ ܥܕܲܡܵܐ ܠܝܵܘ̈ܡܵܬܲܢ. ܘܗܵܟܲܢܵܐ ܡܫܲܪܹܝܢ ܚ݇ܢܲܢ ܠܡܸܬ݂ܢܵܐ ܩܵܢܘܿܢܵܐ ܕܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܢܝܼܩܝܼܵܐ.

ܒܲܫܪܵܪܵܐ ܓܹܝܪ ܕܠܵܐ ܦܘܼܠܵܓ݂ܵܐ:

ܡܗܲܝܡܢܝܼܢܲܢ ܒܚܲܕ݂ ܐܲܠܵܗܵܐ: ܐܲܒ݂ܵܐ ܐܲܚܝܼܕ݂ ܟܠ. ܥܵܒ݂ܘܿܕ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܐܲܝܠܹܝܢ ܕܡܸܬ݂ܚܲܙ̈ܝܵܢ ܘܲܕ݂ܠܵܐ ܡܸܬ݂ܚܲܙ̈ܝܵܢ. ܘܲܒ݂ܚܲܕ݂ ܡܵܪܝܵܐ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ. ܒܪܹܗ ܕܐܲܠܵܗܵܐ ܝܼܚܝܼܕ݂ܵܝܵܐ: ܒܘܼܟ݂ܪܵܐ ܕܟ݂ܠܗܹܝܢ ܒܸܪ̈ܝܵܬ݂ܵܐ. ܗܵܘ̇ ܕܡܢ ܐܲܒ݂ܘܼܗܝ ܐܸܬ݂ܝܼܠܸܕ݂ ܩܕ݂ܵܡ ܟܠܗܘܿܢ ܥܵܠܡܹ̈ܐ: ܘܠܵܐ ܐܸܬ݂ܥܒܸܕ݂. ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܕܡܢ ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܒܲܪ ܟܝܵܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܕܒ݂ܐܝܼ̈ܕ݂ܵܘܗܝ ܐܸܬ݁ܬܲܩܲܢܘ ܥܵܠܡܹ̈ܐ: ܘܐܸܬ݂ܒ݁ܪܝܼ ܟܠ ܡܸܕ݁ܡ. ܗܵܘ̇ ܕܡܸܛܠܵܬܲܢ ܒܢܲܝ̈ܢܵܫܵܐ: ܘܡܸܛܠ ܦܘܼܪܩܵܢܲܢ: ܢܚܸܬ݂ ܡܢ ܫܡܲܝܵܐ: ܘܐܸܬ݂ܓܲܫܲܡ ܡܢ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܘܲܗܘܵܐ ܒܲܪܢܵܫܵܐ: ܘܐܸܬ݂ܒ݁ܛܸܢ ܘܐܸܬ݂ܝܼܠܸܕ݂ ܡܢ ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܘܚܲܫ ܘܐܸܙܕ݁ܩܸܦ: ܒܝܵܘܡܲܝ̈ ܦܲܢܛܝܼܘܿܣ ܦܝܼܠܵܛܘܿܣ. ܘܐܸܬ݂ܩ̄ܒܲܪ: ܘܩܵܡ ܠܲܬ݂ܠܵܬ݂ܵܐ ܝܵܘܡܝܼ̈ܢ ܐܲܝܟ݂ ܕܲܟ݂ܬ݂ܝܼܒ݂. ܘܲܣܠܸܩ ܠܲܫܡܲܝܵܐ: ܘܝܼܬܸܒ݂ ܡܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܘܬ݂ܘܼܒ݂ ܥܬ݂ܝܼܕ݂ ܠܡܹܐܬ݂ܵܐ ܠܲܡܕ݂ܵܢ ܠܡܝܼܬܹ̈ܐ ܘܲܠܚܲܝܹ̈ܐ. ܘܲܒ݂ܚܲܕ݂ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܪܘܼܚܵܐ ܕܲܫܪܵܪܵܐ: ܗܵܘ̇ ܕܡܢ ܐܲܒ݂ܵܐ ܢܵܦܹ݁ܩ: ܪܘܼܚܵܐ ܡܲܚܝܵܢܵܐ. ܘܒܲܚܕ݂ܵܐ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܘܲܫܠܝܼܚܵܝܬܵܐ ܩܵܬ݂ܘܿܠܝܼܩܝܼ. ܘܡܵܘܕܹ݁ܝܢܲܢ ܒܲܚܕ݂ܵܐ ܡܲܥܡܘܿܕ݂ܝܼܬ݂ܵܐ: ܠܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ. ܘܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܪ̈ܲܝܢ: ܘܲܒ݂ܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ: ܐܵܡܹܝܢ.
