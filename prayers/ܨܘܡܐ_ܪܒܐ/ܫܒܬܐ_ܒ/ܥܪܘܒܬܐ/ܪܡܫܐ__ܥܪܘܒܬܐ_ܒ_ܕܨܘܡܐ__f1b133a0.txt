ܕܡܲܓܵܗܲܝ ܥܪܘܼܒܼܬܵܐ ܕܬܲܪܬܹܝܢ ܕܨܵܘܡܵܐ.
ܕܐ݇ܚܖ̈ܵܝܹܐ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܖ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ. ܢܵܘܕܸ݁ܐ ܡܵܪܝ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܡܵܪܘܼܬ݂ܵܟ݂ ܘܫܘܼܒ݂ܚܵܐ ܐܲܡܝܼܢܵܐ ܕܠܵܐ ܫܲܠܘܵܐ ܢܲܣܸܩ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ̣ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ̣ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ: ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ.
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܠܵܟ̣ ܡܲܠܟܵܐ ܕܡܲܠ̈ܟܹ݁ܐ ܣܵܓ݂݁ܕܝܼܢ ܠܹܗ. ܘܟܸܢܫܹ̈ܐ ܘܖܸ̈ܒ݁ܘܵܬ݂ܵܐ ܕܡܲܠܲܐܟܹ̈ܐ ܘܲܕ݂ܖ̈ܲܒܲܝ ܡܲܠܲܐܟܹ̈ܐ. ܒܕܸܚܠܵܐ ܘܒܲܪܬܹܝܬܼܵܐ ܩܵܝܡ݁ܝܼܢ ܘܲܡܫܲܡܫܝܼܢ ܘܲܡܙܲܝܚܝܼܢ ܠܐܝܼܩܵܪܵܐ ܣܓ݂ܝܼܕ݂ܵܐ ܕܪܲܒ݁ܘܼܬܹܗ. ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸܿܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ̣ܠܥܸܕܵܢ ܡܵܪܵܐ ܕܟ̣ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
ܗܘܼܠܵܠܵܐ. ܐܲܠܵܗܵܐ ܩ̣ܵܡ. ܚܙܝܼ: ܡܙܡܘܖ̈ܐ ܦܒ‐ܦܚ.
ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܫܘܼܒ݂ܚܵܐ ܠܵܟ̣ ܐܲܠܵܗܵܐ ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ݁. ܬܢܝܼ. ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ.
 ܘܡܲܩܦܝܼܢ ܐܲܟܼܚܲܕܼ ܬܸܫܒ݁ܘܿܚܬܵܐ. ܡܫܲܒܲܚܬ ܡܵܪܲܢ ܘܠܵܟ̣ ܝܵܐܹܐ ܠܡܲܣܵܩܘܼ ܫܘܼܒ݂ܚܵܐ ܟܠܝܘܿܡ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ܀ ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ̣ܵܘܕܿܝܼܬܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ̣ ܦܘܼܡܲܢ ܘܝܲܗ̣݇ܒ݂ܠܲܢ ܕܲܢܗܲܠܸܠ ܒܬܸܫܒ̇ܚܵܬܹ̈ܗ. ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ̣ܵܘܕܿܝܼܬܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ̣ ܦܘܼܡܲܢ ܘܝܲܗ̣݇ܒ݂ܠܲܢ ܕܲܢܙܲܡܲܪ ܒܬܸܫܒ̇ܚܵܬܹ̈ܗ. ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ̣ܵܘܕܿܝܼܬܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ̣ ܦܘܼܡܲܢ ܘܝܲܗ̣݇ܒ݂ܠܲܢ ܕܲܢܫܲܒܲܚ ܒܬܸܫܒ̇ܚܵܬܹ̈ܗ܀ ܠܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܫܘܼܒ݂ܚܵܐ ܢܲܣܸܩ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܬܢܝܼ ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܢܵܘܕܸܿܐ ܠܵܟ̣ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ̇ܘܼܬ̣ܵܟ̣܀ ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܲܢܩܲܠܣܵܟ̣ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ̇ܘܼܬ̣ܵܟ̣܀ ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܲܢܫܲܒܚܵܟ̣ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ̇ܘܼܬ̣ܵܟ̣܀ ܕܲܠܘܵܬ̣ ܡܵܝܘܿܬܹ̈ܐ ܡܵܪܵܐ ܡܪܲܚܡܵܢܵܐ. ܚܘܼܢܲܝܢ ܒܛܲܝܒ̇ܘܼܬ̣ܵܟ̣ ܘܐܸܬ̣ܪܲܚܲܡܥܠܲܝܢ܀ ܕܠܵܐ ܙܵܟܹ݁ܐ ܩܕ݂ܵܡܲܝܟ ܟܠ ܕܚܲܝ ܒܲܥܒ݂ܝܼܕܹ̈ܐ. ܐܲܢ݇ܬ̇ ܡܵܪܝ ܐܲܦܢ̣ܝܼܬܵܢ ܡ̣ܢ ܟܠܵܗ̇ ܛܘܼܥܝܲܝ. ܕܐܲܢ݇ܬ̇ܘܼ ܐܲܠܵܗܵܐ. ܘܠܵܟ̣ ܝܵܐܹ݁ܐ ܫܘܼܒ݂ܚܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ: ܘܥܲܠ ܐܲܦܲܝ̈ ܟܠܗܘܿܢ ܥܘܼܕ݂ܖ̈ܵܢܲܝܟ ܘܛܲܝ̈ܒ݁ܘܵܬ݂ܵܟ݂ ܕܲܠܘܵܬܲܢ ܕܠܵܐ ܡܸܬ݂ܦܲܪܥܝܼܢ: ܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ ܒܥܹܕ݁ܬܵܟ݂ ܡܟܲܠܲܠܬܵܐ ܡܲܠܝܲܬ݂ ܟܠ ܥܘܼܕ݂ܖ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܵܐ ܘܒ݂ܵܪܘܿܝܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
	ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܬܲܖ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
(ܓܝܼܘܿܪܵܐ ܕܲܒ݂ܥܹܕ݇ܬܵܐ.( ܚܕ݂ܝܼܬ݂ ܟܲܕ݂ ܐܵܡ̇ܪܝܼܢ ܗ݇ܘ̣ܵܘ ܠܝܼ: ܠܒܲܝܬܹ݁ܗ ܕܡܵܪܝܵܐ ܐܵܙ̇ܠ݇ܝܼܢܲܢ (ܘܲܠܒܲܪ ܡ̣ܢ ܥܹܕ݁ܬܵܐ ܐܵܡ̇ܪܝܼܢ) ܒܟ݂ܠܕܘܼܟ ܐܝܼܬܲܝܟ ܐܲܠܵܗܵܐ ܩܲܒܸ݁ܠ ܬܲܟ݂ܫܲܦ̮ܬܲܢ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
 ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.

ܓܥܵܘ ܠܘܵܬ̣ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܚܲܝܠܬ݂ܵܢܵܐ: ܡ̣ܢ ܟܠܹܗ ܠܸܒ݁ܟ̣ܘܿܢ ܡ̣ܢ ܟܠܵܗ݁ ܢܲܦ̮ܫܟ̣ܘܿܢ: ܡܸܛܠ ܕܐܲܠܵܗܵܐ ܐܲܒ݂ܵܐ ܚܲܢܵܢܵܐ ܗ̣݇ܘ ܘܲܡܪܲܚܡܵܢܵܐ ܘܲܡܪܲܚܦܵܢܵܐ: ܘܠܵܐ ܨܵܒܹ݁ܐ ܒܲܓ݂ܒ݂ܝܼܠܬܹܗ ܕܬܹܐܒܲܕ: ܐܸܠܵܐ ܐܸܢ ܕܲܬ̇ܬ̣ܘܼܒ݂ ܠܘܵܬܹܗ ܢܫܲܡܠܸܐ ܥܲܡܲܢ ܛܲܝܒ̇ܘܼܬܹܗ ܘܖ̈ܲܚܡܵܘܗܝ ܘܲܚܢܵܢܹܗ ܥܕܲܡܵܐ ܠܚܲܪܬܵܐ. ܘܥܵܢܹ݁ܝܢ: ܐܵܡܹܝܢ܀ ܩܘܼܡܘ ܒܚܲܝܠܵܐ ܕܐܲܠܵܗܵܐ: ܘܥܵܢܹ݁ܝܢ ܟܸܢܫܵܐ: ܫܘܼܒ݂ܚܵܐ ܠܲܫܡܹܗ ܩܲܕܝܼܫܵܐ: ܘܩܵܝܡ݁ܝܼܢ. ܘܡܲܩܸܦ. ܒܒ݂ܵܥܘܼܬ݂ܵܐ ܘܲܒ݂ܬܲܟ̣ܫܲܦ̮ܬܵܐ ܡܲܠܲܐܟ̣ܵܐ ܕܲܫܠܵܡܵܐ ܘܲܕ݂ܖ̈ܲܚܡܹܐ ܫܵܐ݁ܠܝܼܢܲܢ܀ ܘܥܵܢܹ݁ܝܢ: ܡ̣ܢ ܠܘܵܬ݂ܵܟ̣ ܡܵܪܝܵܐ. ܠܸܠܝܵܐ ܘܐܝܼܡܵܡܵܐ ܟܠܗܘܿܢ ܝܵܘ̈ܵܡܵܬ̣ܵܐ ܕܚܲܝܲܝ̈ܢ: ܫܲܝܢܵܐ ܐܲܡܝܼܢܵܐ ܠܥܹܕܿܬܵܟ̣ ܘܚܲܝܹ̈ܐ ܕܠܵܐ ܚܛܵܗܹ̈ܐ: ܫܵܐ݁ܠܝܼܢܲܢ...܀ ܐܵܘܝܘܼܬ݂ܵܐ ܘܥܵܢܹܝ̇ܢ: ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ.
	ܘܲܡܨܲܠܸܐ ܟܵܗܢܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ. ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ ܡܲܓ݂ܥ̱ܠܝܼܢܲܢ ܦܲܓ݂ܖ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ: ܘܡ̣ܢܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܫܵܐ̇ܠܝܼܢܲܢ ܫܘܼܒ݂ܩܵܢܵܐ ܕܚܵܘܒܹ̈ܐ ܘܕܲܚܛܵܗܹ̈ܐ: ܗܲܒ݂ܠܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܖ̈ܲܚܡܲܝܟ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ ܟܵܪܘܿܙܘܼܬ݂ܵܐ: ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ.

ܘܐܵܡܪܝܼܢ: ܩܲܕܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.

ܩܲܕܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.

ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܒ݂ܵܬܲܪ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ ܒܓ݂ܘܼܕܵܐ ܕܒܼܵܬܲܪ. ܩܲܕ݁ܝܼܫܵܐ ܘܲܫܒ݂ܝܼܚܵܐ ܘܚܲܝܠ̱ܬ݂ܵܢܵܐ ܘܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܕܲܒ݂ܩܲܕ݁ܝܼܫܹ̈ܐ ܥܵܡܲܪ ܘܡܸܬ݁ܬ݁ܢܝܼܚ ܨܸܒ݂ܝܵܢܹܗ: ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ ܐܸܬ݂ܦܲܢ ܡܵܪܝ: ܘܚܘܼܣ ܘܪܲܚܸܡ ܥܠܲܝܢ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.

ܘܐܵܡܲܪ ܪܲܒܵܐ. ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܬܸܫܡܸܫܬܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
	ܘܡܲܩܸܦ ܟܵܗܢܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ ܨܠܘܿܬ݂ܵܐ. ܘܟܲܕ݂ ܫܵܠ̈ܡܵܢ ܢܲܦ̮̈ܫܵܬܲܢ ܠܲܚܕ݂ܵܐ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܓܡܝܼܪܬܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܟܠܲܢ ܒܲܚܕ݂ܵܐ ܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ: ܢܸܫܬ݁ܘܸܐ ܕܢܲܣܸܩ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܥܘܿܢܝܼܬܵܐ ܕܪܲܡܫܵܐ. ܒܐܲܠܵܗܵܐ ܡܸܠܬ̣ܵܐ. ܐܲܝܟ̣ ܐܲܝܠܵܐ ܕܓ݂ܵܥܹ݁ܐ ܥܲܠ ܦܨܝܼܕ݂ܵܐ ܕܡܲܝ̈ܵܐ: ܣܝܼܡ ܕܸܡܥܲܝ̈ ܩܕ݂ܵܡܲܝܟ ܘܲܒ݂ܣܸܦܪܵܟ̣. ܡ݁ܢ ܢܸܬܸ݁ܠ ܠܝܼ ܡܲܒ݁ܘܼܥܵܐ ܕܣܘܿܓ݂ܵܐܐ ܕܕܸܡܥܹ̈ܐ. ܘܚܘܼܫܵܒܹ̈ܐ ܕܪܸܚܡܲܬ݂ ܢܲܟ̣ܦܘܼܬ݂ܵܐ. ܕܐܸܥ̈ܛܸܐ ܒܲܬ݂ܝܵܒ݂ܘܼܬ݂ܵܐ ܚܵܘܒܲܝ̈ ܘܲܚܛܵܗܲܝ̈ ܕܲܣܥܝܼܪܝܼܢ ܠܝܼ ܒܟ̣ܠܥܸܕܵܢ. ܣܢܝܼܩ ܥܲܠ ܖ̈ܲܚܡܲܝܟ ܐܝܼܬܲܝ ܛܵܒ݂ܵܐ ܘܒܲܣܝܼܡܵܐ. ܡܸܛܠ ܕܡܲܚܒܲܬ݁ ܠܐܲܝܠܹܝܢ ܕܬܵܝ݁ܒ݁ܝܼܢ. ܘܡܸܛܠ ܗܵܢܵܐ ܠܵܐ ܦܵܣܹ݁ܩ ܐ݇ܢܵܐ ܣܲܒ݂ܪܵܐ ܕܚܲܝܲܝ̈ ܡܵܪܝܵܐ: ܬܢܝܼ.

ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ.

 ܡ݁ܢ ܢܸܬܸ݁ܠ ܠܝܼ ܡܲܒ݁ܘܼܥܵܐ ܕܣܘܿܓ݂ܵܐܐ ܕܕܸܡܥܹ̈ܐ. ܘܚܘܼܫܵܒܹ̈ܐ ܕܪܸܚܡܲܬ݂ ܢܲܟ̣ܦܘܼܬ݂ܵܐ. ܕܐܸܥ̈ܛܸܐ ܒܲܬ݂ܝܵܒ݂ܘܼܬ݂ܵܐ ܚܵܘܒܲܝ̈ ܘܲܚܛܵܗܲܝ̈ ܕܲܣܥܝܼܪܝܼܢ ܠܝܼ ܒܟ̣ܠܥܸܕܵܢ. ܣܢܝܼܩ ܥܲܠ ܖ̈ܲܚܡܲܝܟ ܐܝܼܬܲܝ ܛܵܒ݂ܵܐ ܘܒܲܣܝܼܡܵܐ. ܡܸܛܠ ܕܡܲܚܒܲܬ݁ ܠܐܲܝܠܹܝܢ ܕܬܵܝ݁ܒ݁ܝܼܢ. ܘܡܸܛܠ ܗܵܢܵܐ ܠܵܐ ܦܵܣܹ݁ܩ ܐ݇ܢܵܐ ܣܲܒ݂ܪܵܐ ܕܚܲܝܲܝ̈ ܡܵܪܝܵܐ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܟ̣ܠܗܹܝܢ ܥܖ̈ܘܼܒ݂ܵܬܼܵܐ ܕܟ̣ܠܵܗ̇ ܫܲܢ݇ܬܵܐ: ܣܛܲܪ ܡ̣ܢ ܕܲܒ݂ܕ݂ܘܼܟ̣ܖ̈ܵܢܹܐ. ܢܲܚܸܡ ܡܵܪܝ ܥܲܢܝܼܕܲܝ̈ܢ ܒܲܚܢܵܢܵܟ̣: ܘܐܲܩܝܼܡ ܐܸܢܘܿܢ ܡܸܢ ܝܲܡܝܼܢܵܟ̣: ܘܐܲܠܒܸ݁ܫ ܐܸܢܘܿܢ ܫܘܼܒ݂ܚܵܐ ܦܲܐܝܵܐ ܒܡܲܠܟ݁ܘܼܬ݂ܵܟ̣: ܘܚܲܠܸܛ ܐܸܢܘܿܢ ܥܲܡ ܟܹܐܢܹ̈ܐ ܘܙܲܕܿܝܼܩܹ̈ܐ ܓܵܡܖ̈ܲܝ ܨܸܒ݂ܝܵܢܵܟ̣ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ. ܡܵܪܵܐ ܕܡܵܘܬܲܢ ܘܲܕ݂ܚܲܝܲܝ̈ܢ: ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
	ܥܵܢܹ݁ܝܢ: ܐܵܡܹܝܢ.
	ܫܘܼܪܵܝܵܐ. ܘܐܲܢ݇ܬ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܘܲܡܪܲܚܦܵܢܵܐ* ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܘܐܲܢ݇ܬ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܘܲܡܪܲܚܦܵܢܵܐ* ܢܲܓ̇ܝܼܪܵܐ ܪܘܼܚܵܐ ܘܣܲܓ̇ܝܼ ܛܲܝܒ̇ܘܼܬ̣ܵܐ ܘܩܘܼܫܬܵܐ* ܐܸܬ̣ܦܲܢ ܥܠܲܝ ܘܪܲܚܸܡ ܥܠܲܝ* ܗܲܒ݂ ܚܲܝܠܵܐ ܠܥܲܒ݂ܕܵܟ̣ ܘܲܦܪܘܿܩ ܠܒܲܪ ܐܲܡܬ̣ܵܟ̣ * ܘܲܥܒܸܕ ܥܲܡܝ ܐܵܬ̣ܵܐ ܛܵܒ݂ܬܵܐ* ܕܢܸܚܙܘܿܢ ܣܵܢ̈ܐܲܝ ܘܢܸܒ݂ܗ̱ܬ̣ܘܼܢ* ܕܐܲܢ݇ܬ̇ ܡܵܪܝܵܐ ܥܲܕܲܪܬܵܢܝ ܘܒܲܝܲܐܬܵܢܝ܀

ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ

ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܐܵܡܹܝܢ

	ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܣܘܼܒܵܥܵܐ. ܡܵܪܝܵܐ ܐܲܡ̣ܠܸܟ̣. ܐ: ܗܘܼܠܵܠܵܐ. ܥܲܡ ܨܠܘܿܬܹܗ. ܚܙܝܼ: ܡܙܡܘܖܹ̈ܐ ܨܓ‐ ܩܐ.
ܘܐܵܡܪܝܼܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	 ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܕܣܘܼܒܵܥܵܐ ܒܓ݂ܘܼܕܵܐ ܕܒܼܵܬܲܪ. ܐܲܫܘܵܐ ܠܲܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܠܪܲܡܫܵܐ ܕܫܲܝܢܵܐ: ܘܲܠܠܸܠܝܵܐ ܕܲܢܝܵܚܵܐ: ܘܲܠܨܲܦܪܵܐ ܕܲܡܣܲܒ݁ܖ̈ܵܢ ܒܹܗ ܛܵܒ̣̈ܵܬ݂ܵܐ: ܘܠܐܝܼܡܵܡܵܐ ܕܲܥܒ݂ܵܕܹ̈ܐ ܛܵܒܹ̈ܐ ܕܙܲܕܿܝܼܩܘܼܬ݂ܵܐ: ܕܲܒ݂ܗܘܿܢ ܢܪܲܥܸܐ ܠܐܲܠܵܗܘܼܬ݂ܵܟ̣ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ̣ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
	ܥܘܿܢ̈ܝܵܬܼܵܐ ܩܵܠܵܐ ܕܥܲܢܝܼܕܹ̈ܐ. ܪܹܫ ܩܵܠܵܐ. ܡܸܛܠ ܕܩܲܠܝܼܠܘܼ ܙܲܒ݂ܢܵܐ ܘܛܵܒ݂ ܘܲܙܥܘܿܪ: ܡܸܛܠ ܕܟ̣ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܐܲܝܟ̣ ܠܲܗܓܵܐ ܐܸܢܘܿܢ ܩܵܝ݁ܡܝܼܢ. ܩܲܠܝܼܠ ܘܲܙܥܘܿܪ ܙܲܒ݂ܢܵܐ ܕܪܵܕܹܝܢܲܢ ܒܥܵܠܡܵܐ ܗܵܢܵܐ. ܚܲܛܵܝܹ̈ܐ ܬܘܼܒ݂ܘܼܢ ܘܲܚܝܵܘ. ܕܡܵܘܬܵܐ ܫܘܼܚܕܵܐ ܠܵܐ ܢܵܣܹ݁ܒ݂. ܕܲܠܥܲܬ݁ܝܼܖܹ̈ܐ ܘܡܸܣܟܹ݁ܢܹ̈ܐ ܐܲܟܲܚ݇ܕ݂ ܕܵܒ̇ܲܪ ܫܲܘܝܵܐܝܼܬ݂: ܬܢܝܼ. ܫܲܒܲܚܘ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ ܚܕܲܬܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܕܓܲܒ݂ܠܲܢ ܘܠܲܒ݂ܪܵܐ ܕܦܲܪܩܲܢ ܒܲܨܠܝܼܒܹܗ. ܘܲܠܪܘܼܚܵܐ ܦܵܪܲܩܠܹܝܛܵܐ. ܐ݇ܪܵܙܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܐ. ܕܗܘ̤ܝܘܼ ܡܢܲܚܸܡ ܥܲܢܝܼ̈ܕܲܝܢ. ܘܡܲܠܒܸ݁ܫ ܫܘܼܒ݂ܚܵܐ ܦܲܓ݂ܖ̈ܲܝܗܘܿܢ. ܓܒܲܠܬܵܢ̣ܝ ܘܣܵܡ̣ܬ݁ ܥܠܲܝ ܐܝܼܕ݂ܵܟ̣. ܒܨܸܒ݂ܝܵܢܵܟ̣ ܐܸܬ݂ܒ݁ܪܝܼܬ݂ ܘܐܲܝܟ̣ ܕܲܫ̣ܦܲܪ ܠܵܟ̣ ܐܲܬ݂ܩܸܢ̣ܬܵܢܝ. ܘܲܓ݂ܒܲܠ̣ܬܵܢܝ ܡ̣ܢ ܥܲܦܪܵܐ. ܒܲܕ݂ܡܘܼܬ݂ ܨܲܠܡܵܟ̣ ܘܲܕ݂ܡܘܼܬ݂ܵܟ̣. ܘܗܵܫܵܐ ܕܫܸܟ̣ܒܹ݁ܬ݂ ܥܲܠ ܣܲܒ݂ܪܵܟ̣. ܐܲܚܵܢܝ ܕܐܹܩܘܼܡ ܘܐܵܘܕܸܿܐ ܠܵܟ̣. ܡܫܲܕܲܪ ܐܲܢ݇ܬ݁ ܪܘܼܚܵܟ̣ ܘܡܸܬ݂ܒ݁ܪܹܝܢ. ܒܦܘܼܩܕܵܢܵܟ̣ ܡܸܬ݂ܢܲܚܡܝܼܢ ܝܲܠܕܵܘ̈ܗܝ ܕܐܵܕ݂ܵܡ ܥܲܦܪܵܢܵܐ. ܘܩܵܝ݁ܡܝܼܢ ܡ̣ܢ ܓܵܘ ܩܲܒ݂ܖܹ̈ܐ. ܘܡܸܬ݂ܝܲܐܒ݂ܝܼܢ ܠܐܲܬ݂ܪܵܐ ܕܢܘܼܗܪܵܐ. ܘܡܲܣܩܝܼܢ ܫܘܼܒ݂ܚܵܐ ܘܬ݂ܵܘܕܿܝܼܬܼܵܐ ܠܲܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܗܘܿܢ. ܚܲܝܹ̈ܐ ܫܲܐܠܵܟ̣ ܘܝܲܗ̣݇ܒ݂ܬ݁ ܠܹܗ. ܒܲܨܠܝܼܒ݂ܵܟ̣ ܝܲܩܝܼܪܵܐ ܚ̣ܝܵܐ ܓܲܝܵܣܵܐ ܕܗܲܝܡܸܢ ܒܵܟ̣. ܘܲܦܪܲܩ̣ܬܵܝܗܝ ܡ̣ܢ ܚܵܘܒܵܘ̈ܗܝ. ܘܐܲܩܝܼܡ̣ܬܵܝܗܝ ܡ̣ܢ ܝܲܡܝܼܢܵܟ̣. ܘܲܚܢܲܢ ܕܐܵܘܕܝܼܢ ܒܲܨܠܝܼܒ݂ܵܟ̣. ܐܲܚܵܢ ܕܲܢܩܘܼܡ ܘܢܵܘܕܸܿܐ ܠܵܟ̣. ܬܵܘ ܢܸܒ݂ܪܘܼܟ̣ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܹܗ. ܢܵܘܕܸܿܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܹܗ ܠܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ. ܕܲܒ݂ܕܸܢܚܵܐ ܕܡܹܐܬ݂ܝܼܬܹܗ. ܐܸܬ̤ܵܐ ܘܦܲܪܩܲܢ ܒܲܨܠܝܼܒܹܗ. ܘܲܥܬ݂ܝܼܕ݂ ܕܢܹܐܬܸܐ ܘܲܢܢܲܚܸܡ. ܝܲܠܕܵܘ̈ܗܝ ܕܐܵܕ݂ܵܡ ܥܲܦܪܵܢܵܐ. ܪܲܚܸܡܥܠܲܝ ܐܲܠܵܗܵܐ ܐܲܝܟ̣ ܛܲܝܒ݁ܘܼܬ݂ܵܟ̣. ܖ̈ܲܚܡܹܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ̣ ܢܹܐܬ݂ܘܿܢ ܠܐܝܼܵܠܝ ܚܲܢܵܢܵܐ. ܕܗܵܐ ܫ̣ܒܲܢܝ ܡܵܘܬܵܐ ܝܲܥܢܵܐ. ܘܚܲܒܸ݁ܠ̣ ܨܘܼܖ̈ܵܬܹܗ ܕܲܩܢܘܿܡܝ. ܘܲܦܪܲܫ̣ ܢܲܦ̮ܫܵܐ ܡ̣ܢ ܦܲܓ݂ܪܵܐ. ܘܲܗܓܲܡ ܘܲܫ̣ܕ݂ܵܢܝ ܒܓ݂ܵܘ ܩܲܒ݂ܪܵܐ. ܘܚܲܛܵܝܹ̈ܐ ܠܘܵܬ݂ܵܟ̣ ܢܸܬ݂ܦܢܘܿܢ. ܕܲܚܛܹ݁ܝܬ݂ ܝܵܕܲܥ݁ܢܵܐ ܘܲܕ݂ܥܸܒ݂ܕܹ݁ܬ݂ ܥܲܠ ܦܘܼܩܕܵܢܵܟ̣. ܪܲܚܸܡܥܠܲܝ ܐܲܠܵܗܵܐ ܒܝܵܘܡܵܐ ܕܕ݂ܵܢ݁ܚܵܐ ܪܲܒ݁ܘܼܬ݂ܵܟ̣. ܘܠܵܐ ܬܸܬ݁ܕܟܲܪ ܠܝܼ ܣܲܟ̣̈ܠ̱ܘܵܬ݂ܝ. ܕܐܹܩܘܼܡ ܘܐܵܘܕܸܿܐ ܠܛܲܝܒ݁ܘܼܬ݂ܵܟ̣. ܣܲܟ̣ܠ̱ܘ̈ܵܬ݂ܵܐ ܕܛܲܠܝܘܼܬ݂ܝ ܠܵܐ ܬܸܬ݁ܕܟܲܪ ܠܝܼ. ܣܲܟ̣̈ܠ̱ܘܵܬ݂ܝ ܐܸܬ݂ܥܲܗ݁ܕܹܬ݂ ܘܐܲܚ̣ܕܲܢܝ ܙܵܘܥܵܐ ܘܲܪܬܹܝܬܼܵܐ. ܡ̣ܢ ܕܝܼܢܵܐ ܕܟܹܐܢܘܼܬ݂ܵܟ̣. ܡܵܪܵܐ ܕܒ݂ܐܝܼܕ݂ܵܘ̈ܗܝ ܐܸܬ݂ܓܲܒ݂ܠܹ݁ܬ݂. ܫܒ݂ܘܿܩ ܠܝܼ ܚܵܘܒܲܝ̈ ܒܹܝܬ݂ ܕܝܼܢܵܐ. ܕܐܸܦܪܘܿܥ ܫܘܼܒ݂ܚܵܐ ܠܡܵܪܘܼܬ݂ܵܟ̣.
	ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܗܵܘ݁ ܩܵܠܵܐ ܕܲܩ̣ܪܵܝܗܝ ܘܢܲܚ̣ܡܹܗ ܠܠܵܥܵܙܲܪ. ܘܲܓ݂ܠ̣ܵܐ ܘܐܵܘܕܲܥ̣ ܒܹܗ ܫܘܼܒ݂ܚܹܗ. ܘܲܡ̣ܠܲܟ̣ ܚܲܝܹ̈ܐ ܘܢܘܼܚܵܡܵܐ. ܕܟ̣ܠ ܐܲܝܢܵܐ ܕܲܡܗܲܝܡܸܢ ܒܝܼ. ܐܝܼܬ݂ ܠܹܗ ܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ.
	ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ. ܫܘܼܒ݂ܚܵܐ ܠܗܵܘ݁ ܩܵܠܵܐ ܕܩ̇ܵܪܹܐ ܠܡܝܼܬܹ̈ܐ ܡ̣ܢ ܩܲܒ݂ܖܹ̈ܐ. ܘܦܵܬܲܚ݁ ܥܘܼܒܵܗ݁ ܕܲܫܝܘܿܠ. ܘܡܸܬ݁ܬ݁ܥܝܼܪܝܼܢ ܫ̇ܵܟ̣ܒܲܝ̈ ܥܲܦܪܵܐ. ܘܡܲܣܩܝܼܢ ܫܘܼܒ݂ܚܵܐ ܘܬ݂ܵܘܕܿܝܼܬܼܵܐ. ܠܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚ ܩܘܼܕ݂ܫܵܐ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܩܵܢܘܿܢܵܐ. ܒܓ݂ܘܼܕܵܐ ܕܲܩܕܼܵܡ. ܢܲܚܸܡ ܡܵܪܝ ܥܲܢܝܼܕܲܝ̈ܢ ܒܲܚܢܵܢܵܟ̣: ܘܐܲܩܝܼܡ ܐܸܢܘܿܢ ܡܸܢ ܝܲܡܝܼܢܵܟ̣: ܘܐܲܠܒܸ݁ܫ ܐܸܢܘܿܢ ܫܘܼܒ݂ܚܵܐ ܦܲܐܝܵܐ ܒܡܲܠܟ݁ܘܼܬ݂ܵܟ̣: ܘܚܲܠܸܛ ܐܸܢܘܿܢ ܥܲܡ ܟܹܐܢܹ̈ܐ ܘܙܲܕܿܝܼܩܹ̈ܐ ܓܵܡܖ̈ܲܝ ܨܸܒ݂ܝܵܢܵܟ̣ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ. ܡܵܪܵܐ ܕܡܵܘܬܲܢ ܘܲܕ݂ܚܲܝܲܝ̈ܢ: ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	ܩܵܢܘܿܢܵܐ. ܕܥܲܢܝܼܕܹ̈ܐ. ܒܥܹܕܬܵܐ ܚܕ݂ܵܝ. ܐܲܠܵܗܵܐ ܢܪܲܚܸܡ ܥܠܲܝܢ* ܢܒܲܪܟܲܢ ܘܢܲܢܗܲܪ ܐܲܦܵܘ̈ܗܝ ܒܲܢ܀ ܡܵܐ ܕܡܸܬ݂ܓ݁ܠܹܝܬ݁ ܘܲܥܒ݂ܵܕܹ̈ܐ ܕܟ̣ܠ ܕܵܝ݁ܢܲܬ݁. ܠܵܐ ܬܸܫܕܹܝܢ ܡ̣ܢ ܩܕ݂ܵܡ ܐܲܦܲܝ̈ܟ ܡܵܪܝܵܐ. ܕܢܸܕܲܥ ܒܐܲܪܥܵܐ ܐܘܼܖ̈ܚܵܬܹܗ* ܘܲܒ݂ܟ̣ܠܗܘܿܢ ܥܲܡܡܹ̈ܐ ܦܘܼܪܩܵܢܹܗ܀ ܢܵܘܕܿܘܿܢ ܠܵܟ̣ ܥܲܡܡܹ̈ܐ ܐܲܠܵܗܵܐ* ܢܵܘܕܘܿܢ ܠܵܟ̣ ܥܲܡܡܹ̈ܐ ܟܠܗܘܿܢ܀ ܢܸܚ̈ܕܝܵܢ ܘܲܢ̈ܫܲܒܚܵܢ ܡܲܠܟ̈ܘܵܬ̣ܵܐ* ܕܕ݂ܵܐܸܢ ܐܲܢ݇ܬ̇ ܠܥܲܡܡܹ̈ܐ ܬܪܝܼܨܵܐܝܼܬ܀ ܘܲܠܡܲܠܟ̈ܘܵܬ̣ܵܐ ܡܕܲܒܲܪ ܐܲܢ݇ܬ ܒܐܲܪܥܵܐ* ܢܵܘܕܘܿܢ ܠܵܟ̣ ܥܲܡܡܹ̈ܐ ܐܲܠܵܗܵܐ܀ ܢܵܘܕܿܘܿܢ ܠܵܟ̣ ܥܲܡܡܹ̈ܐ ܟܠܗܘܿܢ* ܐܲܪܥܵܐ ܝܸܗ݇ܒܲܬ̤ ܦܹܐܖܹ̈ܝܗ̇܀ ܢܒܲܪܟܲܢ ܐܲܠܵܗܵܐ ܐܲܠܵܗܲܢ ܘܲܢܒܲܪܟܲܢ ܐܲܠܵܗܵܐ* ܘܢܸܕ݂ܚܠܘܼܢ ܡܸܢܹܗ ܟܠܗܘܿܢ ܥܸܒ݂ܖܹ̈ܝܗ̇ ܕܐܲܪܥܵܐ܀ ܡܵܐ ܕܡܸܬ݂ܓ݁ܠܹܝܬ݁ ܘܲܥܒ݂ܵܕܹ̈ܐ ܕܟ̣ܠ ܕܵܝ݁ܢܲܬ݁. ܠܵܐ ܬܸܫܕܹܝܢ ܡ̣ܢ ܩܕ݂ܵܡ ܐܲܦܲܝ̈ܟ ܡܵܪܝܵܐ܀
ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ * ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ.  ܒܗܵܝ݁ ܫܵܥܬܵܐ ܕܝܵܘܡܵܟ̣ ܪܲܒܵܐ ܡܸܬ݂ܓ݁ܠܸܐ. ܚܘܼܢܲܝܢ ܡܵܪܝ ܘܲܒ݂ܛܲܝܒ݁ܘܼܬ݂ܵܟ̣ ܖ̈ܲܚܡܹܐ ܥܒܸܕ݂.
ܐܲܠܵܗܵܐ ܢܪܲܚܸܡ ܥܠܲܝܢ* ܢܒܲܪܟܲܢ ܘܢܲܢܗܲܪ ܐܲܦܵܘ̈ܗܝ ܒܲܢ* ܒܗܵܝ݁ ܫܵܥܬܵܐ ܕܝܵܘܡܵܟ̣ ܪܲܒܵܐ ܡܸܬ݂ܓ݁ܠܸܐ. ܚܘܼܢܲܝܢ ܡܵܪܝ ܘܲܒ݂ܛܲܝܒ݁ܘܼܬ݂ܵܟ̣ ܖ̈ܲܚܡܹܐ ܥܒܸܕ݂.
	ܬܸܫܒ݁ܘܿܚܬܵܐ. ܒܗܵܝ݁ ܫܵܥܬ̣ܵܐ ܕܬ݂ܵܩܹ݁ܢ ܟܘܼܪܣܝܵܟ̣ ܠܒ݂ܘܼܚܪܵܢܵܐ. ܠܵܐ ܬܸܥܘܿܠ ܡܵܪܝܵܐ ܒܕ݂ܝܼܢܵܐ ܥܲܡ ܥܲܒ݂ܕܲܝ̈ܟ ܬܢܝܼ:܀ ܟܵܗܢܵܐ ܘܟ̣ܘܼܡܪܵܐ ܕܒܲܕ݂ܡܹܗ ܦܪܲܩ̣ ܥܵܠܡܹ̈ܐ. ܚܲܣܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ̣ ܚܵܘܒܹ̈ܐ ܕܡܲܪܥܝܼܬ݂ܵܟ̣.
ܟܵܪܘܿܙܘܼܬ݂ܵܐ. ܐܲܚܝܼܕ݂ ܟܠ. ܘܡܸܬ݂ܐܲܡܪܵܐ ܒܟ̣ܠܗܘܿܢ ܣܘܼܒܵܥܹ̈ܐ ܕܨܵܘܡܵܐ.
 ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܫܵܐ ܘܒܲܚܦܝܼܛܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܀ ܡܵܪܝܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ ܐܲܚܝܼܕ݂ ܟܠ ܐܲܠܵܗܵܐ ܕܐܲܒ݂̈ܵܗܵܬܲܢ: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟ̣...܀ ܩܲܕܿܝܼܫܵܐ ܘܲܡܫܲܒ̇ܚܵܐ ܕܲܒ݂ܩܲܕܿܝܼܫܹ̈ܐ ܥܵܡܲܪ ܘܡܸܬ̇ܬ̇ܢܝܼܚ ܨܸܒ݂ܝܵܢܹܗ: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟ̣...܀ ܡܲܠܟܵܐ ܕܡܲܠܟܹ̈ܐ ܘܡܵܪܵܐ ܕܡܵܖ̈ܵܘܵܬ̣ܵܐ ܕܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟ̣...܀ ܗ̇ܘ ܕܐ݇ܢܵܫ ܡ̣ܢ ܒܢܲܝ̈ܢܵܫܵܐ ܠܵܐ ܚ̣ܙܵܝܗܝ ܘܐܵܦܠܵܐ ܡܨܸܐ ܠܡܸܚܙܝܹܗ: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟ̣...܀ ܗ̇ܘ ܕܨܵܒܹ݁ܐ ܕܟ̣ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܢܸܚܘܿܢ. ܘܢܸܬ̣ܦܢܘܿܢ ܠܝܼܕܲܥܬ̣ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟ̣...܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡ̣ܢܟ݂...܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ݁ܘ ܕܲܒܼܖ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ̣ܠ: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟܼ...܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒ݂ܐܲܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟܼ...܀ ܫܲܝܢܵܟ̣ ܘܲܫܠܵܡܵܟ̣ ܐܲܫܪܵܐ ܒܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ: ܘܐܸܬ̣ܪܲܚܸܡܥܠܲܝܢ.
 ܘܟܸܢ ܐܲܪܝܼܡܘ ܩܵܠܟ̣ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ. ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ.܀ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ. ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
 ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀
ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܦܵܬܲܚ ܬܲܪܥܹܗ ܠܲܕ݂ܢܵܩܫ݁ܝܼܢ ܒܹܗ: ܡܦܲܢܸܐ ܫܹܐ̈ܠܵܬ݂ܵܐ ܠܲܕ݂ܫܵܐ݁ܠܝܼܢ ܠܹܗ. ܦܬܲܚ ܠܵܗ̇ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܬܲܪܥܵܐ ܕܖ̈ܲܚܡܹܐ ܠܲܨܠܘܿܬܲܢ. ܘܩܲܒܸ݁ܠܹܝܗ̇ ܠܒ݂ܵܥܘܼܬܲܢ. ܘܦܲܢܵܐ ܒܖ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ ܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܵܟ̣ ܥܲܬ݁ܝܼܪܵܐ ܘܲܫܦܝܼܥܵܐ: ܛܵܒ݂ܵܐ ܕܠܵܐ ܟܵܠܹ݁ܐ ܖ̈ܲܚܡܵܘܗܝ ܘܡܵܘܗܒ݂ܵܬܹ̈ܗ: ܡ̣ܢ ܣܢܝܼܩܹ̈ܐ ܘܐܲܠܝܼܨܹ̈ܐ ܥܲܒ݂ܕܵܘ̈ܗܝ ܕܩ̇ܵܪܹܝܢ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܹܗ ܒܟ̣ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ ܒܲܪܸܟ݂ܡܵܪܝ.
	ܚܵܬ݂ܵܗ̇: ܫ̇ܵܡܲܥ ܒܩܵܠܵܐ ܕܟܹܐܢܹ̈ܐ ܘܲܕ݂ܙܲܕܿܝܼܩܹ̈ܐ ܕܲܡܪܲܥܹ݁ܝܢ ܠܹܗ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܘܥܵܒܹ݁ܕ݂ ܨܸܒ݂ܝܵܢܵܐ ܕܕ݂ܵܚܠܵܘ̈ܗܝ. ܫܡܲܥ ܡܵܪܝ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܒܲܚܢܵܢܵܟ̣: ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܒܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ̣: ܘܚܘܼܢ ܠܐܲܠܝܼܨܹ̈ܐ ܘܠܲܡܛܲܖ̈ܦܹܐ ܥܲܒ݂ܕܲܝ̈ܟ ܕܩ̇ܵܪܹܝܢ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܵܟ̣ ܒܟ̣ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ̣ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ ܒܲܪܸܟ݂ܡܵܪܝ.
	ܨܠܘܿܬ݂ܵܐ. ܥܘܼܕ݂ܪܵܢܵܐ ܡܵܪܝ ܕܖ̈ܲܚܡܲܝܟ ܘܣܘܼܝܵܥܵܐ ܪܲܒܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܚܲܝܠܵܐ ܟܲܣܝܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܘܝܲܡܝܼܢܵܟ݂ ܕܡܲܠܝܵܐ ܖ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ: ܐܲܓܸ݁ܢ ܘܐܲܠܘܵܐ ܠܲܡܚܝܼܠܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܡ̣ܢ ܒܲܝܬܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ ܡܠܸܐ ܟܠ ܥܘܼܕ݂ܖ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܒܲܨܠܘܿܬ݂ ܟܠܗܘܿܢ ܩܲܕ݁ܝܼܫܹ̈ܐ ܡܖ̈ܲܥܝܵܢܲܝܟ: ܡܵܪܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
	ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ ܒܲܪܸܟ݂ܡܵܪܝ.
	ܘܡܲܩܦܝܼܢ. ܒܒ݂ܘܼܪܟ݁ܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܢܸܬ݂ܒܲܪܟ݂ܘܼܢ ܥܲܒ݂ܕܲܝ̈ܟ: ܘܒܲܒ݂ܛܝܼܠܘܼܬܹܗ ܕܨܸܒ݂ܝܵܢܵܟ݂ ܢܸܬ݂ܢܲܛܪܘܼܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܘܫܲܝܢܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܲܫܠܵܡܵܐ ܢܲܓ݁ܝܼܪܵܐ ܕܡܵܪܘܼܬ݂ܵܟ݂: ܢܲܡܠܸܟ݂ ܒܥܲܡܵܟ݂ ܘܲܒ݂ܥܹܕ݁ܬܵܟ݂. ܟܠܗܘܿܢ ܝܵܘܡܵܬܹ̈ܗ ܕܥܵܠܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.

	ܘܡܲܩܦܝܼܢ. ܨܠܘܿܬ݂ܵܗ̇ ܡܵܪܝ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܗ̇ ܕܐܸܡܵܐ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܬܲܟ݂ܫܲܦ̮ܬܵܗ̇ ܘܬܲܚܢܲܢܬܵܗ̇ ܕܡܲܠܝܲܬ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܐ: ܡܵܪܬ݁ܝ ܡܲܪܝܲܡ ܛܘܼܒ݂ܵܢܝܼܬ݂ܵܐ: ܘܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ ܙܲܟܵܝܵܐ: ܘܥܘܼܕ݂ܪܵܢܵܐ ܐܲܠܵܗܵܝܵܐ: ܘܒ݂ܵܥܘܼܬܹܗ ܕܡܵܪܝ ܝܘܿܚܲܢܵܢ ܡܲܥ̱ܡܕ݂ܵܢܵܐ: ܘܕܲܫܠܝܼܚܹ̈ܐ ܛܘܼܒ݂ܵܢܹ̈ܐ: ܘܕܲܐܒ݂ܵܗ̈ܵܬ݂ܵܐ ܢܲܨܝܼܚܹ̈ܐ ܡܣܲܒ݁ܖ̈ܵܢܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܘܙܵܖ̈ܥܲܝ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܕܲܐܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܡܵܪܝ ܐܲܒ݂ܵܐ ܩܵܬ݂ܘܿܠܝܼܩܵܐ: ܘܡܵܪܝ ܐܸܣܛܲܦܵܢܘܿܣ ܒܘܼܟ݂ܪܵܐ ܕܣܵܗܕܹ̈ܐ: ܘܲܕ݂ܓܲܢ݇ܒܵܪ ܚܲܝܠܵܐ ܡܵܪܝ ܓܝܼܘܲܪܓܝܼܣ ܣܵܗܕܵܐ ܢܲܨܝܼܚܵܐ: ܘܲܕ݂ܡܵܪܝ ܦܠܵܢ: ܘܲܕ݂ܡܵܪܝ ܐܵܘܓܹ݁ܝܢ ܛܘܼܒ݂ܵܢܵܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܣܝܼܥܬܹ݁ܗ ܪܘܼܚܵܢܵܝܬܵܐ. ܘܕܲܫܡܘܿܢܝܼ ܘܕܲܒ݂ܢܹܝ̈ܗ̇: ܘܲܕ݂ܟ݂ܠܗܘܿܢ ܣܵܗܕܵܘ̈ܗܝ ܘܩܲܕ݁ܝܼ̈ܫܵܘܗܝ ܕܡܵܪܲܢ: ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܫܘܼܪܵܐ ܪܵܡܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܚܲܣܝܼܢܵܐ: ܕܬܸܦܪܘܿܩ ܘܲܬ݂ܦܲܨܸܐ ܘܲܬ݂ܫܵܘܙܸܒ݂ ܘܲܬ݂ܢܲܛܲܪ ܠܦܲܓ݂ܖ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ ܡ̣ܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹܗ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ.
ܘܥܵܢܹ̇ܝܢ: ܐܵܡܹܝܢ.
ܘܟܸܢ ܚܵܬ݂ܡ̇ܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܒܼܵܬܲܪ. ܘܢܵܣܹ̇ܒ݂ ܟܵܗܢܵܐ ܨܠܝܼܒ݂ܵܐ ܒܐܝܼܕܹܗ ܕܝܲܡܝܼܢܵܐ: ܘܩܵܐܹ̇ܡ ܒܓܲܒܵܐ ܬܲܝܡ̱ܢܵܝܵܐ: ܘܡܲܗܦܸܟ݂ ܐܲܦܵܘ̈ܗܝ ܠܥܲܡܵܐ ܘܐܵܡܲܪ: ܒܲܪܸܟ݂ ܡܵܪܝ ܒܦܘܼܩܕܵܢܟ݂ܘܿܢ.
ܘܥܵܢܹ̇ܝܢ: ܒܦܘܼܩܕܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܡܵܪܲܢ: ܘܫܘܼܒ݂ܚܵܐ ܠܲܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ. ܘܡܲܪܟ݁ܢܝܼܢ ܥܲܡܵܐ ܖܹ̈ܫܲܝܗܘܿܢ.
ܘܡܲܩܸܦ ܚܘܼܬܵܡܵܐ. ܬܸܬ݂ܩܲܒܲܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܚܲܠܵܫܹ̈ܐ ܩܕ݂ܵܡ ܒܹܝܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ̣: ܘܬܸܗܘܸܐ ܟܢܝܼܫܘܼܬܲܢ ܗܵܕܹܐ ܠܲܢܝܵܚ ܨܸܒ݂ܝܵܢ ܡܵܪܘܼܬ݂ܵܟ̣: ܠܡܸܣܲܒ݂ ܡܸܢܵܟ̣ ܒܛܲܝܒ݁ܘܼ ܚܘܼܠܡܵܢܵܐ ܕܦܲܓ݂ܪܵܐ: ܘܢܘܼܛܵܪܵܐ ܕܢܲܦ̮ܫܵܐ: ܘܫܘܼܪܬܵܚܵܐ ܕܡܵܙܘܿܢܵܐ: ( ܘܚܘܼܣܵܝܵܐ ܕܚܵܘܒܹ̈ܐ: ܘܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ ): ܘܫܲܝܢܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ: ܘܲܫܠܵܡܵܐ ܢܲܓ݁ܝܼܪܵܐ: ܘܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ ܕܠܵܐ ܥܵܒ݂ܪܵܐ ܘܠܵܐ ܡܸܫܬܲܪܝܵܐ ܡ̣ܢ ܒܲܝܢܵܬܲܢ: ܒܟ̣ܠܹܗ ܙܲܒ݂ܢܵܐ ܕܩܵܐܹܡ ܒܹܗ ܥܵܠܡܵܐ ܗܵܢܵܐ: ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ̣ܝܵܢܲܢ. ܗܵܫܵܐ... ܌
ܘܪܵܫܹ̇ܡ ܨܠܝܼܒ݂ܵܐ ܥܲܠ ܥܲܡܵܐ: ܟܲܕ݂ ܦܵܫܹ̇ܛ ܐܝܼܕܹܗ ܠܥܸܠ ܠܐܲܦܲܝ̈ ܡܲܥ̱ܪܒ݂ܵܐ: ܘܡܲܝܬܸ݁ܐ ܠܵܗ̇ ܠܡܲܕ݂ܢܚܵܐ: ܘܟܸܢ ܦܵܫܹ̇ܛ ܠܵܗ̇ ܠܓܲܒܵܐ ܕܬܲܝ̱ܡܢܵܝܵܐ: ܘܡܵܘܒܸ݁ܠ ܠܵܗ̇ ܠܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ. ܐܲܝܟܲܢܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܸܣܬ݁ܡܸܟ݂ ܥܲܠ ܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܓܲܒܵܐ ܣܸܡܵܠܵܝܵܐ ܕܟܸܢܫܵܐ.
ܘܥܵܢܹ̇ܝܢ: ܡܫܝܼܚܵܐ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬ݂ܵܟ݂: ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܟܵܗܢܘܼܬ݂ܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
ܒܲܫܪܵܪܵܐ ܓܹܝܪ ܕܠܵܐ ܦܘܼܠܵܓ݂ܵܐ: ܡܗܲܝܡܢܝܼܢܲܢ ܒܚܲܕ݂ ܐܲܠܵܗܵܐ: ܐܲܒ݂ܵܐ ܐܲܚܝܼܕ݂ ܟܠ. ܥܵܒ݂ܘܿܕ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܐܲܝܠܹܝܢ ܕܡܸܬ݂ܚܲܙ̈ܝܵܢ ܘܲܕ݂ܠܵܐ ܡܸܬ݂ܚܲܙ̈ܝܵܢ. ܘܲܒ݂ܚܲܕ݂ ܡܵܪܝܵܐ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ. ܒܪܹܗ ܕܐܲܠܵܗܵܐ ܝܼܚܝܼܕ݂ܵܝܵܐ: ܒܘܼܟ݂ܪܵܐ ܕܟ݂ܠܗܹܝܢ ܒܸܖ̈ܝܵܬ݂ܵܐ. ܗܵܘ̇ ܕܡ̣ܢ ܐܲܒ݂ܘܼܗܝ ܐܸܬ݂ܝܼܠܸܕ݂ ܩܕ݂ܵܡ ܟܠܗܘܿܢ ܥܵܠܡܹ̈ܐ: ܘܠܵܐ ܐܸܬ݂ܥ̱ܒܸܕ݂. ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܕܡ̣ܢ ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܒܲܪ ܟܝܵܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܕܒ݂ܐܝܼ̈ܕ݂ܵܘܗܝ ܐܸܬ݁ܬܲܩܲܢܘ ܥܵܠܡܹ̈ܐ: ܘܐܸܬ݂ܒ݁ܪܝܼ ܟܠ ܡܸܕܸ̇ܡ. ܗܵܘ̇ ܕܡܸܛܠܵܬܲܢ ܒܢܲܝ̈ܢܵܫܵܐ: ܘܡܸܛܠ ܦܘܼܪܩܵܢܲܢ: ܢ̣ܚܸܬ݂ ܡ̣ܢ ܫܡܲܝܵܐ: ܘܐܸܬ݂ܓܲܫܲܡ ܡ̣ܢ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܘܲܗܘ̤ܵܐ ܒܲܪܢܵܫܵܐ: ܘܐܸܬ݂ܒ݁ܛܸܢ ܘܐܸܬ݂ܝܼܠܸܕ݂ ܡ̣ܢ ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܘܚܲܫ̣ ܘܐܸܙܕ݁ܩܸܦ: ܒܝܵܘܡܲܝ̈ ܦܲܢܛܝܼܘܿܣ ܦܝܼܠܵܛܘܿܣ. ܘܐܸܬ݂ܩ̄ܒܲܪ: ܘܩ̣ܵܡ ܠܲܬ݂ܠܵܬ݂ܵܐ ܝܵܘܡܝܼ̈ܢ ܐܲܝܟ݂ ܕܲܟ݂ܬ݂ܝܼܒ݂. ܘܲܣ̣ܠܸܩ ܠܲܫܡܲܝܵܐ: ܘܝܼܬܸܒ݂ ܡ̣ܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܘܬ݂ܘܼܒ݂ ܥܬ݂ܝܼܕ݂ ܠܡܹܐܬ݂ܵܐ ܠܲܡܕ݂ܵܢ ܠܡܝܼܬܹ̈ܐ ܘܲܠܚܲܝܹ̈ܐ. ܘܲܒ݂ܚܲܕ݂ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܪܘܼܚܵܐ ܕܲܫܪܵܪܵܐ: ܗܵܘ̇ ܕܡ̣ܢ ܐܲܒ݂ܵܐ ܢܵܦܹ̇ܩ: ܪܘܼܚܵܐ ܡܲܚܝܵܢܵܐ. ܘܒܲܚܕ݂ܵܐ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܘܲܫܠܝܼܚܵܝܬܵܐ ܩܵܬ݂ܘܿܠܝܼܩܝܼ. ܘܡܵܘܕܹ݁ܝܢܲܢ ܒܲܚܕ݂ܵܐ ܡܲܥܡܘܿܕ݂ܝܼܬ݂ܵܐ: ܠܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ. ܘܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܲܒ݂ܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ: ܐܵܡܹܝܢ.
