ܚܕܒܫܒܐ ܕܬܪܝܢ ܕܨܘܡܐ.
	ܕܲܐَܚܖ̈ܵܝܹܐ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܖ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܕ݂ܥܹܐܕܹ̈ܐ.
ܢܵܘܕܸ݁ܐ ܡܵܪܝ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܒܬܸܫ̈ܒ݁ܚܵܬ݂ܵܐ ܖ̈ܘܼܚܵܢܵܝܵܬܵܐ: ܘܢܸܣܓ݁ܘܿܕ݂ ܠܡܵܪܘܼܬ݂ܵܟ݂ ܒܣܸܓ݂̈ܕܵܬ݂ܵܐ ܐܲܖ̈ܥܵܢܵܝܵܬ݂ܵܐ: ܘܲܢܫܲܒܲܚ ܠܲܟ݂ܝܵܢܵܟ݂ ܟܲܣܝܵܐ ܘܲܓ݂ܢܝܼܙܵܐ ܒܡܲܚܫ̈ܒ݂ܵܬ݂ܵܐ ܕܲܟ݂̈ܝܵܬ݂ܵܐ ܘܠܵܐ ܡܛܵܘ̈ܫܵܬ݂ܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܡܲܪܡܝܼܬܼܵܐ. ܠܵܟ̣ ܝܵܐ݁ܝܵܐ. ܚܙܝܼ: ܡܙܡܘܖ̈ܐ ܣܗ‐ܣܙ
ܘܡ̣ܢ ܒܵܬܲܪ ܫܲܒܲܚ. ܘܥܵܠܲܡ. ܐܵܡ̇ܪܝܼܢ ܒܩܵܠܵܐ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܸܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܐܲܝܟ݂ ܥܸܛܪܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܕ݂ܥܹܐܕܹ̈ܐ: ܠܲܫܡܵܐ ܪܲܒܵܐ ܘܲܕ݂ܚܝܼܠܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܘܲܡܒܲܪܟ݂ܵܐ ܘܛܘܼܒ݂ܬ݂ܵܢܵܐ ܘܠܵܐ ܡܸܬ݁ܕܲܪܟܵܢܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܘܲܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܕܲܠܘܵܬ݂ ܓܸܢܣܲܢ: ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܐܲܝܟ݂ ܥܸܛܪܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܒܓ݂ܘܼܕܹ̈ܐ ܡܸܬ݂ܐܲܡܪܵܐ: ܘܲܡܫܲܪܸܐ ܟܵܗܢܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܖ̈ܵܝܹܐ. ܐܸܢ ܫܘܼܪܵܝ ܕܪܲܡܫܵܐ ܕܩܲܕ݂ܡܵܝܹ̈ܐ ܗ݇ܘ̣. ܘܲܒ݂ܗܸܦܟܵܐ.
ܡܵܐ ܚܲܒ݁ܝܼܒ݂ܝܼܢ ܡܲܫܟ݁ܢܲܝ̈ܟ ܡܵܪܝܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ. ܐܲܝܟ݂ ܥܸܛܪܵܐ ܕܒܸܣܡܹ̈ܐ ܛܵܒܹ̈ܐ ܘܪܹܝܚܵܐ ܕܦܝܼܪܡܵܐ ܒܲܣܝܼܡܵܐ: ܩܲܒܸ݁ܠ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ ܒܵܥܘܼܬ݂ܵܐ ܘܲܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ܀ ܣܲܟ݁ܝܲܬ̤ ܢܲܦ̮ܫܝ ܘܐܸܬ݂ܪܲܓ݂ܪ̱ܓܲܬ̤ ܠܕ݂ܵܖ̈ܵܘܗܝ ܕܡܵܪܝܵܐ. ܐܲܝܟ݂ ܥܸܛܪܵܐ...܀ ܫܲܒܲܚ. ܥܵܠܲܡ. ܐܲܝܟ݂ ܥܸܛܪܵܐ...܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܥܲܠ ܦܝܼܪܡܵܐ. ܢܲܣܸܩ ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܟ݂ܠܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܐܵܘ ܗܵܕܹܐ. ܡܫܝܼܚܵܐ ܕܩܲܒܸ݁ܠ̣ ܕܡܵܐ ܕܣܵܗܕܹ̈ܐ ܒܝܘܿܡ ܩܸܛܠܲܝ̈ܗܘܿܢ: ܩܲܒܸ݁ܠ ܒܸܣܡܵܐ ܗܵܢܵܐ ܡ̣ܢ ܐܝܼܕܲܝ̈ ܚܲܠܵܫܘܼܬ݂ܝ ܒܛܲܝܒ݁ܘܼ ܕܲܚܢܵܢܵܟ݂ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܨܠܘܿܬ݂ܵܐ ܕܠܵܟ݂ܘܼ ܡܵܪܵܐ: ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܕ݂ܥܹܐܕܹ̈ܐ ܘܲܕ݁ܕ݂ܘܼܟ݂ܖ̈ܵܢܹܐ. ܘܥܲܠ ܐܲܦܲܝ̈ ܟܠܗܘܿܢ ܥܘܼܕ݂ܖ̈ܵܢܲܝܟ ܘܛܲܝ̈ܒ݁ܘܵܬ݂ܵܟ݂ ܕܲܠܘܵܬܲܢ ܕܠܵܐ ܡܸܬ݂ܦܲܪܥܝܼܢ: ܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ ܒܥܹܕ݁ܬܵܟ݂ ܡܟܲܠܲܠܬܵܐ ܡܲܠܝܲܬ݂ ܟܠ ܥܘܼܕ݂ܖ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܵܐ ܘܒ݂ܵܪܘܿܝܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܐܵܡ̇ܪܝܼܢ ܠܵܗ̇ ܒܓ݂ܘܼܕܹ̈ܐ: ܓ: ܙܲܒ݂ܢ̈ܝܼܢ.
ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ܀ ܚܕ݂ܝܼ̇ܬ݂ ܟܲܕ݂ ܐܵܡ̇ܪܝܼܢ ܗ݇ܘ̣ܵܘ ܠܝܼ: ܠܒܲܝܬܹ݁ܗ ܕܡܵܪܝܵܐ ܐܵܙ̇ܠ݇ܝܼܢ ܚ݇ܢܲܢ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝ̇ܢܲܢ...܀ ܫܲܒܲܚ: ܥܵܠܲܡ. ܠܵܟ݂ܘܼ ܡܪܵܐ ܕܟ݂ܠܵܐ...܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܡ. ܐܲܢ݇ܬ݁ܘܼ ܡܵܪܝ ܫܲܪܝܼܪܵܐܝܼܬ݂ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܘܲܡܢܲܛܪܵܢܵܐ ܐܲܡܝܼܢܵܐ ܕܚܲܝܲܝ̈ܢ: ܘܠܵܟ݂ ܡܵܪܝ ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	ܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܡ. ܛܲܝܒ݁ܘܼܬܹܗ ܕܡܵܪܝܵܐ ܠܥܵܠܲܡ ܐܸܙܡܲܪ* ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܛܲܝܒ݁ܘܼܬܹܗ ܕܡܵܪܝܵܐ ܠܥܵܠܲܡ ܐܸܙܡܲܪ* ܘܲܠܕ݂ܵܪܕܖ̈ܝܼܢ ܐܵܘܕܲܥ ܗܲܝܡܵܢܘܼܬܹܗ ܒܦܘܼܡܝ* ܡܸܛܠ ܕܐܸܡܲܪܬ݁ ܕܥܵܠܡܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܐ ܢܸܬ݂ܒ݁ܢܸܐ* ܘܲܫܡܲܝܵܐ ܬܲܬ݂ܩܸܢ ܗܲܝܡܵܢܘܼܬ݂ܵܟ̣* ܩܝܵܡܵܐ ܐܲܩܝܼܡܹ݁ܬ݂ ܠܓܲܒ݂ܝܝ* ܘܝܼܡ݁ܝܼܬ݂ ܠܕ݂ܵܘܝܼܕ݂ ܥܲܒ݂ܕ݁ܝ* ܕܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܲܬ݂ܩܸܢ ܙܲܪܥܵܟ̣* ܘܐܸܒ݂ܢܸܐ ܠܕ݂ܵܪܕܵܖ̈ܝܼܢ ܟܘܼܪܣܝܵܟ̣* ܫܲܒܲܚ* ܥܵܠܲܡ܀
	ܘܐܵܡܪܝܼܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܡܵܪܝܵܐ ܩܪܹܝ̇ܬ݂ܵܟ݂: ܠܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ: ܘܠܲܒ݂ܛܝܼܠܘܼܬܹܗ ܕܨܸܒ݂ܝܵܢܵܟ݂ ܕܲܥܠܲܝܢ: ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
	ܐܵܡ݁ܪܝܼܢ. ܡܵܪܝܵܐ ܩܪܹܝܬ݂ܵܟ̣ ܐܲܢ݇ܬ݁ ܥܢܝܼܢܝ. ܚܙܝܼ ܡܙܡܘܖ̈ܐ ܕܡܪܝܐ ܕܩܪܝܬܟ
ܕܡܵܪܝܵܐ ܩܪܹܝ݁ܬ݂ܵܟ̣. ܒܥܲܡ ܓܲܒ݂ܖܹ̈ܐ. ܛܘܼܒ݂ܵܘܗܝ ܠܡ̇ܢ ܕܨܵܒܹܐ ܐܲܢ݇ܬ̇ ܒܹܗ* ܘܲܡܩܲܪܸܒ݂ ܐܲܢ݇ܬ̇ ܠܹܗ ܕܢܸܥܡܲܪ ܒܕܲܝܪܵܟ̣* ܘܢܸܣܒܲܥ ܡ̣ܢ ܛܵܒ݂ܵܬܹ̈ܗ ܕܒܲܝܬܵܟ̣* ܘܡ̣ܢ ܩܲܕܿܝܼܫܘܼܬܹܗ ܕܗܲܝܟ݁ܠܵܟ̣* ܘܡ̣ܢ ܙܲܕܿܝܼܩܘܼܬ̣ܵܟ̣ ܕܚܝܼܠܬܵܐ* ܥܢܝܼܢ ܐܲܠܵܗܵܐ ܦܵܪܘܿܩܲܢ.
	ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܒ݂ܵܬܲܪ: ܫܡܲܥ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܒܲܚܢܵܢܵܟ̣: ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܒܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ̣: ܘܚܘܼܢ ܠܚܲܛܵܝܘܼܬܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ̣ ܘܲܒ݂ܖ̈ܲܚܡܲܝܟ: ܐܵܣ݁ܝܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܡܵܪܵܐ ܕܟ̣ܠ... ܘܥܵܢܹ݁ܝܢ: ܐܵܡܹܝܢ.
	ܫܘܼܪܵܝܵܐ ܕܒ݂ܵܬܲܪ. ܢܵܘܕܿܘܿܢ ܫܡܲܝܵܐ ܬܸܕ݂ܡܖ̈ܵܬ݂ܵܟ̣ ܡܵܪܝܵܐ* ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ. ܢܵܘܕ݁ܘܿܢ ܫܡܲܝܵܐ ܬܸܕ݂ܡܖ̈ܵܬ݂ܵܟ̣ ܡܵܪܝܵܐ* ܘܗܲܝܡܵܢܘܼܬ݂ܵܟ̣ ܒܥܹܕܬܵܐ ܕܩܲܕܿܝܼܫܹ̈ܐ* ܡܲܢ̣ܘܼ ܕܦܵܚܹ݁ܡ ܠܡܵܪܝܵܐ ܒܲܫܡܲܝ̈ ܫܡܲܝܵܐ* ܘܲܕܼܕ݂ܵܡܹ݁ܐ ܠܡܵܪܝܵܐ ܒܲܒ݂ܢܲܝ̈ ܡܲܠܲܐܟܹ̈ܐ* ܐܲܠܵܗܵܐ ܩ̣ܵܡ ܒܥܹܕܿܬܵܐ ܕܩܲܕܿܝܼܫܹ̈ܐ* ܪܲܒ݁ ܘܲܕ݂ܚܝܼܠ ܥܲܠ ܟܠܗܘܿܢ ܕܲܚܕ݂ܵܖ̈ܵܘܗܝ* ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ* ܡܲܢܘܼ ܐܲܟ̣ܘܵܬ݂ܵܟ̣ ܥܲܫܝܼܢܵܐ* ܫܲܒܲܚ* ܥܵܠܲܡ܀
	ܘܕܲܥ ܐܘܿ ܐܲܚܘܼܢ. ܕܲܒ݂ܥܹܕܬܵܐ ܐܵܡ݁ܪܝܼܢ ܫܘܼܪܵܝܵܐ: ܡ̣ܢ ܩܕ݂ܵܡ ܥܘܿܢܝܼܬ݂ܵܐ: ܐܸܡܲܬ݂ܝ ܕܢܸܗܘܸܐ ܩܘܼܕ݂ܵܫܵܐ. ܘܟܸܢ ܐܵܡ݁ܪܝܼܢ ܥܘܿܢܝܼܬ݂ܵܐ ܒܗܲܝܟ݁ܠܵܐ: ܘܬ݂ܵܢܹ݁ܝܢ ܠܵܗ݁ ܒܡܲܕ݂ܒ݁ܚܵܐ: ܐܲܝܟ̣ ܕܲܒ݂ܐ݇ܖ̈ܵܙܹܐ܀ ܘܒ݂ܵܬܲܪ ܕܲܡܫܲܒ݁ܚܝܼܢ: ܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܫܲܒܲܚܘ܀ ܘܐܵܡ݁ܪܝܼܢ ܥܘܼܢܵܝܵܐ܀ ܘܒ݂ܵܬܲܪ ܕܲܡܨܲܠܸܐ ܟܵܗܢܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܡܵܪܝ: ܐܵܡ݁ܪܝܼܢ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ. ܘܡܵܐ ܕܫܸܠܡܲܬ̤: ܥܵܒ݂ܪܝܼܢ ܒܲܫܠܵܡܵܐ ܒܗܲܝܟ݁ܠܵܐ. ܘܲܡܨܲܠܹܝܢ ܟܵܗܢܹ̈ܐ ܐܲܝ̤ ܥܝܵܕ݂ܵܐ: ܘܚܵܬܸ݁ܡ ܒܡܲܕ݂ܒ݁ܚܵܐ ܟܵܗܢܵܐ ܕܩܲܕܸܿܫ̣܀ ܘܐܸܢ ܠܲܝܬ݁ ܐ݇ܖ̈ܵܙܹܐ: ܐܵܡ݁ܪܝܼܢ ܠܹܗ ܠܫܘܼܪܵܝܵܐ ܒܵܬܲܪ ܥܘܿܢܝܼܬܼܵܐ.
	ܘܐܵܡܪܝܼܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ:
ܟܵܪܘܿܙܘܼܬܼܵܐ. ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܐܲܒ݂ܵܐ ܕܖ̈ܲܚܡܹܐ ܘܐܲܠܵܗܵܐ ܕܟ݂ܠܹܗ ܒܘܼܝܵܐܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܦܵܪܘܿܩܲܢ ܘܝܲܨܘܼܦܬ݂ܵܢܲܢ ܘܲܡܦܲܪܢ̱ܣܵܢܵܐ ܕܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ ܘܩܘܼܝܵܡܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ ܘܲܕ݂ܟ݂ܠܗܹܝܢ ܥܹ̈ܕ݂ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܐܲܬ݂ܪܲܢ ܘܥܲܠ ܟܠ ܐܲܬ݂ܖ̈ܵܘܵܢ: ܘܥܲܠ ܐܲܝܠܹܝܢ ܕܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܥܵܡ̇ܪܝܼܢ ܒܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܡܘܼܙܵܓ݂ܵܘ̈ܗܝ ܕܐܵܐܲܪ ܘܟܲܗܝܼܢܘܼܬ݂ܵܐ ܕܫܲܢ݇ܬܵܐ ܘܥܲܠ̈ܠܵܬ݂ܵܐ ܕܦܹܐܖܹ̈ܐ ܘܥܲܠ ܬܘܼܩܵܢܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܡܲܠܟܹ̈ܐ ܐܲܚܝܼܕܲܝ̈ ܫܘܼܠܛܵܢܹܗ ܕܥܵܠܡܵܐ ܗܵܢܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܖ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܩܲܫܝܼܫܹ̈ܐ ܘܲܡܫܲܡܫܵܢܹ̈ܐ ܐܲܖ̈ܬܵܕ݁ܘܿܟ݁ܣܹܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ܗ̇ܘ ܕܥܲܬ݁ܝܼܪ ܒܖ̈ܲܚܡܵܘܗܝ ܘܲܫܦܝܼܥ ܒܲܚܢܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܡ̣ܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ ܐܝܼܬ݂ܵܘܗܝ ܘܲܠܕ݂ܵܪܕܵܖ̈ܝܼܢ ܩܲܝܵܡ ܫܘܼܠܛܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܒܲܟ݂ܝܵܢܹܗ ܐܝܼܬ݂ܵܘܗܝ ܛܵܒ݂ܵܐ ܘܝܵܗܘܿܒ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹܐ ܒܡܵܘܬܹ݁ܗ ܕܚܲܛܵܝܵܐ ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡ̣ܢ ܥܵܘ̣ܠܹܗ ܘܢܸܚܸܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܲܒ݂ܨܵܘܡܹܗ ܩܲܕ݁ܝܼܫܵܐ ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܐܲܦܨܲܚ ܠܲܫܡܲܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܠܹܗ ܗܘ̤ ܟܝܵܢܵܐܝܼܬ݂ ܐܝܼܬ݂ ܠܵܐ ܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܘܝܲܬ݁ܝܼܪܵܐܝܼܬ݂ ܕܐܲܝܠܹܝܢ ܕܲܡܗܲܝܡܢܝܼܢ ܒܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܣܓܵܐ ܥܲܡܲܢ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀
ܢܲܦ̮̈ܫܵܬܲܢ ܘܕܲܚܕ݂ܵܕܹ̈ܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܲܓ݂ܥܸܠ. ܘܥܵܢܹ̇ܝܢ: ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ.
ܘܲܡܨܲܠܸܐ ܟܵܗܢܵܐ. ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ ܡܲܓ݂ܥ̱ܠܝܼܢܲܢ ܦܲܓ݂ܖ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ: ܘܡܸܢܵܟܼ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܫܵܐ̇ܠܝܼܢܲܢ ܫܘܼܒ݂ܩܵܢܵܐ ܕܚܵܘܒܹ̈ܐ ܘܕܲܚܛܵܗܹ̈ܐ: ܗܲܒ݂ܠܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܖ̈ܲܚܡܲܝܟ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ ܟܵܪܘܿܙܘܼܬ݂ܵܐ: ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ.
ܘܐܵܡ̇ܪܝܼܢ ܒܩܵܠܵܐ ܦܫܝܼܛܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܫܲܒܲܚ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܥܵܠܲܡ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܒ݂ܵܬܲܪ ܕܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܘܲܕ݂ܥܹܐܕܹ̈ܐ. ܠܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ ܕܩܲܕ݁ܝܼܫ ܒܲܟ݂ܝܵܢܹܗ: ܘܲܫܒ݂ܝܼܚ ܒܐܝܼܬ݂ܘܼܬܹܗ: ܘܪܵܡ ܘܲܡܥܲܠܲܝ ܡ̣ܢ ܟܠ ܒܐܲܠܵܗܘܼܬܹܗ. ܟܝܵܢܵܐ ܩܲܕ݁ܝܼܫܵܐ ܘܲܣܓ݂ܝܼܕ݂ܵܐ ܘܲܡܒܲܪܟ݂ܵܐ ܕܡ̣ܢ ܥܵܠܲܡ. ܡܵܘܕܹ݁ܝܢܲܢ ܘܣܵܓ݂ܕ݁ܝܼܢܲܢ ܘܲܡܫܲܒ݁ܚܝܼܢܲܢ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ. ܒܲܪܸܟ݂ ܡܵܪܝ: ܐܲܪܟܸ݁ܢܘ ܖܹ̈ܫܲܝܟ݁ܘܿܢ ܠܲܣܝܵܡܝܼܕ݂ܵܐ ܘܩܲܒܸ݁ܠܘ ܒܘܼܪܟ݁ܬ݂ܵܐ.
	ܘܐܵܡܲܪ ܪܲܒܵܐ. ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܬܸܫܡܸܫܬܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ. ܘܡܲܩܸܦ ܨܠܘܿܬ݂ܵܐ. ܘܟܲܕ݂ ܫܵܠ̈ܡܵܢ ܢܲܦ̮̈ܫܵܬܲܢ ܠܲܚܕ݂ܵܐ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܓܡܝܼܪܬܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܟܠܲܢ ܒܲܚܕ݂ܵܐ ܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ: ܢܸܫܬ݁ܘܸܐ ܕܢܲܣܸܩ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ... ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	ܕܒ݂ܵܣܵܠܝܼܩܹ̈ܐ. ܒܫܘܼܪܵܝ ܫܠܝܼܚܹ̈ܐ ܩܲܕܿܝܼܫܹ̈ܐ ܠܒ݂ܝܼܫܲܝ̈. ܬܵܘ ܢܫܲܒܲܚ ܠܡܵܪܝܵܐ: ܡܸܛܠ ܕܥܸܫܢܲܬ̤ ܥܠܲܝܢ ܛܲܝܒ݁ܘܼܬܹܗ. ܕܨܘܼܪܬܵܐ. ܗܵܘ݁ ܕܠܹܗ̤ܘ ܫܘܼܒ݂ܚܵܐ ܘܐܝܼܩܵܪܵܐ ܘܐܘܼܚܕܵܢܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ. ܬܵܘ ܢܵܘܕܸܿܐ ܘܲܢܫܲܒܲܚ ܟܠܲܢ ܠܐܲܠܵܗܲܢ ܛܵܒ݂ܵܐ ܐܲܟ̣ܡܵܐ ܕܣܵܦ݁ܩܝܼܢܲܢ. ܥܲܠ ܥܘܼܕ݂ܖ̈ܵܢܵܘܗܝ ܕܲܠܘܵܬ݂ ܓܸܢܣܲܢ. ܝܲܩ̣ܪܹܗ ܓܹܝܪ ܠܬ݂ܘܼܩܵܢܲܢ ܡ̣ܢ ܫܘܼܪܵܝܵܐ. ܒܫܸܡ ܨܲܠܡܹܗ ܝܲܩܝܼܪܵܐ. ܘܟܲܕ݂ ܚ̣ܣܲܡ ܒܐܝܼܩܵܪܲܢ ܒܥܸܠܕܲܪܲܢ. ܘܲܫ̣ܕ݂ܵܢ ܡ̣ܢ ܫܘܼܒ݂ܚܲܢ. ܐܸܬ݂ܓ݁ܠܝܼ ܘܡܲܠܸܠ̣ ܥܲܡܲܢ ܒܲܒ݂ܪܹܗ ܗܵܘ݁ ܕܐܝܼܬ݂ܵܘܗܝ ܝܵܪܬܵܐ ܘܐܲܒ݂ܵܐ ܕܥܵܠܡܵܐ ܕܲܥܬ݂ܝܼܕ݂. ܘܲܒ݂ܝܲܠܕܹܗ ܟܲܢ̣ܫܲܢ ܡ̣ܢ ܦܸܗܝܵܐ ܕܛܵܥ݁ܝܘܼܬ݂ܵܐ ܠܘܵܬ݂ ܝܼܕܲܥܬܼܵܐ ܕܐܲܠܵܗܘܼܬܹܗ. ܘܲܥ̣ܡܲܕ݂ ܘܝܲܗ̣݇ܒ݂ܠܲܢ ܣܝܼܡܲܬ݂ ܒܢܲܝ̈ܵܐ ܫܲܪܝܼܪܬܵܐ. ܘܨ̣ܵܡ ܘܐܲܩ̣ܢܝܼ ܠܘܼܒ݂ܵܒ݂ܵܐ ܠܫܸܦܠܲܢ ܕܢܸܬ݂ܓܲܢ݇ܒܲܪ ܥܲܠ ܣܵܛܵܢܵܐ. ܘܲܒ݂ܡܵܘܬܹ݁ܗ ܚܲܝܒܹܗ ܠܲܛܪܘܼܢܵܐ. ܘܙܲܟ݁ܝܲܢ ܘܐܲܩ̣ܝܼܡܲܢ ܘܐܲܪܝܼܡܲܢ ܥܲܡܹܗ ܒܬܸܫܒ݁ܘܿܚܬܵܐ: ܬܢܝܼ. ܘܫܲܒܲܚ. ܒܵܗ݁.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ. ܚܘܼܣ ܥܠܲܝܢ ܐܲܢ݇ܬ݁ ܚܲܢܵܢܵܐ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܸܬ݂ܦܲܢ ܥܠܲܝܢ ܡܠܸܐ ܖ̈ܲܚܡܹܐ: ܘܠܵܐ ܡܵܪܝ ܬܲܗܡܸܐ ܚܝܵܪܵܟ݂ ܘܲܒ݂ܛܝܼܠܘܼܬ݂ܵܟ݂ ܡܸܢܲܢ: ܕܲܥܠܲܝܟ݁ܘܼ ܣܲܒ݂ܪܲܢ ܘܬ݂ܘܼܟ݂ܠܵܢܲܢ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ: ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	ܫܘܼܪܵܝܵܐ. ܒܛܘܼܒܲܝܗܘܿܢ܀ ܬܵܘ ܢܸܒ݂ܪܘܼܟ̣ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܹܗ* ܘܲܢܒܲܪܟ̣ܝܼܘܗܝ ܠܡܵܪܝܵܐ ܕܥܲܒ݂ܕܲܢ* ܡܸܛܠ ܕܗܘ̤ܝܘ ܐܲܠܵܗܲܢ* ܘܲܚܢܲܢ ܥܲܡܵܐ ܕܝܼܠܹܗ ܘܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬܹܗ.
	ܕܒܹܝܡ. ܬܵܘ ܣܲܒ݂ܘ ܡܲܓܵܢ ܐ݇ܪܵܙ ܚܘܼܣܵܝܵܐ. ܕܝܵܗܹ݁ܒ݂ ܚܲܝܹ̈ܐ ܘܫܵܒܹ݁ܩ ܚܵܘܒܹ̈ܐ. ܒܵܬܹ̈ܐ. ܗܘ̤ܵܐ ܠܲܢ ܡܵܘܬܹ݁ܗ ܥܸܠܲܬ݂ ܚܲܝܹ̈ܐ. ܘܪܲܥ̣ܝܼ ܠܐܲܒ݂ܘܼܗܝ ܥܲܡ ܒܸܖ̈ܝܵܬܹܗ.
	ܬܸܫܒ݁ܘܿܚܬܵܐ. ܐ݇ܖ̈ܵܙܹܐ ܕܲܢܣܲܒܼܢ ܒܗܲܝܡܵܢܘܼܬܼܵܐ: ܢܸܗܘܘ݁ܢ ܠܲܢ ܡܵܪܝ ܠܚܘܼܣܵܝ ܚܵܘܒܹ̈ܐ܀ ܕܡܘܼܬܼܵܐ ܕܥܲܒܼܕܵܐ ܘܲܕܼܥܵܒܼܘ݁ܕܼܵܐ: ܐܲܢ݇ܬ݁ܘܼ ܡܫܝܼܚܵܐ ܡܲܠܟܵܐ ܕܥܵܠܡܹ̈ܐ܀ ܒܦܲܓܼܪܵܟܼ ܘܲܕܼܡܵܟܼ ܚܲܣܝܼܬ݁ ܘܲܫ̣ܒܲܩܬ݁: ܡܘܼܡܹ̈ܐ ܘܚܵܘܒܹ̈ܐ ܕܟܼܠ ܕܗܲܝܡܸܢ̣ܘ ܒܵܟܼ܀ ܐܲܫܘܵܐ ܠܟܼܠܲܢ ܕܲܒܼܓܸܠܝܵܢܵܟ̣: ܒܦܵܪܹܗܣܝܼܵܐ ܢܸܦܘ݁ܩ ܠܐܘܼܪܥܵܟܼ܀ ܘܥܲܡ ܣܸܕܼܖ̈ܲܝܗܘ݁ܢ ܕܲܫܡܲܝܵܢܹ̈ܐ: ܢܲܣܸܩ ܫܘܼܒܼܚܵܐ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀
ܡܲܩܦܝܼܢ ܒܩܵܠܵܐ ܬܲܖ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܪܝܼܢ: ܒܲܪܸܟܼ ܡܵܪܝ
ܘܡܲܩܦܝܼܢ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܥܲܡ ܩܵܢܘܿܢܵܗ̇: ܐܲܝܟ݂ ܕܲܒ݂ܫܘܼܪܵܝܵܐ.
ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܫܡܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܩܲܕ݁ܝܼܫܵܐ ܢܸܫܬܲܒܲܚ: ܘܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܬܸܣܬ݁ܓܸܕ݂: ܘܡܵܪܘܼܬ݂ܵܟ݂ ܬܸܬ݂ܝܲܩܲܪ: ܘܪܲܒ݁ܘܼܬ݂ܵܟ݂ ܬܸܬ݂ܢܲܨܲܚ: ܘܐܝܼܬ݂ܘܼܬ݂ܵܟ݂ ܬܸܬ݂ܪܲܡܪܲܡ: ܘܥܲܠ ܥܲܡܵܟ݂ ܘܥܵܢ̈ܵܐ ܕܡܲܪܥܝܼܬ݂ܵܟ݂: ܢܓ݂ܘܼܠܘܼܢ ܖ̈ܲܚܡܹܐ ܡܬ݂ܘܿܡܵܝܹ̈ܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.

ܐ݇ܚܪܹܬ݂ܵܐ. ܗܘ̤ܝܼ ܡܵܪܝ ܥܲܡܲܢ ܒܖ̈ܲܚܡܲܝܟ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܘܕܲܒܲܪ ܠܲܢ ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܘܼܪܟ݁ܬ݂ܵܐ ܡܵܪܝ ܕܖ̈ܲܚܡܹܐ: ܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܕܖ̈ܲܚܡܹܐ: ܒܝܲܕ݂ ܡܲܠܲܐܟ݂ܵܐ ܕܖ̈ܲܚܡܹܐ: ܡ̣ܢ ܠܘܵܬ݂ ܐܲܠܵܗܵܐ ܣܲܓ݁ܝܼ ܖ̈ܲܚܡܹܐ: ܬܸܫܬܲܕܲܪ ܘܬܸܫܬܲܦܲܥ ܥܲܠ ܟܸܢܫܹ̈ܐ ܕܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ: ܕܐܲܡܝܼܢܝܼܢ ܘܩܵܪܹܝ̇ܢ ܠܲܫܡܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ: ܘܚܵܝ̇ܪܝܼܢ ܘܲܡܣܲܟܹ݁ܝܢ ܠܖ̈ܲܚܡܹܐ: ܘܲܠܦܘܼܪܩܵܢܵܐ ܕܡ̣ܢ ܠܘܵܬ݂ܵܟ݂: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܡܲܩܦܝܼܢ. ܨܠܘܿܬ݂ܵܗ̇ ܡܵܪܝ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܗ̇ ܕܐܸܡܵܐ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܬܲܟ݂ܫܲܦ̮ܬܵܗ̇ ܘܬܲܚܢܲܢܬܵܗ̇ ܕܡܲܠܝܲܬ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܐ: ܡܵܪܬ݁ܝ ܡܲܪܝܲܡ ܛܘܼܒ݂ܵܢܝܼܬ݂ܵܐ: ܘܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ ܙܲܟܵܝܵܐ: ܘܥܘܼܕ݂ܪܵܢܵܐ ܐܲܠܵܗܵܝܵܐ: ܘܒ݂ܵܥܘܼܬܹܗ ܕܡܵܪܝ ܝܘܿܚܲܢܵܢ ܡܲܥ̱ܡܕ݂ܵܢܵܐ: ܘܕܲܫܠܝܼܚܹ̈ܐ ܛܘܼܒ݂ܵܢܹ̈ܐ: ܘܕܲܐܒ݂ܵܗ̈ܵܬ݂ܵܐ ܢܲܨܝܼܚܹ̈ܐ ܡܣܲܒ݁ܖ̈ܵܢܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܘܙܵܖ̈ܥܲܝ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܕܲܐܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܡܵܪܝ ܐܲܒ݂ܵܐ ܩܵܬ݂ܘܿܠܝܼܩܵܐ: ܘܡܵܪܝ ܐܸܣܛܲܦܵܢܘܿܣ ܒܘܼܟ݂ܪܵܐ ܕܣܵܗܕܹ̈ܐ: ܘܲܕ݂ܓܲܢ݇ܒܵܪ ܚܲܝܠܵܐ ܡܵܪܝ ܓܝܼܘܲܪܓܝܼܣ ܣܵܗܕܵܐ ܢܲܨܝܼܚܵܐ: ܘܲܕ݂ܡܵܪܝ ܦܠܵܢ: ܘܲܕ݂ܡܵܪܝ ܐܵܘܓܹ݁ܝܢ ܛܘܼܒ݂ܵܢܵܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܣܝܼܥܬܹ݁ܗ ܪܘܼܚܵܢܵܝܬܵܐ. ܘܕܲܫܡܘܿܢܝܼ ܘܕܲܒ݂ܢܹܝ̈ܗ̇: ܘܲܕ݂ܟ݂ܠܗܘܿܢ ܣܵܗܕܵܘ̈ܗܝ ܘܩܲܕ݁ܝܼ̈ܫܵܘܗܝ ܕܡܵܪܲܢ: ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܫܘܼܪܵܐ ܪܵܡܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܚܲܣܝܼܢܵܐ: ܕܬܸܦܪܘܿܩ ܘܲܬ݂ܦܲܨܸܐ ܘܲܬ݂ܫܵܘܙܸܒ݂ ܘܲܬ݂ܢܲܛܲܪ ܠܦܲܓ݂ܖ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ ܡ̣ܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹܗ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹ̇ܝܢ: ܐܵܡܹܝܢ.
ܘܟܸܢ ܚܵܬ݂ܡ̇ܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܖ̈ܵܝܹܐ. ܘܢܵܣܹ̇ܒ݂ ܟܵܗܢܵܐ ܨܠܝܼܒ݂ܵܐ ܒܐܝܼܕܹܗ ܕܝܲܡܝܼܢܵܐ: ܘܩܵܐܹ̇ܡ ܒܓܲܒܵܐ ܬܲܝܡ̱ܢܵܝܵܐ: ܘܡܲܗܦܸܟ݂ ܐܲܦܵܘ̈ܗܝ ܠܥܲܡܵܐ ܘܐܵܡܲܪ: ܒܲܪܸܟ݂ ܡܵܪܝ ܒܦܘܼܩܕܵܢܟ݂ܘܿܢ. ܘܥܵܢܹ̇ܝܢ: ܒܦܘܼܩܕܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܡܵܪܲܢ: ܘܫܘܼܒ݂ܚܵܐ ܠܲܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ. ܘܡܲܪܟ݁ܢܝܼܢ ܥܲܡܵܐ ܖܹ̈ܫܲܝܗܘܿܢ.
ܘܡܲܩܸܦ ܚܘܼܬܵܡܵܐ.
ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ: ܨܸܡܚܹܗ ܕܐܲܒ݂ܵܐ ܡܬ݂ܘܿܡܵܝܵܐ: ܝܼܠܝܼܕ݂ܵܐ ܕܠܵܐ ܫܘܼܪܵܝܵܐ: ܠܥܸܠ ܡ̣ܢ ܙܲܒ݂ܢܹ̈ܐ ܘܲܗܘܲܝ̈ܵܐ: ܕܠܲܝܬ݁ ܠܲܢ ܣܲܒ݂ܪܵܐ ܘܣܘܼܟܵܝܵܐ: ܐܸܠܵܐ ܐܸܢ ܐܲܢ݇ܬ݁ ܒܵܪܘܿܝܵܐ. ܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܘܲܓ݂ܒܲܝ̈ܵܐ: ܕܲܫ̣ܦܲܪܘ ܠܵܟ݂ ܡ̣ܢ ܫܘܼܪܵܝܵܐ: ܥܒܸܕ݂ ܠܲܚܛܵܗܲܝ̈ܢ ܚܘܼܣܵܝܵܐ: ܘܲܠܣܲܟ݂ܠ̱̈ܘܵܬܲܢ ܫܪܵܝܵܐ: ܘܲܠܥܵܩ̈ܵܬܲܢ ܦܘܼܨܵܝܵܐ: ܘܲܠܫܸܐ̈ܠܵܬܲܢ ܦܘܼܢܵܝܵܐ: ܘܡܲܢܲܥܲܝܢ ܠܢܘܼܗܪܵܟ݂ ܓܲܐܝܵܐ: ܘܢܲܛܲܪܲܝܢ ܒܢܝܼܫܵܟ݂ ܚܲܝܵܐ: ܡ̣ܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܟܣܲܝ̈ܵܐ ܘܲܓ݂ܠܲܝ̈ܵܐ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ). ܗܵܫܵܐ ♰ ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܪܵܫܹ̇ܡ ܨܠܝܼܒ݂ܵܐ ܥܲܠ ܥܲܡܵܐ: ܟܲܕ݂ ܦܵܫܹ̇ܛ ܐܝܼܕܹܗ ܠܥܸܠ ܠܐܲܦܲܝ̈ ܡܲܥ̱ܪܒ݂ܵܐ: ܘܡܲܝܬܸ݁ܐ ܠܵܗ̇ ܠܡܲܕ݂ܢܚܵܐ: ܘܟܸܢ ܦܵܫܹ̇ܛ ܠܵܗ̇ ܠܓܲܒܵܐ ܕܬܲܝ̱ܡܢܵܝܵܐ: ܘܡܵܘܒܸ݁ܠ ܠܵܗ̇ ܠܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ. ܐܲܝܟܲܢܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܸܣܬ݁ܡܸܟ݂ ܥܲܠ ܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܓܲܒܵܐ ܣܸܡܵܠܵܝܵܐ ܕܟܸܢܫܵܐ.
ܘܥܵܢܹ̇ܝܢ: ܡܫܝܼܚܵܐ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬ݂ܵܟ݂: ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܟܵܗܢܘܼܬ݂ܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
ܒܲܫܪܵܪܵܐ ܓܹܝܪ ܕܠܵܐ ܦܘܼܠܵܓ݂ܵܐ: ܡܗܲܝܡܢܝܼܢܲܢ ܒܚܲܕ݂ ܐܲܠܵܗܵܐ: ܐܲܒ݂ܵܐ ܐܲܚܝܼܕ݂ ܟܠ. ܥܵܒ݂ܘܿܕ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܐܲܝܠܹܝܢ ܕܡܸܬ݂ܚܲܙ̈ܝܵܢ ܘܲܕ݂ܠܵܐ ܡܸܬ݂ܚܲܙ̈ܝܵܢ. ܘܲܒ݂ܚܲܕ݂ ܡܵܪܝܵܐ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ. ܒܪܹܗ ܕܐܲܠܵܗܵܐ ܝܼܚܝܼܕ݂ܵܝܵܐ: ܒܘܼܟ݂ܪܵܐ ܕܟ݂ܠܗܹܝܢ ܒܸܖ̈ܝܵܬ݂ܵܐ. ܗܵܘ̇ ܕܡ̣ܢ ܐܲܒ݂ܘܼܗܝ ܐܸܬ݂ܝܼܠܸܕ݂ ܩܕ݂ܵܡ ܟܠܗܘܿܢ ܥܵܠܡܹ̈ܐ: ܘܠܵܐ ܐܸܬ݂ܥ̱ܒܸܕ݂. ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܕܡ̣ܢ ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܒܲܪ ܟܝܵܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܕܒ݂ܐܝܼ̈ܕ݂ܵܘܗܝ ܐܸܬ݁ܬܲܩܲܢܘ ܥܵܠܡܹ̈ܐ: ܘܐܸܬ݂ܒ݁ܪܝܼ ܟܠ ܡܸܕ݁ܡ. ܗܵܘ̇ ܕܡܸܛܠܵܬܲܢ ܒܢܲܝ̈ܢܵܫܵܐ: ܘܡܸܛܠ ܦܘܼܪܩܵܢܲܢ: ܢ̣ܚܸܬ݂ ܡ̣ܢ ܫܡܲܝܵܐ: ܘܐܸܬ݂ܓܲܫܲܡ ܡ̣ܢ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܘܲܗܘ̤ܵܐ ܒܲܪܢܵܫܵܐ: ܘܐܸܬ݂ܒ݁ܛܸܢ ܘܐܸܬ݂ܝܼܠܸܕ݂ ܡ̣ܢ ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܘܚܲܫ̣ ܘܐܸܙܕ݁ܩܸܦ: ܒܝܵܘܡܲܝ̈ ܦܲܢܛܝܼܘܿܣ ܦܝܼܠܵܛܘܿܣ. ܘܐܸܬ݂ܩ̄ܒܲܪ: ܘܩ̣ܵܡ ܠܲܬ݂ܠܵܬ݂ܵܐ ܝܵܘܡܝܼ̈ܢ ܐܲܝܟ݂ ܕܲܟ݂ܬ݂ܝܼܒ݂. ܘܲܣ̣ܠܸܩ ܠܲܫܡܲܝܵܐ: ܘܝܼܬܸܒ݂ ܡ̣ܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܘܬ݂ܘܼܒ݂ ܥܬ݂ܝܼܕ݂ ܠܡܹܐܬ݂ܵܐ ܠܲܡܕ݂ܵܢ ܠܡܝܼܬܹ̈ܐ ܘܲܠܚܲܝܹ̈ܐ. ܘܲܒ݂ܚܲܕ݂ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܪܘܼܚܵܐ ܕܲܫܪܵܪܵܐ: ܗܵܘ̇ ܕܡ̣ܢ ܐܲܒ݂ܵܐ ܢܵܦܹ̇ܩ: ܪܘܼܚܵܐ ܡܲܚܝܵܢܵܐ. ܘܒܲܚܕ݂ܵܐ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܘܲܫܠܝܼܚܵܝܬܵܐ ܩܵܬ݂ܘܿܠܝܼܩܝܼ. ܘܡܵܘܕܹ݁ܝܢܲܢ ܒܲܚܕ݂ܵܐ ܡܲܥܡܘܿܕ݂ܝܼܬ݂ܵܐ: ܠܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ. ܘܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܲܒ݂ܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ: ܐܵܡܹܝܢ.
