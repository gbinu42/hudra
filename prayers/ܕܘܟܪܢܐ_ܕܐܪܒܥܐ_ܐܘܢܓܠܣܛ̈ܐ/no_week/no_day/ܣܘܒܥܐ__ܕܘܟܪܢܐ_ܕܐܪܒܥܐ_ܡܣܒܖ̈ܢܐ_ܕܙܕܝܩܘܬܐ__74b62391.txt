ܣܘܼܒܵܥܵܐ
ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܣܘܼܒܵܥܵܐ. ܐܲܫܘܵܐ ܠܲܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܠܪܲܡܫܵܐ ܕܫܲܝܢܵܐ. ܘܲܠܠܸܠܝܵܐ ܕܲܢܝܵܚܵܐ. ܘܲܠܨܲܦܪܵܐ ܕܡܣܲܒܲܪ ܒܛܵܒ݂ܵܬ݂̈ܵܐ. ܘܠܐܝܼܡܵܡܵܐ ܕܲܥܒ݂ܵܕܹ̈ܐ ܛܵܒܹ̈ܐ. ܕܲܒ݂ܗܘܿܢ ܢܪܲܥܸܐ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘܡܵܬ݂̈ܵܐ ܕܚܲܝܲܝ̈ܢ. ܡܵܪܵܐ ܕܟ݂ܠ.
ܣܘܼܒܵܥܵܐ. ܒܡܵܪܵܐ ܕܟ݂ܠܵܐ. ܪ̈ܵܚܡܵܘܗܝ ܕܡܵܪܝܵܐ ܣܵܢܹܝܢ ܠܒ݂ܝܼܫܬܵܐ: ܒܟ݂ܠ݈ܵܗ̇ ܐܲܪܥܵܐ ܢܸܦܩܲܬ݀ ܣܒܲܪܬ݂ܗܘܿܢ. ܫܠܝܼܚܹ̈ܐ ܪ̈ܵܚܡܵܘܗܝ ܕܲܡܫܝܼܚܵܐ. ܡܲܬܲܝ ܘܡܲܪܩܘܿܣ ܥܲܡ ܠܘܼܩܵܐ. ܐܵܦ ܝܘܿܚܲܢܵܢ ܟܵܪ̈ܘܿܙܹܐ. ܚܕ݂ܘܿܓ݂ܵܘܗ̈ܝ ܕܚܲܬ݂ܢܵܐ ܫܡܲܝܵܢ̈ܵܐ. ܒܥܵܘ ܘܐܸܬ݂ܟܲܫܲܦܘ ܡ̣ܢ ܡܵܪܟ݂ܘܿܢ. ܚܢܵܢܵܐ ܘܪ̈ܲܚܡܹܐ ܥܲܠ ܥܵܠܡܵܐ. ܕܐܲܚܝܼܕ݂ ܓܵܘܣܵܐ ܒܲܨܠܘܿܬ݂ܟ݂ܘܿܢ. ܕܢܸܗܘܸܐ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ: ܬܢܝܼ. ܡܗܲܝܡܲܢܘܼ ܡܵܪܝܵܐ ܒܡܸܠܵܘܗ̈ܝ. ܓܢܘܿܢܵܐ ܕܲܡܠܸܐ ܒܘܼܣܵܡܹ̈ܐ. ܛܲܝܸܒ݂ ܡܫܝܼܚܵܐ ܠܩܲܕܝܼܫܵܘܗ̈ܝ. ܕܲܦܠܲܚܘ ܘܲܥܼܡܲܠܘ ܘܐܸܬ݂ܟܲܫܲܪܘ. ܒܐܸܫܟܵܪ ܚܲܝܹ̈ܐ ܪ̈ܘܼܚܵܢܹܐ. ܗܲܒ݂ܠܲܢ ܡܵܪܲܢ ܒܲܚܢܵܢܵܟ݂. ܕܥܲܡܗܘܿܢ ܢܵܘܕܸܐ ܠܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܘܢܸܙܡܲܪ ܫܘܼܒ݂ܚܵܐ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܕܪܲܡܼܪܸܡܬ݁ ܠܓܸܢܣܲܢ ܡܵܝܘܿܬ݂ܵܐ. ܘܲܒ݂ܢܵܘܗ̈ܝ ܕܲܡܪܲܝܡܵܐ ܟܠܟ݂ܘܿܢ. ܪ̈ܲܒܲܝ ܒܵܬܹ̈ܐ ܕܐܲܠܵܗܵܐ. ܒܢܲܝ̈ ܝܵܪܬ݁ܘܼܬܹܗ ܕܲܡܫܝܼܚܵܐ. ܫܠܝܼܚܹ̈ܐ ܕܬܲܠܼܡܸܕ݂ܘ ܠܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܐܲܦܢܼܝܼܘ ܠܥܲܡܡܹ̈ܐ ܡ̣ܢ ܛܘܼܥܝܲܝ. ܨܠܘܿܬ݂ܟ݂ܘܿܢ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܲܢ. ܘܲܬ݂ܢܲܛܲܪ ܠܲܢ ܡ̣ܢ ܒܝܼܫܵܐ. ܘܢܲܫܘܹܝܢ ܡܫܝܼܚܵܐ ܠܲܚܙܵܬܹܗ. ܒܝܵܘܡܵܐ ܪܲܒܵܐ ܕܓܸܠܝܵܢܹܗ. ܕܲܨܠܘܿܬ݂ܵܐ. ܡ̣ܢ ܒܲܣܝܼܡܘܼܬ݂ܵܐ ܕܙܵܟ݂ܘܼܬ݂ܵܐ ܕܝܲܡܝܼܢܵܟ݂. ܡ̣ܢ ܒܹܝܬ݂ܓܲܙܵܟ݂ ܡܵܪܵܐ ܕܟ݂ܠ. ܫܹܐ̈ܠܵܬ݂ ܥܲܒ݂ܕܲܝ̈ܟ ܢܸܬ݂ܦܲܢ̈ܝܵܢ. ܕܐܲܢ݇ܬ݁ܘܼ ܐܝܼܬܲܝܟ ܐܝܼܵܠܗܘܿܢ. ܘܲܥܠܲܝܟ݁ܘܼ ܣܝܼܡ ܬܘܼܟ݂ܠܵܢܗܘܿܢ. ܢܲܛܲܪ ܐܸܢܘܿܢ ܒܲܚܢܵܢܵܟ݂. ܘܣܲܝܲܥ ܐܸܢܘܿܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. ܘܗܲܒ݂ ܠܗܘܿܢ ܐܝܼܕ݂ܵܐ ܕܥܘܼܕ݂ܪ̈ܵܢܹܐ. ܕܠܵܟ݂ ܡܵܪܝ ܬܸܫܒ݁ܘܿܚܬܵܐ ܢܲܣܩܘܼܢ. ܐܲܝܟ݂ ܥܲܝܢܹ̈ܐ ܕܥܲܒ݂ܕܹ̈ܐ ܠܘܵܬ݂ ܡܵܪ̈ܲܝܗܘܿܢ. ܠܘܵܬ݂ܵܟ݂ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܗܵܐ ܬܠܸܐ ܚܝܵܪܵܐ ܕܲܡܥܵܩܹ̈ܐ. ܐܲܣܵܐ ܟܹܐܒܹ̈ܐ ܘܟ݂ܘܼܪ̈ܗܵܢܹܐ. ܕܐܲܝܠܹܝܢ ܕܲܪܡܹܝܢ ܒܢܸܣܝܘܿܢܹ̈ܐ. ܐܲܝܟ݂ ܕܐܸܫܬܵܘܕܝܼܬ݁ ܒܲܣܒܲܪܬ݂ܵܟ݂. ܡܲܠܝܲܬ݂ ܚܲܝܹ̈ܐ ܘܒ݂ܘܼܣܵܡܹ̈ܐ. ܕܐܲܝܢܵܐ ܕܐܲܡܝܼܢ ܘܢܵܩܹܫ. ܒܲܬ݂ܪܲܥ ܡܵܪܘܼܬ݂ܝ ܡܸܬ݂ܦܬܲܚ ܠܹܗ.  ܕܡܲܪܝܲܡ܀ ܡܸܛܠ ܕܡ̣ܢ ܠܘܵܬܹܗ ܐܸܢܘܿܢ ܪ̈ܲܚܡܹܐ. ܘܣܲܓ݁ܝܼ ܠܘܵܬܹܗ ܦܘܼܪܩܵܢܵܐ. ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܩܲܕܝܼܫܬܵܐ. ܐܲܦܝܼܣܝ ܘܲܒ݂ܥܵܝ ܠܲܡܫܝܼܚܵܐ. ܕܢܸܥܒܸ݁ܕ݂ ܪ̈ܲܚܡܹܐ ܥܲܠ ܥܵܠܡܵܐ. ܕܐܲܚܝܼܕ݂ ܓܵܘܣܵܐ ܒܲܨܠܘܿܬܹܟ݂ܝ. ܘܥܹܕܬܵܐ ܬܸܚܕܸܐ ܒܥܲܕ݂ܥܹܕܹܟ݂ܝ. ܘܝܲܠܕܹ̈ܝܗ̇ ܒܓ݂ܵܘܵܗ̇ ܢܸܬ݂ܢܲܛܪܘܼܢ. ܡ̣ܢ ܢܸܟ݂ܝܵܢܹ̈ܐ ܘܣܲܩܘܼܒ݂̈ܠܹܐ. ܕܐܵܟܹ݁ܠܩܲܪܨܵܐ ܡܵܪܘܿܕ݂ܵܐ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ ܐܲܢ݇ܬ݁ ܥܼܢܲܝܬ݁ ܐܸܢܘܿܢ. ܐܲܠܵܗܵܐ ܬܵܒ݂ܘܿܥܵܐ ܗܘܲܝܼܬ݁ ܠܗܘܿܢ. ܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܘܙܲܕܝܼܩܹ̈ܐ. ܢܒ݂ܝܹ̈ܐ ܫܠܝܼܚܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ. ܘܣܵܗܕܹ̈ܐ ܘܟ݂ܵܗܢܹ̈ܐ ܘܥܵܢܘܵܝܹ̈ܐ. ܕܪܲܥܼܝܘܼܟ݂ ܘܲܫܼܦܲܪܘ ܠܡܵܪܘܼܬ݂ܵܟ݂. ܒܲܛܸܠ ܩܹܐܪ̈ܣܹܐ ܘܲܩܪ̈ܵܒܹܐ. ܘܫܲܝܸܢ ܟܵܗܢܹ̈ܐ ܥܲܡ ܡܲܠܟܹ̈ܐ. ܘܐܲܦܢܵܐ ܒܪ̈ܲܚܡܲܝܟ ܠܪ̈ܲܚܝܼܩܹܐ. ܘܢܲܛܲܪ ܒܪܸܡܙܵܟ݂ ܠܩܲܪ̈ܝܼܒܹܐ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܚܵܘܝܼ ܡܵܪܝܵܐ ܦܘܼܪܩܵܢܹܗ. ܨܠܝܼܒ݂ܵܐ ܕܝܼܫܘܿܥ ܥܠܵܘܗܝ ܐܸܨܛܠܸܒ݂. ܡܵܘܬܵܐ ܘܚܲܝܹܐ ܡܸܢܹܗ ܢܒܲܥܼܘ. ܡܵܘܬܵܐ ܠܥܲܡܵܐ ܕܠܵܐ ܗܲܝܼܡܸܢ. ܘܚܲܝܹ̈ܐ ܠܥܲܡܡܹ̈ܐ ܕܗܲܝܼܡܸܢܘ ܒܹܗ. ܟܹܐܢ ܕܲܝܵܢܵܐ ܘܕܝܼܢܹܗ ܬܪܝܼܨ. ܕܥܲܡܵܐ ܕܨܲܠܒܹܗ ܕܲܝܪܹܗ ܚܪܲܒ݂. ܘܥܲܡܡܹ̈ܐ ܕܗܲܝܼܡܸܢܘ ܒܡܵܪܘܼܬܹܗ. ܫܘܼܒ݂ܚܵܐ ܒܥܹܕܬܹ݁ܗ ܗܵܐ ܙܵܡܪܝܼܢ. ܕܐܲܒ݂ܘܼܢ܀ ܢܸܥܒܸ݁ܕ݂ ܡܵܪܝܵܐ ܟܠܹܗ ܨܸܒ݂ܝܵܢܵܟ݂. ܐܲܒ݂ܘܼܢ ܕܲܢܨܲܚܬ݁ ܒܵܐܓ݂ܘܿܢܵܐ. ܗܵܐ ܒܲܫܡܲܝ̈ܵܐ ܦܘܼܪܥܵܢܵܟ݂. ܡܫܝܼܚܵܐ ܕܨܲܒܸ݁ܬ݁ܬ݁ ܠܹܗ ܩܢܘܿܡܵܟ݂. ܒܥܹܕܬܹ݁ܗ ܐܵܘܪܸܒ݂ ܕܘܼܟ݂ܪܵܢܵܟ݂. ܦܝܼܪܡܵܐ ܕܲܟ݂ܝܵܐ ܚܘܼܒܵܟ݂ ܗܘܵܐ. ܘܪܲܥܼܝܼܬ݁ ܠܡܵܪܵܟ݂ ܒܝܲܕ݂ ܥܲܡܠܲܝ̈ܟ. ܐܲܦܝܼܣ ܥܲܡܲܢ ܕܡܵܐ ܕܡܸܬ݂ܓ݁ܠܸܐ. ܒܫܘܼܒ݂ܚܹܗ ܪܲܒܵܐ ܢܪܲܚܸܡܥܠܲܝܢ.
ܕܥܲܢܝܼܕܹ̈ܐ. ܗܲܠܟܹܬ݂ ܒܗܲܝܡܵܢܘܼܬ݂ܵܐ. ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܵܟ݂ ܡܲܠܟܵܐ ܡܫܝܼܚܵܐ. ܒܗܲܝܡܵܢܘܼܬ݂ܵܐ ܫܠܸܡܼܘ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈. ܫܲܕܲܪܬ݁ ܓܹܝܪ ܡ̣ܢ ܡܪܵܘܡܵܐ ܕܩܘܼܕ݂ܫܵܟ݂. ܘܩܲܒܸ݁ܠܬ݁ ܪܘܼܚܝ ܒܝܲܕ݂ ܦܘܼܩܕܵܢܵܟ݂. ܡܸܟܹ݁ܝܠ ܐܲܢܝܼܚܲܝܢܝ ܒܲܫܠܵܡܵܐ ܒܲܓ݂ܢܘܿܢ ܢܘܼܗܪܵܐ ܕܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܐܲܥܼܒܲܪܬ݁ ܟܠ݈ܵܗ̇ ܚܸܡܬ݂ܵܟ݂. ܥܬ݂ܝܼܕ݂ ܗ݇ܘܼ ܗܵܢܵܐ ܕܡܸܬ݂ܚܲܒܲܠ. ܕܢܸܠܒܲܫ ܠܲܕ݂ܠܵܐ ܒܵܠܹܐ ܘܡܸܬ݂ܚܲܒܲܠ. ܚܲܝ ܗ݇ܘܼ ܠܥܵܠܲܡ ܗܵܘ ܕܲܒ݂ܪܵܝܗܝ ܘܐܲܩܼܝܼܡܹܗ ܡ̣ܢ ܠܵܐ ܡܸܕܸܡ. ܕܲܒ݂ܩܵܠܹܗ ܚܲܝܵܐ ܡܸܬ݂ܢܲܚܲܡ ܡ̣ܢ ܓܵܘ ܩܲܒ݂ܪܵܐ ܠܡܲܠܟ݁ܘܼܬ݂ܵܐ.
ܫܲܒܲܚ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܕܲܓ݂ܒ݂ܵܟ݂ܘܿܢ. ܫܠܝܼܚܹ̈ܐ ܓܒܲܝ̈ܵܐ ܘܩܲܕܝܼܫܹ̈ܐ. ܫܘܼܒ݂ܚܵܐ ܠܲܒ݂ܪܵܐ ܕܲܒ݂ܚܲܝܠܹܗ. ܚܠܵܦܵܘܗܝ ܒܐܵܓ݂ܘܿܢܵܐ ܩܵܡܬ݁ܘܿܢ. ܫܘܼܒ݂ܚܵܐ ܠܪܘܼܚܵܐ ܩܲܕܝܼܫܵܐ. ܕܲܟܠܝܼܠܲܝ̈ܟ݁ܘܿܢ ܨܲܒܸ݁ܬ݂ ܓܕܲܠ. ܨܠܘܿܬ݂ܟ݂ܘܿܢ ܫܘܼܪܵܐ ܬܸܗܘܸܐ ܠܲܢ. ܥܕܲܡܵܐ ܠܫܘܼܠܵܡܹܗ ܕܥܵܠܡܵܐ. ܥܵܠܲܡ. ܘܵܠܹܐ ܘܙܵܕܹܩ ܕܲܢܫܲܒܲܚ. ܟܠܲܢ ܒܝܵܘܡܵܐ ܕܕ݂ܘܼܟ݂ܪܵܢܗܘܿܢ. ܕܐܲܪܒ݁ܥܵܐ ܣܵܦܪܹ̈ܐ ܕܪܘܼܚܩܘܼܕ݂ܫܵܐ. ܡܲܬܲܝ ܘܡܲܪܩܘܿܣ ܥܲܡ ܠܘܼܩܵܐ. ܘܡܵܪܝ ܝܘܿܚܲܢܵܢ ܢܲܨܝܼܚܵܐ. ܕܡܸܠܬܹܗ ܕܐܲܒ݂ܵܐ ܓܠܼܵܐ ܘܒܲܕܸܩ. ܕܢܸܫܬ݁ܘܸܐ ܕܥܲܡܗܘܿܢ ܢܸܬ݂ܒܲܣܲܡ. ܒܗܵܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܠܵܐ ܥܵܒ݂ܪܵܐ.
ܩܵܢܘܿܢܵܐ. ܒܛܘܼܒ݂ ܠܐܲܝܢܵܐ. ܫܲܒܲܚܘ ܙܲܕܝܼܩܹ̈ܐ ܠܡܵܪܝܵܐ. (ܡܙܡܘܪܐ ܠܓ) ܫܠܝܼܚܹ̈ܐ ܩܲܕܝܼܫܹ̈ܐ ܨܲܠܵܘ ܕܥܹܕܵܬ݂̈ܵܐ ܢܸܬ݂ܩܲܝ̈ܡܵܢ. ܘܢܸܬ݂ܥܲܒ݂ܕܘܼܢ ܥܲܕ݂ܥܹܕܲܝ̈ܟ݁ܘܿܢ. ܥܲܝܢܵܘܗ̈ܝ ܕܡܵܪܝܵܐ ܥܲܠ ܙܲܕܝܼܩܹ̈ܐ܀ ܘܐܸܕ݂ܢܵܘܗ̈ܝ ܠܡܸܫܡܲܥ ܐܸܢܘܿܢ܀ ܐܲܦܵܘܗ̈ܝ ܕܡܵܪܝܵܐ ܥܲܠ ܒܝܼܫܹ̈ܐ܀ ܕܢܸܥܛܸܐ ܡ̣ܢ ܐܲܪܥܵܐ ܕܘܼܟ݂ܪܵܢܗܘܿܢ܀ ܓܥܵܘ ܙܲܕܝܼܩܹ̈ܐ ܘܡܵܪܝܵܐ ܫܡܲܥܼ ܐܸܢܘܿܢ ܘܦܲܨܝܼ ܐܸܢܘܿܢ܀ ܩܲܪܝܼܒ݂ܘܼ ܡܵܪܝܵܐ ܠܲܬ݂ܒ݂ܝܼܪ̈ܲܝ ܠܸܒܵܐ܀ ܘܲܠܡܲܟ݁ܝܼܟܲܝ̈ ܪܘܼܚܵܐ ܦܵܪܹܩ܀ ܣܲܓ݁ܝܼ̈ܐܵܢ ܒܝܼܫܵܬܹ̈ܗ ܕܙܲܕܝܼܩܵܐ܀ ܘܡ̣ܢ ܟܠܗܹܝܢ ܡܦܲܨܸܐ ܠܹܗ ܡܵܪܝܵܐ܀ ܘܢܵܛܲܪ ܠܟ݂ܠܗܘܿܢ ܓܲܪ̈ܡܵܘܗܝ܀ ܕܚܲܕ݂ ܡܸܢܗܘܿܢ ܠܵܐ ܢܸܬ݁ܬ݁ܒܲܪ܀ ܩܵܛܠܵܐ ܠܪ̈ܲܫܝܼܥܹܐ ܒܝܼܫܬܵܐ܀ ܘܣܵܢܐܵܘܗ̈ܝ ܕܙܲܕܝܼܩܵܐ ܢܣܘܼܦܘܼܢ܀ ܢܸܦܪܘܿܩ ܡܵܪܝܵܐ ܢܲܦ̮̈ܫܵܬ݂ܵܐ ܕܥܲܒ݂ܕܵܘܗ̈ܝ܀ ܘܠܵܐ ܢܸܬ݂ܚܲܝܒ݂ܘܼܢ ܟܠ ܕܲܡܣܲܒ݁ܪܝܼܢ ܒܹܗ܀ ܒܲܨܠܘܿܬ݂ ܡܲܬܲܝ ܘܡܲܪܩܘܿܣ ܘܲܕ݂ܠܘܼܩܵܐ ܘܲܕ݂ܝܘܿܚܲܢܵܢ. ܛܲܪ ܥܹܕܬܵܟ݂ ܦܵܪܘܿܩܲܢ ܡ̣ܢ ܨܹܢܥܵܬܹ̈ܗ ܕܲܢܟ݂ܝܼܠܵܐ܀ ܫܘܼܒ݂ܚܵܠܵܟ݂ ܡܵܪܲܢ ܝܑܼܫܘܿܥ ܕܠܲܫܠܝܼܚܲܝ̈ܟ ܩܲܕܝܼܫܹ̈ܐ. ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܡܼܠܲܝܬ݁ ܐܸܢܘܿܢ܀ ܐܘܿ ܕܐܲܠܸܦ̮ ܠܲܢ ܕܲܢܨܲܠܸܐ ܘܐܸܫܬܵܘܕܝܼ ܕܩܘܿܫܘ ܘܦܵܬܲܚܢܵܐ. ܦܬܲܚܠܵܗ̇ ܬܲܪܥܵܐ ܠܲܨܠܘܿܬܲܢ.
ܬܸܫܒ݁ܘܿܚܬܵܐ. ܨܠܘܿܬ݂ܗܘܿܢ ܕܩܲܕܝܼܫܲܝ̈ܟ ܕܦܸܬ݂ܚܲܬ݀ ܬܪܲܥ ܪܵܘܡܵܐ. ܗܝ ܡܵܪܝ ܬܸܦ̮ܬܲܚ ܠܵܗ̇ ܬܲܪܥܵܐ ܠܲܨܠܘܿܬܲܢ܀ ܫܠܝܼܚܹ̈ܐ ܩܲܕܝܼܫܹ̈ܐ ܐܲܦܝܼܣܘ ܠܲܡܫܝܼܚܵܐ. ܕܲܒ݂ܝܲܕ݂ ܨܠܵܘܵܬ݂ܟ݂̈ܘܿܢ ܢܸܗܘܘܿܢ ܥܠܲܝܢ ܪ̈ܲܚܡܹܐ.
ܟܵܪܘܿܙܘܼܬ݂ܵܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ܀ ܡܵܪܝܵܐ ܚܲܝܠ݈ܬ݂ܵܢܵܐ ܐܲܚܝܼܕ݂ ܟܠ ܐܲܠܵܗܵܐ ܕܐܲܒ݂ܵܗ̈ܵܬܲܢ. ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܩܲܕܝܼܫܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܲܒ݂ܩܲܕܝܼܫܹ̈ܐ ܥܵܡܲܪ ܘܡܸܬ݁ܬ݁ܢܝܼܚ ܨܸܒ݂ܝܵܢܹܗ. ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܡܲܠܟܵܐ ܕܡܲܠܟܹ̈ܐ ܘܡܵܪܵܐ ܕܡܵܪ̈ܵܘܵܬ݂ܵܐ ܕܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ. ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ ܕܐ݇ܢܵܫ ܡ̣ܢ ܒܢܲܝ̈ܢܵܫܵܐ ܠܵܐ ܚܙܵܝܗܝ. ܘܐܵܦܠܵܐ ܡܨܸܐ ܨܡܸܚܙܝܹܗ. ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܗܵܘ ܕܨܵܒܹ݁ܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܢܸܚܘܿܢ ܘܢܸܬ݂ܦܢܘܿܢ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ. ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ܀ ܗܵܘ ܕܒܲܫܡܲܝ̈ܵܐ܀ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂ ܐܲܫܪܵܐ ܒܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ. ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
