ܛܲܟܼܣܵܐ ܕܬܸܫܡܸܫܬܵܐ
ܕܠܸܠܝܵܐ ܘܕܨܲܦܪܵܐ
 ܒܫܸܡ ܐܲܒܼܵܐ ܘܲܒܼܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕܫܵܐ ܠܥܵܠܡܝܼܢ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒܿܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܖ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒܿܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟܼ: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟܼ: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬܼܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟܼ: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟܼ: ܠܡܲܣܵܩܘܼ ܠܵܟܼ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬܼܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓܼܕ݁ܬ݂ܵܐ ܒܟܼܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟܼܠ.
ܘܥܵܢܹܝ̇ܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ.

ܨܠܘܿܬ݂ܵܐ. ܚܲܝܸܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܠܲܡܚܝܼܠܘܼܬܲܢ ܒܲܚܢܵܢܵܟ݂: ܘܠܲܒܸ݁ܒ݂ ܘܥܲܕܲܪ ܠܲܙܥܘܿܪܘܼܬ݂ ܢܲܦ̮ܫܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܥܝܼܪ ܕܲܡܟ݁ܘܼܬ݂ ܪܸܥܝܵܢܲܢ: ܘܐܲܩܸܠ ܝܘܼܩܪܵܐ ܕܗܲܕܵܡܲܝ̈ܢ: ܘܐܲܫܝܼܓ݂ ܘܲܡܪܘܿܩ ܨܵܐܬ݂ܵܐ ܕܚܵܘܒܲܝ̈ܢ ܘܕܲܚܛܵܗܲܝ̈ܢ: ܘܐܲܢܗܲܪ ܠܚܸܫܘܿܟ݂ܘܼܬܹܗ ܕܡܲܕ݁ܥܲܢ: ܘܐܵܘܫܸܛ ܘܫܲܪܲܪ ܘܗܲܒ݂ ܒܲܢ ܚܲܝܠܵܐ ܘܐܝܼܕ݂ܵܐ ܕܥܘܼܕ݂ܖ̈ܵܢܹܐ: ܕܒ݂ܵܗ̇ ܢܩܘܼܡ ܘܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܨܠܘܿܬ݂ܵܐ. ܚܲܝܠܵܐ ܡܵܪܝ ܟܲܣܝܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܥܘܼܕ݂ܪܵܢܵܐ ܬܡܝܼܗܵܐ ܕܡܵܪܘܼܬ݂ܵܟ݂: ܘܣܘܼܝܵܥܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂: ܗܘ̤ ܢܚܲܝܸܠ ܠܲܡܚܝܼܠܘܼܬܹܗ ܕܲܟ݂ܝܵܢܲܢ ܚܲܠܵܫܵܐ: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܘܥܵܢܹܝ̇ܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܐܲܠܵܗܵܐ. ܡܵܪܝܵܐ ܪܲܚܸܡ ܥܠܲܝܢ.

ܘܡ̣ܢ ܒܵܬܲܪ ܕܲܫܠܸܡܘ ܗܘܼܠܵܠܹ̈ܐ: ܐܸܢ ܡܸܬ݂ܥ̱ܒܸܕ݂ ܩܘܼܪܒܵܢܵܐ ܒܝܵܘܡܵܐ ܗ̇ܘ. ܐܵܡ̇ܪܝܼܢ ܨܠܘܿܬ݂ܵܐ ܕܩܵܠ̈ܝܵܬ݂ܵܐ.
	ܨܠܘܿܬ݂ܵܐ. ܠܲܟ݂ܝܵܢ ܟܲܣܝܘܼܬ݂ܵܟ݂ ܕܠܵܐ ܡܸܬ݁ܕ݁ܪܸܟ݂ ܘܠܵܐ ܡܸܣܬܲܝܲܟ݂ ܡ̣ܢ ܚܘܼܫܵܒ݂ܵܐ ܘܣܘܼܟܵܠܵܐ ܕܟ݂ܠܗܘܿܢ ܥܒ݂ܝܼܕܹ̈ܐ: ܝܵܐܹ̇ܐ ܡܵܪܝ ܫܘܼܒ݂ܚܵܐ ܘܙܵܕ݂ܩ̇ܝܼܢ ܩܵܠܹ̈ܐ ܕܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ: ܘܦܲܐܝܵܐ ܘܡܸܬ݁ܬ݁ܚܝܼܒ݂ܵܐ ܣܸܓ݂ܕ݁ܬ݂ܵܐ: )ܡ̣ܢ ܟܠ ܕܲܒ݂ܪܲܝܬ݁ ܘܐܲܬ݂ܩܸܢܬ݁) ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

 ܒܚܲܝܠܵܟ݂ ܪܲܒܵܐ: ܦܲܓܸܕܲܝܗܝ ܡ݂ܢ ܨܹܐܕܲܝܗܘܿܢ. ܘܚܲܝܸܠ ܐܸܢܘܿܢ: ܕܢܸܫܦܪܘܼܢ ܠܵܟ݂ ܒܕ݂ܘܼܒܵܪ̈ܲܝܗܘܿܢ

܀

 ܘܡܵܐ ܕܡܸܬ݂ܓܠܹܝܬ݁ ܒܬܸܫܒܘܿܚܬܵܐ: ܢܸܦܩܘܼܢ ܠܐܘܼܪܥܵܟ݂ ܥܲܡ ܩܲܕܝܼܫܲܝ̈ܟ. ܘܒܲܓ݂ܢܘܿܢ ܢܘܼܗܪܵܐ ܬܠܝܼܬ݂ܵܐܝܼܬ݂ ܢܪܲܡ̱ܪܡܘܼܢܵܟ݂. ܡܵܪܝܵܐ ܪ̈ܲܚܡܲܝܟ ܠܥܵܠܲܡ. ܫܘܼܒ݂ܚܵܐ ܠܪ̈ܲܚܡܲܝܟ: ܡܫܝܼܚܵܐ ܡܲܠܟܲܢ: ܕܚܵܢ̇ܝܼܢ ܡܲܓܵܢ: ܠܚܲܛܵܝܘܼܬܲܢ. ܘܡܲܫܦܥܝܼܢ ܚܢܵܢܵܐ: ܘܫܘܼܒ݂ܩܵܢ ܚܵܘܒܹ̈ܐ: ܥܲܠ ܡܚܝܼܠܘܼܬܲܢ. ܘܟ݂ܠ ܥܘܼܕ݂ܪ̈ܵܢܝܼܢ: ܡܲܣܪܚܝܼܢ ܠܵܗ̇: ܠܚܲܠܵܫܘܼܬܲܢ

܀

 ܘܡܲܓܪܝܼܢ ܪܘܼܚܵܐ: ܥܲܠ ܣܲܒ݂ܪܵܐ ܕܲܬ݂ܝܵܒ݂ܘܼܬܲܢ. ܘܝܵܗܒܝܼܢ ܚܲܝܠܵܐ: ܘܥܘܼܕ݂ܪܵܢܵܐ ܠܲܨܪܝܼܟ݂ܘܼܬܲܢ

܀

 ܐܲܫܘܵܢ ܡܵܪܝ ܒܛܲܝܒܘܼܬ݂ܵܟ݂: ܕܢܸܫܦܲܪ ܠܵܟ݂ ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܵܟ݂. ܘܲܢܪܲܡܪܸܡ ܘܢܸܣܓܘܿܕ݂ ܠܵܟ݂: ܘܠܲܐܒ݂ܘܼܟ݂ ܥܲܡܵܟ݂ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܣܲܒ݂ܪܵܐ ܕܟ݂ܠܗܘܿܢ ܣܵܘܦܹܝ̈ܗ̇ ܕܐܲܪܥܵܐ. ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܲܢ: ܠܵܟ݂ ܩܵܪܹܝ̇ܢܲܢ: ܬܵܐ ܠܥܘܼܕ݂ܪܵܢܲܢ: ܪܵܚܹ̇ܡ ܐ݇ܢܵܫܵܐ. ܘܲܚܙܝܼ ܠܥܵܠܡܵܐ: ܕܲܪܗܝܼܒ݂ ܘܲܪܕ݂ܝܼܦ: ܡ݂ܢ ܫܘܼܠܛܵܢܹ̈ܐ. ܪܕ݂ܝܼܦܝܼܢ ܥܘܼܡܖܹ̈ܐ: ܫܚܝܼܩܝܼܢ ܡܗܲܝܡܢܹ̈ܐ: ܒܪܸܚܡܲܬ݂ ܟܸܣܦܵܐ. ܪܡܹܝܢ ܒܐܘܼܖ̈ܚܵܬ݂ܵܐ: ܛܠܵܝܹ̈ܐ ܘܢܸܫܹ̈ܐ: ܩܕ݂ܵܡ ܚܲܝ̈ܘܵܬ݂ܵܐ. ܬܠܹܝܢ ܒܲܙܩܝܼܦܵܐ: ܓܲܒ݂ܖܹ̈ܐ ܕܩܘܼܫܬܵܐ: ܘܝܵܨܘܿܦܹ̈ܐ ܕܥܹܕ݁ܬܵܐ

܀

 ܐܘܿܗ ܡ݂ܢ ܐܸܒ݂ܠܵܐ: ܕܥܲܠ ܣܲܓܝܼ̈ܐܹܐ ܗܵܐ ܪܡܸܐ. ܒܸܟ݂ܝܵܐ ܘܐܸܠܝܵܐ: ܡܠܸܐ ܦܘܼܡܵܐ ܕܟ݂ܠ ܒܣܲܪ

܀

 ܐܘܿ ܡܵܪܹܐ ܡܵܪ̈ܵܘܵܬ݂ܵܐ: ܥܩܘܿܪ ܫܸܖ̈ܫܵܘܗܝ ܫܪܝܼ ܫܘܼܠܛܵܢܹܗ. ܣܚܘܿܦ ܟܘܼܪܣܝܹܗ ܕܕ݂ܵܐܹܢ ܥܵܘ̣ܠܵܐ: ܘܠܲܢ ܚܘܼܢܲܝܢ ܒܪ̈ܲܚܡܲܝܟ: ܘܥܲܠ ܥܵܠܡܵܐ ܬܸܟ݂ܪܸܐ ܠܵܟ݂. ܒܐܝܼܡܵܡܵܐ ܘܲܒ݂ܠܸܠܝܵܐ. ܐܸܒ݂ܟܸܐ ܥܲܠ ܢܲܦ̮ܫܝ: ܘܐܲܫܦܲܥ ܕܸܡ̈ܥܹܐ: ܕܡܲܠܝ̈ܵܢ ܚܲܫܵܐ: ܥܲܡ ܟܲܪܝܘܼܬ݂ܵܐ. ܘܐܲܪܩܸܕ݂ ܘܐܲܝܠܸܠ: ܥܲܠ ܩܢܘܿܡܝ ܕܵܘܝܵܐ: ܒܐܸܒ݂ܠܵܐ ܪܲܒܵܐ

܀

 ܕܲܓ݂ܡܲܪܘ ܝܵܘܡܲܝ̈: ܥܲܡ ܫܢܲܝ̈ ܚܲܝܲܝ̈ ܕܠܵܐ ܝܘܼܬ݂ܪܵܢܵܐ. ܘܗܵܐ ܡܲܝܠܸܠ ܐ݇ܢܵܐ: ܡ݂ܢ ܚܵܘܒܲܝ̈ ܘܣܘܼܖ̈ܚܵܢܲܝ. ܕܲܥ̣ܫܸܢ ܟܘܼܪܗܵܢܝ: ܘܠܲܝܬ݁ ܐܵܣܝܵܐ ܕܡܲܐܣܸܐ ܠܝܼ

܀

 ܐܸܬ݁ܬ݁ܥܝܼܪܝ ܢܲܦ̮ܫܝ ܕܵܘܝܼܬ݂ܵܐ: ܘܐܸܬ݂ܟܲܫܲܦܝ ܠܐܵܣܝܵܐ ܛܵܒ݂ܵܐ. ܕܢܲܚܠܸܡ ܟܹܐܒܲܝ̈ܟܝ ܘܲܢܪܲܚܸܡ ܥܠܲܝܟܝ. ܐܲܕ݂ܪ̱ܟ݂ܘܼܢܝ ܚܛܵܗܲܝ̈ ܘܠܵܐ ܐܸܫܟܚܹܬ݂ ܠܡܸܚܙܵܐ. ܚܛܵܗܹ̈ܐ ܕܛܲܠܝܘܼܬ݂ܝ: ܐܸܬ݂ܵܘ ܠܥܘܼܗܕܵܢܝ: ܘܒܲܕ݂ܡܘܼܬ݂ ܨܸܨܸ̈ܐ: ܗܵܐ ܡܩܲܒܥܝܼܢ ܒܝܼ. ܘܕܸܚܠܹܬ݂ ܕܕܲܠܡܵܐ: ܐܵܬܹܐ ܡܵܘܬܵܐ: ܡܒܲܠ̱ܗܝܵܢܵܐ ܕܢܲܦ̮ܫܝ. ܘܢܸܣܲܒ݂ ܘܢܲܪܡܹܝܢܝ: ܒܓ݂ܵܘ ܬܵܘܵܢܹ̈ܐ: ܕܲܡܠܹܝܢ ܚܸܫܟܵܐ

܀

 ܐܸܒ݂ܟܸܐ ܥܲܠ ܢܲܦ̮ܫܝ: ܕܠܵܐ ܩܲܒܠܲܬ̤ ܬܝܵܒ݂ܘܼܬ݂ܵܐ. ܕܝܸܕ݂ܥܹ̇ܬ݂ ܕܬܲܡܵܢ: ܠܲܝܬ݁ ܐܲܬ݂ܪܵܐ ܠܲܬ݂ܝܵܒ݂ܘܼܬ݂ܵܐ

܀

 ܠܘܵܬ݂ ܡܲܢ ܐܹܙܲܠ: ܘܐܸܬ݁ܬܲܠܡܲܕ݂. ܘܲܒ݂ܡܲܢ ܐܹܚܘܿܕ݂ ܒܹܝܬ݂ ܓܵܘܣܵܐ. ܠܘܵܬ݂ܵܟ݂ ܡܫܝܼܚܵܐ ܓܵܥܹܐ ܐ݇ܢܵܐ ܚܘܼܣ ܥܠܲܝ. ܕܡܲܪܝܲܡ܀ ܒܗܸܕ݂ܪ̈ܲܝ ܩܘܼܕ݂ܫܵܐ ܡ݂ܢ ܡܲܪܒܥܵܐ. ܒܬ݂ܘܼܠܬܵܐ ܩܲܕܝܼܫܬܵܐ: ܕܒ݂ܵܗ̇ ܐܸܬ݂ܩܲܕܲܫ: ܓܸܢܣܲܢ ܕܐܸܬ݁ܛܡܲܐ: ܒܢܸܟ݂ܠܵܗ̇ ܕܚܵܘܵܐ. ܐܲܦܝܼܣܝ ܥܲܡܲܢ: ܠܲܡܩܲܕܫܵܢܹܟ݂ܝ: ܕܲܢܩܲܕܸܫ ܠܲܢ. ܘܲܒ݂ܛܸܠܵܠܵܐ: ܕܲܨܠܵܘܵܬܹ̈ܟ݂ܝ: ܢܣܲܬܲܪ ܚܲܝܲܝ̈ܢ

܀

 ܘܢܸܦܪܘܿܣ ܟܸܢ̈ܦܹܐ ܕܲܚܢܵܢܹܗ ܥܲܠ ܫܲܦܠܘܼܬܲܢ. ܘܲܢܫܵܘܙܸܒ݂ ܠܲܢ ܒܟ݂ܠ ܥܸܕܵܢ̈ܝܼܢ ܡ݂ܢ ܢܸܟ݂ܝܵܢܹ̈ܐ

܀

 ܐܘܿ ܐܸܡܹܗ ܕܡܲܚܝܵܢܲܢ. ܐܘܿ ܐܲܡܬܹܗ ܕܥܵܒ݂ܘܿܕܲܢ. ܗܘ̤ܵܝ ܠܲܢ ܫܘܼܪܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܒܟ݂ܠܥܸܕܵܢ. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܡܸܛܠ ܕܬܸܫܒܘܿܚܬܵܐ ܕܥܘܼܫܢܲܢ ܐܲܢ݇ܬ݁ܘܼ. ܡܫܝܼܚܵܐ ܪܹܫܝܼܬܲܢ: ܢܒ݂ܝܼܹ̈ܐ ܒܪܘܼܚܵܐ: ܩܲܕܸܡܘ ܒܲܕܸܩܘ: ܥܲܠ ܡܹܐܬ݂ܝܼܬ݂ܵܟ݂. ܘܒܲܕ݂ܡܵܐ ܕܲܟ݂ܝܵܐ: ܕܲܩܢܘܿܡܲܝ̈ܗܘܿܢ: ܚܬܲܡܘ ܥܲܡܠܲܝ̈ܗܘܿܢ

܀

 ܢܒ݂ܝܼܹ̈ܐ ܒܥܲܡܵܐ: ܘܲܫܠܝܼܚܹ̈ܐ ܒܥܲܡܡܹ̈ܐ. ܬܲܠ̣ܡܸܕ݂ܘ ܘܐܲܠܸܦ̣ܘ: ܬܵܘܕܝܼܬ݂ܵܐ ܓܡܝܼܪܬܵܐ

܀

 ܐܝܼܙܓܲܕܹ̈ܐ ܕܲܫܠܵܡܵܐ. ܟܵܖ̈ܘܿܙܹܐ ܕܬܲܪܥܘܼܬ݂ܵܐ. ܒܥܵܘ ܡ݂ܢ ܡܵܪܟ݂ܘܿܢ ܫܘܼܒ݂ܩܵܢܵܐ ܠܢܲܦ̮̈ܫܵܬܲܢ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܡ݂ܢ ܩܕ݂ܵܡ ܡܵܪܝܵܐ ܗܘ̤ܵܬ݂ ܗܵܕܹܐ ܘܬܸܕ݂ܡܘܼܪܬܵܐ ܗ݇ܝ̣ ܒܥܲܝܢܲܝ̈ܢ. ܙܲܝܢܵܐ ܕܪܘܼܚܵܐ: ܠܒܸܫ ܗ݇ܘ̣ܵܐ ܡܲܠܟܵܐ: ܩܘܼܣܛܲܢܛܝܼܢܘܿܣ: ܘܒܹܗ ܐܸܬ݂ܢܲܨܲܚ. ܘܲܫ̣ܩܲܠ ܥܲܡܹܗ: ܢܝܼܫܵܐ ܚܲܝܵܐ: ܕܲܨܠܝܼܒ݂ ܢܘܼܗܪܵܐ. ܕܢܸܡܚܸܐ ܩܪܵܒ݂ܵܐ: ܥܲܡ ܒܲܪܒܪ̈ܵܝܹܐ: ܕܐܲܩܪܸܒ݂ܘ ܥܲܡܹܗ

܀

 ܘܲܥ̣ܩܲܪ ܘܚܲܒܸܠ: ܥܠܵܘ̈ܵܬ݂ܵܐ ܕܛܘܼܥܝܲܝ. ܘܲܗ̣ܦܲܟ݂ ܘܐܸܬ݂ܵܐ: ܒܙܵܟ݂ܘܼܬ݂ܵܐ ܬܡܝܼܗܬܵܐ

܀

 ܐܘܿ ܡܲܠܟܵܐ ܙܲܟܵܝܵܐ. ܥܵܩܘܿܪܵܐ ܕܚܲܢܦܘܼܬ݂ܵܐ. ܒܪܝܼܟ݂ܘܼ ܕܚܲܝܠܵܟ݂ ܕܬܸܬ݂ܟܲܫܲܪ ܚܠܵܦ ܥܹܕ݁ܬܹܗ. ܕܐܲܒ݂ܘܼܢ܀ ܪܚܸܡ̣ܬ݁ ܙܲܕܝܼܩܘܼܬ݂ܵܐ ܘܲܣ̣ܢܲܝܬ݁ ܥܵܘ̣ܠܵܐ. ܐܲܒ݂ܘܼܢ ܩܲܕܝܼܫܵܐ: ܗܲܝܡܵܢܘܼܬ݂ܵܐ: ܦܲܛܪܘܿܣܵܝܬܵܐ: ܒܵܟ݂ ܐܸܫܬܲܪܪܲܬ̤. ܘܙܲܕܝܼܩܘܼܬ݂ܵܐ: ܗ̇ܝ ܕܒܲܥ̈ܒ݂ܵܕܹܐ: ܒܵܟ݂ ܐܸܬ݂ܚܲܙܝܲܬ̤. ܘܲܒ݂ܬܲܖ̈ܬܲܝܗܹܝܢ: ܗ̤ܘܲܝܬ݁ ܢܲܨܝܼܚܵܐ: ܒܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ

܀

 ܪܲܗܛܵܟ݂ ܫܲܡ̣ܠܝܼܬ݁: ܘܐܵܓ݂ܘܿܢܵܟ݂ ܐܸܬ݂ܓܡܲܪ. ܟܠܝܼܠܵܟ݂ ܨܵܡܲܚ̇: ܥܲܡ ܕܦܵܘܠܘܿܣ ܒܙܲܗܝܘܼܬ݂ܵܐ. ܘܥܹܕ݁ܬܵܐ ܡܙܲܝܚܵܐ: ܠܫܘܼܟ݂ܠܵܠܹܗ ܕܐܵܓ݂ܘܿܢܵܟ݂

܀

 ܐܘܿܐܲܒ݂ܘܼܢ ܩܲܕܝܼܫܵܐ. ܐܘܿ ܪܹܫܵܐ ܕܚܲܣܝܘܼܬ݂ܵܐ. ܨܠܘܿܬ݂ܵܟ݂ ܬܸܗܘܸܐ ܒܹܝܬ݂ ܓܵܘܣܵܐ ܠܢܲܦ̮̈ܫܵܬܲܢ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܡܗܲܝܡܲܢܘܼ ܡܵܪܝܵܐ ܒܡܸܠܵܘ̈ܗܝ. ܐܲܝܟ݂ ܕܐܸܫܬܵܘܕܝܼܬ݁: ܡܵܪܲܢ ܝܼܫܘܿܥ: ܐܲܫܘܵܢ ܐܸܢܘܿܢ: ܠܐܲܝܠܹܝܢ ܕܐܲܚܒ݂ܘܼܟ݂. ܘܥܲܠ ܬܘܼܟ݂ܠܵܢܵܐ: ܕܗܲܝܡܵܢܘܼܬ݂ܵܟ݂: ܥ̣ܢܲܕ݂ܘ ܡ݂ܢ ܥܵܠܡܵܐ. ܘܫܵܘܬܵܦܘܼܬ݂ܵܐ: ܕܦܲܓ݂ܪܵܟ݂ ܘܲܕ݂ܡܵܟ݂: ܬܸܛܲܪ ܐܸܢܘܿܢ

܀

܀

 ܘܢܸܬ݂ܚܲܣܘܿܢ ܡ݂ܢ ܚܵܘܒܲܝ̈ܗܘܿܢ. ܘܲܢܩܘܼܡܘܼܢ ܡ݂ܢ ܝܲܡܝܼܢܵܟ݂. ܘܥܲܡ ܩܲܕܝܼܫܲܝ̈ܟ ܢܸܬ݂ܒܲܣܡܘܼܢ ܒܡܲܠܟܘܼܬ݂ܵܟ݂. ܠܵܐ ܬܸܡܢܹܝܢܝ ܥܲܡ ܪ̈ܲܫܝܼܥܹܐ ܘܥܲܡ ܥܵܒ݂ܕܲܝ̈ ܥܵܘ̣ܠܵܐ. ܒܝܵܘܡܵܐ ܕܡܸܬ݂ܓܠܹܝܬ݁: ܝܼܫܘܿܥ ܡܲܠܟܲܢ: ܘܦܵܪܥܲܬ݁ ܠܟ݂ܠܢܵܫ: ܐܲܓ݂ܪܵܐ ܕܥܲܡܠܵܘ̈ܗܝ. ܕܵܢܚܝܼܢ ܟܹܐܢܹ̈ܐ ܘܢܵܣܒܝܼܢ: ܐܲܓ݂ܪܵܐ ܕܕ݂ܘܼܒܵܪ̈ܲܝܗܘܿܢ

 ܛܵܒܹ̈ܐ ܠܪܵܘܡܵܐ ܡܸܬ݂ܥܲܠܹܝܢ ܒܬܸܫܒܘܿܚܬܵܐ. ܘܒ݂ܝܼܫܹ̈ܐ ܦܵܝܫܝܼܢ ܒܓ݂ܵܘ ܓܹܗܲܢܵܐ ܕܠܵܐ ܕܵܥܟܵܐ

 ܡܵܪܲܢ ܣܚܘܿܦ ܠܥܵܘܵܠܹ̈ܐ: ܘܐܲܒ݂ܗܸܬ݂ ܐܸܢܘܿܢ ܠܪ̈ܲܫܝܼܥܹܐ ܕܢܹܐܙܠ݇ܘܼܢ ܠܚܸܫܟܵܐ ܘܬܲܫܢܝܼܩܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܡܲܠܟܵܐ ܡܫܝܼܚܵܐ ܡܚܲܪܲܪ ܟܠ. ܐܝܼܬ݂ܝܵܐ ܓܢܝܼܙܵܐ ܡܢܲܚܸܡ ܟܠ ܒܛܲܝܒܘܼܬܹܗ

܀

 ܡܸܢ ܦܘܼܡܵܐ ܕܲܥܠܲܝܡܹ̈ܐ ܘܕܲܛ̈ܠܵܝܹܐ. ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ: ܕܲܒ݂ܝܲܕ݂ ܩܵܠܵܟ݂: ܡܝܼ̈ܬܹܐ ܩܵܝ̇ܡܝܼܢ: ܡܸܢ ܓܵܘ ܩܲܒ݂ܖܹ̈ܐ. ܘܡܲܣܩܝܼܢ ܫܘܼܒ݂ܚܵܐ: ܠܵܟ݂ ܥܲܡ ܐܲܒ݂ܘܼܟ݂. ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܒܝܲܕ݂ ܩܵܠܲܝ̈ܗܘܿܢ: ܥܲܡ ܖ̈ܘܼܚܵܢܹܐ: ܒܝܘܿܡ ܢܘܼܚܵܡܵܐ

܀

 ܕܐܲܢ݇ܬ݁ܘܼ ܡܲܠܒܸ݁ܫ ܦܲܓ݂ܪ̈ܲܝܗܘܿܢ ܫܘܼܒ݂ܚܵܐ. ܐܲܝܠܹܝܢ ܕܗܲܝܡܸܢ̣ܘ ܘܐܵܘܕ݁ܝܼܘ ܒܵܟ݂ ܒܲܫܪܵܪܵܐ

܀

 ܐܵܦ ܠܲܢ ܡܵܪܝ ܐܲܫܘܵܢ ܠܲܢ. ܕܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒܚܵܟ݂. ܕܐܲܢ݇ܬ݁ܘܼ ܐܝܼܬܲܝܟ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠ ܒܸܖ̈ܝܵܢ. ܐܲܪܝܼܡ ܩܵܠܹܗ ܘܙܵܥܲܬ̤ ܐܲܪܥܵܐ. ܩ̣ܪܵܐ ܠܠܵܥܵܙܲܪ ܡܸܢ ܓܵܘ ܩܲܒ݂ܪܵܐ ܒܩܵܠܹܗ ܚܲܝܵܐ ܒܲܪ ܡܵܪܹܟܠ. ܘܬܲܒܲܪ ܬܲܖ̈ܥܹܐ ܘܡܘܼܟ݂ܠܹ̈ܐ ܕܲܫܝܘܿܠ ܘܲܢ̣ܦܲܩ ܠܐܘܼܪܥܹܗ. ܘܲܪܥܸܡ̣ ܡܵܘܬܵܐ ܕܲܚ̣ܙܵܐ ܕܐܸܫܬ݁ܪܝܼ ܫܸܡ ܫܘܼܠܛܵܢܹܗ

܀

 ܕܒܹܗ ܚܵܒ݂ ܐܵܕ݂ܵܡ ܒܦܲܪܕܲܝܣܵܐ ܕܲܥܕܸܢ. ܒܡܸܠܟܹ݁ܗ ܕܣܵܢܐܵܐ ܣܵܩܘܿܪܹܗ ܕܓܸܢܣܲܢ

܀

 ܐܲܢ݇ܬ݁ ܐܲܒ݂ܵܐ ܚܲܢܵܢܵܐ: ܒܵܪܘܿܝܵܐ ܡܪܲܚܡܵܢܵܐ. ܚܘܼܢܲܝܢ ܘܚܲܣܵܢ ܩܕ݂ܵܡ ܒܹܝܡܵܟ݂ ܕܚܝܼܠܬܵܐ

܀

ܕܥܲܢܝܼܕܹ̈ܐ. ܗܸܢ̣ܘܿܢ ܥܵܒ݂ܪܝܼܢ ܘܐܲܢ݇ܬ݁ ܩܲܝܵܡ ܐܲܢ݇ܬ݁. ܥܵܠܡܵܐ ܗܵܢܵܐ ܟܠܹܗ ܥܵܒ݂ܘܿܪܵܐ ܗ݇ܘ̣ ܘܒ݂ܵܛܹ̇ܠ ܚܘܼܬ݂ܪܹܗ. ܘܡܵܘܦܸܐ ܫܘܼܒ݂ܗܵܪܹܗ. ܘܗܵܘܹܝ̇ܢ ܕܠܵܐ ܝܘܼܬ݂ܪܵܢ ܟܠ ܕܪܲܚܡ̣ܘܼܗܝ ܘܐܲܚ̣ܒ݂ܘܼܗܝ ܘܟܲܢܸܫ̣ܘ ܩܸܢܝܵܢܵܘ̈ܗܝ. ܘܕ݂ܵܥ̇ܟ݁ܝܼܢ ܠܲܡܦܹܝܕܲܝ̈ܗܘܿܢ ܥܲܡ ܕܚܲܡܸܫ ܣܲܟ݂̈ܠܵܬ݂ܵܐ. ܘܒ݂ܵܟܹܝ̇ܢ ܘܡܸܬ݁ܬܲܢܚܝܼܢ ܡܵܐ ܕܚܵܙܹܝ̇ܢ ܠܲܡܪ̈ܲܚܡܵܢܹܐ ܕܦܵܪܚܝܼܢ ܒܐܵܐܲܪ. ܠܐܘܼܪܥܹܗ ܕܚܲܬ݂ܢܵܐ ܫܒ݂ܝܼܚܵܐ. ܕܢܸܥܠܘܼܢ ܥܲܡܹܗ ܠܗܵܝ̇ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܠܵܐ ܡܸܫܬܲܪܝܵܐ. ܗܵܐ ܠܡܝܼܬܹܐ ܥܵܒܹܕ݂ ܐܲܢ݇ܬ݁ ܬܸܕ݂ܡܪ̈ܵܬ݂ܵܐ. ܗܵܐ ܠܡܝܼ̈ܬܹܐ ܡ݂ܢ ܩܲܒ݂ܖܹ̈ܐ ܩܵܪܹܐ ܐܲܢ݇ܬ݁ ܡܵܪܝܵܐ. ܘܲܠܟ݂ܠܗܘܿܢ ܫܲܘܝܵܐܝܼܬ݂ ܡܢܲܚܸܡ ܐܲܢ݇ܬ݁ ܡ݂ܢ ܫܝܘܿܠ. ܐܸܠܵܐ ܡܦܝܼܣ ܐ݇ܢܵܐ ܠܵܟ݂ ܒܲܚܢܲܓ݂ܬ݂ܵܐ ܡܵܪܵܐ ܕܥܵܠܡܹ̈ܐ. ܕܡܵܐ ܕܡܸܬ݂ܓܲܠ̈ܝܵܢ ܟܲܣ̈ܝܵܬ݂ܵܐ ܕܲܩܢܘܿܡܝ. ܠܵܐ ܬܫܲܐܠܲܢܝ ܒܕ݂ܝܼܢܵܟ݂ ܟܹܐܢܵܐ. ܕܐܸܢܵܐ ܒܲܥܒ݂ܵܕܲܝ̈ ܒܝܼܫܹ̈ܐ ܟܠܝܘܿܡ ܐܲܪܓܸ݁ܙܬܵܟ݂. ܘܚܲܝܵܒ݂ ܐ݇ܢܵܐ ܩܕ݂ܵܡ ܟܹܐܢܘܼܬ݂ܵܟ݂. ܘܠܲܝܬ݁ ܠܝܼ ܓܲܠܝܘܼܬ݂ܵܐ ܐܲܦܹ̈ܐ. ܡܵܪܵܐ ܛܵܒ݂ܵܐ ܐܲܠܵܗܵܐ ܚܘܼܢܲܝܢܝ ܒܪ̈ܲܚܡܲܝܟ ܘܚܲܣܵܐ ܠܝܼ ܚܵܘܒܲܝ̈ ܘܪܲܚܸܡ ܥܠܲܝ.
ܠܐܲܝܟܵܐ ܐܹܙܲܠ ܡ݂ܢ ܪܘܼܚܵܟ݂. ܒܫܵܥܬ݂ܵܐ ܡܵܐ ܕܡ݂ܢ ܓܵܘ ܫܸܠܝܵܐ ܒܕܸܚܠܵܐ ܪܲܒܵܐ ܙܵܥܩܵܐ ܩܲܪܢܵܐ ܕܡܹܐܬ݂ܝܼܬ݂ܵܟ݂. ܡ݂ܢ ܫܡܲܝܵܐ ܛܵܝܣܝܼܢ ܠܡܸܚܲܬ݂ ܒܣܘܼܪܗܵܒ݂ܵܐ. ܬܸܓ݂ܡܹ̈ܐ ܬܡܝܼܗܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ. ܘܡܵܐ ܕܡ݂ܢ ܩܲܒ݂ܖܹ̈ܐ ܩܵܝ̇ܡܝܼܢ ܟܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܒܲܪܬܹܝܬ݂ܵܐ ܠܡܸܫܬܲܐܵܠܘܼ. ܪܵܥ̇ܠܝܼܢ ܚܲܝ̈ܠܵܘܵܬ݂ܵܐ ܥܸܠܵܝܹ̈ܐ. ܡ݂ܢ ܥܘܼܙܹܗ ܕܕ݂ܝܼܢܵܐ ܕܗܵܘܹ̇ܐ ܒܬܲܚܬܵܝܹ̈ܐ. ܘܡܵܐ ܕܲܟ݂ܪ̈ܘܿܒܹܐ ܛܥܝܼܢܝܼܢ ܡܙܲܝܚܝܼܢ ܠܵܟ݂ ܕܲܝܵܢܵܐ ܟܹܐܢܵܐ. ܐܸܠܵܐ ܒܗܵܘ̇ ܕܝܼܢܵܐ ܕܚܝܼܠܵܐ ܡܵܐ ܕܡܸܬ݂ܦܪܲܥ ܥܒ݂ܵܕ݂ܵܘ̈ܗܝ ܕܟ݂ܠܢܵܫ. ܪܲܚܸܡ ܥܠܲܝ ܐܲܢ݇ܬ݁ ܪܵܚܹܡ ܐ݇ܢܵܫܵܐ.
ܘܐܲܝܟܵܐ ܐܸܛܫܸܐ ܡ݂ܢ ܩܕ݂ܵܡܲܝܟ. ܒܫܵܥܬ݂ܵܐ ܗܵܝ̇ ܕܚܲܝ̈ܠܵܘܵܬ݂ܵܐ ܕܥܝܼܖܹ̈ܐ ܟܪܝܼܟ݂ܝܼܢ ܡܙܲܝܚܝܼܢ ܠܪܲܒ݁ܘܼܬ݂ܵܟ݂. ܘܡܵܐ ܕܡܲܙܥܩܝܼܢ ܘܩܵܥܹܝ̇ܢ ܟܠܗܘܿܢ ܒܩܵܠܵܐ ܕܩܲܪܢܵܐ ܘܫܝܼܦܘܿܪܵܐ. ܡܵܚܹܝ̇ܢ ܠܵܗ̇ ܚܸܒ݂̈ܠܹܐ ܩܫܲܝ̈ܵܐ ܠܐܲܪܥܵܐ ܕܬ݂ܵܘܠܸܕ݂ ܥܲܢܝܼ̈ܕܹܐ. ܘܩܵܝ̇ܡܝܼܢ ܝܲܠܕܵܘ̈ܗܝ ܕܐܵܕ݂ܵܡ ܥܲܦܪܵܢܵܐ. ܢܵܦܩ̇ܝܼܢ ܠܐܘܼܪܥܵܟ݂ ܡܵܪܲܢ ܒܐܘܿܫܲܥܢܹ̈ܐ. ܘܡܸܬ݂ܥܲܠܹܝܢ ܛܵܒܹ̈ܐ ܠܪܵܘܡܵܐ. ܘܒ݂ܝܼܫܹ̈ܐ ܦܵܝ̇ܫܝܼܢ ܥܲܠ ܐܲܪܥܵܐ. ܚܘܼܢܲܝܢܝ ܒܗܵܘ̇ ܝܵܘܡܵܟ݂ ܪܲܒܵܐ. ܡܵܐ ܕܐܵܬܹܝܬ݁ ܠܲܡܕ݂ܵܢ ܡܵܪܲܢ. ܘܪܲܚܸܡ ܥܠܲܝ ܐܲܢ݇ܬ݁ ܪܵܚܹܡ ܐ݇ܢܵܫܵܐ.
ܫܲܒܲܚ. ܫܠܝܼܚܹ̈ܐ ܩܲܕ݁ܝܼܫܹ̈ܐ ܕܚܲܫ̣ܘ ܒܲܙܩܝܼܦܵܐ ܡܸܛܠ ܚܘܼܒܹ݁ܗ ܕܡܲܠܟܵܐ ܡܫܝܼܚܵܐ. ܘܙܲܩܦ̣ܘܼܗܝ ܠܥܵܠܡܵܐ ܘܠܲܪ̈ܓ݂ܝܼܓ݂ܵܬܹܗ ܒܗܲܕܵܡܲܝ̈ܗܘܿܢ. ܘܲܫܩܲܠ̣ܘ ܝܘܼܩܪܹܗ ܕܝܵܘܡܵܐ ܘܚܘܼܡܹܗ ܐܲܝܟ݂ ܟܲܫܝܼܖܹ̈ܐ

ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ̣ܵܐ: ܨܠܘܿܬ݂ܗܘܿܢ ܡܵܪܝ ܕܲܫܠܝܼܚܹ̈ܐ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܘܒ݂ܵܥܘܼܬ݂ܗܘܿܢ ܕܟ݂ܵܪ̈ܘܿܙܹܐ ܫܲܪܝܼܖܹ̈ܐ: ܘܬܲܟ݂ܫܲܦ̮ܬ݁ܗܘܿܢ ܘܬܲܚܢܲܢܬ݁ܗܘܿܢ ܕܐܲܬ݂ܠܹܝܛܹ̈ܐ ܢܲܨܝܼܚܹ̈ܐ ܡܣܲܒ݁ܪ̈ܵܢܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܘܙܵܪ̈ܥܲܝ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ: ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܘܡܲܩܸܦ: ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܵܘܡܹܐ ܥܸܠܵܝܹ̈ܐ : ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ݂܀ ܗܵܘܿ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ̇ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡܠܲܟܼ ܛܵܒ݂ܵܬ݂ܵܐ ܠܪܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕ݁ܬܹܗ: ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܠܵܐ ܨܵܒܹܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ:  ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܼܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܗܘ̤ܝܘ̣ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܪ̈ܝܵܬ݂ܵܐ:  ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ ؛ ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܝܼܣ:  ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܝܼܣ : ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈  ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܵܘܿ ܕܲܒ݂ܪܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚܘ ܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂:  ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.

	ܨܠܘܿܬ݂ܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܵܐ: ܫܘܼܒ݂ܚܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ: ܘܗܘܼܠܵܠܹ̈ܐ ܕܠܵܐ ܫܲܠܘܵܐ: ܘܬܸܫܒ̈ܚܵܬ݂ܵܐ ܕܠܵܐ ܣܵܟ݂ܵܐ: ܘܩܵܠܹ̈ܐ ܕܲܡܠܹܝܢ ܬܵܘܕ݁ܝܼܬ݂ܵܐ: ܚܲܝܵܒ݂ ܟܠܹܗ ܟܝܵܢܵܐ ܕܲܡܠܝܼ̈ܠܹܐ ܕܲܒ݂ܪܲܝܬ݁: ܠܡܲܣܵܩܘܼ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ. ܡ̣ܢ ܦܘܼܡܹ̈ܐ ܡܵܪܝ ܡܠܝܼܠܹ̈ܐ ܕܲܒ݂ܪܲܝܬ݁: ܘܡ̣ܢ ܠܸܫܵܢܹ̈ܐ ܡܫܲܒ݁ܚܵܢܹ̈ܐ ܕܪܲܟܸ݁ܒ݂ܬ݁: ܘܡ̣ܢ ܟܸܢܫܹ̈ܐ ܟܠܗܘܿܢ ܕܥܸܠܵܝܹ̈ܐ ܘܲܕ݂ܬܲܚܬܵܝܹ̈ܐ: ܫܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܲܕ݂ܡܵܪܘܼܬ݂ܵܟ݂ ܢܸܣܬ݁ܓܸܕ݂ ܘܢܸܫܬܲܒܲܚ ܘܢܸܬ݂ܝܲܩܲܪ ܘܢܸܬ݂ܪܲܡܪܲܡ ܘܢܸܬ݁ܬܵܘܕܸ݁ܐ ܘܢܸܬ݂ܒܲܪܲܟ݂ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܪܝܼܢ ܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ ܘܬܵܢܹܝܢ ܘܲܡܫܲܒܚܝܼܢ ܒܵܗ̇ ܘܡܲܩܦܝܼܢ:
ܕܠܸܠܝܵܐ. ܒܩܵܠܵܐ ܪܲܒܵܐ. ܪܵܥܝܹܗ ܕܝܼܣܪܵܝܹܠ ܨܘܼܬ݂: ܪܥܝܼ ܐܸܢܘܿܢ ܘܕܲܒܲܪ ܐܸܢܘܿܢ ܥܕܲܡܵܐ ܠܥܵܠܲܡ. ܕܨܘܼܪܬܵܐ. ܪܵܥܝܵܐ ܛܵܒ݂ܵܐ ܢܲܦ̮ܫܹܗ ܣܵܐܹ̇ܡ ܚܠܵܦ ܥܵܢܹ̈ܐ. ܘܬܸܗܘܸܐ ܥܵܢ̈ܵܐ ܟܠܵܗ̇ ܚܕ݂ܵܐ ܘܚܲܕ݂ ܪܵܥܝܵܐ. ܪܵܥܝܵܐ ܫܡܲܝܵܢܵܐ ܕܐܸܬ݂ܪ̱ܟܸܢ ܦܪܲܩ̣ ܥܵܢܹ̈ܐ. ܕܲܚ̣ܙܵܗ̇ ܕܐܸܬ݂ܒܲܕܪܲܬ̤ ܒܛܘܼܥܝܲܝ ܕܚܲܢܦܘܼܬ݂ܵܐ. ܘܲܓ݂ܒ݂ܵܐ ܡܸܢܹܗ ܕܛܘܼܗܡܲܢ ܬܪܸܥܣܲܪ ܦܵܥܠܹ̈ܐ ܟܲܫܝܼܖܹ̈ܐ. ܘܲܡܠ̣ܵܐ ܐܸܢܘܿܢ ܚܸܟ݂ܡܲܬ݂ ܚܲܝܠܵܐ ܕܫ̣ܲܕܲܪ ܡ̣ܢ ܪܵܘܡܵܐ. ܘܐܲܠܸܦ̣ ܐܸܢܘܿܢ ܠܡܸܣܥܲܪ ܚܲܝܠܹ̈ܐ ܬܡܝܼܗܹ̈ܐ ܘܲܬ݂ܗܝܼܖܹ̈ܐ ܘܟܲܕ݂ ܥܵܩ̈ܵܬ݂ܵܐ ܣܲܝܒܲܪܘ ܗ݇ܘ̣ܵܘ ܡ̣ܢ ܖ̈ܵܕ݂ܘܿܦܵܘܗܝ ܕܲܫܪܵܪܵܐ. ܠܵܐ ܐܲܗܡ̣ܝܼܘ ܠܲܡܟܲܢܵܫܘܼ ܠܥܵܢ̈ܵܐ ܩܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܐ ܕܲܟ݂ܝܵܐ. ܕܲܪܘܹܝܢ ܗ݇ܘ̣ܵܘ ܒܚܲܡܪܵܐ ܕܲܥܨܝܼܪ ܒܝܲܕ݂ ܪܘܼܡܚܵܐ. ܕܗ̤ܘ ܐܝܼܬ݂ܵܘܗܝ ܕܡܵܐ ܝܲܩܝܼܪܵܐ. ܕܒܹܗ ܐܸܬ݂ܚܲܣ̈ܝܼ ܚܵܘ̈ܒܵܬܲܢ ܘܐܸܬ݂ܐܲܣܝܼܘ ܟܹܐܒܲܝ̈ܢ. ܐܘܿ ܩܛܝܼܠܹ̈ܐ ܕܚܲܒ݂ܫ̣ܘܼܗ̇ ܠܥܵܡܲܪܬܵܐ ܒܓ݂ܵܘ ܫܘܼܪܵܐ ܚܲܣܝܼܢܵܐ ܕܗܲܝܡܵܢܘܼܬ݂ܵܐ. ܕܠܵܐ ܬܹܐܒܲܕ݂ ܒܢܸܟ݂ܠܹܗ ܕܡܵܪܘܿܕ݂ܵܐ: ܬܢܝܼ. ܘܫܲܒܲܚ. ܒܵܗ̇.
ܥܵܠܲܡ. ܡܫܝܼܚܵܐ ܠܵܐ ܬܲܗܡܸܐ ܡܸܢܲܢ. ܘܠܵܐ ܬܲܪܚܸܩ ܡ݂ܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܒ݂ܵܟ݂ܘܼ ܡܵܪܝ ܐܸܬ݂ܓܵܘܲܣܢܲܢ. ܕܲܒܲܪܲܝܢ ܒܐܘܼܪܚܵܟ݂ ܕܚܲܝܹ̈ܐ ܕܲܢܙܲܡܲܪ ܠܵܟ݂ ܟܠܲܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ. ܢܹܐܡܲܪ. ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܐܸܡܹܗ ܕܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ: ܐܲܦܝܼܣܝ ܥܲܡܲܢ ܠܲܡܫܝܼܚܵܐ ܕܢܲܫܪܸܐ ܫܲܝܢܹܗ ܒܲܝܢܵܬܲܢ: ܘܲܢܢܲܛܲܪ ܠܲܢ ܡ̣ܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ.

܀

 ܒܗܵܠܹܝܢ ܡܸܠܹ̈ܐ ܦܲܠ̣ܚܘܼܗ̇ ܠܐܲܪܥܵܐ ܡܲܠܝܲܬ݂ ܟܘܼܒܹ̈ܐ: ܘܐܵܘܥ̣ܝܼ ܥܘܼܒܵܗ̇ ܦܹܐܖܹ̈ܐ ܕܫܘܼܒ݂ܚܵܐ ܠܲܡܬܲܩܢܵܢܵܐ

܀

ܬܸܫܒܘܿܚܬܵܐ. ܘܲܒ݂ܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܕܹܝܢ ܐܵܡ̇ܪܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܗܵܕܹܐ ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܡܵܪܝ ܢܲܪܣܲܝ ܡܲܠܦܵܢܵܐ.
ܬܵܘܕ݁ܝܼ ܠܛܵܒ݂ܵܐ ܕܚܲܪܲܪ ܓܸܢܣܲܢ: ܡ̣ܢ ܥܲܒ݂ܕ݁ܘܼܬ݂ܵܐ ܕܒ݂ܝܼܫܵܐ ܘܡܵܘܬܵܐ܀ ܫܲܝܸܢ ܥܲܡܲܢ ܟܸܢܫܲܝ̈ ܪܵܘܡܵܐ: ܕܪܲܓ݁ܝܼܙܝܼܢ ܗ݇ܘ̣ܵܘ ܡܸܛܠ ܥܵܘ̣ܠܲܢ܀ ܒܪܝܼܟ݂ ܚܲܢܵܢܵܐ ܕܟܲܕ݂ ܠܵܐ ܒܥܲܝܢ̣ܵܝܗܝ: ܢ̣ܦܲܩ ܒܲܒ݂ܥܵܬܲܢ ܘܲܚ̣ܕ݂ܝܼ ܒܚܲܝܲܝ̈ܢ܀ ܘܨ̣ܵܪ ܕܸܡ̈ܘܵܬ݂ܵܐ ܕܐܲܒ݁ܝܼܕ݂ܘܼܬܲܢ: ܘܲܕ݂ܦܘܼܢܵܝܲܢ ܒܥܸܪܒܵܐ ܕܲܛ̣ܥܵܐ܀ ܝܵܪܬܵܐ ܘܲܒ݂ܪܵܐ ܩ̣ܪܵܝܗܝ ܠܲܟ݂ܝܵܢܲܢ: ܕܲܛ̣ܥܵܐ ܘܲܦܢ̣ܵܐ ܘܡ̣ܝܼܬ݂ ܘܐܸܬ݂ܢܲܚܲܡ܀ ܘܚܲܕ݁ܝܼ ܠܟܸܢܫܹ̈ܐ ܖ̈ܘܼܚܵܢܵܝܹܐ: ܒܲܬ݂ܝܵܒ݂ܘܼܬܲܢ ܘܲܒ݂ܢܘܼܚܵܡܲܢ܀ ܠܵܐ ܡܸܬ݂ܡܲܠܲܠ ܚܘܼܒܵܐ ܪܲܒܵܐ: ܕܚ̣ܵܘܝܼ ܨܹܐܕܲܝܢ ܪܵܚܹ̇ܡ ܓܸܢܣܲܢ܀ ܕܡܸܢܹܗ ܕܓܸܢܣܲܢ ܥ̣ܒܲܕ݂ ܡܸܨܥܵܝܵܐ: ܘܪܲܥ̣ܝܼ ܠܥܵܠܡܵܐ ܥܲܡ ܪܲܒܘܼܬܹܗ܀ ܪܲܒܵܐ ܗ݇ܝ̣ ܡܸܢܲܢ ܘܡ̣ܢ ܟܠ ܒܸܖ̈ܝܵܢ: ܚܕܲܬܵܐ ܕܲܣ̣ܥܲܪ ܨܹܝܕ݂ ܐ݇ܢܵܫܘܼܬܲܢ܀ ܕܥܲܒ݂ܕܹ݁ܗ ܠܦܲܓ݂ܪܲܢ ܗܲܝܟܲܠ ܩܘܼܕ݂ܫܵܐ: ܕܲܢܡܲܠܸܐ ܒܹܗ ܣܸܓ݂ܕܲܬ݂ ܟܠܵܐ܀ ܬܵܘ ܐܲܖ̈ܥܵܢܹܐ ܘܲܫܡܲܝܵܢܹ̈ܐ: ܬܗܲܪܘ ܘܐܸܬ݁ܕܲܡܲܪܘ ܒܪܲܒ݁ܘܼܬ݂ ܕܲܪܓ݂ܵܐ܀ ܕܲܡ̣ܛܵܐ ܓܸܢܣܲܢ ܠܪܵܘܡܵܐ ܪܲܒܵܐ: ܕܐܲܠܵܗܘܼܬ݂ܵܐ ܕܠܵܐ ܡܸܬ݁ܕܲܪܟܵܐ܀ ܫܡܲܝܵܐ ܘܐܲܪܥܵܐ ܘܟ݂ܠܡܵܐ ܕܲܒ݂ܗܘܿܢ: ܢܵܘܕ݁ܘܿܢ ܥܲܡܲܢ ܠܡܵܘܪܸܒ݂ ܓܸܢܣܲܢ܀ ܕܚܲܕܸ݁ܬ݂ ܨܲܠܡܲܢ ܘܲܥ̣ܛܵܐ ܥܵܘ̣ܠܲܢ: ܘܲܩ̣ܪܵܢ ܒܲܫܡܹܗ ܘܫܲܥܒܸ݁ܕ݂ ܠܲܢ ܟܠ܀ ܫܵܘܹ̇ܐ ܠܫܘܼܒ݂ܚܵܐ ܡ̣ܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܗ̇ܘ ܕܐܲܪܝܼܡܲܢ ܠܥܸܠ ܡ̣ܢ ܟܠܵܐ܀ ܘܢܹܐܡܲܪ ܟܠܲܢ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ: ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀
ܘܩܵܝܡ̇ܝܼܢ: ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܐܲܝܟ݂ ܕܲܒ݂ܚܘܼܕ݂ܪܵܐ ܐܵܘ ܒܓܲܙܵܐ. ܐܵܘ ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ. ܕܠܵܐ ܚܲܒ݂ܪ̈ܵܬ݂ܵܗ̇.
(ܘܪܵܡܹܐ ܡܫܲܡܫܵܢܵܐ ܐܘܼܪܵܪܵܐ: ܘܐܵܣܲܪ ܙܘܼܢܵܪܵܐ: ܘܩܵܐܹ̇ܡ ܩܕ݂ܵܡ ܫܟ݂ܝܼܢܬܵܐ ܡܛܲܟ݁ܣܵܐܝܼܬ݂) ܘܐܵܡܲܪ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ: ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ ܘܐܲܠܵܗܵܐ ܕܟ݂ܠܹܗ ܒܘܼܝܵܐܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܦܵܪܘܿܩܲܢ ܘܝܲܨܘܼܦܬ݂ܵܢܲܢ ܘܲܡܦܲܪܢ̱ܣܵܢܵܐ ܕܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ ܘܩܘܼܝܵܡܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ ܘܲܕ݂ܟ݂ܠܗܹܝܢ ܥܹ̈ܕ݂ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܐܲܬ݂ܪܲܢ ܘܥܲܠ ܟܠ ܐܲܬ݂ܪ̈ܵܘܵܢ: ܘܥܲܠ ܐܲܝܠܹܝܢ ܕܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܥܵܡ̇ܪܝܼܢ ܒܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܘܼܙܵܓ݂ܵܘ̈ܗܝ ܕܐܵܐܲܪ ܘܟܲܗܝܼܢܘܼܬ݂ܵܐ ܕܫܲܢ݇ܬܵܐ ܘܥܲܠ̈ܠܵܬ݂ܵܐ ܕܦܹܐܪܹ̈ܐ ܘܥܲܠ ܬܘܼܩܵܢܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ1: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܲܠܟܹ̈ܐ ܐܲܚܝܼܕܲܝ̈ ܫܘܼܠܛܵܢܹܗ ܕܥܵܠܡܵܐ ܗܵܢܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܩܲܫܝܼܫܹ̈ܐ ܘܲܡܫܲܡܫܵܢܹ̈ܐ ܐܲܪ̈ܬܵܕ݁ܘܿܟ݁ܣܹܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ܗ̇ܘ ܕܥܲܬ݁ܝܼܪ ܒܪ̈ܲܚܡܵܘܗܝ ܘܲܫܦܝܼܥ ܒܲܚܢܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܡܸܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ ܐܝܼܬ݂ܵܘܗܝ ܘܲܠܕ݂ܵܪܕܵܪ̈ܝܼܢ ܩܲܝܵܡ ܫܘܼܠܛܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܟ݂ܝܵܢܹܗ ܐܝܼܬ݂ܵܘܗܝ ܛܵܒ݂ܵܐ ܘܝܵܗܘܿܒ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹܐ ܒܡܵܘܬܹ݁ܗ ܕܚܲܛܵܝܵܐ ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܸܢ ܥܵܘ̣ܠܹܗ ܘܢܸܚܸܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܲܒ݂ܝܲܠܕܹ݁ܗ ܐܵܘ ܕܲܒ݂ܕܸܢܚܹܗ ܐܵܘ ܕܲܒ݂ܨܵܘܡܹܗ ܐܵܘ ܕܲܒ݂ܡܲܥܲܠܬܹ݁ܗ ܐܵܘ ܕܒܲܩܝܵܡܬܹ݁ܗ ܐܵܘ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܐܵܘ ܕܲܒ݂ܡܲܚܲܬ݁ܬܹ݁ܗ ܐܵܘ ܕܒܲܨܠܝܼܒܹܗ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܐܲܦܨܲܚ ܠܲܫܡܲܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܹܗ ܗܘ̤ ܟܝܵܢܵܐܝܼܬ݂ ܐܝܼܬ݂ ܠܵܐ ܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܘܝܲܬ݁ܝܼܪܵܐܝܼܬ݂ ܕܐܲܝܠܹܝܢ ܕܲܡܗܲܝܡܢܝܼܢ ܒܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܣܓܵܐ ܥܲܡܲܢ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
