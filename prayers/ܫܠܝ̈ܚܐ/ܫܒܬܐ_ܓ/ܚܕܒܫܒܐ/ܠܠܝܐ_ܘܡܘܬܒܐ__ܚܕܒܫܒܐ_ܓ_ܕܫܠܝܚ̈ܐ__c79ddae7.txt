ܛܲܟ݂ܣܵܐ ܕܬܸܫܡܸܫܬܵܐ
ܕܠܸܠܝܵܐ ܘܕܨܲܦܪܵܐ
 ܒܫܸܡ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕܫܵܐ ܠܥܵܠܡܝܼܢ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܪ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܪܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܩܘܼܡ ܠܲܨܠܘܿܬ݂ܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܢܩܘܼܡ ܡܵܪܝ ܒܚܲܝܠܵܟ݂: ܘܢܸܫܬܲܪܲܪ ܒܣܲܒ݂ܪܵܟ݂: ܘܢܸܬ݁ܬ݁ܪܝܼܡ ܘܢܸܬ݂ܥܲܫܲܢ ܒܲܕ݂ܪܵܥܵܐ ܪܵܡܵܐ ܕܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂: ܘܢܸܫܬ݁ܘܸܐ ܒܥܘܼܕ݂ܪܵܢ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܠܡܲܣܵܩܘܼ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܪ̈ܲܚܡܲܝܟ: ܠܥܘܼܡܪܵܟ݂ ܡܫܲܒܚܵܐ. ܕܢܸܚܙܸܐ ܟܢܘܼܫܝܲܢ ܕܒܲܫܡܲܝܵܐ: ܘܢܸܙܡܲܪ ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ. ܦܬܲܚ ܠܝܼ ܬܲܪ̈ܥܹܐ ܕܙܲܕܝܼܩܘܼܬ݂ܵܐ. ܒܬܲܪܥܵܟ݂ ܡܵܪܲܢ ܨܒܹܝܬ݂ ܕܐܸܩܘܿܫ: ܚܲܢܵܢܵܐ ܥܢܝܼܢܝ. ܘܗܲܒ݂ܠܝܼ ܕܐܸܥܘܿܠ ܘܐܸܬ݂ܩܲܒܲܠ: ܒܪ̈ܲܚܡܹܐ ܕܫܲܠܚܘܼܟ݂. ܕܐܲܢ݇ܬ݁ܘܼ ܐܝܼܬܲܝܟ ܠܡܹܐܢ ܚܲܝܹ̈ܐ: ܕܟ݂ܠܗܘܿܢ ܡܛܲܪ̈ܦܹܐ. ܘܐܲܢ݇ܬ݁ܘܼ ܡܙܲܕܸܩ ܐܲܢ݇ܬ݁ ܡܲܓܵܢ: ܠܐܲܝܠܹܝܢ ܕܩܵܪܹܝܢ ܠܵܟ݂. ܙܲܕܝܼܩܲܬ݁ ܡܵܪܝܵܐ ܘܛܵܒ݂ ܬܪܝܼܨܝܼܢ ܕܝܼܢܲܝ̈ܟ. ܗܵܘ ܕܲܡܙܲܕܸܩ ܡܲܓܵܢ ܠܚܲܛܵܝܹ̈ܐ: ܐܲܢ݇ܬ݁ ܐܝܼܬܲܝܟ ܡܵܪܝ. ܦܬܲܚ ܐܵܦ ܠܝܼ ܬܲܪܥܵܟ݂ ܕܐܸܥܘܿܠ: ܘܐܸܬ݂ܩܲܒܲܠ. ܐܲܝܟ݂ ܕܐܸܫܬܵܘܕܝܼܬ݁ ܒܲܣܒܲܪܬ݂ܵܟ݂: ܠܵܐ ܬܸܓ݂ܠܙܲܢܝ. ܒܝܼܫܵܐ ܓܹܝܪ ܒܲܢܟ݂ܝܼܠܘܼܬܹܗ: ܡܚܵܢܝ ܒܫܲܒ݂ܛܹܗ ܘܐܲܪܥܠܲܢܝ. ܒܓ݂ܵܢܲܝ̈ܟ ܡܫܝܼܚܵܐ ܩܵܪܹܐ ܐ݇ܢܵܐ: ܣܚܘܿܦܲܝܗܝ ܠܲܒ݂ܥܸܠܕܒ݂ܵܒ݂ܝ. ܐܲܠܵܗܵܐ ܦܲܨܵܢܝ ܡܵܪܝܵܐ ܠܥܘܼܕ݂ܪܵܢܝ ܟܲܬܲܪ. ܦܲܨܵܢܝ ܡܵܪܝ ܡܢ ܝܲܕ݂ ܒܝܼܫܵܐ: ܕܲܫܒ݂ܵܢܝ ܡܸܢܵܟ݂. ܒܝܲܕ݂ ܦܲܚܹ̈ܐ ܕܲܪ̈ܓ݂ܝܼܓ݂ܵܬ݂ܵܐ: ܨܵܕܹܗ ܠܪܸܥܝܵܢܝ. ܘܠܵܐ ܡܨܸܐ ܐ݇ܢܵܐ ܕܐܸܙܟܹܝܘܗܝ ܠܝܲܨܪܹܗ. ܐܸܢ ܐܲܢ݇ܬ݁ ܠܵܐ ܬܥܲܕܪܲܢܝ. ܐܝܼܕ݂ܵܐ ܕܪ̈ܲܚܡܹܐ ܗܲܒ݂ܠܝܼ ܡܵܪܝ: ܕܠܵܐ ܢܸܙܟܹܝܢܝ ܒܥܸܠܕܒ݂ܵܒ݂ܝ. ܬܵܘܫܸܛ ܐܝܼܕ݂ܵܟ݂ ܘܬܸܦܪܩܲܢܝ. ܕܲܠܵܢܝ ܡܵܪܝ ܡܢ ܓܵܘ ܝܲܡܵܐ: ܕܚܵܘܒܹ̈ܐ ܕܣܸܥܪܹܬ݂. ܕܗܵܐ ܒܥܸܠܕܒ݂ܵܒ݂ܝ ܫܵܗܲܪ ܟܠܝܘܿܡ: ܕܡܢ ܐܝܼܕܲܝ̈ܟ ܢܸܚܛܦܲܢܝ. ܣܚܘܿܦܲܝܗܝ ܘܒܲܛܸܠ ܫܘܼܠܛܵܢܹܗ: ܕܠܵܐ ܢܸܣܥܸܐ ܥܠܲܝ. ܘܐܲܦܢܵܢܝ ܘܲܫܒ݂ܘܿܩ ܠܝܼ ܚܵܘܒܲܝ̈: ܡܫܝܼܚܵܐ ܚܵܐܹܢ ܟܠ. ܐܸܬ݂ܦܲܢ ܥܠܲܝ ܘܪܲܚܸܡ ܥܠܲܝ. ܚܘܼܢܲܝܢܝ ܡܵܪܝ ܕܲܫܒ݂ܵܢܝ ܒܝܼܫܵܐ: ܒܚܵܘܒܹ̈ܐ ܕܣܸܥܪܹܬ݂. ܕܐܲܢ݇ܬ݁ܘܼ ܒܲܠܚܘܿܕ݂ ܡܙܲܕܸܩ ܐܲܢ݇ܬ݁: ܠܐܲܝܠܹܝܢ ܕܩܵܪܹܝܢ ܠܵܟ݂. ܡܵܪܲܢ ܒܪ̈ܲܚܡܲܝܟ ܚܘܼܣ ܥܠܲܝ: ܘܠܵܐ ܬܲܗܡܸܐ ܡܹܢܝ. ܡܸܛܠ ܕܐܲܢ݇ܬ݁ܘܼ ܐܝܼܵܠܵܐ: ܕܟ݂ܠܗܘܿܢ ܡܛܲܪ̈ܦܹܐ. ܥܢܝܼܢ ܐܲܠܵܗܵܐ ܦܵܪܘܿܩܲܢ. ܥܢܝܼܢܝ ܡܵܪܝܵܐ ܕܠܵܟ݂ ܩܵܪܹܐ ܐ݇ܢܵܐ: ܒܲܕ݂ܡܘܼܬ݂ ܦܲܛܪܘܿܣ. ܠܗ̇ܘ ܕܲܠܝܼܬܵܝܗܝ ܡܢ ܓܵܘ ܝܲܡܵܐ: ܫܓ݂ܝܼܫܵܐ ܘܲܕ݂ܠܝܼܚܵܐ. ܘܠܝܼ ܦܲܨܵܢܝ ܕܗܵܐ ܐܸܬ݂ܚܲܢܩܹܬ݂: ܒܝܲܥܪܵܐ ܕܣܲܢ̈ܝܵܬ݂ܵܐ. ܘܐܲܦܢܵܢܝ ܨܹܐܕܲܝܟ ܒܲܚܢܵܢܵܟ݂: ܡܫܝܼܚܵܐ ܚܵܐܹܢ ܟܠ. ܫܵܘܙܸܒܲܝܢܝ ܘܦܲܨܵܢܝ ܡܢ ܕܵܪܵܐ ܗܵܢܵܐ ܠܥܵܠܲܡ. ܫܵܘܙܸܒܲܝܢܝ ܡܵܪܝ ܡܢ ܝܲܕ݂ ܒܝܼܫܵܐ: ܕܲܫܒ݂ܵܢܝ ܡܸܢܵܟ݂. ܕܗܵܐ ܨܵܒܹܐ ܕܲܡܥܲܪܩܸܠ ܠܝܼ: ܒܢܸܫ̈ܒܹ݁ܐ ܕܲܡܨܝܼ̈ܕ݂ܵܬܹܗ. ܫܲܕܲܪ ܚܲܝܠܵܟ݂ ܡܢ ܪܵܘܡܵܐ: ܘܲܢܦܲܣܸܩ ܢܸܫܒܵܘ̈ܗܝ. ܕܠܵܐ ܢܸܬ݂ܓܲܢ݇ܒܲܪ ܘܢܸܣܥܸܐ ܥܠܲܝ: ܘܒܲܡܨܝܼܕ݁ܬܹ݁ܗ ܢܵܘܒ݁ܕܲܢܝ. ܕܐܘܼܪܚܵܐ܀ ܒܝܵܘ̈ܡܵܬ݂ܵܐ ܩܲܕ݂ܡܵܝܹ̈ܐ. ܝܵܘܣܸܦ ܕܫܸܦܪܲܬ݀ ܠܵܟ݂ ܐܘܼܪܚܹܗ: ܥܢܲܝܬܵܝܗܝ ܡܵܪܝܵܐ. ܘܲܚܠܵܦ ܓܘܼܒܵܐ ܚܸܫܘܿܟ݂ܵܐ: ܬܵܓ݂ܵܐ ܛܲܝܸܒ݂ܬ݁ ܠܹܗ. ܘܐܸܢܵܐ ܕܡܸܛܠ ܬܘܼܪܣܵܝܵܐ: ܒܐܘܼܪܚܵܐ ܚܵܙܹܩ ܐ݇ܢܵܐ. ܥܘܼܕ݂ܪܵܢܵܟ݂ ܢܸܬ݂ܠܵܘܸܐ ܠܝܼ ܡܵܪܝ: ܘܐܸܗܦܘܿܟ݂ ܒܲܫܠܵܡܵܐ. ܕܡܸܛܪܵܐ܀ ܫܡܲܥ ܐܲܠܵܗܵܐ ܩܵܠܝ ܟܲܕ݂ ܐܸܬ݂ܟܲܫܲܦ ܠܵܟ݂. ܐܵܐܲܪ ܕܲܪܦܸܐ ܢܸܬ݂ܚܲܝܲܨ: ܒܪܸܡܙܵܟ݂ ܡܵܪܝܵܐ. ܘܒ݂ܵܥܘܼܬܲܢ ܠܵܐ ܬܸܫܬܵܘܚܲܪ: ܡܸܢ ܩܘܼܒܵܠܵܐ. ܘܐܲܪܥܵܐ ܕܲܠܒ݂ܝܼܫܵܐ ܐܸܒ݂ܠܵܐ: ܒܛܲܠܵܟ݂ ܬܸܬ݂ܦܲܝܲܓ݂. ܘܡܸܛܪܵܐ ܕܪ̈ܲܚܡܹܐ ܫܲܕܲܪ ܠܲܢ: ܝܵܗܘܿܒ݂ܵܐ ܕܚܲܝܲܝ̈ܢ. ܕܡܲܪܝܲܡ܀ ܢܸܗܘܸܐ ܫܡܹܗ ܕܡܵܪܝܵܐ ܡܒܲܪܲܟ݂. ܒܪܝܼܟ݂ܘܼ ܡܸܠܬ݂ܵܐ ܕܡܢ ܐܲܒ݂ܵܐ: ܕܲܨܒ݂ܵܐ ܘܐܸܬ݂ܪܟܸܢ. ܘܲܓ݂ܒ݂ܵܐ ܒܬ݂ܘܼܠܬܵܐ ܡܢ ܓܸܢܣܲܢ: ܘܝܲܩܪܵܗ̇ ܒܚܘܼܒܹܗ. ܘܥܲܒ݂ܕܵܗ̇ ܢܸܒ݂ܥܵܐ ܕܥܘܼܕ݂ܪ̈ܢܹܐ: ܒܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ. ܘܒܲܐܪ̈ܒܲܥܦܸܢܝܵܢ ܡܸܙܕܲܝܲܚ: ܝܵܘܡܵܐ ܕܕ݂ܘܼܟ݂ܪܵܢܵܗ̇. ܕܩܲܕܝܼܫܹ̈ܐ܀ ܬܲܩܝܼܦܵܐ ܘܲܡܫܲܒܚܵܐ. ܠܵܟ݂ܘܼ ܦܲܐܝܵܐ ܬܸܫܒܘܿܚܬܵܐ: ܡܵܘܪܸܒ݂ ܥܲܒ݂ܕܵܘ̈ܗܝ. ܢܒ݂ܝܼܹ̈ܐ ܫܠܝܼܚܹ̈ܐ ܘܡܲܠܦܵܢܹ̈ܐ: ܘܣܵܗܕܹ̈ܐ ܘܟ݂ܠ ܟܹܐܢܝܼ̈ܢ. ܕܒ݂ܵܐܬ݂̈ܘܵܬ݂ܵܐ ܐܵܘܪܸܒ݂ܬ݁ ܐܸܢܘܿܢ: ܒܥܹܕܲܬ݂ ܩܘܼܕ݂ܫܵܐ. ܘܕܲܪ̈ܓܹܐ ܫܒ݂ܝܼܚܹ̈ܐ ܡܠܲܟ݂ܬ݁ ܠܗܘܿܢ: ܒܥܘܼܡܪܵܐ ܕܲܫܡܲܝܵܐ. ܕܲܨܠܝܼܒ݂ܵܐ܀ ܐܲܝܟ݂ ܡܕ݂ܝܼܢ݇ܬܵܐ ܕܲܟ݂ܪܝܼܟ݂ ܠܵܗ̇ ܫܘܼܪܵܐ. ܗܘܝܼ ܡܵܪܝ ܫܘܼܪܵܐ ܚܲܣܝܼܢܵܐ: ܠܢܲܦ̮̈ܫܵܬܲܢ. ܘܢܲܛܲܪ ܠܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܒܚܲܝܠܹܗ ܕܲܨܠܝܼܒ݂ܵܟ݂. ܘܗܲܒ݂ܠܲܢ ܕܢܸܓ݂ܡܘܿܪ ܨܸܒ݂ܝܵܢܵܟ݂: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬܲܢ. ܘܒܲܚܕ݂ܵܐ ܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ: ܢܸܙܡܲܪ ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ. ܕܐܲܒ݂ܘܼܢ܀ ܐܹܒܲܪܸܟ݂ ܠܡܵܪܝܵܐ ܒܟ݂ܠܙܒܲܢ. ܒܪܝܼܟ݂ܘܼ ܕܐܵܘܪܸܒ݂ ܕܘܼܟ݂ܪܵܢܵܟ݂: ܐܲܒ݂ܘܼܢ ܙܲܗܝܵܐ. ܘܥܲܒ݂ܕܵܟ݂ ܢܸܒ݂ܥܵܐ ܕܥܘܼܕ݂ܪ̈ܢܹܐ: ܠܟ݂ܠ ܕܐܲܠܝܼܨܝܼܢ. ܠܹܗ ܐܸܬ݂ܟܲܫܲܦ ܚܠܵܦ ܟܠܲܢ: ܕܲܢܚܘܼܣ ܥܠܲܝܢ. ܡܵܐ ܕܡܸܬ݂ܓܲܠܝܵܐ ܬܸܫܒܘܿܚܬܹ݁ܗ: ܠܲܡܕ݂ܵܢ ܟܠ ܒܸܪ̈ܝܵܢ. ܕܥܲܢܝܼ̈ܕܹܐ܀ ܘܲܪܢܹܝܬ݂ ܘܐܸܬ݁ܛܲܪܦܲܬ݀ ܪܘܼܚܝ. ܪܗܝܼܒ݂ܘܼ ܝܵܘܡܵܐ ܕܡܹܐܬ݂ܝܼܬ݂ܵܟ݂: ܕܒܹܗ ܕܵܝܢܲܬ݁ ܠܟ݂ܠ. ܘܢܵܦܹܠ ܙܵܘܥܵܐ ܘܲܪܬܹܝܬ݂ܵܐ: ܒܟ݂ܠܗܘܿܢ ܥܲܡܡܹ̈ܐ. ܘܚܵܝܪܝܼܢ ܟܠܗܘܿܢ ܘܡܸܬ݂ܒܲܩܹܝܢ: ܒܲܩܢܘܿܡܲܝ̈ܗܘܿܢ. ܘܚܵܕܹܝܢ ܟܹܐܢܹ̈ܐ ܘܡܸܬ݂ܒܲܣܡܝܼܢ: ܒܢܘܼܗܪܵܐ ܕܡܲܠܟܘܼܬ݂ܵܟ݂. ܘܒ݂ܝܼܫܹ̈ܐ ܝܵܪܬ݁ܝܼܢ ܓܹܗܲܢܵܐ: ܕܫܘܼܠܵܡܵܐ ܠܲܝܬ݁ ܠܵܗ̇. ܐܸܬ݁ܕܲܟ݂ܪܹܬ݂ ܐܲܠܵܗܵܐ ܘܐܸܬ݁ܕܲܠܚܹܬ݂. ܠܲܓ݂ܙܵܪ ܕܝܼܢܵܐ ܐܸܬ݂ܥܲܗܕܹܬ݂: ܡܵܪܵܐ ܕܟ݂ܠܵܐ. ܘܚܸܫܹܬ݂ ܣܲܓܝܼ ܥܲܠܚܵܘܒܲܝ̈: ܕܲܒ݂ܚܲܝܲܝ̈ ܐܸܣܬ݁ܥܲܪܘ. ܕܲܥܬ݂ܝܼܕܲܬ݁ ܠܲܡܕ݂ܵܢ ܟܠܢܵܫ: ܕܝܼܢܵܐ ܕܚܝܼܠܵܐ. ܒܲܕ݂ܡܘܼܬܹܗ ܕܗ̇ܘ ܓܲܝܵܣܵܐ: ܕܚܵܢܘܼܗܝ ܪ̈ܲܚܡܲܝܟ. ܐܲܦܢܵܢܝ ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝ: ܡܫܝܼܚܵܐ ܚܵܐܹܢ ܟܠ܀
ܐܹܒܲܪܸܟ݂ ܠܡܵܪܝܵܐ ܒܟ݂ܠܙܒܲܢ. ܒܪܝܼܟ݂ܘܼ ܝܵܘܡܵܟ݂ ܒܲܪ ܡܵܪܹܟܠ. ܕܐܵܬܹܐ ܘܣܵܕܹܩ ܥܘܼܒܵܗ̇ ܕܲܫܝܘܼܠ. ܫܒ݂ܝܼܚ ܢܘܼܚܵܡܵܟ݂ ܕܲܡܣܲܟܹ݁ܝܢ ܠܹܗ. ܕܵܪܹ̈ܐ ܕܲܥܒܲܪܘ ܘܲܕ݂ܩܲܝܵܡܝܼܢ. ܡܸܟܹ݁ܝܠ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ. ܦܘܼܫ ܒܲܫܠܵܡܵܐ ܥܘܼܡܪܵܐ ܕܙܲܒ݂ܢܵܐ. ܕܠܵܐ ܡܨܸܐ ܦܵܪܹܩ̇ ܠܲܕ݂ܩܵܢܹܝܢ ܠܹܗ. ܕܐܹܙܲܠ ܘܐܸܚܙܹܝܘܗܝ ܠܐܲܬ݂ܪܵܐ ܕܢܘܼܗܪܵܐ. ܕܲܡܕܲܝܪܝܼܢ ܒܹܗ ܟܹܐܢܹ̈ܐ ܕܲܥܡܲܠܘ.
ܫܲܒܲܚ. ܒܩܵܠܵܐ ܪܵܡܵܐ ܕܬܸܫܒ݁ܘܿܚܬܵܐ ܢܫܲܒܲܚ ܟܠܲܢ ܠܒ݂ܵܪܘܿܝܵܐ. ܚܲܝܠܵܐ ܪܲܒܵܐ ܘܬܲܩܝܼܦܵܐ. ܕܝܲܗ݇ܒ݂ܠܲܢ ܕܢܸܕ݂ܥܝܼܘܗܝ ܚܲܕ݂ ܟܝܵܢܵܐ. ܘܚܲܕ݂ ܚܲܝܠܵܐ ܕܠܵܐ ܡܸܣܬܲܝܲܟ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܩܘܼܕ݂ܫܵܐ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.

ܫܲܒܲܚ: ܥܵܠܲܡ. ܒܟ݂ܵܗ̇ܢܘܼܬ݂ܵܐ ܕܒܹܝܬ݂ ܐܲܗܪܘܿܢ ܐܸܫܬܲܡܲܫ ܐ݇ܪܵܙܵܐ ܘܛܸܠܵܠܵܐ. ܘܲܒ݂ܝܲܕ݂ ܫܠܝܼܚܹ̈ܐ ܫܪܵܪܵܐ ܓܲܠܝܵܐ ܕܚܹܝܠ ܐܝܼܬ݂ܘܼܬ݂ܵܐ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܩܘܼܕ݂ܫܵܐ ܟܝܵܢܵܐ ܕܠܵܐ ܫܘܼܪܵܝ ܀
ܪܲܚܸܡܥܠܲܝ ܐܲܠܵܗܵܐ ܐܲܝܟ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂. * ܘܐܲܝܟ݂ ܣܘܿܓ݂ܐܵܐ ܕܪ̈ܲܚܡܲܝܟ ܥܛܝܼ ܚܛܵܗܲܝ̈. ܒܟ݂ܵܗ̇ܢܘܼܬ݂ܵܐ ܕܒܹܝܬ݂ ܐܲܗܪܘܿܢ ܐܸܫܬܲܡܲܫ ܐ݇ܪܵܙܵܐ ܘܛܸܠܵܠܵܐ. ܘܲܒ݂ܝܲܕ݂ ܫܠܝܼܚܹ̈ܐ ܫܪܵܪܵܐ ܓܲܠܝܵܐ ܕܚܹܝܠ ܐܝܼܬ݂ܘܼܬ݂ܵܐ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܩܘܼܕ݂ܫܵܐ ܟܝܵܢܵܐ ܕܠܵܐ ܫܘܼܪܵܝ܀

 ܫܠܝܼܚܹ̈ܐ ܩܲܕܝܼܫܹ̈ܐ ܐܲܦܝܼܣܘ ܠܲܡܫܝܼܚܵܐ. ܕܲܒ݂ܝܲܕ݂ ܨܠܵܘ̈ܵܬ݂ܟ݂ܘܿܢ ܢܸܗܘܘܿܢ ܥܠܲܝܢ ܪ̈ܲܚܡܹܐ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܒܩܲܕ݂ܡܵܝܹ̈ܐ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܕܝܵܘܡܵܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܡܵܪܝܵܐ ܚܲܝܠܬ݂ܵܢܵܐ ܐܝܼܬ݂ܝܵܐ ܡܬ݂ܘܿܡܵܝܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܘܡܹܐ ܥܸܠܵܝܹ̈ܐ : ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ݂܀ ܗܵܘܿ ܕܲܒ݂ܚܘܼܒܹ݁ܗ ܣܲܓ݁ܝܼܐܵܐ ܕܐܲܚܒܲܢ ܝܲܩܲܪ ܬܘܼܩܵܢܹܗ ܕܓܸܢܣܲܢ ܒܨܲܠܡܵܐ ܕܐܝܼܩܵܪܹܗ:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܒܲܐܒ݂ܪܵܗܵܡ ܡܗܲܝܡܢܵܐ ܡܠܲܟ݂ ܛܵܒ݂ܵܬ݂ܵܐ ܠܪܵܚܡܵܘܗܝ: ܘܲܒ݂ܓܸܠܝܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܐܸܬ݂ܝܼܕܲܥ ܠܥܹܕ݁ܬܹܗ: ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܠܵܐ ܨܵܒܹܐ ܒܐܲܒ݂ܕܵܢܹܗ ܕܲܟ݂ܝܵܢܲܢ:  ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܼܢ ܛܵܥܝܘܼܬ݂ܵܐ ܕܚܸܫܘܿܟ݂ܵܐ ܠܝܼܕܲܥܬ݂ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܗܘܝܘ ܒܲܠܚܘܿܕ݂ܵܘܗܝ ܥܵܒ݂ܘܿܕ݂ܵܐ ܘܡܲܬ݂ܩܢܵܢܵܐ ܕܒܸܪ̈ܝܵܬ݂ܵܐ:  ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ ؛ ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܝܼܣ:  ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܝܼܣ : ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈  ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗܵܘܿ ܕܲܒ݂ܪܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܗܵܘܿ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚܘ ܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂:  ܒܵܥܹܝܢܲܢ ܢܡܸܢܵܟ ܀ ܙܲܟܵܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܡܹܐܬ݂ܝܼܬ݂ܵܟ݂:  ܘܫܲܝܸܢ ܥܹܕ݁ܬܵܟ݂ ܦܪܝܼܩܲܬ݂ ܒܲܕ݂ܡܵܟ݂ ܝܲܩܝܼܪܵܐ: ܘܐܸܬ݂ܪܲܚܲܡܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨ̈ܠܵܘܵܬ݂ܵܐ ܕܩܵܠܹ̈ܐ ܫܲܗܪܵܐ
ܨܠܘܿܬ݂ܵܐܕܩܲܕ݂ܡܵܝܵܐ. ܢܸܫܦܪܘܼܢ ܠܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܩܵܠܹ̈ܐ ܕܗܘܼܠܵܠܲܝ̈ܢ ܘܢܸܥ̈ܡܵܬ݂ܵܐ ܕܲܙܡܝܼܪ̈ܵܬܲܢ: ܘܩܲܒܸ݁ܠܡܢܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܦܹܐܪܹ̈ܐ ܡܠܝܼܠܹ̈ܐ ܕܣܸܦ̈ܘܵܬܲܢ: ܕܲܒ݂ܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܩܲܪܒ݂ܝܼܢܲܢ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܡܲܩܦܝܼܢ ܘܲܡܫܲܪܹܝܢ ܡܲܪܡܝܼܬ݂ܵܐ ܩܲܕ݂ܡܵܝܬܵܐ ܕܗܘܼܠܵܠܵܐ ܐܲܝܢܵܐ ܕܗܘܟܠܵܗ̇ ܒܩܵܠܵܗ̇: ܘܲܡܫܲܒ݁ܚܝܼܢ ܘܲܡܥܵܠܡܝܼܢ ܘܐܵܡܪܝܼܢ: ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܪܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܐܲܚܪ̈ܵܝܹܐ ܡܲܪܡܝܼܬ݂ܵܐ * ܗܵܝܕܹ݁ܝܢ ܫܲܒܲܚܘ ܡܘܼܫܹܐ ܘܲܒ݂ܢܲܝ̈ ܝܼܣܪܵܝܹܠ. ܡܲܙܡܘܪ̈ܐ ܩܢܐ‐ܩܢܒ.

ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܲܬ݂ܪܲܝܵܢܵܐ. ܐܲܫܘܵܐ ܠܲܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܕܥܲܡ ܥܝܼܪܹ̈ܐ ܘܟܸܢܫܹ̈ܐ ܕܡܲܠܲܐܟܹ̈ܐ: ܒܩܵܠܹ̈ܐ ܕܲܡܠܹܝܢ ܬܵܘܕ݁ܝܼܬ݂ܵܐ: ܢܸܙܡܲܪ ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ. ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܲܡܫܲܡܫܝܼܢ ܟܠܵܗ̇ ܡܲܪܡܝܼܬ݂ܵܐ ܬܪܲܝܵܢܝܼܬ݂ܵܐ ܕܗܘܼܠܵܠܵܐ ܕܐܲܝܢܵܐ ܕܗܘ ܒܩܵܠܵܗ̇. ܐܲܝܟ݂ ܩܲܕ݂ܡܵܝܬܵܐ.
ܘܐܵܡܪܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܩܲܕܡܵܝܹ̈ܐ ܡܲܪܡܝܼܬ݂ܵܐ. ܫܲܒܲܚܘ ܠܡܵܪܝܵܐ ܬܸܫܒ݁ܘܿܚܬܵܐ ܚܕܲܬܵܐ. ܚܙܝܼ: ܨܘܼܬ݂ ܫܡܲܝܵܐ ܘܐܹܡܲܠܸܠ. ܡܙܡܘܪܐ ܩܢܓ
ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܵܐ: ܫܘܼܒ݂ܚܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ: ܘܗܘܼܠܵܠܹ̈ܐ ܕܠܵܐ ܫܲܠܘܵܐ: ܘܬܸܫܒ̈ܚܵܬ݂ܵܐ ܕܠܵܐ ܣܵܟ݂ܵܐ: ܘܩܵܠܹ̈ܐ ܕܲܡܠܹܝܢ ܬܵܘܕ݁ܝܼܬ݂ܵܐ: ܚܲܝܵܒ݂ ܟܠܹܗ ܟܝܵܢܵܐ ܕܲܡܠܝܼ̈ܠܹܐ ܕܲܒ݂ܪܲܝܬ݁: ܠܡܲܣܵܩܘܼ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܲܡܫܲܡܫܝܼܢ ܟܠܵܗ̇ ܡܲܪܡܝܼܬ݂ܵܐ ܬܠܝܼܬ݂ܵܝܬܵܐ ܕܗܘܼܠܵܠܵܐ ܐܲܝܢܵܐ ܕܗܘ ܒܩܵܠܵܗ̇. ܐܲܝܟ݂ ܩܲܕ݂ܡܵܝܬܵܐ.
ܘܐܵܡܪܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܩܲܕܡܵܝܹ̈ܐ ܡܲܪܡܝܼܬ݂ܵܐ. ܘܐܵܦ ܐܸܢܵܐ ܐܲܛܸܢ ܐܸܢܘܿܢ ܒܠܵܐ ܥܲܡܝ. ܡܙܡܘܪܐ ܩܢܕ
 ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ ܕܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ. ܡܢ ܦܘܼܡܹ̈ܐ ܡܵܪܝ ܡܠܝܼܠܹ̈ܐ ܕܲܒ݂ܪܲܝܬ݁: ܘܡܢ ܠܸܫܵܢܹ̈ܐ ܡܫܲܒ݁ܚܵܢܹ̈ܐ ܕܪܲܟܸ݁ܒ݂ܬ݁: ܘܡܢ ܟܸܢܫܹ̈ܐ ܟܠܗܘܿܢ ܕܥܸܠܵܝܹ̈ܐ ܘܲܕ݂ܬܲܚܬܵܝܹ̈ܐ: ܫܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܲܕ݂ܡܵܪܘܼܬ݂ܵܟ݂ ܢܸܣܬ݁ܓܸܕ݂ ܘܢܸܫܬܲܒܲܚ ܘܢܸܬ݂ܝܲܩܲܪ ܘܢܸܬ݂ܪܲܡܪܲܡ ܘܢܸܬ݁ܬܵܘܕܸ݁ܐ ܘܢܸܬ݂ܒܲܪܲܟ݂ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܪܝܼܢ ܥܘܿܢܝܼܬ݂ܵܐ ܕܠܸܠܝܵܐ ܘܬܵܢܹܝܢ ܘܲܡܫܲܒܚܝܼܢ ܒܵܗ̇ ܘܡܲܩܦܝܼܢ:
ܕܠܸܠܝܵܐ. ܒܲܫܠܝܼܚܹ̈ܐ ܓܒܲܝ̈ܵܐ. ܐܹܒܲܪܸܟ݂ ܠܡܵܪܝܵܐ ܒܟ݂ܠܙܒܲܢ: ܘܲܒ݂ܪܝܼܟ݂ܘܼ ܫܡܵܐ ܕܐܝܼܩܵܪܹܗ ܠܥܵܠܲܡ: ܘܲܢܒܲܪܟ݂ܝܼܘܼܗܝ ܠܡܵܪܝܵܐ ܕܥܲܒ݂ܕܲܢ. ܕܨܘܼܪܬܵܐ. ܘܥܲܒ݂ܕܹܗ ܐܲܠܵܗܵܐ ܠܐܵܕ݂ܵܡ ܒܨܲܠܡܹܗ ܘܒܲܕ݂ܡܘܼܬܹܗ. ܒܪܝܼܟ݂ ܛܵܒ݂ܵܐ ܕܲܒ݂ܫܘܼܪܵܝܵܐ ܥܲܒ݂ܕܹܗ ܠܐܵܕ݂ܵܡ ܒܨܲܠܡܹܗ ܘܲܕ݂ܡܘܼܬܹܗ. ܘܲܒ݂ܙܲܒ݂ܢܹ̈ܐ ܐ݇ܚܪ̈ܵܝܹܐ ܚܵܣ ܥܠܵܘܗܝ ܕܲܢܦܲܠ ܡܢ ܫܘܼܒ݂ܚܹܗ. ܘܐܸܬ݀ܵܐ ܡܢ ܫܡܲܝܵܐ ܠܦܘܼܪܩܵܢܲܢ. ܘܲܫܩܲܠ ܡܢ ܓܸܢܣܲܢ ܡܵܝ̇ܘܿܬ݂ܵܐ ܗܡܲܝܪܵܐ ܠܐܝܼܩܵܪܹܗ. ܘܝܲܗ݇ܒ݂ܠܹܗ ܕܢܸܗܘܸܐ ܡܕܲܒ݁ܪܵܢܵܐ ܘܪܹܫܵܐ ܘܡܵܪܵܐ. ܥܲܠ ܟܠ ܡܵܐ ܕܒܲܫܡܲܝܵܐ ܘܡܵܐ ܕܒܲܐܪܥܵܐ. ܘܒܹܗ ܐܸܬ݂ܚܲܕܲܬ݂ ܟܠܹܗ ܟܝܵܢܵܐ. ܕܐܸܬ݂ܚܲܒܲܠ ܗ݇ܘܵܐ ܒܡܵܘܬܵܐ ܢܲܓ݁ܝܼܪܵܐ ܕܐܲܡܠܸܟ݂ ܥܠܵܘܗܝ ܒܲܚܛܵܗܵܘ̈ܗܝ: ܬܢܝܼ. ܘܫܲܒܲܚ ܒܵܗ.
ܥܵܠܲܡ. ܡܫܝܼܚܵܐ ܠܵܐ ܬܲܗܡܸܐ ܡܸܢܲܢ. ܘܠܵܐ ܬܲܪܚܸܩ ܡܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ. ܕܒ݂ܵܟ݂ܘܼ ܡܵܪܝ ܐܸܬ݂ܓܵܘܲܣܢܲܢ. ܕܲܒܲܪܲܝܢ ܒܐܘܼܪܚܵܟ݂ ܕܚܲܝܹ̈ܐ ܕܲܢܙܲܡܲܪ ܠܵܟ݂ ܟܠܲܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ. ܢܹܐܡܲܪ. ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܐܸܡܹܗ ܕܝܼܫܘܿܥ ܦܵܪܘܿܩܲܢ: ܐܲܦܝܼܣܝ ܥܲܡܲܢ ܠܲܡܫܝܼܚܵܐ ܕܢܲܫܪܸܐ ܫܲܝܢܹܗ ܒܲܝܢܵܬܲܢ: ܘܲܢܢܲܛܲܪ ܠܲܢ ܡܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܒܠܸܠܝܵܐ ܘܒ݂ܐܝܼܡܵܡܵܐ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܒܵܚܵܐ. ܠܵܟ݂ ܫܘܼܒ݂ܚܵܐ ܡܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܡܢ ܟܠ ܠܸܫܵܢܝܼ̈ܢ: ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܘܗܸܕ݂ܪܵܐ ܘܪܘܼܡܪܵܡܵܐ ܡܢ ܟܠ ܒܸܪ̈ܝܵܢ: ܐܝܼܬ݂ܝܵܐ ܓܢܝܼܙܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܥܵܡܲܪ ܒܲܡܪ̈ܘܡܹܐ ܥܸܠܵܝܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܡܵܪܝܵܐ.* ܛܥܲܡܘ ܘܲܚܙܵܘ ܕܛܵܒ݂ܘܼ ܡܵܪܝܵܐ. ܘܛܘܼܒܲܝܗܘܿܢ ܠܟ݂ܠ ܕܲܬ݂ܟ݂ܝܼܠܝܼܢ ܥܠܵܘܗܝ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ.* ܥܲܬ݁ܝܼܪܹ̈ܐ ܐܸܬ݂ܡܲܣܟܲܢܘ ܘܲܟ݂ܦܸܢܘ. * ܘܐܲܝܠܹܝܢ ܕܲܠܡܵܪܝܵܐ ܒܵܥܹܝܢ ܠܵܐ ܬܸܚܣܲܪ ܠܗܘܿܢ ܛܵܒ݂ܬ݂ܵܐ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ.* ܬܵܘ ܒܢܲܝ̈ܵܐ ܫܘܼܡܥܘܼܢܝ. * ܘܐܲܠܸܦܟ݂ܘܿܢ ܕܸܚܠܬܹܗ ܕܐܲܠܵܗܵܐ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ.* ܐܲܝܢܵܘ ܓܲܒ݂ܪܵܐ ܕܨܵܒܹ݁ܐ ܚܲܝܹ̈ܐ. * ܘܪܵܚܹܡ ܝܵܘ̈ܡܵܬ݂ܵܐ ܛܵܒܹ̈ܐ ܠܡܸܚܙܵܐ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ.* ܛܲܪ ܠܸܫܵܢܵܟ݂ ܡܢ ܒܝܼܫܬܵܐ. * ܘܣܸܦ̈ܘܵܬ݂ܵܟ݂ ܠܵܐ ܢܡܲܠ̈ܠܵܢ ܢܸܟ݂ܠܵܐ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ.* ܥܒܲܪ ܡܢ ܒܝܼܫܬܵܐ ܘܲܥܒܸܕ݂ ܛܵܒ݂ܬ݂ܵܐ. * ܒܥܝܼ ܫܠܵܡܵܐ ܘܗܲܪܛ ܒܵܬ݂ܪܹܗ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ.* ܥܲܝܢܵܘ̈ܗܝ ܕܡܵܪܝܵܐ ܥܲܠ ܙܲܕܝܼܩܹ̈ܐ. * ܘܐܸܕ݂ܢܵܘ̈ܗܝ ܠܡܸܫܡܲܥ ܐܸܢܘܿܢ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ.* ܐܲܦܵܘ̈ܗܝ ܕܡܵܪܝܵܐ ܥܲܠ ܒܝܼܫܹ̈ܐ. * ܕܢܸܥܛܸܐ ܡܢ ܐܲܪܥܵܐ ܕܘܼܟ݂ܪܵܢܗܘܿܢ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ. * ܓܥܵܘ ܙܲܕܝܼܩܹ̈ܐ ܘܡܵܪܝܵܐ ܫܡܲܥ ܐܸܢܘܿܢ ܘܦܲܨܝܼ ܐܸܢܘܿܢ. * ܩܲܪܝܼܒ݂ܘܼ ܡܵܪܝܵܐ ܠܲܬ݂ܒ݂ܝܼܪ̈ܲܝ ܠܸܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ.* ܘܲܠܡܲܟ݁ܝܼܟܲܝ̈ ܪܘܼܚܵܐ ܦܵܪܹܩ̇. * ܣܲܓܝܼ̈ܐܵܢ ܒܝܼ̈ܫܵܬܹܗ ܕܙܲܕܝܼܩܵܐ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ.* ܘܡܢ ܟܠܗܹܝܢ ܡܦܲܨܸܐ ܠܹܗ ܡܵܪܝܵܐ. * ܘܢܵܛܲܪ ܠܟ݂ܠܗܘܿܢ ܓܲܪ̈ܡܵܘܗܝ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ.* ܕܚܲܕ݂ ܡܸܢܗܘܿܢ ܠܵܐ ܢܸܬ݁ܬ݁ܒܲܪ. * ܩܵܛ̇ܠܵܐ ܠܪ̈ܲܫܝܼܥܹܐ ܒܝܼܫܬܵܐ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ. * ܘܣܵܢܐܵܘ̈ܗܝ ܕܙܲܕܝܼܩܵܐ ܢܣܘܼܦܘܼܢ. * ܢܸܦܪܘܿܩ ܡܵܪܝܵܐ ܢܲܦ̮̈ܫܵܬ݂ܵܐ ܕܥܲܒ݂ܕܵܘ̈ܗܝ. * ܘܠܵܐ ܢܸܬ݂ܚܲܝܒ݂ܘܼܢ ܟܠ ܕܲܡܣܲܒܪܝܼܢ ܒܹܗ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ.
 ܫܲܒܲܚ: ܘܥܵܠܲܡ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ. ܘܥܵܛ̇ܦܝܼܢ ܠܪܹܫܹܗ ܕܡܲܙܡܘܿܪܵܐ ܐܹܒܲܪܸܟ݂ ܠܡܵܪܝܵܐ ܒܟ݂ܠܙܒܲܢ. * ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܬܸܫܒ̈ܚܵܬܹܗ ܒܦܘܼܡܝ. * ܒܡܵܪܝܵܐ ܬܸܫܬܲܒ݂ܗܲܪ ܢܲܦ̮ܫܝ. ܫܘܼܒ݂ܚܵܐ ܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܐܵܗܝܼܢܓܵܗ؛ ܠܵܟ݂ ܡܵܪܝܵܐ..
ܘܐܵܡ̇ܪܝܼܢ ܗܦܵܟ݂ܬܵܐ. ܡܢ ܦܫܝܼܛܵܬ̈ܵܐ ܠܲܫܒ݂̈ܝܼܚܵܬ݂ܵܐ ܕܲܪܸܓ݂ܘ ܐܸܢܘܿܢ: ܕܗܵܟܲܢ ܦܵܩܹܕ݂ ܛܲܟ݂ܣܵܐ ܬܲܩܢܵܐ ܠܐܲܝܢܵܐ ܕܡܲܠܸܦ܀ ܫܸܡ ܒܲܪ ܐ݇ܢܵܫܵܐ ܐܲܗܓ݁ܝܼܘ ܐܸܢܘܿܢ ܐܲܝܟ݂ ܬܲܠܡܝܼܕܹ̈ܐ: ܘܡܢ ܕܐܸܬ݂ܚܲܟܲܡܘ ܩܪܵܐ ܬܲܩܢܵܐܝܼܬ݂ ܡܸܠܬ݂ܵܐ ܘܦܲܓ݂ܪܵܐ.
ܬܸܫܒܘܿܚܬܵܐ. ܘܲܒ݂ܚܲܕ݂ܒ݁ܫܲܒܹ̈ܐ ܕܹܝܢ ܐܵܡ̇ܪܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ ܗܵܕܹܐ ܕܲܥܒ݂ܝܼܕ݂ܵܐ ܠܡܵܪܝ ܢܲܪܣܲܝ ܡܲܠܦܵܢܵܐ.
ܬܵܘܕ݁ܝܼ ܠܛܵܒ݂ܵܐ ܕܚܲܪܲܪ ܓܸܢܣܲܢ: ܡܢ ܥܲܒ݂ܕ݁ܘܼܬ݂ܵܐ ܕܒ݂ܝܼܫܵܐ ܘܡܵܘܬܵܐ܀ ܫܲܝܸܢ ܥܲܡܲܢ ܟܸܢܫܲܝ̈ ܪܵܘܡܵܐ: ܕܪܲܓ݁ܝܼܙܝܼܢ ܗ݇ܘܵܘ ܡܸܛܠ ܥܵܘܠܲܢ܀ ܒܪܝܼܟ݂ ܚܲܢܵܢܵܐ ܕܟܲܕ݂ ܠܵܐ ܒܥܲܝܢܵܝܗܝ: ܢܦܲܩ ܒܲܒ݂ܥܵܬܲܢ ܘܲܚܕ݂ܝܼ ܒܚܲܝܲܝ̈ܢ܀ ܘܨܵܪ ܕܸܡ̈ܘܵܬ݂ܵܐ ܕܐܲܒ݁ܝܼܕ݂ܘܼܬܲܢ: ܘܲܕ݂ܦܘܼܢܵܝܲܢ ܒܥܸܪܒܵܐ ܕܲܛܥܵܐ܀ ܝܵܪܬܵܐ ܘܲܒ݂ܪܵܐ ܩܪܵܝܗܝ ܠܲܟ݂ܝܵܢܲܢ: ܕܲܛܥܵܐ ܘܲܦܢܵܐ ܘܡܝܼܬ݂ ܘܐܸܬ݂ܢܲܚܲܡ܀ ܘܚܲܕ݁ܝܼ ܠܟܸܢܫܹ̈ܐ ܪ̈ܘܼܚܵܢܵܝܹܐ: ܒܲܬ݂ܝܵܒ݂ܘܼܬܲܢ ܘܲܒ݂ܢܘܼܚܵܡܲܢ܀ ܠܵܐ ܡܸܬ݂ܡܲܠܲܠ ܚܘܼܒܵܐ ܪܲܒܵܐ: ܕܚܵܘܝܼ ܨܹܐܕܲܝܢ ܪܵܚܹ̇ܡ ܓܸܢܣܲܢ܀ ܕܡܸܢܹܗ ܕܓܸܢܣܲܢ ܥܒܲܕ݂ ܡܸܨܥܵܝܵܐ: ܘܪܲܥܝܼ ܠܥܵܠܡܵܐ ܥܲܡ ܪܲܒܘܼܬܹܗ܀ ܪܲܒܵܐ ܗ݇ܝ ܡܸܢܲܢ ܘܡܢ ܟܠ ܒܸܪ̈ܝܵܢ: ܚܕܲܬܵܐ ܕܲܣܥܲܪ ܨܹܝܕ݂ ܐ݇ܢܵܫܘܼܬܲܢ܀ ܕܥܲܒ݂ܕܹ݁ܗ ܠܦܲܓ݂ܪܲܢ ܗܲܝܟܲܠ ܩܘܼܕ݂ܫܵܐ: ܕܲܢܡܲܠܸܐ ܒܹܗ ܣܸܓ݂ܕܲܬ݂ ܟܠܵܐ܀ ܬܵܘ ܐܲܪ̈ܥܵܢܹܐ ܘܲܫܡܲܝܵܢܹ̈ܐ: ܬܗܲܪܘ ܘܐܸܬ݁ܕܲܡܲܪܘ ܒܪܲܒ݁ܘܼܬ݂ ܕܲܪܓ݂ܵܐ܀ ܕܲܡܛܵܐ ܓܸܢܣܲܢ ܠܪܵܘܡܵܐ ܪܲܒܵܐ: ܕܐܲܠܵܗܘܼܬ݂ܵܐ ܕܠܵܐ ܡܸܬ݁ܕܲܪܟܵܐ܀ ܫܡܲܝܵܐ ܘܐܲܪܥܵܐ ܘܟ݂ܠܡܵܐ ܕܲܒ݂ܗܘܿܢ: ܢܵܘܕ݁ܘܿܢ ܥܲܡܲܢ ܠܡܵܘܪܸܒ݂ ܓܸܢܣܲܢ܀ ܕܚܲܕܸ݁ܬ݂ ܨܲܠܡܲܢ ܘܲܥܛܵܐ ܥܵܘܠܲܢ: ܘܲܩܪܵܢ ܒܲܫܡܹܗ ܘܫܲܥܒܸ݁ܕ݂ ܠܲܢ ܟܠ܀ ܫܵܘܹ̇ܐ ܠܫܘܼܒ݂ܚܵܐ ܡܢ ܟܠ ܦܘܼܡܝܼ̈ܢ: ܗ̇ܘ ܕܐܲܪܝܼܡܲܢ ܠܥܸܠ ܡܢ ܟܠܵܐ܀ ܘܢܹܐܡܲܪ ܟܠܲܢ ܠܹܗ ܬܸܫܒ݁ܘܿܚܬܵܐ: ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀
ܘܩܵܝܡ̇ܝܼܢ: ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܐܲܝܟ݂ ܕܲܒ݂ܚܘܼܕ݂ܪܵܐ ܐܵܘ ܒܓܲܙܵܐ. ܐܵܘ ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ. ܕܠܵܐ ܚܲܒ݂ܪ̈ܵܬ݂ܵܗ̇.
(ܘܪܵܡܹܐ ܡܫܲܡܫܵܢܵܐ ܐܘܼܪܵܪܵܐ: ܘܐܵܣܲܪ ܙܘܼܢܵܪܵܐ: ܘܩܵܐܹ̇ܡ ܩܕ݂ܵܡ ܫܟ݂ܝܼܢܬܵܐ ܡܛܲܟ݁ܣܵܐܝܼܬ݂) ܘܐܵܡܲܪ ܟܵܪܘܿܙܘܼܬ݂ܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ: ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ.
ܢܩܘܼܡ ܫܲܦܝܼܪ ܟܠܲܢ ܒܚܲܕ݂ܘܼܬ݂ܵܐ ܘܲܦܨܝܼܚܘܼܬ݂ܵܐ ܢܸܒ݂ܥܸܐ ܘܢܹܐܡܲܪ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܡܲܩܸܦ: ܐܲܒ݂ܵܐ ܕܪ̈ܲܚܡܹܐ ܘܐܲܠܵܗܵܐ ܕܟ݂ܠܹܗ ܒܘܼܝܵܐܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܦܵܪܘܿܩܲܢ ܘܝܲܨܘܼܦܬ݂ܵܢܲܢ ܘܲܡܦܲܪܢܣܵܢܵܐ ܕܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ ܘܩܘܼܝܵܡܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ ܘܲܕ݂ܟ݂ܠܗܹܝܢ ܥܹ̈ܕ݂ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܐܲܬ݂ܪܲܢ ܘܥܲܠ ܟܠ ܐܲܬ݂ܪ̈ܵܘܵܢ: ܘܥܲܠ ܐܲܝܠܹܝܢ ܕܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܥܵܡ̇ܪܝܼܢ ܒܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܘܼܙܵܓ݂ܵܘ̈ܗܝ ܕܐܵܐܲܪ ܘܟܲܗܝܼܢܘܼܬ݂ܵܐ ܕܫܲܢ݇ܬܵܐ ܘܥܲܠ̈ܠܵܬ݂ܵܐ ܕܦܹܐܪܹ̈ܐ ܘܥܲܠ ܬܘܼܩܵܢܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ1: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܡܲܠܟܹ̈ܐ ܐܲܚܝܼܕܲܝ̈ ܫܘܼܠܛܵܢܹܗ ܕܥܵܠܡܵܐ ܗܵܢܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܪ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܥܲܠ ܩܲܫܝܼܫܹ̈ܐ ܘܲܡܫܲܡܫܵܢܹ̈ܐ ܐܲܪ̈ܬܵܕ݁ܘܿܟ݁ܣܹܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ܗ̇ܘ ܕܥܲܬ݁ܝܼܪ ܒܪ̈ܲܚܡܵܘܗܝ ܘܲܫܦܝܼܥ ܒܲܚܢܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܡܸܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ ܐܝܼܬ݂ܵܘܗܝ ܘܲܠܕ݂ܵܪܕܵܪ̈ܝܼܢ ܩܲܝܵܡ ܫܘܼܠܛܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܟ݂ܝܵܢܹܗ ܐܝܼܬ݂ܵܘܗܝ ܛܵܒ݂ܵܐ ܘܝܵܗܘܿܒ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹܐ ܒܡܵܘܬܹ݁ܗ ܕܚܲܛܵܝܵܐ ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡܸܢ ܥܵܘܠܹܗ ܘܢܸܚܸܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܲܒ݂ܝܲܠܕܹ݁ܗ ܐܵܘ ܕܲܒ݂ܕܸܢܚܹܗ ܐܵܘ ܕܲܒ݂ܨܵܘܡܹܗ ܐܵܘ ܕܲܒ݂ܡܲܥܲܠܬܹ݁ܗ ܐܵܘ ܕܒܲܩܝܵܡܬܹ݁ܗ ܐܵܘ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܐܵܘ ܕܲܒ݂ܡܲܚܲܬ݁ܬܹ݁ܗ ܐܵܘ ܕܒܲܨܠܝܼܒܹܗ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܐܲܦܨܲܚ ܠܲܫܡܲܝܵܐ: ܒܵܥܹܝܢܲܢ ܡܸܢܵܟ݂܀ ܗ̇ܘ ܕܠܹܗ ܗܘ ܟܝܵܢܵܐܝܼܬ݂ ܐܝܼܬ݂ ܠܵܐ ܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܘܝܲܬ݁ܝܼܪܵܐܝܼܬ݂ ܕܐܲܝܠܹܝܢ ܕܲܡܗܲܝܡܢܝܼܢ ܒܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟ݂܀ ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܣܓܵܐ ܥܲܡܲܢ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ ܫܠܵܡܵܐ ܥܲܡܲܢ.
