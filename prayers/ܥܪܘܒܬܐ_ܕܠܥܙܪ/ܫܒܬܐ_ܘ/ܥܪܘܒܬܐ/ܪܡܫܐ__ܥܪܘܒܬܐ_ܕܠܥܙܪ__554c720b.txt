ܕܡܓܗܝ ܥܪܘܒܬܐ ܕܫܬ
ܕܨܘܡܐ ܘܡܬܩܪܝܐ ܕܠܥܙܪ.
      ܕܐ݇ܚܖ̈ܵܝܹܐ
ܩܲܕ݂ܡܵܝܑܼܬ݂ ܡܫܲܪܸܐ ܪܲܒܵܐ: ܬܸܫܒ݁ܘܿܚܬܵܐ ܠܐܲܠܵܗܵܐ ܒܲܡܖ̈ܵܘܡܹܐ: ܬܢܝܼ: ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܘܥܲܠ ܐܲܪܥܵܐ ܫܠܵܡܵܐ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ: ܒܟ݂ܠ ܥܸܕܵܢ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܘܐܵܡ̇ܪܝܼܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܢܵܘܕܸ݁ܐ ܡܵܪܝ ܠܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܘܢܸܣܓ݁ܘܿܕ݂ ܠܡܵܪܘܼܬ݂ܵܟ݂ ܘܫܘܼܒ݂ܚܵܐ ܐܲܡܝܼܢܵܐ ܕܠܵܐ ܫܲܠܘܵܐ ܢܲܣܸܩ ܠܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ̣ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ: ܫܘܼܒ݂ܚܵܐ ܠܵܟ̣ ܐܲܠܵܗܵܐ. ܗܲܠܸܠܘܼܝܵܗ: ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ. ܘܐܵܡܲܪ ܫܲܡܵܫܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ. ܠܵܟ̣ ܡܲܠܟܵܐ ܕܡܲܠ̈ܟܹ݁ܐ ܣܵܓ݂݁ܕܝܼܢ ܠܹܗ. ܘܟܸܢܫܹ̈ܐ ܘܖܸ̈ܒ݁ܘܵܬ݂ܵܐ ܕܡܲܠܲܐܟܹ̈ܐ ܘܲܕ݂ܖ̈ܲܒܲܝ ܡܲܠܲܐܟܹ̈ܐ. ܒܕܸܚܠܵܐ ܘܒܲܪܬܹܝܬܼܵܐ ܩܵܝܡ݁ܝܼܢ ܘܲܡܫܲܡܫܝܼܢ ܘܲܡܙܲܝܚܝܼܢ ܠܐܝܼܩܵܪܵܐ ܣܓ݂ܝܼܕ݂ܵܐ ܕܪܲܒ݁ܘܼܬܹܗ. ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸܿܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ̣ܠܥܸܕܵܢ ܡܵܪܵܐ ܕܟ̣ܠ. ܐܲܒ݂ܵܐ . . .
ܗܘܼܠܵܠܵܐ. ܐܲܠܵܗܵܐ ܩ̣ܵܡ. ܚܙܝܼ: ܡܙܡܘܖ̈ܐ ܦܒ‐ܦܚ.
      ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܫܘܼܒ݂ܚܵܐ ܠܵܟ̣ ܐܲܠܵܗܵܐ ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ݁. ܬܢܝܼ. ܡܵܪܝܵܐ ܪܲܚܸܡܥܠܲܝܢ.
 ܘܡܲܩܦܝܼܢ ܬܸܫܒ݁ܘܿܚܬܵܐ. ܡܫܲܒܲܚܬ̇ ܡܵܪܲܢ ܘܠܵܟ̣ ܝܵܐܹܐ ܠܡܲܣܵܩܘܼ ܫܘܼܒ݂ܚܵܐ ܟܠܝܘܿܡ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ܀ ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ̣ܵܘܕܿܝܼܬܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ̣ ܦܘܼܡܲܢ ܘܝܲܗ̣݇ܒ݂ܠܲܢ ܕܲܢܗܲܠܸܠ ܒܬܸܫܒ̇ܚܵܬܹ̈ܗ. ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ̣ܵܘܕܿܝܼܬܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ̣ ܦܘܼܡܲܢ ܘܝܲܗ̣݇ܒ݂ܠܲܢ ܕܲܢܙܲܡܲܪ ܒܬܸܫܒ̇ܚܵܬܹ̈ܗ. ܫܘܼܒ݂ܚܵܐ ܠܹܗ ܠܲܡܫܝܼܚܵܐ ܘܬ̣ܵܘܕܿܝܼܬܵܐ ܠܡ̇ܢ ܕܲܦܬܲܚ̣ ܦܘܼܡܲܢ ܘܝܲܗ̣݇ܒ݂ܠܲܢ ܕܲܢܫܲܒܲܚ ܒܬܸܫܒ̇ܚܵܬܹ̈ܗ܀ ܠܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܫܘܼܒ݂ܚܵܐ ܢܲܣܸܩ ܠܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܬܢܝܼ ܓ: ܙܲܒ݂ܢܝܼ̈ܢ. ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܢܵܘܕܸܿܐ ܠܵܟ̣ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ̇ܘܼܬ̣ܵܟ̣܀ ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܲܢܩܲܠܣܵܟ̣ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ̇ܘܼܬ̣ܵܟ̣܀ ܠܵܐ ܣܵܦܹ݁ܩ ܦܘܼܡܲܢ ܕܲܢܫܲܒܚܵܟ̣ ܡܵܪܲܢ ܟܠ ܝܵܘܡܲܝ̈ ܚܲܝܲܝ̈ܢ ܡܸܛܠ ܛܲܝܒ̇ܘܼܬ̣ܵܟ̣܀ ܕܲܠܘܵܬ̣ ܡܵܝܘܿܬܹ̈ܐ ܡܵܪܵܐ ܡܪܲܚܡܵܢܵܐ. ܚܘܼܢܲܝܢ ܒܛܲܝܒ̇ܘܼܬ̣ܵܟ̣ ܘܐܸܬ̣ܪܲܚܲܡܥܠܲܝܢ܀ ܕܠܵܐ ܙܵܟܹ݁ܐ ܩܕ݂ܵܡܲܝܟ ܟܠ ܕܚܲܝ ܒܲܥܒ݂ܝܼܕܹ̈ܐ. ܐܲܢ݇ܬ̇ ܡܵܪܝ ܐܲܦܢ̣ܝܼܬܵܢ ܡ̣ܢ ܟܠܵܗ̇ ܛܘܼܥܝܲܝ. ܕܐܲܢ݇ܬ̇ܘܼ ܐܲܠܵܗܵܐ. ܘܠܵܟ̣ ܝܵܐܹ݁ܐ ܫܘܼܒ݂ܚܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ.
ܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ: ܘܥܲܠ ܐܲܦܲܝ̈ ܟܠܗܘܿܢ ܥܘܼܕ݂ܖ̈ܵܢܲܝܟ ܘܛܲܝ̈ܒ݁ܘܵܬ݂ܵܟ݂ ܕܲܠܘܵܬܲܢ ܕܠܵܐ ܡܸܬ݂ܦܲܪܥܝܼܢ: ܢܵܘܕܸ݁ܐ ܠܵܟ݂ ܘܲܢܫܲܒ݁ܚܵܟ݂ ܕܠܵܐ ܫܲܠܘܵܐ ܒܥܹܕ݁ܬܵܟ݂ ܡܟܲܠܲܠܬܵܐ ܡܲܠܝܲܬ݂ ܟܠ ܥܘܼܕ݂ܖ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܵܪܵܐ ܘܒ݂ܵܪܘܿܝܵܐ ܕܟ݂ܠ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.
ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܬܲܖ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹ݁ܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
(ܓܝܼܘܿܪܵܐ ܕܲܒ݂ܥܹܕ݇ܬܵܐ.( ܚܕ݂ܝܼܬ݂ ܟܲܕ݂ ܐܵܡ̇ܪܝܼܢ ܗ݇ܘ̣ܵܘ ܠܝܼ: ܠܒܲܝܬܹ݁ܗ ܕܡܵܪܝܵܐ ܐܵܙ̇ܠ݇ܝܼܢܲܢ(ܘܲܠܒܲܪ ܡ̣ܢ ܥܹܕ݁ܬܵܐ ܐܵܡ̇ܪܝܼܢ) ܒܟ݂ܠܕܘܼܟ ܐܝܼܬܲܝܟ ܐܲܠܵܗܵܐܩܲܒܸ݁ܠ ܬܲܟ݂ܫܲܦ̮ܬܲܢ. ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.
ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ:ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ.ܠܵܟ݂ܘܼ ܡܵܪܵܐ ܕܟ݂ܠܵܐ ܡܵܘܕܹܝܢܲܢ: ܘܠܵܟ݂ ܝܑܼܫܘܿܥ ܡܫܝܼܚܵܐ ܡܫܲܒ݁ܚܝܼܢܲܢ: ܕܐܲܢ݇ܬ݁ܘܼ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ.ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܡ.ܐܲܢ݇ܬ݁ܘܼ ܡܵܪܝ ܫܲܪܝܼܪܵܐܝܼܬ݂ ܡܢܲܚܡܵܢܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܐܲܢ݇ܬ݁ܘܼ ܦܵܪܘܿܩܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܘܲܡܢܲܛܪܵܢܵܐ ܐܲܡܝܼܢܵܐ ܕܚܲܝܲܝ̈ܢ: ܘܠܵܟ݂ ܡܵܪܝ ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
	      ܫܘܼܪܵܝܵܐ ܕܲܩܕ݂ܵܡ (ܡܙ؛ܩܡܕ؛ 1-12). ܐܹܪܲܡܪ̱ܡܵܟ݂ ܡܵܪܝ ܡܲܠܟܵܐ. ܗܲܠܸܠܘܿܝܵܗ ܗܲܠܸܠܘܿܝܵܗ ܐܹܗܲܠܸܠܘܿܝܵܗ. ܐܹܪܲܡܪ̱ܡܵܟ݂ ܡܵܪܝ ܡܲܠܟܵܐ* ܘܐܹܒܲܪܸܟ݂ ܫܡܵܟ݂ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ* ܒܟ݂ܠܝܘܿܡ ܐܹܒܲܪܟ݂ܵܟ݂ ܘܐܹܫܲܒܲܚ ܫܡܵܟ݂ ܠܥܵܠܲܡ* ܪܲܒܘܼ ܡܵܪܲܢ ܘܲܡܫܲܒܲܚ ܛܵܒ݂* ܘܠܲܝܬ݁ ܣܵܟ݂ܵܐ ܠܪܲܒ݁ܘܼܬܹܗ* ܕܵܪܵܐ ܠܕ݂ܵܪܵܐ ܢܸܫܬܲܥܘܿܢ ܥܒ݂ܵܕܲܝ̈ܟ* ܘܓܲܢ݇ܒܵܪܘܼܬ݂ܵܟ݂ ܢܚܵܘܘܿܢ* ܘܥܘܼܫܢܵܐ ܕܲܕ݂ܚܝܼܠܘܼܬ݂ܵܟ݂ ܢܹܐܡܪܘܼܢ* ܘܪܵܘ̱ܖ̈ܒ݂ܵܬ݂ܵܟ݂ ܢܸܫܬܲܥܘܿܢ* ܘܲܕ݂ܚܝܼܠܘܼܬ݂ ܥܘܼܫܢܵܟ݂ ܢܹܐܡܪܘܼܢ* ܘܪܲܒ݁ܘܼܬ݂ܵܟ݂ ܐܸܫܬܲܥܸܐ* ܕܘܼܟ݂ܪܵܢܵܐ ܕܛܲܝ̈ܒ݁ܘܵܬ݂ܵܟ݂ ܣܲܓ݁ܝ̈ܐܵܬ݂ܵܐ* ܫܲܒܲܚ* ܥܵܠܲܡ*
      ܘܲܦܪܘܿܫ ܥܘܿܢ̈ܝܵܬܼܵܐ ܠܠܵܥܵܙܲܪ. ܕܲܩܕ݂ܵܡ. ܒܦܘܼܫ ܒܲܫܠܵܡܵܐ. ܐܲܪܝܼܡ̣ ܩܵܠܹܗ ܘܙܵܥܲܬ̤ ܐܲܪܥܵܐ: ܘܥܸܠܵܝܵܐ ܝܲܗ݇ܒ݂ ܩܵܠܹܗ. ܐܲܝܟ̣ ܗܵܘ݁ ܩܵܠܵܐ ܕܨܹܝܕ݂ ܠܵܥܵܙܲܪ. ܢܵܦܹ݁ܠ ܩܵܠܹܗ ܕܡܲܚܸܐ ܡܝܼܬܹ̈ܐ. ܘܡܲܪܗܸܒ݂ ܘܲܡܙܝܼܥ ܠܡܵܘܬܵܐ ܝܲܥܢܵܐ. ܘܡܲܝܬܸ݁ܐ ܠܦܲܓ݂ܖܹ̈ܐ ܟܲܕ݂ ܠܵܐ ܡܚܲܒ݁ܠܝܼܢ: ܬܢܝܼ. ܘܫܲܒܲܚ. ܒܗܵܐ ܫ̣ܒ݂ܵܢܝ. ܩܵܠܹܗ ܕܲܒ݂ܪܵܐ ܫ̣ܡܲܥ ܠܲܥܵܙܲܪ ܕܡ̣ܢ ܒܹܝܬ݂ ܥܲܢܝܵܐ. ܘܲܥܢ̣ܵܐ ܘܐܸܡܲܪ ܕܗܵܐ ܐܸܢܵܐ. ܘܐܸܬ݁ܬ݁ܙܝܼܥܝܘ ܩܲܒ݂ܖܹ̈ܐ ܕܒܹܝܬ݂ ܥܲܢܝܼܕܹ̈ܐ. ܘܐܲܝܠܸܠ̣ ܡܵܘܬܵܐ ܘܲܖ̈ܥܸܠ̣ ܫܲܬ݂ܐ̱ܣܹ̈ܝ̈ܗ݁ ܕܲܫܝܘܿܠ. ܬܲܗܪܵܐ ܪܲܒܵܐ ܬܗܲܪ ܟܠ ܥܲܡܡ̈ܝܼܢ ܕܡܵܢܲܐ ܗ݇ܝܼ ܗܵܕܹܐ ܕܐܸܣܬܲܥܪܲܬ̤. ܕܚܲܝܵܐ ܠܡܝܼܬܼܵܐ ܗܵܐ ܩ̇ܵܪܹܐ. ܘܡܝܼܬܼܵܐ ܠܚܲܝܹ̈ܐ ܥܵܢܹ݁ܐ. ܘܗܵܝܕܹܝܢ ܪܸܡܙܵܟ̣ ܚܲܟܸ݁ܡ ܐܸܢܘܿܢ. ܕܗܵܢ̇ܵܘ ܝܼܫܘܿܥ ܒܲܪ ܕܵܘܝܼܕ݂. ܕܠܲܒ݂ܫܹܗ ܡܸܠܬܼܵܐ ܕܡ̣ܢ ܐܲܒ݂ܵܐ: ܘܥܲܒ݂ܕܹܗ ܡܵܪܵܐ ܘܕܲܝ̇ܵܢܵܐ. ܒܪܵܘܡܵܐ ܘܲܒ݂ܥܘܼܡܩܵܐ.
      ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܡܵܪܝܵܐ ܩܪܹܝ̇ܬ݂ܵܟ݂ ܒܓ݂ܘܼܕܵܐ ܕܒܼܵܬܲܪ: ܠܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ: ܘܠܲܒ݂ܛܝܼܠܘܼܬܹܗ ܕܨܸܒ݂ܝܵܢܵܟ݂ ܕܲܥܠܲܝܢ: ܚܲܝܵܒ݂ܝܼܢܲܢ ܕܢܵܘܕܸ݁ܐ ܘܢܸܣܓ݁ܘܿܕ݂ ܘܲܢܫܲܒܲܚ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹܝܢ: ܐܵܡܹܝܢ.

ܨܠܘܿܬ݂ܵܐ ܕܫܘܼܪܵܝܵܐ ܕܒ݂ܵܬܲܪ: ܒܓ݂ܘܼܕܵܐ ܕܒܼܵܬܲܪ. ܫܡܲܥ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܒܲܚܢܵܢܵܟ݂: ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܒܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂: ܘܚܘܼܢ ܠܚܲܛܵܝܘܼܬܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܖ̈ܲܚܡܲܝܟ: ܐܵܣܝܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ ܘܣܲܒ݂ܪܵܐ ܛܵܒ݂ܵܐ ܕܢܲܦ̮̈ܫܵܬܲܢ: ܡܵܪܵܐ ܕܟ݂ܠ.

ܘܡܲܩܸܦ: ܐܲܒ݂ܵܐ ܕܖ̈ܲܚܡܹܐ ܘܐܲܠܵܗܵܐ ܕܟ݂ܠܹܗ ܒܘܼܝܵܐܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܘܥܵܢܹ̇ܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܦܵܪܘܿܩܲܢ ܘܝܲܨܘܼܦܬ݂ܵܢܲܢ ܘܲܡܦܲܪܢ̱ܣܵܢܵܐ ܕܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܫܲܝܢܵܐ ܘܐܵܘܝܘܼܬ݂ܵܐ ܘܩܘܼܝܵܡܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ ܘܲܕ݂ܟ݂ܠܗܹܝܢ ܥܹ̈ܕ݂ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܐܲܬ݂ܪܲܢ ܘܥܲܠ ܟܠ ܐܲܬ݂ܖ̈ܵܘܵܢ: ܘܥܲܠ ܐܲܝܠܹܝܢ ܕܲܒ݂ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܥܵܡ̇ܪܝܼܢ ܒܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܡܘܼܙܵܓ݂ܵܘ̈ܗܝ ܕܐܵܐܲܪ ܘܟܲܗܝܼܢܘܼܬ݂ܵܐ ܕܫܲܢ݇ܬܵܐ ܘܥܲܠ̈ܠܵܬ݂ܵܐ ܕܦܹܐܖܹ̈ܐ ܘܥܲܠ ܬܘܼܩܵܢܵܐ ܕܟ݂ܠܹܗ ܥܵܠܡܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܡܲܠܟܹ̈ܐ ܐܲܚܝܼܕܲܝ̈ ܫܘܼܠܛܵܢܹܗ ܕܥܵܠܡܵܐ ܗܵܢܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ̇ܘ ܕܲܒ݂ܖ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ݂ܠ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܥܲܠ ܩܲܫܝܼܫܹ̈ܐ ܘܲܡܫܲܡܫܵܢܹ̈ܐ ܐܲܖ̈ܬܵܕ݁ܘܿܟ݁ܣܹܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ܗ̇ܘ ܕܥܲܬ݁ܝܼܪ ܒܖ̈ܲܚܡܵܘܗܝ ܘܲܫܦܝܼܥ ܒܲܚܢܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܡ̣ܢ ܩܕ݂ܵܡ ܥܵܠܡܹ̈ܐ ܐܝܼܬ݂ܵܘܗܝ ܘܲܠܕ݂ܵܪܕܵܖ̈ܝܼܢ ܩܲܝܵܡ ܫܘܼܠܛܵܢܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܒܲܟ݂ܝܵܢܹܗ ܐܝܼܬ݂ܵܘܗܝ ܛܵܒ݂ܵܐ ܘܝܵܗܘܿܒ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܛܵܒ݂̈ܵܬ݂ܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܠܵܐ ܨܵܒܹܐ ܒܡܵܘܬܹ݁ܗ ܕܚܲܛܵܝܵܐ ܐܸܠܵܐ ܕܲܢܬ݂ܘܼܒ݂ ܡ̣ܢ ܥܵܘ̣ܠܹܗ ܘܢܸܚܸܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒܲܐܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܗ̇ܘ ܕܲܒ݂ܝܲܠܕܹ݁ܗ ܐܵܘ ܕܲܒ݂ܕܸܢܚܹܗ ܐܵܘ ܕܲܒ݂ܨܵܘܡܹܗ ܐܵܘ ܕܲܒ݂ܡܲܥܲܠܬܹ݁ܗ ܐܵܘ ܕܒܲܩܝܵܡܬܹ݁ܗ ܐܵܘ ܕܲܒ݂ܣܘܼܠܵܩܹܗ ܐܵܘ ܕܲܒ݂ܡܲܚܲܬ݁ܬܹ݁ܗ ܐܵܘ ܕܒܲܨܠܝܼܒܹܗ: ܩܲܕ݁ܝܼܫܵܐ (ܐܵܘ ܩܲܕܝܼܫܬܵܐ) ܚܲܕ݁ܝܼ ܠܐܲܪܥܵܐ ܘܐܲܦܨܲܚ ܠܲܫܡܲܝܵܐ: ܒܵܥܹܝܢܲܢ ܡ̣ܢܟ݂܀ ܗ̇ܘ ܕܠܹܗ ܗܘ̤ ܟܝܵܢܵܐܝܼܬ݂ ܐܝܼܬ݂ ܠܵܐ ܡܵܝܘܿܬ݂ܘܼܬ݂ܵܐ ܘܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܦܵܪܘܿܩܵܐ ܕܟ݂ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܘܝܲܬ݁ܝܼܪܵܐܝܼܬ݂ ܕܐܲܝܠܹܝܢ ܕܲܡܗܲܝܡܢܝܼܢ ܒܹܗ: ܒܵܥܹܝ̇ܢܲܢ ܡܸܢܵܟܼ܀ ܦܪܘܿܩ ܠܟ݂ܠܲܢ ܡܫܝܼܚܵܐ ܡܵܪܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܐܲܣܓܵܐ ܥܲܡܲܢ ܫܲܝܢܵܟ݂ ܘܲܫܠܵܡܵܟ݂: ܘܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀ ܘܥܵܢܹܐ ܥܲܡܵܐ: ܡܵܪܲܢ ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ܀
 ܘܡܵܐ ܕܫܸܠܡܲܬ̤ ܡܲܩܸܦ: ܢܲܦ̮̈ܫܵܬܲܢ ܘܕܲܚܕ݂ܵܕܹ̈ܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܲܓ݂ܥܸܠ.
ܘܥܵܢܹܝ̇ܢ: ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܲܢ.
	ܘܲܡܨܲܠܸܐ ܟܵܗܢܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ. ܠܵܟ݂ ܡܵܪܝܵܐ ܐܲܠܵܗܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ ܡܲܓ݂ܥ̱ܠܝܼܢܲܢ ܦܲܓ݂ܖ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ: ܘܡ̣ܢܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܫܵܐ̇ܠܝܼܢܲܢ ܫܘܼܒ݂ܩܵܢܵܐ ܕܚܵܘܒܹ̈ܐ ܘܕܲܚܛܵܗܹ̈ܐ: ܗܲܒ݂ܠܲܢ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܲܒ݂ܖ̈ܲܚܡܲܝܟ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ ܟܵܪܘܿܙܘܼܬ݂ܵܐ: ܐܲܪܝܼܡܘ ܩܵܠܟ݂ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ.
ܘܐܵܡ̇ܪܝܼܢ ܒܩܵܠܵܐ ܦܫܝܼܛܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܫܲܒܲܚ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܥܵܠܲܡ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܒ݂ܵܬܲܪ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ ܒܓ݂ܘܼܕܵܐ ܕܒܼܵܬܲܪ. ܩܲܕ݁ܝܼܫܵܐ ܘܲܫܒ݂ܝܼܚܵܐ ܘܚܲܝܠ̱ܬ݂ܵܢܵܐ ܘܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܕܲܒ݂ܩܲܕ݁ܝܼܫܹ̈ܐ ܥܵܡܲܪ ܘܡܸܬ݁ܬ݁ܢܝܼܚ ܨܸܒ݂ܝܵܢܹܗ: ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ ܐܸܬ݂ܦܲܢ ܡܵܪܝ: ܘܚܘܼܣ ܘܪܲܚܸܡ ܥܠܲܝܢ ܐܲܝܟ݂ ܕܲܡܥܵܕܲܬ݁ ܒܟ݂ܠ ܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ ܕܐܲܟ݂ܪܸܙ. ܒܲܪܸܟ݂ ܡܵܪܝ: ܐܲܪܟܸ݁ܢܘ ܖܹ̈ܫܲܝܟ݁ܘܿܢ ܠܲܣܝܵܡܝܼܕ݂ܵܐ ܘܩܲܒܸ݁ܠܘ ܒܘܼܪܟ݁ܬ݂ܵܐ.
ܘܐܵܡܲܪ ܪܲܒܵܐ. ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܬܸܫܡܸܫܬܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
	ܘܡܲܩܸܦ ܟܵܗܢܵܐ ܒܓ݂ܘܼܕܵܐ ܕܲܩܕ݂ܵܡ ܨܠܘܿܬ݂ܵܐ. ܘܟܲܕ݂ ܫܵܠ̈ܡܵܢ ܢܲܦ̮̈ܫܵܬܲܢ ܠܲܚܕ݂ܵܐ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܓܡܝܼܪܬܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܟܠܲܢ ܒܲܚܕ݂ܵܐ ܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ: ܢܸܫܬ݁ܘܸܐ ܕܢܲܣܸܩ ܠܵܟ݂ ܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ݁ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ݁ܬ݂ܵܐ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ... ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
      ܥܘܿܢܝܼܬܼܵܐ ܕܪܲܡܫܵܐ. ܒܫܘܼܪܵܝ ܫܠܝܼܚܹ̈ܐ ܓܒܲܝ̈ܵܐ ܫܡܲܥܘ ܗܵܕܹܐ ܟܠܟ̣ܘܿܢ ܥܲܡܡܹ̈ܐ: ܐܸܬ݂ܟܲܪܗܲܬ̤ ܘܐܲܢ݇ܬ݁ ܬܲܩܸܢ̣ܬܵܝܗ݁: ܬܸܬ݂ܟ݁ܬܸܒ݂ ܗܵܕܹܐ ܠܕ݂ܵܪܵܐ ܐ݇ܚܪܹܢܵܐ. ܗܵܝ݁ ܙܵܢܝܼܬܼܵܐ ܕܠܘܼܩܕ݂ܵܡ ܐܝܼܬܹܝܗ݁ ܗ݇ܘ̣ܵܬ݂ ܐܲܝܟ̣ ܒܹܝܬ݂ ܡܲܥ̱ܡܪܵܐ ܕܟ̣ܠ ܒܝܼ̈ܫܵܬܼܵܐ. ܘܟܸܢ ܗܘ̤ܵܬ݂ ܡܩܲܒ݁ܠܵܢܝܼܬܼܵܐ ܕܲܬ݂ܝܵܒ݂ܘܼܬ݂ܵܐ. ܗܵܝ݁ ܕܹܝܢ ܕܡܸܬ݁ܬܲܓ݁ܪܵܐ ܗ݇ܘ̣ܵܬ݂ ܒܟ̣ܠܗܘܿܢ ܚܵܘܒܹ݁ܝܗ݁. ܗܸܦܟܲܬ̤ ܠܵܗ݁ ܦܘܼܪܩܵܢܵܐ ܠܢܲܦ̮ܫܵܗ݁. ܘܲܒ݂ܪܸܡܙܵܐ ܕܥܲܝܢܹܝ̈ܗ݁ ܙܲܠܝܼ̈ܠܵܬܼܵܐ. ܟܲܢܫܲܬ̤ ܠܲܩܢܘܿܡܵܗ݁ ܟܠ ܚܘܼܣܖ̈ܵܢܝܼܢ. ܘܡܸܢܗܹܝܢ ܫܲܦܥܲܬ̤ ܕܸܡܥܹ̈ܐ ܚܢܝܼ̈ܓ݂ܵܬܼܵܐ. ܘܐܲܫܝܼܓܲܬ̤ ܠܖܸ̈ܓ݂ܠܲܝܟ ܩܲܕܿܝܼ̈ܫܵܬܼܵܐ. ܘܲܒ݂ܡܸܫܚܵܐ ܘܲܒ݂ܒ݂ܘܼܣܵܡܹ̈ܐ ܠܲܒ݂ܢܲܝ̈ܢܵܫܵܐ ܨܵܝ݁ܕܵܐܗ݇ܘ̣ܵܬ݂. ܘܲܒ݂ܣܘܼܛܡܹ̈ܐ ܕܥܵܘ̣ܠܵܐ ܟܠܫܵܥ ܐܵܣ݁ܪܵܐ ܗ݇ܘ̣ܵܬ݂. ܡܸܫܚܵܐ ܝܲܩܝܼܪܵܐ ܡܸܫܚܲܬ̤ ܠܵܟ̣ ܡܵܪܝ ܕܐܸܬ݂ܢ̱ܣܸܒ݂ܬ݁ ܡܸܛܠ ܦܘܼܪܩܵܢܲܢ. ܘܥܲܡܵܗ݁ ܠܵܟ̣ܘܼ ܩ̇ܵܪܹܝܢܲܢ ܕܐܲܢ݇ܬ݁ܘܼ ܝܵܕܲܥ݁ ܐܲܢ݇ܬ݁ ܟܠܗܘܿܢ ܚܲܫܲܝ̈ܢ ܡܵܪܹܟ݁ܠ ܫܘܼܒ݂ܚܵܠܵܟ̣: ܬܢܝܼ. ܫܲܒܲܚ. ܒܵܗ݁.
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	ܨܠܘܿܬ݂ܵܐ ܕܡܸܬ݂ܐܲܡܪܵܐ ܒܟ݂ܠܗܹܝܢ ܥܖ̈ܘܼܒ݂ܵܬ݂ܵܐ ܕܟ݂ܠܵܗ̇ ܫܲܢ݇ܬܵܐ: ܣܛܲܪ ܡܸܢ ܕܲܒ݂ܕ݂ܘܼܟ݂ܖ̈ܵܢܹܐ. ܢܲܚܸܡ ܡܵܪܝ ܥܲܢܝܼܕܲܝ̈ܢ ܒܲܚܢܵܢܵܟ݂: ܘܐܲܩܝܼܡ ܐܸܢܘܿܢ ܡܸܢ ܝܲܡܝܼܢܵܟ݂: ܘܐܲܠܒܸ݁ܫ ܐܸܢܘܿܢ ܫܘܼܒ݂ܚܵܐ ܦܲܐܝܵܐ ܒܡܲܠܟ݁ܘܼܬ݂ܵܟ݂: ܘܚܲܠܸܛ ܐܸܢܘܿܢ ܥܲܡ ܟܹܐܢܹ̈ܐ ܘܙܲܕ݁ܝܼܩܹ̈ܐ ܓܵܡܖ̈ܲܝ ܨܸܒ݂ܝܵܢܵܟ݂ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ: ܡܵܪܵܐ ܕܡܵܘܬܲܢ ܘܲܕ݂ܚܲܝܲܝ̈ܢ: ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܠܥܵܠܡܝܼܢ. ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ.
      ܫܘܼܪܵܝܵܐ. ܒܢܸܩܦܲܬ̤܀ ܚܛܵܗܲܝ̈ ܐܵܘܕܲܥܬܵܟ̣ ܘܣܲܟ̣̈ܠ̱ܘܵܬ݂ܝ ܠܵܐ ܟܲܣܝܼ݁ܬ݂ ܡܸܢܵܟ̣* ܘܐܸܡ݁ܪܹܬ݂ ܕܐܵܘܕܸ݁ܐ ܠܡܵܪܝܵܐ ܥܲܠ ܣܲܟ̣̈ܠ̱ܘܵܬ݂ܝ* ܐܲܢ݇ܬ݁ ܫܵܒܹ݁ܩ ܐܲܢ݇ܬ݁ ܠܝܼ ܟܠܗܘܿܢ ܚܛܵܗܲܝ̈* ܡܸܛܠ ܗܵܢܵܐ ܢܨܲܠܸܐ ܠܵܟ̣  ܟܠܡ݁ܢ ܕܲܓ̣ܒܸܐ ܠܵܟ̣ ܒܙܲܒ݂ܢܵܐ ܡܩܲܒ݁ܠܵܐ.
ܗܵܝܕܹ݁ܝܢ ܐܵܡ̇ܪܝܼܢ ܬܲܖ̈ܬܲܝܗܹܝܢ ܓܘܼܕܹ̈ܐ ܐܲܟ݂ܚ݇ܕ݂ܵܐ. ܗܲܠܸܠܘܼܝܵܗ ܗܲܠܸܠܘܼܝܵܗ ܐܹܗܲܠܸܠܘܼܝܵܗ.

      ܕܒܹܝܡ. ܒܪܝܼܟ̣ ܕܲܒ݂ܩܸܛܠܹܗ ܩܲܛ̣ܠܹܗ ܠܡܵܘܬܵܐ. ܘܐܲܚ̣ܝܼ ܠܐܵܕ݂ܵܡ ܒܐ݇ܖ̈ܵܙܵܘܗܝ ܫܒ݂ܝܼܚܹ̈ܐ. ܒܵܬܹ̈ܐ. ܢܵܘܣܹ̈ܐ ܕܟܲܝ̈ܵܐ ܥܲܒ݂ܕܲܢ ܡܵܪܲܢ. ܢܫܝܼܓ݂ ܪܵܥܝܵܢܲܢ ܡ̣ܢ ܣܲܡ ܡܵܘܬܵܐ.

ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
	 ܨܠܘܿܬ݂ܵܐ ܕܲܩܕ݂ܵܡ ܕܣܘܼܒܵܥܵܐ ܒܓ݂ܘܼܕܵܐ ܕܒܼܵܬܲܪ. ܐܲܫܘܵܐ ܠܲܢ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܠܪܲܡܫܵܐ ܕܫܲܝܢܵܐ: ܘܲܠܠܸܠܝܵܐ ܕܲܢܝܵܚܵܐ: ܘܲܠܨܲܦܪܵܐ ܕܲܡܣܲܒ݁ܖ̈ܵܢ ܒܹܗ ܛܵܒ̣̈ܵܬ݂ܵܐ: ܘܠܐܝܼܡܵܡܵܐ ܕܲܥܒ݂ܵܕܹ̈ܐ ܛܵܒܹ̈ܐ ܕܙܲܕܿܝܼܩܘܼܬ݂ܵܐ: ܕܲܒ݂ܗܘܿܢ ܢܪܲܥܸܐ ܠܐܲܠܵܗܘܼܬ݂ܵܟ̣ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ̣ܠ.
      ܥܘܿܢ̈ܝܵܬܼܵܐ. ܒܩܵܠ ܣܵܗܕܵܘ̈ܗܝ ܕܲܒ݂ܪܵܐ. ܕܢܲܚ̣ ܢܘܼܗܪܵܐ ܒܚܸܫܘܿܟ̣ܵܐ ܠܲܬ݂ܖ̈ܝܼܨܹܐ: ܠܐܲܝܠܹܝܢ ܕܲܬ݂ܪܝܼܨܝܼܢ ܒܠܸܒ݁ܗܘܿܢ. ܢܘܼܗܪܵܐ ܕܢܲܗܝܼܪ ܡ̣ܢ ܫܸܡܫܵܐ. ܠܒ݂ܘܼܫܹ̈ܐ ܕܲܦܐܹܝܢ ܡ̣ܢ ܬܲܠܓܵܐ: ܠܵܒ݂ܫܝܼܢ ܒܲܥܕܸܢ ܙܲܕܿܝܼܩܹ̈ܐ. ܘܚܵܕܹ݁ܝܢ ܘܕ݂ܵܝܨ݁ܝܼܢ ܘܲܦܨܝܼܚܝܼܢ. ܒܹܝܬ̣ ܐܝܼܠܵܢܵܘ̈ܗܝ ܕܦܲܪܕܲܝܣܵܐ. ܡܵܐ ܕܫܵܡ݁ܥܝܼܢ ܩܵܠܹܗ ܕܲܒ݂ܪܵܐ: ܬܢܝܼ. ܐܲܩܸܦ. ܒܠܐܲܝܟܵܐ ܡܵܪܲܢ. ܬܵܘ ܒܢܲܝ̈ܵܐ ܫܘܼܡܥܘܼܢܝ. ܢܝܼܪܵܐ ܗ݇ܘ̣ ܚܲܠܝܵܐ ܨܵܘܡܵܐ ܕܲܟ̣ܝܵܐ. ܕܢ̇ܵܓܹ݁ܕ݂ ܠܪܵܘܡܵܐ ܠܐܲܬ݂ܪܵܐ ܕܓܹܐ̈ܘܵܬ݂ܵܐ. ܘܙܲܝܢܵܐ ܗ݇ܘ̣ ܟܲܣܝܵܐ ܚܫܝܼܠ ܡ̣ܢ ܪܘܼܚܵܐ ܕܲܒ݂ܝܲܡܝܼܢܵܐ ܘܣܸܡܵܠܵܐ ܡܲܩܪܸܒ݂. ܒܹܗ ܢܸܬ݂ܓܲܢ݇ܒܲܪ ܥܲܠ ܚܲܫܹ̈ܐ ܘܥܲܠܚܘܼܬ݂ܚܵܬܹ̈ܐ. ܘܢܸܗܘܸܐ ܢܵܘܣܹ̈ܐ ܠܪܘܼܚܩܘܼܕ݂ܫܵܐ. ܡܸܛܠ ܗܵܢܵܐ ܢܨܲܠܸܐܠܵܟ̣ ܥܲܠ ܙܲܒ݂ܢܵܐ ܡܩܲܒ݁ܠܵܐ. ܒܙܲܒ݂ܢܵܐ ܡܩܲܒ݁ܠܵܐ ܕܨܵܘܡܵܐ ܙܲܗܝܵܐ ܢܸܦܠܘܿܚ ܒܕܲܟ̣ܝܘܼ ܩܕ݂ܵܡܵܘܗܝ ܕܲܡܫܝܼܚܵܐ. ܕܢܸܫܬ݁ܘܸܐ ܕܢܸܥܘܿܠ ܠܗܵܘ݁ ܒܘܼܣܵܡܵܐ ܕܲܠܨܲܝ̇ܵܡܹ̈ܐ ܕܡ̣ܢ ܥܵܘ̣ܠܵܐ ܢܛܝܼܪ ܗ݇ܘ̣. ܘܢܸܙܡܲܪ ܥܲܡܗܘܿܢ ܫܘܼܒ݂ܚܵܐ ܩܕ݂ܵܡܘܵܜ ܕܲܡܫܝܼܚܵܐ. ܠܚܲܝܠܵܐ ܕܙܲܟ݁ܝܹܗ ܠܲܟ̣ܝܵܢܲܢ. ܒܝܵܘ̈ܡܵܬܼܵܐ ܩܲܕ݂ܡܵܝܹ̈ܐ. ܒܟ̣ܠܗܘܿܢ ܕܵܖܹ̈ܐ ܨܵܡ̣ܘ ܗ݇ܘ̣ܵܘ ܟܹܐܢܹ̈ܐ. ܘܲܥ̣ܢܲܝܬ݁ ܐܸܢܘܿܢ ܪܵܚܹ݁ܡ ܬܲܝ̇ܵܒܹ̈ܐ. ܨ̣ܵܡ ܕܵܢܝܼܐܹܝܠ ܕܒܹܝܬ݂ ܚܲܢܲܢܝܵܐ. ܐܵܦ ܚܹܙܲܩܝܵܐ ܘܲܦܪܲܩ̣ܬ݁ ܚܲܝܲܝ̈ܗܘܿܢ. ܘܲܚܢܲܢ ܡܵܪܲܢ ܕܲܩܕ݂ܵܡܲܝܟ ܗܵܐ ܩܵܝܡ݁ܝܼܢܲܢ. ܦܲܢܵܐ ܒܖ̈ܲܚܡܲܝܟܫܹܐ̈ܠܵܬܲܢ. ܐܲܩܸܦ. ܪܹܫܩܵܠܵܐ. ܠܐܲܝܟܵܐ ܐܹܙܲܠ ܡ̣ܢ ܩܕ݂ܵܡܲܝܟ. ܠܐܲܝܟܵܐ ܡܵܪܲܢ ܐܸܥܪܘܿܩ ܡܸܢܵܟ̣. ܘܒܲܐܝܢܵܐ ܐܲܬ݂ܪܵܐ ܐܸܛܫܸܐ ܡ̣ܢ ܩܘܼܕ݂ܡܲܝܟ. ܫܡܲܝܵܐ ܟܘܼܪܣܝܵܟ̣ ܘܐܲܪܥܵܐ ܐܲܝܟ̣ ܕܲܡܥܵܕ݂ܵܐ ܗ݇ܝܼ ܛܲܝܒ݁ܘܼܬ݂ܵܟ̣. ܢ̇ܵܚܬܲܝ̈ ܝܲܡܵܐ ܒܐܸܠܦܹ̈ܐ. ܐܲܝܟ̣ ܬܲܓܵܖܹ̈ܐ ܕܲܒ݂ܓ݂ܵܘ ܝܲܡܵܐ ܪܵܕܹ݁ܝܢܲܢ ܒܹܗ ܒܗܵܢ ܥܵܠܡܵܐ ܕܥܵܒܲܪ. ܘܡܵܐ ܕܡܲܛܝܼܢܲܢ ܕܢܸܚܙܘܿܩ ܡܸܢܹܗ. ܕܲܛܥܝܼܢ ܚܵܕܹ݁ܐ ܘܕܲܣܦܝܼܩ ܡܸܬ݁ܬܢܲܚ. ܘܛܘܼܒ݂ܵܘܗܝ ܡܗܲܝܡܲܢܬܵܐ. ܕܡܸܛܠܵܬܹܟ̣ܝ ܗܵܐ ܥܵܒܹ݁ܕ݂ ܐ݇ܢܵܐ ܚܢܵܢܵܐ ܘܖ̈ܲܚܡܹܐ ܥܲܠ ܟܠܵܗ݁ ܒܪܝܼܬܼܵܐ. ܘܐܸܢܗ̤ܘ ܕܬ݂ܵܝܒ݁ܝܼܢ ܚܲܛܵܝܹ̈ܐ ܘܐܵܬܹ݁ܝܢ ܨܹܐܕܲܝ. ܥܵܒܹ݁ܕ݂ ܐ݇ܢܵܐ ܥܠܲܝܗܘܿܢ ܖ̈ܲܚܡܹܐ. ܒܲܥܓܲܠ ܢܩܲܕ݂ܡܘܼܢܵܢ ܖ̈ܲܚܡܲܝܟ. ܐܸܢܗ̤ܘ ܕܡܲܣܟܸ݁ܠ ܥܲܒ݂ܕܵܐ ܒܡܵܪܹܗ ܒܖ̈ܲܚܡܹܐ ܕܡܵܪܹܗ ܐܵܙܹ݁ܠ ܘܡܸܬ݂ܓܵܘܲܣ. ܘܲܒ݂ܒ݂ܵܥܘܼܬ݂ܵܐ ܘܲܒ݂ܬܲܟ̣ܫܲܦ̮ܬܵܐ ܫܵܒܹ݁ܩ ܡܵܪܵܐ ܣܲܟ̣ܠܘܼܬ݂ܵܐ ܕܥܲܒ݂ܕܹܗ. ܘܲܚܢܲܢ ܡܵܪܲܢ ܕܒܲܨܠܝܼܒ݂ܵܟ̣ ܐܸܬ݂ܓܵܘܲܣܢܲܢ. ܒܲܥܓܲܠ ܢܩܲܕ݂ܡܘܼܢܵܢ ܖ̈ܲܚܡܲܝܟ. ܠܵܐ ܬܸܫܕܹܝܢܝ ܡ̣ܢ ܩܕ݂݂ܵܡܲܝܟ. ܒܙܲܒ݂ܢܹ̈ܐ ܐ݇ܚܖ̈ܵܝܹܐ ܡܵܐ ܕܡܸܬ݂ܒܲܚܪܝܼܢ ܥܒ݂ܵܕܹ̈ܐ ܕܟ̣ܠܢܵܫ ܩܕ݂ܵܡܲܝܟ ܚܲܢܵܢܵܐ. ܠܵܐ ܡܵܪܝ ܬܸܫܕܸܿܐ ܡ̣ܢ ܩܕ݂ܵܡ ܐܲܦܲܝ̈ܟ. ܠܐܲܝܠܹܝܢ ܕܐܵܘܕܝܼܘ ܒܲܫܡܵܟ̣ ܩܲܕܿܝܼܫܵܐ. ܐܲܒ݂ܵܐ ܘܲܒ݂ܪܵܐ ܘܪܘܼܚܩܘܼܕ݂ܫܵܐ ܦܵܪܲܩܠܹܝܛܵܐ. ܦܪܘܿܩܲܝܢ ܘܢܲܛܲܪ ܢܲܦ̮ܫ̈ܵܬܲܢ. ܘܐܸܡܲܪ ܕܬ݂ܘܼܒ݂ܘ ܒܢܲܝ̈ܢܵܫܵܐ. ܐܲܟ̣ܪܸܙ ܝܵܘܢܵܢ ܒܹܝܬ݂ ܢܝܼܢܘܵܝܹ̈ܐ ܩܵܠܹ̈ܐ ܚܢܝܼ̈ܓܹܐ ܕܬ݂ܘܼܒ݂ܘ ܡ̣ܢ ܚܵܘܒܲܝ̈ܟ݁ܘܿܢ. ܘܲܓ݂ܙܲܪܘ ܨܵܘܡܵܐ ܘܲܠܒܸܫ̣ܘ ܣܲܩܹ̈ܐ. ܘܲܚ̣ܣܲܠܘ ܛܠܵܝܹ̈ܐ ܘܐܸܡܖܹ̈ܐ ܡ̣ܢ ܚܲܠܒ݂ܵܐ. ܘܲܚܢܲܢ ܐܲܚܲܝ̈ ܢܸܒ݂ܥܸܐ ܖ̈ܲܚܡܹܐ ܡ̣ܢ ܡܵܪܲܢ. ܕܲܡܪܲܚܡܵܢܵܐ ܗ݇ܘ̣ ܐܲܠܵܗܵܐ. ܙܲܕܝܼܩܵܐ ܘܲܬ݂ܪܝܼܨܵܐ. ܨ̣ܵܡ ܕܵܢܝܼܐܹܝܠ ܒܐܲܪܥܵܐ ܕܒ݂ܵܒܹܝܠ. ܕܒܹܝܬ݂ ܚܲܢܲܢܝܵܐ ܒܐܲܪܥܵܐ ܕܟܲܠܕܵܝܹ̈ܐ. ܘܣܲܟܲܪ ܦܘܼܡܵܐ ܕܐܲܖ̈ܝܵܘܵܬ݂ܵܐ ܘܕܲܥܸܟ̣ܘ ܚܲܝܠܵܐ ܕܢܘܼܪܵܐ ܝܵܩܹ݁ܕܬܵܐ. ܨܵܘܡܹܗ ܕܡܵܪܲܢ ܣܲܟ݁ܪܹܗ ܠܦܘܼܡܹܗ ܕܣܵܛܵܢܵܐ. ܘܐܲܒ݂ܗ̱ܬܹܗ ܠܲܒ݂ܥܸܠܕܲܪܵܐ. ܡܸܪܕܵܐ ܪܕ݂ܵܢ̣ܝ ܡܵܪܝܵܐ ܘܲܠܡܵܘܬܵܐ ܠܵܐ ܐܲܫܠ̱ܡܲܢܝ. ܪܕܲܝ̣ܬܵܢ ܡܵܪܲܢ ܡܸܛܠ ܖ̈ܲܚܡܲܝܟ. ܕܲܚܢܲܢ ܢܹܐܪܲܒ݂ ܘܐܲܢ݇ܬ݁ ܡܵܪܝ ܬܸܫܬܲܒܲܚ. ܠܵܐ ܡܵܪܝ ܬܸܪܓܲܙ ܡܸܛܠ ܥܵܘ̣ܠܲܢ. ܕܐܲܢ݇ܬ݁ ܚܲܢܵܢܵܐ ܐܲܢ݇ܬ݁ ܘܠܵܐ ܡܬ݂ܘܿܡ ܪܵܓܹ݁ܙ ܐܲܢ݇ܬ݁. ܐܸܠܵܐ ܡܵܪܝ ܐܲܫܦܲܥ ܖ̈ܲܚܡܲܝܟ ܥܲܠ ܟܠܲܢ. ܒܖ̈ܲܚܡܵܘܗܝ ܕܲܒ݂ܪܵܟ̣ ܚܲܒ݁ܝܼܒ݂ܵܐ. ܗܵܝܕܹܝܢ ܢܹܐܡܪܘܼܢ ܒܹܝܬ݂ ܥܲܡܡܹ̈ܐ. ܛܘܼܒܲܝܟ݁ܝ ܡܲܪܝܲܡ ܡ̣ܢ ܟܠ ܫܲܖ̈ܒ݂ܵܢ. ܘܛܘܼܒܲܝܟ݁ܝ ܢܸܩܥܘܿܢ ܫܡܲܝ̈ܵܐ ܥܲܡ ܐܲܪܥܵܐ. ܕܡ̣ܢ ܓܵܘ ܥܘܼܒܹ݁ܟ̣ܝ ܕܢܲܚ̣ ܠܲܢ ܢܘܼܗܪܵܐ ܕܒܹܗ ܐܸܫܬܲܪܝܲܬ̤ ܛܘܼܥܝܲܝ ܕܚܸܫܘܿܟ̣ܵܐ. ܨܠܘܿܬܹܟ̣ܝ ܬܸܗܘܸܐ ܠܢܲܦ̮ܫ̈ܵܬܲܢ ܒܟ̣ܠ ܥܸܕܵܢܝܼ̈ܢ ܫܘܼܪܵܐ ܪܵܡܵܐ ܕܠܵܐ ܡܸܬ݂ܚܣܸܢ. ܒܟ̣ܠܵܗ݁ ܐܲܪܥܵܐ ܢܸܦܩܲܬ̤ ܣܒܲܪܬ݂ܗܘܿܢ. ܒܟ̣ܠܗܘܿܢ ܣܵܘܦܹ̈ܐ ܕܐܲܪܥܵܐ ܟܠܵܗ݁ ܐܲܟ̣ܪܸܙ̣ܘ ܫܠܝܼܚܹ̈ܐ ܣܒܲܪܬܹܗ ܕܲܡܫܝܼܚܵܐ. ܘܐܲܝܟ̣ ܡܲܒ݁ܘܼܥܹ̈ܐ ܐܲܪܕܝܼܘ ܚܲܝܹ̈ܐ ܘܐܲܫ̣ܩܝܼܘ ܠܢܲܦ̮ܫ̈ܵܬܼܵܐ ܕܲܒ݂ܢܲܝ̈ܢܵܫܵܐ. ܘܗܵܐ ܡܸܬ݂ܒܲܣܡܝܼܢ ܒܗܵܢ݁ܘܿܢ ܛܘܼܒܹ̈ܐ ܫܡܲܝܵܢܹ̈ܐ. ܕܥܲܝܢܵܐ ܠܵܐ ܚܙܵܬ̤ ܐܲܝܟ̣ ܕܲܟ̣ܬ݂ܝܼܒ݂ ܒܲܫܡܲܝܵܐ ܘܒܲܐܪܥܵܐ. ܗܵܐ ܒܓ݂ܵܘ ܥܹܕܿܬܵܐ ܥܲܡ ܩܲܕܝܼܫܹ̈ܐ ܪܫܝܼܡ ܕܘܼܟ̣ܪܵܢܵܟ̣ ܐܲܒ݂ܘܼܢ ܩܲܕܝܼܫܵܐ. ܘܒܲܝܢܵܬ݂ ܓܘܼܕܹ̈ܐ ܕܲܫܡܲܝܵܢܹ̈ܐ ܡܕܲܝܪܵܐ ܢܲܦ̮ܫܵܟ̣ ܘܙܵܡ݁ܪܵܐ ܬܸܫܒ݁ܘܿܚܬܵܐ. ܐܲܦܝܼܣ ܚܠܵܦܲܝܢ ܕܲܒ݂ܗܵܘ݁ ܝܵܘܡܵܐ ܕܦܘܼܪܥܵܢܵܐ. ܢܸܗܘܸܐ ܒܫܘܼܒ݂ܚܵܟ̣ ܫܵܘܬܵܦܹ̈ܐ. ܫܲܒܲܚ. ܦܬ݂ܝܼܚܘܼ ܬܲܪܥܹܗ ܕܦܲܪܕܲܝܣܵܐ. ܘܥܵܠܝܼܢ ܘܫܵܪܹܝܢ ܙܲܕܿܝܼܩܹ̈ܐ. ܥܲܠ ܡܲܒ݁ܘܼܥܵܐ ܕܥܘܼܕ݂ܖ̈ܵܢܹܐ. ܘܲܡܩܲܒ݁ܠܵܐ ܠܗܘܿܢ ܝܲܡܝܼܢܵܐ. ܗܵܝ̇ ܕܲܢܨܲܒ݂ܬܹܗ ܠܦܲܪܕܲܝܣܵܐ. ܘܣܵܝܡܵܐ ܟܠܝܼܠܹ̈ܐ ܒܖܹ̈ܫܲܝܗܘܿܢ.
	 ܩܵܢܘܿܢܵܐ. ܒܚܘܼܣ ܘܚܲܣܵܐ. ܐܲܠܵܗܵܐ ܢܪܲܚܸܡ ܥܠܲܝܢ* ܢܒܲܪܟܲܢ ܘܢܲܢܗܲܪ ܐܲܦܵܘ̈ܗܝ ܒܲܢ*

ܒܗܵܘ݁ ܥܸܕܵܢܵܐ ܕܦܵܩܹ݁ܕ݂ ܪܸܡܙܵܟ̣ ܘܩܵܝ݁ܡܝܼܢ ܡܝܼܬܹ̈ܐ ܡ̣ܢ ܓܵܘ ܩܲܒ݂ܖܹ̈ܐ. ܚܲܕܵܐ ܠܥܲܒ݂ܕܲܝ̈ܟ ܒܲܓ݂ܢܘܿܢܵܐ ܕܡܲܠܟ݁ܘܼܬ݂ܵܟ̣܀

ܕܢܸܕܲܥ ܒܐܲܪܥܵܐ ܐܘܼܖ̈ܚܵܬܹܗ* ܘܲܒ݂ܟ̣ܠܗܘܿܢ ܥܲܡܡܹ̈ܐ ܦܘܼܪܩܵܢܹܗ* ܢܵܘܕܿܘܿܢ ܠܵܟ̣ ܥܲܡܡܹ̈ܐ ܐܲܠܵܗܵܐ* ܢܵܘܕܘܿܢ ܠܵܟ̣ ܥܲܡܡܹ̈ܐ ܟܠܗܘܿܢ* ܢܸܚ̈ܕܝܵܢ ܘܲܢ̈ܫܲܒܚܵܢ ܡܲܠܟ̈ܘܵܬ̣ܵܐ* ܕܕ݂ܵܐܸܢ ܐܲܢ݇ܬ̇ ܠܥܲܡܡܹ̈ܐ ܬܪܝܼܨܵܐܝܼܬ* ܘܲܠܡܲܠܟ̈ܘܵܬ̣ܵܐ ܡܕܲܒܲܪ ܐܲܢ݇ܬ ܒܐܲܪܥܵܐ* ܢܵܘܕܘܿܢ ܠܵܟ̣ ܥܲܡܡܹ̈ܐ ܐܲܠܵܗܵܐ* ܢܵܘܕܿܘܿܢ ܠܵܟ̣ ܥܲܡܡܹ̈ܐ ܟܠܗܘܿܢ* ܐܲܪܥܵܐ ܝܸܗ݇ܒܲܬ̤ ܦܹܐܖܹ̈ܝܗ̇* ܢܒܲܪܟܲܢ ܐܲܠܵܗܵܐ ܐܲܠܵܗܲܢ ܘܲܢܒܲܪܟܲܢ ܐܲܠܵܗܵܐ* ܘܢܸܕ݂ܚܠܘܼܢ ܡܸܢܹܗ ܟܠܗܘܿܢ ܥܸܒ݂ܖܹ̈ܝܗ̇ ܕܐܲܪܥܵܐ*

ܒܗܵܘ݁ ܥܸܕܵܢܵܐ ܕܦܵܩܹ݁ܕ݂ ܪܸܡܙܵܟ̣ ܘܩܵܝ݁ܡܝܼܢ ܡܝܼܬܹ̈ܐ ܡ̣ܢ ܓܵܘ ܩܲܒ݂ܖܹ̈ܐ. ܚܲܕܵܐ ܠܥܲܒ݂ܕܲܝ̈ܟ ܒܲܓ݂ܢܘܿܢܵܐ ܕܡܲܠܟ݁ܘܼܬ݂ܵܟ̣܀

ܘܫܲܒܲܚ܀ ܥܵܠܲܡ.

ܒܗܵܘ݁ ܥܸܕܵܢܵܐ ܕܙܵܥ݁ܩܵܐ ܩܲܪܢܵܐ. ܘܡܸܬ݁ܬ݁ܥܝܼܪܝܼܢ ܬܘܼܒ݂ ܫ̇ܵܟ̣ܒܲܝ ܥܲܦܪܵܐ. ܚܢܵܢܵܟ̣ ܢܸܗܘܸܐ ܡܦܝܼܣܵܢܵܐ ܚܠܵܦ ܟܠܲܢ.

	ܐܲܠܵܗܵܐ ܢܪܲܚܸܡ ܥܠܲܝܢ* ܢܒܲܪܟܲܢ ܘܢܲܢܗܲܪ ܐܲܦܵܘ̈ܗܝ ܒܲܢ*

ܒܗܵܘ݁ ܥܸܕܵܢܵܐ ܕܙܵܥ݁ܩܵܐ ܩܲܪܢܵܐ. ܘܡܸܬ݁ܬ݁ܥܝܼܪܝܼܢ ܬܘܼܒ݂ ܫ̇ܵܟ̣ܒܲܝ ܥܲܦܪܵܐ. ܚܢܵܢܵܟ̣ ܢܸܗܘܸܐ ܡܦܝܼܣܵܢܵܐ ܚܠܵܦ ܟܠܲܢ.

ܬܸܫܒ݁ܘܿܚܬܵܐ. ܩܵܠܵܐ ܕܲܩ̣ܪܵܝܗܝ ܠܠܵܥܵܙܲܪ. ܘܠܲܛܠܝܼܬܼܵܐ ܒܲܪ݇ܬ݂ ܝܘܼܐܵܪܵܫ. ܗ̤ܘ ܩ̇ܵܪܹܐ ܘܲܡܢܲܚܸܡ ܠܲܢ. ܘܲܡܩܝܼܡ ܠܲܢ ܡ̣ܢ ܝܲܡܝܼܢܹܗ܀ ܒܚܲܕ݂ ܩܵܠܵܐ ܕܲܩܥ̣ܵܐ ܡܵܪܲܢ. ܩ̣ܵܡ ܠܵܥܵܙܲܪ ܡ̣ܢ ܩܲܒ݂ܪܵܐ. ܘܲܒ݂ܩܵܠܵܐ ܐ݇ܚܪܹܢܵܐ ܕܩ̇ܵܪܹܐ. ܩܵܝ݁ܡܝܼܢ ܟܠܗܘܿܢ ܥܲܢܝܼܕܹ̈ܐ܀ ܡܘܼܠܟܵܢܵܐ ܕܩܲܒܸ݁ܠ ܡܸܢܵܟ̣. ܓܲܝܵܣܵܐ ܥܲܠ ܓܵܓ݂ܘܼܠܬܵܐ. ܢܩܲܒ݁ܠܘܼܢܵܝܗܝ ܡܵܪܝ ܥܲܢܝܼܕܲܝ̈ܢ. ܕܐܵܘܕܿܝܼܘ ܒܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ̣.
      ܟܵܪܘܿܙܘܼܬ݂ܵܐ. ܐܲܚܝܼܕ݂ ܟܠ. ܘܡܸܬ݂ܐܲܡܪܵܐ ܒܟ̣ܠܗܘܿܢ ܣܘܼܒܵܥܹ̈ܐ ܕܨܵܘܡܵܐ.
 ܢܩܘܼܡ ܫܲܦܝܼܪ...܀ ܡܵܪܝܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ ܐܲܚܝܼܕ݂ ܟܠ ܐܲܠܵܗܵܐ ܕܐܲܒ݂̈ܵܗܵܬܲܢ: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟ̣...܀ ܩܲܕܿܝܼܫܵܐ ܘܲܡܫܲܒ̇ܚܵܐ ܕܲܒ݂ܩܲܕܿܝܼܫܹ̈ܐ ܥܵܡܲܪ ܘܡܸܬ̇ܬ̇ܢܝܼܚ ܨܸܒ݂ܝܵܢܹܗ: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟ̣...܀ ܡܲܠܟܵܐ ܕܡܲܠܟܹ̈ܐ ܘܡܵܪܵܐ ܕܡܵܖ̈ܵܘܵܬ̣ܵܐ ܕܥܵܡܲܪ ܒܢܘܼܗܪܵܐ ܓܲܐܝܵܐ: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟ̣...܀ ܗ̇ܘ ܕܐ݇ܢܵܫ ܡ̣ܢ ܒܢܲܝ̈ܢܵܫܵܐ ܠܵܐ ܚ̣ܙܵܝܗܝ ܘܐܵܦܠܵܐ ܡܨܸܐ ܠܡܸܚܙܝܹܗ: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟ̣...܀ ܗ̇ܘ ܕܨܵܒܹ݁ܐ ܕܟ̣ܠܗܘܿܢ ܒܢܲܝ̈ܢܵܫܵܐ ܢܸܚܘܿܢ. ܘܢܸܬ̣ܦܢܘܿܢ ܠܝܼܕܲܥܬ̣ܵܐ ܕܲܫܪܵܪܵܐ: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟ̣...܀ ܥܲܠ ܚܘܼܠܡܵܢܵܐ ܕܐܲܒ݂ܵܗܲܝ̈ܢ ܩܲܕ݁ܝܼܫܹ̈ܐ: ܡܵܪܝ ܦܠܵܢ ܩܵܬ݂ܘܿܠܝܼܩܵܐ ܦܵܛܲܪܝܵܪܟܵܐ: ܘܡܵܪܝ ܦܠܵܢ ܐܲܦܸܣܩܘܿܦܵܐ ܡܝܼܛܪܵܦܘܿܠܹܝܛܵܐ: ܘܥܲܠ ܟܠܗܘܿܢ ܒܢܲܝ̈ ܬܸܫܡܸܫܬ݁ܗܘܿܢ: ܒܵܥܹܝ̇ܢܲܢ ܡ̣ܢܟ݂...܀ ܐܲܠܵܗܵܐ ܡܪܲܚܡܵܢܵܐ ܗ݁ܘ ܕܲܒܼܖ̈ܲܚܡܹܐ ܡܕܲܒܲܪ ܠܟ̣ܠ: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟܼ...܀ ܗ̇ܘ ܕܒܲܫܡܲܝܵܐ ܡܸܫܬܲܒܲܚ ܘܒ݂ܐܲܪܥܵܐ ܡܸܣܬ݁ܓܸܕ݂: ܒܵܥܹ݁ܝܢܲܢ ܡܸܢܵܟܼ...܀ ܫܲܝܢܵܟ̣ ܘܲܫܠܵܡܵܟ̣ ܐܲܫܪܵܐ ܒܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܡܫܝܼܚܵܐ ܦܵܪܘܿܩܲܢ: ܘܐܸܬ̣ܪܲܚܸܡܥܠܲܝܢ.
 ܘܟܸܢ ܐܲܪܝܼܡܘ ܩܵܠܟ̣ܘܿܢ ܘܫܲܒܲܚܘ ܟܠܹܗ ܥܲܡܵܐ ܠܐܲܠܵܗܵܐ ܚܲܝܵܐ. ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܫܲܒܲܚ܀ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ. ܥܵܠܲܡ܀ ܩܲܕ݁ܝܼܫܵܐ ܐܲܠܵܗܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܚܲܝܠ̱ܬ݂ܵܢܵܐ: ܩܲܕ݁ܝܼܫܵܐ ܠܵܐ ܡܵܝܘܿܬ݂ܵܐ: ܐܸܬ݂ܪܲܚܲܡ ܥܠܲܝܢ.
 ܘܡܲܩܦܝܼܢ ܒܩܵܠܵܐ: ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܢܸܗܘܸܐ ܨܸܒ݂ܝܵܢܵܟ݂: ܐܲܝܟܲܢܵܐ ܕܒܲܫܡܲܝܵܐ ܐܵܦ ܒܐܲܪܥܵܐ. ܗܲܒ݂ܠܲܢ ܠܲܚܡܵܐ ܕܣܘܼܢܩܵܢܲܢ ܝܵܘܡܵܢܵܐ. ܘܲܫܒ݂ܘܿܩ ܠܲܢ ܚܵܘܒܲܝ̈ܢ: ܐܲܝܟܲܢܵܐ ܕܐܵܦ ܚܢܲܢ ܫ̣ܒܲܩܢ ܠܚܲܝܵܒܲܝ̈ܢ. ܘܠܵܐ ܬܲܥܠܲܢ ܠܢܸܣܝܘܿܢܵܐ: ܐܸܠܵܐ ܦܲܨܵܢ ܡ̣ܢ ܒܝܼܫܵܐ: ܡܸܛܠ ܕܕ݂ܝܼܠܵܟ݂ ܗ݇ܝ̣ ܡܲܠܟ݁ܘܼܬ݂ܵܐ ܘܚܲܝܠܵܐ ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܠܥܵܠܲܡ ܥܵܠܡܝܼܢ ܐܵܡܹܝܢ. ܘܡܲܩܸܦ ܪܲܒܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܐܲܒ݂ܵܐ ܘܠܲܒ݂ܪܵܐ ܘܲܠܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ. ܘܥܵܢܹܝ̇ܢ: ܡ̣ܢ ܥܵܠܲܡ ܘܲܥܕܲܡܵܐ ܠܥܵܠܲܡ ܐܵܡܹܝܢ ܘܐܵܡܹܝܢ܀ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܢܸܬ݂ܩܲܕܲܫ ܫܡܵܟ݂. ܬܹܐܬܸܐ ܡܲܠܟ݁ܘܼܬ݂ܵܟ݂. ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁. ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ: ܕܲܡܠܹܝܢ ܫܡܲܝ̈ܵܐ ܘܐܲܪܥܵܐ ܪܲܒ݁ܘܼܬ݂ ܫܘܼܒ݂ܚܵܟ݂: ܥܝܼܖܹ̈ܐ ܘܐ݇ܢܵܫ̈ܵܐ ܩܵܥܹܝ̇ܢ ܠܵܟ݂: ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫ ܩܲܕ݁ܝܼܫܲܬ݁܀
ܘܐܵܡܲܪ ܡܫܲܡܫܵܢܵܐ: ܢܨܲܠܸܐ: ܫܠܵܡܵܐ ܥܲܡܲܢ.
ܨܠܘܿܬ݂ܵܐ. ܦܵܬܲܚ ܬܲܪܥܹܗ ܠܲܕ݂ܢܵܩܫ݁ܝܼܢ ܒܹܗ: ܡܦܲܢܸܐ ܫܹܐ̈ܠܵܬ݂ܵܐ ܠܲܕ݂ܫܵܐ݁ܠܝܼܢ ܠܹܗ. ܦܬܲܚ ܠܵܗ̇ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܬܲܪܥܵܐ ܕܖ̈ܲܚܡܹܐ ܠܲܨܠܘܿܬܲܢ. ܘܩܲܒܸ݁ܠܹܝܗ̇ ܠܒ݂ܵܥܘܼܬܲܢ. ܘܦܲܢܵܐ ܒܖ̈ܲܚܡܲܝܟ ܫܹܐ̈ܠܵܬܲܢ ܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܵܟ̣ ܥܲܬ݁ܝܼܪܵܐ ܘܲܫܦܝܼܥܵܐ: ܛܵܒ݂ܵܐ ܕܠܵܐ ܟܵܠܹ݁ܐ ܖ̈ܲܚܡܵܘܗܝ ܘܡܵܘܗܒ݂ܵܬܹ̈ܗ: ܡ̣ܢ ܣܢܝܼܩܹ̈ܐ ܘܐܲܠܝܼܨܹ̈ܐ ܥܲܒ݂ܕܵܘ̈ܗܝ ܕܩ̇ܵܪܹܝܢ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܹܗ ܒܟ̣ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ̣ܠ.
ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.
ܚܵܬ݂ܵܗ̇: ܫ̇ܵܡܲܥ ܒܩܵܠܵܐ ܕܟܹܐܢܹ̈ܐ ܘܲܕ݂ܙܲܕܿܝܼܩܹ̈ܐ ܕܲܡܪܲܥܹ݁ܝܢ ܠܹܗ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܘܥܵܒܹ݁ܕ݂ ܨܸܒ݂ܝܵܢܵܐ ܕܕ݂ܵܚܠܵܘ̈ܗܝ. ܫܡܲܥ ܡܵܪܝ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܒܲܚܢܵܢܵܟ̣: ܘܩܲܒܸ݁ܠ ܒܵܥܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܒܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ̣: ܘܚܘܼܢ ܠܐܲܠܝܼܨܹ̈ܐ ܘܠܲܡܛܲܖ̈ܦܹܐ ܥܲܒ݂ܕܲܝ̈ܟ ܕܩ̇ܵܪܹܝܢ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܵܟ̣ ܒܟ̣ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ̣ܠ.
ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ: ܒܲܪܸܟ݂ ܡܵܪܝ.

ܨܠܘܿܬ݂ܵܐ. ܥܘܼܕ݂ܪܵܢܵܐ ܡܵܪܝ ܕܖ̈ܲܚܡܲܝܟ ܘܣܘܼܝܵܥܵܐ ܪܲܒܵܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܚܲܝܠܵܐ ܟܲܣܝܵܐ ܘܲܡܫܲܒ݁ܚܵܐ ܕܲܬ݂ܠܝܼܬ݂ܵܝܘܼܬ݂ܵܟ݂ ܡܫܲܒܲܚܬܵܐ: ܘܝܲܡܝܼܢܵܟ݂ ܕܡܲܠܝܵܐ ܖ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ: ܐܲܓܸ݁ܢ ܘܐܲܠܘܵܐ ܠܲܡܚܝܼܠܘܼܬ݂ܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܡ̣ܢ ܒܲܝܬܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ ܡܠܸܐ ܟܠ ܥܘܼܕ݂ܖ̈ܵܢܝܼܢ ܘܟ݂ܠ ܛܘܼܒ݂ܝܼ̈ܢ: ܒܲܨܠܘܿܬ݂ ܟܠܗܘܿܢ ܩܲܕ݁ܝܼܫܹ̈ܐ ܡܖ̈ܲܥܝܵܢܲܝܟ: ܡܵܪܵܐ ܕܟ݂ܠ...

ܘܡܲܩܦܝܼܢ. ܒܒ݂ܘܼܪܟ݁ܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܢܸܬ݂ܒܲܪܟ݂ܘܼܢ ܥܲܒ݂ܕܲܝ̈ܟ: ܘܒܲܒ݂ܛܝܼܠܘܼܬܹܗ ܕܨܸܒ݂ܝܵܢܵܟ݂ ܢܸܬ݂ܢܲܛܪܘܼܢ ܣܵܓ݂ܘܿܕܲܝ̈ܟ: ܘܫܲܝܢܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܲܫܠܵܡܵܐ ܢܲܓ݁ܝܼܪܵܐ ܕܡܵܪܘܼܬ݂ܵܟ݂: ܢܲܡܠܸܟ݂ ܒܥܲܡܵܟ݂ ܘܲܒ݂ܥܹܕ݁ܬܵܟ݂. ܟܠܗܘܿܢ ܝܵܘܡܵܬܹ̈ܗ ܕܥܵܠܡܵܐ: ܡܵܪܵܐ ܕܟ݂ܠ...ܘܥܵܢܹܝ̇ܢ: ܐܵܡܹܝܢ ܒܲܪܸܟ݂ܡܵܪܝ.

ܐ݇ܚܪܹܬ݂ܵܐ. ܒܘܼܪܟ݁ܬܹܗ ܕܲܡܒܲܪܸܟ݂ ܟܠ: ܘܫܲܝܢܹܗ ܕܲܡܫܲܝܸܢ ܟܠ: ܘܲܚܢܵܢܹܗ ܕܚܵܐܹ̇ܢ ܟܠ: ܘܲܡܢܲܛܪܵܢܘܼܬܹܗ ܕܐܲܠܵܗܲܢ ܣܓ݂ܝܼܕ݂ܵܐ: ܬܸܗܘܸܐ ܥܲܡܲܢ ܘܒܲܝܢܵܬܲܢ ܘܲܚܕ݂ܵܖ̈ܲܝܢ: ܘܲܬ݂ܢܲܛܪܲܢ ܡ̣ܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹ̈ܗ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܒ݂ܘܼܪܟ݁ܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܢܸܬ݂ܒܲܪܲܟ݂: ܘܒܲܒ݂ܛܝܼܠܘܼܬ݂ܵܟ݂ ܢܸܬ݂ܢ̱ܛܲܪ: ܘܚܲܝܠܵܟ݂ ܢܹܐܬܸܐ ܠܐܝܼܵܠܲܢ: ܘܥܘܼܕ݂ܪܵܢܵܟ݂ ܢܸܬ݂ܠܵܘܸܐ ܠܲܢ: ܘܝܲܡܝܼܢܵܟ݂ ܬܲܓܸ݁ܢ ܥܠܲܝܢ: ܘܫܲܝܢܵܟ݂ ܢܲܡܠܸܟ݂ ܒܲܝܢܵܬܲܢ: ܘܲܨܠܝܼܒ݂ܵܟ݂ ܫܘܼܪܵܐ ܪܵܡܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܢܸܗܘܸܐ ܠܲܢ: ܘܲܬ݂ܚܹܝܬ݂ ܟܸܢܦܵܘ̈ܗܝ ܢܸܣܬܲܬܲܪ ܡ̣ܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹ̈ܗ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܪܝܼܟ݂ܝܼܢ ܡܵܪܝ ܖ̈ܲܚܡܹܐ ܕܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܲܣܓ݂ܝܼܕ݂ܝܼܢ ܫܘܼܘܕܵܝܹ̈ܐ ܕܡܵܪܘܼܬ݂ܵܟ݂: ܕܲܒ݂ܟ݂ܠܙܒܲܢ ܡܲܠܦܝܼܢ ܠܲܢ ܕܲܢܚܘܼܪ ܠܘܵܬ݂ܵܟ݂ ܘܢܸܫܬܲܒ݂ܗܲܪ ܒܵܟ݂. ܘܠܵܐ ܢܸܦܣܘܿܩ ܣܲܒ݂ܪܲܢ ܡܸܢܵܟܼ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܘܼܪܟ݁ܬ݂ܵܟ݂ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܥܲܠ ܥܲܡܵܟ݂ ܬܸܬ݁ܢܝܼܚ: ܘܲܥܠܲܝܢ ܡܚܝܼܠܹ̈ܐ ܘܚܲܛܵܝܹ̈ܐ ܚܢܵܢܵܟ݂ ܢܸܗܘܸܐ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܣܲܒ݂ܪܲܢ ܛܵܒ݂ܵܐ: ܘܒܹܝܬ݂ ܓܵܘܣܲܢ ܡܠܸܐ ܖ̈ܲܚܡܹܐ: ܘܫܵܒܹܩ̇ ܚܵܘܒܹ̈ܐ ܘܲܚܛܵܗܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܫܲܝܢܹܗ ܕܐܲܒ݂ܵܐ ܥܲܡܲܢ: ܘܚܘܼܒܹ݁ܗ ܕܲܒ݂ܪܵܐ ܒܲܝܢܵܬܲܢ: ܘܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܡܕܲܒܲܪ ܠܲܢ ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܹܗ: ܘܲܥܠܲܝܢ ܖ̈ܲܚܡܵܘܗܝ ܘܲܚܢܵܢܹܗ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܫܲܝܢܵܟ݂ ܡܵܪܝ ܢܸܫܪܸܐ ܒܲܢ: ܘܲܫܠܵܡܵܟ݂ ܢܸܫܬܲܠܲܛ ܒܲܢ: ܘܚܘܼܒܵܟ݂ ܢܸܣܓܸ݁ܐ ܒܲܝܢܵܬܲܢ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܝܲܡܝܼܢܵܟ݂ ܡܵܪܝ ܢܲܛܲܪ ܠܲܢ: ܘܲܬ݂ܚܹܝܬ݂ ܟܸܢܦܲܝ̈ܟ ܣܲܬܲܪ ܠܲܢ: ܘܥܘܼܕ݂ܪܵܢܵܟ݂ ܢܸܬ݂ܠܵܘܸܐ ܠܲܢ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܗܲܒ݂ܠܲܢ ܡܵܪܝ ܫܲܝܢܵܐ ܐܲܡܝܼܢܵܐ: ܘܚܘܼܒܵܐ ܘܪܸܚܡܲܬ݂ ܝܘܼܠܦܵܢܵܐ: ܘܚܲܝܹ̈ܐ ܘܛܘܼܒܹ̈ܐ ܘܚܲܕ݂̈ܘܵܬ݂ܵܐ. ܘܡܵܙܘܿܢܵܐ ܕܠܵܐ ܢܸܚܣܲܪ ܠܲܢ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܗܘ̤ܝܼ ܢܵܛܘܿܪܵܐ ܕܠܵܐ ܕܵܡܹܟ݂: ܠܟܲܪܟ݂ܵܐ ܕܥܵܢ̈ܵܟ݂ ܫܲܪܝܵܐ ܒܹܗ: ܕܠܵܐ ܬܸܬ݂ܢ̱ܟܸܐ ܡ̣ܢ ܕܹܐܒܹ̈ܐ ܨܲܗܝܲܝ̈ ܠܲܕ݂ܡܵܐ ܕܲܓ݂ܙܵܪܵܟ݂: ܕܐܲܢ݇ܬ݁ܘܼ ܝܲܡܵܐ ܕܠܵܐ ܚܵܣܲܪ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܒ݂ܘܼܖ̈ܟܵܬ݂ܵܟ݂ ܡܵܪܲܢ ܒܲܪܸܟܲܝܢ: ܘܫܘܼܪ ܒܛܝܼܠܘܼܬ݂ܵܟ݂ ܐܲܟ݂ܪܸܟܲܝܢ: ܘܡ̣ܢ ܛܵܒ݂̈ܵܢ ܠܵܐ ܬܲܨܪܸܟܲܝܢ: ܘܒܲܓ݂ܢܘܿܢ ܢܘܼܗܪܵܟ݂ ܐܲܣܡܸܟܲܝܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܬܵܐ ܡܵܪܝ ܠܐܝܼܵܠܲܢ ܒܲܚܢܵܢܵܟ݂: ܘܐܸܬ݂ܓܲܠܝ ܠܦܘܼܪܩܵܢܲܢ ܒܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂: ܘܲܬ݂ܪܘܿܨ ܗܲܠ̈ܟ݂ܵܬܲܢ ܒܲܫܒ݂ܝܼܠܹ̈ܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܬܸܕ݂ܢܲܚ ܠܲܢ ܡܵܪܝ ܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܡܵܐ ܕܕ݂ܵܝ̇ܢܵܐ ܠܲܢ ܟܹܐܢܘܼܬ݂ܵܟ݂: ܘܖ̈ܲܚܡܲܝܟ ܢܹܐܬ݂ܘܿܢ ܠܐܝܼܵܠܲܢ: ܒܝܵܘܡܵܐ ܕܕ݂ܵܢ̇ܚܵܐ ܪܲܒ݁ܘܼܬ݂ܵܟ݂: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܘܼܪܟ݁ܬ݂ܵܟ݂ ܘܛܲܝܒ݁ܘܼܬ݂ܵܟ݂ ܘܝܲܡܝܼܢܵܟ݂ ܕܡܲܠܝܵܐ ܖ̈ܲܚܡܹܐ ܘܲܚܢܵܢܵܐ: ܐܲܓܸ݁ܢ ܘܐܲܠܘܵܐ ܥܲܠ ܟܸܢܫܵܐ ܕܣܵܓ݂ܘܿܕܲܝ̈ܟ ܕܩܵܪܹܝ̇ܢ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܵܟ݂ ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܲܪܸܟ݂ ܡܵܪܝ ܒܛܲܝܒ݁ܘܼܬ݂ܵܟ݂: ܘܢܲܛܲܪ ܒܲܡܪܲܚܡܵܢܘܼܬ݂ܵܟ݂ ܠܟܸܢܫܵܐ ܕܲܡܗܲܝܡܢܹ̈ܐ: ܕܲܬ݂ܟ݂ܝܼܠܝܼܢ ܥܠܲܝܟ ܘܩܵܪܹܝ̇ܢ ܠܲܫܡܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ ܘܡܸܬ݂ܟܲܫܦܝܼܢ ܠܵܟ݂ ܒܟ݂ܠܥܸܕܵܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܗܘ̤ܝܼ ܡܵܪܝ ܥܲܡܲܢ ܒܖ̈ܲܚܡܲܝܟ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܘܕܲܒܲܪ ܠܲܢ ܐܲܝܟ݂ ܨܸܒ݂ܝܵܢܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂ ܟܠܗܘܿܢ ܝܵܘ̈ܡܵܬ݂ܵܐ ܕܚܲܝܲܝ̈ܢ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܐ݇ܚܪܹܬ݂ܵܐ. ܒܘܼܪܟ݁ܬ݂ܵܐ ܡܵܪܝ ܕܖ̈ܲܚܡܹܐ: ܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܕܖ̈ܲܚܡܹܐ: ܒܝܲܕ݂ ܡܲܠܲܐܟ݂ܵܐ ܕܖ̈ܲܚܡܹܐ: ܡ̣ܢ ܠܘܵܬ݂ ܐܲܠܵܗܵܐ ܣܲܓ݁ܝܼ ܖ̈ܲܚܡܹܐ: ܬܸܫܬܲܕܲܪ ܘܬܸܫܬܲܦܲܥ ܥܲܠ ܟܸܢܫܹ̈ܐ ܕܲܡܫܲܒ݁ܚܵܢܲܝ̈ܟ: ܕܐܲܡܝܼܢܝܼܢ ܘܩܵܪܹܝ̇ܢ ܠܲܫܡܵܟ݂ ܩܲܕ݁ܝܼܫܵܐ: ܘܚܵܝ̇ܪܝܼܢ ܘܲܡܣܲܟܹ݁ܝܢ ܠܖ̈ܲܚܡܹܐ: ܘܲܠܦܘܼܪܩܵܢܵܐ ܕܡ̣ܢ ܠܘܵܬ݂ܵܟ݂: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܡܲܩܦܝܼܢ. ܨܠܘܿܬ݂ܵܗ̇ ܡܵܪܝ ܕܲܒ݂ܬ݂ܘܼܠܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܒ݂ܵܥܘܼܬ݂ܵܗ̇ ܕܐܸܡܵܐ ܡܒܲܪܲܟ݂ܬܵܐ: ܘܬܲܟ݂ܫܲܦ̮ܬܵܗ̇ ܘܬܲܚܢܲܢܬܵܗ̇ ܕܡܲܠܝܲܬ݂ ܛܲܝܒ݁ܘܼܬ݂ܵܐ: ܡܵܪܬ݁ܝ ܡܲܪܝܲܡ ܛܘܼܒ݂ܵܢܝܼܬ݂ܵܐ: ܘܚܲܝܠܹܗ ܪܲܒܵܐ ܕܲܨܠܝܼܒ݂ܵܐ ܙܲܟܵܝܵܐ: ܘܥܘܼܕ݂ܪܵܢܵܐ ܐܲܠܵܗܵܝܵܐ: ܘܒ݂ܵܥܘܼܬܹܗ ܕܡܵܪܝ ܝܘܿܚܲܢܵܢ ܡܲܥ̱ܡܕ݂ܵܢܵܐ: ܘܕܲܫܠܝܼܚܹ̈ܐ ܛܘܼܒ݂ܵܢܹ̈ܐ: ܘܕܲܐܒ݂ܵܗ̈ܵܬ݂ܵܐ ܢܲܨܝܼܚܹ̈ܐ ܡܣܲܒ݁ܖ̈ܵܢܹܐ ܕܙܲܕ݁ܝܼܩܘܼܬ݂ܵܐ: ܘܙܵܖ̈ܥܲܝ ܫܲܝܢܵܐ ܒܲܒ݂ܪܝܼܬ݂ܵܐ. ܘܕܲܐܒ݂ܘܼܢ ܙܲܗܝܵܐ ܘܩܲܕ݁ܝܼܫܵܐ ܡܵܪܝ ܐܲܒ݂ܵܐ ܩܵܬ݂ܘܿܠܝܼܩܵܐ: ܘܡܵܪܝ ܐܸܣܛܲܦܵܢܘܿܣ ܒܘܼܟ݂ܪܵܐ ܕܣܵܗܕܹ̈ܐ: ܘܲܕ݂ܓܲܢ݇ܒܵܪ ܚܲܝܠܵܐ ܡܵܪܝ ܓܝܼܘܲܪܓܝܼܣ ܣܵܗܕܵܐ ܢܲܨܝܼܚܵܐ: ܘܲܕ݂ܡܵܪܝ ܦܠܵܢ: ܘܲܕ݂ܡܵܪܝ ܐܵܘܓܹ݁ܝܢ ܛܘܼܒ݂ܵܢܵܐ ܘܲܕ݂ܟ݂ܠܵܗ̇ ܣܝܼܥܬܹ݁ܗ ܪܘܼܚܵܢܵܝܬܵܐ. ܘܕܲܫܡܘܿܢܝܼ ܘܕܲܒ݂ܢܹܝ̈ܗ̇: ܘܲܕ݂ܟ݂ܠܗܘܿܢ ܣܵܗܕܵܘ̈ܗܝ ܘܩܲܕ݁ܝܼ̈ܫܵܘܗܝ ܕܡܵܪܲܢ: ܬܸܗܘܸܐ ܠܲܢ ܐܲܡܝܼܢܵܐܝܼܬ݂: ܫܘܼܪܵܐ ܪܵܡܵܐ ܘܒܹܝܬ݂ ܓܵܘܣܵܐ ܚܲܣܝܼܢܵܐ: ܕܬܸܦܪܘܿܩ ܘܲܬ݂ܦܲܨܸܐ ܘܲܬ݂ܫܵܘܙܸܒ݂ ܘܲܬ݂ܢܲܛܲܪ ܠܦܲܓ݂ܖ̈ܲܝܢ ܘܢܲܦ̮̈ܫܵܬܲܢ ܡ̣ܢ ܒܝܼܫܵܐ ܘܚܲܝܠܵܘܵܬܹܗ: ܒܟ݂ܠܗܘܿܢ ܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ: ܡܵܪܵܐ ܕܟ݂ܠ.
ܘܥܵܢܹ̇ܝܢ: ܐܵܡܹܝܢ.
ܘܟܸܢ ܚܵܬ݂ܡ̇ܝܼܢ ܒܓ݂ܘܼܕܵܐ ܕܲܐ݇ܚܖ̈ܵܝܹܐ. ܘܢܵܣܹ̇ܒ݂ ܟܵܗܢܵܐ ܨܠܝܼܒ݂ܵܐ ܒܐܝܼܕܹܗ ܕܝܲܡܝܼܢܵܐ: ܘܩܵܐܹ̇ܡ ܒܓܲܒܵܐ ܬܲܝܡ̱ܢܵܝܵܐ: ܘܡܲܗܦܸܟ݂ ܐܲܦܵܘ̈ܗܝ ܠܥܲܡܵܐ ܘܐܵܡܲܪ: ܒܲܪܸܟ݂ ܡܵܪܝ ܒܦܘܼܩܕܵܢܟ݂ܘܿܢ. ܘܥܵܢܹ̇ܝܢ: ܒܦܘܼܩܕܵܢܹܗ ܕܲܡܫܝܼܚܵܐ ܡܵܪܲܢ: ܘܫܘܼܒ݂ܚܵܐ ܠܲܫܡܹܗ ܩܲܕ݁ܝܼܫܵܐ. ܘܡܲܪܟ݁ܢܝܼܢ ܥܲܡܵܐ ܖܹ̈ܫܲܝܗܘܿܢ.
ܘܡܲܩܸܦ ܚܘܼܬܵܡܵܐ. ܫܘܼܒ݂ܚܵܐ ܠܵܟ݂ ܝܼܫܘܿܥ ܡܲܠܟܲܢ ܙܲܟܵܝܵܐ: ܨܸܡܚܹܗ ܕܐܲܒ݂ܵܐ ܡܬ݂ܘܿܡܵܝܵܐ: ܝܼܠܝܼܕ݂ܵܐ ܕܠܵܐ ܫܘܼܪܵܝܵܐ: ܠܥܸܠ ܡ̣ܢ ܙܲܒ݂ܢܹ̈ܐ ܘܲܗܘܲܝ̈ܵܐ: ܕܠܲܝܬ݁ ܠܲܢ ܣܲܒ݂ܪܵܐ ܘܣܘܼܟܵܝܵܐ: ܐܸܠܵܐ ܐܸܢ ܐܲܢ݇ܬ݁ ܒܵܪܘܿܝܵܐ. ܒܲܨܠܘܿܬ݂ ܟܹܐܢܹ̈ܐ ܘܲܓ݂ܒܲܝ̈ܵܐ: ܕܲܫ̣ܦܲܪܘ ܠܵܟ݂ ܡ̣ܢ ܫܘܼܪܵܝܵܐ: ܥܒܸܕ݂ ܠܲܚܛܵܗܲܝ̈ܢ ܚܘܼܣܵܝܵܐ: ܘܲܠܣܲܟ݂ܠ̱̈ܘܵܬܲܢ ܫܪܵܝܵܐ: ܘܲܠܥܵܩ̈ܵܬܲܢ ܦܘܼܨܵܝܵܐ: ܘܲܠܫܸܐ̈ܠܵܬܲܢ ܦܘܼܢܵܝܵܐ: ܘܡܲܢܲܥܲܝܢ ܠܢܘܼܗܪܵܟ݂ ܓܲܐܝܵܐ: ܘܢܲܛܲܪܲܝܢ ܒܢܝܼܫܵܟ݂ ܚܲܝܵܐ: ܡ̣ܢ ܟܠ ܢܸܟ݂ܝܵܢܝܼ̈ܢ ܟܣܲܝ̈ܵܐ ܘܲܓ݂ܠܲܝ̈ܵܐ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ). ܗܵܫܵܐ ♰ ܘܲܒ݂ܟ݂ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܪܵܫܹ̇ܡ ܨܠܝܼܒ݂ܵܐ ܥܲܠ ܥܲܡܵܐ: ܟܲܕ݂ ܦܵܫܹ̇ܛ ܐܝܼܕܹܗ ܠܥܸܠ ܠܐܲܦܲܝ̈ ܡܲܥ̱ܪܒ݂ܵܐ: ܘܡܲܝܬܸ݁ܐ ܠܵܗ̇ ܠܡܲܕ݂ܢܚܵܐ: ܘܟܸܢ ܦܵܫܹ̇ܛ ܠܵܗ̇ ܠܓܲܒܵܐ ܕܬܲܝ̱ܡܢܵܝܵܐ: ܘܡܵܘܒܸ݁ܠ ܠܵܗ̇ ܠܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ. ܐܲܝܟܲܢܵܐ ܕܩܘܼܕ݂ܫܵܐ ܢܸܣܬ݁ܡܸܟ݂ ܥܲܠ ܓܲܒܵܐ ܓܲܪܒ݁ܝܵܝܵܐ ܕܐܝܼܬ݂ܵܘܗܝ ܓܲܒܵܐ ܣܸܡܵܠܵܝܵܐ ܕܟܸܢܫܵܐ.
ܘܥܵܢܹ̇ܝܢ: ܡܫܝܼܚܵܐ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬ݂ܵܟ݂: ܡܫܝܼܚܵܐ ܢܙܲܗܸܐ ܟܵܗܢܘܼܬ݂ܵܟ݂ ܒܡܲܠܟ݁ܘܼܬ݂ܵܐ ܕܲܫܡܲܝܵܐ.
ܐ݇ܚܪܹܢܵܐ: ܬܸܬ݂ܩܲܒܲܠ ܡܵܪܲܢ ܘܐܲܠܵܗܲܢ ܨܠܘܿܬ݂ܵܐ ܕܥܲܒ݂ܕܲܝ̈ܟ ܚܲܠܵܫܹ̈ܐ: ܩܕ݂ܵܡ ܒܹܝܡܵܐ ܕܐܲܠܵܗܘܼܬ݂ܵܟ݂: ܘܬܸܗܘܸܐ ܟܢܝܼܫܘܼܬܲܢ ܗܵܕܹܐ ܠܲܢܝܵܚ ܨܸܒ݂ܝܵܢ ܡܵܪܘܼܬ݂ܵܟ݂: ܠܡܸܣܲܒ݂ ܡܸܢܵܟܼ ܒܛܲܝܒ݁ܘܼ ܚܘܼܠܡܵܢܵܐ ܕܦܲܓ݂ܪܵܐ: ܘܢܘܼܛܵܪܵܐ ܕܢܲܦ̮ܫܵܐ: ܘܫܘܼܪܬܵܚܵܐ ܕܡܵܙܘܿܢܵܐ:(ܘܚܘܼܣܵܝܵܐ ܕܚܵܘܒܹ̈ܐ ܘܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ):ܘܫܲܝܢܵܐ ܡܵܪܝ ܐܲܡܝܼܢܵܐ: ܘܲܫܠܵܡܵܐ ܢܲܓ̇ܝܼܪܵܐ: ܘܐܵܘܝܘܼܬ݂ܵܐ ܕܚܘܼܒܵܐ ܕܠܵܐ ܥܵܒ݂ܪܵܐ ܘܠܵܐ ܡܸܫܬܲܪܝܵܐ ܡ̣ܢ ܒܲܝܢܵܬܲܢ: ܒܟ݂ܠܹܗ ܙܲܒ݂ܢܵܐ ܕܩܵܐܹ̇ܡ ܒܹܗ ܥܵܠܡܵܐ ܗܵܢܵܐ: )ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܒܪܝܼܟ݂ ܐܲܠܵܗܵܐ ܠܥܵܠܡܝܼܢ: ܘܲܡܫܲܒܲܚ ܫܡܹܗ ܩܲܕ̇ܝܼܫܵܐ ܠܕ݂ܵܪܕܵܖ̈ܝܼܢ: ܕܡ̣ܢܗ ܒܵܥܹܝ̇ܢܲܢ: ܘܲܠܝܲܡܵܐ ܫܦܝܼܥܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܡܸܬ݂ܟܲܫܦܝܼܢܲܢ: ܕܗ̤ܘ ܢܲܫܘܸܐ ܠܲܢ ܠܫܘܼܒ݂ܚܵܐ ܦܲܐܝܵܐ ܕܡܲܠܟ̇ܘܼܬܹܗ: ܘܲܠܒ݂ܘܼܣܵܡܵܐ ܕܥܲܡ ܡܲܠܲܐܟ݂ܵܘ̈ܗܝ ܩܲܕ̇ܝܼܫܹ̈ܐ: ܘܲܠܓܲܠܝܘܼܬ݂ ܐܲܦܹ̈ܐ ܕܲܩܕ݂ܵܡܵܘܗܝ: ܘܲܠܩܵܘܡܵܐ ܕܡ̣ܢ ܝܲܡܝܼܢܹܗ ܒܐܘܿܪܸܫܠܸܡ ܕܲܠܥܸܠ: ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܖ̈ܲܚܡܵܘܗܝ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟ̇ܠ ܕܐܸܬ݂ܟܲܢܲܫܢܲܢ ܒܒܲܝܬܹ݁ܗ: ܘܨܲܠܝܼܢܲܢ ܩܕ݂ܵܡ ܪܲܒ݁ܘܼܬܹܗ: ܥܲܠ ܣܲܒ݂ܪܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܗ̤ܘ ܢܸܫܡܲܥ ܨܠܘܿܬܲܢ ܒܲܚܢܵܢܹܗ: ܘܲܢܩܲܒܸ݁ܠ ܒܵܥܘܼܬܲܢ ܒܚܵܘܣܵܢܹܗ: ܘܲܢܫܝܼܓ݂ ܘܢܸܡܪܘܿܩ ܨܵܐܬ݂ܵܐ ܕܚܵܘܒܲܝ̈ܢ ܘܕܲܚܛܵܗܲܝ̈ܢ ܒܙܘܿܦܵܐ ܫܦܝܼܥܵܐ ܕܚܲܝܘܼܣܬ݂ܵܢܘܼܬܹܗ: ܘܲܢܢܝܼܚ ܢܲܦ̮̈ܫܵܬ݂ܵܐ ܕܥܲܢܝܼܕܲܝ̈ܢ ܒܐܵܘܵܢܹ̈ܐ ܫܒ݂ܝܼܚܹ̈ܐ ܕܡܲܠܟ̇ܘܼܬܹܗ: ܘܢܸܪܘܿܣ ܥܲܠ ܟܠܲܢ ܛܲܠܵܐ ܕܒܲܣܝܼܡܘܼܬܹܗ: ܘܲܥܠܲܝܢ ܘܥܲܠ ܟܠ ܒܸܖ̈ܝܵܢ ܬܲܓܸ̇ܢ ܝܲܡܝܼܢܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬܹܗ: ܒܛܲܝܒ݁ܘܼܬܹܗ ܘܲܒ݂ܖ̈ܲܚܡܵܘܗܝ:(ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.

ܐ݇ܚܪܹܢܵܐ

: ܐܲܠܵܗܵܐ ܡܵܪܹܟܠ ܕܝܲܗ݇ܒ݂ ܬܸܫܒ݁ܚܵܬܹ̈ܗ ܒܦܘܼܡܲܢ: ܘܲܙܡܝܼܖ̈ܵܬܹܗ ܒܠܸܫܵܢܲܢ: ܘܩܝܼܢܵܬܹ̈ܗ ܒܓܲܓ̇ܖ̈ܵܬܲܢ: ܘܬ݂ܵܘܕ̇ܝܼܬܹܗ ܒܣܸܦ̈ܘܵܬܲܢ: ܘܗܲܝܡܵܢܘܼܬܹܗ ܒܠܸܒܵܘ̈ܵܬܲܢ: ܗ̤ܘ ܢܸܫܡܲܥ ܨ̈ܠܵܘܵܬܲܢ: ܘܲܢܩܲܒܸ݁ܠ ܒܵܥ̈ܘܵܬܲܢ: ܘܢܸܬ݂ܪܲܥܸܐ ܒܬܲܟ݂ܫ̈ܦܵܬܲܢ: (ܘܲܢܚܲܣܸܐ ܚܵܘ̈ܒܵܬܲܢ:)ܘܲܢܦܲܢܸܐ ܠܲܢ ܫܸܐ̈ܠܵܬܲܢ ܫܲܦܝܼܖ̈ܵܬ݂ܵܐ ܠܵܐ ܥܕ݂ܝܼ̈ܠܵܬ݂ܵܐ. ܘܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܵܐ ܪܲܒܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ ܢܲܫܦܲܥ ܖ̈ܲܚܡܵܘܗܝ ܘܲܚܢܵܢܹܗ: ܥܠܲܝܢ ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡ̣ܢ ܣܵܟܹܗ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܢܸܫܬܲܒܲܚ ܫܡܹܗ ܕܐܲܠܵܗܵܐ ܡܵܪܹܟܠ ܒܲܝܢܵܬܲܢ: ܡܛܲܟ̇ܣܵܢܗܘܿܢ ܕܙܲܒ݂ܢܹ̈ܐ ܘܥܸܕܵܢܹ̈ܐ. ܘܲܥܠܲܝܢ ܡܚܝܼ̈ܠܹܐ ܘܚܲܛܵܝܹ̈ܐ: ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡ̣ܢ ܣܵܟܹܗ. ܘܥܲܠ ܥܹܕ݁ܬܵܐ ܩܲܕ̇ܝܼܫܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇: ܘܥܲܠ ܐܲܒ݂ܵܗܲܝ̈ܢ ܘܖ̈ܲܒܵܢܲܝܢ ܘܡܲܠܦܵܢܲܝ̈ܢ ܘܥܲܢܝܼܕܲܝ̈ܢ: ܘܥܲܢܝܼܕܹ̈ܐ ܕܲܦܪܲܫ̣ܘ ܘܫܲܢ̣ܝܼܘ ܡ̣ܢ ܒܲܝܢܵܬܲܢ: ܘܥܲܠ ܟܠܵܗ̇ ܐܲܚܘܼܬܲܢ ܕܒܲܡܫܝܼܚܵܐ: ܬܲܓܸ̇ܢ ܝܲܡܝܼܢܵܐ ܕܲܒ݂ܛܝܼܠܘܼܬܹܗ ܘܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܠܐܲܠܵܗܵܐ ܫܘܼܒ݂ܚܵܐ: ܘܲܠܡܲܠܲܐܟܹ̈ܐ ܐܝܼܩܵܪܵܐ: ܘܲܠܣܵܛܵܢܵܐ ܒܸܗܬ݁ܬ݂ܵܐ: ܘܠܲܨܠܝܼܒ݂ܵܐ ܣܸܓ݂ܕ̇ܬ݂ܵܐ: ܘܲܠܥܹܕ݁ܬܵܐ ܪܘܼܡܪܵܡܵܐ: ܘܲܠܥܲܢܝܼܕܹ̈ܐ ܢܘܼܚܵܡܵܐ: ܘܲܠܬܲܝܵܒܹ̈ܐ ܩܘܼܒܵܠܵܐ: ܘܠܲܐܣܝܼܖܹ̈ܐ ܫܪܵܝܵܐ: ܘܲܠܡܲܖ̈ܥܹܐ ܘܲܟ݂ܖ̈ܝܼܗܹܐܚܘܼܠܡܵܢܵܐ ܘܐܵܣܝܘܼܬ݂ܵܐ: ܘܠܲܐܖ̈ܒܲܥܦܸܢܝܵܬܹܗ ܕܥܵܠܡܵܐ ܫܲܝܢܵܐ ܪܲܒܵܐ ܘܲܫܠܵܡܵܐ. ܘܐܵܦ ܥܠܲܝܢ ܕܐܝܼܬܲܝܢ ܡܚܝܼ̈ܠܹܐ ܘܚܲܛܵܝܹ̈ܐ: ܚܢܵܢܹܗ ܘܖ̈ܲܚܡܵܘܗܝ ܕܐܲܠܵܗܲܢ ܣܓ݂ܝܼܕ݂ܵܐ ܢܹܐܬ݂ܘܿܢ ܘܢܲܓ̇ܢܘܼܢ ܘܢܸܫܬܲܦܥܘܼܢ ܘܲܢܩܵܘܘܿܢ ܘܢܲܡ̱ܠܟ݂ܘܼܢ ܒܐܲܡܝܼܢܘܼܬ݂ܵܐ: )ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: )ܒܝܲܡܝܼܢ ܡܵܪܘܼܬ݂ܵܟ݂ ܐܲܒ݂ܘܼܢ ܕܒܲܫܡܲܝܵܐ): ܒܲܪܸܟ݂ܡܵܪܝ ܠܟ݂ܠܲܢ: ܘܢܲܛܲܪ ܠܟ݂ܠܲܢ: ܘܥܲܕܲܪ ܠܟ݂ܠܲܢ: ܘܣܲܝܲܥ ܘܣܲܬܲܪ ܠܟ݂ܠܲܢ: ܘܢܲܚܸܡ ܥܲܢܝܼܕܹ̈ܐ ܕܟ݂ܠܲܢ: ܘܐܲܓܸ̇ܢ ܝܲܡܝܼܢܵܟ݂ ܥܲܠ ܟܠܲܢ: ܘܖ̈ܲܚܡܲܝܟ ܘܲܚܢܵܢܵܟ݂ ܢܸܫܬܲܦܥܘܼܢ ܥܲܠ ܟܠܲܢ:ܘܬܸܫܒ݁ܘܿܚܬܵܐ ܘܐܝܼܩܵܪܵܐ ܘܬ݂ܵܘܕ̇ܝܼܬ݂ܵܐ ܘܣܸܓ݂ܕ̇ܬ݂ܵܐ: ܘܩܘܼܒܵܠܛܲܝܒ݁ܘܼܬ݂ܵܐ ܐܲܡܝܼܢܬܵܐ ܬܸܣܲܩ ܠܵܟ݂ ܡ̣ܢ ܦܘܼܡ ܟܠܲܢ: (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܐ݇ܚܪܹܢܵܐ: ܐܲܠܵܗܵܐ ܡܵܪܹܟ̇ܠ ܢܒܲܪܸܟ݂ ܠܟܸܢܫܲܢ ܒܛܲܝܒ݁ܘܼܬܹܗ: ܘܲܢܢܲܛܲܪ ܠܲܩܝܵܡܲܢ ܒܣܘܿܓ݂ܐܵܐ ܫܦܝܼܥܵܐ ܕܲܡܪܲܚܡܵܢܘܼܬܹܗ: ܘܲܢܦܲܢܸܐ ܫܸܐ̈ܠܵܬܲܢ ܡ̣ܢ ܒܹܝܬ݂ ܓܲܙܹܗ: ܘܥܲܠ ܥܵܠܡܵܐ ܟܠܹܗ ܡ̣ܢ ܣܵܟܹܗ: ܘܥܲܠ ܥܹܕ݁ܬܵܐ ܩܲܕ̇ܝܼܫܬܵܐ ܘܝܲܠܕܹܝ̈ܗ̇: ܘܥܲܠ ܐܲܬ݂ܪܵܐ ܗܵܢܵܐ ܘܥܵܡܘܿܖ̈ܵܘܗܝ: ܘܥܲܠ ܥܘܼܡܪܵܐ ܗܵܢܵܐ ܘܝܲܬܵܒ݂ܵܘ̈ܗܝ: ܘܥܲܠ ܟܠܲܢ ܒܟ݂ܠܲܢ ܟܢܝܼܫܵܐܝܼܬ݂: ܚܢܵܢܹܗ ܘܖ̈ܲܚܡܵܘܗܝ ܕܐܲܠܵܗܲܢ ܛܵܒ݂ܵܐ: ܢܹܐܬ݂ܘܿܢ ܘܢܸܫܬܲܦܥܘܼܢ ܒܐܲܡܝܼܢܘܼ (ܡܫܝܼܚܵܐ ܣܲܒ݂ܪܹܗ ܕܲܟ݂ܝܵܢܲܢ ܗܵܫܵܐ)♰ܘܲܒ݂ܟ̣ܠܙܒܲܢ ܘܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ.
ܘܡܲܩܦܝܼܢ ܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܲܬ݂ܠܵܬ݂ܡܐܵܐ ܘܲܬ݂ܡܲܢܬܲܥܣܲܪ ܐܲܒ݂ܵܗ̈ܵܬ݂ܵܐ ܐܲܦܸܣܩܘܿܦܹ̈ܐ: ܕܐܸܬ݂ܟܲܢܲܫܘ ܒܢܝܼܩܝܼܵܐ ܡܕ݂ܝܼܢ݇ܬܵܐ ܕܒܹܝܬ݂ܘܿܢܝܼܵܐ: ܒܙܲܒ݂ܢܵܐ ܕܩܘܿܣܛܲܢܛܝܼܢܘܿܣ ܡܲܠܟܵܐ ܙܲܟܵܝܵܐ. ܘܥܸܠܬ݂ܵܐ ܕܹܝܢ ܕܲܟ݂ܢܘܼܫܝܼܗܘܿܢ ܡܸܛܠ ܪܘܼܫܥܹܗ ܕܐܲܪܝܼܘܿܣ܀ ܘܐܲܝܟ݂ ܥܝܵܕ݂ܵܐ ܕܹܝܢ ܕܪܵܕܹܐ ܒܲܝܢܵܬ݂ ܒܢܹܝ̈ܗ̇ ܕܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ ܘܲܫܠܝܼܚܵܝܬܵܐ ܕܝܼܠܲܢ ܕܡܲܕ݂ܢ̱ܚܵܐ: ܘܲܟ݂ܒܲܪ ܕܡ̣ܢ ܪܵܘܡܵܐ ܕܝܵܘ̈ܡܵܬ݂ܵܐ ܪܕ݂ܵܐ: ܘܡ̣ܢ ܡܲܫܠ̱ܡܵܢܘܼܬ݂ܵܐ ܘܝܘܼܒܵܠܵܐ ܕܥܲܬ݁ܝܼܩܹ̈ܐ ܡܓ݂ܵܘܝܲܝ̈ ܒܬܹܐܘܿܖ̈ܝܼܵܣ ܕܲܫܪܵܪܵܐ ܐܲܠܵܗܵܝܵܐ ܡܬ݂ܝܼܚ ܗ݇ܘ̣ܵ ܥܕܲܡܵܐ ܠܝܵܘ̈ܡܵܬܲܢ. ܘܗܵܟܲܢܵܐ ܡܫܲܪܹܝܢ ܚ݇ܢܲܢ ܠܡܸܬ݂ܢܵܐ ܩܵܢܘܿܢܵܐ ܕܗܲܝܡܵܢܘܼܬ݂ܵܐ ܕܢܝܼܩܝܼܵܐ.

ܒܲܫܪܵܪܵܐ ܓܹܝܪ ܕܠܵܐ ܦܘܼܠܵܓ݂ܵܐ:

ܡܗܲܝܡܢܝܼܢܲܢ ܒܚܲܕ݂ ܐܲܠܵܗܵܐ: ܐܲܒ݂ܵܐ ܐܲܚܝܼܕ݂ ܟܠ. ܥܵܒ݂ܘܿܕ݂ܵܐ ܕܟ݂ܠܗܹܝܢ ܐܲܝܠܹܝܢ ܕܡܸܬ݂ܚܲܙ̈ܝܵܢ ܘܲܕ݂ܠܵܐ ܡܸܬ݂ܚܲܙ̈ܝܵܢ. ܘܲܒ݂ܚܲܕ݂ ܡܵܪܝܵܐ ܝܼܫܘܿܥ ܡܫܝܼܚܵܐ. ܒܪܹܗ ܕܐܲܠܵܗܵܐ ܝܼܚܝܼܕ݂ܵܝܵܐ: ܒܘܼܟ݂ܪܵܐ ܕܟ݂ܠܗܹܝܢ ܒܸܖ̈ܝܵܬ݂ܵܐ. ܗܵܘ̇ ܕܡ̣ܢ ܐܲܒ݂ܘܼܗܝ ܐܸܬ݂ܝܼܠܸܕ݂ ܩܕ݂ܵܡ ܟܠܗܘܿܢ ܥܵܠܡܹ̈ܐ: ܘܠܵܐ ܐܸܬ݂ܥ̱ܒܸܕ݂. ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܕܡ̣ܢ ܐܲܠܵܗܵܐ ܫܲܪܝܼܪܵܐ: ܒܲܪ ܟܝܵܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܕܒ݂ܐܝܼ̈ܕ݂ܵܘܗܝ ܐܸܬ݁ܬܲܩܲܢܘ ܥܵܠܡܹ̈ܐ: ܘܐܸܬ݂ܒ݁ܪܝܼ ܟܠ ܡܸܕ݁ܡ. ܗܵܘ̇ ܕܡܸܛܠܵܬܲܢ ܒܢܲܝ̈ܢܵܫܵܐ: ܘܡܸܛܠ ܦܘܼܪܩܵܢܲܢ: ܢ̣ܚܸܬ݂ ܡ̣ܢ ܫܡܲܝܵܐ: ܘܐܸܬ݂ܓܲܫܲܡ ܡ̣ܢ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ ܘܲܗܘ̤ܵܐ ܒܲܪܢܵܫܵܐ: ܘܐܸܬ݂ܒ݁ܛܸܢ ܘܐܸܬ݂ܝܼܠܸܕ݂ ܡ̣ܢ ܡܲܪܝܲܡ ܒܬ݂ܘܼܠܬܵܐ ܘܚܲܫ̣ ܘܐܸܙܕ݁ܩܸܦ: ܒܝܵܘܡܲܝ̈ ܦܲܢܛܝܼܘܿܣ ܦܝܼܠܵܛܘܿܣ. ܘܐܸܬ݂ܩ̄ܒܲܪ: ܘܩ̣ܵܡ ܠܲܬ݂ܠܵܬ݂ܵܐ ܝܵܘܡܝܼ̈ܢ ܐܲܝܟ݂ ܕܲܟ݂ܬ݂ܝܼܒ݂. ܘܲܣ̣ܠܸܩ ܠܲܫܡܲܝܵܐ: ܘܝܼܬܸܒ݂ ܡ̣ܢ ܝܲܡܝܼܢܵܐ ܕܐܲܒ݂ܘܼܗܝ. ܘܬ݂ܘܼܒ݂ ܥܬ݂ܝܼܕ݂ ܠܡܹܐܬ݂ܵܐ ܠܲܡܕ݂ܵܢ ܠܡܝܼܬܹ̈ܐ ܘܲܠܚܲܝܹ̈ܐ. ܘܲܒ݂ܚܲܕ݂ ܪܘܼܚܵܐ ܕܩܘܼܕ݂ܫܵܐ: ܪܘܼܚܵܐ ܕܲܫܪܵܪܵܐ: ܗܵܘ̇ ܕܡ̣ܢ ܐܲܒ݂ܵܐ ܢܵܦܹ̇ܩ: ܪܘܼܚܵܐ ܡܲܚܝܵܢܵܐ. ܘܒܲܚܕ݂ܵܐ ܥܹܕ݁ܬܵܐ ܩܲܕ݁ܝܼܫܬܵܐ: ܘܲܫܠܝܼܚܵܝܬܵܐ ܩܵܬ݂ܘܿܠܝܼܩܝܼ. ܘܡܵܘܕܹ݁ܝܢܲܢ ܒܲܚܕ݂ܵܐ ܡܲܥܡܘܿܕ݂ܝܼܬ݂ܵܐ: ܠܫܘܼܒ݂ܩܵܢܵܐ ܕܲܚܛܵܗܹ̈ܐ. ܘܒܲܩܝܵܡܬܵܐ ܕܦܲܓ݂ܖ̈ܲܝܢ: ܘܲܒ݂ܚܲܝܹ̈ܐ ܕܲܠܥܵܠܲܡ ܥܵܠܡܝܼܢ: ܐܵܡܹܝܢ.
